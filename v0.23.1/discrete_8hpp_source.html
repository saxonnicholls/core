68 template <linalg::scalar T, std::size_t K>
69 [[nodiscard]] constexpr result<linalg::fixed_vector<T, K>, hmm_error> normalise(const linalg::fixed_vector<T, K>& p) noexcept(linalg::detail::nothrow_scalar<T>)
70 {
71 T s(0);
72 for (std::size_t i = 0; i < K; ++i)
73 s += p[i];
74 if (!(s > T(0)))
75 return err(hmm_error::zero_probability);
76 return ok(p / s);
77 }

82 [[nodiscard]] constexpr linalg::fixed_matrix<T, R, C> normalise_columns(const linalg::fixed_matrix<T, R, C>& a) noexcept(linalg::detail::nothrow_scalar<T>)
83 {
84 auto m = a;
85 for (std::size_t j = 0; j < C; ++j) {
86 T s(0);

122 const linalg::fixed_matrix<T, L, K>& e) noexcept(linalg::detail::nothrow_scalar<T>)
