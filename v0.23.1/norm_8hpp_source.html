104template <row X, row W, row B, mutable_row Out>
105 requires std::same_as<row_value_t<X>, row_value_t<W>> && std::same_as<row_value_t<X>, row_value_t<B>>
106 && std::same_as<row_value_t<X>, row_value_t<Out>>
107constexpr norm_stats<row_value_t<X>> layer_norm(const X& x, const W& weight, const B& bias, Out&& out,
108 std::type_identity_t<row_value_t<X>> eps = 1e-5)
109{
110 using T = row_value_t<X>;
111 const auto in = as_row(x);
112 const auto w = as_row(weight);
113 const auto b = as_row(bias);
114 const auto y = as_mutable_row(out);
115 const std::size_t n = in.size();
116 assert(y.size() == n && detail::norm::fits<T>(w.size(), n) && detail::norm::fits<T>(b.size(), n));
117 T mean = T(0);
118 for (const T v : in)
119 mean += v;
120 mean = n == 0 ? T(0) : mean / T(n);
121 T var = T(0);
122 for (const T v : in)
123 var += (v - mean) * (v - mean);
124 var = n == 0 ? T(0) : var / T(n);
125 const T rstd = T(1) / math::sqrt(var + eps);
126 for (std::size_t i = 0; i < n; ++i)
127 y[i] = (in[i] - mean) * rstd * detail::norm::param(w, i, T(1)) + detail::norm::param(b, i, T(0));
128 return {mean, rstd};
129}

142template <row D, row X, row W, mutable_row DX, mutable_row DW, mutable_row DB>
143 requires std::same_as<row_value_t<X>, row_value_t<D>> && std::same_as<row_value_t<X>, row_value_t<W>>
144 && std::same_as<row_value_t<X>, row_value_t<DX>> && std::same_as<row_value_t<X>, row_value_t<DW>>
145 && std::same_as<row_value_t<X>, row_value_t<DB>>
146constexpr void layer_norm_backward(const D& dout, const X& x, const W& weight, norm_stats<row_value_t<X>> stats, DX&& dx,
147 DW&& dweight, DB&& dbias)
148{
149 using T = row_value_t<X>;
150 const auto d = as_row(dout);
151 const auto in = as_row(x);
152 const auto w = as_row(weight);
153 const auto gx = as_mutable_row(dx);
154 const auto gw = as_mutable_row(dweight);
155 const auto gb = as_mutable_row(dbias);
156 const std::size_t n = in.size();
157 assert(d.size() == n && gx.size() == n && detail::norm::fits<T>(w.size(), n) && detail::norm::fits<T>(gw.size(), n)
158 && detail::norm::fits<T>(gb.size(), n));
159 if (n == 0)
160 return;
161 T mean_g = T(0), mean_gn = T(0);
162 for (std::size_t i = 0; i < n; ++i) {
163 const T norm = (in[i] - stats.mean) * stats.rstd;
164 const T g = detail::norm::param(w, i, T(1)) * d[i];
165 mean_g += g;
166 mean_gn += g * norm;
167 }
168 mean_g /= T(n);
169 mean_gn /= T(n);
170 for (std::size_t i = 0; i < n; ++i) {
171 const T norm = (in[i] - stats.mean) * stats.rstd;
172 const T g = detail::norm::param(w, i, T(1)) * d[i];
173 if (!gw.empty())
174 gw[i] += d[i] * norm;
175 if (!gb.empty())
176 gb[i] += d[i];
177 gx[i] = stats.rstd * (g - mean_g - norm * mean_gn);
178 }
179}

190template <row X, row W, mutable_row Out>
191 requires std::same_as<row_value_t<X>, row_value_t<W>> && std::same_as<row_value_t<X>, row_value_t<Out>>
192constexpr row_value_t<X> rms_norm(const X& x, const W& weight, Out&& out,
193 std::type_identity_t<row_value_t<X>> eps = std::numeric_limits<row_value_t<X>>::epsilon())
194{
195 using T = row_value_t<X>;
196 const auto in = as_row(x);
197 const auto w = as_row(weight);
198 const auto y = as_mutable_row(out);
199 const std::size_t n = in.size();
200 assert(y.size() == n && detail::norm::fits<T>(w.size(), n));
201 T ms = T(0);
202 for (const T v : in)
203 ms += v * v;
204 ms = n == 0 ? T(0) : ms / T(n);
205 const T rstd = T(1) / math::sqrt(ms + eps);
206 for (std::size_t i = 0; i < n; ++i)
207 y[i] = in[i] * rstd * detail::norm::param(w, i, T(1));
208 return rstd;
209}

220template <row D, row X, row W, mutable_row DX, mutable_row DW>
221 requires std::same_as<row_value_t<X>, row_value_t<D>> && std::same_as<row_value_t<X>, row_value_t<W>>
222 && std::same_as<row_value_t<X>, row_value_t<DX>> && std::same_as<row_value_t<X>, row_value_t<DW>>
223constexpr void rms_norm_backward(const D& dout, const X& x, const W& weight, std::type_identity_t<row_value_t<X>> rstd, DX&& dx,
224 DW&& dweight)
225{
226 using T = row_value_t<X>;
227 const auto d = as_row(dout);
228 const auto in = as_row(x);
229 const auto w = as_row(weight);
230 const auto gx = as_mutable_row(dx);
231 const auto gw = as_mutable_row(dweight);
232 const std::size_t n = in.size();
233 assert(d.size() == n && gx.size() == n && detail::norm::fits<T>(w.size(), n) && detail::norm::fits<T>(gw.size(), n));
234 if (n == 0)
235 return;
236 T mean_gn = T(0);
237 for (std::size_t i = 0; i < n; ++i)
238 mean_gn += detail::norm::param(w, i, T(1)) * d[i] * in[i] * rstd;
239 mean_gn /= T(n);
240 for (std::size_t i = 0; i < n; ++i) {
241 const T norm = in[i] * rstd;
242 const T g = detail::norm::param(w, i, T(1)) * d[i];
243 if (!gw.empty())
244 gw[i] += d[i] * norm;
245 gx[i] = rstd * (g - norm * mean_gn);
246 }

282 std::type_identity_t<row_value_t<X>> eps = 1e-5)

286 const auto w = as_row(weight);

296 continue;

303 for (const T v : group)
304 var += (v - mean) * (v - mean);
305 var = var / T(n);
306 const T rstd = T(1) / math::sqrt(var + eps);
307 for (std::size_t c = c0; c < c1; ++c)

393template <row X, mutable_row Out>
394 requires std::same_as<row_value_t<X>, row_value_t<Out>>
395constexpr row_value_t<X> l2_norm(const X& x, Out&& out, std::type_identity_t<row_value_t<X>> eps = 1e-12)
396{
397 using T = row_value_t<X>;
398 const auto in = as_row(x);
399 const auto y = as_mutable_row(out);
400 assert(y.size() == in.size() && eps >= T(0));
401 T sum = T(0);
402 for (const T v : in)
403 sum += v * v;
404 const T norm = math::sqrt(sum);
405 const T scale = T(1) / (norm > eps ? norm : eps);
406 for (std::size_t i = 0; i < in.size(); ++i)
407 y[i] = in[i] * scale;
408 return norm;
409}

418template <row D, row X, mutable_row DX>
419 requires std::same_as<row_value_t<X>, row_value_t<D>> && std::same_as<row_value_t<X>, row_value_t<DX>>
420constexpr void l2_norm_backward(const D& dout, const X& x, std::type_identity_t<row_value_t<X>> norm, DX&& dx,
421 std::type_identity_t<row_value_t<X>> eps = 1e-12)
422{
423 using T = row_value_t<X>;
424 const auto d = as_row(dout);
425 const auto in = as_row(x);
426 const auto gx = as_mutable_row(dx);
427 assert(d.size() == in.size() && gx.size() == in.size());
428 if (!(norm > eps)) {
429 for (std::size_t i = 0; i < in.size(); ++i)
430 gx[i] = d[i] / eps;
431 return;
432 }
433 const T scale = T(1) / norm;
434 T dot = T(0);
435 for (std::size_t i = 0; i < in.size(); ++i)
436 dot += in[i] * scale * d[i];
437 for (std::size_t i = 0; i < in.size(); ++i)
438 gx[i] = (d[i] - in[i] * scale * dot) * scale;
439}
