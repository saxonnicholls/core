79 namespace detail {
80 // Days since 1970-01-01 of the Monday of week 1 of week-based year y: the Monday on or before 4 January.
81 constexpr std::int64_t iso_week_one(std::int64_t y) noexcept
82 {
83 const std::int64_t jan4 = days_from_civil(y, 1, 4);
84 const unsigned wd = weekday_from_days(jan4); // 0 = Sunday
85 return jan4 - (wd == 0 ? 6 : wd - 1);
86 }
87 } // namespace detail

94 [[nodiscard]] constexpr unsigned iso_weeks_in_year(std::int32_t year) noexcept
95 {
96 return static_cast<unsigned>((detail::iso_week_one(year + std::int64_t{1}) - detail::iso_week_one(year)) / 7);
97 }

103 [[nodiscard]] constexpr iso_week_date to_iso_week_date(std::chrono::sys_days day) noexcept
104 {
105 const auto z = static_cast<std::int64_t>(day.time_since_epoch().count());
106 std::int64_t y = civil_from_days(z).year;
107 if (z >= detail::iso_week_one(y + 1))
108 ++y;
109 else if (z < detail::iso_week_one(y))
110 --y;
111 const unsigned wd = weekday_from_days(z);
112 return {static_cast<std::int32_t>(y), static_cast<unsigned>((z - detail::iso_week_one(y)) / 7 + 1), wd == 0 ? 7U : wd};
113 }

120 [[nodiscard]] constexpr std::chrono::sys_days to_sys_days(const iso_week_date& w) noexcept
121 {
122 const std::int64_t z = detail::iso_week_one(w.year) + (static_cast<std::int64_t>(w.week) - 1) * 7 + (static_cast<std::int64_t>(w.weekday) - 1);
123 return std::chrono::sys_days{std::chrono::days{static_cast<std::chrono::days::rep>(z)}};
124 }

132 [[nodiscard]] constexpr result<iso_week_date, week_date_errc> make_iso_week_date(std::int32_t year, unsigned week, unsigned weekday) noexcept
133 {
134 if (week < 1 || week > iso_weeks_in_year(year))
135 return err(week_date_errc::invalid_week);
136 if (weekday < 1 || weekday > 7)
137 return err(week_date_errc::invalid_weekday);
138 return ok(iso_week_date{year, week, weekday});
139 }
