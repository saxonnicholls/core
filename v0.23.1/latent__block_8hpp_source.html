126template <transformer_config C, typename S, math::math_number T, mutable_row X, latent_kv_store<T> Cache, typename R>
127 requires std::same_as<row_value_t<X>, T> && rope_source<R>
128constexpr void block_forward(const latent_block_weights<dense_weights<S>, T>& w, X&& x, std::size_t pos, const Cache& cache,
129 std::size_t layer, const R& rope_src)
130{
131 const auto s = as_mutable_row(x);
132 assert(s.size() == C.d_model);
133 std::array<T, C.d_model> h{}, a{};
134 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
135 latent_attention_forward<C>(w.attention, h, pos, cache, layer, rope_src, a);
136 for (std::size_t i = 0; i < C.d_model; ++i)
137 s[i] += a[i];
138 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
139 mlp_forward<C>(w.mlp, h, a);
140 for (std::size_t i = 0; i < C.d_model; ++i)
141 s[i] += a[i];
142}

153template <transformer_config C, typename B, math::math_number T, typename R>
154 requires linalg::blas_backend<B, T> && rope_source<R>
155constexpr void block_forward_sequence(const B& backend, const latent_block_weights<dense_weights<T>, T>& w,
156 std::type_identity_t<std::span<T>> xs, const R& rope_src, const latent_block_activations<T>& act,
157 std::size_t sequences = 1)
158{
159 assert(xs.size() % C.d_model == 0 && act.x0.size() == xs.size() && act.h1.size() == xs.size() && act.x1.size() == xs.size()
160 && act.h2.size() == xs.size());
161 copy_rows(backend, std::span<const T>(xs), act.x0);
162 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm1_weight, w.norm1_bias, act.h1);
163 latent_attention_forward_sequence<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, xs, sequences);
164 add_rows(backend, std::span<const T>(act.x0), xs);
165 copy_rows(backend, std::span<const T>(xs), act.x1);
166 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm2_weight, w.norm2_bias, act.h2);
167 mlp_forward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), xs, mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden});
168 add_rows(backend, std::span<const T>(act.x1), xs);
169}

181template <transformer_config C, typename B, math::math_number T, typename R>
182 requires linalg::blas_backend<B, T> && rope_source<R>
183constexpr void block_backward(const B& backend, const latent_block_weights<dense_weights<T>, T>& w, const R& rope_src,
184 const latent_block_activations<T>& act, std::type_identity_t<std::span<T>> dxs,
185 const latent_block_gradients<T>& grads, std::size_t sequences = 1)
186{
187 assert(dxs.size() == act.x0.size() && grads.scratch.size() == dxs.size());
188 fill_rows(backend, grads.scratch, T(0));
189 mlp_backward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden},
190 std::span<const T>(dxs), grads.scratch, grads.mlp);
191 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x1), w.norm2_weight, w.norm2_bias,
192 dxs, grads.norm2_weight, grads.norm2_bias);
193 fill_rows(backend, grads.scratch, T(0));
194 latent_attention_backward<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(dxs),
195 grads.scratch, grads.attention, sequences);
196 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x0), w.norm1_weight, w.norm1_bias,
197 dxs, grads.norm1_weight, grads.norm1_bias);
198}
