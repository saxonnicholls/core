61 public:
63 using handler = std::function<http_response(const http_request&)>;

70 capability_result<void> add(http_method method, std::string path, handler h)
71 {
72 if (path.empty() || path.front() != '/')
73 return fail(capability_errc::invalid_argument, {}, "a path starts with /");
74 if (!h)
75 return fail(capability_errc::invalid_argument, {}, "an empty handler");
76 auto& methods = routes_[std::move(path)];
77 if (!methods.emplace(method, std::move(h)).second)
78 return fail(capability_errc::conflict, {}, "the route is already registered");
79 return ok();
80 }

85 http_response handle(const http_request& r) const
86 {
87 const auto path = routes_.find(r.path);
88 if (path == routes_.end())
89 return respond(404, "not found");
90 const auto& methods = path->second;
91 auto route = methods.find(r.method);
92 const bool head_as_get = route == methods.end() && r.method == http_method::head;
93 if (head_as_get)
94 route = methods.find(http_method::get);
95 if (route == methods.end()) {
96 std::string allow;
97 for (const auto& [m, h] : methods)
98 allow += (allow.empty() ? "" : ", ") + std::string(to_string(m));
99 http_response out = respond(405, "method not allowed");
100 out.headers.push_back({"Allow", std::move(allow)});
101 return out;
102 }
103 http_response out;
104#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
105 try {
106 out = route->second(r);
107 } catch (const std::bad_alloc&) {
108 throw;
109 } catch (...) {
110 return respond(500, "internal error");
111 }
112#else
113 out = route->second(r);
114#endif
115 if (head_as_get)
116 out.body.clear();
117 return out;
118 }

122 [[nodiscard]] std::size_t size() const noexcept
123 {
124 std::size_t n = 0;
125 for (const auto& [p, methods] : routes_)
126 n += methods.size();
127 return n;
128 }
129
130 private:
131 std::map<std::string, std::map<http_method, handler>, std::less<>> routes_;
132 };
