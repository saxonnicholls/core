67 [[nodiscard]] constexpr std::optional<std::string_view> check_token(const T& t) noexcept
68 {
69 if (const auto failed = check_asset(t))
70 return failed;
71 if (const auto failed = check_chain(t.chain()))
72 return failed;
73 const std::string_view symbol = t.symbol();
74 const std::string_view contract = t.contract();
75 if (symbol.empty())
76 return "token: symbol() is empty";
77 if (!detail::is_token_code(t.code(), symbol, detail::chain_id_of(t.chain())))
78 return "token: code() is neither symbol() nor symbol():chain_id";
79 if (const auto failed = detail::check_token_address(t.chain().kind(), contract))
80 return failed;
81 if (std::string_view{t.symbol()} != symbol || std::string_view{t.contract()} != contract)
82 return "token: symbol() or contract() changes between calls";
83 return std::nullopt;
84 }
