70 template <point A, point B>

72 [[nodiscard]] constexpr T reduced(const A& a, const B& b) const noexcept(linalg::detail::nothrow_scalar<T> && noexcept(coordinate(a, 0)) && noexcept(coordinate(b, 0)))

74 linalg::fixed_vector<T, N> d{};
