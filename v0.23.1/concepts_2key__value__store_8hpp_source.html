238 [[nodiscard]] constexpr kv_bytes lower() const noexcept { return lower_; }
241 [[nodiscard]] constexpr bool lower_open() const noexcept { return lower_open_; }

251
255 [[nodiscard]] constexpr bool contains(kv_bytes key) const noexcept
256 {
257 if (lower_open_ ? !kv_less(lower_, key) : kv_less(key, lower_))
258 return false;
259 return !beyond(key);
260 }
261
265 [[nodiscard]] constexpr bool beyond(kv_bytes key) const noexcept
266 {
267 return !kv_starts_with(key, prefix_) || (bounded_ && !kv_less(key, upper_));
268 }

272 [[nodiscard]] constexpr bool empty() const noexcept { return bounded_ && !kv_less(lower_, upper_); }

277 kv_bytes prefix_{};
278 bool lower_open_ = false;
279 bool bounded_ = false;

319 class kv_write_batch {
320 public:
328 template <kv_byte_source K, kv_byte_source V>
329 kv_write_batch& put(const K& key, const V& value) { return add(kv_op::put, kv_bytes_of(key), kv_bytes_of(value)); }
330
338 template <kv_byte_source K, kv_byte_source V>
339 kv_write_batch& insert(const K& key, const V& value) { return add(kv_op::insert, kv_bytes_of(key), kv_bytes_of(value)); }
340
346 template <kv_byte_source K>
347 kv_write_batch& erase(const K& key) { return add(kv_op::erase, kv_bytes_of(key), {}); }
348
351 [[nodiscard]] std::span<const kv_write> writes() const noexcept { return writes_; }
354 operator std::span<const kv_write>() const noexcept { return writes_; } // NOLINT(google-explicit-constructor)
357 [[nodiscard]] std::size_t size() const noexcept { return writes_.size(); }
360 [[nodiscard]] bool empty() const noexcept { return writes_.empty(); }
362 void clear() noexcept
363 {
364 writes_.clear();
365 storage_.clear();
366 }
367
368 private:
369 kv_write_batch& add(kv_op op, kv_bytes key, kv_bytes value)
370 {
371 auto block = std::make_unique<std::byte[]>(key.size() + value.size() + 1); // + 1: never a zero-length array
372 std::copy(key.begin(), key.end(), block.get());
373 std::copy(value.begin(), value.end(), block.get() + key.size());
374 writes_.reserve(writes_.size() + 1);
375 storage_.reserve(storage_.size() + 1);
376 writes_.push_back({op, kv_bytes(block.get(), key.size()), kv_bytes(block.get() + key.size(), value.size())});
377 storage_.push_back(std::move(block)); // cannot throw: room reserved
378 return *this;
379 }
380
381 std::vector<kv_write> writes_;
382 std::vector<std::unique_ptr<std::byte[]>> storage_;
383 };

402 template <typename S>
403 concept kv_readable = requires(const S& s, kv_bytes key, const kv_range& range, kv_read_visitor rv, kv_scan_visitor sv) {
404 { s.read(key, rv) } -> std::same_as<result<bool, kv_errc>>;
405 { s.contains(key) } -> std::same_as<result<bool, kv_errc>>;
406 { s.scan(range, sv) } -> std::same_as<result<std::size_t, kv_errc>>;
407 { s.sequence() } -> std::same_as<result<std::uint64_t, kv_errc>>;
408 };

420 template <typename S>
421 concept kv_writable = kv_readable<S> &&
422 requires(S& s, kv_bytes key, kv_bytes value, const kv_range& range, std::span<const kv_write> batch) {
423 { s.put(key, value) } -> std::same_as<result<void, kv_errc>>;

589 template <kv_readable S, typename F>
590 requires std::is_invocable_v<F&, kv_bytes, kv_bytes>
591 result<std::size_t, kv_errc> kv_scan(const S& store, const kv_range& range, F&& fn)
592 {
593 using answer = std::invoke_result_t<F&, kv_bytes, kv_bytes>;
594 static_assert(std::is_void_v<answer> || std::is_convertible_v<answer, bool>,
595 "snicholls::kv_scan: the visitor must return void or bool");
596 if constexpr (std::is_void_v<answer>)
597 return store.scan(range, [&](kv_bytes k, kv_bytes v) {
598 fn(k, v);
599 return true;
600 });
601 else
602 return store.scan(range, kv_scan_visitor(fn));
603 }

649 template <kv_readable S>
650 result<digest<32>, kv_errc> kv_digest_v1(const S& store, const kv_range& range = kv_range::all())
651 {
652 const auto u64_le = [](std::uint64_t n) {
653 std::array<std::byte, 8> out{};
654 for (std::size_t i = 0; i < 8; ++i)
655 out[i] = static_cast<std::byte>(n >> (8 * i));
656 return out;
657 };
658 sha256 h;
659 h.update(kv_digest_v1_tag);
660 auto pairs = store.scan(range, [&](kv_bytes k, kv_bytes v) {
661 h.update(u64_le(k.size())).update(k).update(u64_le(v.size())).update(v);
662 return true;
663 });
664 if (pairs.is_err())
665 return err(pairs.unwrap_err_unchecked());
666 h.update(u64_le(*pairs));
667 return ok(h.finish());
668 }
