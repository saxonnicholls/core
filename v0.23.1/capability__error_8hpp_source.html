123 [[nodiscard]] static constexpr bool is_code(std::string_view text) noexcept
124 {
125 if (text.empty() || text.size() > capacity)
126 return false;
127 for (const char c : text)
128 if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_' ||
129 c == '-' || c == '.' || c == ':'))
130 return false;
131 return true;
132 }

135 constexpr provider_code() noexcept = default;

140 template <std::size_t M>

142 consteval provider_code(const char (&literal)[M]) // NOLINT(google-explicit-constructor)
143 : text_(literal)
144 {
145 if (!is_code(text_.view()))

147 }
