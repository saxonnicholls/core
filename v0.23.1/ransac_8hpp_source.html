87 [[nodiscard]] constexpr std::size_t ransac_iterations(double confidence, double inlier_fraction, std::size_t s) noexcept
88 {
89 if (inlier_fraction >= 1.0)
90 return 1;
91 const double ws = math::pow(inlier_fraction, static_cast<double>(s));
92 if (!(ws > 0.0) || !(confidence > 0.0))
93 return ws > 0.0 ? 1 : static_cast<std::size_t>(-1);
94 const double n = math::log(1.0 - (confidence < 1.0 ? confidence : 1.0 - 1e-16)) / math::log1p(-ws); // log1p: 1 - ws rounds to 1 for ws below 2^-53
95 if (!(n < 1e18))
96 return static_cast<std::size_t>(-1);
97 const auto k = static_cast<std::size_t>(n);
98 return static_cast<double>(k) < n ? k + 1 : (k == 0 ? 1 : k);
99 }

116 template <std::size_t S, typename Fit, typename Error, full_range_generator G>
117 [[nodiscard]] constexpr auto ransac(std::size_t n, Fit&& fit, Error&& error, double threshold, std::span<std::size_t> inliers, G& g,
118 std::size_t max_iterations = 1000, double confidence = 0.99)
119 noexcept(noexcept(g()) && noexcept(fit(std::declval<std::span<const std::size_t>>()))
120 && noexcept(error(*fit(std::declval<std::span<const std::size_t>>()), std::size_t{}))
121 && std::is_nothrow_copy_constructible_v<std::remove_cvref_t<decltype(*fit(std::declval<std::span<const std::size_t>>()))>>
122 && std::is_nothrow_copy_assignable_v<std::remove_cvref_t<decltype(*fit(std::declval<std::span<const std::size_t>>()))>>)
123 -> result<ransac_result<std::remove_cvref_t<decltype(*fit(std::declval<std::span<const std::size_t>>()))>>, ransac_error>
124 {
125 using model_t = std::remove_cvref_t<decltype(*fit(std::declval<std::span<const std::size_t>>()))>;
126 static_assert(S > 0, "snicholls::stats::ransac: a sample has at least one datum");
127 if (n < S || inliers.size() < n)
128 return err(ransac_error::too_few_data);
129 bool have = false;
130 model_t best{};
131 std::size_t best_count = 0;
132 double best_sum = 0;
133 std::size_t needed = max_iterations;
134 std::size_t tries = 0;
135 std::array<std::size_t, S> sample{};
136 for (; tries < needed && tries < max_iterations; ++tries) {
137 for (std::size_t k = 0; k < S; ++k) {
138 bool fresh = false;
139 while (!fresh) {
140 sample[k] = static_cast<std::size_t>(uniform_index(g, n));
141 fresh = true;
142 for (std::size_t q = 0; q < k; ++q)
143 fresh = fresh && sample[q] != sample[k];
144 }
145 }
146 auto m = fit(std::span<const std::size_t>(sample));
147 if (!m)
148 continue;
149 std::size_t count = 0;
150 double sum = 0;
151 for (std::size_t i = 0; i < n; ++i) {
152 const double e = static_cast<double>(error(*m, i));
153 if (e <= threshold) {
154 ++count;
155 sum += e;
156 }
157 }
158 if (count > best_count || (count == best_count && have && sum < best_sum)) {
159 best = *m;
160 best_count = count;
161 best_sum = sum;
162 have = true;
163 const std::size_t want = ransac_iterations(confidence, static_cast<double>(count) / static_cast<double>(n), S);
164 if (want < needed)
165 needed = want;
166 }
167 }
168 if (!have || best_count < S)
169 return err(ransac_error::no_model);
170 std::size_t k = 0;
171 for (std::size_t i = 0; i < n; ++i)
172 if (static_cast<double>(error(best, i)) <= threshold)
173 inliers[k++] = i;
174 return ok(ransac_result<model_t>{best, k, tries});
175 }
