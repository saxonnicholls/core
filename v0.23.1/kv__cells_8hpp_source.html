184 constexpr std::size_t seq_rm(kv_seq s, kv_pos p0, kv_pos p1) noexcept
185 {
186 assert(s >= -1 && s < kv_max_sequences);
187 std::size_t first = Capacity;
188 for (std::size_t i = 0; i < Capacity; ++i) {
189 if (is_empty(i) || !in(i, p0, p1) || (s >= 0 && !seq_has(i, s)))
190 continue;
191 seq_[i] = s < 0 ? 0 : seq_[i] & ~bit(s);
192 if (is_empty(i)) {
193 clear(i);
194 first = first < i ? first : i;
195 }
196 }
197 return first;
198 }

205 constexpr void seq_cp(kv_seq src, kv_seq dst, kv_pos p0, kv_pos p1) noexcept
206 {
207 if (src == dst)
208 return;
209 for (std::size_t i = 0; i < Capacity; ++i)
210 if (!is_empty(i) && in(i, p0, p1) && seq_has(i, src))
211 seq_[i] |= bit(dst);
212 }

217 constexpr std::size_t seq_keep(kv_seq s) noexcept
218 {
219 std::size_t first = Capacity;
220 for (std::size_t i = 0; i < Capacity; ++i) {
221 if (is_empty(i))
222 continue;
223 if (seq_has(i, s)) {
224 seq_[i] = bit(s);
225 continue;
226 }
227 seq_[i] = 0;
228 clear(i);
229 first = first < i ? first : i;
230 }
231 return first;

240 constexpr void seq_add(kv_seq s, kv_pos p0, kv_pos p1, kv_pos delta) noexcept
241 {
242 if (delta == 0)
243 return;
244 for (std::size_t i = 0; i < Capacity; ++i) {
245 if (is_empty(i) || !in(i, p0, p1) || !seq_has(i, s))
246 continue;
247 pos_[i] += delta;
248 shift_[i] += delta;
249 has_shift_ = true;
250 if (pos_[i] < 0) {
251 seq_[i] = 0;
252 clear(i);
253 }
254 }
255 }

263 constexpr void seq_div(kv_seq s, kv_pos p0, kv_pos p1, kv_pos d) noexcept
264 {
265 assert(d > 0);
266 if (d == 1)
267 return;
268 for (std::size_t i = 0; i < Capacity; ++i) {
269 if (is_empty(i) || !in(i, p0, p1) || !seq_has(i, s))
270 continue;
271 const kv_pos old = pos_[i];
272 pos_[i] = old / d;
273 shift_[i] += pos_[i] - old;
274 has_shift_ = true;
275 }
276 }

281 [[nodiscard]] constexpr kv_pos seq_pos_min(kv_seq s) const noexcept
282 {
283 kv_pos m = -1;
284 for (std::size_t i = 0; i < Capacity; ++i)
285 if (!is_empty(i) && seq_has(i, s) && (m < 0 || pos_[i] < m))

293 [[nodiscard]] constexpr kv_pos seq_pos_max(kv_seq s) const noexcept
294 {
295 kv_pos m = -1;
296 for (std::size_t i = 0; i < Capacity; ++i)
297 if (!is_empty(i) && seq_has(i, s) && pos_[i] > m)
298 m = pos_[i];
299 return m;

316 [[nodiscard]] constexpr bool visible(std::size_t i, kv_seq s, kv_pos query_pos, std::size_t window = 0) const noexcept
317 {
318 return !is_empty(i) && seq_has(i, s) && pos_[i] <= query_pos
319 && (window == 0 || static_cast<std::size_t>(query_pos - pos_[i]) < window);

323 static constexpr std::uint64_t bit(kv_seq s) noexcept
324 {
325 assert(s >= 0 && s < kv_max_sequences);

329 constexpr bool in(std::size_t i, kv_pos p0, kv_pos p1) const noexcept
330 {
331 const kv_pos lo = p0 < 0 ? 0 : p0, hi = p1 < 0 ? std::numeric_limits<kv_pos>::max() : p1;
332 return pos_[i] >= lo && pos_[i] < hi;
333 }

336 {
337 pos_[i] = -1;

355template <transformer_config C, math::math_number T, std::size_t Capacity>
356constexpr void kv_shift_keys(std::span<T> keys, std::size_t layers, kv_cells<Capacity>& cells)
357{
358 static_assert(valid(C) && C.rope != rope_kind::none && !C.latent(),
359 "snicholls::ml::kv_shift_keys: the configuration rotates no keys, or its keys are latent");
360 assert((keys.size() == kv_cache_view<C, T>::required_size(layers, Capacity)));
361 constexpr std::size_t d = C.head_dim, h = d / 2;
362 if (!cells.has_shift())
363 return;
364 for (std::size_t i = 0; i < Capacity; ++i) {
365 const kv_pos delta = cells.shift(i);
366 if (cells.is_empty(i) || delta == 0)
367 continue;
368 std::array<T, h> c{}, s{};
369 rope_angles(static_cast<std::size_t>(delta < 0 ? -delta : delta), c, s, rope_base_v<C, T>, rope_scaling_v<C, T>);
370 for (std::size_t layer = 0; layer < layers; ++layer)
371 for (std::size_t kv = 0; kv < C.n_kv_heads; ++kv) {
372 const auto row = keys.subspan(((layer * C.n_kv_heads + kv) * Capacity + i) * d, d);
373 if (delta > 0)
374 rope(row, c, s, C.layout());
375 else
376 rope_inverse(row, c, s, C.layout());
377 }
378 }
379 cells.reset_shift();
380}

397template <transformer_config C, math::math_number T, std::size_t Capacity, row Q, mutable_row Out>
398 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<Out>, T>
399constexpr T attend_cells(const kv_cache_view<C, T>& cache, const kv_cells<Capacity>& cells, std::size_t layer, std::size_t kv_head,
400 const Q& q, Out&& out, kv_seq seq, kv_pos query_pos, std::type_identity_t<T> slope = 0)
401{
402 const auto qs = as_row(q);
403 const auto y = as_mutable_row(out);
404 assert(cache.capacity() == Capacity && qs.size() == C.head_dim && y.size() == C.head_dim);
405 const T scale = T(1) / math::sqrt(T(C.head_dim));
406 std::fill(y.begin(), y.end(), T(0));
407 T max = -std::numeric_limits<T>::infinity(), sum = T(0);
408 for (std::size_t i = 0; i < Capacity; ++i) {
409 if (!cells.visible(i, seq, query_pos, C.window))
410 continue;
411 const auto run = cache.visible(layer, kv_head, key_range{i, i + 1});
412 T s = scale * detail::attention::dot(qs, run.keys[0]);
413 if constexpr (C.attention_softcap > 0)
414 s = softcap(s, attention_softcap_v<C, T>);
415 if (slope != T(0))
416 s = s + alibi_bias(slope, query_pos, cells.pos(i));
417 detail::attention::fold(max, sum, y, s, run.values[0]);
418 }
419 for (T& v : y)
420 v = sum == T(0) ? T(0) : v / sum;
421 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
422}
