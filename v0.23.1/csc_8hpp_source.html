140 template <typename T, typename I>
141 [[nodiscard]] constexpr bool is_valid(const csc_view<T, I>& a) noexcept
142 {
143 if (a.col_ptr.size() < a.cols + 1 || a.col_ptr[0] != 0)
144 return false;
145 for (std::size_t j = 0; j < a.cols; ++j)
146 if (a.col_ptr[j + 1] < a.col_ptr[j])
147 return false;
148 const std::size_t nnz = a.nnz();
149 if (a.row_idx.size() < nnz || a.values.size() < nnz)
150 return false;
151 for (std::size_t j = 0; j < a.cols; ++j)
152 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
153 if (a.row_idx[k] < 0 || detail::at(a.row_idx[k]) >= a.rows)
154 return false;
155 if (k > detail::at(a.col_ptr[j]) && !(a.row_idx[k - 1] < a.row_idx[k]))
156 return false;
157 }
158 return true;
159 }

174 template <typename T, typename I>
175 [[nodiscard]] constexpr result<std::size_t, sparse_error> from_triplets(std::size_t rows, std::size_t cols, std::span<const I> ti, std::span<const I> tj, std::span<const T> tv,
176 csc_view<T, I>& out, std::span<I> work) noexcept
177 {
178 const std::size_t n = ti.size();
179 if (tj.size() != n || tv.size() != n)
180 return err(sparse_error::dimension_mismatch);
181 if (out.col_ptr.size() < cols + 1 || out.row_idx.size() < n || out.values.size() < n || work.size() < (rows > cols ? rows : cols))
182 return err(sparse_error::insufficient_capacity);
183 for (std::size_t k = 0; k < n; ++k)
184 if (ti[k] < 0 || tj[k] < 0 || detail::at(ti[k]) >= rows || detail::at(tj[k]) >= cols)
185 return err(sparse_error::invalid_index);
186 out.rows = rows;
187 out.cols = cols;
188 // Count per column, then place (a counting sort by column, stable in triplet order)
189 for (std::size_t j = 0; j <= cols; ++j)
190 out.col_ptr[j] = 0;
191 for (std::size_t k = 0; k < n; ++k)
192 ++out.col_ptr[detail::at(tj[k]) + 1];
193 for (std::size_t j = 0; j < cols; ++j)
194 out.col_ptr[j + 1] += out.col_ptr[j];
195 for (std::size_t j = 0; j < cols; ++j)
196 work[j] = out.col_ptr[j];
197 for (std::size_t k = 0; k < n; ++k) {
198 const std::size_t p = detail::at(work[detail::at(tj[k])]++);
199 out.row_idx[p] = ti[k];
200 out.values[p] = tv[k];
201 }
202 // Within each column: sort by row (insertion: columns are short) and sum duplicates, compacting in place
203 std::size_t write = 0;
204 for (std::size_t j = 0; j < cols; ++j) {
205 const std::size_t lo = detail::at(out.col_ptr[j]), hi = detail::at(out.col_ptr[j + 1]);
206 for (std::size_t a = lo + 1; a < hi; ++a)
207 for (std::size_t b = a; b > lo && out.row_idx[b] < out.row_idx[b - 1]; --b) {
208 const I ri = out.row_idx[b];
209 out.row_idx[b] = out.row_idx[b - 1];
210 out.row_idx[b - 1] = ri;
211 const T vi = out.values[b];
212 out.values[b] = out.values[b - 1];
213 out.values[b - 1] = vi;
214 }
215 out.col_ptr[j] = static_cast<I>(write);
216 for (std::size_t k = lo; k < hi; ++k) {
217 if (write > detail::at(out.col_ptr[j]) && out.row_idx[write - 1] == out.row_idx[k])
218 out.values[write - 1] += out.values[k];
219 else {
220 out.row_idx[write] = out.row_idx[k];
221 out.values[write] = out.values[k];
222 ++write;
223 }
224 }
225 }
226 out.col_ptr[cols] = static_cast<I>(write);
227 return ok(write);
228 }

240 template <typename T, typename I, typename U>
241 constexpr void spmv(const csc_view<T, I>& a, std::span<const U> x, std::span<U> y, U alpha = U(1), U beta = U(0)) noexcept
242 {
243 for (std::size_t i = 0; i < a.rows; ++i)
244 y[i] = beta == U(0) ? U(0) : beta * y[i];
245 for (std::size_t j = 0; j < a.cols; ++j) {
246 const U xj = alpha * x[j];
247 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k)
248 y[detail::at(a.row_idx[k])] += static_cast<U>(a.values[k]) * xj;
249 }
250 }

262 template <typename T, typename I, typename U>
263 constexpr void spmv_transpose(const csc_view<T, I>& a, std::span<const U> x, std::span<U> y, U alpha = U(1), U beta = U(0)) noexcept
264 {
265 for (std::size_t j = 0; j < a.cols; ++j) {
266 U s(0);
267 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k)
268 s += static_cast<U>(a.values[k]) * x[detail::at(a.row_idx[k])];
269 y[j] = (beta == U(0) ? U(0) : beta * y[j]) + alpha * s;
270 }

284 template <typename T, typename I, typename U>
285 constexpr void spmv_symmetric_upper(const csc_view<T, I>& a, std::span<const U> x, std::span<U> y, U alpha = U(1), U beta = U(0)) noexcept
286 {
287 for (std::size_t i = 0; i < a.rows; ++i)
288 y[i] = beta == U(0) ? U(0) : beta * y[i];
289 for (std::size_t j = 0; j < a.cols; ++j)
290 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
291 const std::size_t i = detail::at(a.row_idx[k]);
292 if (i > j)
293 continue;
294 const U v = alpha * static_cast<U>(a.values[k]);
295 y[i] += v * x[j];
296 if (i != j)
297 y[j] += v * x[i];
298 }

312 template <typename T, typename I, typename U, typename J>
313 [[nodiscard]] constexpr result<void, sparse_error> transpose(const csc_view<T, I>& a, csc_view<U, J>& out, std::span<J> work) noexcept
314 {
315 const std::size_t nnz = a.nnz();
316 if (out.col_ptr.size() < a.rows + 1 || out.row_idx.size() < nnz || out.values.size() < nnz || work.size() < a.rows)
317 return err(sparse_error::insufficient_capacity);
318 out.rows = a.cols;
319 out.cols = a.rows;
320 for (std::size_t i = 0; i <= a.rows; ++i)
321 out.col_ptr[i] = 0;
322 for (std::size_t k = 0; k < nnz; ++k)
323 ++out.col_ptr[detail::at(a.row_idx[k]) + 1];
324 for (std::size_t i = 0; i < a.rows; ++i)
325 out.col_ptr[i + 1] += out.col_ptr[i];
326 for (std::size_t i = 0; i < a.rows; ++i)
327 work[i] = out.col_ptr[i];
328 // Columns in order, so each output column's rows come out ascending
329 for (std::size_t j = 0; j < a.cols; ++j)
330 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
331 const std::size_t p = detail::at(work[detail::at(a.row_idx[k])]++);
332 out.row_idx[p] = static_cast<J>(j);
333 out.values[p] = static_cast<U>(a.values[k]);
334 }
335 return ok();
336 }

350 template <typename T, typename I>
351 [[nodiscard]] constexpr result<void, sparse_error> symmetric_permute_upper(const csc_view<const T, const I>& a, std::span<const I> perm, csc_view<T, I>& out,
352 std::span<I> work, std::span<I> map = {}) noexcept
353 {
354 const std::size_t n = a.cols, nnz = a.nnz();
355 if (a.rows != n || perm.size() != n)
356 return err(sparse_error::dimension_mismatch);
357 if (out.col_ptr.size() < n + 1 || out.row_idx.size() < nnz || out.values.size() < nnz || work.size() < 2 * n || (!map.empty() && map.size() < nnz))
358 return err(sparse_error::insufficient_capacity);
359 const auto inverse = work.first(n); // inverse[old] = new
360 const auto next = work.subspan(n, n);
361 for (std::size_t k = 0; k < n; ++k)
362 inverse[k] = -1;
363 for (std::size_t k = 0; k < n; ++k) {
364 if (perm[k] < 0 || detail::at(perm[k]) >= n || inverse[detail::at(perm[k])] != -1)
365 return err(sparse_error::invalid_index);
366 inverse[detail::at(perm[k])] = static_cast<I>(k);
367 }
368 out.rows = out.cols = n;
369 for (std::size_t j = 0; j <= n; ++j)
370 out.col_ptr[j] = 0;
371 // Count: entry (i, j), i ≤ j, goes to column max(new i, new j)
372 for (std::size_t j = 0; j < n; ++j)
373 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
374 const std::size_t i = detail::at(a.row_idx[k]);
375 if (i > j)
376 continue;
377 const auto ni = inverse[i], nj = inverse[j];
378 ++out.col_ptr[detail::at(ni > nj ? ni : nj) + 1];
379 }
380 for (std::size_t j = 0; j < n; ++j)
381 out.col_ptr[j + 1] += out.col_ptr[j];
382 for (std::size_t j = 0; j < n; ++j)
383 next[j] = out.col_ptr[j];
384 for (std::size_t j = 0; j < n; ++j)
385 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
386 const std::size_t i = detail::at(a.row_idx[k]);
387 if (i > j)
388 continue;
389 const auto ni = inverse[i], nj = inverse[j];
390 const auto col = ni > nj ? ni : nj, row = ni > nj ? nj : ni;
391 const std::size_t p = detail::at(next[detail::at(col)]++);
392 out.row_idx[p] = row;
393 out.values[p] = a.values[k];
394 if (!map.empty())
395 map[k] = static_cast<I>(p);
396 }
397 // Rows ascending within each column (insertion, carrying the map's back-pointers by a second pass)
398 for (std::size_t j = 0; j < n; ++j) {
399 const std::size_t lo = detail::at(out.col_ptr[j]), hi = detail::at(out.col_ptr[j + 1]);
400 for (std::size_t s = lo + 1; s < hi; ++s)
401 for (std::size_t b = s; b > lo && out.row_idx[b] < out.row_idx[b - 1]; --b) {
402 const I r = out.row_idx[b];
403 out.row_idx[b] = out.row_idx[b - 1];
404 out.row_idx[b - 1] = r;
405 const T v = out.values[b];
406 out.values[b] = out.values[b - 1];
407 out.values[b - 1] = v;
408 }
409 }
410 if (!map.empty()) {
411 // Recompute the map against the sorted columns: find each original entry's row in its target column
412 for (std::size_t j = 0; j < n; ++j)
413 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
414 const std::size_t i = detail::at(a.row_idx[k]);
415 if (i > j) {
416 map[k] = -1;
417 continue;
418 }
419 const auto ni = inverse[i], nj = inverse[j];
420 const auto col = detail::at(ni > nj ? ni : nj);
421 const auto row = ni > nj ? nj : ni;
422 for (std::size_t p = detail::at(out.col_ptr[col]); p < detail::at(out.col_ptr[col + 1]); ++p)
423 if (out.row_idx[p] == row) {
424 map[k] = static_cast<I>(p);
425 break;
426 }
427 }
428 }
429 return ok();
430 }

440 template <typename T, typename I, typename U>
441 constexpr void scale(const csc_view<T, I>& a, std::span<const U> r, std::span<const U> c) noexcept
442 {
443 for (std::size_t j = 0; j < a.cols; ++j)
444 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
445 T v = a.values[k];
446 if (!r.empty())

448 if (!c.empty())
449 v *= static_cast<T>(c[j]);
450 a.values[k] = v;
451 }

459 constexpr void column_inf_norms(const csc_view<T, I>& a, std::span<U> out) noexcept
460 {
461 for (std::size_t j = 0; j < a.cols; ++j) {
462 U m(0);
463 for (std::size_t k = detail::at(a.col_ptr[j]); k < detail::at(a.col_ptr[j + 1]); ++k) {
464 const U v = static_cast<U>(a.values[k] < T(0) ? -a.values[k] : a.values[k]);
465 m = v > m ? v : m;
466 }
467 out[j] = m;
468 }
