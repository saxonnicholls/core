73template <math::math_number T>
74struct biquad
75{
76 T b0 = T(1);
77 T b1 = T(0);
78 T b2 = T(0);
79 T a1 = T(0);
80 T a2 = T(0);

85 [[nodiscard]] constexpr complex<T> response(T omega) const noexcept
86 {
87 const complex<T> z1{math::cos(omega), -math::sin(omega)}; // e^(-i omega)
88 const complex<T> z2 = z1 * z1;
89 const complex<T> num = complex<T>{b0, T(0)} + z1 * b1 + z2 * b2;
90 const complex<T> den = complex<T>{T(1), T(0)} + z1 * a1 + z2 * a2;
91 return num / den;
92 }
93
98 friend constexpr bool operator==(const biquad& a, const biquad& b) noexcept = default;
99};

104template <math::math_number T>
105struct biquad_state
106{
107 T z1 = T(0);
108 T z2 = T(0);
109
114 constexpr T process(const biquad<T>& s, T x) noexcept
115 {
116 const T y = s.b0 * x + z1;
117 z1 = s.b1 * x - s.a1 * y + z2;
118 z2 = s.b2 * x - s.a2 * y;
119 return y;
120 }
121};

145template <math::math_number T>
146constexpr biquad<T> rbj_biquad(biquad_kind kind, T f0, std::type_identity_t<T> fs, std::type_identity_t<T> q,
147 std::type_identity_t<T> gain_db = T(0)) noexcept
148{
149 using C = compute_t<T>;
150 assert(f0 > T(0) && f0 < fs / T(2) && q > T(0));
151 const C w0 = C(2) * math::math_limits<C>::pi * C(f0) / C(fs), cw = math::cos(w0), sw = math::sin(w0);
152 const C alpha = sw / (C(2) * C(q)), a = math::pow(C(10), C(gain_db) / C(40)), ra = C(2) * math::sqrt(a) * alpha;
153 std::array<C, 6> c{}; // b0 b1 b2 a0 a1 a2
154 switch (kind) {
155 case biquad_kind::lowpass: c = {(1 - cw) / 2, 1 - cw, (1 - cw) / 2, 1 + alpha, -2 * cw, 1 - alpha}; break;
156 case biquad_kind::highpass: c = {(1 + cw) / 2, -(1 + cw), (1 + cw) / 2, 1 + alpha, -2 * cw, 1 - alpha}; break;
157 case biquad_kind::bandpass: c = {alpha, 0, -alpha, 1 + alpha, -2 * cw, 1 - alpha}; break;
158 case biquad_kind::notch: c = {1, -2 * cw, 1, 1 + alpha, -2 * cw, 1 - alpha}; break;
159 case biquad_kind::allpass: c = {1 - alpha, -2 * cw, 1 + alpha, 1 + alpha, -2 * cw, 1 - alpha}; break;
160 case biquad_kind::peaking: c = {1 + alpha * a, -2 * cw, 1 - alpha * a, 1 + alpha / a, -2 * cw, 1 - alpha / a}; break;
161 case biquad_kind::lowshelf:
162 c = {a * ((a + 1) - (a - 1) * cw + ra), 2 * a * ((a - 1) - (a + 1) * cw), a * ((a + 1) - (a - 1) * cw - ra),
163 (a + 1) + (a - 1) * cw + ra, -2 * ((a - 1) + (a + 1) * cw), (a + 1) + (a - 1) * cw - ra};
164 break;
165 case biquad_kind::highshelf:
166 c = {a * ((a + 1) + (a - 1) * cw + ra), -2 * a * ((a - 1) + (a + 1) * cw), a * ((a + 1) + (a - 1) * cw - ra),
167 (a + 1) - (a - 1) * cw + ra, 2 * ((a - 1) - (a + 1) * cw), (a + 1) - (a - 1) * cw - ra};
168 break;
169 }
170 return {static_cast<T>(c[0] / c[3]), static_cast<T>(c[1] / c[3]), static_cast<T>(c[2] / c[3]), static_cast<T>(c[4] / c[3]),
171 static_cast<T>(c[5] / c[3])};
172}

195 // Bilinear transform s = (1 - z^-1) / (1 + z^-1) of (n2 s^2 + n1 s + n0) / (s^2 + d1 s + d0), normalised by a0.
196 template <typename C, typename T>
197 constexpr biquad<T> bilinear(C n2, C n1, C n0, C d1, C d0) noexcept
198 {
199 const C a0 = C(1) + d1 + d0, a1 = C(2) * d0 - C(2), a2 = C(1) - d1 + d0;
200 const C b0 = n2 + n1 + n0, b1 = C(2) * n0 - C(2) * n2, b2 = n2 - n1 + n0;
201 return {static_cast<T>(b0 / a0), static_cast<T>(b1 / a0), static_cast<T>(b2 / a0), static_cast<T>(a1 / a0), static_cast<T>(a2 / a0)};
202 }
203
204 // Bilinear transform of the first-order (n1 s + n0) / (s + d0).
205 template <typename C, typename T>
206 constexpr biquad<T> bilinear1(C n1, C n0, C d0) noexcept
207 {
208 const C a0 = C(1) + d0, a1 = d0 - C(1);
209 return {static_cast<T>((n1 + n0) / a0), static_cast<T>((n0 - n1) / a0), T(0), static_cast<T>(a1 / a0), T(0)};
210 }
211} // namespace detail::iir

220template <math::math_number T>
221constexpr void butterworth(std::size_t order, iir_band band, T w1, std::type_identity_t<T> w2, std::span<biquad<std::type_identity_t<T>>> sections) noexcept
222{
223 using C = compute_t<T>;
224 const bool two = band == iir_band::bandpass || band == iir_band::bandstop;
225 assert(order >= 1 && w1 > T(0) && w1 < T(1) && (!two || (w2 > w1 && w2 < T(1))));
226 assert(sections.size() == butterworth_sections(order, band));
227 const C pi = math::math_limits<C>::pi;
228 const C o1 = math::tan(pi * C(w1) / C(2)); // prewarped edges for s = (1 - z^-1) / (1 + z^-1)
229 const C o2 = two ? math::tan(pi * C(w2) / C(2)) : C(0);
230 const C bw = o2 - o1, o0sq = o1 * o2;
231 std::size_t out = 0;
232 // The prototype's poles in the upper half-plane, p_k = e^(i theta), theta = pi (2k + order + 1) / (2 order), k < order / 2;
233 // and for odd order the real pole -1.
234 for (std::size_t k = 0; k < order / 2; ++k) {
235 const C theta = pi * C(static_cast<long long>(2 * k + order + 1)) / C(static_cast<long long>(2 * order));
236 const complex<C> p{math::cos(theta), math::sin(theta)};
237 switch (band) {
238 case iir_band::lowpass: // Omega_c^2 / (s^2 - 2 Re(p) Omega_c s + Omega_c^2)
239 sections[out++] = detail::iir::bilinear<C, T>(C(0), C(0), o1 * o1, -C(2) * p.re * o1, o1 * o1);
240 break;
241 case iir_band::highpass: // s^2 / (s^2 - 2 Re(p) Omega_c s + Omega_c^2)
242 sections[out++] = detail::iir::bilinear<C, T>(C(1), C(0), C(0), -C(2) * p.re * o1, o1 * o1);
243 break;
244 case iir_band::bandpass:
245 case iir_band::bandstop: {
246 // Band-pass: s^2 - p BW s + Omega_0^2 = 0; band-stop: s^2 - (BW / p) s + Omega_0^2 = 0. Two roots, each with its conjugate.
247 const complex<C> c = band == iir_band::bandpass ? p * bw : conj(p) * bw; // 1 / p = conj p on the unit circle
248 const complex<C> disc = sqrt(c * c - complex<C>{C(4) * o0sq, C(0)});
249 for (const complex<C> r : std::array<complex<C>, 2>{(c + disc) * C(0.5), (c - disc) * C(0.5)}) {
250 const C d1 = -C(2) * r.re, d0 = norm(r);
251 sections[out++] = band == iir_band::bandpass ? detail::iir::bilinear<C, T>(C(0), bw, C(0), d1, d0)
252 : detail::iir::bilinear<C, T>(d0 / o0sq, C(0), d0, d1, d0);
253 }
254 break;
255 }
256 }
257 }
258 if (order % 2 == 1) { // the real pole p = -1
259 switch (band) {
260 case iir_band::lowpass: sections[out++] = detail::iir::bilinear1<C, T>(C(0), o1, o1); break;
261 case iir_band::highpass: sections[out++] = detail::iir::bilinear1<C, T>(C(1), C(0), o1); break;
262 case iir_band::bandpass: sections[out++] = detail::iir::bilinear<C, T>(C(0), bw, C(0), bw, o0sq); break;
263 case iir_band::bandstop: sections[out++] = detail::iir::bilinear<C, T>(C(1), C(0), o0sq, bw, o0sq); break;
264 }
265 }
266}

303template <math::math_number T>
304constexpr void sosfilt(std::span<const biquad<T>> sections, std::span<biquad_state<std::type_identity_t<T>>> state,
305 std::span<const std::type_identity_t<T>> x, std::span<std::type_identity_t<T>> y) noexcept
306{
307 assert(state.size() == sections.size() && y.size() == x.size());
308 for (std::size_t n = 0; n < x.size(); ++n) {
309 T v = x[n];
310 for (std::size_t s = 0; s < sections.size(); ++s)
311 v = state[s].process(sections[s], v);
312 y[n] = v;
313 }
314}
