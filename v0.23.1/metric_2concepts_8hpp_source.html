82 template <typename P>
83 concept sized_indexable = requires(const P& p, std::size_t i) {
84 { p.size() } -> std::convertible_to<std::size_t>;
85 p[i];
86 } && std::is_arithmetic_v<std::remove_cvref_t<decltype(std::declval<const P&>()[std::size_t{}])>>
87 // a bool read by value is a packed-bit proxy (libstdc++'s const std::vector<bool>::operator[]): not a point
88 && (!std::is_same_v<std::remove_cvref_t<decltype(std::declval<const P&>()[std::size_t{}])>, bool>
89 || std::is_reference_v<decltype(std::declval<const P&>()[std::size_t{}])>);

250 template <typename M, typename T>
251 concept decomposable = requires(const M& m, std::size_t axis, T x) {
252 { m.term(axis, x) } -> std::convertible_to<T>;
253 { m.combine(x, x) } -> std::convertible_to<T>;
254 { m.expand(x) } -> std::convertible_to<T>;
255 { m.contract(x) } -> std::convertible_to<T>;

338 template <typename M, point A, point B>
339 requires compatible_points<A, B> && decomposable<M, distance_t<A, B>>
340 [[nodiscard]] constexpr distance_t<A, B> accumulate(const M& m, const A& a, const B& b) noexcept(noexcept(m.combine(distance_t<A, B>{}, m.term(std::size_t{}, distance_t<A, B>{}))) && noexcept(coordinate(a, 0)) && noexcept(coordinate(b, 0)))
341 {
342 using T = distance_t<A, B>;
343 T acc(0);
344 const std::size_t n = dimension(a);
345 for (std::size_t i = 0; i < n; ++i)
346 acc = m.combine(acc, m.term(i, static_cast<T>(coordinate(a, i)) - static_cast<T>(coordinate(b, i))));
347 return acc;
348 }
