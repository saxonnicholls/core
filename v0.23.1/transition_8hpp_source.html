126 // Gauss-Legendre nodes and weights on [-1, 1], by Newton's method on P_n, at compile time
127 template <std::size_t N>
128 struct gauss_legendre {
129 std::array<double, N> x{}, w{};
130 };
131 template <std::size_t N>
132 consteval gauss_legendre<N> make_gauss_legendre()
133 {
134 gauss_legendre<N> g{};
135 for (std::size_t i = 0; i < N; ++i) {
136 double z = math::cos(3.14159265358979323846 * (static_cast<double>(i) + 0.75) / (static_cast<double>(N) + 0.5));
137 double dp = 0;
138 for (int it = 0; it < 100; ++it) {
139 double p0 = 1, p1 = z;
140 for (std::size_t k = 2; k <= N; ++k) {
141 const double p2 = ((2.0 * static_cast<double>(k) - 1.0) * z * p1 - (static_cast<double>(k) - 1.0) * p0) / static_cast<double>(k);
142 p0 = p1;
143 p1 = p2;
144 }
145 dp = static_cast<double>(N) * (z * p1 - p0) / (z * z - 1.0);
146 const double dz = p1 / dp;
147 z -= dz;
148 if (dz < 1e-17 && dz > -1e-17)
149 break;
150 }
151 g.x[i] = z;
152 g.w[i] = 2.0 / ((1.0 - z * z) * dp * dp);
153 }
154 return g;
155 }
156 inline constexpr gauss_legendre<16> gl16 = make_gauss_legendre<16>();

158 // The series of sin a / a, (1 - cos a) / a and their derivatives, 12 terms: exact to rounding for |a| < 0.5,
159 // where the closed forms lose digits to cancellation
160 template <typename T>
161 struct turn_series {
162 T f, g, df, dg;
163 };
164 template <typename T>
165 constexpr turn_series<T> turn_series_at(T a) noexcept(linalg::detail::nothrow_scalar<T>)
166 {
167 turn_series<T> r{T(0), T(0), T(0), T(0)};
168 T p(1); // a^(2k)
169 T prev(0); // a^(2k - 1), for the derivative of sin a / a (no division by a, so a = 0 is fine)
170 T fact(1); // (2k + 1)!
171 for (int k = 0; k < 12; ++k) {
172 const T sign = (k % 2 == 0) ? T(1) : T(-1);
173 const T f2 = fact * T(2 * k + 2); // (2k + 2)!
174 r.f += sign * p / fact;
175 r.g += sign * p * a / f2;
176 r.dg += sign * T(2 * k + 1) * p / f2;
177 r.df += sign * T(2 * k) * prev / fact;
178 prev = p * a;
179 p *= a * a;
180 fact = f2 * T(2 * k + 3);
181 }
182 return r;
183 }

185 // sin(a)/w and (1 - cos(a))/w for a = w dt, by series when |a| is small
186 template <typename T>
187 struct turn_terms {
188 T s, c, sw, cw; // sin a, cos a, sin a / w, (1 - cos a) / w
189 };
190 template <typename T>
191 constexpr turn_terms<T> turn(T w, T dt) noexcept(linalg::detail::nothrow_scalar<T>)
192 {
193 const T a = w * dt;
194 const T s = static_cast<T>(math::sin(a)), c = static_cast<T>(math::cos(a));
195 if (abs(a) < T(0.5)) {
196 const auto r = turn_series_at(a);
197 return {s, c, dt * r.f, dt * r.g};
198 }
199 return {s, c, s / w, (T(1) - c) / w};
200 }

207 template <linalg::scalar T, std::size_t N>
208 struct constant_nth_derivative {
209 using value_type = T;
210 static constexpr std::size_t dim = N + 1;
211 T q{};
212
214 [[nodiscard]] constexpr linalg::fixed_matrix<T, dim, dim> matrix(T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
215 {
216 auto f = linalg::fixed_matrix<T, dim, dim>::identity();
217 for (std::size_t i = 0; i < dim; ++i)
218 for (std::size_t j = i + 1; j < dim; ++j)
219 f(i, j) = detail::transition::power(dt, j - i) / static_cast<T>(detail::transition::factorial(j - i));
220 return f;
221 }

223 [[nodiscard]] constexpr linalg::fixed_vector<T, dim> apply(const linalg::fixed_vector<T, dim>& x, T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
224 {
225 return matrix(dt) * x;
226 }

228 [[nodiscard]] constexpr linalg::fixed_matrix<T, dim, dim> jacobian(const linalg::fixed_vector<T, dim>& x, T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
229 {
230 static_cast<void>(x);
231 return matrix(dt);
232 }
233
235 [[nodiscard]] constexpr linalg::fixed_matrix<T, dim, dim> covariance(T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
236 {
237 const T t = detail::transition::abs(dt);
238 linalg::fixed_matrix<T, dim, dim> m{};
239 for (std::size_t i = 0; i < dim; ++i)
240 for (std::size_t j = 0; j < dim; ++j) {
241 const std::size_t e = 2 * N + 1 - i - j;
242 m(i, j) = q * detail::transition::power(t, e) /
243 static_cast<T>(detail::transition::factorial(N - i) * detail::transition::factorial(N - j) * static_cast<double>(e));
244 }
245 return m;
246 }

267 template <linalg::scalar T, std::size_t N>
268 struct nth_derivative_decay {
269 static_assert(N >= 1, "snicholls::tracking::nth_derivative_decay: N is at least 1");
270 using value_type = T;
271 static constexpr std::size_t dim = N + 1;
272 T q{};
273 T k{};
274
276 [[nodiscard]] constexpr linalg::fixed_matrix<T, dim, dim> matrix(T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
277 {
278 auto f = linalg::fixed_matrix<T, dim, dim>::identity();
279 const T e = static_cast<T>(math::exp(-k * dt));
280 for (std::size_t i = 0; i < dim; ++i) {
281 const std::size_t r = N - i;
282 T v = e * ((r % 2 == 0) ? T(1) : T(-1)) / detail::transition::power(k, r);
283 for (std::size_t n = 1; n <= r; ++n)
284 v -= ((n % 2 == 0) ? T(1) : T(-1)) * detail::transition::power(dt, r - n) / (static_cast<T>(detail::transition::factorial(r - n)) * detail::transition::power(k, n));
285 f(i, N) = v;
286 for (std::size_t j = i + 1; j < N; ++j)
287 f(i, j) = detail::transition::power(dt, j - i) / static_cast<T>(detail::transition::factorial(j - i));
288 }
289 return f;
290 }

298 {
299 static_cast<void>(x);
300 return matrix(dt);

303 [[nodiscard]] constexpr linalg::fixed_matrix<T, dim, dim> covariance(T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
304 {
305 const T t = detail::transition::abs(dt);
306 linalg::fixed_matrix<T, dim, dim> m{};
307 if (t == T(0))
308 return m;
309 const T kt = k * t;
310 std::size_t panels = 1;
311 while (panels < 64 && static_cast<T>(panels) < kt)
312 panels *= 2;
313 const T h = t / static_cast<T>(panels);
314 for (std::size_t p = 0; p < panels; ++p) {
315 const T a = h * static_cast<T>(p);
316 for (std::size_t g = 0; g < 16; ++g) {
317 const T tau = a + h * (T(1) + static_cast<T>(detail::transition::gl16.x[g])) / T(2);
318 const auto f = matrix(tau);
319 const T w = static_cast<T>(detail::transition::gl16.w[g]) * h / T(2);
320 for (std::size_t i = 0; i < dim; ++i)
321 for (std::size_t j = 0; j < dim; ++j)
322 m(i, j) += w * f(i, N) * f(j, N);
323 }
324 }
325 return linalg::symmetrise(m * q);
326 }

397 template <linalg::scalar T>
398 struct constant_turn {
399 using value_type = T;
400 static constexpr std::size_t dim = 5;
401 T q_x{};
402 T q_y{};
403 T q_omega{};
404
406 [[nodiscard]] constexpr linalg::fixed_vector<T, 5> apply(const linalg::fixed_vector<T, 5>& x, T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
407 {
408 const auto t = detail::transition::turn(x[4], dt);
409 return {{x[0] + t.sw * x[1] - t.cw * x[3], x[1] * t.c - x[3] * t.s, t.cw * x[1] + x[2] + t.sw * x[3], x[1] * t.s + x[3] * t.c, x[4]}};
410 }
411
412 [[nodiscard]] constexpr linalg::fixed_matrix<T, 5, 5> jacobian(const linalg::fixed_vector<T, 5>& x, T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
413 {
414 const T w = x[4], vx = x[1], vy = x[3];
415 const auto t = detail::transition::turn(w, dt);
416 // d(sin a / w)/dw and d((1 - cos a)/w)/dw, by series when a = w dt is small
417 const T a = w * dt;
418 T dsw, dcw;
419 if (detail::transition::abs(a) < T(0.5)) {
420 // dt² times the derivatives of sin a / a and (1 - cos a) / a
421 const auto r = detail::transition::turn_series_at(a);
422 dsw = dt * dt * r.df;
423 dcw = dt * dt * r.dg;
424 } else {
425 dsw = (dt * t.c * w - t.s) / (w * w);
426 dcw = (dt * t.s * w - (T(1) - t.c)) / (w * w);
427 }
428 return {{T(1), t.sw, T(0), -t.cw, vx * dsw - vy * dcw,
429 T(0), t.c, T(0), -t.s, -dt * (t.s * vx + t.c * vy),
430 T(0), t.cw, T(1), t.sw, vx * dcw + vy * dsw,
431 T(0), t.s, T(0), t.c, dt * (t.c * vx - t.s * vy),
432 T(0), T(0), T(0), T(0), T(1)}};
433 }
434
436 [[nodiscard]] constexpr linalg::fixed_matrix<T, 5, 5> covariance(T dt) const noexcept(linalg::detail::nothrow_scalar<T>)
437 {
438 auto m = linalg::block_diagonal(constant_velocity<T>{q_x}.covariance(dt), constant_velocity<T>{q_y}.covariance(dt));
439 linalg::fixed_matrix<T, 5, 5> r{};
440 linalg::set_block<0, 0>(r, m);
441 r(4, 4) = q_omega * detail::transition::abs(dt);
442 return r;
443 }

464 struct combined {
