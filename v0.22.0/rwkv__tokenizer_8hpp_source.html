72constexpr result<std::size_t, rwkv_error> rwkv_unescape(std::string_view escaped, std::span<char> out) noexcept
73{
74 const auto hex = [](char c) -> int {
75 if (c >= '0' && c <= '9')
76 return c - '0';
77 if (c >= 'a' && c <= 'f')
78 return c - 'a' + 10;
79 if (c >= 'A' && c <= 'F')
80 return c - 'A' + 10;
81 return -1;
82 };
83 std::size_t at = 0;
84 for (std::size_t i = 0; i < escaped.size(); ++i) {
85 char c = escaped[i];
86 if (c == '\\') {
87 if (++i == escaped.size())
88 break; // a trailing backslash escapes nothing, as llama.cpp reads it
89 const char e = escaped[i];
90 if (e == 'x') {
91 if (i + 2 >= escaped.size())
92 return err(rwkv_error::bad_escape);
93 const int hi = hex(escaped[i + 1]), lo = hex(escaped[i + 2]);
94 if (hi < 0 || lo < 0)
95 return err(rwkv_error::bad_escape);
96 c = static_cast<char>(hi * 16 + lo);
97 i += 2;
98 } else {
99 c = e == 't' ? '\t' : e == 'n' ? '\n' : e == 'r' ? '\r' : e;
100 }
101 }
102 if (at == out.size())
103 return err(rwkv_error::output_too_small);
104 out[at++] = c;
105 }
106 return ok(at);
107}

115constexpr result<std::size_t, rwkv_error> rwkv_encode(std::string_view text, const token_trie& vocab, std::uint32_t unknown,
116 std::span<std::uint32_t> out) noexcept
117{
118 std::size_t count = 0;
119 for (std::size_t at = 0; at < text.size();) {
120 std::size_t n = 0;
121 std::uint32_t id = vocab.longest_prefix(text.substr(at), n);
122 if (id == no_token) {
123 id = unknown;
124 n = 1;
125 }
126 if (count == out.size())
127 return err(rwkv_error::output_too_small);
128 out[count++] = id;
129 at += n;
130 }
131 return ok(count);
132}
