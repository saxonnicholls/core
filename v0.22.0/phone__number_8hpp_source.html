94inline constexpr std::array<std::uint64_t, 16> e164_pow10 = [] {
95 std::array<std::uint64_t, 16> t{};
96 std::uint64_t p = 1;
97 for (auto& v : t) {
98 v = p;
99 p *= 10;
100 }
101 return t;
102}();
103
104// Per code 0..999: bit 0 assigned, bit 1 a leading 0 after it is the trunk prefix
105inline constexpr std::array<std::uint8_t, 1000> e164_code_flags = [] {
106 std::array<std::uint8_t, 1000> t{};
107 for (const auto& e : e164::country_codes)
108 t[e.code] = static_cast<std::uint8_t>(1U | (e.trunk_zero ? 2U : 0U));
109 return t;
110}();
111
112// The representation: the digits as an integer scaled to 15 digits (so integer order is string order), then the
113// digit count (4 bits) and the code length (2 bits). 10^15 < 2^50: 56 bits in all.
114constexpr std::uint64_t e164_aligned(std::uint64_t packed) noexcept { return packed >> 6; }

186 static constexpr std::size_t max_digits = 15;

190 [[nodiscard]] constexpr bool empty() const noexcept { return packed == 0; }
191
194 [[nodiscard]] constexpr std::size_t digit_count() const noexcept { return (packed >> 2) & 0xFU; }
195
198 [[nodiscard]] constexpr std::size_t country_code_length() const noexcept { return packed & 0x3U; }
199
202 [[nodiscard]] constexpr std::uint16_t country_code() const noexcept
203 {
204 return static_cast<std::uint16_t>(detail::e164_aligned(packed) /
205 detail::e164_pow10[max_digits - country_code_length()]);
206 }
207
210 [[nodiscard]] constexpr std::size_t national_number_length() const noexcept
211 {
212 return digit_count() - country_code_length();
213 }
214
219 [[nodiscard]] constexpr char digit(std::size_t i) const noexcept
220 {
221 return static_cast<char>('0' + detail::e164_aligned(packed) / detail::e164_pow10[max_digits - 1 - i] % 10);
222 }
223
226 [[nodiscard]] constexpr phone_number_text national_number() const noexcept
227 {
228 phone_number_text t;
229 const std::size_t c = country_code_length();
230 for (std::size_t i = c; i < digit_count(); ++i)
231 t.chars[i - c] = digit(i);
232 t.length = static_cast<std::uint8_t>(national_number_length());
233 return t;
234 }

266constexpr std::to_chars_result to_chars(char* first, char* last, const phone_number& p) noexcept
267{
268 const std::size_t n = p.digit_count();
269 if (n == 0) return {first, std::errc{}};
270 if (last - first < static_cast<std::ptrdiff_t>(n + 1)) return {last, std::errc::value_too_large};
271 // All 15 places by constant division (a multiply), then the first n: no division by a variable power of 10
272 std::array<char, phone_number::max_digits> all{};
273 std::uint64_t v = detail::e164_aligned(p.packed);
274 for (std::size_t k = all.size(); k > 0; --k) {
275 all[k - 1] = static_cast<char>('0' + v % 10);
276 v /= 10;
277 }
278 first[0] = '+';
279 for (std::size_t k = 0; k < n; ++k)
280 first[k + 1] = all[k];
281 return {first + n + 1, std::errc{}};
282}

338[[nodiscard]] constexpr result<phone_number, phone_errc> parse_phone_number(std::string_view text) noexcept
339{
340 std::array<std::uint8_t, phone_number::max_digits> d{};
341 std::size_t n = 0;
342 bool international = false;
343 bool open = false;
344 for (std::size_t i = 0; i < text.size(); ++i) {
345 const char c = text[i];
346 if (c >= '0' && c <= '9') {
347 if (!international) {
348 if (c != '0' || i + 1 == text.size() || text[i + 1] != '0') return err(phone_errc::missing_country_code);
349 international = true;
350 ++i;
351 continue;
352 }
353 if (n < d.size()) d[n] = static_cast<std::uint8_t>(c - '0');
354 ++n;
355 } else if (c == '+') {
356 if (international) return err(phone_errc::invalid_character);
357 international = true;
358 } else if (c == '(') {
359 if (open) return err(phone_errc::unbalanced_parentheses);
360 open = true;
361 } else if (c == ')') {
362 if (!open) return err(phone_errc::unbalanced_parentheses);
363 open = false;
364 } else if (c != ' ' && c != '-' && c != '.' && c != '/') {
365 return err(phone_errc::invalid_character);
366 }
367 }
368 if (open) return err(phone_errc::unbalanced_parentheses);
369 if (!international) return err(phone_errc::empty);
370 if (n > phone_number::max_digits) return err(phone_errc::too_long);
371 if (n == 0) return err(phone_errc::too_short);
372 if (d[0] == 0) return err(phone_errc::unknown_country_code);
373 std::size_t cc_length = 0;
374 unsigned code = 0;
375 for (std::size_t k = 0; k < 3 && k < n && cc_length == 0; ++k) {
376 code = code * 10 + d[k];
377 if ((detail::e164_code_flags[code] & 1U) != 0) cc_length = k + 1;
378 }
379 if (cc_length == 0) return err(phone_errc::unknown_country_code);
380 if (n - cc_length < 2) return err(phone_errc::too_short);
381 if (d[cc_length] == 0 && (detail::e164_code_flags[code] & 2U) != 0) return err(phone_errc::trunk_prefix);
382 std::uint64_t v = 0;
383 for (std::size_t k = 0; k < n; ++k)
384 v = v * 10 + d[k];
385 v *= detail::e164_pow10[phone_number::max_digits - n];
386 return ok(phone_number{(v << 6) | (static_cast<std::uint64_t>(n) << 2) | cc_length});
387}
