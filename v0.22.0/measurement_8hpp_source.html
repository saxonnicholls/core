79 template <typename M>
80 concept measurement_model = requires(const M& m, const linalg::fixed_vector<typename M::value_type, M::state_dim>& x) {
81 { M::kinds } -> std::convertible_to<geometry::angle_kinds<M::measurement_dim>>;
82 { m.measure(x) } -> std::same_as<linalg::fixed_vector<typename M::value_type, M::measurement_dim>>;
83 { m.jacobian(x) } -> std::same_as<linalg::fixed_matrix<typename M::value_type, M::measurement_dim, M::state_dim>>;
84 { m.covariance() } -> std::same_as<linalg::fixed_matrix<typename M::value_type, M::measurement_dim, M::measurement_dim>>;
85 };

128 // Writes the M x D derivative with respect to the mapped components into the M x N Jacobian
129 template <std::size_t N, auto Map, typename T, std::size_t M, std::size_t D>
130 constexpr linalg::fixed_matrix<T, M, N> scatter(const linalg::fixed_matrix<T, M, D>& d) noexcept(linalg::detail::nothrow_scalar<T>)
131 {
132 linalg::fixed_matrix<T, M, N> j{};
133 for (std::size_t r = 0; r < M; ++r)
134 for (std::size_t k = 0; k < D; ++k)
135 j(r, static_cast<std::size_t>(Map[k])) += d(r, k);
136 return j;
137 }

139 // d(bearing)/dp and d(range)/dp for p in the plane (rows), zero at the origin
140 template <typename T>
141 struct polar_rows {
142 T phi, rho;
143 linalg::fixed_matrix<T, 1, 2> dphi, drho;
144 };
145 template <typename T>
146 constexpr polar_rows<T> polar(const linalg::fixed_vector<T, 2>& p) noexcept(linalg::detail::nothrow_scalar<T>)
147 {
148 const T r2 = p[0] * p[0] + p[1] * p[1];
149 if (r2 == T(0))
150 return {T(0), T(0), {}, {}};
151 const T r = static_cast<T>(math::sqrt(r2));
152 return {static_cast<T>(math::atan2(p[1], p[0])), r, {{-p[1] / r2, p[0] / r2}}, {{p[0] / r, p[1] / r}}};
153 }

155 // (elevation, bearing, range) and their derivatives for p in space
156 template <typename T>
157 struct sphere_rows {
158 T theta, phi, rho;
159 linalg::fixed_matrix<T, 1, 3> dtheta, dphi, drho;
160 };
161 template <typename T>
162 constexpr sphere_rows<T> sphere(const linalg::fixed_vector<T, 3>& p) noexcept(linalg::detail::nothrow_scalar<T>)
163 {
164 const T h2 = p[0] * p[0] + p[1] * p[1];
165 const T r2 = h2 + p[2] * p[2];
166 if (r2 == T(0))
167 return {T(0), T(0), T(0), {}, {}, {}};
168 const T r = static_cast<T>(math::sqrt(r2)), h = static_cast<T>(math::sqrt(h2));
169 sphere_rows<T> s{static_cast<T>(math::asin(p[2] / r)), h2 == T(0) ? T(0) : static_cast<T>(math::atan2(p[1], p[0])), r, {}, {}, {{p[0] / r, p[1] / r, p[2] / r}}};
170 if (h2 > T(0)) {
171 s.dtheta = {{-p[0] * p[2] / (r2 * h), -p[1] * p[2] / (r2 * h), h / r2}};
172 s.dphi = {{-p[1] / h2, p[0] / h2, T(0)}};
173 }
174 return s;
175 }

379 template <linalg::scalar T, std::size_t N, std::array Map, std::array VelMap>
380 struct bearing_range_rate {
381 static constexpr std::size_t d = Map.size();
382 static_assert((d == 2 || d == 3) && VelMap.size() == d && detail::measurement::valid_map<N, Map>() && detail::measurement::valid_map<N, VelMap>(),
383 "snicholls::tracking::bearing_range_rate: map 2 or 3 position and as many velocity components inside the state");
384 using value_type = T;
385 static constexpr std::size_t state_dim = N;
386 static constexpr std::size_t measurement_dim = 3;
387 static constexpr geometry::angle_kinds<3> kinds{geometry::angle_kind::bearing, geometry::angle_kind::linear, geometry::angle_kind::linear};
388 linalg::fixed_matrix<T, 3, 3> noise{};
389 linalg::fixed_vector<T, d> translation{};
390 linalg::fixed_matrix<T, 3, 3> rotation = linalg::fixed_matrix<T, 3, 3>::identity();
391 linalg::fixed_vector<T, d> velocity{};
392
393 private:
394 struct parts {
395 linalg::fixed_vector<T, d> rel, vel;
396 T phi, rho, rate;
397 linalg::fixed_matrix<T, 1, d> dphi;
398 };
399 constexpr parts compute(const linalg::fixed_vector<T, N>& x) const noexcept(linalg::detail::nothrow_scalar<T>)
400 {
401 const auto a = detail::measurement::upper_left<T, d>(rotation);
402 const auto rel = detail::measurement::gather<T, d, N, Map>(x) - translation;
403 const auto vel = detail::measurement::gather<T, d, N, VelMap>(x) - velocity;
404 const auto p = a * rel;
405 const linalg::fixed_vector<T, 2> pxy{{p[0], p[1]}};
406 const auto pr = detail::measurement::polar(pxy);
407 const T rho = linalg::norm(rel);
408 linalg::fixed_matrix<T, 1, d> dphi_p{};
409 dphi_p(0, 0) = pr.dphi(0, 0);
410 dphi_p(0, 1) = pr.dphi(0, 1);
411 return {rel, vel, pr.phi, rho, rho > T(0) ? linalg::dot(rel, vel) / rho : T(0), dphi_p * a};
412 }
413
414 public:
416 [[nodiscard]] constexpr linalg::fixed_vector<T, 3> measure(const linalg::fixed_vector<T, N>& x) const noexcept(linalg::detail::nothrow_scalar<T>)
417 {
418 const auto c = compute(x);
419 return {{c.phi, c.rho, c.rate}};
420 }
421
422 [[nodiscard]] constexpr linalg::fixed_matrix<T, 3, N> jacobian(const linalg::fixed_vector<T, N>& x) const noexcept(linalg::detail::nothrow_scalar<T>)
423 {
424 const auto c = compute(x);
425 linalg::fixed_matrix<T, 3, N> j{};
426 if (c.rho == T(0))
427 return j;
428 for (std::size_t k = 0; k < d; ++k) {
429 const auto pk = static_cast<std::size_t>(Map[k]), vk = static_cast<std::size_t>(VelMap[k]);
430 j(0, pk) += c.dphi(0, k);
431 j(1, pk) += c.rel[k] / c.rho;
432 j(2, pk) += c.vel[k] / c.rho - c.rate * c.rel[k] / (c.rho * c.rho);
433 j(2, vk) += c.rel[k] / c.rho;
434 }
435 return j;
436 }
437
438 [[nodiscard]] constexpr linalg::fixed_matrix<T, 3, 3> covariance() const noexcept { return noise; }
439 };
