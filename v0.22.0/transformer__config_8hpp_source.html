280[[nodiscard]] constexpr bool valid_long_context(const transformer_config& c) noexcept
281{
282 const bool standard = c.q_lora_rank == 0 && c.qk_rope_dim == 0 && c.v_head_dim == 0;
283 const bool latent = c.qk_rope_dim > 0 && c.qk_rope_dim % 2 == 0 && c.qk_rope_dim < c.head_dim && c.v_head_dim > 0
284 && c.rope != rope_kind::none && c.n_kv_heads == c.n_heads && c.mask == attention_mask::causal;
285 const bool yarn = c.rope_scaling == rope_scaling_kind::yarn || c.rope_scaling == rope_scaling_kind::yarn_index_ramp;
286 const bool scaling = c.rope_factor >= 1 && c.yarn_mscale >= 0
287 && (!yarn || (c.rope_original_context > 0 && c.yarn_beta_fast > c.yarn_beta_slow && c.yarn_beta_slow > 0));
288 const bool mtp = c.mtp_loss_weight >= 0 && (c.mtp_depth > 0 || c.mtp_loss_weight == 0);
289 const bool options = c.kv_lora_rank > 0 || (!c.attention_sinks && c.sparse_top_k == 0); // the latent attention's options
290 return (c.kv_lora_rank == 0 ? standard : latent) && scaling && mtp && options;
291}

299[[nodiscard]] constexpr bool valid_mixer(const transformer_config& c) noexcept
300{
301 if (c.mixer == mixer_kind::none)
302 return c.mixer_heads == 0 && c.mixer_head_dim == 0 && c.mixer_state == 0 && c.mixer_groups == 0 && c.conv_width == 0
303 && c.mixer_chunk == 0 && c.hybrid_period == 0 && c.hybrid_attention_offset == 0;
304 const bool rwkv = c.mixer == mixer_kind::rwkv6 || c.mixer == mixer_kind::rwkv7;
305 const bool conv = c.mixer == mixer_kind::mamba2 || c.mixer == mixer_kind::gated_deltanet;
306 return c.mixer_heads > 0 && c.mixer_head_dim > 0 && c.mixer_state > 0 && c.mixer_groups > 0 && c.mixer_heads % c.mixer_groups == 0
307 && c.mixer_chunk > 0 && (!rwkv || (c.mixer_groups == c.mixer_heads && c.mixer_state == c.mixer_head_dim))
308 && (!conv || c.conv_width > 0)
309 && (c.hybrid_period == 0 ? c.hybrid_attention_offset == 0 : c.hybrid_attention_offset < c.hybrid_period);
310}

325[[nodiscard]] constexpr bool layer_is_attention(const transformer_config& c, std::size_t layer) noexcept
326{
327 if (c.mixer == mixer_kind::none)
328 return true;
329 return c.hybrid_period != 0 && layer % c.hybrid_period == c.hybrid_attention_offset;

339[[nodiscard]] constexpr bool valid(const transformer_config& c) noexcept
340{
341 const bool dense = c.experts == 0 && c.top_k == 0 && c.shared_experts == 0 && c.expert_d_ff == 0 && c.expert_groups == 0 && c.group_top_k == 0;
342 const bool groups = (c.expert_groups == 0 && c.group_top_k == 0)
343 || (c.expert_groups > 0 && c.experts % c.expert_groups == 0 && c.group_top_k > 0 && c.group_top_k <= c.expert_groups
344 && c.group_top_k * (c.experts / c.expert_groups) >= c.top_k);
345 const bool moe = c.experts > 0 && c.experts <= 0xffffffffu && c.top_k > 0 && c.top_k <= c.experts && c.expert_d_ff > 0
346 && c.route_scale > 0 && c.bias_update_rate >= 0 && c.balance_loss_weight >= 0 && c.z_loss_weight >= 0 && groups;
347 return c.d_model > 0 && c.n_heads > 0 && c.n_kv_heads > 0 && c.head_dim > 0 && c.d_ff > 0 && c.n_heads % c.n_kv_heads == 0
348 && (c.rope == rope_kind::none || c.head_dim % 2 == 0) && c.norm_eps > 0 && c.attention_softcap >= 0
349 && (c.rope == rope_kind::none || c.rope_base > 1) && (dense || moe) && (c.mask == attention_mask::causal || c.window == 0)
350 && ((c.mask == attention_mask::block_causal) == (c.mask_block > 0)) && valid_long_context(c) && c.glu_alpha > 0
351 && c.glu_limit > 0 && valid_mixer(c);
352}
