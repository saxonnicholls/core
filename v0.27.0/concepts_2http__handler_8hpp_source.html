194 [[nodiscard]] inline bool match_route(std::string_view pattern, std::string_view path, std::vector<http_header>& captured)
195 {
196 std::vector<http_header> found;
197 std::size_t i = 0, j = 0;
198 while (i < pattern.size() && j < path.size()) {
199 if (pattern[i] != '/' || path[j] != '/')
200 return false;
201 ++i;
202 ++j;
203 const std::size_t pe = std::min(pattern.find('/', i), pattern.size());
204 const std::size_t qe = std::min(path.find('/', j), path.size());
205 const std::string_view seg = pattern.substr(i, pe - i);
206 const std::string_view got = path.substr(j, qe - j);
207 if (!seg.empty() && seg.front() == ':') {
208 if (got.empty())
209 return false;
210 found.push_back({std::string(seg.substr(1)), std::string(got)});
211 } else if (seg != got) {
212 return false;
213 }
214 i = pe;
215 j = qe;
216 }
217 if (i != pattern.size() || j != path.size())
218 return false;
219 captured.insert(captured.end(), found.begin(), found.end());
220 return true;
221 }
