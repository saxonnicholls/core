66 inline http_request kit_echo(http_method m = http_method::get)
67 {
68 http_request r;
69 r.method = m;
70 r.path = "/echo";
71 r.headers.push_back({"X-Echo", "header value"});
72 r.body = std::string("binary\0body\xff", 12);
73 return r;
74 }

81 template <typename Probe>
82 requires detail::http_client_probe<Probe>
83 [[nodiscard]] capability_conformance_report run_http_client_conformance()
84 {
85 capability_conformance_report report;
86
87 detail::capability_check<Probe>(report, "a request reaches the server and its response returns", [](Probe& p, const auto& expect) {
88 const http_request r = detail::kit_echo();
89 const auto got = p.client().send(r);
90 expect(got.is_ok(), "the request is sent");
91 if (!got)
92 return;
93 expect(got->status == 200, "the server's status");
94 expect(got->body == r.body, "the body, byte for byte");
95 expect(find_header(got->headers, "x-echo") == std::optional<std::string_view>{"header value"}, "the header, either way");
96 });
97
98 detail::capability_check<Probe>(report, "an error status is a response", [](Probe& p, const auto& expect) {
99 const auto got = http_get(p.client(), "/no/such/path?x=1");
100 expect(got.is_ok() && got->status == 404, "a 404 is a response, not an error");
101 });
102
103 detail::capability_check<Probe>(report, "HEAD returns no body", [](Probe& p, const auto& expect) {
104 const auto got = p.client().send(detail::kit_echo(http_method::head));
105 expect(got.is_ok() && got->body.empty(), "HEAD has no body");
106 });
107
108 detail::capability_check<Probe>(report, "a malformed request is refused", [](Probe& p, const auto& expect) {
109 http_request no_path = detail::kit_echo();
110 no_path.path = "echo";
111 http_request split = detail::kit_echo();
112 split.headers.push_back({"X-Bad", "a\r\nInjected: yes"});
113 for (const http_request* r : {&no_path, &split}) {
114 const auto got = p.client().send(*r);
115 expect(got.is_err() && got.unwrap_err().code == capability_errc::invalid_argument, "invalid_argument");
116 }
117 });
118
119 detail::capability_check<Probe>(report, "a transport failure leaks nothing", [](Probe& p, const auto& expect) {
120 p.transport_fails(capability_errc::unavailable, capability_provider_response);
121 const auto got = p.client().send(detail::kit_echo());
122 expect(got.is_err() && got.unwrap_err().code == capability_errc::unavailable, "no response is an error");
123 if (got.is_err())
124 expect(!capability_error_leaks(got.unwrap_err(), capability_redaction_marker), "nothing received is kept");
125 });
126
127 return report;
128 }
