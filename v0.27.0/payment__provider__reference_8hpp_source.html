123 capability_result<collection> collect(const collection_request& c)
124 {
125 if (auto bad = check_request(c.idempotency_key, c.our_reference, c.amount))
126 return err(*bad);
127 if (const auto it = collect_keys_.find(c.idempotency_key); it != collect_keys_.end()) {
128 if (it->second.first != fingerprint(c))
129 return fail(capability_errc::conflict, {}, "idempotency key reused for another request");
130 return ok(it->second.second);
131 }
132 if (c.method != payment_method::card && c.method != payment_method::bank_transfer)
133 return fail(capability_errc::unsupported, {}, "card and bank transfer only");
134 if (auto e = take_injected())
135 return err(*e);
136 collection out;
137 out.pay = new_payment("mem_pay_", movement::collection, c.our_reference, c.method, c.amount);
138 if (c.method == payment_method::card) {
139 out.pay.status = payment_status::requires_action;
140 out.next.kind = next_step_kind::redirect;
141 out.next.url = "memory://checkout/" + std::string(out.pay.payment_id.view());
142 } else {
143 auto reference = c.payer_reference.empty() ? make_id<short_text>("MEM-" + std::to_string(++next_reference_))
144 : make_id<short_text>(c.payer_reference);
145 if (!reference)
146 return err(reference.unwrap_err());
147 out.next.kind = next_step_kind::bank_transfer;
148 out.next.bank = bank_details{"Memory Payments", "Memory Bank", "000000001", "000-000", "MEMOXXXX", {}, *reference};
149 awaiting_credit_.emplace(std::string(reference->view()), std::string(out.pay.payment_id.view()));
150 }
151 payments_.insert_or_assign(std::string(out.pay.payment_id.view()), out.pay);
152 collect_keys_.emplace(std::string(c.idempotency_key), std::make_pair(fingerprint(c), out));
153 return ok(std::move(out));
154 }

159 capability_result<payment> payout(const payout_request& o)
160 {
161 if (auto bad = check_request(o.idempotency_key, o.our_reference, o.amount))
162 return err(*bad);
163 if (const auto it = payout_keys_.find(o.idempotency_key); it != payout_keys_.end()) {
164 if (it->second.first != fingerprint(o))
165 return fail(capability_errc::conflict, {}, "idempotency key reused for another request");
166 return ok(it->second.second);
167 }
168 if (auto e = take_injected())
169 return err(*e);
170 if (!beneficiaries_.contains(o.beneficiary_id))
171 return fail(capability_errc::not_found, {}, "no such beneficiary");
172 if (o.quote_id) {
173 const auto q = quotes_.find(*o.quote_id);
174 if (q == quotes_.end())
175 return fail(capability_errc::not_found, {}, "no such quote");
176 if (stamp() > q->second.expires_at)
177 return fail(capability_errc::stale, {}, "the quote expired");
178 }
179 if (auto taken = take(o.amount); !taken)
180 return err(taken.unwrap_err());
181 payment p = new_payment("mem_out_", movement::payout, o.our_reference, payment_method::bank_transfer, o.amount);
182 p.status = payment_status::succeeded;
183 p.received = o.amount;
184 payments_.insert_or_assign(std::string(p.payment_id.view()), p);
185 record(std::string(p.payment_id.view()), fact_kind::payout, -o.amount, money::zero(o.amount.asset_of()),
186 fact_refs{{}, {}, {}, {}, p.payment_id});
187 payout_keys_.emplace(std::string(o.idempotency_key), std::make_pair(fingerprint(o), p));
188 return ok(std::move(p));
189 }

194 capability_result<payment> refund(const refund_request& r)
195 {
196 if (r.idempotency_key.empty() || r.idempotency_key.size() > client_id::capacity())
197 return fail(capability_errc::invalid_argument, {}, "an idempotency key of 1 to 64 characters");
198 if (const auto it = refund_keys_.find(r.idempotency_key); it != refund_keys_.end()) {
199 if (it->second.first != fingerprint(r))
200 return fail(capability_errc::conflict, {}, "idempotency key reused for another request");
201 return ok(it->second.second);
202 }
203 if (auto e = take_injected())
204 return err(*e);
205 const auto found = payments_.find(r.payment_id);
206 if (found == payments_.end() || found->second.kind != movement::collection)
207 return fail(capability_errc::not_found);
208 payment& original = found->second;
209 if (original.status != payment_status::succeeded && original.status != payment_status::partially_refunded)
210 return fail(capability_errc::conflict, {}, "only a succeeded collection is refunded");
211 const auto remaining = subtract(original.received, original.refunded);
212 if (!remaining)
213 return fail(capability_errc::internal);
214 const money amount = r.amount ? *r.amount : *remaining;
215 if (amount.asset_of() != original.received.asset_of())
216 return fail(capability_errc::invalid_argument, {}, "a refund is in the collection's currency");
217 if (amount.is_negative() || amount.is_zero())
218 return fail(capability_errc::invalid_argument, {}, "an amount is positive");
219 if (amount.magnitude() > remaining->magnitude())
220 return fail(capability_errc::conflict, {}, "more than remains to refund");
221 if (auto taken = take(amount); !taken)
222 return err(taken.unwrap_err());
223 original.refunded = *add(original.refunded, amount);
224 original.status = original.refunded == original.received ? payment_status::refunded : payment_status::partially_refunded;
225 original.updated_at = stamp();
226 payment p = new_payment("mem_ref_", movement::refund, original.our_reference.view(), original.method, amount);
227 p.status = payment_status::succeeded;
228 p.received = amount;
229 payments_.insert_or_assign(std::string(p.payment_id.view()), p);
230 record(std::string(p.payment_id.view()), fact_kind::refund, -amount, money::zero(amount.asset_of()),
231 fact_refs{original.payment_id, {}, p.payment_id, {}, {}});
232 refund_keys_.emplace(std::string(r.idempotency_key), std::make_pair(fingerprint(r), p));
233 return ok(std::move(p));
234 }

239 capability_result<fx_quote> quote(const quote_request& q)
240 {
241 if (q.amount.is_negative() || q.amount.is_zero())
242 return fail(capability_errc::invalid_argument, {}, "an amount is positive");
243 const asset source_unit = q.amount_is_source ? q.amount.asset_of() : q.other;
244 const asset target_unit = q.amount_is_source ? q.other : q.amount.asset_of();
245 const auto r = rates_.find(key(source_unit, target_unit));
246 if (r == rates_.end())
247 return fail(capability_errc::unsupported, {}, "no rate for that pair");
248 money source, target, fee;
249 if (q.amount_is_source) {
250 source = q.amount;
251 fee = fee_on(q.amount);
252 const auto net = subtract(q.amount, fee);
253 const auto converted = net ? convert(*net, r->second, rounding::toward_zero)
254 : result<conversion, money_errc>(err(money_errc::overflow));
255 if (!converted)
256 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
257 target = converted->converted;
258 } else {
259 target = q.amount;
260 const auto back = r->second.inverse();
261 const auto net = back ? convert(q.amount, *back, rounding::ceiling)
262 : result<conversion, money_errc>(err(money_errc::overflow));
263 if (!net)
264 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
265 fee = fee_on(net->converted);
266 source = *add(net->converted, fee);
267 }
268 const fx_quote out{id<provider_id>("mem_q_" + std::to_string(++next_id_)), source, target, fee, r->second,
269 stamp() + quote_lifetime};
270 quotes_.insert_or_assign(std::string(out.quote_id.view()), out);
271 return ok(out);
272 }

320 {
321 std::vector<payment_fact> out;

323 return ok(std::move(out));
324 for (const auto& [related, f] : ledger_)

337 capability_result<statement_page> statement(const statement_query& q)
338 {
339 std::size_t from = 0;
340 if (!q.cursor.empty()) {
341 std::int64_t at = 0;
342 if (!detail::memory_webhook::parse_int(q.cursor, at) || at < 0)
343 return fail(capability_errc::invalid_argument, {}, "not a cursor this provider gave");
344 from = static_cast<std::size_t>(at);
345 }
346 statement_page page;
347 std::size_t i = from;
348 for (; i < ledger_.size() && page.facts.size() < q.limit; ++i) {
349 const payment_fact& f = ledger_[i].second;
350 if (f.settled.gross.asset_of() == q.currency && (q.payout_id.empty() || f.refs.payout == q.payout_id))
351 page.facts.push_back(f);
352 }
353 if (i < ledger_.size())
354 page.next_cursor = id<provider_id>(std::to_string(i));
355 return ok(std::move(page));
356 }

393 capability_result<credit> receive_credit(const money& net, std::string_view reference, std::string_view sender)
394 {
395 if (net.is_negative() || net.is_zero())
396 return fail(capability_errc::invalid_argument, {}, "a credit is positive");
397 auto ref = make_id<short_text>(reference);
398 auto from = make_id<short_text>(sender);
399 if (!ref || !from)
400 return fail(capability_errc::invalid_argument, {}, "a reference or sender longer than 255 characters");
401 const credit c{id<provider_id>("mem_cr_" + std::to_string(++next_id_)), net, money::zero(net.asset_of()), *ref,
402 *from, stamp()};
403 credit_into(net);
404 credits_.push_back(c);
405 std::string related;
406 if (const auto waiting = awaiting_credit_.find(reference); waiting != awaiting_credit_.end()) {
407 payment& p = payments_.at(waiting->second);
408 if (p.amount.asset_of() == net.asset_of()) {
409 p.status = payment_status::succeeded;
410 p.received = net;
411 p.updated_at = c.occurred_at;
412 related = std::string(p.payment_id.view());
413 awaiting_credit_.erase(waiting);
414 }
415 }
416 record(related, fact_kind::bank_credit, net, money::zero(net.asset_of()),
417 fact_refs{id<provider_id>(related), {}, {}, {}, {}}, std::string(c.credit_id.view()));
418 return ok(c);

471 // Every field of a borrowed request, owned, separated: equal fingerprints, equal requests.
472 static void field(std::string& out, std::string_view v)
473 {
474 out += std::to_string(v.size());
475 out += ':';
476 out += v;
477 out += '\x1f';
478 }
479 static void field(std::string& out, const money& m)
480 {
481 field(out, to_string(m));
482 field(out, m.asset_of().code());
483 field(out, std::to_string(m.asset_of().scale()));
484 }
485 static std::string fingerprint(const collection_request& c)
486 {
487 std::string out;
488 for (const std::string_view v : {c.idempotency_key, c.our_reference, c.payer_reference, c.return_url, c.cancel_url})
489 field(out, v);
490 field(out, c.amount);
491 field(out, std::to_string(static_cast<int>(c.method)));
492 for (const auto& [k, v] : c.metadata) {
493 field(out, k);
494 field(out, v);
495 }
496 return out;
497 }
498 static std::string fingerprint(const payout_request& o)
499 {
500 std::string out;
501 for (const std::string_view v : {o.idempotency_key, o.our_reference, o.beneficiary_id, o.reference})
502 field(out, v);
503 field(out, o.amount);
504 field(out, o.quote_id ? "1" : "0");
505 field(out, o.quote_id.value_or(std::string_view{}));
506 return out;
507 }
508 static std::string fingerprint(const refund_request& r)
509 {
510 std::string out;
511 for (const std::string_view v : {r.idempotency_key, r.payment_id, r.reason})
512 field(out, v);
513 if (r.amount)
514 field(out, *r.amount);
515 else
516 field(out, "-");
517 return out;
518 }

526 std::optional<capability_error> check_request(std::string_view idempotency_key, std::string_view our_reference,
527 const money& amount) const
528 {
529 if (idempotency_key.empty() || idempotency_key.size() > client_id::capacity())
530 return capability_error{capability_errc::invalid_argument, {}, "an idempotency key of 1 to 64 characters"};
531 if (our_reference.size() > client_id::capacity())
532 return capability_error{capability_errc::invalid_argument, {}, "our reference is at most 64 characters"};
533 if (amount.is_negative() || amount.is_zero())
534 return capability_error{capability_errc::invalid_argument, {}, "an amount is positive"};
535 if (std::find(currencies_.begin(), currencies_.end(), amount.asset_of()) == currencies_.end())
536 return capability_error{capability_errc::unsupported, {}, "the currency is not held"};
537 return std::nullopt;
538 }

555 money fee_on(const money& m) const
556 {
557 const auto r = rate::make(m.asset_of(), m.asset_of(), fee_bps_, 10000);
558 const auto f = r ? convert(m, *r, rounding::ceiling) : result<conversion, money_errc>(err(money_errc::overflow));
559 return f ? f->converted : money::zero(m.asset_of());

562 void settle(payment& p, const money& gross)
563 {
564 p.status = payment_status::succeeded;
565 p.received = gross;
566 p.fee = fee_on(gross);
567 p.updated_at = stamp();

572 // One balance transaction: gross, the fee, and net = gross - fee, so balances(settled) holds by construction.
573 void record(std::string related, fact_kind kind, const money& gross, const money& fee, const fact_refs& refs,
574 const std::string& line_id = {})
575 {
576 payment_fact f;
577 f.source = "memory";
578 f.provider_account = provider_account_;
579 f.fact_id = id<fact_id>(line_id.empty() ? "mem_txn_" + std::to_string(++next_id_) : line_id);
580 f.kind = kind;
581 f.dates.trade_at = stamp();
582 f.dates.known_at = f.dates.trade_at;
583 f.dates.settled_at = f.dates.trade_at; // the reference settles at once
584 f.dates.settles_on = f.dates.trade_at.days();
585 f.settled = settlement{gross, fee, *subtract(gross, fee)};
586 f.refs = refs;
587 ledger_.emplace_back(std::move(related), f);
588 }

590 void credit_into(const money& m)
591 {
592 const auto it = balances_.find(m.asset_of().code());
593 if (it == balances_.end())
594 balances_.emplace(std::string(m.asset_of().code()), m);
595 else
596 it->second = *add(it->second, m);
597 }

599 capability_result<void> take(const money& m)
600 {
601 const auto it = balances_.find(m.asset_of().code());
602 if (it == balances_.end() || it->second.asset_of() != m.asset_of() || it->second.magnitude() < m.magnitude())
603 return fail(capability_errc::insufficient_funds);
604 it->second = *subtract(it->second, m);
605 return ok();
606 }
