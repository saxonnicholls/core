75 // The rows of kv_role_store: keys and values as length-prefixed bytes.

77 static void u8(std::string& o, std::size_t n) { o.push_back(static_cast<char>(static_cast<unsigned char>(n))); }
78 static void text(std::string& o, std::string_view t)
79 {
80 u8(o, t.size());
81 o.append(t);
82 }
83 static void id(std::string& o, const uuid& u)
84 {
85 for (const auto b : u.bytes)
86 o.push_back(static_cast<char>(b));
87 }
88 static void u64(std::string& o, std::uint64_t v)
89 {
90 for (int i = 7; i >= 0; --i)
91 o.push_back(static_cast<char>(static_cast<unsigned char>(v >> (8 * i))));
92 }
93 static void time(std::string& o, utc_s t) { u64(o, static_cast<std::uint64_t>(t.unix_seconds())); }
94
95 static std::string tenant_prefix(char kind, std::string_view tenant)
96 {
97 std::string k(1, kind);
98 text(k, tenant);
99 return k;
100 }
101 static std::string grant_key(std::string_view tenant, const uuid& party, std::string_view role)
102 {
103 std::string k = tenant_prefix('g', tenant);
104 id(k, party);
105 k.append(role);
106 return k;
107 }
108 static void grant(std::string& o, const role_grant& g)
109 {
110 id(o, g.subject);
111 text(o, g.tenant.view());
112 text(o, g.role.view());
113 id(o, g.granted_by);

128 // Reads what the writers wrote; any shortfall or overflow clears `ok`.
129 struct reader {
130 std::string_view in;
131 bool ok = true;
132
133 std::string_view take(std::size_t n)
134 {
135 if (!ok || in.size() < n) {
136 ok = false;
137 return {};
138 }
139 const auto out = in.substr(0, n);
140 in.remove_prefix(n);
141 return out;
142 }
143 std::size_t u8() { const auto b = take(1); return ok ? static_cast<unsigned char>(b[0]) : 0; }
144 template <typename Text>
145 Text text()
146 {
147 const auto t = take(u8());
148 auto r = Text::make(t);
149 ok = ok && r.is_ok();
150 return ok ? *r : Text{};
151 }
152 uuid id()
153 {
154 uuid u;
155 const auto b = take(16);
156 for (std::size_t i = 0; ok && i < 16; ++i)
157 u.bytes[i] = static_cast<std::uint8_t>(b[i]);
158 return u;
159 }
160 std::uint64_t u64()
161 {
162 std::uint64_t v = 0;
163 for (const char c : take(8))
164 v = (v << 8) | static_cast<unsigned char>(c);
165 return v;
166 }
167 utc_s time() { return utc_s::from_unix_seconds(static_cast<std::int64_t>(u64())); }
168 role_grant grant()
169 {
170 role_grant g;
171 g.subject = id();
172 g.tenant = text<tenant_id>();
173 g.role = text<role_id>();
174 g.granted_by = id();
175 g.reason = text<short_text>();
176 g.granted_at = time();
177 const bool expires = u8() != 0;
178 const utc_s at = time();
179 if (expires)
180 g.expires_at = at;
181 return g;
182 }
183 role_event event()
184 {
185 role_event e;
186 e.granted = u8() != 0;
187 e.grant = grant();
188 e.confirmation = text<short_text>();
189 ok = ok && in.empty();
190 return e;
191 }
192 };
193 };

238 capability_result<void> grant(const role_grant& g, std::string_view confirmation)
239 {
240 if (auto e = take_injected())
241 return err(*e);
242 const auto c = short_text::make(confirmation);
243 if (!c)
244 return fail(capability_errc::invalid_argument, {}, "a confirmation longer than 255 characters");
245 std::string value;
246 detail::role_rows::grant(value, g);
247 return append(detail::role_rows::grant_key(g.tenant.view(), g.subject, g.role.view()), value, {true, g, *c});
248 }

258 capability_result<void> revoke(const uuid& party, std::string_view tenant, std::string_view role, const uuid& actor,
259 std::string_view reason, utc_s now)
260 {
261 if (auto e = take_injected())
262 return err(*e);
263 const auto why = short_text::make(reason);
264 if (!why)
265 return fail(capability_errc::invalid_argument, {}, "a reason longer than 255 characters");
266 if (tenant.size() > tenant_id::capacity() || role.size() > role_id::capacity())
267 return fail(capability_errc::not_found, {}, "no such grant");
268 const std::string key = detail::role_rows::grant_key(tenant, party, role);
269 const auto row = kv_get(kv_, key);
270 if (row.is_err())
271 return err(backend_error(row.unwrap_err()));
272 if (!*row)
273 return fail(capability_errc::not_found, {}, "no such grant");
274 detail::role_rows::reader in{**row};
275 role_grant g = in.grant();
276 if (!in.ok || !in.in.empty())
277 return fail(capability_errc::internal, {}, "a stored grant is malformed");
278 g.granted_by = actor;
279 g.reason = *why;
280 g.granted_at = now;
281 return append(key, std::nullopt, {false, g, {}});
282 }

322 static capability_error backend_error(kv_errc e) noexcept
323 {
324 if (e == kv_errc::conflict)
325 return {capability_errc::conflict, {}, "the key-value store refused the write"};
326 if (e == kv_errc::corrupt || e == kv_errc::invalid_argument || e == kv_errc::limit)
327 return {capability_errc::internal, {}, "the key-value store failed"};
328 return {capability_errc::unavailable, {}, "the key-value store is unavailable"};

339 {
340 if (auto e = take_injected())
341 return err(*e);
342 std::vector<role_grant> out;

345 detail::role_rows::reader in{as_text(v)};

349 out.push_back(g);

358 // One batch: the grant row (put, or erased when `value` is empty), the event, the next sequence number.
359 capability_result<void> append(const std::string& key, std::optional<std::string_view> value, const role_event& e)
360 {
361 const auto next = kv_get(kv_, std::string_view{"s"});
362 if (next.is_err())
363 return err(backend_error(next.unwrap_err()));
364 std::uint64_t seq = 0;
365 if (*next) {
366 detail::role_rows::reader in{**next};
367 seq = in.u64();
368 if (!in.ok)
369 return fail(capability_errc::internal, {}, "the stored sequence is malformed");
370 }
371 std::string event_key = detail::role_rows::tenant_prefix('e', e.grant.tenant.view());
372 detail::role_rows::u64(event_key, seq);
373 std::string next_seq;
374 detail::role_rows::u64(next_seq, seq + 1);
375 kv_write_batch batch;
376 if (value)
377 batch.put(key, *value);
378 else
379 batch.erase(key);
380 batch.put(event_key, detail::role_rows::event(e));
381 batch.put(std::string_view{"s"}, next_seq);
382 const auto applied = kv_.apply(batch);
383 if (applied.is_err())
384 return err(backend_error(applied.unwrap_err()));
385 return ok();
386 }

389 mutable std::optional<capability_error> injected_;
