176 // OCaml's compare-style sort: cmp(a, b) < 0 when a comes first.
177 struct sort_fn {
178 template <class Cmp, class L>
179 constexpr auto operator()(Cmp&& cmp, L&& l) const
180 {
181 return fn::sort_by([&cmp](const auto& a, const auto& b) { return std::invoke(cmp, a, b) < 0; }, std::forward<L>(l));
