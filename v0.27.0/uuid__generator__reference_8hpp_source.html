97 // A random_source_like source: eight bytes, or nothing when it fails (it fails closed). constexpr, so a constexpr
98 // source (test_random_source) composes identifiers at compile time.
99 template <typename R>
100 requires uuid_fill_source<R>
101 constexpr std::optional<std::uint64_t> uuid_draw(R& r) noexcept
102 {
103 std::array<std::byte, 8> b{};
104 if (r.fill(b.data(), b.size()) != std::errc{})
105 return std::nullopt;
106 std::uint64_t v = 0;
107 for (const std::byte x : b)
108 v = (v << 8) | static_cast<std::uint64_t>(x);
109 return v;
110 }

133 template <uuid_random_source Random, uuid_clock Clock>
134 class basic_uuid_generator {
135 public:
139 constexpr basic_uuid_generator(Random random, Clock clock) noexcept(
140 std::is_nothrow_move_constructible_v<Random> && std::is_nothrow_move_constructible_v<Clock>)
141 : random_(std::move(random)), clock_(std::move(clock))
142 {
143 }
144
147 constexpr capability_result<uuid> v4()
148 {
149 const auto hi = detail::uuid_draw(random_);
150 if (!hi)
151 return fail(capability_errc::unavailable, {}, "the random source failed");
152 const auto lo = detail::uuid_draw(random_);
153 if (!lo)
154 return fail(capability_errc::unavailable, {}, "the random source failed");
155 std::array<std::uint8_t, 16> bytes{};
156 for (std::size_t i = 0; i < 8; ++i) {
157 bytes[i] = static_cast<std::uint8_t>(*hi >> (56 - 8 * i));
158 bytes[8 + i] = static_cast<std::uint8_t>(*lo >> (56 - 8 * i));
159 }
160 return ok(make_uuid_v4(bytes));
161 }
162
165 constexpr capability_result<uuid> v7()
166 {
167 const std::int64_t now = static_cast<std::int64_t>(clock_.now().unix_millis());
168 if (now < 0 || now > detail::uuid_v7_max_ms)
169 return fail(capability_errc::invalid_argument, {}, "the clock is outside version 7's range");
170 const auto r = detail::uuid_draw(random_);
171 if (!r)
172 return fail(capability_errc::unavailable, {}, "the random source failed");
173 std::int64_t ms = now;
174 std::uint32_t counter = static_cast<std::uint32_t>(*r >> 53); // 11 bits: room to count within the ms
175 if (started_ && ms <= last_ms_) {
176 ms = last_ms_;
177 counter = last_counter_ + 1;
178 if (counter > 0x0FFFu) {
179 if (ms == detail::uuid_v7_max_ms)
180 return fail(capability_errc::invalid_argument, {}, "the clock is outside version 7's range");
181 ++ms;
182 counter = static_cast<std::uint32_t>(*r >> 53);
183 }
184 }
185 started_ = true;
186 last_ms_ = ms;
187 last_counter_ = counter;
188 return ok(make_uuid_v7(static_cast<std::uint64_t>(ms), static_cast<std::uint16_t>(counter), *r));
189 }
190
193 [[nodiscard]] constexpr Random& random() noexcept { return random_; }
196 [[nodiscard]] constexpr Clock& clock() noexcept { return clock_; }
197
198 private:
199 Random random_;
200 Clock clock_;
201 bool started_ = false;
202 std::int64_t last_ms_ = 0;
203 std::uint32_t last_counter_ = 0;
204 };
