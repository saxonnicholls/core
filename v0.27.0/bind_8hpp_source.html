113 // What a bound argument becomes in the call; Args is a std::tuple of references to the call's arguments.
114 // Three cases that cannot overlap: a placeholder, a nested bind, a value.
115 // The position is checked in the template parameters, not a requires-clause, so that a compiler that
116 // substitutes into the return type first (Apple Clang 17) never names an element that does not exist.
117 template <class B, class Args>
118 inline constexpr bool names_an_argument_v = false;
119 template <std::size_t N, class Args>
120 inline constexpr bool names_an_argument_v<arg_t<N>, Args> = (N <= std::tuple_size_v<Args>);
121
122 template <class B, class Args,
123 std::enable_if_t<names_an_argument_v<std::remove_cvref_t<B>, std::remove_reference_t<Args>>, int> = 0>
124 constexpr auto resolve(B&&, Args&& args)
125 SNICHOLLS_FN_RETURNS(std::get<std::remove_cvref_t<B>::position - 1>(std::move(args)))
126
127 template <class B, class Args>
128 requires is_bind_v<std::remove_cvref_t<B>>
129 constexpr auto resolve(B&& b, Args&& args)
130 SNICHOLLS_FN_RETURNS(detail::apply_tuple(std::forward<B>(b), args, tuple_indices<Args>{}))
131
132 template <class B, class Args>
133 requires(!is_placeholder_v<std::remove_cvref_t<B>> && !is_bind_v<std::remove_cvref_t<B>>)
134 constexpr B&& resolve(B&& b, Args&&) noexcept
135 {
136 return std::forward<B>(b);
137 }

139 // Part 0 is f, parts 1... the bound arguments.
140 struct bind_op {
141 template <std::size_t... I, class S, class Args>
142 static constexpr auto with(std::index_sequence<0, I...>, S&& s, Args&& args)
143 SNICHOLLS_FN_RETURNS(std::invoke(detail::part<0>(std::forward<S>(s)),
144 detail::resolve(detail::part<I>(std::forward<S>(s)), args)...))
145
146 template <std::size_t... I, class S, class... A>
147 static constexpr auto call(std::index_sequence<I...> is, S&& s, A&&... a)
148 SNICHOLLS_FN_RETURNS(bind_op::with(is, std::forward<S>(s), std::forward_as_tuple(std::forward<A>(a)...)))
149 };
