138template <math::math_number T>
139struct goertzel_coefficients
140{
141 T coeff = T(2);
142 T cos_w = T(1);
143 T sin_w = T(0);
144 complex<T> block_phase{T(1), T(0)};
145 T bin = T(0);
146 std::size_t n = 1;
147
152 static constexpr goertzel_coefficients at_bin(std::size_t k, std::size_t n) noexcept
153 {
154 assert(n >= 1);
155 using C = compute_t<T>;
156 k %= n;
157 const complex<C> u = detail::complex_::unit<C>(k, n); // (cos, sin) of 2 pi k / n
158 return {static_cast<T>(C(2) * u.re), static_cast<T>(u.re), static_cast<T>(u.im), {T(1), T(0)},
159 static_cast<T>(static_cast<long long>(k)), n};
160 }
161
166 static constexpr goertzel_coefficients at_fractional_bin(compute_t<T> k, std::size_t n) noexcept
167 {
168 assert(n >= 1);
169 using C = compute_t<T>;
170 const C nn = C(static_cast<long long>(n));
171 k -= nn * math::floor(k / nn); // in [0, n]
172 const C whole = math::round(k);
173 if (math::abs(k - whole) <= C(8) * nn / C(math::math_limits<C>::whole_from)) // 4 eps n: on a bin
174 return at_bin(static_cast<std::size_t>(static_cast<long long>(whole)), n);
175 const C two_pi = C(2) * math::math_limits<C>::pi;
176 const C c = math::cos(two_pi * (k / nn)), s = math::sin(two_pi * (k / nn));
177 const C frac = two_pi * (k - math::floor(k)); // e^(-i w n) = e^(-2 pi i frac(k))
178 return {static_cast<T>(C(2) * c), static_cast<T>(c), static_cast<T>(s),
179 {static_cast<T>(math::cos(frac)), static_cast<T>(-math::sin(frac))}, static_cast<T>(k), n};
180 }
181
186 static constexpr goertzel_coefficients at_cycles(compute_t<T> cycles, std::size_t n) noexcept
187 {
188 return at_fractional_bin(cycles * compute_t<T>(static_cast<long long>(n)), n);
189 }
190
196 static constexpr goertzel_coefficients at_frequency(double frequency, double sample_rate, std::size_t n) noexcept
197 {
198 assert(sample_rate > 0);
199 using C = compute_t<T>;
200 return at_fractional_bin(C(frequency) * C(static_cast<long long>(n)) / C(sample_rate), n);
201 }
202
206 [[nodiscard]] constexpr complex<T> phase_after(std::size_t c) const noexcept
207 {
208 if (c == n)
209 return block_phase;
210 using C = compute_t<T>;
211 const C t = C(bin) * C(static_cast<long long>(c)) / C(static_cast<long long>(n)); // w c / (2 pi)
212 const C a = C(2) * math::math_limits<C>::pi * (t - math::floor(t));
213 return {static_cast<T>(math::cos(a)), static_cast<T>(-math::sin(a))};
214 }
215};

220template <math::math_number T>
221struct goertzel_state
222{
223 T s1 = T(0);
224 T s2 = T(0);
225 std::size_t count = 0;
226
230 constexpr void push(T coeff, T x) noexcept
231 {
232 const T s0 = (x - s2) + coeff * s1; // x - s2 is off the critical path: one multiply and one add per sample
233 s2 = s1;
234 s1 = s0;
235 ++count;
236 }
237
239 constexpr void reset() noexcept { *this = goertzel_state{}; }
240
244 [[nodiscard]] constexpr T power(const goertzel_coefficients<T>& c) const noexcept
245 {
246 const T re = c.cos_w * s1 - s2, im = c.sin_w * s1;
247 return re * re + im * im;
248 }
249
253 [[nodiscard]] constexpr complex<T> bin(const goertzel_coefficients<T>& c) const noexcept
254 {
255 const complex<T> y{c.cos_w * s1 - s2, c.sin_w * s1};
256 return c.phase_after(count) * y; // exactly y for an integer bin over a full block
257 }
258};

263 template <typename T>
264 constexpr T weighted(T x, std::span<const T> window, std::size_t at) noexcept
265 {
266 assert(at < window.size());
267 return x * window[at];
