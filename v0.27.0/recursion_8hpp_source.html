108 struct hylo_fn {
109 template <class Alg, class Z, class Coalg, class S, class V = std::remove_cvref_t<Z>>
110 constexpr V operator()(Alg&& alg, Z&& z, Coalg&& coalg, S&& seed) const
111 {
112 V acc(std::forward<Z>(z));
113 std::remove_cvref_t<S> state(std::forward<S>(seed));
114 for (auto step = std::invoke(coalg, std::as_const(state)); step.has_value();
115 step = std::invoke(coalg, std::as_const(state))) {
116 acc = std::invoke(alg, std::move(acc), std::move(step->first));
117 state = std::move(step->second);
118 }
119 return acc;
120 }

154 struct trampoline_fn {
155 template <class Step, class S, class B = std::remove_cvref_t<std::invoke_result_t<Step&, std::remove_cvref_t<S>>>,
156 class R = typename bounce_types<B>::result>
157 constexpr R operator()(Step&& step, S&& start) const
158 {
159 B b = more_t<typename bounce_types<B>::state>{std::forward<S>(start)};
160 while (b.index() == 0) b = std::invoke(step, std::move(std::get<0>(b).state));
161 return std::move(std::get<1>(b).result);
162 }

201 template <std::size_t N, class Coalg, class S>
202 [[nodiscard]] constexpr auto ana(Coalg&& coalg, S&& seed)
203 {
204 using state_t = std::remove_cvref_t<S>;
205 using step_t = std::remove_cvref_t<std::invoke_result_t<Coalg&, const state_t&>>;
206 using T = std::remove_cvref_t<typename step_t::value_type::first_type>;
207 containers::inplace_vector<T, N> out;
208 state_t state(std::forward<S>(seed));
209 while (out.size() < N) {
210 step_t step = std::invoke(coalg, std::as_const(state));
211 if (!step.has_value()) break;
212 out.push_back(std::move(step->first));
213 state = std::move(step->second);
214 }
215 return out;
216 }
