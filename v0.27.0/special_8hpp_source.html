76// Halley's stopping tolerance: 1e-16, as for double, unless the type is wider (80-bit, quad), where twice its epsilon
77template <typename T>
78constexpr T halley_tolerance_() { return std::numeric_limits<T>::epsilon() < T(5e-17) ? T(2) * std::numeric_limits<T>::epsilon() : T(1e-16); }

80// The tolerance at -1/e: a rounded -1/e may fall an ulp below it. 1e-12 in double and wider; a float's epsilon for a
81// float argument, whose nearest value to -1/e lies 9e-9 below it.
82template <typename X>
83constexpr double branch_fuzz_() { return std::numeric_limits<X>::epsilon() > 1e-12 ? double(std::numeric_limits<X>::epsilon()) : 1e-12; }

85// W0 by Corless et al. (1996): a Puiseux seed at the branch point, a log-asymptotic seed for large z, Halley's iteration.
86template <typename T>
87constexpr T lambert_w0_c(T z, T fuzz)
88{
89 if (is_nan(z))
90 return z;
91 if (is_inf(z))
92 return z > T(0) ? inf_t<T> : nan_t<T>;
93 if (z <= -inv_e_t<T>) {
94 // tolerate the one-ulp fuzz of a rounded -1/E (which is itself just below -1/e); refuse below
95 if (z < -inv_e_t<T> - fuzz)
96 return nan_t<T>;
97 return T(-1);
98 }
99 if (z == T(0))
100 return T(0);
101
102 // ---- seed ---------------------------------------------------------------
103 T w = T(0);
104 const T p2 = T(2) * (e_t<T> * z + T(1)); // 2(1 + E z) >= 0 on the domain
105 if (p2 < T(2.0e-2)) {
106 // branch-point Puiseux series in p = sqrt(2(1 + E z)) (Corless et
107 // al. 1996, eq. 4.22): W = -1 + p - p^2/3 + 11 p^3/72 - 43 p^4/540
108 const T p = sqrt(p2);
109 w = T(-1) + p * (T(1) + p * (T(-1) / T(3) + p * (T(11) / T(72) - p * T(43) / T(540))));
110 // The series' error is about p^5: 1e-25 at p = 1e-5, below double's and 80-bit's rounding; a quad needs p < 1e-8
111 // (Halley below would divide by about 0)
112 if (p < (std::numeric_limits<T>::epsilon() < T(1e-20) ? T(1e-8) : T(1e-5)))
113 return w;
114 } else if (z < T(3)) {
115 w = z >= T(0) ? log(T(1) + z) : z * (T(1) - z); // crude, Halley finishes
116 } else {
117 // asymptotic seed W ~ L1 - L2 + L2/L1 + L2(L2-2)/(2 L1^2)
118 // (Corless et al. 1996, eq. 4.19)
119 const T l1 = log(z);
120 const T l2 = log(l1);
121 w = l1 - l2 + l2 / l1 + l2 * (l2 - T(2)) / (T(2) * l1 * l1);
122 constexpr T big = std::numeric_limits<T>::max_exponent10 > 308 ? std::numeric_limits<T>::max() / T(1e8) : T(1.0e300);
123 if (z > big)
124 return w; // w e^w would overflow inside Halley; the asymptotic series is ~1e-9 relative here
125 }
126
127 // ---- Halley (cubic convergence; see Corless et al. 1996, sec. 5) --------
128 for (int i = 0; i < 32; ++i) {
129 const T ew = exp(w);
130 const T f = w * ew - z;
131 const T wp1 = w + T(1);
132 const T denom = ew * wp1 - (w + T(2)) * f / (T(2) * wp1);
133 const T step = f / denom;
134 w -= step;
135 if (abs(step) <= halley_tolerance_<T>() * (abs(w) + T(1e-300)))
136 break;
137 }
138 return w;
139}

141// W-1 on [-1/e, 0): the Puiseux series with p = -sqrt(2(1 + E z)) near the branch point, the log-asymptotic seed
142// L1 - L2 + L2/L1 (L1 = log(-z), L2 = log(-L1)) towards 0, then Halley's iteration on g(w) = w + log(-w) - log(-z),
143// which is w e^w = z in logarithms and stays finite where e^w underflows.
144template <typename T>
145constexpr T lambert_wm1_c(T z, T fuzz)
146{
147 if (is_nan(z))
148 return z;
149 if (z >= T(0))
150 return z == T(0) ? -inf_t<T> : nan_t<T>;
151 if (z <= -inv_e_t<T>)
152 return z < -inv_e_t<T> - fuzz ? nan_t<T> : T(-1);
153
154 T w = T(0);
155 if (z < T(-0.25)) {
156 const T p2 = T(2) * (e_t<T> * z + T(1));
157 const T p = -sqrt(p2 > T(0) ? p2 : T(0));
158 w = T(-1) + p * (T(1) + p * (T(-1) / T(3) + p * (T(11) / T(72) - p * T(43) / T(540))));
159 if (p > T(-1e-5))
160 return w;
161 } else {
162 const T l1 = log(-z);
163 const T l2 = log(-l1);
164 w = l1 - l2 + l2 / l1;
165 }
166 const T lz = log(-z);
167 for (int i = 0; i < 32; ++i) {
168 const T g = w + log(-w) - lz;
169 const T g1 = T(1) + T(1) / w; // g'
170 const T g2 = T(-1) / (w * w); // g''
171 const T step = g / (g1 - g * g2 / (T(2) * g1));
172 const T before = w;
173 w -= step;
174 if (!(w < T(-1))) // never cross onto the principal branch
175 w = (before + T(-1)) / T(2);
176 if (abs(step) <= halley_tolerance_<T>() * abs(w))
177 break;
178 }
179 return w;
180}

182template <typename T>
183constexpr T softplus_c(T x)
184{
185 // Beyond digits * ln 2, e^-|x| is below half an ulp of the result: softplus is x above and e^x below.
186 constexpr T cut = T(std::numeric_limits<T>::digits) * T(0.693147180559945309417232121458176568L);
187 if (is_nan(x)) return x;
188 if (x > cut) return x;
189 if (x < -cut) return exp(x);
190 return x > T(0) ? x + log1p(exp(-x)) : log1p(exp(x));
191}

193template <typename T>
194constexpr T logistic_c(T x)
195{
196 if (x >= T(0))
197 return T(1) / (T(1) + exp(-x));
198 const T ex = exp(x); // x < 0 or NaN
199 return ex / (T(1) + ex);
200}

202// log Phi(x) as SciPy's log_ndtr: log1p(-Phi(-x)) for x >= 0, log(Phi(x)) down to -20, and below it the asymptotic
203// series of Abramowitz and Stegun 26.2.12 for the Mills ratio, Phi(x) = phi(x) / |x| * (1 + sum_n (-1)^n (2n-1)!! / x^2n),
204// taken in logarithms. At x = -20 its terms fall by (2n-1) / 400 each, so a double needs about ten.
205template <typename T>
206constexpr T log_normcdf_c(T x)
207{
208 if (is_nan(x)) return x;
209 if (x >= T(0)) return log1p(-normal_cdf(-x));
210 if (x > T(-20)) return log(normal_cdf(x));
211 if (is_inf(x)) return x;
212 const T x2 = x * x;
213 T term = T(1), sum = T(0);
214 for (int n = 1; n < 100; ++n) {
215 term *= -T(2 * n - 1) / x2;
216 sum += term;
217 if (abs(term) <= std::numeric_limits<T>::epsilon() / T(16))
218 break;
219 }
220 return -x2 / T(2) - log(-x) - limits<T>::half_log_2pi + log1p(sum);
221}

223template <typename T>
224constexpr T entr_c(T x)
225{
226 if (is_nan(x)) return x;
227 if (x < T(0)) return -inf_t<T>;
228 if (x == T(0)) return T(0);
229 return -x * log(x);

232template <typename T>
233constexpr T rel_entr_c(T x, T y)
234{
235 if (is_nan(x) || is_nan(y)) return nan_t<T>;
236 if (x > T(0) && y > T(0)) {
237 // log(x / y) is accurate when x is near y; once x / y overflows or underflows, the difference of logarithms
238 const T r = x / y;
239 if (!is_inf(r) && r >= std::numeric_limits<T>::min())
240 return x * log(r);
241 return x * (log(x) - log(y));
242 }
243 if (x == T(0) && y >= T(0)) return T(0);
244 return inf_t<T>;
245}

247template <typename T>
248constexpr T kl_div_c(T x, T y)
249{
250 if (is_nan(x) || is_nan(y)) return nan_t<T>;
251 if (x > T(0) && y > T(0)) {
252 if (is_inf(x) || is_inf(y))
253 return is_inf(x) && is_inf(y) ? nan_t<T> : inf_t<T>; // x log(x/y) - x + y grows without bound
254 return rel_entr_c(x, y) - x + y;
255 }
256 if (x == T(0) && y >= T(0)) return y;
257 return inf_t<T>;
258}

260template <typename T>
261constexpr T huber_c(T x, T m)
262{
263 if (is_nan(x) || is_nan(m) || m < T(0)) return nan_t<T>;
264 const T a = abs(x);
265 return a <= m ? x * x : T(2) * m * a - m * m;
266}
