153 settles_on_estimated == other.settles_on_estimated && settled_at == other.settled_at &&
154 available_at == other.available_at;
155 }

157
163 [[nodiscard]] constexpr capability_result<void> validate(const fact_dates& d) noexcept
164 {
165 if (d.known_at < d.trade_at)
166 return fail(capability_errc::invalid_argument, {}, "a fact is known before it happened");
167 if (d.settled_at && *d.settled_at < d.trade_at)
168 return fail(capability_errc::invalid_argument, {}, "a fact settled before it happened");
169 if (d.settles_on_estimated && !d.settles_on)
170 return fail(capability_errc::invalid_argument, {}, "an estimate without a settlement date");
171 return ok();
172 }
