368 // Every function applied to every value, functions outermost: Haskell's list <*>.
369 template <class MF, class MA, std::size_t NF = std::tuple_size_v<std::remove_cvref_t<MF>>,
370 std::size_t NA = std::tuple_size_v<std::remove_cvref_t<MA>>,
371 class R = std::remove_cvref_t<std::invoke_result_t<const typename std::remove_cvref_t<MF>::value_type&,
372 const typename std::remove_cvref_t<MA>::value_type&>>>
373 static constexpr std::array<R, NF * NA> ap(MF&& mf, MA&& ma)
374 {
375 return [&]<std::size_t... J>(std::index_sequence<J...>) {
376 return std::array<R, NF * NA>{{std::invoke(std::as_const(mf[J / NA]), std::as_const(ma[J % NA]))...}};
377 }(std::make_index_sequence<NF * NA>{});
378 }
379 };
380 template <class T, std::size_t N>

382 static constexpr bool defined = true;
383 // Every value through k, the arrays concatenated: Haskell's list >>= (concatMap).
384 template <class M, class K, class RA = std::remove_cvref_t<std::invoke_result_t<K&, const T&>>,
385 std::size_t NK = std::tuple_size_v<RA>, class R = typename RA::value_type>
386 static constexpr std::array<R, N * NK> bind(M&& m, K&& k)
387 {
