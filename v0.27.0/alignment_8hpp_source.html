71 template <linalg::scalar T>
72 struct transform_2d {
73 linalg::fixed_matrix<T, 2, 2> rotation = linalg::fixed_matrix<T, 2, 2>::identity();
74 linalg::fixed_vector<T, 2> translation{};
75 T scale = T(1);
76
78 [[nodiscard]] constexpr T angle() const noexcept(linalg::detail::nothrow_scalar<T>) { return static_cast<T>(math::atan2(rotation(1, 0), rotation(0, 0))); }
80 [[nodiscard]] constexpr linalg::fixed_vector<T, 2> apply(const linalg::fixed_vector<T, 2>& x) const noexcept(linalg::detail::nothrow_scalar<T>)
81 {
82 return rotation * x * scale + translation;
83 }
84
85 [[nodiscard]] constexpr transform_2d inverse() const noexcept(linalg::detail::nothrow_scalar<T>)
86 {
87 const auto rt = linalg::transpose(rotation);
88 return {rt, -(rt * translation) / scale, T(1) / scale};
89 }
90 };

115 template <typename T, std::size_t N>
116 struct centred {
117 linalg::fixed_vector<T, N> source_mean{};
118 linalg::fixed_vector<T, N> target_mean{};
119 T weight{};
120 T source_spread{}; // Σ w |s - s̄|²
121 };
122
123 template <typename T, std::size_t N>
124 constexpr centred<T, N> centre(std::span<const linalg::fixed_vector<T, N>> s, std::span<const linalg::fixed_vector<T, N>> d, std::span<const T> w) noexcept(linalg::detail::nothrow_scalar<T>)
125 {
126 centred<T, N> c{};
127 for (std::size_t i = 0; i < s.size(); ++i) {
128 const T wi = w.empty() ? T(1) : w[i];
129 c.weight += wi;
130 c.source_mean += s[i] * wi;
131 c.target_mean += d[i] * wi;
132 }
133 c.source_mean = c.source_mean / c.weight;
134 c.target_mean = c.target_mean / c.weight;
135 for (std::size_t i = 0; i < s.size(); ++i)
136 c.source_spread += (w.empty() ? T(1) : w[i]) * linalg::squared_norm(s[i] - c.source_mean);
137 return c;
138 }

140 constexpr bool lengths_fit(std::size_t need, std::size_t s, std::size_t d, std::size_t w) noexcept
141 {
142 return s >= need && s == d && (w == 0 || w == s);
143 }

154 template <linalg::scalar T>
155 [[nodiscard]] constexpr result<transform_2d<T>, alignment_error> align_2d(std::span<const linalg::fixed_vector<T, 2>> source, std::span<const linalg::fixed_vector<T, 2>> target,
156 std::span<const T> weights = {}, bool with_scale = false) noexcept(linalg::detail::nothrow_scalar<T>)
157 {
158 if (!detail::alignment::lengths_fit(2, source.size(), target.size(), weights.size()))
159 return err(alignment_error::too_few_points);
160 const auto c = detail::alignment::centre<T, 2>(source, target, weights);
161 if (!(c.source_spread > T(0)) || !(c.weight > T(0)))
162 return err(alignment_error::degenerate);
163 T a(0), b(0);
164 for (std::size_t i = 0; i < source.size(); ++i) {
165 const T wi = weights.empty() ? T(1) : weights[i];
166 const auto s = source[i] - c.source_mean;
167 const auto d = target[i] - c.target_mean;
168 a += wi * (s[0] * d[0] + s[1] * d[1]);
169 b += wi * (s[0] * d[1] - s[1] * d[0]);
170 }
171 const T h = static_cast<T>(math::sqrt(a * a + b * b));
172 if (!(h > T(0)))
173 return err(alignment_error::degenerate);
174 transform_2d<T> x{};
175 x.rotation = {{a / h, -b / h, b / h, a / h}};
176 x.scale = with_scale ? h / c.source_spread : T(1);
177 x.translation = c.target_mean - x.rotation * c.source_mean * x.scale;
178 return ok(x);
179 }

189 template <linalg::scalar T>
190 [[nodiscard]] constexpr result<transform_3d<T>, alignment_error> align_3d(std::span<const linalg::fixed_vector<T, 3>> source, std::span<const linalg::fixed_vector<T, 3>> target,
191 std::span<const T> weights = {}, bool with_scale = false) noexcept(linalg::detail::nothrow_scalar<T>)
192 {
193 if (!detail::alignment::lengths_fit(3, source.size(), target.size(), weights.size()))
194 return err(alignment_error::too_few_points);
195 const auto c = detail::alignment::centre<T, 3>(source, target, weights);
196 if (!(c.source_spread > T(0)) || !(c.weight > T(0)))
197 return err(alignment_error::degenerate);
198 linalg::fixed_matrix<T, 3, 3> m{}; // Σ w s' d'ᵀ
199 for (std::size_t i = 0; i < source.size(); ++i) {
200 const T wi = weights.empty() ? T(1) : weights[i];
201 m += linalg::outer(source[i] - c.source_mean, target[i] - c.target_mean) * wi;
202 }
203 const T sxx = m(0, 0), sxy = m(0, 1), sxz = m(0, 2), syx = m(1, 0), syy = m(1, 1), syz = m(1, 2), szx = m(2, 0), szy = m(2, 1), szz = m(2, 2);
204 const linalg::fixed_matrix<T, 4, 4> n{{sxx + syy + szz, syz - szy, szx - sxz, sxy - syx,
205 syz - szy, sxx - syy - szz, sxy + syx, szx + sxz,
206 szx - sxz, sxy + syx, -sxx + syy - szz, syz + szy,
207 sxy - syx, szx + sxz, syz + szy, -sxx - syy + szz}};
208 const auto e = linalg::symmetric_eigen(n);
209 if (!e)
210 return err(alignment_error::degenerate);
211 // A unique rotation needs the largest eigenvalue to stand clear of the next (collinear points tie them)
212 const T top = e->values[3], next = e->values[2];
213 const T scale = linalg::norm(n);
214 if (!(top - next > T(64) * std::numeric_limits<T>::epsilon() * scale))
215 return err(alignment_error::degenerate);
216 const T q0 = e->vectors(0, 3), qx = e->vectors(1, 3), qy = e->vectors(2, 3), qz = e->vectors(3, 3);
217 transform_3d<T> x{};
218 x.rotation = {{q0 * q0 + qx * qx - qy * qy - qz * qz, T(2) * (qx * qy - q0 * qz), T(2) * (qx * qz + q0 * qy),
219 T(2) * (qy * qx + q0 * qz), q0 * q0 - qx * qx + qy * qy - qz * qz, T(2) * (qy * qz - q0 * qx),
220 T(2) * (qz * qx - q0 * qy), T(2) * (qz * qy + q0 * qx), q0 * q0 - qx * qx - qy * qy + qz * qz}};
221 if (with_scale) {
222 T num(0);
223 for (std::size_t i = 0; i < source.size(); ++i) {
224 const T wi = weights.empty() ? T(1) : weights[i];
225 num += wi * linalg::dot(target[i] - c.target_mean, x.rotation * (source[i] - c.source_mean));
226 }
227 x.scale = num / c.source_spread;
228 }
229 x.translation = c.target_mean - x.rotation * c.source_mean * x.scale;
230 return ok(x);
231 }

243 template <typename X, linalg::scalar T, std::size_t N>
244 [[nodiscard]] constexpr T rms_error(const X& x, std::span<const linalg::fixed_vector<T, N>> source, std::span<const linalg::fixed_vector<T, N>> target,
245 std::span<const T> weights = {}) noexcept(linalg::detail::nothrow_scalar<T>)
246 {
247 T s(0), w(0);
248 for (std::size_t i = 0; i < source.size() && i < target.size(); ++i) {
249 const T wi = weights.empty() ? T(1) : weights[i];
250 s += wi * linalg::squared_norm(target[i] - x.apply(source[i]));
251 w += wi;
252 }
253 return w > T(0) ? static_cast<T>(math::sqrt(s / w)) : T(0);
254 }
