136 template <std::floating_point T>
137 struct compensated_sum {
138 T total{};
139 T compensation{};
140
143 constexpr void add(T x) noexcept
144 {
145 // Neumaier: the low part lost when the smaller of the two is added to the larger
146 const T t = total + x;
147 const T abs_total = total < 0 ? -total : total;
148 const T abs_x = x < 0 ? -x : x;
149 compensation += abs_total >= abs_x ? (total - t) + x : (x - t) + total;
150 total = t;
151 }
152
154 constexpr void merge(const compensated_sum& other) noexcept
155 {
156 add(other.total);
157 compensation += other.compensation;
158 }
159
161 [[nodiscard]] constexpr T value() const noexcept { return total + compensation; }
162 };

252 template <std::floating_point T>
253 struct variance {
254 std::uint64_t n = 0;
255 T m{};
256 T m2{};
257
260 constexpr void add(T x) noexcept
261 {
262 ++n;
263 const T delta = x - m;
264 m += delta / static_cast<T>(n);
265 m2 += delta * (x - m);
266 }
267
269 constexpr void merge(const variance& other) noexcept
270 {
271 if (other.n == 0)
272 return;
273 const std::uint64_t total = n + other.n;
274 const T delta = other.m - m;
275 m += delta * static_cast<T>(other.n) / static_cast<T>(total);
276 m2 += other.m2 + delta * delta * static_cast<T>(n) * static_cast<T>(other.n) / static_cast<T>(total);
277 n = total;
278 }
279
281 [[nodiscard]] constexpr T value() const noexcept { return n < 2 ? std::numeric_limits<T>::quiet_NaN() : m2 / static_cast<T>(n - 1); }
284 [[nodiscard]] constexpr T mean() const noexcept { return n == 0 ? std::numeric_limits<T>::quiet_NaN() : m; }
287 [[nodiscard]] constexpr T standard_deviation() const noexcept { return math::sqrt(value()); }
290 [[nodiscard]] constexpr std::uint64_t count() const noexcept { return n; }
293 [[nodiscard]] constexpr T population_variance() const noexcept { return n == 0 ? std::numeric_limits<T>::quiet_NaN() : m2 / static_cast<T>(n); }
294 };

299 template <std::floating_point T>
300 struct covariance {
301 std::uint64_t n = 0;
302 T mx{};
303 T my{};
304 T cxy{};
305 T m2x{};
306 T m2y{};
307
311 constexpr void add(T x, T y) noexcept
312 {
313 ++n;
314 const T dx = x - mx;
315 mx += dx / static_cast<T>(n);
316 const T dy = y - my;
317 my += dy / static_cast<T>(n);
318 cxy += dx * (y - my);
319 m2x += dx * (x - mx);
320 m2y += dy * (y - my);
321 }
322
325 template <typename P>
326 requires requires(const P& q) {
327 T(q.first);
328 T(q.second);
329 }
330 constexpr void add(const P& p) noexcept
331 {
332 add(static_cast<T>(p.first), static_cast<T>(p.second));
333 }
334
336 constexpr void merge(const covariance& other) noexcept
337 {
338 if (other.n == 0)
339 return;
340 const std::uint64_t total = n + other.n;
341 const T dx = other.mx - mx, dy = other.my - my;
342 const T w = static_cast<T>(n) * static_cast<T>(other.n) / static_cast<T>(total);
343 cxy += other.cxy + dx * dy * w;
344 m2x += other.m2x + dx * dx * w;
345 m2y += other.m2y + dy * dy * w;
346 mx += dx * static_cast<T>(other.n) / static_cast<T>(total);
347 my += dy * static_cast<T>(other.n) / static_cast<T>(total);
348 n = total;
349 }
350
352 [[nodiscard]] constexpr T value() const noexcept { return n < 2 ? std::numeric_limits<T>::quiet_NaN() : cxy / static_cast<T>(n - 1); }
355 [[nodiscard]] constexpr T correlation() const noexcept
356 {
357 return (n < 2 || m2x <= 0 || m2y <= 0) ? std::numeric_limits<T>::quiet_NaN() : cxy / math::sqrt(m2x * m2y);
358 }
359
361 [[nodiscard]] constexpr T beta() const noexcept { return (n < 2 || m2x <= 0) ? std::numeric_limits<T>::quiet_NaN() : cxy / m2x; }
364 [[nodiscard]] constexpr std::uint64_t count() const noexcept { return n; }
365 };

370 template <std::floating_point T>
371 struct ewma {
372 T alpha = T(0.06);
373 T m{};
374 T var{};
375 bool started = false;
376
379 constexpr void add(T x) noexcept
380 {
381 if (!started) {
382 m = x;
383 started = true;
384 return;
385 }
386 const T delta = x - m;
387 m += alpha * delta;
388 var = (1 - alpha) * (var + alpha * delta * delta);
389 }
390
392 [[nodiscard]] constexpr T value() const noexcept { return started ? m : std::numeric_limits<T>::quiet_NaN(); }
395 [[nodiscard]] constexpr T variance() const noexcept { return started ? var : std::numeric_limits<T>::quiet_NaN(); }
398 [[nodiscard]] constexpr T volatility() const noexcept { return math::sqrt(variance()); }

405 template <std::floating_point T, std::size_t Bins>
406 struct histogram {
407 static_assert(Bins >= 1, "snicholls::stats::histogram: at least one bin");
408 T lo = 0;
409 T hi = 1;
410 std::array<std::uint64_t, Bins> bins{};
411 std::uint64_t below = 0;
412 std::uint64_t above = 0;
413
416 constexpr void add(T x) noexcept
417 {
418 if (!(x >= lo)) {
419 ++below;
420 return;
421 }
422 if (x >= hi) {
423 ++above;
424 return;
425 }
426 auto i = static_cast<std::size_t>((x - lo) / (hi - lo) * static_cast<T>(Bins));
427 ++bins[i < Bins ? i : Bins - 1];
428 }
429
431 constexpr void merge(const histogram& other) noexcept
432 {
433 for (std::size_t i = 0; i < Bins; ++i)
434 bins[i] += other.bins[i];
435 below += other.below;
436 above += other.above;
437 }
438
440 [[nodiscard]] constexpr const std::array<std::uint64_t, Bins>& value() const noexcept { return bins; }
443 [[nodiscard]] constexpr std::uint64_t count() const noexcept
444 {
445 std::uint64_t total = below + above;
446 for (const std::uint64_t b : bins)
447 total += b;
448 return total;
449 }
450
453 [[nodiscard]] constexpr T edge(std::size_t i) const noexcept { return lo + (hi - lo) * static_cast<T>(i) / static_cast<T>(Bins); }
454 };

485 template <typename Acc, std::size_t B = 200>
486 struct bootstrapped {
487 static_assert(B >= 2, "snicholls::stats::bootstrapped: at least two replicates");
488 Acc estimate{};
489 std::array<Acc, B> replicates{};
490 wyrand rng{0x5eed};
491
494 constexpr void reseed(std::uint64_t seed) noexcept { rng = wyrand(seed); }
495
499 template <typename T>
500 constexpr void add(const T& x)
501 {
502 estimate.add(x);
503 for (std::size_t b = 0; b < B; b += 2) {
504 const std::uint64_t bits = rng();
505 for (std::uint32_t k = detail::poisson1_from(static_cast<std::uint32_t>(bits)); k > 0; --k)
506 replicates[b].add(x);
507 if (b + 1 < B)
508 for (std::uint32_t k = detail::poisson1_from(static_cast<std::uint32_t>(bits >> 32)); k > 0; --k)
509 replicates[b + 1].add(x);
510 }
511 }
512
515 constexpr void merge(const bootstrapped& other)
516 requires mergeable<Acc>
517 {
518 estimate.merge(other.estimate);
519 for (std::size_t b = 0; b < B; ++b)
520 replicates[b].merge(other.replicates[b]);
521 }
522
526 [[nodiscard]] bootstrap_distribution value() const
527 {
528 bootstrap_distribution out;
529 out.estimate = static_cast<double>(estimate.value());
530 out.replicates.reserve(B);
531 for (const Acc& r : replicates)
532 out.replicates.push_back(static_cast<double>(r.value()));
533 return out;
534 }
535 };
