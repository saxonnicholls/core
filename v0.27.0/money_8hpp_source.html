123 template <std::signed_integral I>
124 requires(sizeof(I) <= sizeof(std::uint64_t))
125 [[nodiscard]] constexpr std::uint64_t magnitude_of(I units) noexcept
126 {
127 const auto wide = static_cast<std::int64_t>(units);
128 return wide < 0 ? std::uint64_t{0} - static_cast<std::uint64_t>(wide) : static_cast<std::uint64_t>(wide);
129 }

131 // The rounding test on r and rest = d - r, of any unsigned type: r > rest is above half, r == rest a tie.
132 template <typename U>
133 [[nodiscard]] constexpr bool rounds_away_by(const U& r, const U& rest, bool odd, bool negative, rounding mode) noexcept
134 {
135 switch (mode) {
136 case rounding::toward_zero: return false;
137 case rounding::away_from_zero: return true;
138 case rounding::floor: return negative;
139 case rounding::ceiling: return !negative;
140 case rounding::half_up: return r >= rest;
141 case rounding::half_down: return r > rest;
142 case rounding::half_even: return r > rest || (r == rest && odd);
143 }
144 return false;
145 }
146
147 // Whether q + r/d, of the given sign, rounds up in magnitude (to q + 1) under `mode`. 0 <= r < d. When r and d
148 // fit in 64 bits the comparisons are on std::uint64_t, with the same answer.
149 [[nodiscard]] constexpr bool rounds_away(const uint256& q, const uint256& r, const uint256& d, bool negative,
150 rounding mode) noexcept
151 {
152 if (r.is_zero())
153 return false;
154 if ((r.limb(1) | r.limb(2) | r.limb(3) | d.limb(1) | d.limb(2) | d.limb(3)) == 0)
155 return rounds_away_by(r.limb(0), d.limb(0) - r.limb(0), q.bit(0), negative, mode);
156 return rounds_away_by(r, d - r, q.bit(0), negative, mode);
157 }

349 [[nodiscard]] constexpr std::optional<signed_magnitude> signed_add(bool an, const uint256& a, bool bn,
350 const uint256& b) noexcept
351 {
352 if (an == bn) {
353 const auto sum = uint256::checked_add(a, b);
354 if (!sum)
355 return std::nullopt;
356 return signed_magnitude{an, *sum};

450 [[nodiscard]] constexpr result<money, money_errc> parse_money(const asset& a, std::string_view text,
451 std::optional<rounding> mode = std::nullopt) noexcept
452 {
453 bool negative = false;
454 if (!text.empty() && (text.front() == '-' || text.front() == '+')) {
455 negative = text.front() == '-';
456 text.remove_prefix(1);
457 }
458 const std::size_t point = text.find('.');
459 const std::string_view whole = text.substr(0, point);
460 const std::string_view fraction = point == std::string_view::npos ? std::string_view{} : text.substr(point + 1);
461 if (whole.empty() && fraction.empty())
462 return err(money_errc::malformed);
463 const auto is_digits = [](std::string_view s) {
464 for (const char c : s)
465 if (c < '0' || c > '9')
466 return false;
467 return true;
468 };
469 if (!is_digits(whole) || !is_digits(fraction))
470 return err(money_errc::malformed);
471 const unsigned scale = a.scale();
472 uint256 units;
473 const auto push = [&units](char c) {
474 const auto t = uint256::checked_mul(units, 10);
475 if (!t)
476 return false;
477 const auto s = uint256::checked_add(*t, static_cast<std::uint64_t>(c - '0'));
478 if (!s)
479 return false;
480 units = *s;
481 return true;
482 };
483 for (const char c : whole)
484 if (!push(c))
485 return err(money_errc::overflow);
486 for (std::size_t i = 0; i < fraction.size() && i < scale; ++i)
487 if (!push(fraction[i]))
488 return err(money_errc::overflow);
489 for (std::size_t i = fraction.size(); i < scale; ++i)
490 if (!push('0'))
491 return err(money_errc::overflow);
492 if (fraction.size() > scale) {
493 const std::string_view rest = fraction.substr(scale);
494 bool any = false;
495 for (const char c : rest)
496 any = any || c != '0';
497 if (any) {
498 if (!mode)
499 return err(money_errc::too_many_digits);
500 // the rest as a fraction r / d of one minor unit, compared without forming 10^len when it is long
501 const std::size_t len = rest.size() > 77 ? 77 : rest.size();
502 uint256 r;
503 for (std::size_t i = 0; i < len; ++i)
504 r = r * 10 + static_cast<std::uint64_t>(rest[i] - '0');
505 const uint256 d = uint256::pow10(static_cast<unsigned>(len));
506 bool beyond = false; // nonzero digits past the 77th make a tie impossible and r slightly larger
507 for (std::size_t i = len; i < rest.size(); ++i)
508 beyond = beyond || rest[i] != '0';
509 bool away = detail::rounds_away(units, r, d, negative, *mode);
510 if (beyond && r == d - r && (*mode == rounding::half_down || *mode == rounding::half_even))
511 away = true; // above the tie, not on it
512 if (away) {
513 const auto s = uint256::checked_add(units, 1);
514 if (!s)
515 return err(money_errc::overflow);
516 units = *s;
517 }
518 }
519 }
520 return ok(money::from_units(a, units, negative));
521 }

529 [[nodiscard]] constexpr result<money, money_errc> from_decimal(const asset& a, const decimal& d,
530 std::optional<rounding> mode = std::nullopt) noexcept
531 {
532 const bool negative = d.is_negative();
533 const std::int64_t m = d.mantissa();
534 const uint256 magnitude = m < 0 ? uint256(0ull - static_cast<std::uint64_t>(m)) : uint256(static_cast<std::uint64_t>(m));
535 const int shift = d.exponent() + static_cast<int>(a.scale()); // units = magnitude * 10^shift
536 if (shift >= 0) {
537 const auto units = uint256::checked_mul(magnitude, uint256::pow10(static_cast<unsigned>(shift)));
538 if (!units)
539 return err(money_errc::overflow);
540 return ok(money::from_units(a, *units, negative));
541 }
542 const uint256 divisor = uint256::pow10(static_cast<unsigned>(-shift));
543 auto [q, r] = magnitude.divmod(divisor);
544 if (!r.is_zero()) {
545 if (!mode)
546 return err(money_errc::too_many_digits);
547 if (detail::rounds_away(q, r, divisor, negative, *mode))
548 q = q + 1;
549 }
550 return ok(money::from_units(a, q, negative));
551 }

556 [[nodiscard]] constexpr std::optional<decimal> to_decimal(const money& m) noexcept
557 {
558 uint256 units = m.magnitude();
559 int exponent = -static_cast<int>(m.asset_of().scale());
560 while (!units.is_zero()) { // trailing zeros belong in the exponent
561 const auto [q, r] = units.divmod(10);
562 if (!r.is_zero())
563 break;
564 units = q;
565 ++exponent;
566 }
567 const auto small = units.to_uint64();
568 if (!small || *small > 999'999'999'999'999'999ull)
569 return std::nullopt;
570 const auto mantissa = static_cast<std::int64_t>(*small);
571 return decimal::from_parts(m.is_negative() ? -mantissa : mantissa, units.is_zero() ? 0 : exponent);
572 }

586 constexpr std::to_chars_result to_chars(char* first, char* last, const money& m,
587 money_digits digits = money_digits::minimal) noexcept
588 {
589 // the magnitude's decimal digits, least significant first
590 std::array<char, 80> rev{};
591 std::size_t n = 0;
592 uint256 v = m.magnitude();
593 const uint256 chunk = uint256::pow10(19);
594 do {
595 const auto [q, r] = v.divmod(chunk);
596 std::uint64_t part = r.low64();
597 const bool last_chunk = q.is_zero();
598 for (int i = 0; i < 19 && (part != 0 || !last_chunk); ++i) {
599 rev[n++] = static_cast<char>('0' + part % 10);
600 part /= 10;
601 }
602 v = q;
603 } while (!v.is_zero());
604 if (n == 0)
605 rev[n++] = '0';
606 const std::size_t scale = m.asset_of().scale();
607 while (n <= scale) // at least one digit before the point
608 rev[n++] = '0';
609 std::size_t frac = scale;
610 if (digits == money_digits::minimal)
611 while (frac > 0 && rev[scale - frac] == '0')
612 --frac;
613 const std::size_t whole = n - scale;
614 // the text in a local buffer first, then copied with a loop bounded by both ends: GCC's -Wstringop-overflow
615 // cannot see that a size check alone keeps the writes inside a small destination
616 std::array<char, 82> text{};
617 std::size_t size = 0;
618 if (m.is_negative())
619 text[size++] = '-';
620 for (std::size_t i = 0; i < whole; ++i)
621 text[size++] = rev[n - 1 - i];
622 if (frac > 0) {
623 text[size++] = '.';
624 for (std::size_t i = 0; i < frac; ++i)
625 text[size++] = rev[scale - 1 - i];
626 }
627 if (static_cast<std::size_t>(last - first) < size)
628 return {last, std::errc::value_too_large};
629 char* out = first;
630 for (std::size_t i = 0; i < size && out != last; ++i)
631 *out++ = text[i];
632 return {out, std::errc{}};
633 }
