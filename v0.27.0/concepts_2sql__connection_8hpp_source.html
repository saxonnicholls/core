94 template <typename C>
95 concept sql_connection = requires(C& c, const C& cc, std::string_view sql, std::span<const sql_param> params,
96 sql_row_visitor visit, sql_row_filler fill, std::size_t n) {
97 requires sql_dialect<std::remove_cvref_t<decltype(cc.dialect())>>;
98 { c.prepare(sql) } -> std::same_as<sql_result<void>>;
99 { c.execute(sql, params) } -> std::same_as<sql_result<std::size_t>>;
100 { c.query(sql, params, visit) } -> std::same_as<sql_result<std::size_t>>;
101 { c.execute_batch(sql, n, n, fill) } -> std::same_as<sql_result<std::size_t>>;
102 { c.begin() } -> std::same_as<sql_result<void>>;
103 { c.commit() } -> std::same_as<sql_result<void>>;
104 { c.rollback() } -> std::same_as<sql_result<void>>;
105 { cc.in_transaction() } noexcept -> std::same_as<bool>;
106 { c.ping() } -> std::same_as<sql_result<void>>;
107 { cc.is_open() } noexcept -> std::same_as<bool>;
108 };

172 class sql_transaction {
173 public:
177 [[nodiscard]] static sql_result<sql_transaction> begin(C& c)
178 {
179 SNICHOLLS_TRY_VOID(c.begin());
180 return ok(sql_transaction(c));
181 }
182
183 sql_transaction(const sql_transaction&) = delete;
184 sql_transaction& operator=(const sql_transaction&) = delete;
187 sql_transaction(sql_transaction&& other) noexcept : c_(std::exchange(other.c_, nullptr)) {}
188 sql_transaction& operator=(sql_transaction&&) = delete;
189
191 ~sql_transaction()
192 {
193 if (c_ != nullptr && c_->in_transaction())
194 static_cast<void>(c_->rollback()); // a failed rollback leaves the connection to report itself broken
195 }

199 sql_result<void> commit()
200 {
201 C* c = std::exchange(c_, nullptr);
202 return c->commit();
203 }
204
207 sql_result<void> rollback()
208 {
209 C* c = std::exchange(c_, nullptr);
210 return c->rollback();
211 }

213 private:
214 explicit sql_transaction(C& c) noexcept : c_(&c) {}
215 C* c_;
216 };
