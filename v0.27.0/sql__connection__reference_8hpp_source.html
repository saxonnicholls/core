254 static bool tokenize(std::string_view s, std::vector<token>& out)
255 {
256 std::size_t i = 0;
257 while (i < s.size()) {
258 const char c = s[i];
259 if (c == ' ' || c == '\n' || c == '\t' || c == '\r') {
260 ++i;
261 } else if (c == '"') {
262 std::string id;
263 ++i;
264 for (;;) {
265 if (i >= s.size())
266 return false;
267 if (s[i] == '"') {
268 if (i + 1 < s.size() && s[i + 1] == '"') {
269 id += '"';
270 i += 2;
271 continue;
272 }
273 ++i;
274 break;
275 }
276 id += s[i++];
277 }
278 out.push_back({tk::ident, std::move(id), 0});
279 } else if (c == '?' || c == '$') {
280 std::size_t n = 0, j = i + 1;
281 while (j < s.size() && s[j] >= '0' && s[j] <= '9')
282 n = n * 10 + static_cast<std::size_t>(s[j++] - '0');
283 if (j == i + 1 || n == 0)
284 return false;
285 out.push_back({tk::param, {}, n});
286 i = j;
287 } else if (c >= '0' && c <= '9') {
288 std::size_t n = 0;
289 while (i < s.size() && s[i] >= '0' && s[i] <= '9')
290 n = n * 10 + static_cast<std::size_t>(s[i++] - '0');
291 out.push_back({tk::number, {}, n});
292 } else if (std::isalpha(static_cast<unsigned char>(c)) || c == '_') {
293 std::string w;
294 while (i < s.size() && (std::isalnum(static_cast<unsigned char>(s[i])) || s[i] == '_'))
295 w += static_cast<char>(std::toupper(static_cast<unsigned char>(s[i++])));
296 out.push_back({tk::word, std::move(w), 0});
297 } else if ((c == '>' || c == '<') && i + 1 < s.size() && s[i + 1] == '=') {
298 out.push_back({tk::punct, std::string{c, '='}, 0});
299 i += 2;
300 } else if (std::string_view("(),=<>*;.").find(c) != std::string_view::npos) {
301 out.push_back({tk::punct, std::string(1, c), 0});
302 ++i;
303 } else {
304 return false;
305 }
306 }
307 while (!out.empty() && out.back().kind == tk::punct && out.back().text == ";")
308 out.pop_back();
309 out.push_back({});
310 return true;
311 }

313 struct parser {
314 const std::vector<token>& t;
315 std::size_t i = 0;
316 statement s{};
317 bool good = true;
318
319 const token& peek() const { return t[i]; }
320 bool word(std::string_view w)
321 {
322 if (t[i].kind == tk::word && t[i].text == w) {
323 ++i;
324 return true;
325 }
326 return false;
327 }
328 bool punct(std::string_view p)
329 {
330 if (t[i].kind == tk::punct && t[i].text == p) {
331 ++i;
332 return true;
333 }
334 return false;
335 }
336 void need_word(std::string_view w) { good = good && word(w); }
337 void need_punct(std::string_view p) { good = good && punct(p); }
338 std::string ident()
339 {
340 if (t[i].kind != tk::ident) {
341 good = false;
342 return {};
343 }
344 return t[i++].text;
345 }
346 std::size_t param()
347 {
348 if (t[i].kind != tk::param) {
349 good = false;
350 return 0;
351 }
352 s.params = std::max(s.params, t[i].n);
353 return t[i++].n;
354 }
355 std::vector<std::string> ident_list()
356 {
357 std::vector<std::string> out;
358 need_punct("(");
359 do
360 out.push_back(ident());
361 while (good && punct(","));
362 need_punct(")");
363 return out;
364 }
365 void where_clause()
366 {
367 if (!word("WHERE"))
368 return;
369 do {
370 condition c;
371 c.column = ident();
372 if (punct("="))
373 c.rel = '=';
374 else if (punct(">="))
375 c.rel = 'g';
376 else if (punct("<"))
377 c.rel = '<';
378 else
379 good = false;
380 c.param = param();
381 s.where.push_back(std::move(c));
382 } while (good && word("AND"));
383 }
384 sql_kind type_kind()
385 {
386 std::string name;
387 while (t[i].kind == tk::word && t[i].text != "NOT")
388 name += (name.empty() ? "" : " ") + t[i++].text;
389 if (name == "INTEGER" || name == "BIGINT")
390 return sql_kind::int64;
391 if (name == "BOOLEAN")
392 return sql_kind::boolean;
393 if (name == "REAL" || name == "DOUBLE PRECISION")
394 return sql_kind::float64;
395 if (name == "TEXT")
396 return sql_kind::text;
397 if (name == "BLOB" || name == "BYTEA")
398 return sql_kind::blob;
399 good = false;
400 return sql_kind::null;
401 }

403 void parse()
404 {
405 if (word("CREATE")) {
406 s.kind = verb::create;
407 need_word("TABLE");
408 if (word("IF")) {
409 need_word("NOT");
410 need_word("EXISTS");
411 s.flag = true;
412 }
413 s.table = ident();
414 need_punct("(");
415 do {
416 if (word("PRIMARY")) {
417 need_word("KEY");
418 s.key = ident_list();
419 continue;
420 }
421 column c;
422 c.name = ident();
423 c.kind = type_kind();
424 c.nullable = true;
425 if (word("NOT")) {
426 need_word("NULL");
427 c.nullable = false;
428 }
429 s.columns.push_back(std::move(c));
430 } while (good && punct(","));
431 need_punct(")");
432 if (word("WITHOUT"))
433 need_word("ROWID");
434 } else if (word("DROP")) {
435 s.kind = verb::drop;
436 need_word("TABLE");
437 if (word("IF")) {
438 need_word("EXISTS");
439 s.flag = true;
440 }
441 s.table = ident();
442 } else if (word("INSERT")) {
443 s.kind = verb::insert;
444 need_word("INTO");
445 s.table = ident();
446 s.names = ident_list();
447 need_word("VALUES");
448 need_punct("(");
449 do
450 s.values.push_back(param());
451 while (good && punct(","));
452 need_punct(")");
453 good = good && s.values.size() == s.names.size();
454 if (word("ON")) {
455 need_word("CONFLICT");
456 static_cast<void>(ident_list());
457 need_word("DO");
458 if (word("NOTHING")) {
459 s.conflict = sql_conflict::ignore;
460 } else {
461 need_word("UPDATE");
462 need_word("SET");
463 s.conflict = sql_conflict::update;
464 do {
465 std::string target = ident();
466 need_punct("=");
467 need_word("EXCLUDED");
468 need_punct(".");
469 good = good && ident() == target;
470 s.updates.push_back(std::move(target));
471 } while (good && punct(","));
472 }
473 }
474 } else if (word("SELECT")) {
475 s.kind = verb::select;
476 if (word("COUNT")) {
477 s.kind = verb::count;
478 need_punct("(");
479 need_punct("*");
480 need_punct(")");
481 } else {
482 do
483 s.names.push_back(ident());
484 while (good && punct(","));
485 }
486 need_word("FROM");
487 s.table = ident();
488 where_clause();
489 if (word("ORDER")) {
490 need_word("BY");
491 do {
492 order o{ident(), false};
493 if (word("DESC"))
494 o.descending = true;
495 else
496 static_cast<void>(word("ASC"));
497 s.order_by.push_back(std::move(o));
498 } while (good && punct(","));
499 }
500 if (word("LIMIT")) {
501 if (peek().kind == tk::number)
502 s.limit = t[i++].n;
503 else
504 good = false;
505 }
506 } else if (word("DELETE")) {
507 s.kind = verb::erase;
508 need_word("FROM");
509 s.table = ident();
510 where_clause();
511 } else if (word("BEGIN")) {
512 s.kind = verb::begin;
513 static_cast<void>(word("IMMEDIATE"));
514 } else if (word("COMMIT")) {
515 s.kind = verb::commit;
516 } else if (word("ROLLBACK")) {
517 s.kind = verb::rollback;
518 } else {
519 good = false;
520 }
521 good = good && peek().kind == tk::end;
522 }
523 };

525 sql_result<const statement*> statement_for(std::string_view sql)
526 {
527 if (const auto it = cache_.find(sql); it != cache_.end())
528 return ok(&it->second);
529 std::vector<token> tokens;
530 if (!tokenize(sql, tokens))
531 return sql_fail(sql_errc::syntax_error, "the memory engine does not understand this", "42601");
532 parser p{tokens};
533 p.parse();
534 if (!p.good)
535 return sql_fail(sql_errc::syntax_error, "the memory engine does not understand this", "42601");
536 const auto [it, inserted] = cache_.emplace(std::string(sql), std::move(p.s));
537 static_cast<void>(inserted);
538 return ok(&it->second);
539 }

543 // -1, 0, 1: NULL first, then numbers, text, blobs.
544 static int compare(const sql_value& a, const sql_value& b)
545 {
546 const auto rank = [](const sql_value& v) {
547 switch (kind_of(v)) {
548 case sql_kind::null: return 0;
549 case sql_kind::boolean:
550 case sql_kind::int64:
551 case sql_kind::float64: return 1;
552 case sql_kind::text: return 2;
553 case sql_kind::blob: return 3;
554 }
555 return 4;
556 };
557 const int ra = rank(a), rb = rank(b);
558 if (ra != rb)
559 return ra < rb ? -1 : 1;
560 if (ra == 1) {
561 const auto as_int = [](const sql_value& v) { return kind_of(v) == sql_kind::boolean ? std::int64_t{std::get<1>(v)} : std::get<2>(v); };
562 if (kind_of(a) != sql_kind::float64 && kind_of(b) != sql_kind::float64) {
563 const std::int64_t x = as_int(a), y = as_int(b);
564 return x < y ? -1 : (x > y ? 1 : 0);
565 }
566 const double x = kind_of(a) == sql_kind::float64 ? std::get<3>(a) : static_cast<double>(as_int(a));
567 const double y = kind_of(b) == sql_kind::float64 ? std::get<3>(b) : static_cast<double>(as_int(b));
568 return x < y ? -1 : (x > y ? 1 : 0);
569 }
570 if (ra == 2)
571 return std::get<4>(a).compare(std::get<4>(b)) < 0 ? -1 : (std::get<4>(a) == std::get<4>(b) ? 0 : 1);
572 if (ra == 3) {
573 const auto& x = std::get<5>(a);
574 const auto& y = std::get<5>(b);
575 if (std::lexicographical_compare(x.begin(), x.end(), y.begin(), y.end()))
576 return -1;
577 return x == y ? 0 : 1;
578 }
579 return 0;
580 }

582 // The parameter as the column stores it, or the error.
583 static sql_result<sql_value> coerce(const column& c, const sql_param& p)
584 {
585 const sql_kind k = kind_of(p);
586 if (k == sql_kind::null) {
587 if (!c.nullable)
588 return sql_fail(sql_errc::constraint_violation, "NULL in a NOT NULL column", "23502");
589 return ok(sql_value{sql_null});
590 }
591 if (k == c.kind)
592 return ok(to_value(p));
593 if (c.kind == sql_kind::int64 && k == sql_kind::boolean)
594 return ok(sql_value{std::int64_t{std::get<1>(p) ? 1 : 0}});
595 if (c.kind == sql_kind::float64 && k == sql_kind::int64)
596 return ok(sql_value{static_cast<double>(std::get<2>(p))});
597 if (c.kind == sql_kind::boolean && k == sql_kind::int64 && (std::get<2>(p) == 0 || std::get<2>(p) == 1))
598 return ok(sql_value{std::get<2>(p) == 1});
599 return sql_fail(sql_errc::type_mismatch, "a value the column's type cannot hold", "42804");
600 }

610 sql_result<std::size_t> run(std::string_view sql, std::span<const sql_param> params, sql_row_visitor* visit)
611 {
612 if (!open_)
613 return sql_fail(sql_errc::connection_lost, "the connection is broken");
614 const auto found = statement_for(sql);
615 if (!found)
616 return err(found.unwrap_err());
617 const statement& s = **found;
618 if (params.size() != s.params)
619 return sql_fail(sql_errc::misuse, "the wrong number of parameters");
620 switch (s.kind) {
621 case verb::begin: SNICHOLLS_TRY_VOID(begin()); return ok(std::size_t{0});
622 case verb::commit: SNICHOLLS_TRY_VOID(commit()); return ok(std::size_t{0});
623 case verb::rollback: SNICHOLLS_TRY_VOID(rollback()); return ok(std::size_t{0});
624 case verb::create: return create(s);
625 case verb::drop: {
626 const auto it = tables_.find(s.table);
627 if (it == tables_.end()) {
628 if (s.flag)
629 return ok(std::size_t{0});
630 return sql_fail(sql_errc::undefined_object, "no such table", "42P01");
631 }
632 tables_.erase(it);
633 return ok(std::size_t{0});
634 }
635 default: break;
636 }
637 const auto it = tables_.find(s.table);
638 if (it == tables_.end())
639 return sql_fail(sql_errc::undefined_object, "no such table", "42P01");
640 table& t = it->second;
641 if (s.kind == verb::insert)
642 return insert(t, s, params);
643 // the rows the WHERE selects
644 std::vector<std::size_t> picked;
645 std::vector<std::pair<std::size_t, char>> conds;
646 for (const condition& c : s.where) {
647 const auto col = index_of(t, c.column);
648 if (!col)
649 return sql_fail(sql_errc::undefined_object, "no such column", "42703");
650 conds.emplace_back(*col, c.rel);
651 }
652 std::vector<sql_value> bound;
653 for (const condition& c : s.where)
654 bound.push_back(to_value(params[c.param - 1]));
655 for (std::size_t r = 0; r < t.rows.size(); ++r) {
656 bool keep = true;
657 for (std::size_t k = 0; k < conds.size() && keep; ++k) {
658 const sql_value& cell = t.rows[r][conds[k].first];
659 if (kind_of(cell) == sql_kind::null || kind_of(bound[k]) == sql_kind::null) {
660 keep = false;
661 break;
662 }
663 const int c = compare(cell, bound[k]);
664 keep = conds[k].second == '=' ? c == 0 : (conds[k].second == 'g' ? c >= 0 : c < 0);
665 }
666 if (keep)
667 picked.push_back(r);
668 }
669 if (s.kind == verb::erase) {
670 for (auto r = picked.rbegin(); r != picked.rend(); ++r)
671 t.rows.erase(t.rows.begin() + static_cast<std::ptrdiff_t>(*r));
672 return ok(picked.size());
673 }
674 if (s.kind == verb::count) {
675 const sql_param cell = static_cast<std::int64_t>(picked.size());
676 const std::string_view name = "count(*)";
677 if (visit != nullptr)
678 static_cast<void>((*visit)(sql_row(std::span<const std::string_view>(&name, 1), std::span<const sql_param>(&cell, 1))));
679 return ok(std::size_t{1});
680 }
681 // select
682 std::vector<std::pair<std::size_t, bool>> order_cols;
683 for (const order& o : s.order_by) {
684 const auto col = index_of(t, o.column);
685 if (!col)
686 return sql_fail(sql_errc::undefined_object, "no such column", "42703");
687 order_cols.emplace_back(*col, o.descending);
688 }
689 std::stable_sort(picked.begin(), picked.end(), [&](std::size_t a, std::size_t b) {
690 for (const auto& [col, desc] : order_cols) {
691 const int c = compare(t.rows[a][col], t.rows[b][col]);
692 if (c != 0)
693 return desc ? c > 0 : c < 0;
694 }
695 return false;
696 });
697 if (s.limit != 0 && picked.size() > s.limit)
698 picked.resize(s.limit);
699 std::vector<std::size_t> cols;
700 std::vector<std::string_view> names;
701 for (const std::string& n : s.names) {
702 const auto col = index_of(t, n);
703 if (!col)
704 return sql_fail(sql_errc::undefined_object, "no such column", "42703");
705 cols.push_back(*col);
706 names.push_back(t.columns[*col].name);
707 }
708 std::size_t seen = 0;
709 std::vector<sql_param> cells(cols.size());
710 for (const std::size_t r : picked) {
711 for (std::size_t k = 0; k < cols.size(); ++k)
712 cells[k] = view_of(t.rows[r][cols[k]]);
713 ++seen;
714 if (visit != nullptr && !(*visit)(sql_row(names, cells)))
715 break;
716 }
717 return ok(seen);
718 }

720 sql_result<std::size_t> create(const statement& s)
721 {
722 if (tables_.find(s.table) != tables_.end()) {
723 if (s.flag)
724 return ok(std::size_t{0});
725 return sql_fail(sql_errc::misuse, "the table exists", "42P07");
726 }
727 table t;
728 t.columns = s.columns;
729 for (const std::string& k : s.key) {
730 const auto col = index_of(t, k);
731 if (!col)
732 return sql_fail(sql_errc::undefined_object, "no such column", "42703");
733 t.key.push_back(*col);
734 t.columns[*col].nullable = false;
735 }
736 tables_.emplace(s.table, std::move(t));
737 return ok(std::size_t{0});

740 static sql_result<std::size_t> insert(table& t, const statement& s, std::span<const sql_param> params)
741 {
742 std::vector<sql_value> row(t.columns.size(), sql_value{sql_null});
743 std::vector<bool> given(t.columns.size(), false);
744 for (std::size_t k = 0; k < s.names.size(); ++k) {
745 const auto col = index_of(t, s.names[k]);
746 if (!col)
747 return sql_fail(sql_errc::undefined_object, "no such column", "42703");

749 if (!v)

753 }
754 for (std::size_t c = 0; c < t.columns.size(); ++c)
755 if (!given[c] && !t.columns[c].nullable)
756 return sql_fail(sql_errc::constraint_violation, "NULL in a NOT NULL column", "23502");
757 if (!t.key.empty())
758 for (auto& existing : t.rows) {

760 for (const std::size_t k : t.key)
761 same = same && compare(existing[k], row[k]) == 0;
762 if (!same)
763 continue;
764 if (s.conflict == sql_conflict::fail)

766 if (s.conflict == sql_conflict::ignore)
767 return ok(std::size_t{0});
768 for (const std::string& u : s.updates)
769 if (const auto col = index_of(t, u))
770 existing[*col] = row[*col];
771 return ok(std::size_t{1});
772 }
773 t.rows.push_back(std::move(row));
774 return ok(std::size_t{1});
