92 signature_verifier<std::remove_cvref_t<decltype(std::declval<Probe&>().verifier())>> &&

104 template <typename Probe>
105 requires detail::signature_verifier_probe<Probe>
106 [[nodiscard]] capability_conformance_report run_signature_verifier_conformance()
107 {
108 capability_conformance_report report;
109 using detail::signature_kit_message;
110
111 detail::capability_check<Probe>(report, "a valid signature verifies", [](Probe& p, const auto& expect) {
112 const signed_sample s = p.sign(signature_kit_message());
113 expect(p.verifier().verify(s.request(signature_kit_message())).is_ok(), "the probe's signature verifies");
114 });
115
116 detail::capability_check<Probe>(report, "a tampered message is signature_invalid", [](Probe& p, const auto& expect) {
117 const signed_sample s = p.sign(signature_kit_message());
118 std::vector<std::byte> tampered(signature_kit_message().begin(), signature_kit_message().end());
119 tampered.back() ^= std::byte{1};
120 const auto r = p.verifier().verify(s.request(tampered));
121 expect(r.is_err() && r.unwrap_err().code == capability_errc::signature_invalid, "one flipped bit is signature_invalid");
122 });
123
124 detail::capability_check<Probe>(report, "a tampered signature is signature_invalid", [](Probe& p, const auto& expect) {
125 signed_sample s = p.sign(signature_kit_message());
126 expect(!s.signature.empty(), "the signature has bytes");
127 if (s.signature.empty())
128 return;
129 s.signature.front() ^= std::byte{0x80};
130 const auto r = p.verifier().verify(s.request(signature_kit_message()));
131 expect(r.is_err() && r.unwrap_err().code == capability_errc::signature_invalid, "one flipped bit is signature_invalid");
132 });
133
134 detail::capability_check<Probe>(report, "a truncated signature never verifies", [](Probe& p, const auto& expect) {
135 signed_sample s = p.sign(signature_kit_message());
136 if (!s.signature.empty())
137 s.signature.pop_back();
138 expect(p.verifier().verify(s.request(signature_kit_message())).is_err(), "one byte short does not verify");
139 s.signature.clear();
140 expect(p.verifier().verify(s.request(signature_kit_message())).is_err(), "an empty signature does not verify");
141 });
142
143 detail::capability_check<Probe>(report, "an algorithm it cannot check is unsupported", [](Probe& p, const auto& expect) {
144 signed_sample s = p.sign(signature_kit_message());
145 s.algorithm = p.unsupported();
146 const auto r = p.verifier().verify(s.request(signature_kit_message()));
147 expect(r.is_err() && r.unwrap_err().code == capability_errc::unsupported, "the algorithm is unsupported, never ok");
148 });
149
150 detail::capability_check<Probe>(report, "a provider failure never leaks its response", [](Probe& p, const auto& expect) {
151 const signed_sample s = p.sign(signature_kit_message());
152 p.provider_fails(capability_errc::unavailable, capability_provider_response);
153 const auto r = p.verifier().verify(s.request(signature_kit_message()));
154 expect(r.is_err() && r.unwrap_err().code == capability_errc::unavailable, "the provider's failure is returned");
155 if (r.is_err())
156 expect(!capability_error_leaks(r.unwrap_err(), capability_redaction_marker),
157 "neither provider_code nor message carries the provider's response");
158 });
159
160 return report;
161 }
