110 // An array of N elements, element J being make(J); the elements need not be default-constructible.
111 template <std::size_t N, class Make>
112 constexpr auto generate(Make&& make)
113 {
114 return [&]<std::size_t... J>(std::index_sequence<J...>) {
115 using T = std::remove_cvref_t<decltype(make(std::size_t{0}))>;
116 return std::array<T, N>{{make(J)...}};
117 }(std::make_index_sequence<N>{});
118 }

276 struct group_by_fn {
277 template <class Eq, std_array A, std::size_t N = size_v<A>>
278 constexpr vec<vec<elem_t<A>, N>, N> operator()(Eq&& eq, const A& xs) const
279 {
280 vec<vec<elem_t<A>, N>, N> out;
281 for (std::size_t i = 0; i < N;) {
282 vec<elem_t<A>, N> g;
283 g.push_back(xs[i]);
284 std::size_t j = i + 1;
285 for (; j < N && std::invoke(eq, xs[i], xs[j]); ++j) g.push_back(xs[j]);
286 out.push_back(g);
287 i = j;
288 }
289 return out;
290 }

352 struct subsequences_fn {
353 template <std_array A, std::size_t N = size_v<A>>
354 requires(N <= 20)
355 constexpr auto operator()(const A& xs) const
356 {
357 std::array<vec<elem_t<A>, N>, (std::size_t{1} << N)> out{};
358 // Haskell's order: by the binary counter, lowest element varying fastest.
359 for (std::size_t mask = 0; mask < out.size(); ++mask)
360 for (std::size_t i = 0; i < N; ++i)
361 if (mask >> i & 1U) out[mask].push_back(xs[i]);
362 return out;
363 }
