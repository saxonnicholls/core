135 struct replace_repeated_fn {
136 template <class T, class R>
137 constexpr fixed_point_result<T> operator()(T x, R&& r, std::size_t limit) const
138 {
139 fixed_point_result<T> out{std::move(x), 0, false};
140 for (; out.iterations < limit; ++out.iterations) {
141 std::optional<T> y = std::invoke(r, std::as_const(out.value));
142 if (!y.has_value()) break;
143 out.value = std::move(*y);
144 }
145 out.converged = out.iterations < limit || !std::invoke(r, std::as_const(out.value)).has_value();
146 return out;
147 }
148 };

166 struct rewrite_up_fn {
167 template <class C, class R, class T>
168 constexpr T operator()(const C& children, const R& r, const T& x) const
169 {
170 T y = std::invoke(children, [&](const T& c) -> T { return (*this)(children, r, c); }, x);
171 std::optional<T> z = std::invoke(r, std::as_const(y));
172 return z.has_value() ? (*this)(children, r, *z) : y;
173 }
