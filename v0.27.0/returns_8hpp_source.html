129 // A money amount as signed minor units, when it fits in 64 bits.
130 constexpr std::optional<std::int64_t> minor_units(const money& m) noexcept
131 {
132 const auto u = m.magnitude().to_uint64();
133 if (!u || *u > static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()))
134 return std::nullopt;
135 const auto v = static_cast<std::int64_t>(*u);
136 return m.is_negative() ? -v : v;
137 }

139 // A money amount as a double of minor units (the asset's scale cancels in a rate).
140 constexpr double minor_units_double(const money& m) noexcept
141 {
142 double v = 0;
143 for (std::size_t i = 4; i-- > 0;)
144 v = v * 18446744073709551616.0 + static_cast<double>(m.magnitude().limb(i));
145 return m.is_negative() ? -v : v;

151 // The rate search's ends: 1 / (1 + r) and 1 + r from this up to 1.
152 inline constexpr double rate_floor = 1e-6;
153 inline constexpr int rate_steps = 1024;
154
155 // The net present value's sign at x = 1 / (1 + r) in (0, 1] (r >= 0), as sum c x^t, or at y = 1 + r in (0, 1]
156 // (r <= 0), as y^T times the value, sum c y^(T - t): both bounded, with the value's sign.
157 template <class Flows>
158 constexpr double npv_scaled(const Flows& f, bool below_zero, double v) noexcept
159 {
160 double s = 0;
161 for (std::size_t i = 0; i < f.size(); ++i)
162 s += f.amount(i) * math::pow(v, below_zero ? f.span() - f.time(i) : f.time(i));
163 return s;
164 }

166 constexpr double to_rate(bool below_zero, double v) noexcept { return below_zero ? v - 1 : 1 / v - 1; }

168 template <class Flows>
169 constexpr double close(const Flows& f, bool below_zero, double lo, double hi) noexcept
170 {
171 const auto g = [&](double v) noexcept { return npv_scaled(f, below_zero, v); };
172 return to_rate(below_zero, math::brent(g, lo, hi).x);
173 }

175 // Every rate in range, ascending; the count found (written up to out.size()).
176 template <class Flows>
177 constexpr std::size_t all_rates(const Flows& f, std::span<double> out) noexcept
178 {
179 std::size_t found = 0;
180 const auto keep = [&](double r) noexcept { // the smallest out.size() rates, ascending
181 std::size_t j = found < out.size() ? found : out.size();
182 if (j == out.size() && (j == 0 || out[j - 1] <= r)) {
183 ++found;
184 return;
185 }
186 if (j == out.size())
187 --j; // the largest kept makes way
188 for (; j > 0 && out[j - 1] > r; --j)
189 out[j] = out[j - 1];
190 out[j] = r;
191 ++found;
192 };
193 for (const bool below_zero : {true, false}) {
194 double prev_v = rate_floor, prev_g = npv_scaled(f, below_zero, rate_floor);
195 if (prev_g == 0)
196 keep(to_rate(below_zero, prev_v));
197 for (int k = 1; k <= rate_steps; ++k) {
198 const double v = k == rate_steps ? 1.0 : rate_floor + (1 - rate_floor) * k / rate_steps;
199 const double g = npv_scaled(f, below_zero, v);
200 if (g == 0) {
201 if (!(below_zero && k == rate_steps)) // r = 0 is counted once, from above
202 keep(to_rate(below_zero, v));
203 } else if (prev_g != 0 && (prev_g < 0) != (g < 0)) {
204 keep(close(f, below_zero, prev_v, v));
205 }
206 prev_v = v;
207 prev_g = g;
208 }
209 }
210 return found;
211 }

213 // The one rate, or why there is not one.
214 template <class Flows>
215 constexpr result<double, returns_errc> one_rate(const Flows& f) noexcept
216 {
217 int changes = 0, last = 0;
218 for (std::size_t i = 0; i < f.size(); ++i) {
219 const double a = f.amount(i);
220 const int s = a > 0 ? 1 : a < 0 ? -1 : 0;
221 if (s != 0 && last != 0 && s != last)
222 ++changes;
223 if (s != 0)
224 last = s;
225 }
226 if (changes == 0)
227 return err(returns_errc::no_root);
228 if (changes == 1) { // exactly one rate in (-1, inf): find which side of zero, and close on it
229 const double at_zero = npv_scaled(f, false, 1);
230 if (at_zero == 0)
231 return ok(0.0);
232 for (const bool below_zero : {false, true}) {
233 const double edge = npv_scaled(f, below_zero, rate_floor);
234 if (edge != 0 && (edge < 0) != (at_zero < 0))
235 return ok(close(f, below_zero, rate_floor, 1.0));
236 if (edge == 0)
237 return ok(to_rate(below_zero, rate_floor));
238 }
239 return err(returns_errc::no_root);
240 }
241 double rates[2]{};
242 const std::size_t n = all_rates(f, rates);
243 if (n == 0)
244 return err(returns_errc::no_root);
245 if (n > 1)
246 return err(returns_errc::multiple_roots);
247 return ok(rates[0]);
248 }

287 // Flows within [start, end], in order, all of one asset.
288 constexpr std::optional<returns_errc> check_period(const money& v0, const money& v1, std::span<const cash_flow> flows,
289 std::chrono::sys_days start, std::chrono::sys_days end) noexcept
290 {
291 if (end <= start)
292 return returns_errc::invalid_dates;
293 if (v0.asset_of() != v1.asset_of())
294 return returns_errc::asset_mismatch;
295 std::chrono::sys_days previous = start;
296 for (const cash_flow& f : flows) {
297 if (f.amount.asset_of() != v0.asset_of())
298 return returns_errc::asset_mismatch;
299 if (f.on < previous || f.on > end)
300 return returns_errc::invalid_dates;
301 previous = f.on;
302 }
303 return std::nullopt;
304 }

311 [[nodiscard]] constexpr result<ratio, returns_errc> period_return(const valuation& v) noexcept
312 {
313 if (v.flow.asset_of() != v.start.asset_of() || v.end.asset_of() != v.start.asset_of())
314 return err(returns_errc::asset_mismatch);
315 const auto s = detail::minor_units(v.start), f = detail::minor_units(v.flow), e = detail::minor_units(v.end);
316 const auto capital = s && f ? detail::sum(*s, *f) : std::nullopt;
317 const auto gain = capital && e ? checked_sub(*e, *capital) : std::nullopt;
318 if (!gain)
319 return err(returns_errc::out_of_range);
320 if (*capital <= 0)
321 return err(returns_errc::zero_capital);
322 return ok(ratio{*gain, *capital});
323 }

330 [[nodiscard]] constexpr result<ratio, returns_errc> time_weighted_return(std::span<const valuation> periods) noexcept
331 {
332 if (periods.empty())
333 return err(returns_errc::empty);
334 ratio growth{1, 1};
335 for (const valuation& v : periods) {
336 if (v.start.asset_of() != periods.front().start.asset_of())
337 return err(returns_errc::asset_mismatch);
338 const auto r = period_return(v);
339 if (!r)
340 return err(r.unwrap_err_unchecked());
341 const auto step = add(*r, ratio{1, 1});
342 const auto chained = step ? multiply(growth, *step) : std::nullopt;
343 if (!chained)
344 return err(returns_errc::out_of_range);
345 growth = *chained;
346 }
347 const auto twr = subtract(growth, ratio{1, 1});
348 if (!twr)
349 return err(returns_errc::out_of_range);
350 return ok(*twr);
351 }

357 [[nodiscard]] constexpr double time_weighted_return(std::span<const double> period_returns) noexcept
358 {
359 double growth = 1;
360 for (const double r : period_returns)
361 growth *= 1 + r;

374 [[nodiscard]] constexpr result<ratio, returns_errc> modified_dietz(const money& start_value, const money& end_value,
375 std::span<const cash_flow> flows,
376 std::chrono::sys_days start,
377 std::chrono::sys_days end) noexcept
378 {
379 if (const auto bad = detail::check_period(start_value, end_value, flows, start, end))
380 return err(*bad);
381 const std::int64_t days = (end - start).count();
382 const auto v0 = detail::minor_units(start_value), v1 = detail::minor_units(end_value);
383 if (!v0 || !v1)
384 return err(returns_errc::out_of_range);
385 auto gain = checked_sub(*v1, *v0); // V1 - V0 - sum F
386 auto capital = detail::mul(*v0, days); // D V0 + sum (D - d) F
387 for (const cash_flow& f : flows) {
388 const auto a = detail::minor_units(f.amount);
389 const auto weighted = a ? detail::mul(*a, days - (f.on - start).count()) : std::nullopt;
390 gain = gain && a ? checked_sub(*gain, *a) : std::nullopt;
391 capital = capital && weighted ? detail::sum(*capital, *weighted) : std::nullopt;
392 }
393 const auto scaled = gain ? detail::mul(*gain, days) : std::nullopt;
394 if (!scaled || !capital)
395 return err(returns_errc::out_of_range);
396 if (*capital <= 0)
397 return err(returns_errc::zero_capital);
398 return ok(ratio{*scaled, *capital});
399 }
