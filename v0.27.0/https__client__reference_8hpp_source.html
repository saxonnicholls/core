119 [[nodiscard]] static recorded_exchange record(const https_request& r, const https_response& response)
120 {
121 recorded_exchange x;
122 x.method = r.method;
123 x.url = r.url;
124 for (const auto& h : r.headers)
125 if (!ascii_iequals(h.name, "Authorization"))
126 x.request_headers.push_back(h);
127 if (!r.idempotency_key.empty())
128 x.request_headers.push_back({"Idempotency-Key", std::string(r.idempotency_key.view())});
129 x.request_body = r.sensitive_body ? std::string(redacted) : r.body;
130 x.had_authorization = !r.authorization.reveal().empty();
131 x.response.status = response.status;
132 x.response.body = response.body;
133 for (const auto& h : response.headers)
134 if (!ascii_iequals(h.name, "Set-Cookie"))
135 x.response.headers.push_back(h);
136 return x;
137 }

162 [[nodiscard]] static capability_result<cassette> load(std::string_view text)
163 {
164 constexpr capability_message malformed = "not a cassette";
165 cassette c;
166 std::size_t at = 0;
167 const auto line = [&](std::string_view& out) {
168 const auto nl = text.find('\n', at);
169 if (at >= text.size() || nl == std::string_view::npos)
170 return false;
171 out = text.substr(at, nl - at);
172 at = nl + 1;
173 return true;
174 };
175 const auto bytes = [&](std::string_view count, std::string& out) {
176 std::size_t n = 0;
177 const auto [end, ec] = std::from_chars(count.data(), count.data() + count.size(), n);
178 if (ec != std::errc{} || end != count.data() + count.size() || n > text.size() - at ||
179 text.size() - at - n < 1 || text[at + n] != '\n')
180 return false;
181 out.assign(text.substr(at, n));
182 at += n + 1;
183 return true;
184 };
185 const auto header = [](std::string_view l, std::vector<http_header>& out) {
186 const auto colon = l.find(": ", 4);
187 if (colon == std::string_view::npos)
188 return false;
189 out.push_back({std::string(l.substr(4, colon - 4)), std::string(l.substr(colon + 2))});
190 return true;
191 };
192 std::string_view l;
193 while (line(l)) {
194 recorded_exchange x;
195 const auto space = l.find(' ', 2);
196 const auto method = l.starts_with("> ") && space != std::string_view::npos
197 ? parse_http_method(l.substr(2, space - 2)) : std::nullopt;
198 if (!method)
199 return fail(capability_errc::invalid_argument, {}, malformed);
200 x.method = *method;
201 x.url = std::string(l.substr(space + 1));
202 bool have_body = false;
203 while (!have_body && line(l)) {
204 if (l == "> a")
205 x.had_authorization = true;
206 else if (!(l.starts_with("> h ") && header(l, x.request_headers)) &&
207 !(l.starts_with("> b ") && (have_body = bytes(l.substr(4), x.request_body))))
208 return fail(capability_errc::invalid_argument, {}, malformed);
209 }
210 if (!have_body || !line(l) || !l.starts_with("< "))
211 return fail(capability_errc::invalid_argument, {}, malformed);
212 const auto status = l.substr(2);
213 const auto [end, ec] = std::from_chars(status.data(), status.data() + status.size(), x.response.status);
214 if (ec != std::errc{} || end != status.data() + status.size())
215 return fail(capability_errc::invalid_argument, {}, malformed);
216 bool have_response_body = false;
217 while (!have_response_body && line(l))
218 if (!(l.starts_with("< h ") && header(l, x.response.headers)) &&
219 !(l.starts_with("< b ") && (have_response_body = bytes(l.substr(4), x.response.body))))
220 return fail(capability_errc::invalid_argument, {}, malformed);
221 if (!have_response_body)
222 return fail(capability_errc::invalid_argument, {}, malformed);
223 c.exchanges.push_back(std::move(x));
224 }
225 return ok(std::move(c));
226 }

229 friend bool operator==(const cassette&, const cassette&) = default;

267 template <typename H>
268 requires http_handler<H>
269 memory_https_client& serve(std::string_view host, H& h)
270 {
271 return route(host, [p = &h](const https_request& r) -> capability_result<https_response> {
272 const auto url = parse_https_url(r.url);
273 const std::string_view target = url->target.empty() ? std::string_view{"/"} : url->target;
274 const auto q = target.find('?');
275 http_request in;
276 in.method = r.method;
277 in.path = target.substr(0, q);
278 if (in.path.empty())
279 in.path = "/";
280 in.query = q == std::string_view::npos ? std::string{} : std::string(target.substr(q + 1));
281 in.headers = r.headers;
282 if (!r.authorization.reveal().empty())
283 in.headers.push_back({"Authorization", r.authorization.reveal()});
284 if (!r.idempotency_key.empty())
285 in.headers.push_back({"Idempotency-Key", std::string(r.idempotency_key.view())});
286 in.body = r.body;
287 http_response out = p->handle(in);
288 return ok(https_response{out.status, std::move(out.headers), std::move(out.body)});
289 });

296 capability_result<https_response> send(const https_request& r)
297 {
298 if (injected_) {
299 const capability_error e = *injected_;
300 injected_.reset();
301 ++failures_;
302 return err(e);
303 }
304 const auto url = parse_https_url(r.url);
305 if (url.is_err())
306 return err(url.unwrap_err());
307 for (const auto& h : r.headers)
308 if (!is_sendable(h))
309 return fail(capability_errc::invalid_argument, {}, "a header that would split the request");
310 if (!is_sendable({"Authorization", r.authorization.reveal()}) ||
311 !is_sendable({"Idempotency-Key", std::string(r.idempotency_key.view())}))
312 return fail(capability_errc::invalid_argument, {}, "a header that would split the request");
313 const handler* answer = fallback_ ? &fallback_ : nullptr;
314 for (const auto& route : routes_)
315 if (ascii_iequals(route.host, url->host))
316 answer = &route.answer;
317 if (!answer) {
318 ++failures_;
319 return fail(capability_errc::unavailable, {}, "no route to the host");
320 }
321 auto response = (*answer)(r);
322 if (response.is_ok())
323 seen_.exchanges.push_back(cassette::record(r, *response));
324 else
325 ++failures_;
326 return response;
327 }

402 capability_result<https_response> send(const https_request& r)
403 {
404 for (std::size_t i = 0; i < tape_.exchanges.size(); ++i) {
405 const auto& x = tape_.exchanges[i];
406 if (used_[i] || x.method != r.method || x.url != r.url || x.idempotency_key() != r.idempotency_key.view())
407 continue;
408 if (x.request_body != cassette::redacted && x.request_body != r.body)
409 continue;
410 used_[i] = true;
411 return ok(x.response);
412 }
413 return fail(capability_errc::not_found, {}, "no such exchange recorded");
414 }
