122 // Calls f with the kind as a constant, so a row's loop is compiled once per kind with the switch folded away
123 template <typename F>
124 constexpr void dispatch(glu_kind k, F&& f)
125 {
126 using K = glu_kind;
127 switch (k) {
128 case K::reglu: f(std::integral_constant<K, K::reglu>{}); return;
129 case K::geglu: f(std::integral_constant<K, K::geglu>{}); return;

146template <math::math_number T>
147constexpr T glu(glu_kind kind, T g, T u, const glu_parameters<T>& p = {})
148{
149 switch (kind) {
150 case glu_kind::reglu: return relu(g) * u;
151 case glu_kind::geglu: return gelu_tanh(g) * u;
152 case glu_kind::geglu_erf: return gelu(g) * u;
153 case glu_kind::geglu_quick: return gelu_quick(g) * u;
154 case glu_kind::swiglu: return silu(g) * u;
155 case glu_kind::swiglu_oai: {
156 const T x = detail::glu::min(g, p.limit), y = detail::glu::clamp(u, -p.limit, p.limit);
157 const T out = x * sigmoid(p.alpha * x);
158 return out * (y + T(1));
159 }
160 case glu_kind::swiglu_clamp: {
161 const T x = detail::glu::min(g, p.limit), y = detail::glu::clamp(u, -p.limit, p.limit);
162 return silu(x) * y;
163 }
164 }
165 return T(0);
166}

176template <math::math_number T>
177constexpr glu_gradient<T> glu_backward(glu_kind kind, T g, T u, T dh, const glu_parameters<T>& p = {})
178{
179 switch (kind) {
180 case glu_kind::reglu: return {dh * u * relu_derivative(g), dh * relu(g)};
181 case glu_kind::geglu: return {dh * u * gelu_tanh_derivative(g), dh * gelu_tanh(g)};
182 case glu_kind::geglu_erf: return {dh * u * gelu_derivative(g), dh * gelu(g)};
183 case glu_kind::geglu_quick: return {dh * u * gelu_quick_derivative(g), dh * gelu_quick(g)};
184 case glu_kind::swiglu: return {dh * u * silu_derivative(g), dh * silu(g)};
185 case glu_kind::swiglu_oai: {
186 const T x = detail::glu::min(g, p.limit), y = detail::glu::clamp(u, -p.limit, p.limit);
187 const T dx = g <= p.limit ? T(1) : T(0), dy = detail::glu::inside(u, -p.limit, p.limit);
188 const auto [s, one_minus_s] = detail::activations::sigmoid_both(p.alpha * x);
189 const T dact = s + p.alpha * x * s * one_minus_s; // d/dx (x sigmoid(α x))
190 return {dh * (y + T(1)) * dact * dx, dh * x * s * dy};
191 }
192 case glu_kind::swiglu_clamp: {
193 const T x = detail::glu::min(g, p.limit), y = detail::glu::clamp(u, -p.limit, p.limit);
194 const T dx = g <= p.limit ? T(1) : T(0), dy = detail::glu::inside(u, -p.limit, p.limit);
195 return {dh * y * silu_derivative(x) * dx, dh * silu(x) * dy};
196 }
197 }
198 return {T(0), T(0)};
199}

207template <row G, row U, mutable_row Out>
208 requires std::same_as<row_value_t<G>, row_value_t<U>> && std::same_as<row_value_t<G>, row_value_t<Out>>
209constexpr void glu_rows(glu_kind kind, const G& gate, const U& up, Out&& out, const glu_parameters<row_value_t<G>>& p = {})
210{
211 const auto g = as_row(gate);
212 const auto u = as_row(up);
213 const auto y = as_mutable_row(out);
214 assert(u.size() == g.size() && y.size() == g.size());
215 detail::glu::dispatch(kind, [&](auto k) {
216 for (std::size_t i = 0; i < g.size(); ++i)
217 y[i] = glu(decltype(k)::value, g[i], u[i], p);
218 });
219}

229template <row G, row U, row D, mutable_row DG, mutable_row DU>
230 requires std::same_as<row_value_t<G>, row_value_t<U>> && std::same_as<row_value_t<G>, row_value_t<D>>
231 && std::same_as<row_value_t<G>, row_value_t<DG>> && std::same_as<row_value_t<G>, row_value_t<DU>>
232constexpr void glu_rows_backward(glu_kind kind, const G& gate, const U& up, const D& dout, DG&& dgate, DU&& dup,
233 const glu_parameters<row_value_t<G>>& p = {})
234{
235 const auto g = as_row(gate);
236 const auto u = as_row(up);
237 const auto d = as_row(dout);
238 const auto dg = as_mutable_row(dgate);
239 const auto du = as_mutable_row(dup);
240 assert(u.size() == g.size() && d.size() == g.size() && dg.size() == g.size() && du.size() == g.size());
241 detail::glu::dispatch(kind, [&](auto k) {
242 for (std::size_t i = 0; i < g.size(); ++i) {
243 const auto r = glu_backward(decltype(k)::value, g[i], u[i], d[i], p);
244 dg[i] = r.gate;
245 du[i] = r.up;
246 }
247 });
248}

257template <row X, mutable_row Out>
258 requires std::same_as<row_value_t<X>, row_value_t<Out>>
259constexpr void glu_split(glu_kind kind, const X& x, Out&& out, bool swapped = false, const glu_parameters<row_value_t<X>>& p = {})
260{
261 const auto in = as_row(x);
262 const auto y = as_mutable_row(out);
263 const std::size_t n = y.size();
264 assert(in.size() == 2 * n);
265 glu_rows(kind, in.subspan(swapped ? n : 0, n), in.subspan(swapped ? 0 : n, n), y, p);
266}

275template <row X, row D, mutable_row DX>

277constexpr void glu_split_backward(glu_kind kind, const X& x, const D& dout, DX&& dx, bool swapped = false,
