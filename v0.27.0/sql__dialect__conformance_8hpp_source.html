75 template <typename Probe>
76 requires detail::sql_dialect_probe<Probe>
77 [[nodiscard]] capability_conformance_report run_sql_dialect_conformance()
78 {
79 capability_conformance_report report;
80
81 detail::capability_check<Probe>(report, "every type has a name", [](Probe& p, const auto& expect) {
82 const auto& d = p.dialect();
83 bool named = true;
84 for (const sql_type t : sql_types) {
85 const auto n = d.column_type(t);
86 named = named && n && !n->empty();
87 }
88 expect(named, "column_type names every sql_type");
89 expect(!d.name().empty(), "the dialect has a name");
90 });
91
92 detail::capability_check<Probe>(report, "identifiers are quoted", [](Probe& p, const auto& expect) {
93 const auto& d = p.dialect();
94 std::string out = "x";
95 expect(d.append_identifier(out, "Price") && out == "x\"Price\"", "a name is quoted and keeps its case");
96 out.clear();
97 expect(d.append_identifier(out, "a\"b") && out == "\"a\"\"b\"", "an embedded quote is doubled");
98 out.clear();
99 expect(d.append_identifier(out, "select; drop table x") && out.front() == '"' && out.back() == '"',
100 "a keyword or punctuation stays inside the quotes");
101 });
102
103 detail::capability_check<Probe>(report, "unquotable identifiers are refused", [](Probe& p, const auto& expect) {
104 const auto& d = p.dialect();
105 std::string out = "kept";
106 expect(!d.append_identifier(out, ""), "an empty identifier is refused");
107 expect(!d.append_identifier(out, std::string_view("a\0b", 3)), "a NUL is refused");
108 expect(out == "kept", "nothing is appended when refused");
109 });
110
111 detail::capability_check<Probe>(report, "placeholders are distinct", [](Probe& p, const auto& expect) {
112 const auto& d = p.dialect();
113 std::set<std::string> seen;
114 bool numbered = true;
115 for (std::size_t n = 1; n <= 100; ++n) {
116 std::string out;
117 d.append_placeholder(out, n);
118 numbered = numbered && out.find(std::to_string(n)) != std::string::npos;
119 seen.insert(out);
120 }
121 expect(seen.size() == 100, "100 numbers make 100 placeholders");
122 expect(numbered, "each placeholder carries its number");
123 });
124
125 detail::capability_check<Probe>(report, "a transaction begins with BEGIN", [](Probe& p, const auto& expect) {
126 expect(p.dialect().begin_transaction().substr(0, 5) == "BEGIN", "begin_transaction starts with BEGIN");
127 });
128
129 detail::capability_check<Probe>(report, "the generators use the dialect", [](Probe& p, const auto& expect) {
130 const auto& d = p.dialect();
131 const sql_table t = detail::kit_sql_table();
132 const auto create = sql_create_table(d, t);
133 expect(create.is_ok(), "create_table succeeds");
134 if (create) {
135 expect(create->find("\"kit \"\"quotes\"\"\"") != std::string::npos, "the table's name is quoted");
136 expect(create->find(std::string(*d.column_type(sql_type::timestamp))) != std::string::npos, "the dialect's types are used");
137 expect(create->find("PRIMARY KEY (\"id\")") != std::string::npos, "the primary key is declared");
138 }
139 const auto insert = sql_insert(d, t, sql_conflict::update);
140 std::string third;
141 d.append_placeholder(third, 3);
142 expect(insert.is_ok() && insert->find(third) != std::string::npos, "an insert binds every column");
143 sql_table bad = t;
144 bad.columns[1].name = std::string("n\0", 2);
145 expect(sql_create_table(d, bad).is_err() && sql_create_table(d, bad).unwrap_err().code == sql_errc::misuse,
146 "a column that cannot be quoted is misuse");
147 });
148
149 return report;
150 }
