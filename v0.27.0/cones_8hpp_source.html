213 // A root of an increasing function on (a, b), f(x) returning {value, derivative}: Newton, kept inside the
214 // bracket by bisection. f(a) <= 0 <= f(b) is assumed, never evaluated.
215 template <typename T, typename F>
216 constexpr T increasing_root(F f, T a, T b)
217 {
218 const T eps = std::numeric_limits<T>::epsilon();
219 T x = a + (b - a) / T(2);
220 for (int i = 0; i < 400; ++i) {
221 const std::pair<T, T> g = f(x);
222 if (g.first == T(0))
223 return x;
224 if (g.first < T(0))
225 a = x;
226 else
227 b = x;
228 T next = x - g.first / g.second;
229 if (!(next > a && next < b))
230 next = a + (b - a) / T(2);
231 if (next == x || math::abs(next - x) <= T(2) * eps * math::abs(next) || !(b - a > T(2) * eps * max_(math::abs(a), math::abs(b))))
232 return next;
233 x = next;
234 }
235 return x;
236 }

254 // The projection of (r0, s0, t0), its largest magnitude 1, onto the exponential cone (Friberg 2021).
255 template <typename T>
256 constexpr std::array<T, 3> exp_unit(T r0, T s0, T t0)
257 {
258 if (in_exp(r0, s0, t0))
259 return {r0, s0, t0};
260 if (in_exp_dual(-r0, -s0, -t0))
261 return {T(0), T(0), T(0)};
262 const std::array<T, 3> heuristic{min_(r0, T(0)), T(0), max_(t0, T(0))};
263 if (r0 <= T(0) && s0 <= T(0))
264 return heuristic;
265 // The interval where (rho - 1) r0 + s0 > 0 and r0 - rho s0 > 0
266 T lo = -inf<T>(), hi = inf<T>();
267 if (r0 > T(0))
268 lo = T(1) - s0 / r0;
269 else if (r0 < T(0))
270 hi = T(1) - s0 / r0;
271 if (s0 > T(0))
272 hi = min_(hi, r0 / s0);
273 else if (s0 < T(0))
274 lo = max_(lo, r0 / s0);
275 // h(rho) e^{-|rho|} and its derivative
276 auto g = [&](T p) {
277 const T e = math::exp(-math::abs(p)), e2 = e * e;
278 const T ep = p >= T(0) ? T(1) : e2, em = p >= T(0) ? e2 : T(1);
279 const T a = (p - T(1)) * r0 + s0, b = r0 - p * s0, q = p * p - p + T(1);
280 const T h = a * ep - b * em - q * t0 * e;
281 const T dh = (p * r0 + s0) * ep + (r0 - (p - T(1)) * s0) * em - (T(2) * p - T(1)) * t0 * e;
282 return std::pair<T, T>{h, p >= T(0) ? dh - h : dh + h};
283 };
284 // An unbounded end: step out, doubling, until h changes sign (it must: h e^{-|rho|} tends to a positive
285 // value above and a negative one below). 60 doublings keep rho^2 finite in float.
286 if (!(lo > -inf<T>())) {
287 T step(1);
288 lo = hi - step;
289 for (int i = 0; i < 60 && !(g(lo).first < T(0)); ++i) {
290 step *= T(2);
291 lo = hi - step;
292 }
293 }
294 if (!(hi < inf<T>())) {
295 T step(1);
296 hi = lo + step;
297 for (int i = 0; i < 60 && !(g(hi).first > T(0)); ++i) {
298 step *= T(2);
299 hi = lo + step;
300 }
301 }
302 const T rho = increasing_root(g, lo, hi);
303 // The projection onto the ray (rho, 1, e^rho), normalised so that nothing overflows
304 std::array<T, 3> ray{};
305 if (rho > T(0)) {
306 const T e = math::exp(-rho);
307 ray = {rho * e, e, T(1)};
308 } else {
309 const T k = max_(-rho, T(1));
310 ray = {rho / k, T(1) / k, math::exp(rho) / k};
311 }
312 const T scale = (r0 * ray[0] + s0 * ray[1] + t0 * ray[2]) / (ray[0] * ray[0] + ray[1] * ray[1] + ray[2] * ray[2]);
313 const std::array<T, 3> candidate{scale * ray[0], scale * ray[1], scale * ray[2]};
314 if (scale > T(0) && finite3(candidate) && dist3(candidate, r0, s0, t0) <= dist3(heuristic, r0, s0, t0))
315 return candidate;
316 return heuristic;
317 }

349 // The projection of (x0, y0, z0), its largest magnitude 1, onto the power cone (Hien 2015).
350 template <typename T>
351 constexpr std::array<T, 3> pow_unit(T x0, T y0, T z0, T alpha)
352 {
353 if (in_pow(x0, y0, z0, alpha))
354 return {x0, y0, z0};
355 if (in_pow_dual(-x0, -y0, -z0, alpha))
356 return {T(0), T(0), T(0)};
357 const std::array<T, 3> heuristic{max_(x0, T(0)), max_(y0, T(0)), T(0)};
358 const T az = math::abs(z0);
359 if (az == T(0))
360 return heuristic;
361 // x and y depend on r through r(|z0| - r), symmetric about |z0|/2. The root is found in t on (0, |z0|/2),
362 // r = t below the middle and r = |z0| - t above it, so that a root next to either end (x or y near 0) is
363 // resolved to the type's precision rather than to one ulp of |z0|.
364 const T half = az / T(2);
365 auto xy = [&](T t) {
366 const T cx = T(4) * alpha * t * (az - t), cy = T(4) * (T(1) - alpha) * t * (az - t);
367 const T rx = math::sqrt(x0 * x0 + cx), ry = math::sqrt(y0 * y0 + cy);
368 return std::array<T, 4>{half_root(x0, cx, rx), half_root(y0, cy, ry), rx, ry};
369 };
370 const std::array<T, 4> mid = xy(half);
371 const bool upper = !(pow_mean(mid[0], mid[1], alpha) < half);
372 auto g = [&](T t) {
373 const std::array<T, 4> p = xy(t);
374 const T m = pow_mean(p[0], p[1], alpha);
375 const T dx = alpha * (az - T(2) * t) / p[2], dy = (T(1) - alpha) * (az - T(2) * t) / p[3];
376 const T dm = m * (alpha * dx / p[0] + (T(1) - alpha) * dy / p[1]);
377 return upper ? std::pair<T, T>{m - (az - t), dm + T(1)} : std::pair<T, T>{t - m, T(1) - dm};
378 };
379 const T t = increasing_root(g, T(0), half);
380 const T r = upper ? az - t : t;
381 const std::array<T, 4> p = xy(t);
382 const std::array<T, 3> candidate{p[0], p[1], z0 < T(0) ? -r : r};
383 if (finite3(candidate) && dist3(candidate, x0, y0, z0) <= dist3(heuristic, x0, y0, z0))
384 return candidate;
385 return heuristic;
386 }

388 // Scale a 3-vector to largest magnitude 1, project with f, scale back
389 template <typename T, typename F>
390 constexpr void project3(std::span<T, 3> v, F f)
391 {
392 const T m = max_(max_(math::abs(v[0]), math::abs(v[1])), math::abs(v[2]));
393 if (!(m > T(0)) || !(m < inf<T>()))
394 return;
395 const std::array<T, 3> p = f(v[0] / m, v[1] / m, v[2] / m);
396 for (std::size_t i = 0; i < 3; ++i)
397 v[i] = p[i] * m;
398 }

400 // ---- Simplex thresholds ----
401 // Condat (2016), Algorithm 1: the threshold tau of the projection of y onto {x >= 0, sum x = r}, r > 0. The
402 // workspace holds Condat's two lists side by side: the waiting list at [0, nt), the candidate list at
403 // [nt, nt + nv), so that moving the candidates to the waiting list is nt += nv.
404 template <typename T, typename Y>
405 constexpr T condat_tau(std::size_t n, Y y, T r, std::span<T> ws)
406 {
407 std::size_t nt = 0, nv = 1;
408 ws[0] = y(0);
409 T rho = ws[0] - r;
410 for (std::size_t i = 1; i < n; ++i) {
411 const T yi = y(i);
412 if (yi > rho) {
413 rho += (yi - rho) / static_cast<T>(nv + 1);
414 if (rho > yi - r) {
415 ws[nt + nv++] = yi;
416 } else {
417 nt += nv;
418 ws[nt] = yi;
419 nv = 1;
420 rho = yi - r;
421 }
422 }
423 }
424 for (std::size_t k = nt; k-- > 0;) {
425 const T yk = ws[k];
426 if (yk > rho) {
427 ws[k] = ws[nt - 1];
428 ws[nt - 1] = yk;
429 --nt;
430 ++nv;
431 rho += (yk - rho) / static_cast<T>(nv);
432 }
433 }
434 for (bool changed = true; changed;) {
435 changed = false;
436 for (std::size_t k = 0; k < nv && nv > 1;) {
437 const T yk = ws[nt + k];
438 if (yk <= rho) {
439 ws[nt + k] = ws[nt + nv - 1];
440 --nv;
441 rho += (rho - yk) / static_cast<T>(nv);
442 changed = true;
443 } else {
444 ++k;
445 }
446 }
447 }
448 return rho;
449 }

451 // Michelot (1986): the same threshold without a workspace, a sweep per change of the active set.
452 template <typename T, typename Y>
453 constexpr T michelot_tau(std::size_t n, Y y, T r)
454 {
455 T sum(0);
456 for (std::size_t i = 0; i < n; ++i)
457 sum += y(i);
458 T tau = (sum - r) / static_cast<T>(n);
459 std::size_t count = n;
460 for (std::size_t sweep = 0; sweep <= n; ++sweep) {
461 T s(0);
462 std::size_t c = 0;
463 for (std::size_t i = 0; i < n; ++i)
464 if (y(i) > tau) {
465 s += y(i);
466 ++c;
467 }
468 if (c == count || c == 0)
469 break;
470 count = c;
471 tau = (s - r) / static_cast<T>(c);
472 }
473 return tau;
474 }

765 // X+ from an eigen decomposition (values w, vectors in the columns of V, row-major n x n), written as an
766 // svec into x, from whichever of the positive or the negative eigenvalues are fewer (x holds svec(X)).
767 template <typename T, typename V>
768 constexpr void psd_from_eigen(std::span<T> x, std::size_t n, const T* w, V vec)
769 {
770 std::size_t negative = 0;
771 for (std::size_t k = 0; k < n; ++k)
772 negative += w[k] < T(0) ? 1 : 0;
773 if (negative == 0)
774 return;
775 const bool from_negative = negative <= n - negative;
776 std::size_t idx = 0;
777 for (std::size_t j = 0; j < n; ++j)

1164 template <linalg::scalar T>
1165 constexpr result<void, error> project_capped_simplex(std::span<T> x, std::type_identity_t<T> r, std::type_identity_t<T> u = T(1)) noexcept(detail::cones::nt<T>)
1166 {
1167 const std::size_t n = x.size();
1168 if (!(u > T(0)) || !(r >= T(0)) || !(r <= static_cast<T>(n) * u))
1169 return err(error::infeasible);
1170 if (n == 0)
1171 return ok();
1172 auto clamp = [u](T v) { return v < T(0) ? T(0) : u < v ? u : v; };
1173 auto sum = [&](T tau) {
1174 T s(0);
1175 for (const T& v : x)
1176 s += clamp(v - tau);
1177 return s;
1178 };
1179 T lo = x[0], hi = x[0];
1180 for (const T& v : x) {
1181 lo = detail::cones::min_(lo, v);
1182 hi = detail::cones::max_(hi, v);
1183 }
1184 lo -= u; // sum(lo) = n u >= r, sum(hi) = 0 <= r
1185 for (int i = 0; i < 200; ++i) {
1186 const T mid = lo + (hi - lo) / T(2);
1187 if (!(mid > lo && mid < hi))
1188 break;
1189 if (sum(mid) > r)
1190 lo = mid;
1191 else
1192 hi = mid;
1193 }
1194 T tau = lo + (hi - lo) / T(2);
1195 // The exact threshold for the active set at tau
1196 T fixed(0);
1197 std::size_t free = 0;
1198 for (const T& v : x) {
1199 const T d = v - tau;
1200 if (d >= u)
1201 fixed += u;
1202 else if (d > T(0)) {
1203 fixed += v;
1204 ++free;
1205 }
1206 }
1207 if (free > 0) {
1208 const T exact = (fixed - r) / static_cast<T>(free);
1209 if (math::abs(sum(exact) - r) <= math::abs(sum(tau) - r))
1210 tau = exact;
1211 }
1212 for (T& v : x)
1213 v = clamp(v - tau);
1214 return ok();
1215 }
