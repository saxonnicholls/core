63 template <typename Probe>
64 concept encryptor_probe = std::is_default_constructible_v<Probe> &&
65 encryptor<std::remove_reference_t<decltype(std::declval<Probe&>().encryptor())>> && requires(Probe& p) {
66 { p.key() } -> std::convertible_to<std::string_view>;
67 { p.other_key() } -> std::convertible_to<std::string_view>;
68 p.forget(std::string_view{});
69 };

71 inline std::vector<std::byte> kit_bytes(std::string_view s)
72 {
73 std::vector<std::byte> out;
74 for (const char c : s)
75 out.push_back(static_cast<std::byte>(c));
76 return out;
77 }

79 inline bool kit_contains(std::span<const std::byte> hay, std::span<const std::byte> needle)
80 {
81 if (needle.empty() || hay.size() < needle.size())
82 return false;
83 for (std::size_t i = 0; i + needle.size() <= hay.size(); ++i) {
84 bool same = true;
85 for (std::size_t j = 0; j < needle.size() && same; ++j)
86 same = hay[i + j] == needle[j];
87 if (same)
88 return true;
89 }
90 return false;
91 }

100 [[nodiscard]] capability_conformance_report run_encryptor_conformance()
101 {
102 capability_conformance_report report;
103 using detail::kit_bytes;
104 static const std::vector<std::byte> secret = kit_bytes("card 4111 1111 1111 1111, " + std::string(capability_redaction_marker));
105 static const std::vector<std::byte> aad = kit_bytes("users/42/card");

107 detail::capability_check<Probe>(report, "what is sealed opens", [](Probe& p, const auto& expect) {
108 auto& e = p.encryptor();
109 const std::vector<std::byte> large(10000, std::byte{0x5a});
110 for (const auto* plain : {&secret, &large}) {
111 const auto sealed = e.encrypt(p.key(), *plain, aad);
112 expect(sealed.is_ok(), "encrypt succeeds");
113 if (!sealed)
114 return;
115 const auto back = e.decrypt(*sealed, aad);
116 expect(back.is_ok() && *back == *plain, "decrypt returns the plaintext");
117 }
118 const auto empty = e.encrypt(p.key(), {}, {});
119 expect(empty.is_ok() && e.decrypt(*empty, {}).is_ok() && e.decrypt(*empty, {})->empty(), "the empty plaintext round-trips");
120 expect(!e.algorithm().empty(), "the algorithm has a name");
121 });
122
123 detail::capability_check<Probe>(report, "every seal is fresh", [](Probe& p, const auto& expect) {
124 auto& e = p.encryptor();
125 const auto a = e.encrypt(p.key(), secret, aad);
126 const auto b = e.encrypt(p.key(), secret, aad);
127 expect(a.is_ok() && b.is_ok() && *a != *b, "two seals of the same plaintext differ");
128 if (a.is_ok())
129 expect(!detail::kit_contains(*a, secret), "the envelope does not contain the plaintext");
130 });

132 detail::capability_check<Probe>(report, "a changed envelope does not open", [](Probe& p, const auto& expect) {
133 auto& e = p.encryptor();
134 const auto sealed = e.encrypt(p.key(), secret, aad);
135 if (!sealed) {
136 expect(false, "encrypt succeeds");
137 return;
138 }
139 const std::size_t header = 2 + p.key().size();
140 bool all_refused = true, none_leaked = true;
141 for (std::size_t i = header; i < sealed->size(); ++i) { // every byte after the key id
142 std::vector<std::byte> changed = *sealed;
143 changed[i] ^= std::byte{0x01};
144 const auto r = e.decrypt(changed, aad);
145 all_refused = all_refused && r.is_err() && r.unwrap_err().code == capability_errc::signature_invalid;
146 none_leaked = none_leaked && r.is_err();
147 }
148 expect(all_refused, "any flipped bit is signature_invalid");
149 expect(none_leaked, "no plaintext comes back from a changed envelope");
150 std::vector<std::byte> shorter = *sealed;
151 shorter.pop_back();
152 const auto r = e.decrypt(shorter, aad);
153 expect(r.is_err(), "a truncated envelope does not open");
154 });
