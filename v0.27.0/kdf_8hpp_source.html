112template <hash_function H, hash_input P, hash_input I>
113constexpr result<void, kdf_error> hkdf_expand(const P& prk, const I& info, std::span<std::byte> out) noexcept
114{
115 if (out.empty())
116 return err(kdf_error::empty_output);
117 if (out.size() > 255 * H::digest_size)
118 return err(kdf_error::output_too_long);
119 const hmac<H> keyed{prk};
120 digest_of_t<H> t{};
121 std::size_t done = 0;
122 for (std::uint8_t counter = 1; done < out.size(); ++counter) {
123 hmac<H> m = keyed;
124 if (counter > 1)
125 m.update(t);
126 m.update(info);
127 m.update(std::array<std::uint8_t, 1>{counter});
128 t = m.finish();
129 for (std::size_t i = 0; i < H::digest_size && done < out.size(); ++i)
130 out[done++] = t.bytes[i];
131 }
132 secure_wipe(t);
133 return ok();
134}

188 // The single-step KDF's blocks (NIST SP 800-56A 5.8.2.1.1): H(counter || Z || FixedInfo) for counter = 1, 2, ...
189 // until out is full; `fixed(h)` feeds FixedInfo into h, so a caller can stream fields without a buffer.
190 template <hash_function H, hash_input Z, typename Fixed>
191 constexpr result<void, kdf_error> concat_kdf_blocks(const Z& z, Fixed&& fixed, std::span<std::byte> out) noexcept
192 {
193 if (out.empty())
194 return err(kdf_error::empty_output);
195 if (static_cast<std::uint64_t>(out.size()) > 0xFFFF'FFFFULL * H::digest_size)
196 return err(kdf_error::output_too_long);
197 digest_of_t<H> block{};
198 std::size_t done = 0;
199 for (std::uint32_t counter = 1; done < out.size(); ++counter) {
200 H h;
201 h.update(std::array<std::uint8_t, 4>{static_cast<std::uint8_t>(counter >> 24), static_cast<std::uint8_t>(counter >> 16),
202 static_cast<std::uint8_t>(counter >> 8), static_cast<std::uint8_t>(counter)});
203 h.update(z);
204 fixed(h);
205 block = h.finish();
206 for (std::size_t i = 0; i < H::digest_size && done < out.size(); ++i)
207 out[done++] = block.bytes[i];
208 }
209 secure_wipe(block);
210 return ok();
211 }

251template <hash_function H, hash_input Z, hash_input A, hash_input U, hash_input V>
252constexpr result<void, kdf_error> jwa_concat_kdf(const Z& z, const A& algorithm, const U& party_u, const V& party_v,
253 std::span<std::byte> out) noexcept
254{
255 if (static_cast<std::uint64_t>(out.size()) > 0xFFFF'FFFFULL / 8)
256 return err(kdf_error::output_too_long);
257 const auto field = [](H& h, const auto& bytes) {
258 h.update(detail::be32(detail::input_size(bytes)));
259 h.update(bytes);
260 };
261 return detail::concat_kdf_blocks<H>(z, [&](H& h) {
262 field(h, algorithm);
263 field(h, party_u);
264 field(h, party_v);
265 h.update(detail::be32(static_cast<std::uint64_t>(out.size()) * 8));
266 }, out);
267}

271 // PBKDF2's blocks (RFC 8018 5.2): T_i for i = 1, 2, ... until `length` bytes; `salt(m)` feeds the salt into an
272 // HMAC, `emit(offset, T_i, count)` takes the first `count` bytes of T_i for output offset `offset`.
273 template <hash_function H, hash_input P, typename Salt, typename Emit>
274 constexpr void pbkdf2_blocks(const P& password, Salt&& salt, std::uint32_t iterations, std::size_t length, Emit&& emit) noexcept
275 {
276 const hmac<H> keyed{password};
277 digest_of_t<H> u{}, t{};
278 std::size_t done = 0;
279 for (std::uint32_t block = 1; done < length; ++block) {
280 hmac<H> m = keyed;
281 salt(m);
282 const std::array<std::uint8_t, 4> index{static_cast<std::uint8_t>(block >> 24), static_cast<std::uint8_t>(block >> 16),
283 static_cast<std::uint8_t>(block >> 8), static_cast<std::uint8_t>(block)};
284 m.update(index);
285 u = m.finish();
286 t = u;
287 for (std::uint32_t i = 1; i < iterations; ++i) {
288 hmac<H> next = keyed;
289 next.update(u);
290 u = next.finish();
291 for (std::size_t j = 0; j < H::digest_size; ++j)
292 t.bytes[j] ^= u.bytes[j];
293 }
294 const std::size_t count = length - done < H::digest_size ? length - done : H::digest_size;
295 emit(done, t, count);
296 done += count;
297 }
298 secure_wipe(u);
299 secure_wipe(t);
300 }

313template <hash_function H, hash_input P, hash_input S>
314constexpr result<void, kdf_error> pbkdf2_hmac(const P& password, const S& salt, std::uint32_t iterations,
315 std::span<std::byte> out) noexcept
316{
317 if (iterations == 0)
318 return err(kdf_error::zero_iterations);
319 if (out.empty())
320 return err(kdf_error::empty_output);
321 if (out.size() / H::digest_size >= 0xffffffffull)
322 return err(kdf_error::output_too_long);
323 detail::pbkdf2_blocks<H>(password, [&salt](hmac<H>& m) { m.update(salt); }, iterations, out.size(),
324 [&out](std::size_t offset, const digest_of_t<H>& t, std::size_t count) {
325 for (std::size_t j = 0; j < count; ++j)
326 out[offset + j] = t.bytes[j];
327 });
328 return ok();
329}

342 static_assert(L >= 1, "snicholls::pbkdf2_hmac: L is at least 1 byte");
