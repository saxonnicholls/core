164 template <class SM>
165 [[nodiscard]] std::string to_record(const SM& machine, std::uint64_t version)
166 {
167 std::string out(persist_format);
168 out += "\nmachine ";
169 out += detail::short_name(type_name<typename SM::definition>());
170 out += "\nversion ";
171 detail::append_number(out, version);
172 out += "\nregions ";
173 detail::append_number(out, static_cast<std::uint64_t>(SM::slot_count));
174 for (const auto& slot : machine.snapshot()) {
175 out += '\n';
176 detail::append_number(out, slot.value);
177 out += ' ';
178 out += slot.name;
179 }
180 out += '\n';
181 return out;
182 }

190 template <class SM>
191 result<std::uint64_t, fsm_errc> from_record(std::string_view record, SM& machine) noexcept
192 {
193 std::string_view rest = record;
194 std::string_view name;
195 std::string_view field;
196 std::uint64_t version = 0;
197 std::size_t regions = 0;
198 if (detail::next_line(rest) != persist_format || !detail::parse_field(detail::next_line(rest), "machine", name) ||
199 !detail::parse_field(detail::next_line(rest), "version", field) || !detail::parse_number(field, version) ||
200 !detail::parse_field(detail::next_line(rest), "regions", field) || !detail::parse_number(field, regions))
201 return err(fsm_errc::corrupt);
202 if (name != detail::short_name(type_name<typename SM::definition>()))
203 return err(fsm_errc::wrong_machine);
204 if (regions != SM::slot_count)
205 return err(fsm_errc::shape_mismatch);
206 std::array<wire_state, SM::slot_count> slots{};
207 for (auto& slot : slots) {
208 const std::string_view line = detail::next_line(rest);
209 const std::size_t space = line.find(' ');
210 if (space == std::string_view::npos || !detail::parse_number(line.substr(0, space), slot.value))
211 return err(fsm_errc::corrupt);
212 slot.name = line.substr(space + 1);
213 }
214 if (!rest.empty())
215 return err(fsm_errc::corrupt);
216 const auto checked = SM::check_snapshot(slots);
217 if (checked.is_err())
218 return err(checked.unwrap_err_unchecked());
219 const auto restored = machine.restore_snapshot(slots);
220 if (restored.is_err())
221 return err(restored.unwrap_err_unchecked());
222 return ok(version);
223 }

234 template <kv_writable Store, kv_byte_source K, class SM>
235 result<std::uint64_t, fsm_errc> save(Store& store, const K& key, const SM& machine, std::uint64_t expected_version)
236 {
237 const std::uint64_t version = expected_version + 1;
238 const kv_bytes key_bytes = kv_bytes_of(key);
239 const std::string_view key_text(reinterpret_cast<const char*>(key_bytes.data()), key_bytes.size());
240 const std::string record = to_record(machine, version);
241 kv_write_batch batch;
242 batch.insert(version_key(key_text, version), record);
243 batch.put(key_text, record);
244 const auto applied = store.apply(batch);
245 if (applied.is_err())
246 return err(applied.unwrap_err_unchecked() == kv_errc::conflict ? fsm_errc::stale
247 : detail::from_kv(applied.unwrap_err_unchecked()));
248 return ok(version);
249 }

260 result<std::uint64_t, fsm_errc> load(const Store& store, const K& key, SM& machine)
261 {
262 std::string record;
263 const auto found = kv_get(store, key, record);
264 if (found.is_err())
265 return err(detail::from_kv(found.unwrap_err_unchecked()));
266 if (!*found)
267 return err(fsm_errc::not_found);
268 return from_record(record, machine);
269 }
