69 [[nodiscard]] capability_result<void> verify(const verify_request& r) const
70 {
71 if (injected_) {
72 const capability_error e = *injected_;
73 injected_.reset();
74 return err(e);
75 }
76 if (r.algorithm != signature_algorithm::hs256)
77 return fail(capability_errc::unsupported, {}, "the reference verifies HS256 only");
78 if (r.key_version == 0)
79 return fail(capability_errc::invalid_argument, {}, "an HS256 key needs its version");
80 const auto id = provider_id::make(r.key_id);
81 const auto current = store_->info(r.key_id);
82 if (!id || current.is_err() || r.key_version > current->version)
83 return fail(capability_errc::not_found, {}, "no such key version");
84 const signature s{signature_algorithm::hs256, *id, r.key_version,
85 std::vector<std::byte>(r.signature.begin(), r.signature.end())};
86 if (!store_->verify(s, r.message))
87 return fail(capability_errc::signature_invalid);
88 return ok();
89 }
