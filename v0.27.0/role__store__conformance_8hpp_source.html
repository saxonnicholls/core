62 template <typename Probe>
63 concept role_store_probe = std::is_default_constructible_v<Probe> &&
64 role_store<std::remove_reference_t<decltype(std::declval<Probe&>().store())>> &&
65 requires(Probe& p, capability_errc c, std::string_view r) { p.provider_fails(c, r); };

67 inline constexpr uuid role_kit_alice = uuid::from_u64(0xa11ce, 1);
68 inline constexpr uuid role_kit_bob = uuid::from_u64(0xb0b, 2);
69 inline constexpr uuid role_kit_actor = uuid::from_u64(0xac7, 3);
70 inline constexpr utc_s role_kit_t0 = utc_s::from_unix_seconds(1'800'000'000);

72 inline role_grant role_kit_grant(const uuid& party, std::string_view tenant, std::string_view role)
73 {
74 return role_grant{party, *tenant_id::make(tenant), *role_id::make(role), role_kit_actor, "because", role_kit_t0, std::nullopt};
75 }
