65 consteval std::optional<std::size_t> checked_permutation_count(std::size_t n, std::size_t r) noexcept
66 {
67 if (r > n)
68 return std::size_t{0};
69 std::size_t count = 1;
70 for (std::size_t i = 0; i < r; ++i) {
71 const std::size_t factor = n - i;
72 if (count > std::numeric_limits<std::size_t>::max() / factor)
73 return std::nullopt;
74 count *= factor;
75 }
76 return count;
77 }

79 // The one walk behind every permutation in core: each position takes each value with some
80 // left, in order, and recurses. remaining[v] is how many of value v are still unused, so
81 // repeated values give each distinct arrangement once, and all ones gives plain permutations.
82 template <std::size_t K, std::size_t R, typename F>
83 constexpr void for_each_arrangement(std::array<std::size_t, K>& remaining, std::array<std::size_t, R>& row, std::size_t depth,
84 F& f) noexcept(noexcept(f(std::as_const(row))))
85 {
86 if (depth == row.size()) {
87 f(std::as_const(row));
88 return;
89 }
90 for (std::size_t value = 0; value < remaining.size(); ++value) {
91 if (remaining[value] == 0)
92 continue;
93 --remaining[value];
94 row[depth] = value;
95 for_each_arrangement(remaining, row, depth + 1, f);
96 ++remaining[value];
97 }
98 }

134 template <std::size_t N, std::size_t R = N, typename F>
135 constexpr void for_each_permutation(F&& f) noexcept(noexcept(f(std::declval<const std::array<std::size_t, R>&>())))
136 {
137 std::array<std::size_t, N> remaining = detail::all_ones<N>();
138 std::array<std::size_t, R> row{};
139 detail::for_each_arrangement(remaining, row, 0, f);
140 }

147 template <std::size_t N, std::size_t R = N>
148 [[nodiscard]] constexpr std::array<std::array<std::size_t, R>, permutation_count_v<N, R>> index_permutations() noexcept
149 {
150 std::array<std::array<std::size_t, R>, permutation_count_v<N, R>> rows{};
151 std::size_t next = 0;
152 for_each_permutation<N, R>([&rows, &next](const std::array<std::size_t, R>& row) noexcept { rows[next++] = row; });
153 return rows;
154 }
