238// An instant as whole seconds (floored) + nanoseconds in [0, 1e9), with the fraction's excess-precision flag.
239struct split_instant
240{
241 std::int64_t seconds;
242 std::int64_t nanos;
243 bool excess; // non-zero digits finer than 1 ns
244};
245
246// (seconds, nanos) -> ticks of D. Errors: excess precision under reject, or out of D's range.
247template <utc_duration D>
248constexpr std::errc to_ticks(split_instant s, precision_policy policy, typename D::rep& out) noexcept
249{
250 using rep = typename D::rep;
251 using period = typename D::period;
252 std::int64_t ticks = 0;
253 bool excess = s.excess;
254 if constexpr (period::den > 1) { // 1/10^k s
255 constexpr std::int64_t den = period::den;
256 constexpr std::int64_t ns_per_tick = 1000000000 / den;
257 excess |= s.nanos % ns_per_tick != 0;
258 if (excess && policy == precision_policy::reject) return std::errc::result_out_of_range;
259 constexpr std::int64_t max = std::numeric_limits<std::int64_t>::max();
260 constexpr std::int64_t min = std::numeric_limits<std::int64_t>::min();
261 constexpr std::int64_t hi = max / den;
262 constexpr std::int64_t lo = min / den; // truncated toward 0: lo - 1 may still hold a valid instant
263 if (s.seconds > hi || s.seconds < lo - 1) return std::errc::result_out_of_range;
264 const std::int64_t sub = s.nanos / ns_per_tick; // [0, den)
265 if (s.seconds >= 0) {
266 ticks = s.seconds * den;
267 if (sub > max - ticks) return std::errc::result_out_of_range;
268 ticks += sub;
269 } else { // (seconds + 1) * den - (den - sub), never overflowing on the way
270 ticks = (s.seconds + 1) * den;
271 const std::int64_t back = den - sub;
272 if (ticks < min + back) return std::errc::result_out_of_range;
273 ticks -= back;
274 }
275 } else { // whole seconds (num >= 1)
276 constexpr std::int64_t num = period::num;
277 const std::int64_t q = floor_div(s.seconds, num);
278 excess |= s.nanos != 0 || s.seconds - q * num != 0;
279 if (excess && policy == precision_policy::reject) return std::errc::result_out_of_range;
280 ticks = q;
281 }
282 if constexpr (sizeof(rep) < sizeof(std::int64_t)) {
283 if (ticks < static_cast<std::int64_t>(std::numeric_limits<rep>::min()) ||
284 ticks > static_cast<std::int64_t>(std::numeric_limits<rep>::max()))
285 return std::errc::result_out_of_range;
286 }
287 out = static_cast<rep>(ticks);
288 return std::errc{};
289}

293// days_from_civil for y in 0..9999 in 32-bit unsigned arithmetic (no negative-era branch): years are shifted by
294// one 400-year era so y - 1 never goes below zero, then the era's 146097 days are taken back off.
295constexpr std::int64_t days_from_civil_0000_9999(unsigned y, unsigned m, unsigned d) noexcept
296{
297 y += 400 - (m <= 2);
298 const unsigned era = y / 400;
299 const unsigned yoe = y - era * 400;
300 const unsigned doy = (153 * (m > 2 ? m - 3 : m + 9) + 2) / 5 + d - 1;
301 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
302 return static_cast<std::int64_t>(era * 146097 + doe) - 719468 - 146097;
303}
304constexpr unsigned digit(char c) noexcept { return static_cast<unsigned>(c - '0'); }
305constexpr unsigned two(const char* p) noexcept { return digit(p[0]) * 10 + digit(p[1]); }

307// "YYYY-MM-DD" at p (at least 10 chars available). Syntax and validity.
308constexpr bool read_date(const char* p, unsigned& y, unsigned& m, unsigned& d) noexcept
309{
310 if (!(is_digit(p[0]) && is_digit(p[1]) && is_digit(p[2]) && is_digit(p[3]) && p[4] == '-' &&
311 is_digit(p[5]) && is_digit(p[6]) && p[7] == '-' && is_digit(p[8]) && is_digit(p[9])))
312 return false;
313 y = two(p) * 100 + two(p + 2);
314 m = two(p + 5);
315 d = two(p + 8);
316 if (m - 1 > 11 || d - 1 > 30) return false; // unsigned: also rejects 0
317 if (d <= 28) return true;
318 const bool leap = y % 4 == 0 && (y % 100 != 0 || y % 400 == 0); // 32-bit, y in 0..9999
319 return d <= (m == 2 ? 28U + leap : (m == 4 || m == 6 || m == 9 || m == 11) ? 30U : 31U);
320}

370template <utc_duration D>
371constexpr std::to_chars_result to_chars(char* first, char* last, const utc<D>& t, int precision) noexcept
372{
373 using period = typename D::period;
374 constexpr int digits = utc<D>::fraction_digits;
375 if (precision < 0 || precision > 9) return {last, std::errc::invalid_argument};
376
377 const std::int64_t count = static_cast<std::int64_t>(t.time_since_epoch().count());
378 std::int64_t seconds = 0;
379 std::int64_t sub = 0; // ticks of D below one second (0 for whole-second D)
380 if constexpr (period::den > 1) {
381 seconds = detail::floor_div(count, period::den);
382 sub = count - seconds * period::den;
383 } else {
384 constexpr std::int64_t num = period::num;
385 constexpr std::int64_t lo = -(-detail::rfc3339_min_seconds / num); // ceil(min / num), min < 0
386 constexpr std::int64_t hi = detail::rfc3339_max_seconds / num;
387 if (count < lo || count > hi) return {last, std::errc::value_too_large};
388 seconds = count * num;
389 }
390 if (seconds < detail::rfc3339_min_seconds || seconds > detail::rfc3339_max_seconds)
391 return {last, std::errc::value_too_large};
392
393 const std::size_t n = rfc3339_length(precision);
394 if (static_cast<std::size_t>(last - first) < n) return {last, std::errc::value_too_large};
395
396 const std::int64_t days = detail::floor_div(seconds, 86400);
397 const auto sod = static_cast<unsigned>(seconds - days * 86400);
398 const civil_date c = civil_from_days_fast(days); // years 0000..9999, checked above
399 const auto y = static_cast<unsigned>(c.year);
400
401 char* p = first;
402 detail::put2(p, y / 100);
403 detail::put2(p + 2, y % 100);
404 p[4] = '-';
405 detail::put2(p + 5, c.month);
406 p[7] = '-';
407 detail::put2(p + 8, c.day);
408 p[10] = 'T';
409 detail::put2(p + 11, sod / 3600);
410 p[13] = ':';
411 detail::put2(p + 14, sod / 60 % 60);
412 p[16] = ':';
413 detail::put2(p + 17, sod % 60);
414 p += 19;
415 if (precision > 0) {
416 std::int64_t frac = precision <= digits ? sub / detail::pow10[static_cast<std::size_t>(digits - precision)]
417 : sub * detail::pow10[static_cast<std::size_t>(precision - digits)];
418 *p = '.';
419 char* q = p + precision; // last digit
420 int left = precision;
421 for (; left >= 2; left -= 2, q -= 2) {
422 detail::put2(q - 1, static_cast<unsigned>(frac % 100));
423 frac /= 100;
424 }
425 if (left == 1) *q = static_cast<char>('0' + frac % 10);
426 p += precision + 1;
427 }
428 *p++ = 'Z';
429 return {p, std::errc{}};
430}

558struct rfc3339_scan
559{
560 split_instant instant{0, 0, false}; // the UTC instant, with second 60 read as 59
561 bool leap = false; // second 60 (already checked to be 23:59:60 UTC)
562 const char* end = nullptr; // one past the last character consumed
563 std::errc ec{}; // errc{} or invalid_argument
564};
565
566constexpr rfc3339_scan scan_rfc3339(const char* first, const char* last, const parse_options& options) noexcept
567{
568 rfc3339_scan r;
569 r.ec = std::errc::invalid_argument;
570 if (last - first < 19) return r;
571
572 unsigned y = 0, mo = 0, d = 0;
573 if (!read_date(first, y, mo, d)) return r;
574 const char* p = first + 10;
575 if (!(*p == 'T' || *p == 't' || *p == ' ')) return r;
576 if (!(is_digit(p[1]) && is_digit(p[2]) && p[3] == ':' && is_digit(p[4]) && is_digit(p[5]) && p[6] == ':' &&
577 is_digit(p[7]) && is_digit(p[8])))
578 return r;
579 const unsigned hh = two(p + 1), mi = two(p + 4), ss = two(p + 7);
580 if (hh > 23 || mi > 59 || ss > 60) return r;
581 p += 9;
582
583 if (p != last && *p == '.') {
584 ++p;
585 if (p == last || !is_digit(*p)) return r;
586 const char* const digits = p;
587 const char* const nine = last - p > 9 ? p + 9 : last;
588 std::uint32_t nanos = 0;
589 for (; p != nine && is_digit(*p); ++p) nanos = nanos * 10 + digit(*p);
590 nanos *= static_cast<std::uint32_t>(pow10[static_cast<std::size_t>(9 - (p - digits))]);
591 r.instant.nanos = nanos;
592 for (; p != last && is_digit(*p); ++p) r.instant.excess |= *p != '0'; // beyond 1 ns
593 }
594
595 std::int64_t offset = 0; // seconds east of UTC
596 if (p != last && (*p == 'Z' || *p == 'z')) {
597 ++p;
598 } else if (p != last && (*p == '+' || *p == '-')) {
599 const bool negative = *p == '-';
600 if (last - p < 3 || !is_digit(p[1]) || !is_digit(p[2])) return r;
601 const unsigned oh = two(p + 1);
602 const char* q = p + 3;
603 if (last - q >= 3 && *q == ':' && is_digit(q[1]) && is_digit(q[2]))
604 q += 1;
605 else if (!(options.basic_offset && last - q >= 2 && is_digit(q[0]) && is_digit(q[1])))
606 return r;
607 const unsigned om = two(q);
608 if (oh > 23 || om > 59) return r;
609 offset = (static_cast<std::int64_t>(oh) * 3600 + om * 60) * (negative ? -1 : 1);
610 p = q + 2;
611 } else if (options.require_offset) {
612 return r;
613 }
614
615 r.leap = ss == 60;
616 r.instant.seconds =
617 days_from_civil_0000_9999(y, mo, d) * 86400 + (hh * 3600 + mi * 60 + (r.leap ? 59 : ss)) - offset;
618 if (r.leap && r.instant.seconds - floor_div(r.instant.seconds, 86400) * 86400 != 86399) return r; // not 23:59 UTC
619 r.end = p;
620 r.ec = std::errc{};
621 return r;
622}

624template <utc_duration D>
625constexpr std::errc evaluate_rfc3339(const rfc3339_scan& s, const parse_options& options, typename D::rep& ticks) noexcept
626{
627 if (!s.leap) return to_ticks<D>(s.instant, options.precision, ticks);
628 if (options.leap_second == leap_second_policy::reject) return std::errc::result_out_of_range;
629 // The last representable instant before the insertion: one tick before ceil<D>(next second).
630 const split_instant next_second{s.instant.seconds + 1, 0, false};
631 typename D::rep next{};
632 if (to_ticks<D>(next_second, precision_policy::reject, next) == std::errc{}) { // D divides the second
633 if (next == std::numeric_limits<typename D::rep>::min()) return std::errc::result_out_of_range;
634 ticks = static_cast<typename D::rep>(next - 1);
635 return std::errc{};
636 }
637 return to_ticks<D>(next_second, precision_policy::truncate, ticks); // the floor already lies before it
638}

648template <utc_duration D>
649constexpr std::from_chars_result from_chars(const char* first, const char* last, utc<D>& out,
650 parse_options options = {}) noexcept
651{
652 const detail::rfc3339_scan s = detail::scan_rfc3339(first, last, options);
653 if (s.ec != std::errc{}) return {first, s.ec};
654 typename D::rep ticks{};
655 const std::errc ec = detail::evaluate_rfc3339<D>(s, options, ticks);
656 if (ec != std::errc{}) return {first, ec};
657 out = utc<D>(typename utc<D>::time_point(D(ticks)));
658 return {s.end, std::errc{}};
659}

677// One past the syntactically valid Unix-number text -?DIGITS(.DIGITS)? at first, or nullptr.
678constexpr const char* scan_unix(const char* first, const char* last) noexcept
679{
680 const char* p = first;
681 if (p != last && *p == '-') ++p;
682 if (p == last || !is_digit(*p)) return nullptr;
683 while (p != last && is_digit(*p)) ++p;
684 if (p != last && *p == '.') {
685 ++p;
686 if (p == last || !is_digit(*p)) return nullptr;
687 while (p != last && is_digit(*p)) ++p;
688 }
689 return p;
690}

692// [first, end) is syntactically valid (scan_unix).
693template <utc_duration Unit, utc_duration D>
694constexpr std::errc evaluate_unix(const char* first, const char* end, const parse_options& options,
695 typename D::rep& ticks) noexcept
696{
697 constexpr int unit_digits = utc<Unit>::fraction_digits; // s 0, ms 3, us 6, ns 9
698 const char* p = first;
699 const bool negative = *p == '-';
700 if (negative) ++p;
701
702 // Magnitude in Unit, up to 2^63 (so INT64_MIN ticks parse) or 2^63 - 1.
703 constexpr std::uint64_t int64_max = static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
704 const std::uint64_t limit = negative ? int64_max + 1 : int64_max;
705 std::uint64_t whole = 0;
706 for (; p != end && is_digit(*p); ++p) {
707 const unsigned dg = digit(*p);
708 if (whole > (limit - dg) / 10) return std::errc::result_out_of_range;
709 whole = whole * 10 + dg;
710 }
711 const auto unit_scale = static_cast<std::uint64_t>(pow10[unit_digits]);
712 const std::uint64_t whole_seconds = whole / unit_scale; // <= 2^63
713 split_instant s{0, static_cast<std::int64_t>(whole % unit_scale) * pow10[9 - unit_digits], false};
714 if (p != end) { // '.' DIGITS: digit n of the fraction is worth 10^(8 - unit_digits - n) ns
715 ++p;
716 for (int n = 0; p != end; ++p, ++n) {
717 if (n < 9 - unit_digits)
718 s.nanos += digit(*p) * pow10[static_cast<std::size_t>(8 - unit_digits - n)];
719 else if (*p != '0')
720 s.excess = true;
721 }
722 }
723 if (!negative) {
724 s.seconds = static_cast<std::int64_t>(whole_seconds); // <= 2^63 - 1
725 } else if (s.nanos == 0 && !s.excess) {
726 s.seconds = whole_seconds == 0 ? 0 : -static_cast<std::int64_t>(whole_seconds - 1) - 1; // 2^63 -> INT64_MIN
727 } else {
728 // -(W + n + e) with 0 < n + e < 1 s floors to -(W + 1) s + (1e9 - n - [e > 0]) ns: digits finer than
729 // 1 ns move a negative value one more ns into the past.
730 if (whole_seconds > int64_max) return std::errc::result_out_of_range; // only 2^63 itself
731 s.seconds = -static_cast<std::int64_t>(whole_seconds) - 1;
732 s.nanos = 1000000000 - s.nanos - (s.excess ? 1 : 0);
733 }
734 return to_ticks<D>(s, options.precision, ticks);
735}

752{
