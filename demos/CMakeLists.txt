# Demos. The layout mirrors include/snicholls/core/: <sub>/<name>.hpp has
# <sub>/<name>_demo.cpp here. Each prints what it does and exits non-zero if the
# result is not what it documents, so every demo is also a CTest (label demo).
# scripts/check_mirror.py fails CI if a public header has no demo or the demo
# is not listed below.
function(snicholls_core_add_demo name source)
    add_executable(snicholls_core_${name} ${source})
    target_link_libraries(snicholls_core_${name} PRIVATE snicholls::core)
    snicholls_core_test_options(snicholls_core_${name})
    add_test(NAME demo.${name} COMMAND snicholls_core_${name})
    set_tests_properties(demo.${name} PROPERTIES LABELS demo)
endfunction()
