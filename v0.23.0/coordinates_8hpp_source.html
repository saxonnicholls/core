78 template <linalg::scalar T>
79 [[nodiscard]] constexpr linalg::fixed_vector<T, 3> cart_to_sphere(T x, T y, T z) noexcept(linalg::detail::nothrow_scalar<T>)
80 {
81 const T r = static_cast<T>(math::sqrt(x * x + y * y + z * z));
82 if (r == T(0))
83 return {{T(0), T(0), T(0)}};
84 const T h = static_cast<T>(math::sqrt(x * x + y * y));
85 return {{r, h == T(0) ? T(0) : static_cast<T>(math::atan2(y, x)), static_cast<T>(math::asin(z / r))}};
86 }

123 template <linalg::scalar T>
124 [[nodiscard]] constexpr linalg::fixed_vector<T, 3> az_el_range_to_cart(T azimuth, T elevation, T range) noexcept(linalg::detail::nothrow_scalar<T>)
125 {
126 const T sa = static_cast<T>(math::sin(azimuth)), se = static_cast<T>(math::sin(elevation));
127 const T q = T(1) - sa * sa - se * se;
128 return {{range * sa, range * se, range * static_cast<T>(math::sqrt(q > T(0) ? q : T(0)))}};
129 }
