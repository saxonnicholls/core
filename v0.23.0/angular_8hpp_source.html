94 template <point A, point B>
95 requires compatible_points<A, B>
96 [[nodiscard]] constexpr distance_t<A, B> cosine_similarity(const A& a, const B& b) noexcept(noexcept(coordinate(a, 0)) && noexcept(coordinate(b, 0)))
97 {
98 using T = distance_t<A, B>;
99 T ab(0), aa(0), bb(0);
100 const std::size_t n = dimension(a);
101 for (std::size_t i = 0; i < n; ++i) {
102 const T x = static_cast<T>(coordinate(a, i)), y = static_cast<T>(coordinate(b, i));
103 ab += x * y;
104 aa += x * x;
105 bb += y * y;
106 }
107 if (!(aa > T(0)) || !(bb > T(0)))
108 return T(0);
109 const T c = ab / static_cast<T>(math::sqrt(aa * bb));
110 return c > T(1) ? T(1) : c < T(-1) ? T(-1) : c;
111 }
