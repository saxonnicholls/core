96 [[nodiscard]] static constexpr result<rate, money_errc> parse(const asset& from, const asset& to,
97 std::string_view text) noexcept
98 {
99 const std::size_t point = text.find('.');
100 if (text.empty() || text == "." ||
101 (point != std::string_view::npos && text.find('.', point + 1) != std::string_view::npos))
102 return err(money_errc::malformed);
103 const std::size_t places = point == std::string_view::npos ? 0 : text.size() - point - 1;
104 if (places > 77)
105 return err(money_errc::overflow);
106 uint256 n;
107 for (const char c : text) {
108 if (c == '.')
109 continue;
110 if (c < '0' || c > '9')
111 return err(money_errc::malformed);
112 const auto t = uint256::checked_mul(n, 10);
113 const auto s = t ? uint256::checked_add(*t, static_cast<std::uint64_t>(c - '0')) : std::nullopt;
114 if (!s)
115 return err(money_errc::overflow);
116 n = *s;
117 }
118 return make(from, to, n, uint256::pow10(static_cast<unsigned>(places)));
119 }

126 [[nodiscard]] static constexpr result<rate, money_errc> from_decimal(const asset& from, const asset& to,
127 const decimal& d) noexcept
128 {
129 if (d.is_negative())
130 return err(money_errc::malformed);
131 const uint256 m = static_cast<std::uint64_t>(d.mantissa());
132 const int e = d.exponent();
133 if (e >= 0)
134 return make(from, to, m * uint256::pow10(static_cast<unsigned>(e)), 1);
135 return make(from, to, m, uint256::pow10(static_cast<unsigned>(-e)));
136 }

186 [[nodiscard]] constexpr result<money, money_errc> at_scale(const asset& target, unsigned extra) const noexcept
187 {
188 const auto finer = make_asset(target.code(), target.scale() + extra);

214 [[nodiscard]] constexpr result<conversion, money_errc> convert(const money& amount, const rate& r, rounding mode) noexcept
215 {
216 if (amount.asset_of() != r.from())
217 return err(money_errc::asset_mismatch);
218 const auto n = uint256::checked_mul(r.numerator(), uint256::pow10(r.to().scale()));
219 const auto d = uint256::checked_mul(r.denominator(), uint256::pow10(r.from().scale()));
220 if (!n || !d)
221 return err(money_errc::overflow);
222 const auto qr = detail::mul_div(amount.magnitude(), *n, *d);
223 if (!qr)
224 return err(money_errc::overflow);
225 auto [q, rem] = *qr;
226 residual left{amount.is_negative(), rem, *d};
227 if (detail::rounds_away(q, rem, *d, amount.is_negative(), mode)) {
228 const auto up = uint256::checked_add(q, 1);
229 if (!up)
230 return err(money_errc::overflow);
231 q = *up;
232 left = residual{!amount.is_negative(), *d - rem, *d}; // the rounded value overshoots by (d - r) / d
233 }
234 if (left.numerator.is_zero())
235 left.negative = false;
236 return ok(conversion{money::from_units(r.to(), q, amount.is_negative()), left});
237 }
