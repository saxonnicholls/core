60 // The integers std::cmp_equal takes: not bool, not the character types
61 template <typename X>
62 concept comparable_integer = std::integral<X> && !std::same_as<X, bool> && !std::same_as<X, char> && !std::same_as<X, wchar_t> &&
63 !std::same_as<X, char8_t> && !std::same_as<X, char16_t> && !std::same_as<X, char32_t>;
64
65 // Equality of two coordinates: std::cmp_equal for two integers (exact across signedness, no warning), == otherwise
66 template <comparable_integer X, comparable_integer Y>
67 constexpr bool same(X x, Y y) noexcept
68 {
69 return std::cmp_equal(x, y);
70 }
71 template <typename X, typename Y>
72 requires(!(comparable_integer<X> && comparable_integer<Y>))
73 constexpr bool same(const X& x, const Y& y) noexcept(noexcept(x == y))
74 {
75 return x == y;
76 }

106 struct jaccard {
107 static constexpr bool is_metric = true;
110 template <bit_vector A, bit_vector B>
111 requires compatible_points<A, B>
112 [[nodiscard]] constexpr double operator()(const A& a, const B& b) const noexcept(noexcept(coordinate(a, 0)) && noexcept(coordinate(b, 0)))
113 {
114 using W = std::common_type_t<value_t<A>, value_t<B>>;
115 std::size_t both = 0, either = 0;
116 const std::size_t n = dimension(a);
117 for (std::size_t i = 0; i < n; ++i) {
118 const W x = static_cast<W>(coordinate(a, i)), y = static_cast<W>(coordinate(b, i));
119 both += static_cast<std::size_t>(std::popcount(static_cast<W>(x & y)));
120 either += static_cast<std::size_t>(std::popcount(static_cast<W>(x | y)));
121 }
122 return either == 0 ? 0.0 : 1.0 - static_cast<double>(both) / static_cast<double>(either);
123 }
124 };
