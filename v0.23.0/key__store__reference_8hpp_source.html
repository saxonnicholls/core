108 capability_result<key_info> create(const key_spec& spec)
109 {
110 if (auto e = take_injected())
111 return err(*e);
112 if (spec.algorithm != signature_algorithm::hs256)
113 return fail(capability_errc::unsupported, {}, "the reference holds HS256 keys only");
114 auto label = make_id<short_text>(spec.label);
115 if (!label)
116 return err(label.unwrap_err());
117 const auto id = *provider_id::make("key-" + std::to_string(++next_id_));
118 entry e{spec.algorithm, *label, {}};
119 e.versions.push_back(make_version(1));
120 const auto [it, inserted] = keys_.emplace(std::string(id.view()), std::move(e));
121 static_cast<void>(inserted);
122 return ok(describe(it->first, it->second, it->second.versions.back()));
123 }

128 capability_result<key_info> rotate(std::string_view id)
129 {
130 if (auto e = take_injected())
131 return err(*e);
132 const auto it = keys_.find(id);
133 if (it == keys_.end())
134 return fail(capability_errc::not_found);
135 auto& versions = it->second.versions;
136 if (versions.back().state != key_state::active)
137 return fail(capability_errc::conflict, {}, "a retired key does not rotate");
138 versions.push_back(make_version(static_cast<std::uint32_t>(versions.size() + 1)));
139 versions[versions.size() - 2].state = key_state::retired;
140 return ok(describe(it->first, it->second, versions.back()));
141 }

146 capability_result<void> retire(std::string_view id)
147 {
148 if (auto e = take_injected())
149 return err(*e);
150 const auto it = keys_.find(id);
151 if (it == keys_.end())
152 return fail(capability_errc::not_found);
153 for (auto& v : it->second.versions)
154 v.state = key_state::retired;
155 return ok();
156 }

161 capability_result<void> destroy(std::string_view id)

197 capability_result<signature> sign(const sign_request& r)
198 {
199 if (auto e = take_injected())
200 return err(*e);
201 if (r.kind == payload_kind::digest)
202 return fail(capability_errc::unsupported, {}, "HS256 hashes for itself: no digests");
203 const auto it = keys_.find(r.key_id);
204 if (it == keys_.end())
205 return fail(capability_errc::not_found);
206 const version& v = it->second.versions.back();
207 if (v.state != key_state::active)
208 return fail(capability_errc::conflict, {}, "the key is retired");
209 return ok(signature{signature_algorithm::hs256, *provider_id::make(it->first), v.number, mac(v, r.payload)});
210 }

215 [[nodiscard]] capability_result<public_key> public_key_of(const key_ref& k) const
216 {
217 const auto it = keys_.find(k.key_id);
218 if (it == keys_.end() || k.version > it->second.versions.size())
219 return fail(capability_errc::not_found);
220 return fail(capability_errc::unsupported, {}, "HS256 has no public key");
221 }
222
227 [[nodiscard]] bool verify(const signature& s, std::span<const std::byte> payload) const
228 {
229 const auto it = keys_.find(s.key_id.view());
230 if (it == keys_.end() || s.algorithm != signature_algorithm::hs256 || s.key_version == 0 ||
231 s.key_version > it->second.versions.size())
232 return false;
233 const auto expected = mac(it->second.versions[s.key_version - 1], payload);
234 return constant_time_equal(std::span<const std::byte>(expected), std::span<const std::byte>(s.bytes));
235 }

267 version make_version(std::uint32_t number)
268 {
269 std::string bytes(key_bytes, '\0');
270 random_(std::as_writable_bytes(std::span<char>(bytes.data(), bytes.size())));
271 return version{number, key_state::active, clock_(), secret<std::string>{std::move(bytes)}};
272 }

274 static std::vector<std::byte> mac(const version& v, std::span<const std::byte> payload)
275 {
276 const auto tag = hmac<sha256>::mac(v.material.reveal(), payload);
277 return {tag.bytes.begin(), tag.bytes.end()};
278 }

280 static key_info describe(const std::string& id, const entry& e, const version& v)

282 return key_info{*provider_id::make(id), v.number, e.algorithm, v.state, e.label, v.created_at};
283 }
