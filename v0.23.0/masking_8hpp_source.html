180 // Python's round(): to nearest, halves to even
181 constexpr double round_even(double x) noexcept
182 {
183 const double f = math::floor(x), r = x - f;
184 if (r > 0.5)
185 return f + 1;
186 if (r < 0.5)
187 return f;
188 return math::floor(f / 2) * 2 == f ? f : f + 1;
189 }

203 constexpr box place(const token_grid& g, block_extent e, block_rule rule, stats::wyrand& rng) noexcept
204 {
205 const std::size_t extra = rule == block_rule::vjepa ? 1 : 0;
206 const std::size_t t0 = e.frames < g.frames ? std::size_t(stats::uniform_index(rng, g.frames - e.frames + 1)) : 0;
207 const std::size_t y0 = std::size_t(stats::uniform_index(rng, g.height - e.height + extra));
208 const std::size_t x0 = std::size_t(stats::uniform_index(rng, g.width - e.width + extra));
209 return {rule == block_rule::vjepa ? t0 : 0, y0, x0, e};
210 }

212 // The sorted tokens of b that no box of others[0, n) contains
213 template <typename Out>
214 constexpr std::size_t write_box(const token_grid& g, const box& b, std::span<const box> others, Out out) noexcept
215 {
216 std::size_t count = 0;
217 for (std::size_t t = b.t0; t < b.t0 + b.e.frames; ++t)
218 for (std::size_t y = b.y0; y < b.y0 + b.e.height; ++y)
219 for (std::size_t x = b.x0; x < b.x0 + b.e.width; ++x) {
220 bool hidden = false;
221 for (const box& o : others)
222 hidden = hidden || o.contains(t, y, x);
223 if (!hidden) {
224 if (!out.empty())
225 out[count] = std::uint32_t((t * g.height + y) * g.width + x);
226 ++count;
227 }
228 }
229 return count;
230 }

239[[nodiscard]] constexpr block_extent sample_block_extent(const token_grid& g, const block_sampler& s, double u, block_rule rule) noexcept
240{
241 const double scale = s.scale_min + u * (s.scale_max - s.scale_min);
242 const double aspect = s.aspect_min + u * (s.aspect_max - s.aspect_min);
243 const double temporal = s.temporal_min + u * (s.temporal_max - s.temporal_min);
244 const double keep = double(std::size_t(double(g.height * g.width) * scale));
245 std::size_t h = std::size_t(detail::masking::round_even(math::sqrt(keep * aspect)));
246 std::size_t w = std::size_t(detail::masking::round_even(math::sqrt(keep / aspect)));
247 if (rule == block_rule::ijepa) {
248 while (h >= g.height && h > 1)
249 --h;
250 while (w >= g.width && w > 1)
251 --w;
252 } else {
253 h = std::min(h, g.height);
254 w = std::min(w, g.width);
255 }
256 const std::size_t f = std::size_t(double(g.frames) * temporal);
257 return {std::clamp<std::size_t>(f, 1, g.frames), std::max<std::size_t>(h, 1), std::max<std::size_t>(w, 1)};
258}

269constexpr mask_counts ijepa_masks(const ijepa_mask_config& c, std::uint64_t seed, std::uint64_t step, std::uint64_t sample,
270 std::span<std::uint32_t> context, std::span<std::uint32_t> targets,
271 std::span<std::size_t> target_counts) noexcept
272{
273 using detail::masking::box;
274 const std::size_t n = c.grid.tokens(), m = c.targets.blocks;
275 assert(c.grid.frames == 1 && m <= max_mask_blocks && c.context.blocks == 1 && context.size() >= n && targets.size() >= m * n
276 && target_counts.size() == m);
277 const std::uint64_t key = detail::masking::key(seed, step);
278 stats::wyrand sizes = stats::wyrand::stream(key, 0), rng = stats::wyrand::stream(key, sample + 1);
279 const block_extent pe = sample_block_extent(c.grid, c.targets, stats::uniform_real(sizes), block_rule::ijepa);
280 const block_extent ee = sample_block_extent(c.grid, c.context, stats::uniform_real(sizes), block_rule::ijepa);
281 assert(pe.tokens() > c.min_keep && ee.tokens() > c.min_keep);
282 std::array<box, max_mask_blocks> placed{};
283 mask_counts counts{};
284 for (std::size_t b = 0; b < m; ++b) {
285 placed[b] = detail::masking::place(c.grid, pe, block_rule::ijepa, rng); // a whole block always exceeds min_keep
286 target_counts[b] = detail::masking::write_box(c.grid, placed[b], {}, targets.subspan(b * n, n));
287 counts.targets += target_counts[b];
288 }
289 for (std::size_t tries = 0, timeout = 0;;) {
290 const box e = detail::masking::place(c.grid, ee, block_rule::ijepa, rng);
291 const std::size_t constrain = c.allow_overlap ? 0 : (m > tries ? m - tries : 0);
292 const auto others = std::span<const box>(placed).first(constrain);
293 if (detail::masking::write_box(c.grid, e, others, std::span<std::uint32_t>{}) > c.min_keep) {
294 counts.context = detail::masking::write_box(c.grid, e, others, context);
295 return counts;
296 }
297 if (++timeout % 20 == 0)
298 ++tries;
299 }
300}

310constexpr mask_counts vjepa_masks(const vjepa_mask_config& c, std::uint64_t seed, std::uint64_t step, std::uint64_t sample,
311 std::span<std::uint32_t> context, std::span<std::uint32_t> targets) noexcept
312{
313 using detail::masking::box;
314 [[maybe_unused]] const std::size_t n = c.grid.tokens();
315 const std::size_t m = c.targets.blocks;
316 assert(m <= max_mask_blocks && context.size() >= n && targets.size() >= n);
317 const std::uint64_t key = detail::masking::key(seed, step);
318 stats::wyrand sizes = stats::wyrand::stream(key, 0), rng = stats::wyrand::stream(key, sample + 1);
319 const block_extent e = sample_block_extent(c.grid, c.targets, stats::uniform_real(sizes), block_rule::vjepa);
320 std::array<box, max_mask_blocks> placed{};
321 for (;;) {
322 for (std::size_t b = 0; b < m; ++b)
323 placed[b] = detail::masking::place(c.grid, e, block_rule::vjepa, rng);
324 const auto boxes = std::span<const box>(placed).first(m);
325 mask_counts counts{};
326 for (std::size_t t = 0; t < c.grid.frames; ++t)
327 for (std::size_t y = 0; y < c.grid.height; ++y)
328 for (std::size_t x = 0; x < c.grid.width; ++x) {
329 bool hidden = false;
330 for (const box& b : boxes)
331 hidden = hidden || b.contains(t, y, x);
332 const auto token = std::uint32_t((t * c.grid.height + y) * c.grid.width + x);
333 if (hidden)
334 targets[counts.targets++] = token;
335 else
336 context[counts.context++] = token;
337 }
338 if (counts.context > 0)
339 return counts;
340 }

352constexpr mask_counts random_masks(std::size_t tokens, double ratio, std::uint64_t seed, std::uint64_t step, std::uint64_t sample,
353 std::span<std::uint32_t> context, std::span<std::uint32_t> targets) noexcept
354{
355 assert(ratio >= 0 && ratio <= 1 && context.size() >= tokens && targets.size() >= tokens);
356 stats::wyrand rng = stats::wyrand::stream(detail::masking::key(seed, step), sample + 1);
357 const auto all = context.first(tokens);
358 for (std::size_t i = 0; i < tokens; ++i)
359 all[i] = std::uint32_t(i);
360 stats::shuffle(all, rng);
361 const auto keep = std::size_t(double(tokens) * (1.0 - ratio));
362 std::copy(all.begin() + std::ptrdiff_t(keep), all.end(), targets.begin());
363 std::sort(all.begin(), all.begin() + std::ptrdiff_t(keep));
364 std::sort(targets.begin(), targets.begin() + std::ptrdiff_t(tokens - keep));
365 return {keep, tokens - keep};
366}

389constexpr void collate_indices(std::span<const std::uint32_t> lists, std::size_t stride, std::span<const std::size_t> counts,
390 std::size_t width, std::span<std::uint32_t> out, std::span<std::size_t> lengths) noexcept
391{
392 assert(out.size() == counts.size() * width && (lengths.empty() || lengths.size() == counts.size()));
393 for (std::size_t s = 0; s < counts.size(); ++s) {
394 assert(counts[s] > 0 && counts[s] <= stride && s * stride + counts[s] <= lists.size());
395 const std::size_t real = std::min(counts[s], width);
396 for (std::size_t j = 0; j < width; ++j)
397 out[s * width + j] = lists[s * stride + std::min(j, real - 1)];
398 if (!lengths.empty())
399 lengths[s] = real;
400 }
401}
