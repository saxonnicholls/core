79template <math::math_number T>
80constexpr T sinc(T x) noexcept
81{
82 if (x == T(0))
83 return T(1);
84 using C = compute_t<T>;
85 const C y = math::math_limits<C>::pi * C(x);
86 return static_cast<T>(math::sin(y) / y);
87}

97template <math::math_number T, typename W = hamming>
98constexpr void fir_design(std::span<T> h, fir_band band, std::type_identity_t<T> f1, std::type_identity_t<T> f2 = T(0), const W& window = W{}) noexcept
99{
100 using C = compute_t<T>;
101 const std::size_t taps = h.size();
102 [[maybe_unused]] const bool two_edges = band == fir_band::bandpass || band == fir_band::bandstop;
103 assert(f1 > T(0) && f1 < T(1) && (!two_edges || (f2 > f1 && f2 < T(1))));
104 assert((band != fir_band::highpass && band != fir_band::bandstop) || taps % 2 == 1);
105 // The pass bands, as scipy's hstack of [0] * pass_zero, cutoffs, [1] * pass_nyquist.
106 std::array<C, 4> edges{};
107 std::size_t count = 0;
108 const auto push = [&](C v) { edges[count++] = v; };
109 switch (band) {
110 case fir_band::lowpass: push(C(0)); push(C(f1)); break;
111 case fir_band::highpass: push(C(f1)); push(C(1)); break;
112 case fir_band::bandpass: push(C(f1)); push(C(f2)); break;
113 case fir_band::bandstop: push(C(0)); push(C(f1)); push(C(f2)); push(C(1)); break;
114 }
115 const C alpha = C(0.5) * C(static_cast<long long>(taps) - 1);
116 const C left0 = edges[0], right0 = edges[1];
117 const C scale_frequency = left0 == C(0) ? C(0) : (right0 == C(1) ? C(1) : C(0.5) * (left0 + right0));
118 // The taps are symmetric: compute the first half and mirror it (half the work, which matters in constant evaluation).
119 C s = C(0);
120 for (std::size_t i = 0; 2 * i < taps; ++i) {
121 const C m = C(static_cast<long long>(i)) - alpha;
122 C v = C(0);
123 for (std::size_t b = 0; b < count; b += 2)
124 v += edges[b + 1] * sinc(edges[b + 1] * m) - (edges[b] == C(0) ? C(0) : edges[b] * sinc(edges[b] * m));
125 v *= C(window_value<T>(window, i, taps, window_symmetry::symmetric));
126 h[i] = h[taps - 1 - i] = static_cast<T>(v);
127 const C c = scale_frequency == C(0) ? C(1) : math::cos(math::math_limits<C>::pi * m * scale_frequency);
128 s += (2 * i + 1 == taps ? C(1) : C(2)) * v * c;
129 }
130 for (auto& v : h)
131 v = static_cast<T>(C(v) / s);
132}

143template <math::math_number T, std::size_t N, typename W = hamming>

145{
146 std::array<T, N> h{};

156template <math::math_number T>
157constexpr complex<T> fir_response(std::span<const T> h, std::type_identity_t<T> omega) noexcept
158{
159 complex<T> acc{};
160 for (std::size_t k = 0; k < h.size(); ++k) {
161 const T a = -omega * T(static_cast<long long>(k));
162 acc += complex<T>{math::cos(a), math::sin(a)} * h[k];
163 }
164 return acc;
165}

172template <math::math_number T>
173constexpr void fir_filter(std::span<const T> h, std::span<const std::type_identity_t<T>> x, std::span<std::type_identity_t<T>> y) noexcept
174{
175 assert(y.size() == x.size());
176 for (std::size_t n = 0; n < x.size(); ++n) {
177 T acc = T(0);
178 const std::size_t kmax = n + 1 < h.size() ? n + 1 : h.size();
179 for (std::size_t k = 0; k < kmax; ++k)
180 acc += h[k] * x[n - k];
181 y[n] = acc;
182 }
183}
