227 template <transformer_config C>
228 struct dims
229 {
230 static_assert(valid(C), "snicholls::ml::latent_attention: the transformer_config is not valid");
231 static_assert(C.latent(), "snicholls::ml::latent_attention: the transformer_config has no kv_lora_rank");
232 static constexpr std::size_t h = C.n_heads, dqk = C.head_dim, dn = C.qk_nope_dim(), dr = C.qk_rope_dim,
233 dv = C.v_head_dim, dc = C.kv_lora_rank, dq = C.q_lora_rank, d = C.d_model,
234 kvw = dn + dv, qw = h * dqk, kvn = h * kvw, yw = h * dv, cw = dc + dr;
235 };

243 // The keys token t of a sequence attends to: their count, and the position of key j
244 template <transformer_config C>
245 struct keys
246 {
247 key_range r;
248 std::span<const std::uint32_t> sel;
249 [[nodiscard]] constexpr std::size_t size() const
250 {
251 return C.sparse_top_k == 0 || sel.empty() ? r.size() : std::min(sel.size(), r.size());
252 }
253 [[nodiscard]] constexpr std::size_t operator[](std::size_t j) const
254 {
255 if (C.sparse_top_k == 0 || sel.empty())
256 return r.first + j;
257 assert(sel[j] >= r.first && sel[j] < r.last);
258 return sel[j];
259 }
260 };

262 template <transformer_config C, typename T>
263 constexpr keys<C> keys_of(const latent_attention_activations<T>& act, std::size_t row, std::size_t t)
264 {
265 const key_range r = attention_window(t, C.window);
266 if constexpr (C.sparse_top_k > 0) {
267 assert(act.selected.size() >= (row + 1) * C.sparse_top_k);
268 return {r, act.selected.subspan(row * C.sparse_top_k, C.sparse_top_k)};
269 } else {
270 return {r, {}};
271 }

284[[nodiscard]] constexpr std::size_t latent_attention_parameter_count(const transformer_config& c) noexcept
285{
286 const std::size_t h = c.n_heads, dq = c.q_lora_rank, d = c.d_model, dc = c.kv_lora_rank;
287 return dq * d + dq + h * c.head_dim * (dq > 0 ? dq : d) + (dc + c.qk_rope_dim) * d + dc + h * (c.qk_nope_dim() + c.v_head_dim) * dc
288 + d * h * c.v_head_dim + (c.attention_sinks ? h : 0);
289}

294[[nodiscard]] constexpr std::size_t latent_attention_activation_values(const transformer_config& c, std::size_t rows) noexcept
295{
296 const std::size_t h = c.n_heads, dc = c.kv_lora_rank, dr = c.qk_rope_dim;
297 return rows * (2 * c.q_lora_rank + h * c.head_dim + (dc + dr) + dc + dr + h * (c.qk_nope_dim() + c.v_head_dim) + h * c.v_head_dim + h);
298}

303[[nodiscard]] constexpr std::size_t latent_attention_gradient_work(const transformer_config& c, std::size_t rows) noexcept
304{
305 const std::size_t h = c.n_heads, dc = c.kv_lora_rank, dr = c.qk_rope_dim;
306 return rows * (h * c.v_head_dim + h * c.head_dim + h * (c.qk_nope_dim() + c.v_head_dim) + dr + dc + (dc + dr) + 2 * c.q_lora_rank);
307}

314template <transformer_config C, math::math_number T>
315 requires(C.latent())
316constexpr latent_attention_weights<dense_weights<T>, T> latent_attention_weights_in(std::span<const T> params)
317{
318 using D = detail::latent_attention::dims<C>;
319 assert(params.size() == latent_attention_parameter_count(C));
320 constexpr std::size_t qin = D::dq > 0 ? D::dq : D::d;
321 detail::cutter<const T> c{params};
322 latent_attention_weights<dense_weights<T>, T> w{};
323 w.q_down = {c.take(D::dq * D::d), D::dq, D::dq > 0 ? D::d : 0};
324 w.q_norm = c.take(D::dq);
325 w.q_up = {c.take(D::qw * qin), D::qw, qin};
326 w.kv_down = {c.take(D::cw * D::d), D::cw, D::d};
327 w.kv_norm = c.take(D::dc);
328 w.kv_up = {c.take(D::kvn * D::dc), D::kvn, D::dc};
329 w.o = {c.take(D::d * D::yw), D::d, D::yw};
330 w.sinks = c.take(C.attention_sinks ? D::h : 0);
331 return w;
332}

340template <transformer_config C, math::math_number T>
341 requires(C.latent())
342constexpr latent_attention_activations<T> latent_attention_activations_in(std::span<T> values, std::size_t rows,
343 std::span<const std::uint32_t> selected = {})
344{
345 using D = detail::latent_attention::dims<C>;
346 assert(values.size() == latent_attention_activation_values(C, rows));
347 detail::cutter<T> v{values};
348 latent_attention_activations<T> a{};
349 a.cq_raw = v.take(rows * D::dq);
350 a.cq = v.take(rows * D::dq);
351 a.q = v.take(rows * D::qw);
352 a.ckv_raw = v.take(rows * D::cw);
353 a.ckv = v.take(rows * D::dc);
354 a.k_rope = v.take(rows * D::dr);
355 a.kv = v.take(rows * D::kvn);
356 a.y = v.take(rows * D::yw);
357 a.lse = v.take(rows * D::h);
358 a.selected = selected;
359 return a;
360}

369template <transformer_config C, math::math_number T>
370 requires(C.latent())
371constexpr latent_attention_gradients<T> latent_attention_gradients_in(std::span<T> grads, std::span<T> work, std::size_t rows)
372{
373 using D = detail::latent_attention::dims<C>;
374 assert(grads.size() == latent_attention_parameter_count(C) && work.size() == latent_attention_gradient_work(C, rows));
375 constexpr std::size_t qin = D::dq > 0 ? D::dq : D::d;
376 detail::cutter<T> c{grads}, w{work};
377 latent_attention_gradients<T> g{};
378 g.q_down = c.take(D::dq * D::d);
379 g.q_norm = c.take(D::dq);
380 g.q_up = c.take(D::qw * qin);
381 g.kv_down = c.take(D::cw * D::d);
382 g.kv_norm = c.take(D::dc);
383 g.kv_up = c.take(D::kvn * D::dc);
384 g.o = c.take(D::d * D::yw);
385 g.sinks = c.take(C.attention_sinks ? D::h : 0);
386 g.y = w.take(rows * D::yw);
387 g.q = w.take(rows * D::qw);
388 g.kv = w.take(rows * D::kvn);
389 g.k_rope = w.take(rows * D::dr);
390 g.ckv = w.take(rows * D::dc);
391 g.ckv_raw = w.take(rows * D::cw);
392 g.cq = w.take(rows * D::dq);
393 g.cq_raw = w.take(rows * D::dq);
394 return g;
395}

406template <transformer_config C, typename B, typename W, math::math_number T>
407constexpr void latent_attention_norm_rows(const B&, const latent_attention_weights<W, T>& w, const latent_attention_activations<T>& act,
408 std::size_t rows)
409{
410 using D = detail::latent_attention::dims<C>;
411 constexpr T eps = norm_eps_v<C, T>;
412 for (std::size_t r = 0; r < rows; ++r) {
413 if constexpr (D::dq > 0)
414 rms_norm(act.cq_raw.subspan(r * D::dq, D::dq), w.q_norm, act.cq.subspan(r * D::dq, D::dq), eps);
415 rms_norm(act.ckv_raw.subspan(r * D::cw, D::dc), w.kv_norm, act.ckv.subspan(r * D::dc, D::dc), eps);
416 }
417}

423template <transformer_config C, typename B, math::math_number T, typename R>
424constexpr void latent_attention_rope_rows(const B&, const latent_attention_activations<T>& act, std::size_t n, const R& rope_src,
425 std::size_t sequences = 1)
426{
427 using D = detail::latent_attention::dims<C>;
428 for (std::size_t s = 0; s < sequences; ++s)
429 for (std::size_t t = 0; t < n; ++t) {
430 const std::size_t row = s * n + t;
431 const auto kr = act.k_rope.subspan(row * D::dr, D::dr);
432 std::copy_n(act.ckv_raw.begin() + std::ptrdiff_t(row * D::cw + D::dc), D::dr, kr.begin());
433 rope(kr, rope_src.cos_at(t), rope_src.sin_at(t), C.layout());
434 for (std::size_t h = 0; h < D::h; ++h)
435 rope(act.q.subspan(row * D::qw + h * D::dqk + D::dn, D::dr), rope_src.cos_at(t), rope_src.sin_at(t), C.layout());
436 }

442template <transformer_config C, typename B, typename W, math::math_number T>
443constexpr void latent_attention_attend_rows(const B&, const latent_attention_weights<W, T>& w, const latent_attention_activations<T>& act,
444 std::size_t n, std::size_t sequences = 1)
445{
446 using D = detail::latent_attention::dims<C>;
447 const T scale = latent_attention_scale<C, T>();
448 for (std::size_t s = 0; s < sequences; ++s)
449 for (std::size_t t = 0; t < n; ++t) {
450 const std::size_t row = s * n + t, base = s * n;
451 const auto keys = detail::latent_attention::keys_of<C>(act, row, t);
452 for (std::size_t h = 0; h < D::h; ++h) {
453 const auto q = std::span<const T>(act.q).subspan(row * D::qw + h * D::dqk, D::dqk);
454 act.lse[row * D::h + h] = detail::attention_sink::forward(
455 keys.size(),
456 [&](std::size_t j) {
457 const std::size_t k = base + keys[j];
458 T acc = T(0);
459 for (std::size_t i = 0; i < D::dn; ++i)
460 acc += q[i] * act.kv[k * D::kvn + h * D::kvw + i];
461 for (std::size_t i = 0; i < D::dr; ++i)
462 acc += q[D::dn + i] * act.k_rope[k * D::dr + i];
463 return scale * acc;
464 },
465 [&](std::size_t j) { return std::span<const T>(act.kv).subspan((base + keys[j]) * D::kvn + h * D::kvw + D::dn, D::dv); },
466 act.y.subspan(row * D::yw + h * D::dv, D::dv), detail::latent_attention::sink_of(w.sinks, h, C.attention_sinks));
467 }
468 }

475template <transformer_config C, typename B, typename W, math::math_number T>
476constexpr void latent_attention_attend_rows_backward(const B&, const latent_attention_weights<W, T>& w,
477 const latent_attention_activations<T>& act, std::type_identity_t<std::span<const T>> dy,
478 const latent_attention_gradients<T>& grads, std::size_t n, std::size_t sequences = 1)
479{
480 using D = detail::latent_attention::dims<C>;
481 const T scale = latent_attention_scale<C, T>();
482 for (std::size_t s = 0; s < sequences; ++s)
483 for (std::size_t t = 0; t < n; ++t) {
484 const std::size_t row = s * n + t, base = s * n;
485 const auto keys = detail::latent_attention::keys_of<C>(act, row, t);
486 for (std::size_t h = 0; h < D::h; ++h) {
487 const auto q = std::span<const T>(act.q).subspan(row * D::qw + h * D::dqk, D::dqk);
488 const auto gq = grads.q.subspan(row * D::qw + h * D::dqk, D::dqk);
489 const auto g = dy.subspan(row * D::yw + h * D::dv, D::dv);
490 std::fill(gq.begin(), gq.end(), T(0));
491 const T sink = detail::latent_attention::sink_of(w.sinks, h, C.attention_sinks);
492 const T dsink = detail::attention_sink::backward(
493 keys.size(),
494 [&](std::size_t j) {
495 const std::size_t k = base + keys[j];
496 T acc = T(0);
497 for (std::size_t i = 0; i < D::dn; ++i)
498 acc += q[i] * act.kv[k * D::kvn + h * D::kvw + i];
499 for (std::size_t i = 0; i < D::dr; ++i)
500 acc += q[D::dn + i] * act.k_rope[k * D::dr + i];
501 return scale * acc;
502 },
503 [&](std::size_t j) { return std::span<const T>(act.kv).subspan((base + keys[j]) * D::kvn + h * D::kvw + D::dn, D::dv); },
504 std::span<const T>(act.y).subspan(row * D::yw + h * D::dv, D::dv), g, act.lse[row * D::h + h], sink,
505 [&](std::size_t j, T p, T ds) {
506 const std::size_t k = base + keys[j], at = k * D::kvn + h * D::kvw;
507 for (std::size_t i = 0; i < D::dv; ++i)
508 grads.kv[at + D::dn + i] += p * g[i];
509 for (std::size_t i = 0; i < D::dn; ++i) {
510 gq[i] += scale * ds * act.kv[at + i];
511 grads.kv[at + i] += scale * ds * q[i];
512 }
513 for (std::size_t i = 0; i < D::dr; ++i) {
514 gq[D::dn + i] += scale * ds * act.k_rope[k * D::dr + i];
515 grads.k_rope[k * D::dr + i] += scale * ds * q[D::dn + i];
516 }
517 });
518 if constexpr (C.attention_sinks)
519 grads.sinks[h] += dsink;
520 }
521 }
522}

528constexpr void latent_attention_rope_rows_backward(const B&, const latent_attention_gradients<T>& grads, std::size_t n, const R& rope_src,

538 C.layout());
539 }

546template <transformer_config C, typename B, typename W, math::math_number T>
547constexpr void latent_attention_norm_rows_backward(const B&, const latent_attention_weights<W, T>& w,
548 const latent_attention_activations<T>& act, const latent_attention_gradients<T>& grads,
549 std::size_t rows)
550{
551 using D = detail::latent_attention::dims<C>;
552 constexpr T eps = norm_eps_v<C, T>;
553 std::array<T, D::dc> scratch{};
554 for (std::size_t r = 0; r < rows; ++r) {
555 if constexpr (D::dq > 0) {
556 std::array<T, D::dq> sq{};
557 const auto x = std::span<const T>(act.cq_raw).subspan(r * D::dq, D::dq);
558 const T rstd = rms_norm(x, w.q_norm, sq, eps);
559 rms_norm_backward(std::span<const T>(grads.cq).subspan(r * D::dq, D::dq), x, w.q_norm, rstd,
560 grads.cq_raw.subspan(r * D::dq, D::dq), grads.q_norm);
561 }
562 const auto x = std::span<const T>(act.ckv_raw).subspan(r * D::cw, D::dc);
563 const T rstd = rms_norm(x, w.kv_norm, scratch, eps);
564 rms_norm_backward(std::span<const T>(grads.ckv).subspan(r * D::dc, D::dc), x, w.kv_norm, rstd,
565 grads.ckv_raw.subspan(r * D::cw, D::dc), grads.kv_norm);
566 std::copy_n(grads.k_rope.begin() + std::ptrdiff_t(r * D::dr), D::dr, grads.ckv_raw.begin() + std::ptrdiff_t(r * D::cw + D::dc));
567 }

582template <transformer_config C, typename B, math::math_number T, typename R>
583 requires linalg::blas_backend<B, T> && rope_source<R>
584constexpr void latent_attention_forward_sequence(const B& backend, const latent_attention_weights<dense_weights<T>, T>& w,
585 std::type_identity_t<std::span<const T>> xs, const R& rope_src,
586 const latent_attention_activations<T>& act, std::type_identity_t<std::span<T>> out,
587 std::size_t sequences = 1)
588{
589 using D = detail::latent_attention::dims<C>;
590 const std::size_t rows = xs.size() / D::d, n = rows / sequences;
591 assert(sequences > 0 && xs.size() == n * sequences * D::d && out.size() == xs.size());
592 assert(act.q.size() == rows * D::qw && act.ckv_raw.size() == rows * D::cw && act.ckv.size() == rows * D::dc
593 && act.k_rope.size() == rows * D::dr && act.kv.size() == rows * D::kvn && act.y.size() == rows * D::yw
594 && act.lse.size() == rows * D::h && (D::dq == 0 || (act.cq_raw.size() == rows * D::dq && act.cq.size() == rows * D::dq)));
595 assert(!C.attention_sinks || w.sinks.size() == D::h);
596 if constexpr (D::dq > 0)
597 linear_sequence(backend, w.q_down, xs, act.cq_raw);
598 linear_sequence(backend, w.kv_down, xs, act.ckv_raw);
599 latent_attention_norm_rows<C>(backend, w, act, rows);
600 if constexpr (D::dq > 0)
601 linear_sequence(backend, w.q_up, std::span<const T>(act.cq), act.q);
602 else
603 linear_sequence(backend, w.q_up, xs, act.q);
604 latent_attention_rope_rows<C>(backend, act, n, rope_src, sequences);
605 linear_sequence(backend, w.kv_up, std::span<const T>(act.ckv), act.kv);
606 latent_attention_attend_rows<C>(backend, w, act, n, sequences);
607 linear_sequence(backend, w.o, std::span<const T>(act.y), out);
608}

625template <transformer_config C, typename B, math::math_number T, typename R>
626 requires linalg::blas_backend<B, T> && rope_source<R>
627constexpr void latent_attention_backward(const B& backend, const latent_attention_weights<dense_weights<T>, T>& w,
628 std::type_identity_t<std::span<const T>> xs, const R& rope_src,
629 const latent_attention_activations<T>& act, std::type_identity_t<std::span<const T>> dout,
630 std::type_identity_t<std::span<T>> dxs, const latent_attention_gradients<T>& grads,
631 std::size_t sequences = 1)
632{
633 using D = detail::latent_attention::dims<C>;
634 const std::size_t rows = xs.size() / D::d, n = rows / sequences;
635 assert(sequences > 0 && rows == n * sequences && dout.size() == xs.size() && (dxs.empty() || dxs.size() == xs.size()));
636 assert(grads.y.size() == rows * D::yw && grads.q.size() == rows * D::qw && grads.kv.size() == rows * D::kvn
637 && grads.k_rope.size() == rows * D::dr && grads.ckv.size() == rows * D::dc && grads.ckv_raw.size() == rows * D::cw
638 && (D::dq == 0 || (grads.cq.size() == rows * D::dq && grads.cq_raw.size() == rows * D::dq)));
639 fill_rows(backend, grads.y, T(0));
640 linear_sequence_backward(backend, w.o, std::span<const T>(act.y), dout, grads.y, grads.o);
641 fill_rows(backend, grads.kv, T(0));
642 fill_rows(backend, grads.k_rope, T(0));
643 latent_attention_attend_rows_backward<C>(backend, w, act, std::span<const T>(grads.y), grads, n, sequences);
644 latent_attention_rope_rows_backward<C>(backend, grads, n, rope_src, sequences);
645 fill_rows(backend, grads.ckv, T(0));
646 linear_sequence_backward(backend, w.kv_up, std::span<const T>(act.ckv), std::span<const T>(grads.kv), grads.ckv, grads.kv_up);

648 fill_rows(backend, grads.cq, T(0));
649 linear_sequence_backward(backend, w.q_up, std::span<const T>(act.cq), std::span<const T>(grads.q), grads.cq, grads.q_up);
650 } else {
651 linear_sequence_backward(backend, w.q_up, xs, std::span<const T>(grads.q), dxs, grads.q_up);
652 }
653 latent_attention_norm_rows_backward<C>(backend, w, act, grads, rows);
654 linear_sequence_backward(backend, w.kv_down, xs, std::span<const T>(grads.ckv_raw), dxs, grads.kv_down);
655 if constexpr (D::dq > 0)
656 linear_sequence_backward(backend, w.q_down, xs, std::span<const T>(grads.cq_raw), dxs, grads.q_down);

672template <transformer_config C, math::math_number T = float>
673class latent_kv_cache_view
674{
675 static_assert(valid(C) && C.latent(), "snicholls::ml::latent_kv_cache_view: the transformer_config is not a valid latent one");
676
677public:
679 static constexpr std::size_t width = C.kv_lora_rank + C.qk_rope_dim;
680
684 [[nodiscard]] static constexpr std::size_t required_size(std::size_t layers, std::size_t capacity) noexcept
685 {
686 return layers * capacity * width;
687 }
688
690 constexpr latent_kv_cache_view() noexcept = default;
691
696 constexpr latent_kv_cache_view(std::span<T> rows, std::size_t layers, std::size_t capacity) noexcept
697 : rows_(rows), layers_(layers), capacity_(capacity)
698 {
699 assert(rows.size() == required_size(layers, capacity));
700 assert(C.window == 0 || capacity == 0 || capacity >= C.window);
701 }
702
705 template <row Row>
706 requires std::same_as<row_value_t<Row>, T>
707 constexpr void append(std::size_t layer, std::size_t pos, const Row& r) const
708 {
709 const auto x = as_row(r);
710 assert(layer < layers_ && capacity_ > 0 && x.size() == width);
711 std::copy(x.begin(), x.end(), rows_.begin() + std::ptrdiff_t(offset(layer, pos)));
712 }
713
717 [[nodiscard]] constexpr std::span<const T> row(std::size_t layer, std::size_t pos) const
718 {
719 assert(layer < layers_ && capacity_ > 0);
720 return std::span<const T>(rows_).subspan(offset(layer, pos), width);
721 }
722
724 [[nodiscard]] constexpr std::size_t layers() const noexcept { return layers_; }
726 [[nodiscard]] constexpr std::size_t capacity() const noexcept { return capacity_; }
727
728private:
729 constexpr std::size_t offset(std::size_t layer, std::size_t pos) const noexcept { return (layer * capacity_ + pos % capacity_) * width; }
730
731 std::span<T> rows_{};
732 std::size_t layers_ = 0;
733 std::size_t capacity_ = 0;
734};

741{
742public:

746
747private:

762template <transformer_config C, typename S, math::math_number T, row X, latent_kv_store<T> Cache, typename R, mutable_row Out>
763 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T> && rope_source<R>
764constexpr void latent_attention_forward(const latent_attention_weights<dense_weights<S>, T>& w, const X& x, std::size_t pos,
765 const Cache& cache, std::size_t layer, const R& rope_src, Out&& out,
766 std::span<const std::uint32_t> selected = {})
767{
768 using D = detail::latent_attention::dims<C>;
769 constexpr T eps = norm_eps_v<C, T>;
770 const auto xs = as_row(x);
771 assert(xs.size() == D::d && w.kv_up.rows == D::kvn && w.kv_up.cols == D::dc && (!C.attention_sinks || w.sinks.size() == D::h));
772 std::array<T, D::qw> q{};
773 if constexpr (D::dq > 0) {
774 std::array<T, D::dq> cq{};
775 linear(w.q_down, xs, cq);
776 rms_norm(cq, w.q_norm, cq, eps);
777 linear(w.q_up, cq, q);
778 } else {
779 linear(w.q_up, xs, q);
780 }
781 std::array<T, D::cw> c{};
782 linear(w.kv_down, xs, c);
783 const auto latent = std::span<T>(c).first(D::dc);
784 rms_norm(latent, w.kv_norm, latent, eps);
785 rope(std::span<T>(c).subspan(D::dc, D::dr), rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
786 cache.append(layer, pos, c);
787 const auto keys = detail::latent_attention::keys<C>{attention_window(pos, C.window), selected};
788 const T scale = latent_attention_scale<C, T>();
789 std::array<T, D::yw> y{};
790 std::array<T, D::cw> qa{};
791 std::array<T, D::dc> o{};
792 for (std::size_t h = 0; h < D::h; ++h) {
793 const auto qh = std::span<T>(q).subspan(h * D::dqk, D::dqk);
794 rope(qh.subspan(D::dn, D::dr), rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
795 for (std::size_t k = 0; k < D::dc; ++k) { // (W^UK_h)ᵀ q^C_h
796 T acc = T(0);
797 for (std::size_t i = 0; i < D::dn; ++i) {
798 const T p = detail::linear::widen<T>(w.kv_up.row(h * D::kvw + i)[k]) * qh[i];
799 acc += p;
800 }
801 qa[k] = acc;
802 }
803 std::copy_n(qh.begin() + std::ptrdiff_t(D::dn), D::dr, qa.begin() + std::ptrdiff_t(D::dc));
804 detail::attention_sink::forward(
805 keys.size(), [&](std::size_t j) { return scale * detail::attention::dot(std::span<const T>(qa), cache.row(layer, keys[j])); },
806 [&](std::size_t j) { return cache.row(layer, keys[j]).first(D::dc); }, std::span<T>(o),
807 detail::latent_attention::sink_of(w.sinks, h, C.attention_sinks));
808 for (std::size_t v = 0; v < D::dv; ++v) { // W^UV_h ō_h
809 const auto r = w.kv_up.row(h * D::kvw + D::dn + v);
810 T acc = T(0);
811 for (std::size_t k = 0; k < D::dc; ++k) {
812 const T p = detail::linear::widen<T>(r[k]) * o[k];
813 acc += p;
814 }
815 y[h * D::dv + v] = acc;
816 }
817 }
818 linear(w.o, y, out);
819}
