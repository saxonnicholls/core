143template <math::math_number T>
144constexpr void ordered_sum(std::span<const std::span<const T>> parts, std::span<T> out) noexcept(std::is_floating_point_v<T>)
145{
146 assert(!parts.empty());
147 for ([[maybe_unused]] const auto& part : parts)
148 assert(part.size() == out.size());
149 const auto first = parts.front();
150 for (std::size_t i = 0; i < out.size(); ++i)
151 out[i] = first[i];
152 for (const auto part : parts.subspan(1)) // no index into parts: GCC 15's -Warray-bounds misreads one
153 for (std::size_t i = 0; i < out.size(); ++i)
154 out[i] = out[i] + part[i];
155}

215constexpr void compress_bf16(std::span<const float> delta, std::span<float> residual, std::span<std::uint16_t> out) noexcept
216{
217 assert(residual.size() == delta.size() && out.size() == delta.size());
218 for (std::size_t i = 0; i < delta.size(); ++i) {
219 const float v = delta[i] + residual[i];
220 const bfloat16 b = to_bfloat16(v);
221 out[i] = b.bits;
222 const float back = to_float(b);
223 residual[i] = detail::diloco_::finite(v) && detail::diloco_::finite(back) ? v - back : 0.0f;
224 }
225}

235constexpr void ordered_sum_bf16(std::span<const std::span<const std::uint16_t>> parts, std::span<float> out) noexcept
236{
237 assert(!parts.empty());
238 for ([[maybe_unused]] const auto& part : parts)
239 assert(part.size() == out.size());
240 const auto first = parts.front();
241 for (std::size_t i = 0; i < out.size(); ++i)
242 out[i] = to_float(bfloat16::from_bits(first[i]));
243 for (const auto part : parts.subspan(1)) // no index into parts: GCC 15's -Warray-bounds misreads one
244 for (std::size_t i = 0; i < out.size(); ++i)
245 out[i] = out[i] + to_float(bfloat16::from_bits(part[i]));
246}

286template <math::math_number T>
287constexpr void diloco_outer_step(std::span<T> global, std::span<const T> mean_delta, std::span<T> momentum,
288 const diloco_config& c) noexcept(std::is_floating_point_v<T>)
289{
290 assert(mean_delta.size() == global.size() && momentum.size() == global.size());
291 const T mu = T(c.momentum);
292 const T lr = T(c.outer_lr);
293 for (std::size_t i = 0; i < global.size(); ++i) {
294 const T g = mean_delta[i];
295 const T decayed = mu * momentum[i];
296 momentum[i] = decayed + g;
297 T step = momentum[i];
298 if (c.nesterov) {
299 const T ahead = mu * momentum[i];
300 step = g + ahead;
301 }
302 const T change = lr * step;
303 global[i] = global[i] - change;
304 }
305}
