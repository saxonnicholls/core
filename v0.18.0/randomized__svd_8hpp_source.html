130 struct rademacher_sketch {
131 std::uint64_t seed = 0;
132
134 template <scalar T>
135 constexpr void operator()(std::span<T> omega, std::size_t rows, std::size_t cols) const
136 {
137 stats::wyrand g(seed);
138 std::uint64_t bits = 0;
139 for (std::size_t i = 0; i < rows * cols; ++i) {
140 if (i % 64 == 0)
141 bits = g();
142 omega[i] = (bits >> (i % 64)) & 1 ? T(1) : T(-1);
143 }
144 }
145 };

153 [[nodiscard]] constexpr std::size_t truncated_svd_work_size(std::size_t m, std::size_t n, std::size_t k,
154 const truncated_svd_options& options = {}) noexcept
155 {
156 const std::size_t l = k == 0 ? 0 : std::min({k + options.oversample, m, n});
157 return 2 * n * l + 2 * m * l + l * l + 2 * l;
158 }

169 template <typename B, typename T>
170 constexpr result<void, error> orthonormalise(const B& b, matrix_view<T, layout_right> y, std::span<T> tau,
171 matrix_view<T, layout_right> q)
172 {
173 if (auto f = qr_factor(b, y, tau); f.is_err())

191 template <typename B, typename EA, typename LA, scalar T, typename LU, typename LV, typename S = gaussian_sketch<>>
192 requires(std::same_as<detail::value_t<EA>, T> && blas_backend<B, T> && factorisation_backend<B, T> && decomposition_backend<B, T>
193 && sketch<S, T>)
194 constexpr result<void, error> truncated_svd(const B& b, const md_span<EA, 2, LA>& a, std::span<T> s, const md_span<T, 2, LU>& u,
195 const md_span<T, 2, LV>& v, std::span<T> work, const truncated_svd_options& options = {},
196 const S& sk = S{})
197 {
198 using right = layout_right;
199 const std::size_t m = a.extent(0), n = a.extent(1), k = s.size();
200 if (k > std::min(m, n) || u.extent(0) != m || u.extent(1) != k || v.extent(0) != n || v.extent(1) != k
201 || work.size() < truncated_svd_work_size(m, n, k, options))
202 return err(error::dimension_mismatch);
203 if (k == 0)
204 return ok();
205 const std::size_t l = std::min({k + options.oversample, m, n});
206 std::size_t at = 0;
207 auto take = [&](std::size_t size) {
208 const auto part = work.subspan(at, size);
209 at += size;
210 return part;
211 };
212 const auto n1 = take(n * l), n2 = take(n * l), y = take(m * l), q = take(m * l), vb = take(l * l), sl = take(l), tau = take(l);
213 const matrix_view<const T> av(a);
214 const matrix_view<T, right> n1v(n1.data(), {n, l}), n2v(n2.data(), {n, l}), yv(y.data(), {m, l}), qv(q.data(), {m, l}),
215 vbv(vb.data(), {l, l});
216 const matrix_view<const T, right> n1c(n1.data(), {n, l}), n2c(n2.data(), {n, l}), qc(q.data(), {m, l});
217 using detail::randomized::orthonormalise;
218 using detail::randomized::product;
219
220 // 1. Y = A Ω, Ω from the sketch
221 sk(n1, n, l);
222 product<B, T>(b, av, n1c, yv);
223
224 // 2. Power iterations: Y = A Q(Aᵀ Q(Y))
225 for (int it = 0; it < options.power_iterations; ++it) {
226 if (auto r = orthonormalise(b, yv, tau, qv); r.is_err())
227 return r;
228 product<B, T>(b, transposed(av), qc, n1v); // Z = Aᵀ Q, n x l
229 if (auto r = orthonormalise(b, n1v, tau, n2v); r.is_err())
230 return r;
231 product<B, T>(b, av, n2c, yv); // Y = A Q(Z)
232 }
233
234 // 3. Q, then 4. the exact SVD of Aᵀ Q = U_b S V_bᵀ (n x l, n >= l)
235 if (auto r = orthonormalise(b, yv, tau, qv); r.is_err())
236 return r;
237 product<B, T>(b, transposed(av), qc, n1v);
238 if (auto r = svd(b, n1v, sl, vbv); r.is_err())
239 return r;
240
241 // 5. U = Q V_b and V = U_b, the first k columns; s the first k values
242 product<B, T>(b, qc, matrix_view<const T>(vb.data(), {l, k}, {l, 1}), u);
243 for (std::size_t i = 0; i < n; ++i)
244 for (std::size_t j = 0; j < k; ++j)
245 v(i, j) = n1[i * l + j];
246 std::copy_n(sl.begin(), k, s.begin());
247 return ok();
248 }
