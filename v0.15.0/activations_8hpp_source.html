64 // sigmoid(v) and sigmoid(-v) = 1 - sigmoid(v), both from one e^-|v|, so neither cancels.
65 template <typename T>
66 struct sigmoid_pair
67 {
68 T s;
69 T one_minus_s;
70 };
71
72 template <typename T>
73 constexpr sigmoid_pair<T> sigmoid_both(T v)
74 {
75 const T e = math::exp(-math::abs(v));
76 const T r = T(1) / (T(1) + e);
77 if (v >= T(0))
78 return {r, e * r};
79 return {e * r, r};
80 }
81} // namespace detail::activations

87template <math::math_argument X>
88constexpr math::real_t<X> sigmoid(X x)
89{
90 using T = math::real_t<X>;
91 const T v = static_cast<T>(x);
92 if (v >= T(0))
93 return T(1) / (T(1) + math::exp(-v));
94 const T e = math::exp(v);
95 return e / (T(1) + e);
96}

124template <math::math_argument X>
125constexpr math::real_t<X> softplus(X x)
126{
127 using T = math::real_t<X>;
128 const T v = static_cast<T>(x);
129 if (math::is_nan(v))
130 return v;
131 const T positive = v > T(0) ? v : T(0);
132 return positive + math::log1p(math::exp(-math::abs(v)));
133}

150template <math::math_argument X>
151constexpr math::real_t<X> logit(X p)
152{
153 using T = math::real_t<X>;
154 const T v = static_cast<T>(p);
155 if (!(v >= T(0) && v <= T(1))) // NaN, or outside [0, 1]: no arithmetic on a NaN, which a constant expression refuses
156 return std::numeric_limits<T>::quiet_NaN();
157 return math::log(v) - math::log1p(-v);
158}

256template <math::math_argument X>
257constexpr math::real_t<X> gelu_derivative(X x)
258{
259 using T = math::real_t<X>;
260 using L = math::detail::limits<T>;
261 const T v = static_cast<T>(x);
262 const T cdf = T(0.5) * math::erfc(-v * T(L::inv_sqrt2));
263 const T pdf = math::exp(T(-0.5) * v * v) / T(L::sqrt_2pi);
264 return cdf + v * pdf;

269 // u = sqrt(2/pi) (x + 0.044715 x^3), GPT-2's approximation of GELU; returns 2u.
270 template <typename T>
271 constexpr T gelu_tanh_2u(T v)
272 {
273 using L = math::detail::limits<T>;
274 const T sqrt_2_over_pi = T(2) / T(L::sqrt_2pi);
275 return T(2) * sqrt_2_over_pi * (v + T(0.044715) * v * v * v);
276 }

295template <math::math_argument X>
296constexpr math::real_t<X> gelu_tanh_derivative(X x)
297{
298 using T = math::real_t<X>;
299 using L = math::detail::limits<T>;
300 const T v = static_cast<T>(x);
301 const auto [s, one_minus_s] = detail::activations::sigmoid_both(detail::activations::gelu_tanh_2u(v));
302 const T du = T(2) / T(L::sqrt_2pi) * (T(1) + T(3) * T(0.044715) * v * v);
303 return s + T(2) * v * s * one_minus_s * du;
304}
