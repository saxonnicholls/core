71template <row Table>
72constexpr std::span<const row_value_t<Table>> embedding_row(const Table& table, std::size_t id, std::size_t channels)
73{
74 const auto t = as_row(table);
75 assert((id + 1) * channels <= t.size());
76 return t.subspan(id * channels, channels);
77}

114template <row D, mutable_row DTable>
115 requires std::same_as<row_value_t<D>, row_value_t<DTable>>
116constexpr void embedding_backward(const D& dout, std::size_t id, DTable&& dtable)
117{
118 const auto d = as_row(dout);
119 const auto t = as_mutable_row(dtable);
120 assert((id + 1) * d.size() <= t.size());
121 for (std::size_t c = 0; c < d.size(); ++c)
122 t[id * d.size() + c] += d[c];
123}

130template <typename B, math::math_number T>
131constexpr void embed_rows(const B&, std::type_identity_t<std::span<const T>> table, std::type_identity_t<std::span<const T>> positions,
132 std::span<const std::uint32_t> ids, std::span<T> out)
133{
134 const std::size_t n = ids.size();
135 assert(n == 0 || (out.size() % n == 0 && (positions.empty() || positions.size() == out.size())));
136 const std::size_t d = n == 0 ? 0 : out.size() / n;
137 for (std::size_t t = 0; t < n; ++t)
138 for (std::size_t c = 0; c < d; ++c)
139 out[t * d + c] = positions.empty() ? table[ids[t] * d + c] : table[ids[t] * d + c] + positions[t * d + c];
140}

145template <typename B, math::math_number T>
146constexpr void embed_rows_backward(const B&, std::type_identity_t<std::span<const T>> dout, std::span<const std::uint32_t> ids,
147 std::span<T> dtable, std::type_identity_t<std::span<T>> dpositions)
148{
149 const std::size_t n = ids.size();

151 const std::size_t d = n == 0 ? 0 : dout.size() / n;
