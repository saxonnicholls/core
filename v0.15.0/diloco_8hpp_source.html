106template <math::math_number T>
107constexpr void ordered_sum(std::span<const std::span<const T>> parts, std::span<T> out) noexcept(std::is_floating_point_v<T>)
108{
109 assert(!parts.empty());
110 for ([[maybe_unused]] const auto& part : parts)
111 assert(part.size() == out.size());
112 const auto first = parts.front();
113 for (std::size_t i = 0; i < out.size(); ++i)
114 out[i] = first[i];
115 for (const auto part : parts.subspan(1)) // no index into parts: GCC 15's -Warray-bounds misreads one
116 for (std::size_t i = 0; i < out.size(); ++i)
117 out[i] = out[i] + part[i];
118}

128template <math::math_number T>
129constexpr void ordered_mean(std::span<const std::span<const T>> parts, std::span<T> out) noexcept(std::is_floating_point_v<T>)
130{
131 ordered_sum(parts, out);
132 const T count = T(parts.size());
133 for (std::size_t i = 0; i < out.size(); ++i)
134 out[i] = out[i] / count;
135}

145template <math::math_number T>
146constexpr void pseudo_gradient(std::span<const T> global, std::span<const T> local, std::span<T> out) noexcept(std::is_floating_point_v<T>)
147{
148 assert(global.size() == out.size() && local.size() == out.size());
149 for (std::size_t i = 0; i < out.size(); ++i)
150 out[i] = global[i] - local[i];
151}

173template <math::math_number T>
174constexpr void diloco_outer_step(std::span<T> global, std::span<const T> mean_delta, std::span<T> momentum,
175 const diloco_config& c) noexcept(std::is_floating_point_v<T>)
176{
177 assert(mean_delta.size() == global.size() && momentum.size() == global.size());
178 const T mu = T(c.momentum);
179 const T lr = T(c.outer_lr);
180 for (std::size_t i = 0; i < global.size(); ++i) {
181 const T g = mean_delta[i];
182 const T decayed = mu * momentum[i];
183 momentum[i] = decayed + g;
184 T step = momentum[i];
185 if (c.nesterov) {
186 const T ahead = mu * momentum[i];
187 step = g + ahead;
188 }
189 const T change = lr * step;
190 global[i] = global[i] - change;
191 }
192}
