104 // Whether a carrier holds a value, and the value: overloads, one per carrier.
105 template <class T>
106 constexpr bool engaged(const std::optional<T>& m) noexcept { return m.has_value(); }
107 template <class V, class E>
108 constexpr bool engaged(const snicholls::result<V, E>& m) noexcept { return m.is_ok(); }

119 // The first error, left to right; one of them is an error.
120 template <class E, class M0>
121 constexpr E first_error(M0&& m0) noexcept(
122 std::is_nothrow_constructible_v<E, decltype(std::forward<M0>(m0).unwrap_err_unchecked())>)
123 {
124 return std::forward<M0>(m0).unwrap_err_unchecked();
125 }
126
127 // The noexcept is spelled out: naming this overload in its own specification fails for three or more.
128 template <class E, class M0, class M1, class... M>
129 constexpr E first_error(M0&& m0, M1&& m1, M&&... m) noexcept(
130 ((std::is_nothrow_constructible_v<E, decltype(std::declval<M0>().unwrap_err_unchecked())> &&
131 std::is_nothrow_constructible_v<E, decltype(std::declval<M1>().unwrap_err_unchecked())>) && ... &&
132 std::is_nothrow_constructible_v<E, decltype(std::declval<M>().unwrap_err_unchecked())>))
133 {
134 if (m0.is_err()) return detail::first_error<E>(std::forward<M0>(m0));
135 return detail::first_error<E>(std::forward<M1>(m1), std::forward<M>(m)...);
136 }

138 struct lift_op {
139 // Every argument an optional.
140 template <std::size_t... I, class S, optional_carrier... M, class R = opt_lift_t<part_t<0, S>, M...>>
141 requires(sizeof...(M) > 0) && std::is_invocable_v<part_t<0, S>, opt_ref_t<M>...>
142 static constexpr R call(std::index_sequence<I...>, S&& s, M&&... m) noexcept(
143 std::is_nothrow_invocable_v<part_t<0, S>, opt_ref_t<M>...> &&
144 std::is_nothrow_move_constructible_v<typename R::value_type>)
145 {
146 if (!(m.has_value() && ...)) return std::nullopt;
147 return R(std::in_place, std::invoke(detail::part<0>(std::forward<S>(s)), *std::forward<M>(m)...));
148 }
149
150 // Every argument a result with a value type, all with the first one's error type.
151 template <std::size_t... I, class S, result_carrier M0, result_carrier... M, class E = error_of_t<M0>,
152 class V = std::remove_cvref_t<std::invoke_result_t<part_t<0, S>, res_ref_t<M0>, res_ref_t<M>...>>>
153 requires(std::is_same_v<E, error_of_t<M>> && ...) && (!std::is_void_v<V>)
154 static constexpr snicholls::result<V, E> call(std::index_sequence<I...>, S&& s, M0&& m0, M&&... m) noexcept(
155 std::is_nothrow_invocable_v<part_t<0, S>, res_ref_t<M0>, res_ref_t<M>...> &&
156 std::is_nothrow_move_constructible_v<V> && std::is_nothrow_copy_constructible_v<E> &&
157 std::is_nothrow_move_constructible_v<E>)
158 {
159 if (m0.is_ok() && (m.is_ok() && ...))
160 return snicholls::result<V, E>(std::in_place,
161 std::invoke(detail::part<0>(std::forward<S>(s)),
162 std::forward<M0>(m0).unwrap_unchecked(),
163 std::forward<M>(m).unwrap_unchecked()...));
164 return snicholls::err(detail::first_error<E>(std::forward<M0>(m0), std::forward<M>(m)...));
165 }
166 };

245 struct either_fn {
246 template <class F, class G, result_carrier M,
247 class R = std::common_type_t<
248 std::remove_cvref_t<std::invoke_result_t<F, decltype(std::declval<M>().unwrap_err_unchecked())>>,
249 std::remove_cvref_t<std::invoke_result_t<G, res_ref_t<M>>>>>
250 constexpr R operator()(F&& on_error, G&& on_value, M&& m) const
251 {
252 if (m.is_ok()) return R(std::invoke(std::forward<G>(on_value), std::forward<M>(m).unwrap_unchecked()));
253 return R(std::invoke(std::forward<F>(on_error), std::forward<M>(m).unwrap_err_unchecked()));
254 }

257 struct fold_m_fn {
258 template <class F, class Z, std::ranges::input_range R, class V = std::remove_cvref_t<Z>,
259 class M = std::remove_cvref_t<std::invoke_result_t<F&, V, std::ranges::range_reference_t<R>>>>
260 requires optional_carrier<M> || result_carrier<M>
261 constexpr M operator()(F&& f, Z&& z, R&& xs) const
262 {
263 M acc = pure<M>(V(std::forward<Z>(z)));
264 for (auto&& x : xs) {
265 if (!detail::engaged(acc)) break;
266 acc = std::invoke(f, detail::value_of(std::move(acc)), x);
267 }
268 return acc;
269 }
