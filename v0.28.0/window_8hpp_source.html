94 // a0 - a1 cos 2 pi x + a2 cos 4 pi x - a3 cos 6 pi x, with i folded to min(i, m - i) so the symmetric form is exact.
95 template <typename T>
96 constexpr T cosine_sum(std::size_t i, std::size_t m, double a0, double a1, double a2, double a3) noexcept
97 {
98 using C = compute_t<T>;
99 if (m == 0)
100 return T(1);
101 const std::size_t f = 2 * i > m ? m - i : i;
102 const C x = C(static_cast<long long>(f)) / C(static_cast<long long>(m));
103 return static_cast<T>(C(a0) - C(a1) * cos2pi(x) + C(a2) * cos2pi(C(2) * x) - C(a3) * cos2pi(C(3) * x));
104 }
105} // namespace detail::window

221struct tukey
222{
223 double alpha = 0.5;
224
230 template <math::math_number T>
231 constexpr T at(std::size_t i, std::size_t m) const noexcept
232 {
233 using C = compute_t<T>;
234 if (alpha <= 0 || m == 0)
235 return T(1);
236 if (alpha >= 1)
237 return hann{}.at<T>(i, m);
238 const C a = C(alpha), mm = C(static_cast<long long>(m)), pi = math::math_limits<C>::pi;
239 const auto width = static_cast<std::size_t>(math::floor(a * mm / C(2)));
240 const C x = C(static_cast<long long>(i));
241 if (i <= width)
242 return static_cast<T>(C(0.5) * (C(1) + math::cos(pi * (C(-1) + C(2) * x / a / mm))));
243 if (i < m - width)
244 return T(1);
245 return static_cast<T>(C(0.5) * (C(1) + math::cos(pi * (C(-2) / a + C(1) + C(2) * x / a / mm))));
246 }
247};

302 // The sums of w^Power over every position modulo the hop, as scipy's check_COLA / check_NOLA fold them.
303 template <int Power, typename T, typename Check>
304 constexpr bool fold(std::span<const T> w, std::size_t hop, Check check) noexcept
305 {
306 if (hop == 0 || hop > w.size())
307 return false;
308 for (std::size_t r = 0; r < hop; ++r) {
309 T s = T(0);
310 for (std::size_t i = r; i < w.size(); i += hop)
311 s += Power == 1 ? w[i] : w[i] * w[i];
312 if (!check(r, s))
313 return false;
314 }
315 return true;
316 }
317} // namespace detail::window

325template <math::math_number T>
326constexpr bool is_cola(std::span<const T> w, std::size_t hop, T tolerance = T(1e-10)) noexcept
327{
328 T first = T(0);
329 return detail::window::fold<1>(w, hop, [&](std::size_t r, T s) {
330 if (r == 0)
331 first = s;
332 return math::abs(s - first) <= tolerance;
333 });
334}
