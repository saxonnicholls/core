309template <transformer_config C, typename B, math::math_number T, typename Attention, typename Mlp, typename AttentionActivations,
310 typename MlpActivations, typename R>
311 requires linalg::blas_backend<B, T>
312constexpr moe_losses<T> block_forward_sequence(const B& backend, const composed_block_weights<Attention, Mlp, T>& w,
313 std::type_identity_t<std::span<T>> xs, const R& rope_src,
314 const composed_block_activations<AttentionActivations, MlpActivations, T>& act,
315 std::size_t sequences = 1)
316{
317 assert(xs.size() % C.d_model == 0 && act.x0.size() == xs.size() && act.h1.size() == xs.size() && act.x1.size() == xs.size()
318 && act.h2.size() == xs.size());
319 copy_rows(backend, std::span<const T>(xs), act.x0);
320 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm1_weight, w.norm1_bias, act.h1);
321 detail::composed_block::attention_pass<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, xs, sequences);
322 add_rows(backend, std::span<const T>(act.x0), xs);
323 copy_rows(backend, std::span<const T>(xs), act.x1);
324 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm2_weight, w.norm2_bias, act.h2);
325 const auto losses = detail::composed_block::mlp_pass<C>(backend, w.mlp, std::span<const T>(act.h2), xs, act.mlp, sequences);
326 add_rows(backend, std::span<const T>(act.x1), xs);
327 return losses;
328}

341template <transformer_config C, typename B, math::math_number T, typename Attention, typename Mlp, typename AttentionActivations,
342 typename MlpActivations, typename AttentionGradients, typename MlpGradients, typename R>
343 requires linalg::blas_backend<B, T>
344constexpr void block_backward(const B& backend, const composed_block_weights<Attention, Mlp, T>& w, const R& rope_src,
345 const composed_block_activations<AttentionActivations, MlpActivations, T>& act,
346 std::type_identity_t<std::span<T>> dxs, const composed_block_gradients<AttentionGradients, MlpGradients, T>& grads,
347 std::size_t sequences = 1)
348{
349 assert(dxs.size() == act.x0.size() && grads.scratch.size() == dxs.size());
350 fill_rows(backend, grads.scratch, T(0));
351 detail::composed_block::mlp_pass_backward<C>(backend, w.mlp, std::span<const T>(act.h2), act.mlp, std::span<const T>(dxs), grads.scratch,
352 grads.mlp, sequences);
353 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x1), w.norm2_weight, w.norm2_bias,
354 dxs, grads.norm2_weight, grads.norm2_bias);
355 fill_rows(backend, grads.scratch, T(0));
356 detail::composed_block::attention_pass_backward<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention,
357 std::span<const T>(dxs), grads.scratch, grads.attention, sequences);
358 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x0), w.norm1_weight, w.norm1_bias,
359 dxs, grads.norm1_weight, grads.norm1_bias);
360}

371template <transformer_config C, typename Attention, typename Mlp, math::math_number T, mutable_row X, typename Cache, typename R>
372 requires std::same_as<row_value_t<X>, T>
373constexpr void block_forward(const composed_block_weights<Attention, Mlp, T>& w, X&& x, std::size_t pos, const Cache& cache, std::size_t layer,
374 const R& rope_src)
375{
376 const auto s = as_mutable_row(x);
377 assert(s.size() == C.d_model);
378 std::array<T, C.d_model> h{}, a{};
379 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
380 detail::composed_block::attention_token<C>(w.attention, std::span<const T>(h), pos, cache, layer, rope_src, std::span<T>(a));
381 for (std::size_t i = 0; i < C.d_model; ++i)
382 s[i] += a[i];
383 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
384 mlp_forward<C>(w.mlp, h, a);
385 for (std::size_t i = 0; i < C.d_model; ++i)
386 s[i] += a[i];
387}

437template <transformer_config C, math::math_number T>
438 requires(C.latent() && C.moe())
439constexpr latent_moe_block_weights<T> latent_moe_block_weights_in(std::span<const T> params, std::span<const T> selection_bias = {})
440{
441 assert(params.size() == latent_moe_block_parameter_count(C));
442 constexpr std::size_t d = C.d_model, b = C.norm == norm_kind::layer ? d : 0;
443 detail::cutter<const T> c{params};
444 latent_moe_block_weights<T> w{};
445 w.norm1_weight = c.take(d);
446 w.norm1_bias = c.take(b);
447 w.attention = latent_attention_weights_in<C>(c.take(latent_attention_parameter_count(C)));
448 w.norm2_weight = c.take(d);
449 w.norm2_bias = c.take(b);
450 w.mlp = moe_weights_in<C>(c.take(moe_parameter_count(C)), selection_bias);
451 return w;
452}

461template <transformer_config C, math::math_number T>
462 requires(C.latent() && C.moe())
463constexpr latent_moe_block_activations<T> latent_moe_block_activations_in(std::span<T> values, std::span<std::uint32_t> indices,
464 std::size_t rows, std::span<const std::uint32_t> selected = {})
465{
466 assert(values.size() == latent_moe_block_activation_values(C, rows) && indices.size() == latent_moe_block_activation_indices(C, rows));
467 detail::cutter<T> v{values};
468 latent_moe_block_activations<T> a{};
469 a.x0 = v.take(rows * C.d_model);
470 a.h1 = v.take(rows * C.d_model);
471 a.x1 = v.take(rows * C.d_model);
472 a.h2 = v.take(rows * C.d_model);
473 a.attention = latent_attention_activations_in<C>(v.take(latent_attention_activation_values(C, rows)), rows, selected);
474 a.mlp = moe_activations_in<C>(v.take(moe_activation_values(C, rows)), indices, rows);
475 return a;
476}

484template <transformer_config C, math::math_number T>
485 requires(C.latent() && C.moe())
486constexpr latent_moe_block_gradients<T> latent_moe_block_gradients_in(std::span<T> grads, std::span<T> work, std::size_t rows)
487{
488 assert(grads.size() == latent_moe_block_parameter_count(C) && work.size() == latent_moe_block_gradient_work(C, rows));
489 constexpr std::size_t d = C.d_model, b = C.norm == norm_kind::layer ? d : 0;
490 detail::cutter<T> c{grads}, w{work};
491 latent_moe_block_gradients<T> g{};
492 g.norm1_weight = c.take(d);
493 g.norm1_bias = c.take(b);
494 g.attention = latent_attention_gradients_in<C>(c.take(latent_attention_parameter_count(C)), w.take(latent_attention_gradient_work(C, rows)),
495 rows);
496 g.norm2_weight = c.take(d);
497 g.norm2_bias = c.take(b);
498 g.mlp = moe_gradients_in<C>(c.take(moe_parameter_count(C)), w.take(moe_gradient_work(C, rows)), rows);
499 g.scratch = w.take(rows * d);
500 return g;
501}
