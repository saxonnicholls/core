186 template <time_zone Z, utc_duration D>
187 [[nodiscard]] constexpr result<utc<D>, time_zone_errc> from_local(const Z& zone, const std::chrono::local_time<D>& local,
188 local_options options = {})
189 {
190 const local_info li = zone.info(std::chrono::floor<std::chrono::seconds>(local));
191 const auto at = [&](std::chrono::seconds offset) {
192 return utc<D>(typename utc<D>::time_point(local.time_since_epoch() - std::chrono::duration_cast<D>(offset)));
193 };
194 switch (li.kind) {
195 case local_time_kind::unique: return ok(at(li.first.offset));
196 case local_time_kind::nonexistent:
197 if (options.gap == gap_policy::next_valid)
198 return ok(utc<D>(typename utc<D>::time_point(std::chrono::duration_cast<D>(li.second.begin.time_since_epoch()))));
199 if (options.gap == gap_policy::shift_forward)
200 return ok(at(li.first.offset));
201 return err(time_zone_errc::nonexistent_local_time);
202 case local_time_kind::ambiguous:
203 if (options.fold == fold_policy::earliest)
204 return ok(at(li.first.offset));
205 if (options.fold == fold_policy::latest)
206 return ok(at(li.second.offset));
207 return err(time_zone_errc::ambiguous_local_time);
208 }
209 return err(time_zone_errc::nonexistent_local_time);
210 }

218 template <utc_duration D>
219 [[nodiscard]] constexpr rfc3339_text to_text(const utc<D>& t, std::chrono::seconds offset, int precision = utc<D>::fraction_digits) noexcept
220 {
221 rfc3339_text text;
222 const auto s = offset.count();
223 if (s % 60 != 0 || s <= -86400 || s >= 86400) {
224 text.ec = std::errc::not_supported;
225 return text;
226 }
227 using C = std::common_type_t<D, std::chrono::seconds>;
228 const auto ticks = std::chrono::duration_cast<C>(t.time_since_epoch());
229 const auto shift = std::chrono::duration_cast<C>(offset);
230 if ((shift.count() > 0 && ticks.count() > C::max().count() - shift.count()) ||
231 (shift.count() < 0 && ticks.count() < C::min().count() - shift.count())) {
232 text.ec = std::errc::value_too_large;
233 return text;
234 }
235 const utc<C> local_as_utc(typename utc<C>::time_point(ticks + shift));
236 text = to_text(local_as_utc, precision);
237 if (!text || s == 0)
238 return text;
239 // replace the trailing 'Z' with +HH:MM
240 const auto minutes = static_cast<unsigned>((s < 0 ? -s : s) / 60);
241 char* p = text.buffer.data() + text.size - 1;
242 p[0] = s < 0 ? '-' : '+';
243 p[1] = static_cast<char>('0' + minutes / 600);
244 p[2] = static_cast<char>('0' + minutes / 60 % 10);
245 p[3] = ':';
246 p[4] = static_cast<char>('0' + minutes % 60 / 10);
247 p[5] = static_cast<char>('0' + minutes % 10);
248 text.size = static_cast<std::uint8_t>(text.size + 5);
249 return text;
250 }
