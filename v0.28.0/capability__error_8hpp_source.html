118 class provider_code {
119 public:
121 static constexpr std::size_t capacity = 32;
122
126 [[nodiscard]] static constexpr bool is_code(std::string_view text) noexcept
127 {
128 if (text.empty() || text.size() > capacity)
129 return false;
130 for (const char c : text)
131 if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_' ||
132 c == '-' || c == '.' || c == ':'))
133 return false;
134 return true;
135 }
136
138 constexpr provider_code() noexcept = default;
139
143 template <std::size_t M>
144 requires(M >= 2 && M <= capacity + 1)
145 consteval provider_code(const char (&literal)[M]) // NOLINT(google-explicit-constructor)
146 : text_(literal)
147 {
148 if (!is_code(text_.view()))
149 throw "snicholls::provider_code: not a provider code (1 to 32 of A-Z a-z 0-9 _ - . :)";
150 }

155 [[nodiscard]] static constexpr result<provider_code, capability_errc> make(std::string_view text) noexcept
156 {
157 if (!is_code(text))
158 return err(capability_errc::invalid_argument);
159 provider_code out;
160 out.text_ = *inplace_string<capacity>::make(text);
161 return ok(out);
162 }
