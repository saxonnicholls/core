67 // |n| of a ratio part, never INT64_MIN.
68 constexpr std::uint64_t magnitude(std::int64_t v) noexcept
69 {
70 return static_cast<std::uint64_t>(v < 0 ? -v : v);

73 // magnitude * num / den rounded under `mode` with the result's sign, or none past 256 bits.
74 constexpr std::optional<uint256> scaled(const uint256& magnitude, const uint256& num, const uint256& den, bool negative,
75 rounding mode) noexcept
76 {
77 const auto product = uint256::checked_mul(magnitude, num);
78 const auto qr = product ? uint256::checked_divmod(*product, den) : std::nullopt;
79 if (!qr)
80 return std::nullopt;
81 if (!snicholls::detail::rounds_away(qr->first, qr->second, den, negative, mode))
82 return qr->first;
83 return uint256::checked_add(qr->first, uint256::from_limbs(1));
84 }

86 // m * (num / den) as money of `out`, rounded once.
87 constexpr std::optional<money> apply(const money& m, std::int64_t num, std::int64_t den, const asset& out,
88 rounding mode) noexcept
89 {
90 if (den == 0)
91 return std::nullopt;
92 const bool negative = (m.is_negative() != (num < 0)) != (den < 0) && !m.is_zero() && num != 0;
93 const auto top = uint256::checked_mul(uint256::from_limbs(magnitude(num)), uint256::pow10(out.scale()));
94 const auto bottom = uint256::checked_mul(uint256::from_limbs(magnitude(den)), uint256::pow10(m.asset_of().scale()));
95 const auto v = top && bottom ? scaled(m.magnitude(), *top, *bottom, negative, mode) : std::nullopt;
96 if (!v)
97 return std::nullopt;
98 return money::from_units(out, *v, negative && !v->is_zero());
99 }

106 [[nodiscard]] constexpr std::optional<ratio> exact_value(const money& m) noexcept
107 {
108 const auto units = m.magnitude().to_uint64();
109 if (!units || *units > detail::max_part || m.asset_of().scale() > 18)
110 return std::nullopt;
111 const auto n = static_cast<std::int64_t>(*units);
112 std::int64_t d = 1;
113 for (unsigned i = 0; i < m.asset_of().scale(); ++i)
114 d *= 10;
115 return ratio{m.is_negative() ? -n : n, d};
116 }

171 [[nodiscard]] constexpr std::optional<money> to_money(const ratio& r, const asset& out, rounding mode) noexcept
172 {
173 const bool negative = r.is_negative();
174 const auto v = detail::scaled(uint256::from_limbs(detail::magnitude(r.numerator())), uint256::pow10(out.scale()),
175 uint256::from_limbs(static_cast<std::uint64_t>(r.denominator())), negative, mode);
176 if (!v)

186 [[nodiscard]] constexpr std::optional<ratio> ratio_of(const money& a, const money& b) noexcept
187 {
188 const auto x = exact_value(a), y = exact_value(b);

190 return x && inv ? multiply(*x, *inv) : std::nullopt;
