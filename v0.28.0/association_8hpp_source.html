85 template <linalg::scalar T>
86 constexpr bool pda_weights(std::span<const T> log_likelihoods, T pd, T pg, T clutter_density, std::span<T> weights) noexcept(linalg::detail::nothrow_scalar<T>)
87 {
88 if (weights.size() != log_likelihoods.size() + 1 || !(clutter_density > T(0)))
89 return false;
90 const T miss = T(1) - pd * pg;
91 weights[0] = miss > T(0) ? static_cast<T>(math::log(miss)) : -std::numeric_limits<T>::infinity();
92 const T base = static_cast<T>(math::log(pd)) - static_cast<T>(math::log(clutter_density));
93 for (std::size_t i = 0; i < log_likelihoods.size(); ++i)
94 weights[i + 1] = log_likelihoods[i] + base;
95 const T z = stats::normalise_log_weights(weights);
96 if (!(z > -std::numeric_limits<T>::infinity()) || !(z < std::numeric_limits<T>::infinity()))
97 return false;
98 for (T& w : weights)
99 w = static_cast<T>(math::exp(w));
100 return true;
101 }

113 template <linalg::scalar T, std::size_t N, std::size_t M>
114 [[nodiscard]] constexpr result<gaussian<T, N>, linalg::error> pda_update(const gaussian<T, N>& prediction, const measurement_prediction<T, N, M>& mp,
115 std::span<const linalg::fixed_vector<T, M>> innovations, std::span<const T> weights) noexcept(linalg::detail::nothrow_scalar<T>)
116 {
117 if (weights.size() != innovations.size() + 1)
118 return err(linalg::error::dimension_mismatch);
119 const auto l = linalg::cholesky(mp.covariance);
120 if (!l)
121 return err(l.unwrap_err_unchecked());
122 const auto k = linalg::transpose(linalg::cholesky_solve(*l, linalg::transpose(mp.cross_covariance)));
123 linalg::fixed_vector<T, M> nu{};
124 linalg::fixed_matrix<T, M, M> spread{};
125 for (std::size_t i = 0; i < innovations.size(); ++i) {
126 nu += innovations[i] * weights[i + 1];
127 spread += linalg::outer(innovations[i], innovations[i]) * weights[i + 1];
128 }
129 spread -= linalg::outer(nu, nu);
130 const T b0 = weights[0];
131 const auto pc = prediction.covariance - linalg::sandwich(k, mp.covariance);
132 return ok(gaussian<T, N>{prediction.mean + k * nu, linalg::symmetrise(prediction.covariance * b0 + pc * (T(1) - b0) + linalg::sandwich(k, spread))});
133 }

142 template <linalg::scalar T, std::size_t M>
143 [[nodiscard]] constexpr result<T, linalg::error> gate_volume(T threshold, const linalg::fixed_matrix<T, M, M>& s) noexcept(linalg::detail::nothrow_scalar<T>)
144 {
145 const auto l = linalg::cholesky(s);
146 if (!l)
147 return err(l.unwrap_err_unchecked());
148 const T half = static_cast<T>(M) / T(2);
149 const T log_v = half * static_cast<T>(math::log(static_cast<T>(std::numbers::pi_v<long double>))) - static_cast<T>(math::lgamma(half + T(1))) +
150 half * static_cast<T>(math::log(threshold)) + linalg::cholesky_log_det(*l) / T(2);
151 return ok(static_cast<T>(math::exp(log_v)));
152 }

165 template <std::size_t MaxTracks, std::size_t MaxDetections, linalg::scalar T>
166 constexpr bool jpda_marginals(std::span<const T> likelihoods, std::size_t tracks, std::size_t detections, std::span<T> marginals) noexcept(linalg::detail::nothrow_scalar<T>)
167 {
168 const std::size_t cols = detections + 1;
169 if (tracks > MaxTracks || detections > MaxDetections || likelihoods.size() < tracks * cols || marginals.size() < tracks * cols)
170 return false;
171 for (std::size_t i = 0; i < tracks * cols; ++i)
172 marginals[i] = T(0);
173 if (tracks == 0)
174 return true;
175 std::array<std::size_t, MaxTracks + 1> choice{}; // the hypothesis chosen at each depth, or cols when exhausted
176 std::array<T, MaxTracks + 1> partial{}; // the product of the chosen weights above each depth
177 std::array<std::uint8_t, MaxDetections + 1> used{};
178 T total(0);
179 std::size_t depth = 0;
180 partial[0] = T(1);
181 choice[0] = 0;
182 bool fresh = true; // whether choice[depth] is still to be tried from its current value
183 for (;;) {
184 if (!fresh) {
185 // Release the detection this depth used, then advance to the next choice
186 if (choice[depth] > 0)
187 used[choice[depth]] = 0;
188 ++choice[depth];
189 }
190 fresh = false;
191 while (choice[depth] < cols) {
192 const std::size_t c = choice[depth];
193 if (likelihoods[depth * cols + c] > T(0) && (c == 0 || !used[c]))
194 break;
195 ++choice[depth];
196 }
197 if (choice[depth] >= cols) {
198 if (depth == 0)
199 break;
200 --depth;
201 continue;
202 }
203 const std::size_t c = choice[depth];
204 if (c > 0)
205 used[c] = 1;
206 const T p = partial[depth] * likelihoods[depth * cols + c];
207 if (depth + 1 == tracks) {
208 total += p;
209 for (std::size_t t = 0; t < tracks; ++t)
210 marginals[t * cols + choice[t]] += p;
211 continue; // not fresh: release and advance this depth
212 }
213 partial[depth + 1] = p;
214 ++depth;
215 choice[depth] = 0;
216 fresh = true;
217 }
218 if (!(total > T(0)))
219 return false;
220 for (std::size_t i = 0; i < tracks * cols; ++i)
221 marginals[i] /= total;
222 return true;
223 }
