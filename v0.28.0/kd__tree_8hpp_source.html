97 template <metric::point Point, std::size_t K, typename Metric = metric::euclidean>
98 requires metric::decomposable<Metric, double>
99 class kd_tree {
100 static_assert(K > 0, "snicholls::containers::kd_tree: the dimension is at least 1");
101 static constexpr std::size_t max_depth = 64;
102 template <typename Q>
103 static constexpr bool nothrow_read = noexcept(metric::coordinate(std::declval<const Point&>(), std::size_t{})) &&
104 noexcept(metric::coordinate(std::declval<const Q&>(), std::size_t{}));
105
106 struct frame {
107 std::size_t lo, hi, depth;
108 double bound; // a lower bound on the reduced distance of every point in the range
109 };

111 static constexpr double at(const auto& p, std::size_t axis) noexcept(noexcept(metric::coordinate(p, axis)))
112 {
113 return static_cast<double>(metric::coordinate(p, axis));
114 }

116 public:
121 constexpr kd_tree(std::span<const Point> points, std::span<std::uint32_t> indices, const Metric& m = Metric{}) noexcept(nothrow_read<Point>)
122 : points_(points), index_(indices.first(indices.size() < points.size() ? indices.size() : points.size())), metric_(m)
123 {
124 for (std::size_t i = 0; i < index_.size(); ++i)
125 index_[i] = static_cast<std::uint32_t>(i);
126 std::array<frame, max_depth> stack{};
127 std::size_t top = 0;
128 if (index_.size() > 1)
129 stack[top++] = {0, index_.size(), 0, 0};
130 while (top > 0) {
131 const frame f = stack[--top];
132 const std::size_t mid = f.lo + (f.hi - f.lo) / 2;
133 const std::size_t axis = f.depth % K;
134 std::nth_element(index_.begin() + static_cast<std::ptrdiff_t>(f.lo), index_.begin() + static_cast<std::ptrdiff_t>(mid),
135 index_.begin() + static_cast<std::ptrdiff_t>(f.hi),
136 [&](std::uint32_t a, std::uint32_t b) { return at(points_[a], axis) < at(points_[b], axis); });
137 if (mid - f.lo > 1)
138 stack[top++] = {f.lo, mid, f.depth + 1, 0};
139 if (f.hi - (mid + 1) > 1)
140 stack[top++] = {mid + 1, f.hi, f.depth + 1, 0};
141 }
142 }

145 [[nodiscard]] constexpr std::size_t size() const noexcept { return index_.size(); }
147 [[nodiscard]] constexpr bool empty() const noexcept { return index_.empty(); }
149 [[nodiscard]] constexpr const Metric& distance_metric() const noexcept { return metric_; }

155 template <metric::point Q>
156 [[nodiscard]] constexpr std::optional<kd_neighbour> nearest(const Q& q) const noexcept(nothrow_read<Q>)
157 {
158 if (empty())
159 return std::nullopt;
160 kd_neighbour best{0, std::numeric_limits<double>::infinity()};
161 search(q, [&](std::uint32_t i, double r) {
162 if (before(r, i, best)) // at an equal distance, the lower index
163 best = {i, r};
164 }, [&] { return best.distance; });
165 best.distance = metric::expand(metric_, best.distance);
166 return best;
167 }

174 template <metric::point Q>
175 constexpr std::size_t k_nearest(const Q& q, std::span<kd_neighbour> out) const noexcept(nothrow_read<Q>)
176 {
177 const std::size_t k = out.size() < size() ? out.size() : size();
178 if (k == 0)
179 return 0;
180 std::size_t have = 0;
181 search(q, [&](std::uint32_t i, double r) {
182 if (have == k && !before(r, i, out[k - 1]))
183 return;
184 std::size_t j = have < k ? have++ : k - 1;
185 for (; j > 0 && before(r, i, out[j - 1]); --j)
186 out[j] = out[j - 1];
187 out[j] = {i, r};
188 }, [&] { return have < k ? std::numeric_limits<double>::infinity() : out[k - 1].distance; });
189 for (std::size_t j = 0; j < k; ++j)
190 out[j].distance = metric::expand(metric_, out[j].distance);
191 return k;
192 }

203 template <metric::point Q>
204 constexpr std::size_t k_nearest_within(const Q& q, double radius, std::span<kd_neighbour> out) const noexcept(nothrow_read<Q>)
205 {
206 const std::size_t k = out.size() < size() ? out.size() : size();
207 if (k == 0)
208 return 0;
209 const double bound = metric::contract(metric_, radius);
210 std::size_t have = 0;
211 search(q, [&](std::uint32_t i, double r) {
212 if (!(r < bound) || (have == k && !before(r, i, out[k - 1])))
213 return;
214 std::size_t j = have < k ? have++ : k - 1;
215 for (; j > 0 && before(r, i, out[j - 1]); --j)
216 out[j] = out[j - 1];
217 out[j] = {i, r};
218 }, [&] { return have < k ? bound : (out[k - 1].distance < bound ? out[k - 1].distance : bound); });
219 for (std::size_t j = 0; j < have; ++j)
220 out[j].distance = metric::expand(metric_, out[j].distance);
221 return have;
222 }

230 template <metric::point Q, typename F>
231 constexpr void for_each_within(const Q& q, double radius, F&& f) const noexcept(nothrow_read<Q> && noexcept(f(std::uint32_t{}, double{})))
232 {
233 const double r = metric::contract(metric_, radius);
234 search(q, [&](std::uint32_t i, double d) {
235 if (d <= r)
236 f(i, metric::expand(metric_, d));
237 }, [&] { return r; });
238 }

246 template <metric::point Q, typename F>
247 constexpr void for_each_in_box(const Q& lo, const Q& hi, F&& f) const noexcept(nothrow_read<Q> && noexcept(f(std::uint32_t{})))
248 {
249 std::array<frame, max_depth> stack{};
250 std::size_t top = 0;
251 if (!empty())
252 stack[top++] = {0, index_.size(), 0, 0};
253 while (top > 0) {
254 const frame fr = stack[--top];
255 const std::size_t mid = fr.lo + (fr.hi - fr.lo) / 2;
256 const std::size_t axis = fr.depth % K;
257 const auto& p = points_[index_[mid]];
258 bool inside = true;
259 for (std::size_t a = 0; a < K && inside; ++a) {
260 const double c = at(p, a);
261 inside = c >= at(lo, a) && c <= at(hi, a);
262 }
263 if (inside)
264 f(index_[mid]);
265 const double c = at(p, axis);
266 if (mid > fr.lo && at(lo, axis) <= c)
267 stack[top++] = {fr.lo, mid, fr.depth + 1, 0};
268 if (mid + 1 < fr.hi && at(hi, axis) >= c)
269 stack[top++] = {mid + 1, fr.hi, fr.depth + 1, 0};
270 }
271 }

274 // The order of neighbours: nearer first, and at an equal distance the lower point index first, so a query's
275 // result is one well-defined list (it matches a reference that breaks ties by index, bit for bit)
276 static constexpr bool before(double r, std::uint32_t i, const kd_neighbour& n) noexcept
277 {
278 return r < n.distance || (r == n.distance && i < n.index);
279 }

281 // The reduced distance, stopping early once it exceeds bound
282 template <typename Q>
283 constexpr double reduced(const Q& q, std::uint32_t i, double bound) const noexcept(nothrow_read<Q>)
284 {
285 double acc = 0;
286 for (std::size_t a = 0; a < K && !(acc > bound); ++a)
287 acc = metric_.combine(acc, metric_.term(a, at(points_[i], a) - at(q, a)));
288 return acc;
289 }

291 // Visits candidates nearest-first along the tree, pruning ranges whose bound is beyond bound()
292 template <typename Q, typename Visit, typename Bound>
293 constexpr void search(const Q& q, Visit&& visit, Bound&& bound) const
294 {
295 std::array<frame, max_depth> stack{};
296 std::size_t top = 0;
297 if (!empty())
298 stack[top++] = {0, index_.size(), 0, 0};
299 while (top > 0) {
300 const frame fr = stack[--top];
301 if (fr.bound > bound())
302 continue;
303 const std::size_t mid = fr.lo + (fr.hi - fr.lo) / 2;
304 const std::size_t axis = fr.depth % K;
305 const std::uint32_t node = index_[mid];
306 visit(node, reduced(q, node, bound()));
307 const double diff = at(q, axis) - at(points_[node], axis);
308 const double t = metric_.term(axis, diff);
309 const double far_bound = fr.bound > t ? fr.bound : t;
310 const bool left_near = diff < 0;
311 const frame lo_child{fr.lo, mid, fr.depth + 1, left_near ? fr.bound : far_bound};
312 const frame hi_child{mid + 1, fr.hi, fr.depth + 1, left_near ? far_bound : fr.bound};
313 // Push the far child first so the near one is searched first
314 const frame& near = left_near ? lo_child : hi_child;
315 const frame& far = left_near ? hi_child : lo_child;
316 if (far.hi > far.lo)
317 stack[top++] = far;
318 if (near.hi > near.lo)
319 stack[top++] = near;
320 }
321 }
