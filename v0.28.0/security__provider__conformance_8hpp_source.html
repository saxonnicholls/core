73 template <typename Probe>
74 concept security_provider_probe = std::is_default_constructible_v<Probe> &&
75 security_provider<std::remove_reference_t<decltype(std::declval<Probe&>().provider())>> &&
76 requires(Probe& p, std::string_view order, const money& m, const webhook_event& e, capability_errc code) {
77 { p.quantity() } -> std::same_as<money>;
78 { p.consideration() } -> std::same_as<money>;
79 p.trade(order, m, m);
80 p.settle(order);
81 { p.deliver(e) } -> std::same_as<owned_delivery>;
82 { p.deliver_late(e) } -> std::same_as<owned_delivery>;
83 p.provider_fails(code, order);
84 };

86 inline webhook_event security_kit_event(std::string_view order = "ord-1")
87 {
88 webhook_event e;
89 e.source = "test";
90 e.event_id = "evt_sec_1";
91 e.kind = event_kind::security_updated;
92 e.type = "trade.filled";
93 e.payment_id = *provider_id::make(order);
94 e.occurred_at = utc_s::from_unix_seconds(1'800'000'000);
95 return e;

98 template <typename Provider>
99 std::vector<security_fact> kit_resolve(Provider& provider, std::string_view order)
100 {
101 const auto r = provider.resolve(security_kit_event(order));
102 return r.is_ok() ? *r : std::vector<security_fact>{};
103 }

122 capability_conformance_report report;
123 const auto code_is = [](const auto& r, capability_errc c) { return r.is_err() && r.unwrap_err().code == c; };

125 detail::capability_check<Probe>(report, "capabilities name the kinds it reports", [](Probe& p, const auto& expect) {
126 const auto caps = p.provider().capabilities();
127 expect(!caps.kinds.empty(), "at least one kind");
128 expect(std::find(caps.kinds.begin(), caps.kinds.end(), security_fact_kind::buy) != caps.kinds.end(),
129 "a provider the kit can drive reports buys");
130 });

132 detail::capability_check<Probe>(report, "a trade resolves to one fact that validates", [](Probe& p, const auto& expect) {
133 p.trade("ord-1", p.quantity(), p.consideration());
134 const auto facts = detail::kit_resolve(p.provider(), "ord-1");
135 expect(facts.size() == 1, "the trade resolves to one fact");
136 if (facts.size() != 1)
137 return;
138 const security_fact& f = facts.front();
139 expect(validate(f).is_ok(), "the fact validates");
140 expect(f.kind == security_fact_kind::buy, "a positive quantity is a buy");
141 expect(f.quantity == p.quantity() && f.consideration == p.consideration(), "quantity and cash are the trade's");
142 expect(f.event_id == detail::security_kit_event().event_id, "the fact carries the event that led to it");
143 expect(!f.dates.settles_on || *f.dates.settles_on >= f.dates.trade_at.days(), "it settles on or after its trade date");
144 const auto again = detail::kit_resolve(p.provider(), "ord-1");
145 expect(again.size() == 1 && again.front().fact_id == f.fact_id, "a redelivery points at the same fact");
146 });
147
148 detail::capability_check<Probe>(report, "an event that books nothing resolves to no facts", [](Probe& p, const auto& expect) {
149 p.trade("ord-1", p.quantity(), p.consideration());
150 webhook_event other = detail::security_kit_event();
151 other.kind = event_kind::other;
152 const auto none = p.provider().resolve(other);
153 expect(none.is_ok() && none->empty(), "an other event resolves to no facts");
154 webhook_event test = detail::security_kit_event();
155 test.is_test = true;
156 const auto booked = p.provider().resolve(test);
157 expect(booked.is_ok() && booked->empty(), "a test delivery resolves to no facts");
158 const auto unknown = p.provider().resolve(detail::security_kit_event("ord-none"));
159 expect(unknown.is_ok() && unknown->empty(), "an unknown order resolves to no facts");
160 });
161
162 detail::capability_check<Probe>(report, "settlement moves the settled holding", [](Probe& p, const auto& expect) {
163 if (!p.provider().capabilities().holdings)
164 return;
165 const money q = p.quantity();
166 p.trade("ord-1", q, p.consideration());
167 const auto traded = detail::kit_resolve(p.provider(), "ord-1");
168 expect(traded.size() == 1, "the trade resolves");
169 if (traded.size() != 1)
170 return;
171 const auto day = traded.front().dates.trade_at.days();
172 const auto before = p.provider().holdings(day - std::chrono::days{1});
173 expect(before.is_ok() && detail::kit_find(*before, q.asset_of()) == nullptr, "nothing is held the day before");
174 const auto pending = p.provider().holdings(day);
175 const security_holding* h = pending.is_ok() ? detail::kit_find(*pending, q.asset_of()) : nullptr;
176 expect(h && h->traded == q && h->settled.is_zero(), "traded on the trade date, not yet settled");
177 p.settle("ord-1");
178 const auto facts = detail::kit_resolve(p.provider(), "ord-1");
179 const auto settled_fact = std::find_if(facts.begin(), facts.end(),
180 [](const security_fact& f) { return f.kind == security_fact_kind::settle; });
181 expect(settled_fact != facts.end() && validate(*settled_fact).is_ok() && settled_fact->dates.settled_at,
182 "the settlement is a settle fact with its time");
183 if (settled_fact == facts.end() || !settled_fact->dates.settled_at)
184 return;
185 const auto after = p.provider().holdings(settled_fact->dates.settled_at->days());
186 const security_holding* s = after.is_ok() ? detail::kit_find(*after, q.asset_of()) : nullptr;
187 expect(s && s->traded == q && s->settled == q, "settled once the settle fact is in");
188 });
189
190 detail::capability_check<Probe>(report, "a webhook verifies and carries its event", [](Probe& p, const auto& expect) {
191 const webhook_event e = detail::security_kit_event();
192 const owned_delivery d = p.deliver(e);
193 const auto v = p.provider().verify_webhook(d.view());
194 expect(v.is_ok(), "a signed delivery verifies");
195 if (!v.is_ok())
196 return;
197 expect(v->event_id == e.event_id && v->kind == e.kind && v->payment_id == e.payment_id, "the event round-trips");
198 const auto again = p.provider().verify_webhook(d.view());
199 expect(again.is_ok() && *again == *v, "the event id is stable across deliveries");
200 });

202 detail::capability_check<Probe>(report, "a webhook fails closed", [&](Probe& p, const auto& expect) {
203 auto& pp = p.provider();
204 owned_delivery tampered = p.deliver(detail::security_kit_event());
205 tampered.body[tampered.body.size() / 2] = static_cast<char>(tampered.body[tampered.body.size() / 2] ^ 1);
206 expect(code_is(pp.verify_webhook(tampered.view()), capability_errc::signature_invalid), "a tampered body");
207 owned_delivery unsigned_one = p.deliver(detail::security_kit_event());
208 unsigned_one.headers.clear();
209 expect(code_is(pp.verify_webhook(unsigned_one.view()), capability_errc::signature_invalid), "no signature");
210 const owned_delivery late = p.deliver_late(detail::security_kit_event());
211 expect(code_is(pp.verify_webhook(late.view()), capability_errc::stale), "outside the window is stale");
212 const webhook_delivery garbage{{}, "\x01\x02 not a webhook", {}};
213 expect(code_is(pp.verify_webhook(garbage), capability_errc::signature_invalid), "garbage");
214 });

216 detail::capability_check<Probe>(report, "a statement pages through every fact once", [](Probe& p, const auto& expect) {
217 for (const std::string_view order : {"ord-s1", "ord-s2", "ord-s3"})
218 p.trade(order, p.quantity(), p.consideration());
219 p.settle("ord-s1");
220 std::vector<std::string> seen;
221 security_statement_query q{"", std::nullopt, 2};
222 std::string cursor;
223 for (int guard = 0; guard < 50; ++guard) {
224 const auto page = p.provider().statement(q);
225 expect(page.is_ok(), "every page answers");
226 if (!page.is_ok())
227 return;
228 expect(page->facts.size() <= 2, "a page holds at most the limit");
229 for (const security_fact& f : page->facts) {
230 expect(validate(f).is_ok(), "every fact validates");
231 seen.emplace_back(f.fact_id.view());
232 }
233 if (page->next_cursor.empty())
234 break;
235 cursor = std::string(page->next_cursor.view());
236 q.cursor = cursor;
237 }
238 expect(seen.size() >= 4, "every trade and the settlement are facts");
239 std::sort(seen.begin(), seen.end());
240 expect(std::adjacent_find(seen.begin(), seen.end()) == seen.end(), "no fact twice");
241 security_statement_query other{"", p.quantity().asset_of(), 100};
242 const auto one = p.provider().statement(other);
243 expect(one.is_ok() && one->facts.size() >= 4, "the instrument's facts are its own");
244 });

246 detail::capability_check<Probe>(report, "a provider failure never leaks its response", [](Probe& p, const auto& expect) {
247 p.provider_fails(capability_errc::unavailable, capability_provider_response);
248 const auto r = p.provider().statement({});
249 expect(r.is_err() && r.unwrap_err().code == capability_errc::unavailable, "the provider's failure is returned");
250 if (r.is_err())
251 expect(!capability_error_leaks(r.unwrap_err(), capability_redaction_marker),
252 "neither provider_code nor message carries the provider's response");
253 });
254
255 return report;
