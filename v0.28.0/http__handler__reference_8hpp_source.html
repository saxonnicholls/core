73 capability_result<void> add(http_method method, std::string path, handler h)
74 {
75 if (path.empty() || path.front() != '/')
76 return fail(capability_errc::invalid_argument, {}, "a path starts with /");
77 if (!h)
78 return fail(capability_errc::invalid_argument, {}, "an empty handler");
79 auto& methods = routes_[std::move(path)];

91 capability_result<void> add_pattern(http_method method, std::string pattern, handler h)
92 {
93 if (pattern.empty() || pattern.front() != '/')
94 return fail(capability_errc::invalid_argument, {}, "a path starts with /");
95 if (!h)
96 return fail(capability_errc::invalid_argument, {}, "an empty handler");
97 for (auto& [p, methods] : patterns_)
98 if (p == pattern) {
99 if (!methods.emplace(method, std::move(h)).second)
100 return fail(capability_errc::conflict, {}, "the route is already registered");
101 return ok();
102 }
103 patterns_.emplace_back(std::move(pattern), std::map<http_method, handler>{});
104 patterns_.back().second.emplace(method, std::move(h));
105 return ok();
106 }

111 http_response handle(const http_request& r) const
112 {
113 const auto path = routes_.find(r.path);
114 if (path != routes_.end())
115 return dispatch(path->second, r);
116 for (const auto& [pattern, methods] : patterns_) {
117 http_request captured = r;
118 if (match_route(pattern, r.path, captured.path_params))
119 return dispatch(methods, captured);
120 }
121 return respond(404, "not found");
122 }

126 [[nodiscard]] std::size_t size() const noexcept
127 {
128 std::size_t n = 0;
129 for (const auto& [p, methods] : routes_)
130 n += methods.size();
131 for (const auto& [p, methods] : patterns_)
132 n += methods.size();
133 return n;
134 }

136 private:
137 static http_response dispatch(const std::map<http_method, handler>& methods, const http_request& r)
138 {
139 auto route = methods.find(r.method);
140 const bool head_as_get = route == methods.end() && r.method == http_method::head;
141 if (head_as_get)
142 route = methods.find(http_method::get);
143 if (route == methods.end()) {
144 std::string allow;
145 for (const auto& [m, h] : methods)
146 allow += (allow.empty() ? "" : ", ") + std::string(to_string(m));
147 http_response out = respond(405, "method not allowed");
148 out.headers.push_back({"Allow", std::move(allow)});
149 return out;
150 }
151 http_response out;
152#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
153 try {
154 out = route->second(r);
155 } catch (const std::bad_alloc&) {
156 throw;
157 } catch (...) {
158 return respond(500, "internal error");
159 }
160#else
161 out = route->second(r);
162#endif
163 if (head_as_get)
164 out.body.clear();
165 return out;
166 }
167
168 std::map<std::string, std::map<http_method, handler>, std::less<>> routes_;
169 std::vector<std::pair<std::string, std::map<http_method, handler>>> patterns_;
170 };
