98 capability_result<void> add_key(std::string_view key_id, std::span<const std::byte> material)
99 {
100 if (!is_key_id(key_id))
101 return fail(capability_errc::invalid_argument, {}, "not a key id");
102 if (material.size() != key_size)
103 return fail(capability_errc::invalid_argument, {}, "a key is 32 bytes");
104 std::array<std::byte, key_size> k{};

129 capability_result<std::vector<std::byte>> encrypt(std::string_view key_id, std::span<const std::byte> plaintext,
130 std::span<const std::byte> aad)
131 {
132 if (!is_key_id(key_id))
133 return fail(capability_errc::invalid_argument, {}, "not a key id");
134 const auto it = keys_.find(key_id);
135 if (it == keys_.end())
136 return fail(capability_errc::not_found, {}, "no such key");
137 std::vector<std::byte> out = sealed_header(key_id);
138 const std::size_t header = out.size();
139 ++counter_;
140 for (std::size_t i = 0; i < nonce_size; ++i)
141 out.push_back(i < 8 ? static_cast<std::byte>(counter_ >> (56 - 8 * i)) : std::byte{0});
142 const std::span<const std::byte> nonce(out.data() + header, nonce_size);
143 const std::array<std::byte, nonce_size> n = copy_nonce(nonce);
144 out.resize(out.size() + plaintext.size());
145 xor_stream(it->second, n, plaintext, std::span<std::byte>(out.data() + header + nonce_size, plaintext.size()));
146 const auto t = tag(it->second, std::span<const std::byte>(out.data(), out.size()), aad);
147 out.insert(out.end(), t.begin(), t.end());
148 return ok(std::move(out));
149 }

155 [[nodiscard]] capability_result<std::vector<std::byte>> decrypt(std::span<const std::byte> sealed,
156 std::span<const std::byte> aad) const
157 {
158 const auto id = sealed_key_id(sealed);
159 if (!id)
160 return err(id.unwrap_err());
161 const std::size_t header = 2 + id->size();
162 if (sealed.size() < header + nonce_size + tag_size)
163 return fail(capability_errc::invalid_argument, {}, "the envelope is too short");
164 const auto it = keys_.find(*id);
165 if (it == keys_.end())
166 return fail(capability_errc::not_found, {}, "no such key");
167 const auto body = sealed.first(sealed.size() - tag_size);
168 const auto expected = tag(it->second, body, aad);
169 if (!constant_time_equal(std::span<const std::byte>(expected), sealed.last(tag_size)))
170 return fail(capability_errc::signature_invalid, {}, "the envelope is not authentic");
171 const auto n = copy_nonce(sealed.subspan(header, nonce_size));
172 const auto ciphertext = body.subspan(header + nonce_size);
173 std::vector<std::byte> plain(ciphertext.size());
174 xor_stream(it->second, n, ciphertext, plain);
175 return ok(std::move(plain));
176 }

187 std::array<std::byte, nonce_size> n{};

193 static void xor_stream(const key_type& key, const std::array<std::byte, nonce_size>& nonce,
194 std::span<const std::byte> in, std::span<std::byte> out) noexcept
195 {
196 for (std::size_t block = 0; block * 32 < in.size(); ++block) {
197 std::array<std::byte, 8> counter{};
198 for (std::size_t i = 0; i < 8; ++i)
199 counter[i] = static_cast<std::byte>(static_cast<std::uint64_t>(block) >> (56 - 8 * i));
200 sha256 h;
201 h.update(std::span<const std::byte>(key)).update(std::span<const std::byte>(nonce)).update(std::span<const std::byte>(counter));
202 auto ks = h.finish();
203 for (std::size_t i = 0; i < 32 && block * 32 + i < in.size(); ++i)
204 out[block * 32 + i] = in[block * 32 + i] ^ ks.bytes[i];
205 secure_wipe(ks);
206 }
207 }

209 static std::array<std::byte, tag_size> tag(const key_type& key, std::span<const std::byte> body, std::span<const std::byte> aad) noexcept
210 {
211 std::array<std::byte, 8> aad_length{};
212 for (std::size_t i = 0; i < 8; ++i)
213 aad_length[i] = static_cast<std::byte>(static_cast<std::uint64_t>(aad.size()) >> (56 - 8 * i));
214 sha256 h;
215 h.update(std::string_view{"tag"}).update(std::span<const std::byte>(key)).update(std::span<const std::byte>(aad_length));
216 h.update(aad).update(body);
217 const auto d = h.finish();
218 std::array<std::byte, tag_size> t{};
219 for (std::size_t i = 0; i < tag_size; ++i)
220 t[i] = d.bytes[i];
221 return t;
222 }

224 std::map<std::string, key_type, std::less<>> keys_;
225 std::uint64_t counter_ = 0;
