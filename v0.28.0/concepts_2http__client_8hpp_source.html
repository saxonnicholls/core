71 [[nodiscard]] inline capability_result<void> validate_request(const http_request& r) noexcept
72 {
73 if (r.path.empty() || r.path.front() != '/')
74 return fail(capability_errc::invalid_argument, {}, "a path starts with /");
75 const auto breaks = [](std::string_view s) { return s.find_first_of("\r\n") != std::string_view::npos; };
76 if (breaks(r.path) || breaks(r.query))
77 return fail(capability_errc::invalid_argument, {}, "a line break in the target");
78 for (const http_header& h : r.headers)
79 if (h.name.empty() || breaks(h.name) || breaks(h.value) || h.name.find(':') != std::string::npos)
80 return fail(capability_errc::invalid_argument, {}, "a header that cannot be sent");
81 return ok();
82 }

97 template <http_client C>
98 capability_result<http_response> http_get(C& c, std::string_view target, std::vector<http_header> headers = {})
99 {
100 http_request r;
101 const auto q = target.find('?');
102 r.path = std::string(target.substr(0, q));
103 if (q != std::string_view::npos)
104 r.query = std::string(target.substr(q + 1));
105 r.headers = std::move(headers);
106 return c.send(r);
107 }
