81 [[nodiscard]] capability_conformance_report run_token_verifier_conformance()
82 {
83 capability_conformance_report report;
84
85 detail::capability_check<Probe>(report, "a valid token names its party", [](Probe& p, const auto& expect) {
86 const std::string token = p.valid_token();
87 const auto r = p.verifier().verify(token);
88 expect(r.is_ok(), "the valid token verifies");
89 if (r.is_ok())
90 expect(r->subject == p.subject() && !r->subject.is_nil(), "the claims name the token's party");
91 });
92
93 detail::capability_check<Probe>(report, "garbage and the empty token are unauthorized", [](Probe& p, const auto& expect) {
94 for (const std::string_view t : {std::string_view{}, std::string_view{"garbage"},
95 std::string_view{"eyJhbGciOiJub25lIn0.eyJzdWIiOiJ4In0."}}) {
96 const auto r = p.verifier().verify(t);
97 expect(r.is_err() && r.unwrap_err().code == capability_errc::unauthorized,
98 "the empty token, garbage and an unsigned JWT are unauthorized");
99 }
100 });
101
102 detail::capability_check<Probe>(report, "an altered token is refused", [](Probe& p, const auto& expect) {
103 std::string t = p.valid_token();
104 expect(!t.empty(), "the valid token is not empty");
105 if (t.empty())
106 return;
107 t.back() = t.back() == 'A' ? 'B' : 'A';
108 const auto r = p.verifier().verify(t);
109 expect(r.is_err() && (r.unwrap_err().code == capability_errc::unauthorized ||
110 r.unwrap_err().code == capability_errc::signature_invalid),
111 "a token with its last character changed is unauthorized (or signature_invalid)");
112 });
113
114 detail::capability_check<Probe>(report, "an expired token is stale", [](Probe& p, const auto& expect) {
115 const std::string t = p.expired_token();
116 const auto r = p.verifier().verify(t);
117 expect(r.is_err() && r.unwrap_err().code == capability_errc::stale, "an expired token is stale");
118 });
119
120 detail::capability_check<Probe>(report, "a refused token is never echoed", [](Probe& p, const auto& expect) {
121 const std::string planted = p.valid_token() + std::string(capability_redaction_marker);
122 for (const std::string& t : {std::string(capability_redaction_marker), planted}) {
123 const auto r = p.verifier().verify(t);
124 expect(r.is_err(), "a token carrying the marker is refused");
125 if (r.is_err())
126 expect(!capability_error_leaks(r.unwrap_err(), capability_redaction_marker),
127 "neither provider_code nor message carries the token");
128 }
129 });
130
131 detail::capability_check<Probe>(report, "a provider failure never leaks its response", [](Probe& p, const auto& expect) {
132 const std::string t = p.valid_token();
133 p.provider_fails(capability_errc::unavailable, capability_provider_response);
134 const auto r = p.verifier().verify(t);
135 expect(r.is_err() && r.unwrap_err().code == capability_errc::unavailable, "the provider's failure is returned");
136 if (r.is_err())
137 expect(!capability_error_leaks(r.unwrap_err(), capability_redaction_marker),
138 "neither provider_code nor message carries the provider's response");
139 });
140
141 return report;
142 }
