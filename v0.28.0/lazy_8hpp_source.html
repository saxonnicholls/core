359 template <std::size_t N, class P, class St>
360 [[nodiscard]] constexpr auto take_while(P p, const St& s)
361 {
362 containers::inplace_vector<detail::value_t<St>, N> out;
363 auto state = s.state;
364 while (out.size() < N) {
365 auto r = std::invoke(s.step, state);
366 if (!std::invoke(p, std::as_const(r.first))) break;
367 out.push_back(std::move(r.first));
368 state = std::move(r.second);
369 }
370 return out;
371 }
