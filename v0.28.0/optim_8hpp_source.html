109constexpr T adam_bias_correction(T beta, std::uint64_t step) noexcept(std::is_floating_point_v<T>)
110{
111 return -math::expm1(T(step) * math::log1p(-(T(1) - beta)));

138template <mutable_row P, row G, mutable_row M, mutable_row V>
139 requires std::same_as<row_value_t<P>, row_value_t<G>> && std::same_as<row_value_t<P>, row_value_t<M>>
140 && std::same_as<row_value_t<P>, row_value_t<V>>
141constexpr void adamw_step(P&& params, const G& grads, M&& m, V&& v, std::uint64_t step, const adamw_config<row_value_t<P>>& config)
142{
143 using T = row_value_t<P>;
144 const auto p = as_mutable_row(params);
145 const auto g = as_row(grads);
146 const auto mm = as_mutable_row(m);
147 const auto vv = as_mutable_row(v);
148 assert(step >= 1 && g.size() == p.size() && mm.size() == p.size() && vv.size() == p.size());
149 const T c1 = adam_bias_correction(config.beta1, step);
150 const T c2 = adam_bias_correction(config.beta2, step);
151 for (std::size_t i = 0; i < p.size(); ++i) {
152 mm[i] = config.beta1 * mm[i] + (T(1) - config.beta1) * g[i];
153 vv[i] = config.beta2 * vv[i] + (T(1) - config.beta2) * g[i] * g[i];
154 const T m_hat = mm[i] / c1;
155 const T v_hat = vv[i] / c2;
156 p[i] -= config.learning_rate * (m_hat / (math::sqrt(v_hat) + config.eps) + config.weight_decay * p[i]);
157 }
158}

167template <math::math_number T>
168constexpr T warmup_cosine(std::uint64_t step, std::uint64_t warmup, std::uint64_t total, T max_lr, std::type_identity_t<T> min_lr)
169{
170 assert(warmup <= total);
171 if (step < warmup)
172 return max_lr * T(step + 1) / T(warmup);
173 if (step >= total)
174 return min_lr;
175 const T ratio = T(step - warmup) / T(total - warmup);
176 const T coefficient = T(0.5) * (T(1) + math::cos(std::numbers::pi_v<T> * ratio));
177 return min_lr + coefficient * (max_lr - min_lr);
178}

187template <math::math_number T>
188constexpr T warmup_warmdown(std::uint64_t step, std::uint64_t total, std::uint64_t warmup, std::uint64_t warmdown, T final_fraction)
189{
190 assert(warmup + warmdown <= total);
191 if (step < warmup)
192 return T(step + 1) / T(warmup);
193 if (step + warmdown <= total)
194 return T(1);
195 const T progress = step >= total ? T(0) : T(total - step) / T(warmdown);
196 return progress + (T(1) - progress) * final_fraction;
197}
