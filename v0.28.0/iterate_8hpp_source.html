149 template <typename F, typename T, typename Same = std::equal_to<>>
150 [[nodiscard]] constexpr fixed_point_result<T> fixed_point(F&& f, T x, std::size_t limit, Same same = {})
151 {
152 fixed_point_result<T> out{x, 0, false};
153 while (out.iterations < limit) {
154 T next = std::invoke(f, std::as_const(out.value));
155 ++out.iterations;
156 if (std::invoke(same, std::as_const(out.value), std::as_const(next))) {
157 out.value = std::move(next);
158 out.converged = true;
159 return out;
160 }
161 out.value = std::move(next);
162 }
163 return out;
164 }
