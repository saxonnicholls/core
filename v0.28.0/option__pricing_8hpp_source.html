107 namespace detail {
108 // d1, d2 and the two discount factors, shared by the price and every greek
109 template <std::floating_point T>
110 struct bsm_terms {
111 T d1, d2, sd, qdf, rdf;
112 };
113 template <std::floating_point T>
114 constexpr bsm_terms<T> bsm(const vanilla_option<T>& o) noexcept
115 {
116 const T sd = o.vol * math::sqrt(o.expiry);
117 const T d1 = (math::log(o.spot / o.strike) + (o.rate - o.yield + o.vol * o.vol / 2) * o.expiry) / sd;
118 return {d1, d1 - sd, sd, math::exp(-o.yield * o.expiry), math::exp(-o.rate * o.expiry)};
119 }
120 } // namespace detail

127 template <std::floating_point T>
128 [[nodiscard]] constexpr T black_scholes(const vanilla_option<T>& o) noexcept
129 {
130 const auto b = detail::bsm(o);
131 const T w = o.sign();
132 return w * (o.spot * b.qdf * math::normal_cdf(w * b.d1) - o.strike * b.rdf * math::normal_cdf(w * b.d2));
133 }

140 template <std::floating_point T>
141 [[nodiscard]] constexpr greeks<T> black_scholes_greeks(const vanilla_option<T>& o) noexcept
142 {
143 const auto b = detail::bsm(o);
144 const T w = o.sign(), density = math::normal_pdf(b.d1), nd1 = math::normal_cdf(w * b.d1), nd2 = math::normal_cdf(w * b.d2);
145 const T vega = o.spot * b.qdf * density * math::sqrt(o.expiry);
146 return {.delta = w * b.qdf * nd1,
147 .gamma = b.qdf * density / (o.spot * b.sd),
148 .vega = vega,
149 .vanna = -b.qdf * density * b.d2 / o.vol,
150 .volga = vega * b.d1 * b.d2 / o.vol,
151 .theta = -o.spot * b.qdf * density * o.vol / (2 * math::sqrt(o.expiry)) - w * o.rate * o.strike * b.rdf * nd2
152 + w * o.yield * o.spot * b.qdf * nd1,
153 .rho = w * o.strike * o.expiry * b.rdf * nd2};
154 }

177 template <std::size_t Steps, std::floating_point T>
178 [[nodiscard]] constexpr binomial_price<Steps, T> binomial(const vanilla_option<T>& o, exercise_style style = exercise_style::american) noexcept
179 {
180 static_assert(Steps >= 1, "snicholls::finance::binomial: a tree needs at least one step");
181 const T dt = o.expiry / T(Steps), jump = o.vol * math::sqrt(dt);
182 const T up = math::exp(jump), down = math::exp(-jump), p = (math::exp((o.rate - o.yield) * dt) - down) / (up - down);
183 std::array<T, 2 * Steps + 1> level{}; // level[k] = spot * up^(k - Steps); node j of step t is level Steps + 2j - t
184 for (std::size_t k = 0; k <= 2 * Steps; ++k)
185 level[k] = o.spot * math::exp(jump * (static_cast<T>(k) - static_cast<T>(Steps)));
186 const auto spot_at = [&](std::size_t t, std::size_t j) { return level[Steps + 2 * j - t]; };
187 const bool american = style == exercise_style::american;
188 const T w = o.sign(), strike = o.strike; // by value: the payoff is the inner loop
189 const auto g = [&level, w, strike, american](std::size_t t, std::size_t j) {
190 const T x = w * (level[Steps + 2 * j - t] - strike);
191 return american || t == Steps ? (x > T(0) ? x : T(0)) : std::numeric_limits<T>::lowest();
192 };
193 const auto tree = stochastic::lattice_envelope<Steps>(std::array<T, 2>{1 - p, p}, g, math::exp(-o.rate * dt));
194
195 binomial_price<Steps, T> out{};
196 out.value = tree.value;
197 out.delta = (tree.head[1][1] - tree.head[1][0]) / (spot_at(1, 1) - spot_at(1, 0));
198 if constexpr (Steps >= 2) {
199 const T upper = (tree.head[2][2] - tree.head[2][1]) / (spot_at(2, 2) - spot_at(2, 1));
200 const T lower = (tree.head[2][1] - tree.head[2][0]) / (spot_at(2, 1) - spot_at(2, 0));
201 out.gamma = (upper - lower) / ((spot_at(2, 2) - spot_at(2, 0)) / 2);
202 }
203 for (std::size_t t = 0; t < Steps; ++t)
204 out.boundary[t] = !tree.exercises(t) ? T(0)
205 : o.kind == option_kind::put ? spot_at(t, tree.last_exercise[t])
206 : spot_at(t, tree.first_exercise[t]);
207 return out;
208 }
