107inline constexpr std::array<std::uint8_t, 6> ternary_pow3 = [] {
108 std::array<std::uint8_t, 6> p{};
109 unsigned v = 1;
110 for (auto& e : p) {
111 e = static_cast<std::uint8_t>(v);
112 v *= 3;
113 }
114 return p;
115}();

118inline constexpr std::array<std::array<std::uint8_t, 5>, 256> ternary_trits = [] {
119 std::array<std::array<std::uint8_t, 5>, 256> t{};
120 for (unsigned b = 0; b < 256; ++b)
121 for (unsigned k = 0; k < 5; ++k) {
122 const auto q = static_cast<std::uint8_t>(b * ternary_pow3[k]);
123 t[b][k] = static_cast<std::uint8_t>((unsigned(q) * 3) >> 8);
124 }
125 return t;
126}();

130 // (int) lroundf(x) + 1. Out of a long's range (an infinite inverse scale) lroundf saturates, as Apple's libm
131 // does where the fixture was made: LONG_MAX, whose low word is -1, for +inf; LONG_MIN, low word 0, for -inf and NaN
132 constexpr int trit(float x)
133 {
134 if (!(x >= -2147483648.0f && x < 2147483648.0f))
135 return x > 0.0f ? 0 : 1;
136 return quantize::round_to_int(x) + 1;
137 }

139 // ceil(q · 256 / 243): five trits as a byte
140 constexpr std::uint8_t pack5(unsigned q)
141 {
142 return static_cast<std::uint8_t>((std::uint16_t(std::uint8_t(q)) * 256 + 242) / 243);
143 }
144
145 constexpr int trit_of(std::uint8_t b, std::size_t k)
146 {
147 const auto q = static_cast<std::uint8_t>(b * ternary_pow3[k]);
148 return int((unsigned(q) * 3) >> 8);
149 }

151 constexpr float amax_of(std::span<const float> b)
152 {
153 float amax = 0.0f;
154 for (const float v : b)
155 amax = quantize::fmax(amax, quantize::abs(v));
156 return amax;
157 }

164constexpr void quantize_row(std::span<const float> x, std::span<const float> importance, std::span<block_tq1_0> y)
165{
166 using namespace detail::ternary;
167 detail::quantize::check(256, y.size(), x.size(), importance.size());
168 for (std::size_t i = 0; i < y.size(); ++i) {
169 const auto b = x.subspan(i * 256, 256);
170 const float d = amax_of(b);
171 const float id = d != 0.0f ? 1.0f / d : 0.0f;
172 y[i].d = detail::quantize::fp16(d);
173 const auto code = [&](std::size_t at) {
174 const float v = b[at] * id;
175 return trit(v);
176 };
177 for (std::size_t m = 0; m < 32; ++m) {
178 unsigned q = 0;
179 for (std::size_t n = 0; n < 5; ++n)
180 q = std::uint8_t(std::uint8_t(q * 3) + code(m + n * 32));
181 y[i].qs[m] = pack5(q);
182 }
183 for (std::size_t m = 0; m < 16; ++m) {
184 unsigned q = 0;
185 for (std::size_t n = 0; n < 5; ++n)
186 q = std::uint8_t(std::uint8_t(q * 3) + code(160 + m + n * 16));
187 y[i].qs[32 + m] = pack5(q);
188 }
189 for (std::size_t j = 0; j < 4; ++j) {
190 unsigned q = 0;
191 for (std::size_t m = 0; m < 4; ++m)
192 q = std::uint8_t(std::uint8_t(q * 3) + code(240 + j + m * 4));
193 y[i].qh[j] = pack5(std::uint8_t(q * 3));
194 }
195 }
196}

202constexpr void quantize_row(std::span<const float> x, std::span<const float> importance, std::span<block_tq2_0> y)
203{
204 using namespace detail::ternary;
205 detail::quantize::check(256, y.size(), x.size(), importance.size());
206 for (std::size_t i = 0; i < y.size(); ++i) {
207 const auto b = x.subspan(i * 256, 256);
208 const float d = amax_of(b);
209 const float id = d != 0.0f ? 1.0f / d : 0.0f;
210 y[i].d = detail::quantize::fp16(d);
211 for (std::size_t g = 0; g < 2; ++g)
212 for (std::size_t m = 0; m < 32; ++m) {
213 unsigned q = 0;
214 for (std::size_t n = 0; n < 4; ++n) {
215 const float v = b[128 * g + m + n * 32] * id;
216 q = std::uint8_t(q + unsigned((trit(v) & 3) << (2 * n)));
217 }
218 y[i].qs[32 * g + m] = static_cast<std::uint8_t>(q);
219 }
220 }

226constexpr void dequantize_row(std::span<const block_tq1_0> x, std::span<float> y)
227{
228 assert(y.size() == 256 * x.size());
229 std::size_t o = 0;
230 for (std::size_t i = 0; i < x.size(); ++i) {
231 const float d = to_float(x[i].d);
232 for (std::size_t n = 0; n < 5; ++n)
233 for (std::size_t m = 0; m < 32; ++m)
234 y[o++] = float(ternary_trits[x[i].qs[m]][n] - 1) * d;
235 for (std::size_t n = 0; n < 5; ++n)
236 for (std::size_t m = 0; m < 16; ++m)
237 y[o++] = float(ternary_trits[x[i].qs[32 + m]][n] - 1) * d;
238 for (std::size_t n = 0; n < 4; ++n)
239 for (std::size_t j = 0; j < 4; ++j)
240 y[o++] = float(ternary_trits[x[i].qh[j]][n] - 1) * d;
241 }

264constexpr float dot(std::span<const block_tq1_0> x, std::span<const block_q8_k> y)
265{
266 assert(x.size() == y.size());
267 float sum = 0.0f;
268 for (std::size_t i = 0; i < x.size(); ++i) {
269 int s = 0;
270 for (std::size_t l = 0; l < 5; ++l)
271 for (std::size_t m = 0; m < 32; ++m)
272 s += (detail::ternary::trit_of(x[i].qs[m], l) - 1) * int(y[i].qs[l * 32 + m]);
273 for (std::size_t l = 0; l < 5; ++l)
274 for (std::size_t m = 0; m < 16; ++m)
275 s += (detail::ternary::trit_of(x[i].qs[32 + m], l) - 1) * int(y[i].qs[160 + l * 16 + m]);
276 for (std::size_t l = 0; l < 4; ++l)
277 for (std::size_t j = 0; j < 4; ++j)
278 s += (detail::ternary::trit_of(x[i].qh[j], l) - 1) * int(y[i].qs[240 + l * 4 + j]);
279 const float scale = to_float(x[i].d) * y[i].d;
280 const float term = float(s) * scale;
281 sum += term;
282 }
283 return sum;
284}

290constexpr float dot(std::span<const block_tq2_0> x, std::span<const block_q8_k> y)
291{
292 assert(x.size() == y.size());
293 float sum = 0.0f;
294 for (std::size_t i = 0; i < x.size(); ++i) {
295 int s = 0;
296 for (std::size_t g = 0; g < 2; ++g)
297 for (std::size_t l = 0; l < 4; ++l)
298 for (std::size_t k = 0; k < 32; ++k)
299 s += int(y[i].qs[128 * g + l * 32 + k]) * (int((x[i].qs[32 * g + k] >> (2 * l)) & 3) - 1);
300 const float d = y[i].d * to_float(x[i].d);
301 const float term = float(s) * d;
302 sum += term;
303 }
304 return sum;
