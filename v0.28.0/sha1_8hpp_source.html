70 // Five rounds at a time with the roles of a..e rotated through the calls; each run of 20 rounds has its
71 // own function and constant, so the selection is outside the rounds.
72 template <typename F>
73 static constexpr void twenty(word& a, word& b, word& c, word& d, word& e, word (&w)[16], std::size_t t0, word k,
74 F f) noexcept
75 {
76 const auto round = [&](word x0, word& x1, word x2, word x3, word& x4, std::size_t t) {
77 if (t >= 16)
78 w[t & 15] = rotl<word>(w[(t + 13) & 15] ^ w[(t + 8) & 15] ^ w[(t + 2) & 15] ^ w[t & 15], 1);
79 x4 = static_cast<word>(x4 + rotl<word>(x0, 5) + f(x1, x2, x3) + k + w[t & 15]);
80 x1 = rotl<word>(x1, 30);
81 };
82 for (std::size_t t = t0; t < t0 + 20; t += 5) {
83 round(a, b, c, d, e, t);
84 round(e, a, b, c, d, t + 1);
85 round(d, e, a, b, c, t + 2);
86 round(c, d, e, a, b, t + 3);
87 round(b, c, d, e, a, t + 4);
88 }
89 }

91 static constexpr void compress(state& h, const word (&block)[16]) noexcept
92 {
93 word w[16]{};
94 for (std::size_t i = 0; i < 16; ++i)
95 w[i] = block[i];
96 word a = h[0], b = h[1], c = h[2], d = h[3], e = h[4];
97 twenty(a, b, c, d, e, w, 0, 0x5a827999u, [](word x, word y, word z) { return static_cast<word>((x & y) | (~x & z)); });
98 twenty(a, b, c, d, e, w, 20, 0x6ed9eba1u, [](word x, word y, word z) { return static_cast<word>(x ^ y ^ z); });
99 twenty(a, b, c, d, e, w, 40, 0x8f1bbcdcu, [](word x, word y, word z) { return static_cast<word>((x & y) | (x & z) | (y & z)); });
100 twenty(a, b, c, d, e, w, 60, 0xca62c1d6u, [](word x, word y, word z) { return static_cast<word>(x ^ y ^ z); });
101 h[0] += a;
102 h[1] += b;
103 h[2] += c;
104 h[3] += d;
105 h[4] += e;
106 }
107 };
