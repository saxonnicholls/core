116 capability_result<order> buy(const buy_request& b)
117 {
118 if (b.client_order_id.empty())
119 return fail(capability_errc::invalid_argument, {}, "a buy needs a client order id");
120 auto client = make_id<client_id>(b.client_order_id);
121 if (!client)
122 return err(client.unwrap_err());
123 if (const auto it = buys_.find(b.client_order_id); it != buys_.end()) {
124 if (!it->second.matches(b))
125 return fail(capability_errc::conflict, {}, "client order id reused for another buy");
126 return ok(orders_.at(it->second.order_id));
127 }
128 if (auto e = take_injected())
129 return err(*e);
130 auto q = price(b.unit, b.quote_currency, b.amount);
131 if (!q)
132 return err(q.unwrap_err());
133 if (b.max_price && greater(q->price, *b.max_price))
134 return fail(capability_errc::rejected, {}, "the price is above max_price");
135 if (!has(q->spend))
136 return fail(capability_errc::insufficient_funds);
137 move_balance(q->spend, true);
138 move_balance(q->receive, false);
139 const std::string n = std::to_string(++next_id_);
140 order o{id<provider_id>("mem_ord_" + n), *client, order_status::filled, q->receive, q->spend, q->fee, stamp()};
141 orders_.emplace(std::string(o.order_id.view()), o);
142 onramp_fact f = fact("mem_trd_" + n, onramp_fact_kind::trade_fill);
143 f.debit = venue(*subtract(q->spend, q->fee));
144 f.credit = venue(q->receive);
145 f.fee = venue(q->fee);
146 f.price = q->price;
147 f.order_id = o.order_id;
148 f.client_order_id = o.client_order_id;
149 fill_facts_.push_back(f);
150 buys_.emplace(std::string(b.client_order_id),
151 kept_buy{b.unit, b.quote_currency, b.amount, b.max_price, std::string(o.order_id.view())});
152 return ok(o);
153 }

170 capability_result<withdrawal> withdraw(const withdraw_request& w)
171 {
172 if (w.client_reference.empty())
173 return fail(capability_errc::invalid_argument, {}, "a withdrawal needs a client reference");
174 auto client = make_id<client_id>(w.client_reference);
175 if (!client)
176 return err(client.unwrap_err());
177 if (const auto it = withdrawals_by_ref_.find(w.client_reference); it != withdrawals_by_ref_.end()) {
178 if (!it->second.matches(w))
179 return fail(capability_errc::conflict, {}, "client reference reused for another withdrawal");
180 return ok(withdrawals_.at(it->second.withdrawal_id));
181 }
182 if (auto e = take_injected())
183 return err(*e);
184 const auto a = std::find_if(assets_.begin(), assets_.end(),
185 [&](const onramp_asset& x) { return x.unit == w.amount.asset_of(); });
186 if (a == assets_.end())
187 return fail(capability_errc::invalid_argument, {}, "the venue does not deliver that asset");
188 if (std::find(a->chain_ids.begin(), a->chain_ids.end(), w.chain_id) == a->chain_ids.end())
189 return fail(capability_errc::invalid_argument, {}, "the asset is not withdrawn on that chain");
190 if (!is_evm_address(w.address))
191 return fail(capability_errc::invalid_argument, {}, "the address is not an EVM address");
192 if (w.amount.is_negative() || w.amount.is_zero())
193 return fail(capability_errc::invalid_argument, {}, "a withdrawal is positive");
194 const money fee = fee_for(w.amount.asset_of());
195 const auto gross = add(w.amount, fee);
196 if (!gross || !has(*gross))
197 return fail(capability_errc::insufficient_funds);
198 move_balance(*gross, true);
199 withdrawal out;
200 out.withdrawal_id = id<provider_id>("mem_wd_" + std::to_string(++next_id_));
201 out.client_reference = *client;
202 out.status = withdrawal_status::broadcast;
203 out.amount = w.amount;
204 out.fee = fee;
205 out.chain_id = w.chain_id;
206 out.address = id<short_text>(w.address);
207 out.transaction_hash = id<provider_id>("0x" + std::string(sha256::hash(out.withdrawal_id.view()).hex().view()));
208 out.updated_at = stamp();
209 withdrawals_.emplace(std::string(out.withdrawal_id.view()), out);
210 onramp_fact f = fact(std::string(out.withdrawal_id.view()), onramp_fact_kind::withdrawal);
211 f.debit = venue(w.amount);
212 f.credit = w.amount; // the 1:1 move to the chain asset
213 f.fee = venue(fee);
214 f.withdrawal_id = out.withdrawal_id;
215 f.client_reference = out.client_reference;
216 f.chain_id = w.chain_id;
217 f.address = out.address;
218 f.transaction_hash = out.transaction_hash;
219 withdrawal_facts_.push_back(f);
220 withdrawals_by_ref_.emplace(std::string(w.client_reference),
221 kept_withdrawal{w.amount, w.chain_id, std::string(w.address), std::string(w.memo),
222 std::string(out.withdrawal_id.view())});
223 return ok(out);
224 }

250 capability_result<onramp_statement_page> statement(const onramp_statement_query& q)
251 {
252 const auto& all = q.source == onramp_statement_source::fills ? fill_facts_ : withdrawal_facts_;
253 if (q.limit == 0 || q.limit > 100)
254 return fail(capability_errc::invalid_argument, {}, "a page holds 1 to 100 facts");
255 std::size_t skip = 0;
256 for (const char c : q.cursor) {
257 if (c < '0' || c > '9' || skip > all.size())
258 return fail(capability_errc::invalid_argument, {}, "not a cursor this venue gave");
259 skip = skip * 10 + static_cast<std::size_t>(c - '0');
260 }
261 onramp_statement_page page;
262 const std::size_t end = all.size() > skip ? all.size() - skip : 0;
263 const std::size_t begin = end > q.limit ? end - q.limit : 0;
264 page.facts.assign(all.begin() + static_cast<std::ptrdiff_t>(begin), all.begin() + static_cast<std::ptrdiff_t>(end));
265 if (begin > 0)
266 page.next_cursor = id<provider_id>(std::to_string(skip + (end - begin)));
267 return ok(std::move(page));
268 }

357 static bool greater(const rate& a, const rate& b) noexcept
358 {
359 const auto l = uint256::checked_mul(a.numerator(), b.denominator());
360 const auto r = uint256::checked_mul(b.numerator(), a.denominator());
361 return !l || (r && *l > *r); // an overflowing left side counts as greater: refuse rather than fill

364 capability_result<onramp_quote> price(const asset& unit, const asset& quote_currency, const money& amount) const
365 {
366 if (amount.is_negative() || amount.is_zero())
367 return fail(capability_errc::invalid_argument, {}, "an amount is positive");
368 if (amount.asset_of() != unit && amount.asset_of() != quote_currency)
369 return fail(capability_errc::invalid_argument, {}, "the amount is in neither asset");
370 const auto p = prices_.find(pair_key(unit, quote_currency));
371 if (p == prices_.end())
372 return fail(capability_errc::unsupported, {}, "no price for that pair");
373 money spend, receive, fee;
374 if (amount.asset_of() == quote_currency) {
375 spend = amount;
376 fee = fee_on(amount);
377 const auto per_asset = p->second.inverse();
378 const auto net = subtract(amount, fee);
379 const auto got = per_asset && net ? convert(*net, *per_asset, rounding::toward_zero)
380 : result<conversion, money_errc>(err(money_errc::overflow));
381 if (!got)
382 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
383 receive = got->converted;
384 } else {
385 receive = amount;
386 const auto cost = convert(amount, p->second, rounding::ceiling);
387 if (!cost)
388 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
389 fee = fee_on(cost->converted);
390 spend = *add(cost->converted, fee);
391 }
392 if (receive.is_zero())
393 return fail(capability_errc::invalid_argument, {}, "the ticket buys nothing");
394 const utc_s now = stamp();
395 return ok(onramp_quote{p->second, spend, receive, fee, now, now + quote_lifetime});
396 }

398 money fee_on(const money& m) const
399 {
400 const auto r = rate::make(m.asset_of(), m.asset_of(), fee_bps_, 10000);
401 const auto f = r ? convert(m, *r, rounding::ceiling) : result<conversion, money_errc>(err(money_errc::overflow));
402 return f ? f->converted : money::zero(m.asset_of());
403 }

405 money fee_for(const asset& a) const
406 {
407 const auto it = withdrawal_fees_.find(a.code());
408 return it == withdrawal_fees_.end() || it->second.asset_of() != a ? money::zero(a) : it->second;
409 }

411 bool has(const money& m) const
412 {
413 const auto it = balances_.find(m.asset_of().code());
414 return it != balances_.end() && it->second.asset_of() == m.asset_of() && it->second.magnitude() >= m.magnitude();
415 }

417 void move_balance(const money& m, bool out)
418 {
419 auto it = balances_.find(m.asset_of().code());
420 if (it == balances_.end())
421 it = balances_.emplace(std::string(m.asset_of().code()), money::zero(m.asset_of())).first;
422 it->second = out ? *subtract(it->second, m) : *add(it->second, m);
423 }
