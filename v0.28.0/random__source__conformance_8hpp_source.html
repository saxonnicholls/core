65 template <typename Source>
66 [[nodiscard]] std::vector<std::string> check_random_source(Source& source)
67 {
68 static_assert(is_random_source_like_v<Source>, "snicholls::check_random_source: not a random source");
69 std::vector<std::string> failures;
70 constexpr std::byte sentinel{0xA5};
71 for (const std::size_t n : {std::size_t{1}, std::size_t{7}, std::size_t{33}, std::size_t{255}, std::size_t{256},
72 std::size_t{257}, std::size_t{1000}}) {
73 std::vector<std::byte> buffer(n + 16, sentinel);
74 const std::errc e = source.fill(buffer.data(), n);
75 if (random_failed(e)) {
76 failures.push_back("a fill of " + std::to_string(n) + " bytes failed: " + std::make_error_code(e).message());
77 continue;
78 }
79 bool past_untouched = true;
80 for (std::size_t i = n; i < buffer.size(); ++i)
81 past_untouched = past_untouched && buffer[i] == sentinel;
82 if (!past_untouched)
83 failures.push_back("a fill of " + std::to_string(n) + " bytes wrote past its end");
84 bool all_zero = n >= 7;
85 for (std::size_t i = 0; i < n; ++i)
86 all_zero = all_zero && buffer[i] == std::byte{0};
87 if (all_zero)
88 failures.push_back("a fill of " + std::to_string(n) + " bytes is all zero");
89 }
90 if (const std::errc e = source.fill(nullptr, 0); random_failed(e))
91 failures.push_back("a zero-length fill failed: " + std::make_error_code(e).message());
92 // Padded like the buffers above: a source that writes past its end (one the kit exists to catch) must not
93 // corrupt the kit's own stack.
94 std::array<std::byte, 32 + 16> a{}, b{};
95 const std::errc ea = source.fill(a.data(), 32);
96 const std::errc eb = source.fill(b.data(), 32);
97 if (random_failed(ea) || random_failed(eb))
98 failures.push_back("two 32-byte fills did not both succeed");
99 else if (std::equal(a.begin(), a.begin() + 32, b.begin()))
100 failures.push_back("two fills gave the same bytes");
101 return failures;
102 }

109 template <typename Source>
110 [[nodiscard]] std::vector<std::string> check_random_source_fails_closed(Source& source)
111 {
112 static_assert(is_random_source_like_v<Source>, "snicholls::check_random_source_fails_closed: not a random source");
113 std::vector<std::string> failures;
114 for (const std::size_t n : {std::size_t{1}, std::size_t{32}, std::size_t{257}}) {
115 std::vector<std::byte> buffer(n + 16); // padded: a broken source must not write outside the kit's buffer
116 if (!random_failed(source.fill(buffer.data(), n)))
117 failures.push_back("an unavailable source claimed to fill " + std::to_string(n) + " bytes");
118 }
119 return failures;
120 }
