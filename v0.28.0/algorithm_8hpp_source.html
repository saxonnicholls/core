77 // Whether walking It (step, compare, dereference) can throw: part of every function's noexcept below.
78 template <class It>
79 inline constexpr bool walk_noexcept_v = noexcept(++std::declval<It&>()) &&
80 noexcept(std::declval<const It&>() != std::declval<const It&>()) &&
81 noexcept(*std::declval<It&>());
82
83 // Whether exchanging *a and *b by three moves can throw: traits, not expressions, since `*x = std::move(*x)` in
84 // an unevaluated operand still draws GCC's -Wself-move when A and B are one type.
85 template <class A, class B, class RA = decltype(*std::declval<A&>()), class RB = decltype(*std::declval<B&>()),
86 class VA = std::remove_cv_t<std::remove_reference_t<RA>>>
87 inline constexpr bool exchange_noexcept_v =
88 walk_noexcept_v<A> && walk_noexcept_v<B> &&
89 std::is_nothrow_constructible_v<VA, std::remove_reference_t<RA>&&> &&
90 std::is_nothrow_assignable_v<RA, std::remove_reference_t<RB>&&> && std::is_nothrow_assignable_v<RB, VA&&>;
91
92 // Exchanges two elements by three moves: std::swap is constexpr only from C++20.
93 template <class A, class B>
94 constexpr void exchange_elements(A a, B b) noexcept(exchange_noexcept_v<A, B>)
95 {
96 auto held = std::move(*a);
97 *a = std::move(*b);
98 *b = std::move(held);
99 }
