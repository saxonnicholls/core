126 [[nodiscard]] constexpr zone_info info(std::chrono::sys_seconds t) const noexcept
127 {
128 const auto local_day = std::chrono::floor<std::chrono::days>(t + std_);
129 const int y = static_cast<int>(std::chrono::year_month_day(local_day).year());
130 std::array<transition, 6> ts{};
131 for (int i = 0; i < 3; ++i) {
132 ts[2 * i] = {when(into_, y - 1 + i, std_), true};
133 ts[2 * i + 1] = {when(out_, y - 1 + i, std_ + save_), false};
134 }
135 std::sort(ts.begin(), ts.end(), [](const transition& a, const transition& b) { return a.at < b.at; });
136 std::size_t k = 0;
137 while (k + 1 < ts.size() && ts[k + 1].at <= t)
138 ++k;
139 const bool dst = ts[k].into_dst;
140 return {ts[k].at, k + 1 < ts.size() ? ts[k + 1].at : std::chrono::sys_seconds::max(), dst ? std_ + save_ : std_,
141 dst ? save_ : std::chrono::minutes{0}, dst ? dst_abbrev_ : std_abbrev_};
142 }
143
147 [[nodiscard]] constexpr local_info info(std::chrono::local_seconds l) const noexcept
148 {
149 const std::chrono::seconds dst_offset = std_ + save_;
150 const zone_info as_std = info(std::chrono::sys_seconds(l.time_since_epoch() - std_));
151 const zone_info as_dst = info(std::chrono::sys_seconds(l.time_since_epoch() - dst_offset));
152 const bool std_ok = as_std.offset == std_;
153 const bool dst_ok = as_dst.offset == dst_offset;
154 if (std_ok && dst_ok) // the earlier instant is the daylight one
155 return {local_time_kind::ambiguous, as_dst, as_std};
156 if (std_ok)
157 return {local_time_kind::unique, as_std, as_std};
158 if (dst_ok)
159 return {local_time_kind::unique, as_dst, as_dst};
160 // in the gap: interpreted on the standard offset it is already daylight time, and the other way round
161 return {local_time_kind::nonexistent, as_dst, as_std};
162 }

166 std::chrono::sys_seconds at{};
167 bool into_dst = false;

170 // The instant of `r` in year `y`, on the clock at `offset`.
171 static constexpr std::chrono::sys_seconds when(const dst_rule& r, int y, std::chrono::seconds offset) noexcept
172 {
173 const std::chrono::year year{y};
174 const std::chrono::sys_days day = r.nth >= 5 ? std::chrono::sys_days(year / r.month / r.weekday[std::chrono::last])
175 : std::chrono::sys_days(year / r.month / r.weekday[r.nth]);
176 return std::chrono::sys_seconds(day) + r.at - offset;
177 }
