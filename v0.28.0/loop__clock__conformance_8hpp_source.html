94 template <typename Clock, typename Step>
95 std::vector<std::string> check_loop_clock_moves(Clock& clock, Step step)
96 {
97 std::vector<std::string> fail = check_loop_clock(clock);
98 int woken = 0;
99 clock.set_waker([&woken] { ++woken; });
100 const auto before = clock.now();
101 step();
102 if (!(before < clock.now()))
103 fail.emplace_back("a step moves now forward");
104 if (woken != 1)
105 fail.emplace_back("a step wakes the loop once");
106 if (clock.block_for(before) != 0)
107 fail.emplace_back("the passed deadline no longer blocks");
108 clock.set_waker(std::function<void()>{});
