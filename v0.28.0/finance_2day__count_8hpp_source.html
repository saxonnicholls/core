75 constexpr std::int64_t serial(std::chrono::sys_days d) noexcept
76 {
77 return static_cast<std::int64_t>(d.time_since_epoch().count());
78 }
79 // 360 * (y2 - y1) + 30 * (m2 - m1) + (d2 - d1), after a convention has adjusted d1 and d2
80 constexpr std::int64_t thirty(const civil_date& a, std::int64_t d1, const civil_date& b, std::int64_t d2) noexcept
81 {
82 return 360 * (b.year - a.year) + 30 * (static_cast<std::int64_t>(b.month) - static_cast<std::int64_t>(a.month)) + (d2 - d1);
83 }

206 static constexpr finance::year_fraction year_fraction(std::chrono::sys_days start, std::chrono::sys_days end) noexcept
207 {
208 std::int64_t a = detail::serial(start), b = detail::serial(end), sign = 1;
209 if (b < a) {
210 const std::int64_t t = a;
211 a = b;
212 b = t;
213 sign = -1;
214 }
215 // 365 * 366 = 133590: a common-year day is 366 / 133590 of a year, a leap-year day 365 / 133590
216 const auto weight = [](std::int64_t year) noexcept { return is_leap_year(year) ? std::int64_t{365} : std::int64_t{366}; };
217 const std::int64_t y1 = civil_from_days(a).year, y2 = civil_from_days(b).year;
218 if (y1 == y2)
219 return {sign * (b - a) * weight(y1), 133590};
220 const std::int64_t first = days_from_civil(y1 + 1, 1, 1) - a; // to the end of the first year
221 const std::int64_t last = b - days_from_civil(y2, 1, 1); // from the start of the last year
222 return {sign * ((y2 - y1 - 1) * 133590 + first * weight(y1) + last * weight(y2)), 133590};
223 }
224 };
