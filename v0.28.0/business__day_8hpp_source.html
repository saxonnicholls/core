134 // The first business day at or after `day` stepping by `step` (+1 or -1) days, within the search limit.
135 template <typename C>
136 constexpr result<std::chrono::sys_days, business_day_errc> roll(std::chrono::sys_days day, int step, const C& calendar)
137 {
138 for (int i = 0; i <= max_business_day_search; ++i, day += std::chrono::days(step))
139 if (static_cast<bool>(calendar.is_business_day(day)))
140 return ok(day);
141 return err(business_day_errc::no_business_day);
142 }

144 constexpr std::chrono::month month_of(std::chrono::sys_days day) noexcept
145 {
146 return std::chrono::year_month_day(day).month();
147 }
148
149 // Roll by `step`; if that leaves the month, roll the other way.
150 template <typename C>
151 constexpr result<std::chrono::sys_days, business_day_errc> roll_modified(std::chrono::sys_days day, int step, const C& calendar)
152 {
153 const auto r = roll(day, step, calendar);
154 if (r && month_of(*r) != month_of(day))
155 return roll(day, -step, calendar);
156 return r;
157 }

168 template <holiday_calendar C>
169 [[nodiscard]] constexpr result<std::chrono::sys_days, business_day_errc>
170 adjust(std::chrono::sys_days day, business_day_convention convention, const C& calendar) noexcept(noexcept(calendar.is_business_day(day)))
171 {
172 switch (convention) {
173 case business_day_convention::following: return detail::roll(day, 1, calendar);
174 case business_day_convention::modified_following: return detail::roll_modified(day, 1, calendar);
175 case business_day_convention::preceding: return detail::roll(day, -1, calendar);
176 case business_day_convention::modified_preceding: return detail::roll_modified(day, -1, calendar);
177 case business_day_convention::half_month_modified_following: {
178 const auto r = detail::roll_modified(day, 1, calendar);
179 if (r && static_cast<unsigned>(std::chrono::year_month_day(day).day()) <= 15 &&
180 static_cast<unsigned>(std::chrono::year_month_day(*r).day()) > 15)
181 return detail::roll(day, -1, calendar);
182 return r;
183 }
184 case business_day_convention::nearest:
185 for (int i = 0; i <= max_business_day_search; ++i) {
186 if (static_cast<bool>(calendar.is_business_day(day + std::chrono::days(i))))
187 return ok(day + std::chrono::days(i));
188 if (static_cast<bool>(calendar.is_business_day(day - std::chrono::days(i))))
189 return ok(day - std::chrono::days(i));
190 }
191 return err(business_day_errc::no_business_day);
192 case business_day_convention::unadjusted: break;
193 }
194 return ok(day);
195 }

228 template <holiday_calendar C>
229 [[nodiscard]] constexpr result<std::chrono::sys_days, business_day_errc> advance_business_days(std::chrono::sys_days day, int n,
230 const C& calendar)
231 {
232 const int step = n < 0 ? -1 : 1;
233 for (; n != 0; n -= step) {
234 const auto next = detail::roll(day + std::chrono::days(step), step, calendar);
235 if (!next)
236 return next;
237 day = *next;
238 }
239 return ok(day);
240 }

253 [[nodiscard]] constexpr result<std::chrono::sys_days, business_day_errc>
254 advance_months(std::chrono::sys_days day, int months, business_day_convention convention, bool end_of_month_rule, const C& calendar)

271 template <holiday_calendar C>
272 [[nodiscard]] constexpr std::int64_t count_business_days(std::chrono::sys_days from, std::chrono::sys_days to, const C& calendar,
273 bool include_first = true, bool include_last = false)
274 {
275 const auto count = [&calendar](std::chrono::sys_days a, std::chrono::sys_days b, bool first, bool last) {
276 std::int64_t n = last && static_cast<bool>(calendar.is_business_day(b)) ? 1 : 0;
277 for (auto x = first ? a : a + std::chrono::days(1); x < b; x += std::chrono::days(1))
278 n += static_cast<bool>(calendar.is_business_day(x)) ? 1 : 0;
279 return n;
280 };
281 if (from < to)
282 return count(from, to, include_first, include_last);
283 if (to < from)
284 return -count(to, from, include_last, include_first);
285 return include_first && include_last && static_cast<bool>(calendar.is_business_day(from)) ? 1 : 0;
286 }

328 {
329 const bool a = static_cast<bool>(first.is_business_day(day));
