95 // GPT-2's bytes_to_unicode: printable Latin-1 bytes map to themselves, the other 68 to U+0100 onwards in order
96 consteval std::array<char32_t, 256> make_byte_to_unicode()
97 {
98 std::array<char32_t, 256> map{};
99 char32_t next = 256;
100 for (std::size_t b = 0; b < 256; ++b) {
101 const bool printable = (b >= 0x21 && b <= 0x7E) || (b >= 0xA1 && b <= 0xAC) || (b >= 0xAE && b <= 0xFF);
102 map[b] = printable ? static_cast<char32_t>(b) : next++;
103 }
104 return map;
105 }
106} // namespace detail::bpe_merges

164 template <typename Key>
165 constexpr bool before(const merge_entry<Key>& x, const merge_entry<Key>& y) noexcept
166 {
167 return x.key < y.key || (x.key == y.key && x.a < y.a);
168 }

170 template <typename Key>
171 constexpr void push(merge_entry<Key>* heap, std::size_t& size, merge_entry<Key> e) noexcept
172 {
173 std::size_t i = size++;
174 while (i > 0) {
175 const std::size_t p = (i - 1) / 2;
176 if (!before(e, heap[p]))
177 break;
178 heap[i] = heap[p];
179 i = p;
180 }
181 heap[i] = e;
182 }
183
184 template <typename Key>
185 constexpr merge_entry<Key> pop(merge_entry<Key>* heap, std::size_t& size) noexcept
186 {
187 const merge_entry<Key> top = heap[0];
188 const merge_entry<Key> e = heap[--size];
189 std::size_t i = 0;
190 for (;;) {
191 std::size_t c = 2 * i + 1;
192 if (c >= size)
193 break;
194 if (c + 1 < size && before(heap[c + 1], heap[c]))
195 ++c;
196 if (!before(heap[c], e))
197 break;
198 heap[i] = heap[c];
199 i = c;
200 }
201 if (size > 0)
202 heap[i] = e;
203 return top;
204 }

206 // Pushes the candidate of the part at i and its successor, if they merge
207 template <typename Key, typename Priority>
208 constexpr void candidate(std::string_view piece, Priority& priority, merge_entry<Key>* link, merge_entry<Key>* heap,
209 std::size_t& size, std::uint32_t i)
210 {
211 const std::uint32_t j = link[i].a;
212 if (j >= piece.size()) {
213 link[i].key = merge_never<Key>;
214 return;
215 }
216 const std::uint32_t k = link[j].a;
217 const Key key = priority(piece.substr(i, j - i), piece.substr(j, k - j));
218 link[i].key = key;
219 if (key != merge_never<Key>)
220 push(heap, size, merge_entry<Key>{key, i, k});
221 }

237template <typename Priority, typename Token, typename ByteToken,
238 typename Key = std::remove_cvref_t<std::invoke_result_t<Priority&, std::string_view, std::string_view>>>
239constexpr result<std::size_t, bpe_error> merge_encode(std::string_view piece, Priority&& priority, Token&& token, ByteToken&& byte_token,
240 std::span<std::uint32_t> out, std::type_identity_t<std::span<merge_entry<Key>>> scratch,
241 merge_options options = {})
242{
243 namespace d = detail::bpe_merges;
244 const std::size_t n = piece.size();
245 if (n == 0)
246 return ok(std::size_t{0});
247 if (options.ignore_merges)
248 if (const std::uint32_t whole = token(piece); whole != no_token) {
249 if (out.empty())
250 return err(bpe_error::output_too_small);
251 out[0] = whole;
252 return ok(std::size_t{1});
253 }
254 if (out.size() < n)
255 return err(bpe_error::output_too_small);
256 if (scratch.size() < merge_scratch_size(n))
257 return err(bpe_error::scratch_too_small);
258
259 // link[i], for a part starting at byte i: a = the next part's start, b = the previous part's start (n for
260 // none) with d::dead set once merged away, key = the priority of the part and its successor
261 merge_entry<Key>* link = scratch.data();
262 merge_entry<Key>* heap = link + n;
263 std::size_t size = 0;
264 std::uint32_t prev = static_cast<std::uint32_t>(n);
265 for (std::uint32_t i = 0; i < n;) {
266 const std::uint32_t len = options.units == merge_units::bytes ? 1u : decode_utf8(piece, i).length;
267 link[i] = {merge_never<Key>, i + len, prev};
268 prev = i;
269 i += len;
270 }
271 for (std::uint32_t i = 0; i < n; i = link[i].a)
272 d::candidate(piece, priority, link, heap, size, i);
273
274 while (size > 0) {
275 const merge_entry<Key> e = d::pop(heap, size);
276 const std::uint32_t i = e.a;
277 if ((link[i].b & d::dead) || link[i].key != e.key || link[i].a >= n || link[link[i].a].a != e.b)
278 continue; // merged away, or its pair has changed since
279 const std::uint32_t j = link[i].a, k = link[j].a;
280 link[j].b |= d::dead;
281 link[i].a = k;
282 if (k < n)
283 link[k].b = i;
284 d::candidate(piece, priority, link, heap, size, i);
285 if (const std::uint32_t p = link[i].b; p < n)
286 d::candidate(piece, priority, link, heap, size, p);
287 }
288
289 std::size_t count = 0;
290 for (std::uint32_t i = 0; i < n; i = link[i].a) {
291 const std::string_view part = piece.substr(i, link[i].a - i);
292 if (const std::uint32_t id = token(part); id != no_token) {
293 out[count++] = id;
294 continue;
295 }
296 for (std::size_t k = 0; k < part.size();) {
297 // A byte of the part, or with byte_level the byte its next code point spells
298 std::uint32_t b = static_cast<unsigned char>(part[k]);
299 std::size_t len = 1;
300 if (options.byte_level) {
301 const utf8_decoded d = decode_utf8(part, k);
302 if (!d.valid || d.code_point >= unicode_to_byte.size() || unicode_to_byte[d.code_point] < 0)
303 return err(bpe_error::unknown_bytes);
304 b = static_cast<std::uint32_t>(unicode_to_byte[d.code_point]);
305 len = d.length;
306 }
307 const std::uint32_t id = byte_token(static_cast<unsigned char>(b));
308 if (id == no_token)
309 return err(bpe_error::unknown_bytes);
310 out[count++] = id;
311 k += len;
312 }
313 }
314 return ok(count);
315}

319struct merge_table
320{
321 std::span<const std::pair<std::string_view, std::string_view>> merges;
322 std::span<const std::uint32_t> order;
323
328 constexpr std::uint32_t rank(std::string_view left, std::string_view right) const noexcept
329 {
330 const auto less = [&](std::uint32_t m) {
331 const auto& [l, r] = merges[m];
332 return l < left || (l == left && r < right);
333 };
334 std::size_t lo = 0, hi = order.size();
335 while (lo < hi) {
336 const std::size_t mid = lo + (hi - lo) / 2;
337 if (less(order[mid]))
338 lo = mid + 1;
339 else
340 hi = mid;
341 }
342 if (lo < order.size() && merges[order[lo]].first == left && merges[order[lo]].second == right)
343 return order[lo];
344 return merge_never<std::uint32_t>;
345 }
346
351 constexpr std::uint32_t operator()(std::string_view left, std::string_view right) const noexcept { return rank(left, right); }
352};

357constexpr void build_merge_index(std::span<const std::pair<std::string_view, std::string_view>> merges, std::span<std::uint32_t> order)
358{
359 for (std::size_t i = 0; i < merges.size(); ++i)
360 order[i] = static_cast<std::uint32_t>(i);
361 const auto before = [&](std::uint32_t x, std::uint32_t y) { return merges[x] < merges[y] || (merges[x] == merges[y] && x < y); };
362 detail::vocab_index::sort(order.first(merges.size()), before);
363}

381constexpr result<std::size_t, bpe_error> byte_level_encode(std::string_view bytes, std::span<char> out) noexcept
382{
383 std::size_t at = 0;
384 for (const char c : bytes) {
385 const char32_t cp = byte_to_unicode[static_cast<unsigned char>(c)];
386 const std::size_t len = utf8_length(cp);
387 if (out.size() - at < len)
388 return err(bpe_error::output_too_small);
389 at += encode_utf8(cp, out.data() + at);
390 }
391 return ok(at);
392}

398constexpr result<std::size_t, bpe_error> byte_level_decode(std::string_view text, std::span<char> out) noexcept
399{
400 std::size_t at = 0;
401 for (std::size_t i = 0; i < text.size();) {
402 const utf8_decoded d = decode_utf8(text, i);
403 if (!d.valid || d.code_point >= unicode_to_byte.size() || unicode_to_byte[d.code_point] < 0)
404 return err(bpe_error::unknown_bytes);
405 if (at == out.size())
406 return err(bpe_error::output_too_small);
407 out[at++] = static_cast<char>(unicode_to_byte[d.code_point]);
408 i += d.length;
409 }
410 return ok(at);
411}
