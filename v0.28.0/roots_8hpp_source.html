116template <typename T>
117constexpr bool opposite_(T a, T b) { return (a < T(0) && b > T(0)) || (a > T(0) && b < T(0)); }

119// Brent's convergence radius: half the caller's absolute tolerance plus two units of relative precision, never 0.
120template <typename T>
121constexpr T root_tol_(T x, T tolerance)
122{
123 return T(2) * std::numeric_limits<T>::epsilon() * abs_(x) + tolerance / T(2) + std::numeric_limits<T>::min();
124}
125} // namespace detail

137template <math_number T, typename F>
138 requires std::invocable<F&, T>
139constexpr root_result<T> bisect(F&& f, T lo, T hi, std::type_identity_t<T> tolerance = T(0),
140 int max_iterations = default_root_iterations<T>)
141{
142 if (hi < lo)
143 std::swap(lo, hi);
144 T flo = static_cast<T>(f(lo)), fhi = static_cast<T>(f(hi));
145 if (flo == T(0)) return {lo, flo, 0, root_status::converged};
146 if (fhi == T(0)) return {hi, fhi, 0, root_status::converged};
147 if (!detail::opposite_(flo, fhi))
148 return {lo, flo, 0, root_status::not_bracketed};
149 const auto best = [&](int n, root_status s) {
150 return detail::abs_(flo) <= detail::abs_(fhi) ? root_result<T>{lo, flo, n, s} : root_result<T>{hi, fhi, n, s};
151 };
152 for (int n = 0;; ++n) {
153 const T mid = lo + (hi - lo) / T(2);
154 if ((hi - lo) / T(2) <= detail::root_tol_(mid, tolerance) || mid == lo || mid == hi)
155 return best(n, root_status::converged);
156 if (n == max_iterations)
157 return best(n, root_status::max_iterations);
158 const T fm = static_cast<T>(f(mid));
159 if (fm == T(0))
160 return {mid, fm, n + 1, root_status::converged};
161 if (detail::opposite_(flo, fm)) { hi = mid; fhi = fm; }
162 else { lo = mid; flo = fm; }
163 }
164}

181template <math_number T, typename F>
182 requires std::invocable<F&, T>
183constexpr root_result<T> brent(F&& f, T lo, T hi, std::type_identity_t<T> tolerance = T(0),
184 int max_iterations = default_root_iterations<T>)
185{
186 using detail::abs_;
187 T a = lo, b = hi;
188 T fa = static_cast<T>(f(a)), fb = static_cast<T>(f(b));
189 if (fa == T(0)) return {a, fa, 0, root_status::converged};
190 if (fb == T(0)) return {b, fb, 0, root_status::converged};
191 if (!detail::opposite_(fa, fb))
192 return {a, fa, 0, root_status::not_bracketed};
193 T c = a, fc = fa, d = b - a, e = d;
194 for (int n = 0;; ++n) {
195 if (detail::opposite_(fb, fc) == false) { c = a; fc = fa; d = e = b - a; } // keep the root between b and c
196 if (abs_(fc) < abs_(fb)) { a = b; b = c; c = a; fa = fb; fb = fc; fc = fa; } // b is the best point
197 const T tol1 = detail::root_tol_(b, tolerance);
198 const T xm = (c - b) / T(2);
199 if (abs_(xm) <= tol1 || fb == T(0))
200 return {b, fb, n, root_status::converged};
201 if (n == max_iterations)
202 return {b, fb, n, root_status::max_iterations};
203 if (abs_(e) >= tol1 && abs_(fa) > abs_(fb)) {
204 T p, q;
205 const T s = fb / fa;
206 if (a == c) { // the secant step
207 p = T(2) * xm * s;
208 q = T(1) - s;
209 } else { // inverse quadratic interpolation
210 const T qa = fa / fc, r = fb / fc;
211 p = s * (T(2) * xm * qa * (qa - r) - (b - a) * (r - T(1)));
212 q = (qa - T(1)) * (r - T(1)) * (s - T(1));
213 }
214 if (p > T(0)) q = -q;
215 else p = -p;
216 const T bound1 = T(3) * xm * q - abs_(tol1 * q), bound2 = abs_(e * q);
217 if (T(2) * p < (bound1 < bound2 ? bound1 : bound2)) { e = d; d = p / q; } // accept the interpolation
218 else { d = xm; e = d; } // bisect
219 } else {
220 d = xm;
221 e = d;
222 }
223 a = b;
224 fa = fb;
225 b += abs_(d) > tol1 ? d : (xm > T(0) ? tol1 : -tol1);
226 fb = static_cast<T>(f(b));
227 }
228}

245template <math_number T, typename FDF>
246 requires std::invocable<FDF&, T>
247constexpr root_result<T> newton_safeguarded(FDF&& f_and_df, T lo, T hi, std::type_identity_t<T> x0,
248 std::type_identity_t<T> tolerance = T(0),
249 int max_iterations = default_root_iterations<T>)
250{
251 using detail::abs_;
252 const auto [flo_, dlo_] = f_and_df(lo);
253 const auto [fhi_, dhi_] = f_and_df(hi);
254 (void)dlo_;
255 (void)dhi_;
256 const T flo = static_cast<T>(flo_), fhi = static_cast<T>(fhi_);
257 if (flo == T(0)) return {lo, flo, 0, root_status::converged};
258 if (fhi == T(0)) return {hi, fhi, 0, root_status::converged};
259 if (!detail::opposite_(flo, fhi))
260 return {lo, flo, 0, root_status::not_bracketed};
261 T neg = flo < T(0) ? lo : hi, pos = flo < T(0) ? hi : lo; // f(neg) < 0 < f(pos)
262 const T low = lo < hi ? lo : hi, high = lo < hi ? hi : lo;
263 T x = x0 > low && x0 < high ? x0 : low + (high - low) / T(2);
264 T step = high - low, step_before = step;
265 T best_x = abs_(flo) <= abs_(fhi) ? lo : hi, best_f = abs_(flo) <= abs_(fhi) ? flo : fhi;
266 for (int n = 1;; ++n) {
267 const auto [fx_, dx_] = f_and_df(x);
268 const T fx = static_cast<T>(fx_), dfx = static_cast<T>(dx_);
269 if (!(abs_(fx) > abs_(best_f))) { best_x = x; best_f = fx; }
270 const T tol1 = detail::root_tol_(x, tolerance);
271 if (fx == T(0) || abs_(step) <= tol1 || abs_(pos - neg) / T(2) <= tol1)
272 return {x, fx, n, root_status::converged};
273 if (n == max_iterations)
274 return {best_x, best_f, n, root_status::max_iterations};
275 if (fx < T(0)) neg = x;
276 else pos = x;
277 const T a = neg < pos ? neg : pos, b = neg < pos ? pos : neg;
278 const T xn = x - fx / dfx;
279 step_before = step;
280 if (dfx != T(0) && xn > a && xn < b && abs_(T(2) * fx) <= abs_(step_before * dfx)) {
281 step = x - xn;
282 x = xn;
283 } else {
284 step = (b - a) / T(2);
285 x = a + step;
286 }
287 }

303template <math_number T, typename F>
304 requires std::invocable<F&, T>
305constexpr bracket_result<T> bracket(F&& f, T x0, std::type_identity_t<T> step, std::type_identity_t<T> growth = T(1.6),
306 int max_iterations = 60)
307{
308 step = detail::abs_(step);
309 T a = x0 - step, b = x0 + step;
310 if (!(step > T(0)) || detail::inf_(step))
311 return {a, b, T(0), T(0), 0, root_status::not_bracketed};
312 T fa = static_cast<T>(f(a)), fb = static_cast<T>(f(b));
313 for (int n = 0;; ++n) {
314 if (fa == T(0) || fb == T(0) || detail::opposite_(fa, fb))
315 return {a, b, fa, fb, n, root_status::converged};
316 if (n == max_iterations)
317 return {a, b, fa, fb, n, root_status::max_iterations};
318 if (detail::abs_(fa) < detail::abs_(fb)) { a -= growth * (b - a); fa = static_cast<T>(f(a)); }
319 else { b += growth * (b - a); fb = static_cast<T>(f(b)); }
320 }
321}
