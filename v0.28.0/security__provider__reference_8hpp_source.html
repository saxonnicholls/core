155 capability_result<security_statement_page> statement(const security_statement_query& q)
156 {
157 if (auto bad = take_injected())
158 return err(*bad);
159 std::size_t from = 0;
160 if (!q.cursor.empty()) {
161 std::int64_t at = 0;
162 if (!detail::memory_webhook::parse_int(q.cursor, at) || at < 0 || static_cast<std::size_t>(at) > facts_.size())
163 return fail(capability_errc::invalid_argument, {}, "not a cursor this provider gave");
164 from = static_cast<std::size_t>(at);
165 }
166 security_statement_page page;
167 std::size_t i = from;
168 for (; i < facts_.size() && page.facts.size() < q.limit; ++i) {
169 const security_fact& f = facts_[i];
170 if (!q.instrument || f.instrument == *q.instrument || f.other_instrument == q.instrument)
171 page.facts.push_back(f);
172 }
173 if (i < facts_.size())
174 page.next_cursor = *provider_id::make(std::to_string(i));
175 return ok(std::move(page));
176 }

182 capability_result<std::vector<security_holding>> holdings(std::chrono::sys_days as_of)
183 {
184 if (auto bad = take_injected())
185 return err(*bad);
186 std::map<std::string, security_holding, std::less<>> by_code;
187 const auto move = [&](const money& q, bool traded, bool settled) {
188 const std::string key = std::string(q.asset_of().code()) + ":" + std::to_string(q.asset_of().scale());
189 auto [it, fresh] = by_code.try_emplace(key, security_holding{q.asset_of(), money::zero(q.asset_of()),
190 money::zero(q.asset_of())});
191 if (traded)
192 it->second.traded = *add(it->second.traded, q);
193 if (settled)
194 it->second.settled = *add(it->second.settled, q);
195 };
196 for (const security_fact& f : facts_) {
197 const bool settlement = f.kind == security_fact_kind::settle;
198 const bool traded = !settlement && f.kind != security_fact_kind::settle_failed && f.dates.trade_at.days() <= as_of;
199 const bool settled = f.kind != security_fact_kind::settle_failed && f.dates.settled_at &&
200 f.dates.settled_at->days() <= as_of;
201 move(f.quantity, traded, settled);
202 if (f.other_quantity)
203 move(*f.other_quantity, traded, settled);
204 }
205 std::vector<security_holding> out;
206 for (const auto& [code, h] : by_code)
207 if (!h.traded.is_zero() || !h.settled.is_zero())
208 out.push_back(h);
209 return ok(std::move(out));
210 }

216 capability_result<security_fact> book(security_fact f)
217 {
218 if (f.source.empty())
219 f.source = "memory";
220 if (f.provider_account.empty())
221 f.provider_account = provider_account_;
222 if (f.fact_id.empty())
223 f.fact_id = *fact_id::make("mem_sf_" + std::to_string(++next_id_));
224 if (f.dates.known_at == utc_s{})
225 f.dates.known_at = stamp();
226 if (auto valid = validate(f); !valid)
227 return err(valid.unwrap_err());
228 for (const security_fact& g : facts_)
229 if (g.fact_id == f.fact_id) {
230 if (g.kind == f.kind && g.quantity == f.quantity && g.consideration == f.consideration &&
231 g.dates.same_content(f.dates))
232 return ok(g);
233 return fail(capability_errc::conflict, {}, "the fact id is booked with other content");
234 }
235 facts_.push_back(f);
236 return ok(f);
237 }

246 capability_result<security_fact> execute(std::string_view order, const money& quantity, const money& consideration,
247 std::optional<money> fees = std::nullopt)
248 {
249 auto id = make_id<provider_id>(order);
250 if (!id || order.empty() || quantity.is_zero())
251 return fail(capability_errc::invalid_argument, {}, "an order id and a quantity");
252 for (const security_fact& f : facts_)
253 if (f.refs.order == *id)
254 return fail(capability_errc::invalid_argument, {}, "the order is already filled");
255 security_fact f;
256 f.kind = quantity.is_negative() ? security_fact_kind::sell : security_fact_kind::buy;
257 f.instrument = quantity.asset_of();
258 f.quantity = quantity;
259 f.consideration = consideration;
260 f.fees = fees ? *fees : money::zero(consideration.asset_of());
261 f.dates.trade_at = stamp();
262 f.dates.known_at = f.dates.trade_at;
263 f.dates.settles_on = f.dates.trade_at.days() + settlement_lag;
264 f.dates.settles_on_estimated = true;
265 f.refs.order = *id;
266 f.refs.execution = *provider_id::make("mem_exe_" + std::to_string(next_id_ + 1));
267 return book(f);
268 }

315 capability_result<security_fact> conclude(std::string_view order, security_fact_kind kind)
316 {
317 const security_fact* trade = nullptr;
318 for (const security_fact& f : facts_)
319 if (f.refs.order.view() == order && !order.empty()) {
320 if (f.kind == security_fact_kind::settle || f.kind == security_fact_kind::settle_failed)
321 return fail(capability_errc::conflict, {}, "the order is already settled or failed");
322 trade = &f;
323 }
324 if (!trade)
325 return fail(capability_errc::not_found);
326 security_fact f = *trade;
327 f.fact_id = {};
328 f.kind = kind;
329 f.dates.known_at = stamp();
330 if (kind == security_fact_kind::settle)
331 f.dates.settled_at = stamp();
332 f.refs.settlement = *provider_id::make("mem_set_" + std::to_string(next_id_ + 1));
333 return book(f);
334 }
