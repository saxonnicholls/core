102 constexpr ratio(std::int64_t numerator, std::int64_t denominator) noexcept
103 {
104 assert(denominator != 0 && numerator != min_ && denominator != min_);
105 if (denominator < 0) {
106 numerator = -numerator;
107 denominator = -denominator;
108 }
109 const std::int64_t g = std::gcd(numerator, denominator);
110 numerator_ = g == 0 ? 0 : numerator / g;
111 denominator_ = g == 0 ? 1 : denominator / g;
112 }

118 [[nodiscard]] static constexpr result<ratio, ratio_errc> make(std::int64_t numerator, std::int64_t denominator) noexcept
119 {
120 if (denominator == 0)
121 return err(ratio_errc::zero_denominator);
122 if (numerator == min_ || denominator == min_)
123 return err(ratio_errc::out_of_range);
124 return ok(ratio{numerator, denominator});
125 }

131 [[nodiscard]] static constexpr result<ratio, ratio_errc> parse(std::string_view text) noexcept
132 {
133 const bool negative = !text.empty() && text.front() == '-';
134 if (negative)
135 text.remove_prefix(1);
136 std::int64_t n = 0, d = 1;
137 std::size_t i = 0;
138 const auto digits = [&](std::int64_t& v, bool scale) -> result<std::size_t, ratio_errc> {
139 const std::size_t start = i;
140 for (; i < text.size() && text[i] >= '0' && text[i] <= '9'; ++i) {
141 const auto t = checked_mul<std::int64_t>(v, 10);
142 const auto u = t ? checked_add<std::int64_t>(*t, text[i] - '0') : t;
143 const auto s = scale ? checked_mul<std::int64_t>(d, 10) : std::optional<std::int64_t>{d};
144 if (!u || !s)
145 return err(ratio_errc::out_of_range);
146 v = *u;
147 d = *s;
148 }
149 return ok(i - start);
150 };
151 const auto whole = digits(n, false);
152 if (!whole)
153 return err(whole.unwrap_err_unchecked());
154 if (*whole == 0)
155 return err(ratio_errc::malformed);
156 if (i < text.size() && text[i] == '/') {
157 ++i;
158 std::int64_t den = 0;
159 const auto below = digits(den, false);
160 if (!below)
161 return err(below.unwrap_err_unchecked());
162 if (*below == 0 || i != text.size())
163 return err(ratio_errc::malformed);
164 return make(negative ? -n : n, den);
165 }
166 if (i < text.size() && text[i] == '.') {
167 ++i;
168 const auto fraction = digits(n, true);
169 if (!fraction)
170 return err(fraction.unwrap_err_unchecked());
171 if (*fraction == 0)
172 return err(ratio_errc::malformed);
173 }
174 if (i < text.size() && text[i] == '%') {
175 ++i;
176 const auto hundredths = checked_mul<std::int64_t>(d, 100);
177 if (!hundredths)
178 return err(ratio_errc::out_of_range);
179 d = *hundredths;
180 }
181 if (i != text.size())
182 return err(ratio_errc::malformed);
183 return make(negative ? -n : n, d);
184 }

220 friend constexpr bool operator==(const ratio&, const ratio&) noexcept = default;
225 friend constexpr std::strong_ordering operator<=>(const ratio& a, const ratio& b) noexcept
226 {
227 // Cross-multiplied while the products fit (one multiply each, the usual case); otherwise
228 // compare a/b with c/d (b, d > 0) by their continued fractions: whole parts first, then the remainders'
229 // inverses, with the order flipping at each inversion. Every step divides; nothing is multiplied.
230 const auto left = checked_mul(a.numerator_, b.denominator_), right = checked_mul(b.numerator_, a.denominator_);
231 if (left && right)
232 return *left <=> *right;
233 std::int64_t p = a.numerator_, q = a.denominator_, r = b.numerator_, s = b.denominator_;
234 bool flipped = false;
235 for (;;) {
236 const std::int64_t x = floor_div(p, q), y = floor_div(r, s);
237 const std::int64_t pr = p - x * q, rr = r - y * s; // 0 <= remainder < denominator
238 if (x != y || pr == 0 || rr == 0) {
239 const std::strong_ordering o = x != y ? x <=> y : pr <=> rr; // a zero remainder is the smaller
240 return flipped ? 0 <=> o : o;
241 }
242 p = q;
243 q = pr;
244 r = s;
245 s = rr;
246 flipped = !flipped;
247 }
248 }
249
250 private:
251 static constexpr std::int64_t min_ = std::numeric_limits<std::int64_t>::min();
252 static constexpr std::int64_t floor_div(std::int64_t p, std::int64_t q) noexcept
253 {
254 const std::int64_t d = p / q;
255 return (p % q != 0 && p < 0) ? d - 1 : d;
256 }
257
258 std::int64_t numerator_ = 0;
259 std::int64_t denominator_ = 1;

267 [[nodiscard]] constexpr std::optional<ratio> multiply(const ratio& a, const ratio& b) noexcept
268 {
269 const std::int64_t g1 = std::gcd(a.numerator(), b.denominator()), g2 = std::gcd(b.numerator(), a.denominator());
270 const auto n = checked_mul<std::int64_t>(a.numerator() / g1, b.numerator() / g2);
271 const auto d = checked_mul<std::int64_t>(a.denominator() / g2, b.denominator() / g1);
272 if (!n || !d)
273 return std::nullopt;
274 const auto r = ratio::make(*n, *d);
275 return r ? std::optional<ratio>{*r} : std::nullopt;
276 }

283 [[nodiscard]] constexpr std::optional<ratio> add(const ratio& a, const ratio& b) noexcept
284 {
285 const std::int64_t g = std::gcd(a.denominator(), b.denominator());
286 const auto left = checked_mul<std::int64_t>(a.numerator(), b.denominator() / g);
287 const auto right = checked_mul<std::int64_t>(b.numerator(), a.denominator() / g);
288 const auto n = left && right ? checked_add<std::int64_t>(*left, *right) : std::nullopt;
289 const auto d = checked_mul<std::int64_t>(a.denominator(), b.denominator() / g);
290 if (!n || !d)
291 return std::nullopt;
292 const auto r = ratio::make(*n, *d);
293 return r ? std::optional<ratio>{*r} : std::nullopt;
294 }
