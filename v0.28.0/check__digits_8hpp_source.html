88 constexpr bool digit(char c) noexcept { return c >= '0' && c <= '9'; }
89 constexpr bool upper(char c) noexcept { return c >= 'A' && c <= 'Z'; }
90 // 0-9 for a digit, 10-35 for A-Z
91 constexpr int value(char c) noexcept { return digit(c) ? c - '0' : c - 'A' + 10; }

93 constexpr bool all_digits(std::string_view s) noexcept
94 {
95 for (const char c : s)
96 if (!digit(c))
97 return false;
98 return true;
99 }
100 constexpr bool all_alphanumeric(std::string_view s) noexcept
101 {
102 for (const char c : s)
103 if (!digit(c) && !upper(c))
104 return false;
105 return true;

108 // Luhn over the digits of `s` read right to left (a letter A-Z as its two digits), doubling the rightmost
109 // when `double_first`. The sum modulo 10.
110 constexpr int luhn_sum(std::string_view s, bool double_first) noexcept
111 {
112 int sum = 0;
113 bool twice = double_first;
114 const auto add = [&](int d) noexcept {
115 if (twice)
116 d = d * 2 > 9 ? d * 2 - 9 : d * 2;
117 sum += d;
118 twice = !twice;
119 };
120 for (std::size_t i = s.size(); i-- > 0;) {
121 const int v = value(s[i]);
122 add(v % 10);
123 if (v > 9)
124 add(v / 10);
125 }
126 return sum % 10;
127 }

129 // ISO 7064 MOD 97-10 remainder of the number spelled by `s` (a letter A-Z as its two digits), from `r`.
130 constexpr int mod97(std::string_view s, int r = 0) noexcept
131 {
132 for (const char c : s) {
133 const int v = value(c);
134 r = v > 9 ? (r * 100 + v) % 97 : (r * 10 + v) % 97;
135 }
136 return r;
137 }

139 constexpr std::array<char, 2> two_digits(int n) noexcept
140 {
141 return {static_cast<char>('0' + n / 10), static_cast<char>('0' + n % 10)};

144 // The IBAN remainder: BBAN, then country, then check digits.
145 constexpr int iban_mod97(std::string_view country, std::string_view check, std::string_view bban) noexcept

150 // The CUSIP value of one character: digits, A-Z, then *, @, #; -1 for anything else.
151 constexpr int cusip_value(char c) noexcept
152 {
153 return digit(c) || upper(c) ? value(c) : c == '*' ? 36 : c == '@' ? 37 : c == '#' ? 38 : -1;
154 }

156 constexpr bool sedol_character(char c) noexcept
157 {
158 return digit(c) || (upper(c) && c != 'A' && c != 'E' && c != 'I' && c != 'O' && c != 'U');
159 }

168 [[nodiscard]] constexpr result<char, identifier_errc> luhn_check_digit(std::string_view payload) noexcept
169 {
170 if (!detail::check_digits::all_digits(payload))
171 return err(identifier_errc::invalid_character);
172 return ok(static_cast<char>('0' + (10 - detail::check_digits::luhn_sum(payload, true)) % 10));
173 }

234 std::string_view bban) noexcept
235 {
236 using namespace detail::check_digits;
237 if (country.size() != 2 || bban.empty() || bban.size() > 30)
238 return err(identifier_errc::wrong_length);
239 if (!all_alphanumeric(country) || !all_alphanumeric(bban))
240 return err(identifier_errc::invalid_character);

250 [[nodiscard]] constexpr result<void, identifier_errc> validate_iban(std::string_view text) noexcept
251 {
252 using namespace detail::check_digits;
253 if (text.size() < 5 || text.size() > 34)
254 return err(identifier_errc::wrong_length);
255 if (!all_alphanumeric(text))
256 return err(identifier_errc::invalid_character);
257 if (!upper(text[0]) || !upper(text[1]) || !digit(text[2]) || !digit(text[3]))
258 return err(identifier_errc::invalid_format);
259 const int check = value(text[2]) * 10 + value(text[3]);
260 if (check < 2 || check > 98 || iban_mod97(text.substr(0, 2), text.substr(2, 2), text.substr(4)) != 1)
261 return err(identifier_errc::invalid_check_digit);
262 return ok();

305 [[nodiscard]] constexpr result<char, identifier_errc> cusip_check_digit(std::string_view payload) noexcept
306 {
307 if (payload.size() != 8)
308 return err(identifier_errc::wrong_length);
309 int sum = 0;
310 for (std::size_t i = 0; i < 8; ++i) {
311 int v = detail::check_digits::cusip_value(payload[i]);
312 if (v < 0)
313 return err(identifier_errc::invalid_character);
314 v *= i % 2 == 1 ? 2 : 1;
315 sum += v / 10 + v % 10;
316 }
317 return ok(static_cast<char>('0' + (10 - sum % 10) % 10));
318 }

324 [[nodiscard]] constexpr result<void, identifier_errc> validate_cusip(std::string_view text) noexcept
325 {
326 if (text.size() != 9)
327 return err(identifier_errc::wrong_length);
328 const auto check = cusip_check_digit(text.substr(0, 8));
329 if (!check)
330 return err(check.unwrap_err_unchecked());
331 if (detail::check_digits::cusip_value(text[8]) < 0)
332 return err(identifier_errc::invalid_character);
333 if (!detail::check_digits::digit(text[8]))
334 return err(identifier_errc::invalid_format);
335 if (*check != text[8])
336 return err(identifier_errc::invalid_check_digit);
337 return ok();
338 }

346 [[nodiscard]] constexpr result<char, identifier_errc> sedol_check_digit(std::string_view payload) noexcept
347 {
348 using namespace detail::check_digits;
349 if (payload.size() != 6)
350 return err(identifier_errc::wrong_length);
351 constexpr std::array<int, 6> weights{1, 3, 1, 7, 3, 9};
352 int sum = 0;
353 for (std::size_t i = 0; i < 6; ++i) {
354 if (!sedol_character(payload[i]))
355 return err(identifier_errc::invalid_character);
356 sum += weights[i] * value(payload[i]);
357 }
358 if (digit(payload[0]) && !all_digits(payload))
359 return err(identifier_errc::invalid_format);
360 return ok(static_cast<char>('0' + (10 - sum % 10) % 10));
361 }

383 // ---- GTIN (EAN-13, EAN-8, UPC-A, GTIN-14) -----------------------------------------------------------------------
384
392 [[nodiscard]] constexpr result<char, identifier_errc> gtin_check_digit(std::string_view payload) noexcept
393 {
394 if (payload.size() != 7 && payload.size() != 11 && payload.size() != 12 && payload.size() != 13)
395 return err(identifier_errc::wrong_length);
396 if (!detail::check_digits::all_digits(payload))
397 return err(identifier_errc::invalid_character);
398 int sum = 0;
399 for (std::size_t i = 0; i < payload.size(); ++i)
400 sum += (payload[payload.size() - 1 - i] - '0') * (i % 2 == 0 ? 3 : 1);
401 return ok(static_cast<char>('0' + (10 - sum % 10) % 10));
402 }

408 [[nodiscard]] constexpr result<void, identifier_errc> validate_gtin(std::string_view text) noexcept
409 {
410 if (text.empty())
411 return err(identifier_errc::wrong_length);
412 const auto check = gtin_check_digit(text.substr(0, text.size() - 1));
413 if (!check)
414 return err(check.unwrap_err_unchecked());
415 if (!detail::check_digits::digit(text.back()))
416 return err(identifier_errc::invalid_character);
417 if (*check != text.back())
418 return err(identifier_errc::invalid_check_digit);
419 return ok();
420 }
