96 // Drops and creates `t`; false (and a recorded failure) when the engine refuses.
97 template <typename C, typename E>
98 bool kit_fresh_table(C& c, const sql_table& t, const E& expect)
99 {
100 const auto drop = sql_drop_table(c.dialect(), t.name);
101 const auto create = sql_create_table(c.dialect(), t);
102 const bool good = drop && create && c.execute(*drop, {}).is_ok() && c.execute(*create, {}).is_ok();
103 expect(good, "the kit's table is created");
104 return good;
105 }

107 template <typename C>
108 std::int64_t kit_count(C& c, const sql_table& t)
109 {
110 std::int64_t n = -1;
111 const auto q = sql_count(c.dialect(), t);
112 if (q)
113 static_cast<void>(sql_query(c, *q, std::span<const sql_param>{}, [&n](const sql_row& r) { n = r.get_int64(0).unwrap_or(-1); }));
114 return n;
