81 concept sql_dialect = requires(const D& d, std::string& out, std::string_view id, std::size_t n, sql_type t) {
82 { d.name() } -> std::same_as<std::string_view>;
83 { d.column_type(t) } -> std::same_as<std::optional<std::string_view>>;
84 { d.append_identifier(out, id) } -> std::same_as<bool>;
85 { d.append_placeholder(out, n) } -> std::same_as<void>;
86 { d.begin_transaction() } -> std::same_as<std::string_view>;
87 { d.clustered_suffix() } -> std::same_as<std::string_view>;
88 };

145 inline sql_result<void> sql_check_table(const sql_table& t)
146 {
147 if (t.name.empty() || t.columns.empty())
148 return sql_fail(sql_errc::misuse, "a table needs a name and columns");
149 for (const std::size_t k : t.primary_key)
150 if (k >= t.columns.size())
151 return sql_fail(sql_errc::misuse, "a key column past the table");
152 return ok();
153 }

155 // " WHERE ... ORDER BY ... LIMIT n", numbering placeholders from `n`.
156 template <sql_dialect D>
157 sql_result<void> sql_append_where(const D& d, std::string& out, const sql_table& t, const sql_select& w, bool order)
158 {
159 const auto col = [&](std::size_t i) -> sql_result<void> {
160 if (i >= t.columns.size())
161 return sql_fail(sql_errc::misuse, "a column index past the table");
162 return sql_append_id(d, out, t.columns[i].name);
163 };
164 std::size_t n = 1;
165 bool first = true;
166 const auto conjunct = [&] {
167 out += first ? " WHERE " : " AND ";
168 first = false;
169 };
170 for (const std::size_t i : w.equal) {
171 conjunct();
172 SNICHOLLS_TRY_VOID(col(i));
173 out += " = ";
174 d.append_placeholder(out, n++);
175 }
176 if ((w.has_from || w.has_to) && !w.range_column)
177 return sql_fail(sql_errc::misuse, "a range without a column");
178 if (w.has_from) {
179 conjunct();
180 SNICHOLLS_TRY_VOID(col(*w.range_column));
181 out += " >= ";
182 d.append_placeholder(out, n++);
183 }
184 if (w.has_to) {
185 conjunct();
186 SNICHOLLS_TRY_VOID(col(*w.range_column));
187 out += " < ";
188 d.append_placeholder(out, n++);
189 }
190 if (!order)
191 return ok();
192 for (std::size_t k = 0; k < w.order_by.size(); ++k) {
193 out += k == 0 ? " ORDER BY " : ", ";
194 SNICHOLLS_TRY_VOID(col(w.order_by[k]));
195 if (w.descending)
196 out += " DESC";
197 }
198 if (w.limit != 0)
199 out += " LIMIT " + std::to_string(w.limit);
200 return ok();
201 }

209 template <sql_dialect D>
210 [[nodiscard]] sql_result<std::string> sql_create_table(const D& d, const sql_table& t)
211 {
212 SNICHOLLS_TRY_VOID(detail::sql_check_table(t));
213 std::string out = "CREATE TABLE IF NOT EXISTS ";
214 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.name));
215 out += " (";
216 for (std::size_t i = 0; i < t.columns.size(); ++i) {
217 if (i != 0)
218 out += ", ";
219 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.columns[i].name));
220 const auto type = d.column_type(t.columns[i].type);
221 if (!type)
222 return sql_fail(sql_errc::misuse, "a type the dialect cannot name");
223 out += ' ';
224 out += *type;
225 if (!t.columns[i].nullable)
226 out += " NOT NULL";
227 }
228 if (!t.primary_key.empty()) {
229 out += ", PRIMARY KEY (";
230 for (std::size_t k = 0; k < t.primary_key.size(); ++k) {
231 if (k != 0)
232 out += ", ";
233 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.columns[t.primary_key[k]].name));
234 }
235 out += ')';
236 }
237 out += ')';
238 if (t.clustered && !t.primary_key.empty())
239 out += d.clustered_suffix();
240 return ok(std::move(out));
241 }

263 template <sql_dialect D>
264 [[nodiscard]] sql_result<std::string> sql_insert(const D& d, const sql_table& t, sql_conflict conflict = sql_conflict::fail)
265 {
266 SNICHOLLS_TRY_VOID(detail::sql_check_table(t));
267 if (conflict != sql_conflict::fail && t.primary_key.empty())
268 return sql_fail(sql_errc::misuse, "a conflict clause needs a primary key");
269 std::string out = "INSERT INTO ";
270 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.name));
271 out += " (";
272 for (std::size_t i = 0; i < t.columns.size(); ++i) {
273 if (i != 0)
274 out += ", ";
275 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.columns[i].name));
276 }
277 out += ") VALUES (";
278 for (std::size_t i = 0; i < t.columns.size(); ++i) {
279 if (i != 0)
280 out += ", ";
281 d.append_placeholder(out, i + 1);
282 }
283 out += ')';
284 if (conflict == sql_conflict::fail)
285 return ok(std::move(out));
286 out += " ON CONFLICT (";
287 for (std::size_t k = 0; k < t.primary_key.size(); ++k) {
288 if (k != 0)
289 out += ", ";
290 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.columns[t.primary_key[k]].name));
291 }
292 out += ')';
293 bool any = false;
294 if (conflict == sql_conflict::update)
295 for (std::size_t i = 0; i < t.columns.size(); ++i) {
296 bool key = false;
297 for (const std::size_t k : t.primary_key)
298 key = key || k == i;
299 if (key)
300 continue;
301 out += any ? ", " : " DO UPDATE SET ";
302 any = true;
303 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.columns[i].name));
304 out += " = excluded.";
305 SNICHOLLS_TRY_VOID(detail::sql_append_id(d, out, t.columns[i].name));
306 }
307 if (!any)
308 out += " DO NOTHING";
309 return ok(std::move(out));
310 }

371 inline bool append_quoted_identifier(std::string& out, std::string_view id)
372 {
373 if (id.empty() || id.find('\0') != std::string_view::npos)
374 return false;
375 out += '"';
376 for (const char c : id) {
377 if (c == '"')
378 out += '"';
379 out += c;
380 }
381 out += '"';
382 return true;
383 }
