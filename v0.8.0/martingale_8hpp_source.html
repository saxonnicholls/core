72 [[nodiscard]] constexpr doob_parts doob_decomposition(std::span<const double> x, std::span<const double> predicted)
73 {
74 assert(!x.empty() && predicted.size() + 1 == x.size());
75 doob_parts out;
76 out.martingale.assign(x.size(), 0.0);
77 out.compensator.assign(x.size(), 0.0);
78 for (std::size_t t = 1; t < x.size(); ++t) {
79 out.compensator[t] = out.compensator[t - 1] + (predicted[t - 1] - x[t - 1]);
80 out.martingale[t] = out.martingale[t - 1] + (x[t] - predicted[t - 1]);
81 }
82 return out;
83 }

91 [[nodiscard]] constexpr std::vector<double> martingale_transform(std::span<const double> h, std::span<const double> x)
92 {
93 assert(h.size() == x.size());
94 std::vector<double> out(x.size(), 0.0);
95 for (std::size_t t = 1; t < x.size(); ++t)
96 out[t] = out[t - 1] + h[t] * (x[t] - x[t - 1]);

116 public:
122 markov_chain(std::size_t n, std::vector<double> transitions) : n_(n), p_(std::move(transitions))
123 {
124 assert(p_.size() == n * n);
125 for (std::size_t i = 0; i < n; ++i) {
126 double row = 0;
127 for (std::size_t j = 0; j < n; ++j) {
128 assert(p_[i * n + j] >= 0.0 && p_[i * n + j] <= 1.0);
129 row += p_[i * n + j];
130 }
131 assert(std::abs(row - 1.0) < 1e-9 && "snicholls::stochastic::markov_chain: a row does not sum to 1");
132 static_cast<void>(row);
133 }
134 }

150 [[nodiscard]] double expect(std::size_t i, std::span<const double> f) const noexcept
151 {
152 double e = 0;
153 for (std::size_t j = 0; j < n_; ++j)
154 e += p_[i * n_ + j] * f[j];
155 return e;

163 [[nodiscard]] doob_parts doob(std::span<const std::size_t> states, std::span<const double> f) const
164 {
165 std::vector<double> x(states.size()), predicted(states.empty() ? 0 : states.size() - 1);
166 for (std::size_t t = 0; t < states.size(); ++t) {
167 x[t] = f[states[t]];
168 if (t + 1 < states.size())
169 predicted[t] = expect(states[t], f);
170 }
171 return doob_decomposition(x, predicted);
172 }

175 std::size_t n_;
176 std::vector<double> p_;

195 [[nodiscard]] inline test_result variance_ratio_test(std::span<const double> y, std::size_t q)
196 {
197 assert(q >= 2 && y.size() >= q + 2);
198 const std::size_t nq = y.size() - 1;
199 const double n = static_cast<double>(nq), qd = static_cast<double>(q);
200 const double mu = (y.back() - y.front()) / n;
201 double sigma_a = 0;
202 for (std::size_t t = 1; t < y.size(); ++t)
203 sigma_a += (y[t] - y[t - 1] - mu) * (y[t] - y[t - 1] - mu);
204 sigma_a /= n - 1;
205 double sigma_b = 0;
206 for (std::size_t t = q; t < y.size(); ++t)
207 sigma_b += (y[t] - y[t - q] - qd * mu) * (y[t] - y[t - q] - qd * mu);
208 const double m = qd * (n - qd + 1) * (1 - qd / n);
209 sigma_b /= m;
210 const double vr = sigma_b / sigma_a;
211 double denominator = 0;
212 for (std::size_t t = 1; t < y.size(); ++t)
213 denominator += (y[t] - y[t - 1] - mu) * (y[t] - y[t - 1] - mu);
214 double theta = 0;
215 for (std::size_t j = 1; j < q; ++j) {
216 double delta = 0;
217 for (std::size_t t = j + 1; t < y.size(); ++t) {
218 const double a = y[t] - y[t - 1] - mu, b = y[t - j] - y[t - j - 1] - mu;
219 delta += a * a * b * b;
220 }
221 delta *= n / (denominator * denominator);
222 const double w = 2 * (qd - static_cast<double>(j)) / qd;
223 theta += w * w * delta;
224 }
225 test_result out;
226 out.statistic = std::sqrt(n) * (vr - 1) / std::sqrt(theta); // Lo and MacKinlay's z*: sqrt(nq) M / sqrt(theta)
227 out.p_value = 2 * (1 - math::normal_cdf(std::abs(out.statistic)));
228 return out;
229 }

236 [[nodiscard]] inline test_result ljung_box(std::span<const double> x, std::size_t lags)
237 {
238 assert(lags >= 1 && x.size() > lags);
239 const double n = static_cast<double>(x.size());
240 double mean = 0;
241 for (const double v : x)
242 mean += v;
243 mean /= n;
244 double c0 = 0;
245 for (const double v : x)
246 c0 += (v - mean) * (v - mean);
247 double q = 0;
248 for (std::size_t k = 1; k <= lags; ++k) {
249 double ck = 0;
250 for (std::size_t t = k; t < x.size(); ++t)
251 ck += (x[t] - mean) * (x[t - k] - mean);
252 const double rho = ck / c0;
253 q += rho * rho / (n - static_cast<double>(k));
254 }
255 test_result out;
256 out.statistic = n * (n + 2) * q;
257 out.p_value = 1 - math::chi_square_cdf(out.statistic, static_cast<double>(lags));
258 return out;
259 }
