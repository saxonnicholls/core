109template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
110 requires std::same_as<row_value_t<X>, T>
111constexpr void block_forward(const block_weights<W, T>& w, X&& x, std::size_t pos, const S& cache, std::size_t layer,
112 const R& rope_src)
113{
114 const auto s = as_mutable_row(x);
115 assert(s.size() == C.d_model);
116 std::array<T, C.d_model> h{}, a{};
117 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
118 attention_forward<C>(w.attention, h, pos, cache, layer, rope_src, a);
119 for (std::size_t i = 0; i < C.d_model; ++i)
120 s[i] += a[i];
121 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
122 mlp_forward<C>(w.mlp, h, a);
123 for (std::size_t i = 0; i < C.d_model; ++i)
124 s[i] += a[i];
125}

135template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
136 requires std::same_as<row_value_t<X>, T>
137constexpr void block_prefill(const block_weights<W, T>& w, X&& xs, std::size_t first, const S& cache, std::size_t layer,
138 const R& rope_src)
139{
140 const auto s = as_mutable_row(xs);
141 assert(s.size() % C.d_model == 0);
142 for (std::size_t t = 0; t < s.size() / C.d_model; ++t)
143 block_forward<C>(w, s.subspan(t * C.d_model, C.d_model), first + t, cache, layer, rope_src);
144}
