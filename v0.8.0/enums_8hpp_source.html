96 // The part of the signature that is V: a name if V is an enumerator, a cast of a number if not.
97 template <auto V>
98 constexpr std::string_view enum_spelling() noexcept
99 {
100 constexpr std::string_view sig = enum_signature<V>();
101#if defined(__clang__)
102 constexpr std::string_view head = "[V = ";
103 constexpr char tail = ']';
104#elif defined(__GNUC__)
105 constexpr std::string_view head = "[with auto V = ";
106 constexpr char tail = ';';
107#elif defined(_MSC_VER)
108 constexpr std::string_view head = "enum_signature<";
109 constexpr char tail = '>';
110#else
111 constexpr std::string_view head = "";
112 constexpr char tail = ']';
113#endif
114 const std::size_t start = sig.find(head);
115 if (start == std::string_view::npos)
116 return {};
117 const std::size_t from = start + head.size();
118 std::size_t end = sig.find(tail, from);
119 if (end == std::string_view::npos)
120 end = sig.find(']', from);
121 if (end == std::string_view::npos)
122 return {};
123 return sig.substr(from, end - from);
124 }

126 // The enumerator's own name, or empty when V is not an enumerator
127 template <auto V>
128 constexpr std::string_view enum_value_name() noexcept
129 {
130 // The last component: an identifier for an enumerator ("(anonymous namespace)::side::sell"),
131 // a number for anything else ("(side)9", "(enum side)0x9", "((anonymous namespace)::side)-1").
132 constexpr std::string_view spelling = enum_spelling<V>();
133 const std::size_t colon = spelling.rfind("::");
134 const std::size_t paren = spelling.rfind(')');
135 std::size_t from = 0;
136 if (colon != std::string_view::npos)
137 from = colon + 2;
138 if (paren != std::string_view::npos && paren + 1 > from)
139 from = paren + 1;
140 const std::string_view last = spelling.substr(from);
141 const bool identifier = !last.empty() && (last.front() == '_' || (last.front() >= 'a' && last.front() <= 'z') ||
142 (last.front() >= 'A' && last.front() <= 'Z'));
143 return identifier ? last : std::string_view{};
144 }

146 template <typename E>
147 consteval long long enum_scan_min() noexcept
148 {
149 using U = std::underlying_type_t<E>;
150 const long long lowest = std::is_signed_v<U> ? static_cast<long long>(std::numeric_limits<U>::min()) : 0;
151 return enum_range<E>::min > lowest ? enum_range<E>::min : lowest;
152 }
153
154 template <typename E>
155 consteval long long enum_scan_max() noexcept
156 {
157 using U = std::underlying_type_t<E>;
158 const auto highest = static_cast<unsigned long long>(std::numeric_limits<U>::max());
159 const unsigned long long wanted = enum_range<E>::max < 0 ? 0 : static_cast<unsigned long long>(enum_range<E>::max);
160 return static_cast<long long>(wanted < highest ? wanted : highest);
161 }

175 template <typename E>
176 struct enum_table {
177 static_assert(reflectable_enum<E>, "snicholls::enum reflection: give the enumeration a fixed underlying type "
178 "(enum e : int { ... }); without one its out-of-range values are undefined");
179 static constexpr long long min = enum_scan_min<E>();
180 static constexpr long long max = enum_scan_max<E>();
181 static_assert(max >= min && max - min < 65536, "snicholls::enum reflection: enum_range spans more than 65536 values");
182 static constexpr auto span = std::make_index_sequence<static_cast<std::size_t>(max - min + 1)>{};
183 static constexpr auto valid = enum_valid<E, min>(span);
184 static constexpr auto scanned = enum_scanned_names<E, min>(span);
185
186 static consteval std::size_t count() noexcept
187 {
188 std::size_t n = 0;
189 for (const bool v : valid)
190 n += v ? 1 : 0;
191 return n;
192 }
193
194 static consteval std::array<E, count()> values() noexcept
195 {
196 std::array<E, count()> out{};
197 std::size_t k = 0;
198 for (std::size_t i = 0; i < valid.size(); ++i)
199 if (valid[i])
200 out[k++] = static_cast<E>(min + static_cast<long long>(i));
201 return out;
202 }
203
204 // The position of each scanned value in values(), or none: lookup by value is one load.
205 static consteval std::array<std::uint16_t, valid.size()> positions() noexcept
206 {
207 std::array<std::uint16_t, valid.size()> out{};
208 std::uint16_t k = 0;
209 for (std::size_t i = 0; i < valid.size(); ++i)
210 out[i] = valid[i] ? k++ : std::uint16_t{0xFFFF};
211 return out;
212 }
213
214 static consteval std::array<std::string_view, count()> names() noexcept
215 {
216 std::array<std::string_view, count()> out{};
217 std::size_t k = 0;
218 for (std::size_t i = 0; i < valid.size(); ++i)
219 if (valid[i])
220 out[k++] = scanned[i];
221 return out;
222 }
223 };

252 template <reflectable_enum E>
253 [[nodiscard]] constexpr std::optional<std::size_t> enum_index(E value) noexcept
254 {
255 using table = detail::enum_table<E>;
256 const auto v = static_cast<long long>(static_cast<std::underlying_type_t<E>>(value));
257 if (v < table::min || v > table::max)
258 return std::nullopt;
259 const std::uint16_t position = detail::enum_positions_v<E>[static_cast<std::size_t>(v - table::min)];
260 return position == 0xFFFF ? std::nullopt : std::optional<std::size_t>(position);
261 }
