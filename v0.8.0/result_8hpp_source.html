366template <class E>
367[[nodiscard]] std::string error_text(const E& e)
368{
369 using T = std::remove_cvref_t<E>;
370 if constexpr (result_detail::adl_error_text<T>)
371 return std::string(snicholls_error_text(e));
372 else if constexpr (result_detail::has_what<T>)
373 return std::string(e.what());
374 else if constexpr (std::is_same_v<T, std::exception_ptr>)
375 {
376 if (!e)
377 return "null exception_ptr";
378#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
379 try
380 {
381 std::rethrow_exception(e);
382 }
383 catch (const std::exception& x)
384 {
385 return std::string(x.what());
386 }
387 catch (...)
388 {
389 return "unknown exception";
390 }
391#else
392 return "exception";
393#endif
394 }
395 else if constexpr (result_detail::string_viewable<T>)
396 return std::string(std::string_view(e));
397 else if constexpr (result_detail::has_message<T>)
398 return std::string(e.message());
399 else if constexpr (result_detail::std_formattable<T>)
400 return std::format("{}", e);
401 else if constexpr (std::is_enum_v<T>)
402 return "error " + std::to_string(static_cast<std::underlying_type_t<T>>(e));
403 else
404 return "error";
405}

469template <class E>
470[[noreturn]] void throw_error(const E& e)
471{
472 using T = std::remove_cvref_t<E>;
473#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
474 if constexpr (result_detail::has_custom_raise<T>)
475 error_exception<T>::raise(e);
476 else if constexpr (std::is_same_v<T, std::exception_ptr>)
477 {
478 if (e)
479 std::rethrow_exception(e);
480 throw bad_result_access<void>("snicholls::result: rethrow of a null exception_ptr");
481 }
482 else if constexpr (std::is_base_of_v<std::exception, T> && std::is_copy_constructible_v<T>)
483 throw e;
484 else if constexpr (std::is_same_v<T, std::error_code>)
485 throw std::system_error(e);
486 else if constexpr (std::is_copy_constructible_v<T>)
487 throw bad_result_access<T>("snicholls::result: error: " + error_text(e), e);
488 else
489 throw bad_result_access<void>("snicholls::result: error: " + error_text(e));
490#else
491 std::fputs("snicholls::result: error: ", stderr);
492 std::fputs(error_text(e).c_str(), stderr);
493 std::fputs("\n", stderr);
494 std::terminate();
495#endif
496 // Every branch above either throws or terminates; this keeps [[noreturn]] honest for the compiler.
497 std::terminate();
498}

696 template <class... A>
697 constexpr void reinit_value(A&&... a)
698 {
699 if (ok_)
700 {
701 // Assign, never destroy-then-construct: if constructing the new value threw after the old one was
702 // destroyed, ok_ would still say "value" and the destructor would destroy it a second time.
703 if constexpr (sizeof...(A) == 1 && (std::is_assignable_v<V&, A> && ...))
704 ((v_ = std::forward<A>(a)), ...);
705 else
706 v_ = V(std::forward<A>(a)...);
707 return;
708 }
709 if constexpr (std::is_nothrow_constructible_v<V, A...>)
710 {
711 std::destroy_at(std::addressof(e_));
712 std::construct_at(std::addressof(v_), std::forward<A>(a)...);
713 }
714 else if constexpr (std::is_nothrow_move_constructible_v<V>)
715 {
716 V tmp(std::forward<A>(a)...);
717 std::destroy_at(std::addressof(e_));
718 std::construct_at(std::addressof(v_), std::move(tmp));
719 }
720 else
721 {
722 E saved(std::move(e_));
723 std::destroy_at(std::addressof(e_));
724#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
725 try
726 {
727 std::construct_at(std::addressof(v_), std::forward<A>(a)...);
728 }
729 catch (...)
730 {
731 std::construct_at(std::addressof(e_), std::move(saved));
732 throw;
733 }
734#else
735 std::construct_at(std::addressof(v_), std::forward<A>(a)...);
736#endif
737 }
738 ok_ = true;
739 }

742 template <class... A>
743 constexpr void reinit_error(A&&... a)
744 {
745 if (!ok_)
746 {
747 // Assign, as reinit_value does: destroy-then-construct would destroy the error twice if construction threw.
748 if constexpr (sizeof...(A) == 1 && (std::is_assignable_v<E&, A> && ...))
749 ((e_ = std::forward<A>(a)), ...);
750 else
751 e_ = E(std::forward<A>(a)...);
752 return;
753 }
754 if constexpr (std::is_nothrow_constructible_v<E, A...>)
755 {
756 std::destroy_at(std::addressof(v_));
757 std::construct_at(std::addressof(e_), std::forward<A>(a)...);
758 }
759 else if constexpr (std::is_nothrow_move_constructible_v<E>)
760 {
761 E tmp(std::forward<A>(a)...);
762 std::destroy_at(std::addressof(v_));
763 std::construct_at(std::addressof(e_), std::move(tmp));
764 }
765 else
766 {
767 V saved(std::move(v_));
768 std::destroy_at(std::addressof(v_));
769#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
770 try
771 {
772 std::construct_at(std::addressof(e_), std::forward<A>(a)...);
773 }
774 catch (...)
775 {
776 std::construct_at(std::addressof(v_), std::move(saved));
777 throw;
778 }
779#else
780 std::construct_at(std::addressof(e_), std::forward<A>(a)...);
781#endif
782 }
783 ok_ = false;
784 }
