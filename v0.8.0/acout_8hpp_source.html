198 // The rule for colour_mode::automatic, apart from reading the variables. NO_COLOR is the
199 // no-color.org convention. A Windows console shows escape codes only when told to, which takes a
200 // platform call, so there colour needs a terminal known to be on: Windows Terminal, or one that sets TERM.
201 constexpr bool environment_allows_colour(const char* no_color, const char* term, const char* wt_session, bool on_windows) noexcept
202 {
203 if (no_color != nullptr && no_color[0] != '\0')
204 return false;
205 if (term != nullptr && std::string_view{term} == "dumb")
206 return false;
207 return !on_windows || wt_session != nullptr || term != nullptr;
208 }

234 // The message's buffer: 256 bytes in place, a std::string beyond. A flush requested through the
235 // stream (std::endl, std::flush) is recorded, not performed, until the message is emitted.

237 public:
238 message_buffer(std::ostream* target, std::FILE* file) noexcept : target_(target), file_(file)
239 {
240 setp(inline_, inline_ + sizeof inline_);
241 }
242
243 std::ostream* target() const noexcept { return target_; }
244 std::FILE* file() const noexcept { return file_; }

246 // Whether this message writes escape codes. A message printed inside another (a type's
247 // print(acout&) reached through a container) asks the message it will end up in.
248 bool colour_wanted() const noexcept
249 {
250 const colour_mode mode = acout_colour_mode.load(std::memory_order_relaxed);
251 if (mode != colour_mode::automatic)
252 return mode == colour_mode::always;
253 if (file_ != nullptr)
254 return (file_ == stdout || file_ == stderr) && colour_allowed_by_environment();
255 if (target_ == &std::cout || target_ == &std::cerr || target_ == &std::clog)
256 return colour_allowed_by_environment();
257 const message_buffer* outer = enclosing();
258 return outer != nullptr && outer->colour_wanted();
259 }

261 std::optional<colour> active_colour() const noexcept { return active_; }
262 void set_active_colour(std::optional<colour> c) noexcept { active_ = c; }

264 // The colour the enclosing message is in, which this one must hand back.
265 std::optional<colour> inherited_colour() const noexcept
266 {
267 const message_buffer* outer = enclosing();
268 return outer != nullptr ? outer->current_colour() : std::nullopt;
269 }

271 std::optional<colour> current_colour() const noexcept { return active_ ? active_ : inherited_colour(); }

273 // The whole message, contiguous. Spills the in-place part when a std::string is in use.
274 std::string_view text()
275 {
276 if (spilled_.empty())
277 return {pbase(), static_cast<std::size_t>(pptr() - pbase())};
278 spilled_.append(pbase(), pptr());
279 setp(inline_, inline_ + sizeof inline_);
280 return spilled_;
281 }
282
283 // A copy of the message so far, without disturbing it.
284 std::string copy() const
285 {
286 std::string out = spilled_;
287 out.append(pbase(), pptr());
288 return out;
289 }
290
291 bool flush_requested() const noexcept { return flush_; }
292 void request_flush() noexcept { flush_ = true; }
293
294 void reset() noexcept
295 {
296 spilled_.clear();
297 setp(inline_, inline_ + sizeof inline_);
298 flush_ = false;
299 }
300
301 protected:
302 int_type overflow(int_type ch) override
303 {
304 spilled_.append(pbase(), pptr());
305 if (!traits_type::eq_int_type(ch, traits_type::eof()))
306 spilled_.push_back(traits_type::to_char_type(ch));
307 setp(inline_, inline_ + sizeof inline_);
308 return traits_type::not_eof(ch);
309 }
310
311 int sync() override
312 {
313 flush_ = true;
314 return 0;
315 }
316
317 private:
318 // Only ever asked on the coloured path, so a message without colour pays nothing. Without
319 // RTTI a nested message cannot see the enclosing one, and prints without colour.
320 const message_buffer* enclosing() const noexcept
321 {
322#if defined(__cpp_rtti)
323 return target_ != nullptr ? dynamic_cast<const message_buffer*>(target_->rdbuf()) : nullptr;
324#else
325 return nullptr;
326#endif
327 }

331 std::ostream* target_;
332 std::FILE* file_;
333 std::optional<colour> active_;
334 bool flush_ = false;

518 void copy_into(std::ostream& into)
519 {
520 close_colour();
521 const std::string_view text = buffer_.text();
522 if (!text.empty())
523 into.write(text.data(), static_cast<std::streamsize>(text.size()));
524 buffer_.reset();
525 }
526
529 void emit()
530 {
531 close_colour(); // a colour never outlives its message, so it cannot tint another thread's
532 const std::string_view text = buffer_.text();
533 const bool flush = buffer_.flush_requested();
534 if (text.empty() && !flush)
535 return;
536 if (std::FILE* const file = buffer_.file()) { // the C library locks the FILE for the one call
537 if (!text.empty())
538 std::fwrite(text.data(), 1, text.size(), file);
539 if (flush)
540 std::fflush(file);
541 } else {
542 std::ostream* const target = buffer_.target();
543 const std::lock_guard<std::mutex> lock(detail::acout_write_lock);
544 if (!text.empty())
545 target->write(text.data(), static_cast<std::streamsize>(text.size()));
546 if (flush)
547 target->flush();
548 }
549 buffer_.reset();
550 }
551
552 private:
553 // Unformatted, so an escape code does not use up a std::setw meant for the value after it.
554 void write_escape(std::string_view code) { stream_.write(code.data(), static_cast<std::streamsize>(code.size())); }
555
556 void start_colour(colour c)
557 {
558 write_escape(ansi_start(c).view());
559 buffer_.set_active_colour(c);
560 }
561
562 void return_to_colour(std::optional<colour> to)
563 {
564 if (to)
565 write_escape(ansi_start(*to).view());
566 else
567 write_escape(ansi_reset);
568 // What the enclosing message is in is not this message's to close
569 buffer_.set_active_colour(to == buffer_.inherited_colour() ? std::nullopt : to);
570 }
571
572 void close_colour()
573 {
574 if (buffer_.active_colour())
575 return_to_colour(buffer_.inherited_colour());
576 }

654 // The engine is a first_match chain: the first branch that accepts the value prints it, so a
655 // std::string is text and not a range of chars, a std::filesystem::path is its own operator<<
656 // and not a range of paths, and a std::map is {k: v} and not a range of pairs. Every branch
657 // states its result type, so only the chosen branch's body is ever instantiated. The branches
658 // are named so that two chains can share them: the full one behind print_to, and the lent one
659 // behind print_operators, which must never ask "is it streamable?" of its own argument, or a
660 // type made streamable by that very operator would stream itself forever.
661 void print_to_chain(std::ostream& os, const auto& value, bool nested);
662
663 template <typename Range>
664 void print_elements(std::ostream& os, const Range& range, char open, char close)
665 {
666 os << open;
667 const char* separator = "";
668 for (const auto& element : range) {
669 os << separator;
670 print_to_chain(os, element, true);
671 separator = ", ";
672 }
673 os << close;
674 }

678 inline constexpr auto print_acout_form = [](std::ostream& os, const printable_to_acout auto& v, bool) -> void {
679 acout message(os);

706 // A scoped enumeration without its own operator<< prints by name (meta/enums.hpp), or as its integer
707 // when the value is not an enumerator; unary + so that a std::uint8_t enumeration is not a character.
708 inline constexpr auto print_enum = []<typename E>(std::ostream& os, const E& e, bool) -> void
709 requires(scoped_enumeration<E> && !streamable<E>)
710 {

713 os << name;
714 else

716 };

724 if (o)

730 if (v.valueless_by_exception())

735 inline constexpr auto print_result_void = [](std::ostream& os, const result_of_void auto& r, bool) -> void {
736 if (r.is_ok()) {

791 // A value's text under its display attributes. A number with any of number_format, scale or units: scaled,
792 // formatted (as a double), the scale's suffix, then the units. Else its label. Else as it prints.
793 template <typename Set, typename T>
794 requires(number_like<T> && (Set::template has<number_format> || Set::template has<scale> || Set::template has<units>))
795 void write_displayed(std::ostream& os, const Set& s, const T& v, bool, display::rank<2>)
796 {
797 const scale sc = attribute_lookup::pick<scale>(s, scale{1.0, ""});
798 const number_format nf = attribute_lookup::pick<number_format>(s, number_format{"{}"});
799 const std::string_view unit = attribute_lookup::pick<units>(s, units{""}).text;
800 const double x = static_cast<double>(numeric_of(v)) / sc.factor; // number_format's check assumes a double

802 os << sc.suffix;
803 if (!unit.empty())
804 os << ' ' << unit;
805 }
