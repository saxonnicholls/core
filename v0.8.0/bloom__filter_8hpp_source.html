7// The algorithm is the same: k bits per key from two 64-bit hashes by double hashing (Kirsch and
8// Mitzenmacher), sized for a target false-positive rate. Core hashes with wyhash instead of FNV-1a, stores
9// plain words instead of atomics (the lock-free read side is a wrapper's business), and adds clear, count,

46 [[nodiscard]] inline bloom_parameters bloom_sizing(std::size_t expected_items, double false_positive_rate) noexcept
47 {
48 if (expected_items < 1)
49 expected_items = 1;
50 if (!(false_positive_rate > 0.0 && false_positive_rate < 1.0))
51 false_positive_rate = 0.01;
52 constexpr double ln2 = 0.6931471805599453;
53 const double n = static_cast<double>(expected_items);
54 const double m = -n * std::log(false_positive_rate) / (ln2 * ln2);
55 std::size_t bits = static_cast<std::size_t>(m) + 1;
56 bits = ((bits + 63) / 64) * 64;
57 const double k = std::round((m / n) * ln2);
58 const std::uint32_t hashes = k < 1.0 ? 1u : k > 32.0 ? 32u : static_cast<std::uint32_t>(k);
59 return {bits, hashes};
60 }

214 {
215 if (count_ == 0)

217 const double k = hashes_;

226 private:
227 // The k bit positions of a key: h1 + i * h2 (double hashing), h2 odd.
228 void add_hash(std::uint64_t h1) noexcept
229 {
230 const std::uint64_t h2 = splitmix64(h1) | 1u;
231 for (std::uint32_t i = 0; i < hashes_; ++i) {
232 const std::uint64_t bit = (h1 + i * h2) % bits_;
233 words_[bit >> 6] |= std::uint64_t{1} << (bit & 63);
234 }
235 ++count_;
236 }
237
238 [[nodiscard]] bool contains_hash(std::uint64_t h1) const noexcept
239 {
240 const std::uint64_t h2 = splitmix64(h1) | 1u;
241 for (std::uint32_t i = 0; i < hashes_; ++i) {
242 const std::uint64_t bit = (h1 + i * h2) % bits_;
243 if ((words_[bit >> 6] & (std::uint64_t{1} << (bit & 63))) == 0)
244 return false;
245 }
246 return true;
247 }
