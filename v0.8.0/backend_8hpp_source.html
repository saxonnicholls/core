455 template <typename B, typename EA, typename LA, scalar T, typename LI>
456 requires std::same_as<detail::value_t<EA>, T>
457 constexpr result<void, error> lu_inverse(const B& b, const md_span<EA, 2, LA>& lu, std::span<const pivot_index> pivots, const md_span<T, 2, LI>& inverse)
458 {
459 // Every check before the identity is written, so a failure leaves every argument unchanged.
460 if (!detail::is_square(lu.extent(0), lu.extent(1)) || pivots.size() != lu.extent(0) || inverse.extent(0) != lu.extent(0) ||
461 inverse.extent(1) != lu.extent(0))
462 return err(error::dimension_mismatch);
463 if (detail::same_storage(inverse, lu) || detail::same_storage(inverse, pivots))
464 return err(error::aliased);
465 for (std::size_t i = 0; i < inverse.extent(0); ++i)
466 for (std::size_t j = 0; j < inverse.extent(1); ++j)
467 inverse(i, j) = i == j ? T(1) : T(0);
468 return lu_solve(b, lu, pivots, inverse);
469 }

478 template <typename EA, typename LA>
479 requires scalar<detail::value_t<EA>>
480 [[nodiscard]] constexpr result<signed_log_det<detail::value_t<EA>>, error> lu_log_det(const md_span<EA, 2, LA>& lu, std::span<const pivot_index> pivots)
481 {
482 using T = detail::value_t<EA>;
483 if (!detail::is_square(lu.extent(0), lu.extent(1)) || pivots.size() != lu.extent(0))
484 return err(error::dimension_mismatch);
485 signed_log_det<T> d;
486 for (std::size_t i = 0; i < lu.extent(0); ++i) {
487 const T u = lu(i, i);
488 if (static_cast<std::size_t>(pivots[i]) != i)
489 d.sign = -d.sign;
490 if (u == T(0))
491 return ok(signed_log_det<T>{0, -std::numeric_limits<T>::infinity()});
492 if (u < T(0))
493 d.sign = -d.sign;
494 d.log_abs += static_cast<T>(math::log(math::abs(u)));
495 }
496 return ok(d);
497 }

737 template <typename B, scalar T, typename LA, std::size_t RB, typename LB>
738 requires(RB == 1 || RB == 2)
739 constexpr result<void, error> least_squares(const B& b, const md_span<T, 2, LA>& a, std::span<std::type_identity_t<T>> tau, const md_span<T, RB, LB>& rhs)
740 {
741 // Every check qr_factor and qr_solve make, before qr_factor writes A.
742 if (rhs.extent(0) != a.extent(0) || a.extent(0) < a.extent(1) || tau.size() != a.extent(1))
743 return err(error::dimension_mismatch);
744 if (detail::same_storage(rhs, a) || detail::same_storage(tau, a) || detail::same_storage(tau, rhs))
745 return err(error::aliased);
746 const result<void, error> f = qr_factor(b, a, tau);
747 if (!f)
748 return f;
749 return qr_solve(b, md_span<const T, 2, LA>(a), std::span<const T>(tau), rhs);
750 }
