94template <mutable_row P, row G, mutable_row M, mutable_row V>
95 requires std::same_as<row_value_t<P>, row_value_t<G>> && std::same_as<row_value_t<P>, row_value_t<M>>
96 && std::same_as<row_value_t<P>, row_value_t<V>>
97constexpr void adamw_step(P&& params, const G& grads, M&& m, V&& v, std::uint64_t step, const adamw_config<row_value_t<P>>& config)
98{
99 using T = row_value_t<P>;
100 const auto p = as_mutable_row(params);
101 const auto g = as_row(grads);
102 const auto mm = as_mutable_row(m);
103 const auto vv = as_mutable_row(v);
104 assert(step >= 1 && g.size() == p.size() && mm.size() == p.size() && vv.size() == p.size());
105 const T c1 = T(1) - math::pow(config.beta1, T(step));
106 const T c2 = T(1) - math::pow(config.beta2, T(step));
107 for (std::size_t i = 0; i < p.size(); ++i) {
108 mm[i] = config.beta1 * mm[i] + (T(1) - config.beta1) * g[i];
109 vv[i] = config.beta2 * vv[i] + (T(1) - config.beta2) * g[i] * g[i];
110 const T m_hat = mm[i] / c1;
111 const T v_hat = vv[i] / c2;
112 p[i] -= config.learning_rate * (m_hat / (math::sqrt(v_hat) + config.eps) + config.weight_decay * p[i]);
113 }
114}

123template <math::math_number T>
124constexpr T warmup_cosine(std::uint64_t step, std::uint64_t warmup, std::uint64_t total, T max_lr, std::type_identity_t<T> min_lr)
125{
126 assert(warmup <= total);
127 if (step < warmup)
128 return max_lr * T(step + 1) / T(warmup);
129 if (step >= total)
130 return min_lr;
131 const T ratio = T(step - warmup) / T(total - warmup);
132 const T coefficient = T(0.5) * (T(1) + math::cos(std::numbers::pi_v<T> * ratio));
133 return min_lr + coefficient * (max_lr - min_lr);
134}

143template <math::math_number T>
144constexpr T warmup_warmdown(std::uint64_t step, std::uint64_t total, std::uint64_t warmup, std::uint64_t warmdown, T final_fraction)
145{
146 assert(warmup + warmdown <= total);
147 if (step < warmup)
148 return T(step + 1) / T(warmup);
149 if (step + warmdown <= total)
150 return T(1);
151 const T progress = step >= total ? T(0) : T(total - step) / T(warmdown);
152 return progress + (T(1) - progress) * final_fraction;
153}
