122 constexpr float abs(float x) { return x < 0.0f ? -x : x; }

124 // roundf for |x| below 2^31, halves away from zero: the fraction x - trunc(x) is exact
125 constexpr int round_to_int(float x)
126 {
127 const int t = static_cast<int>(x);
128 const float frac = x - static_cast<float>(t);
129 return t + (frac >= 0.5f ? 1 : 0) - (frac <= -0.5f ? 1 : 0);
130 }

132 // 2^(e - 128): half the MX scale, as ggml_e8m0_to_fp32_half, to pair with the doubled values
133 constexpr float e8m0_half(std::uint8_t e)
134 {
135 return std::bit_cast<float>(e < 2 ? std::uint32_t{0x00200000} << e : std::uint32_t(e - 1) << 23);
136 }

138 // floor(log2 x) for a positive finite float, from its bits
139 constexpr int floor_log2(float x)
140 {
141 const std::uint32_t b = std::bit_cast<std::uint32_t>(x);
142 const int biased = int((b >> 23) & 0xff);
143 if (biased != 0)
144 return biased - 127;
145 return std::bit_width(b & 0x7fffffu) - 1 - 149; // subnormal: m · 2^-149
146 }

148 constexpr block_q8_0 quantize(std::type_identity<block_q8_0>, block x)
149 {
150 float amax = 0.0f;
151 for (const float v : x)
152 amax = amax < abs(v) ? abs(v) : amax;
153 const float d = amax / 127.0f;
154 const float id = d != 0.0f ? 1.0f / d : 0.0f;
155 block_q8_0 b{to_minifloat<float16>(d), {}};
156 for (std::size_t j = 0; j < quant_block_size; ++j) {
157 const float scaled = x[j] * id;
158 b.qs[j] = static_cast<std::int8_t>(round_to_int(scaled));
159 }
160 return b;
161 }

163 constexpr block_q4_0 quantize(std::type_identity<block_q4_0>, block x)
164 {
165 float amax = 0.0f, max = 0.0f;
166 for (const float v : x)
167 if (amax < abs(v)) {
168 amax = abs(v);
169 max = v;
170 }
171 const float d = max / -8.0f;
172 const float id = d != 0.0f ? 1.0f / d : 0.0f;
173 block_q4_0 b{to_minifloat<float16>(d), {}};
174 const auto code = [id](float v) {
175 const float scaled = v * id;
176 const float shifted = scaled + 8.5f;
177 const int q = static_cast<std::int8_t>(shifted);
178 return static_cast<std::uint8_t>(q < 15 ? q : 15);
179 };
180 for (std::size_t j = 0; j < quant_block_size / 2; ++j)
181 b.qs[j] = static_cast<std::uint8_t>(code(x[j]) | (code(x[j + quant_block_size / 2]) << 4));
182 return b;
183 }

185 constexpr std::uint8_t nearest_fp4(float v, float half_scale)
186 {
187 std::uint8_t best = 0;
188 float best_err = abs(float(fp4_doubled[0]) * half_scale - v);
189 for (std::uint8_t c = 1; c < 16; ++c) {
190 const float err = abs(float(fp4_doubled[c]) * half_scale - v);
191 if (err < best_err) {
192 best = c;
193 best_err = err;
194 }
195 }
196 return best;
197 }

199 constexpr block_mxfp4 quantize(std::type_identity<block_mxfp4>, block x)
200 {
201 float amax = 0.0f;
202 for (const float v : x)
203 amax = amax < abs(v) ? abs(v) : amax;
204 const int exponent = amax > 0.0f ? floor_log2(amax) - 2 + 127 : 0;
205 block_mxfp4 b{static_cast<std::uint8_t>(exponent < 0 ? 0 : exponent), {}};
206 const float half = e8m0_half(b.e);
207 for (std::size_t j = 0; j < quant_block_size / 2; ++j)
208 b.qs[j] = static_cast<std::uint8_t>(nearest_fp4(x[j], half) | (nearest_fp4(x[j + quant_block_size / 2], half) << 4));
209 return b;
210 }

214 const float d = to_float(b.d);

219 constexpr void dequantize(const block_q4_0& b, std::span<float, quant_block_size> out)
220 {
221 const float d = to_float(b.d);
222 for (std::size_t j = 0; j < quant_block_size / 2; ++j) {
223 out[j] = float((b.qs[j] & 0x0f) - 8) * d;
224 out[j + quant_block_size / 2] = float((b.qs[j] >> 4) - 8) * d;
225 }
226 }

228 constexpr void dequantize(const block_mxfp4& b, std::span<float, quant_block_size> out)
229 {
230 const float half = e8m0_half(b.e);
231 for (std::size_t j = 0; j < quant_block_size / 2; ++j) {
232 out[j] = half * float(fp4_doubled[b.qs[j] & 0x0f]);
233 out[j + quant_block_size / 2] = half * float(fp4_doubled[b.qs[j] >> 4]);
234 }

247template <quant_block Block, std::size_t N, row X>
248 requires std::same_as<row_value_t<X>, float>
249constexpr void quantize(const X& x, std::span<Block, N> blocks)
250{
251 const auto in = as_row(x);
252 assert(in.size() == quant_block_size * blocks.size());
253 for (std::size_t i = 0; i < blocks.size(); ++i)
254 blocks[i] = detail::quantize::quantize(std::type_identity<Block>{}, in.subspan(i * quant_block_size).template first<quant_block_size>());

274constexpr float dot(std::span<const block_q8_0> x, std::span<const block_q8_0> y)
275{
276 assert(x.size() == y.size());
277 float sum = 0.0f;
278 for (std::size_t i = 0; i < x.size(); ++i) {
279 int s = 0;
280 for (std::size_t j = 0; j < quant_block_size; ++j)
281 s += int(x[i].qs[j]) * int(y[i].qs[j]);
282 const float scale = to_float(x[i].d) * to_float(y[i].d);
283 const float term = float(s) * scale;
284 sum += term;
285 }
286 return sum;
287}

293constexpr float dot(std::span<const block_q4_0> x, std::span<const block_q8_0> y)
294{
295 assert(x.size() == y.size());
296 constexpr std::size_t h = quant_block_size / 2;
297 float sum = 0.0f;
298 for (std::size_t i = 0; i < x.size(); ++i) {
299 int s0 = 0, s1 = 0;
300 for (std::size_t j = 0; j < h; ++j) {
301 s0 += ((x[i].qs[j] & 0x0f) - 8) * int(y[i].qs[j]);
302 s1 += ((x[i].qs[j] >> 4) - 8) * int(y[i].qs[j + h]);
303 }
304 const float term = float(s0 + s1) * to_float(x[i].d) * to_float(y[i].d);
305 sum += term;
306 }
307 return sum;
308}

314constexpr float dot(std::span<const block_mxfp4> x, std::span<const block_q8_0> y)
315{
316 assert(x.size() == y.size());
317 constexpr std::size_t h = quant_block_size / 2;
318 float sum = 0.0f;
319 for (std::size_t i = 0; i < x.size(); ++i) {
320 const float d = to_float(y[i].d) * detail::quantize::e8m0_half(x[i].e);
321 int s1 = 0, s2 = 0;
322 for (std::size_t j = 0; j < h; ++j) {
323 s1 += int(y[i].qs[j]) * detail::quantize::fp4_doubled[x[i].qs[j] & 0x0f];
324 s2 += int(y[i].qs[j + h]) * detail::quantize::fp4_doubled[x[i].qs[j] >> 4];
325 }
326 const float term = d * float(s1 + s2);
327 sum += term;
328 }
329 return sum;
330}

336template <row X>
337 requires std::same_as<row_value_t<X>, float>
338constexpr float quantize_int8(const X& x, std::span<std::int8_t> q)
339{
340 const auto in = as_row(x);
341 assert(q.size() == in.size());
342 float amax = 0.0f;
343 for (const float v : in)
344 amax = amax < detail::quantize::abs(v) ? detail::quantize::abs(v) : amax;
345 const float scale = amax / 127.0f;
346 const float inverse = scale != 0.0f ? 1.0f / scale : 0.0f;
347 for (std::size_t i = 0; i < in.size(); ++i) {
348 const float scaled = in[i] * inverse;
349 q[i] = static_cast<std::int8_t>(detail::quantize::round_to_int(scaled));
350 }
351 return scale;
352}
