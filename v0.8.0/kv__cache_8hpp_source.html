135 template <row K, row V>
136 requires std::same_as<row_value_t<K>, T> && std::same_as<row_value_t<V>, T>
137 constexpr void append(std::size_t layer, std::size_t pos, const K& k, const V& v) const
138 {
139 const auto ks = as_row(k);
140 const auto vs = as_row(v);
141 assert(layer < layers_ && capacity_ > 0 && ks.size() == C.kv_size() && vs.size() == C.kv_size());
142 for (std::size_t h = 0; h < C.n_kv_heads; ++h) {
143 const std::size_t at = offset(layer, h, pos % capacity_);
144 for (std::size_t i = 0; i < C.head_dim; ++i) {
145 keys_[at + i] = ks[h * C.head_dim + i];
146 values_[at + i] = vs[h * C.head_dim + i];
147 }
148 }
149 }

156 [[nodiscard]] constexpr kv_runs<T> visible(std::size_t layer, std::size_t kv_head, key_range r) const
157 {
158 assert(layer < layers_ && kv_head < C.n_kv_heads && r.size() <= capacity_);
159 kv_runs<T> out{};
160 if (r.size() == 0)
161 return out;
162 const std::size_t first = r.first % capacity_;
163 const std::size_t n1 = first + r.size() <= capacity_ ? r.size() : capacity_ - first;
164 const std::size_t base = offset(layer, kv_head, 0);
165 out.keys[0] = std::span<const T>(keys_).subspan(base + first * C.head_dim, n1 * C.head_dim);
166 out.values[0] = std::span<const T>(values_).subspan(base + first * C.head_dim, n1 * C.head_dim);
167 out.keys[1] = std::span<const T>(keys_).subspan(base, (r.size() - n1) * C.head_dim);
168 out.values[1] = std::span<const T>(values_).subspan(base, (r.size() - n1) * C.head_dim);
169 return out;
170 }
171
174 [[nodiscard]] constexpr std::size_t layers() const noexcept { return layers_; }
177 [[nodiscard]] constexpr std::size_t capacity() const noexcept { return capacity_; }
178
179private:
180 constexpr std::size_t offset(std::size_t layer, std::size_t h, std::size_t slot) const noexcept
181 {
182 return ((layer * C.n_kv_heads + h) * capacity_ + slot) * C.head_dim;
183 }
184
185 std::span<T> keys_{};
186 std::span<T> values_{};
187 std::size_t layers_ = 0;
188 std::size_t capacity_ = 0;

256 template <row K, row V>
257 requires std::same_as<row_value_t<K>, T> && std::same_as<row_value_t<V>, T>
258 constexpr void append(std::size_t layer, std::size_t pos, const K& k, const V& v) const
259 {
260 const auto ks = as_row(k);
261 const auto vs = as_row(v);
262 assert(layer < layers_ && pos / page_size_ < table_.size() && ks.size() == C.kv_size() && vs.size() == C.kv_size());
263 for (std::size_t h = 0; h < C.n_kv_heads; ++h) {
264 const std::size_t at = offset(table_[pos / page_size_], layer, h, pos % page_size_);
265 for (std::size_t i = 0; i < C.head_dim; ++i) {
266 keys_[at + i] = ks[h * C.head_dim + i];
267 values_[at + i] = vs[h * C.head_dim + i];
268 }
269 }
270 }

277 [[nodiscard]] constexpr kv_runs<T, MaxPages> visible(std::size_t layer, std::size_t kv_head, key_range r) const
278 {
279 assert(layer < layers_ && kv_head < C.n_kv_heads);
280 kv_runs<T, MaxPages> out{};
281 std::size_t run = 0;
282 for (std::size_t p = r.first; p < r.last; ++run) {
283 assert(run < MaxPages && p / page_size_ < table_.size());
284 const std::size_t slot = p % page_size_, n = std::min(page_size_ - slot, r.last - p);
285 const std::size_t at = offset(table_[p / page_size_], layer, kv_head, slot);
286 out.keys[run] = std::span<const T>(keys_).subspan(at, n * C.head_dim);
287 out.values[run] = std::span<const T>(values_).subspan(at, n * C.head_dim);
288 p += n;
289 }
290 return out;
291 }

299 {
300 return (((page * layers_ + layer) * C.n_kv_heads + h) * page_size_ + slot) * C.head_dim;
301 }

320template <typename Token>
321 requires std::integral<Token>
322[[nodiscard]] constexpr std::uint64_t kv_page_hash(std::uint64_t parent, std::span<const Token> tokens) noexcept
323{
324 std::uint64_t h = wymix(parent ^ 0xa0761d6478bd642full, tokens.size() ^ 0xe7037ed1a0b428dbull);
325 for (const Token t : tokens)
326 h = wymix(h ^ static_cast<std::uint64_t>(static_cast<std::make_unsigned_t<Token>>(t)), 0x8ebc6af09c88c6e3ull);
327 return h;
328}
