239// An instant as whole seconds (floored) + nanoseconds in [0, 1e9), with the fraction's excess-precision flag.
240struct split_instant
241{
242 std::int64_t seconds;
243 std::int64_t nanos;
244 bool excess; // non-zero digits finer than 1 ns
245};

247// (seconds, nanos) -> ticks of D. Errors: excess precision under reject, or out of D's range.
248template <utc_duration D>
249constexpr std::errc to_ticks(split_instant s, precision_policy policy, typename D::rep& out) noexcept
250{
251 using rep = typename D::rep;
252 using period = typename D::period;
253 std::int64_t ticks = 0;
254 bool excess = s.excess;
255 if constexpr (period::den > 1) { // 1/10^k s
256 constexpr std::int64_t den = period::den;
257 constexpr std::int64_t ns_per_tick = 1000000000 / den;
258 excess |= s.nanos % ns_per_tick != 0;
259 if (excess && policy == precision_policy::reject) return std::errc::result_out_of_range;
260 constexpr std::int64_t max = std::numeric_limits<std::int64_t>::max();
261 constexpr std::int64_t min = std::numeric_limits<std::int64_t>::min();
262 constexpr std::int64_t hi = max / den;
263 constexpr std::int64_t lo = min / den; // truncated toward 0: lo - 1 may still hold a valid instant
264 if (s.seconds > hi || s.seconds < lo - 1) return std::errc::result_out_of_range;
265 const std::int64_t sub = s.nanos / ns_per_tick; // [0, den)
266 if (s.seconds >= 0) {
267 ticks = s.seconds * den;
268 if (sub > max - ticks) return std::errc::result_out_of_range;
269 ticks += sub;
270 } else { // (seconds + 1) * den - (den - sub), never overflowing on the way
271 ticks = (s.seconds + 1) * den;
272 const std::int64_t back = den - sub;
273 if (ticks < min + back) return std::errc::result_out_of_range;
274 ticks -= back;
275 }
276 } else { // whole seconds (num >= 1)
277 constexpr std::int64_t num = period::num;
278 const std::int64_t q = floor_div(s.seconds, num);
279 excess |= s.nanos != 0 || s.seconds - q * num != 0;
280 if (excess && policy == precision_policy::reject) return std::errc::result_out_of_range;
281 ticks = q;
282 }
283 if constexpr (sizeof(rep) < sizeof(std::int64_t)) {
284 if (ticks < static_cast<std::int64_t>(std::numeric_limits<rep>::min()) ||
285 ticks > static_cast<std::int64_t>(std::numeric_limits<rep>::max()))
286 return std::errc::result_out_of_range;
287 }
288 out = static_cast<rep>(ticks);
289 return std::errc{};
290}

294// days_from_civil for y in 0..9999 in 32-bit unsigned arithmetic (no negative-era branch): years are shifted by
295// one 400-year era so y - 1 never goes below zero, then the era's 146097 days are taken back off.
296constexpr std::int64_t days_from_civil_0000_9999(unsigned y, unsigned m, unsigned d) noexcept
297{
298 y += 400 - (m <= 2);
299 const unsigned era = y / 400;
300 const unsigned yoe = y - era * 400;
301 const unsigned doy = (153 * (m > 2 ? m - 3 : m + 9) + 2) / 5 + d - 1;
302 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
303 return static_cast<std::int64_t>(era * 146097 + doe) - 719468 - 146097;
304}
305constexpr unsigned digit(char c) noexcept { return static_cast<unsigned>(c - '0'); }
306constexpr unsigned two(const char* p) noexcept { return digit(p[0]) * 10 + digit(p[1]); }

308// "YYYY-MM-DD" at p (at least 10 chars available). Syntax and validity.
309constexpr bool read_date(const char* p, unsigned& y, unsigned& m, unsigned& d) noexcept
310{
311 if (!(is_digit(p[0]) && is_digit(p[1]) && is_digit(p[2]) && is_digit(p[3]) && p[4] == '-' &&
312 is_digit(p[5]) && is_digit(p[6]) && p[7] == '-' && is_digit(p[8]) && is_digit(p[9])))
313 return false;
314 y = two(p) * 100 + two(p + 2);
315 m = two(p + 5);
316 d = two(p + 8);
317 if (m - 1 > 11 || d - 1 > 30) return false; // unsigned: also rejects 0
318 if (d <= 28) return true;
319 const bool leap = y % 4 == 0 && (y % 100 != 0 || y % 400 == 0); // 32-bit, y in 0..9999
320 return d <= (m == 2 ? 28U + leap : (m == 4 || m == 6 || m == 9 || m == 11) ? 30U : 31U);
321}

371template <utc_duration D>
372constexpr std::to_chars_result to_chars(char* first, char* last, const utc<D>& t, int precision) noexcept
373{
374 using period = typename D::period;
375 constexpr int digits = utc<D>::fraction_digits;
376 if (precision < 0 || precision > 9) return {last, std::errc::invalid_argument};
377
378 const std::int64_t count = static_cast<std::int64_t>(t.time_since_epoch().count());
379 std::int64_t seconds = 0;
380 std::int64_t sub = 0; // ticks of D below one second (0 for whole-second D)
381 if constexpr (period::den > 1) {
382 seconds = detail::floor_div(count, period::den);
383 sub = count - seconds * period::den;
384 } else {
385 constexpr std::int64_t num = period::num;
386 constexpr std::int64_t lo = -(-detail::rfc3339_min_seconds / num); // ceil(min / num), min < 0
387 constexpr std::int64_t hi = detail::rfc3339_max_seconds / num;
388 if (count < lo || count > hi) return {last, std::errc::value_too_large};
389 seconds = count * num;
390 }
391 if (seconds < detail::rfc3339_min_seconds || seconds > detail::rfc3339_max_seconds)
392 return {last, std::errc::value_too_large};
393
394 const std::size_t n = rfc3339_length(precision);
395 if (static_cast<std::size_t>(last - first) < n) return {last, std::errc::value_too_large};
396
397 const std::int64_t days = detail::floor_div(seconds, 86400);
398 const auto sod = static_cast<unsigned>(seconds - days * 86400);
399 const civil_date c = civil_from_days_fast(days); // years 0000..9999, checked above
400 const auto y = static_cast<unsigned>(c.year);
401
402 char* p = first;
403 detail::put2(p, y / 100);
404 detail::put2(p + 2, y % 100);
405 p[4] = '-';
406 detail::put2(p + 5, c.month);
407 p[7] = '-';
408 detail::put2(p + 8, c.day);
409 p[10] = 'T';
410 detail::put2(p + 11, sod / 3600);
411 p[13] = ':';
412 detail::put2(p + 14, sod / 60 % 60);
413 p[16] = ':';
414 detail::put2(p + 17, sod % 60);
415 p += 19;
416 if (precision > 0) {
417 std::int64_t frac = precision <= digits ? sub / detail::pow10[static_cast<std::size_t>(digits - precision)]
418 : sub * detail::pow10[static_cast<std::size_t>(precision - digits)];
419 *p = '.';
420 char* q = p + precision; // last digit
421 int left = precision;
422 for (; left >= 2; left -= 2, q -= 2) {
423 detail::put2(q - 1, static_cast<unsigned>(frac % 100));
424 frac /= 100;
425 }
426 if (left == 1) *q = static_cast<char>('0' + frac % 10);
427 p += precision + 1;
428 }
429 *p++ = 'Z';
430 return {p, std::errc{}};
431}

559struct rfc3339_scan
560{
561 split_instant instant{0, 0, false}; // the UTC instant, with second 60 read as 59
562 bool leap = false; // second 60 (already checked to be 23:59:60 UTC)
563 const char* end = nullptr; // one past the last character consumed
564 std::errc ec{}; // errc{} or invalid_argument
565};
566
567constexpr rfc3339_scan scan_rfc3339(const char* first, const char* last, const parse_options& options) noexcept
568{
569 rfc3339_scan r;
570 r.ec = std::errc::invalid_argument;
571 if (last - first < 19) return r;
572
573 unsigned y = 0, mo = 0, d = 0;
574 if (!read_date(first, y, mo, d)) return r;
575 const char* p = first + 10;
576 if (!(*p == 'T' || *p == 't' || *p == ' ')) return r;
577 if (!(is_digit(p[1]) && is_digit(p[2]) && p[3] == ':' && is_digit(p[4]) && is_digit(p[5]) && p[6] == ':' &&
578 is_digit(p[7]) && is_digit(p[8])))
579 return r;
580 const unsigned hh = two(p + 1), mi = two(p + 4), ss = two(p + 7);
581 if (hh > 23 || mi > 59 || ss > 60) return r;
582 p += 9;
583
584 if (p != last && *p == '.') {
585 ++p;
586 if (p == last || !is_digit(*p)) return r;
587 const char* const digits = p;
588 const char* const nine = last - p > 9 ? p + 9 : last;
589 std::uint32_t nanos = 0;
590 for (; p != nine && is_digit(*p); ++p) nanos = nanos * 10 + digit(*p);
591 nanos *= static_cast<std::uint32_t>(pow10[static_cast<std::size_t>(9 - (p - digits))]);
592 r.instant.nanos = nanos;
593 for (; p != last && is_digit(*p); ++p) r.instant.excess |= *p != '0'; // beyond 1 ns
594 }
595
596 std::int64_t offset = 0; // seconds east of UTC
597 if (p != last && (*p == 'Z' || *p == 'z')) {
598 ++p;
599 } else if (p != last && (*p == '+' || *p == '-')) {
600 const bool negative = *p == '-';
601 if (last - p < 3 || !is_digit(p[1]) || !is_digit(p[2])) return r;
602 const unsigned oh = two(p + 1);
603 const char* q = p + 3;
604 if (last - q >= 3 && *q == ':' && is_digit(q[1]) && is_digit(q[2]))
605 q += 1;
606 else if (!(options.basic_offset && last - q >= 2 && is_digit(q[0]) && is_digit(q[1])))
607 return r;
608 const unsigned om = two(q);
609 if (oh > 23 || om > 59) return r;
610 offset = (static_cast<std::int64_t>(oh) * 3600 + om * 60) * (negative ? -1 : 1);
611 p = q + 2;
612 } else if (options.require_offset) {
613 return r;
614 }
615
616 r.leap = ss == 60;
617 r.instant.seconds =
618 days_from_civil_0000_9999(y, mo, d) * 86400 + (hh * 3600 + mi * 60 + (r.leap ? 59 : ss)) - offset;
619 if (r.leap && r.instant.seconds - floor_div(r.instant.seconds, 86400) * 86400 != 86399) return r; // not 23:59 UTC
620 r.end = p;
621 r.ec = std::errc{};
622 return r;
623}

625template <utc_duration D>
626constexpr std::errc evaluate_rfc3339(const rfc3339_scan& s, const parse_options& options, typename D::rep& ticks) noexcept
627{
628 if (!s.leap) return to_ticks<D>(s.instant, options.precision, ticks);
629 if (options.leap_second == leap_second_policy::reject) return std::errc::result_out_of_range;
630 // The last representable instant before the insertion: one tick before ceil<D>(next second).
631 const split_instant next_second{s.instant.seconds + 1, 0, false};
632 typename D::rep next{};
633 if (to_ticks<D>(next_second, precision_policy::reject, next) == std::errc{}) { // D divides the second
634 if (next == std::numeric_limits<typename D::rep>::min()) return std::errc::result_out_of_range;
635 ticks = static_cast<typename D::rep>(next - 1);
636 return std::errc{};
637 }
638 return to_ticks<D>(next_second, precision_policy::truncate, ticks); // the floor already lies before it
639}

649template <utc_duration D>
650constexpr std::from_chars_result from_chars(const char* first, const char* last, utc<D>& out,
651 parse_options options = {}) noexcept
652{
653 const detail::rfc3339_scan s = detail::scan_rfc3339(first, last, options);
654 if (s.ec != std::errc{}) return {first, s.ec};
655 typename D::rep ticks{};
656 const std::errc ec = detail::evaluate_rfc3339<D>(s, options, ticks);
657 if (ec != std::errc{}) return {first, ec};
658 out = utc<D>(typename utc<D>::time_point(D(ticks)));
659 return {s.end, std::errc{}};
660}

678// One past the syntactically valid Unix-number text -?DIGITS(.DIGITS)? at first, or nullptr.
679constexpr const char* scan_unix(const char* first, const char* last) noexcept
680{
681 const char* p = first;
682 if (p != last && *p == '-') ++p;
683 if (p == last || !is_digit(*p)) return nullptr;
684 while (p != last && is_digit(*p)) ++p;
685 if (p != last && *p == '.') {
686 ++p;
687 if (p == last || !is_digit(*p)) return nullptr;
688 while (p != last && is_digit(*p)) ++p;
689 }
690 return p;
691}

693// [first, end) is syntactically valid (scan_unix).
694template <utc_duration Unit, utc_duration D>
695constexpr std::errc evaluate_unix(const char* first, const char* end, const parse_options& options,
696 typename D::rep& ticks) noexcept
697{
698 constexpr int unit_digits = utc<Unit>::fraction_digits; // s 0, ms 3, us 6, ns 9
699 const char* p = first;
700 const bool negative = *p == '-';
701 if (negative) ++p;
702
703 // Magnitude in Unit, up to 2^63 (so INT64_MIN ticks parse) or 2^63 - 1.
704 constexpr std::uint64_t int64_max = static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
705 const std::uint64_t limit = negative ? int64_max + 1 : int64_max;
706 std::uint64_t whole = 0;
707 for (; p != end && is_digit(*p); ++p) {
708 const unsigned dg = digit(*p);
709 if (whole > (limit - dg) / 10) return std::errc::result_out_of_range;
710 whole = whole * 10 + dg;
711 }
712 const auto unit_scale = static_cast<std::uint64_t>(pow10[unit_digits]);
713 const std::uint64_t whole_seconds = whole / unit_scale; // <= 2^63
714 split_instant s{0, static_cast<std::int64_t>(whole % unit_scale) * pow10[9 - unit_digits], false};
715 if (p != end) { // '.' DIGITS: digit n of the fraction is worth 10^(8 - unit_digits - n) ns
716 ++p;
717 for (int n = 0; p != end; ++p, ++n) {
718 if (n < 9 - unit_digits)
719 s.nanos += digit(*p) * pow10[static_cast<std::size_t>(8 - unit_digits - n)];
720 else if (*p != '0')
721 s.excess = true;
722 }
723 }
724 if (!negative) {
725 s.seconds = static_cast<std::int64_t>(whole_seconds); // <= 2^63 - 1
726 } else if (s.nanos == 0 && !s.excess) {
727 s.seconds = whole_seconds == 0 ? 0 : -static_cast<std::int64_t>(whole_seconds - 1) - 1; // 2^63 -> INT64_MIN
728 } else {
729 // -(W + n + e) with 0 < n + e < 1 s floors to -(W + 1) s + (1e9 - n - [e > 0]) ns: digits finer than
730 // 1 ns move a negative value one more ns into the past.
731 if (whole_seconds > int64_max) return std::errc::result_out_of_range; // only 2^63 itself
732 s.seconds = -static_cast<std::int64_t>(whole_seconds) - 1;
733 s.nanos = 1000000000 - s.nanos - (s.excess ? 1 : 0);
734 }
735 return to_ticks<D>(s, options.precision, ticks);
736}
