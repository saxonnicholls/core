87 template <typename T, std::size_t... Extents>
88 class md_array {
89 static_assert(sizeof...(Extents) >= 1 && ((Extents >= 1) && ...), "snicholls::containers::md_array: extents must be at least 1");
90
91 public:
93 static constexpr std::size_t rank = sizeof...(Extents);
95 static constexpr std::array<std::size_t, rank> extents{Extents...};
97 static constexpr std::size_t size = (Extents * ...);
99 static constexpr std::array<std::size_t, rank> strides = [] {
100 std::array<std::size_t, rank> s{};
101 std::size_t stride = 1;
102 for (std::size_t d = rank; d-- > 0;) {
103 s[d] = stride;
104 stride *= extents[d];
105 }
106 return s;
107 }();
108
109 std::array<T, size> elements{};
110

116 requires(sizeof...(Is) == rank)

118 {
119 return elements[offset(is...)];
120 }
121
126 template <typename... Is>
127 requires(sizeof...(Is) == rank)
128 [[nodiscard]] constexpr const T& operator()(Is... is) const noexcept
129 {
130 return elements[offset(is...)];
131 }
132
138 template <typename... Is>
139 requires(sizeof...(Is) == rank)
140 [[nodiscard]] constexpr T& at(Is... is)
141 {
142 std::size_t d = 0;
143 const bool inside = ((static_cast<std::size_t>(is) < extents[d++]) && ...);
144 if (!inside)

187 // The state every mapping shares: extents and strides, both run-time values.
188 template <std::size_t Rank>
189 struct strided_extents {
190 std::array<std::size_t, Rank> extents{};
191 std::array<std::size_t, Rank> strides{};
192
193 [[nodiscard]] constexpr std::size_t size() const noexcept
194 {
195 std::size_t n = 1;
196 for (std::size_t d = 0; d < Rank; ++d)
197 n *= extents[d];
198 return n;
199 }
200 [[nodiscard]] constexpr std::size_t required_span_size() const noexcept
201 {
202 if (size() == 0)
203 return 0;
204 std::size_t last = 0;
205 for (std::size_t d = 0; d < Rank; ++d)
206 last += (extents[d] - 1) * strides[d];
207 return last + 1;
208 }
209 };

559 // A range slice: a std::pair of any two integral types, {first, last}.
560 template <typename S>
561 inline constexpr bool is_index_pair = false;
562 template <std::integral I, std::integral J>
563 inline constexpr bool is_index_pair<std::pair<I, J>> = true;
564
565 template <typename S>
566 inline constexpr bool keeps_dimension = std::same_as<S, full_extent_t> || is_index_pair<S>;
567
568 // A slice's first index and its extent, by overload: an index, the whole dimension, or [first, last).
569 constexpr std::size_t slice_first(std::size_t i) noexcept { return i; }
570 constexpr std::size_t slice_first(full_extent_t) noexcept { return 0; }
571 constexpr std::size_t slice_first(const std::pair<std::size_t, std::size_t>& r) noexcept { return r.first; }
572 constexpr std::size_t slice_extent(full_extent_t, std::size_t extent) noexcept { return extent; }
573 constexpr std::size_t slice_extent(const std::pair<std::size_t, std::size_t>& r, std::size_t) noexcept { return r.second - r.first; }
574
575 template <typename S>
576 using slice_t = std::conditional_t<std::same_as<std::remove_cvref_t<S>, full_extent_t>, full_extent_t,
577 std::conditional_t<is_index_pair<std::remove_cvref_t<S>>, std::pair<std::size_t, std::size_t>, std::size_t>>;
578 } // namespace detail

593 template <typename T, std::size_t Rank, typename Layout, typename... Slices>
594 requires(sizeof...(Slices) == Rank &&
595 ((std::integral<std::remove_cvref_t<Slices>> || detail::keeps_dimension<std::remove_cvref_t<Slices>>) && ...) &&
596 (detail::keeps_dimension<std::remove_cvref_t<Slices>> || ...))
597 [[nodiscard]] constexpr auto submdspan(const md_span<T, Rank, Layout>& s, Slices... slices) noexcept
598 {
599 constexpr std::size_t kept = (std::size_t{detail::keeps_dimension<std::remove_cvref_t<Slices>>} + ...);
600 const std::tuple<detail::slice_t<Slices>...> all{static_cast<detail::slice_t<Slices>>(slices)...};
601 std::array<std::size_t, kept> extents{};
602 std::array<std::size_t, kept> strides{};
603 std::size_t offset = 0;
604 std::size_t k = 0;
605 constexpr_for<0, Rank, 1>([&](auto d) {
606 using S = std::tuple_element_t<d, std::tuple<detail::slice_t<Slices>...>>;
607 const auto& slice = std::get<d>(all);
608 offset += detail::slice_first(slice) * s.stride(d);
609 if constexpr (detail::keeps_dimension<S>) {
610 extents[k] = detail::slice_extent(slice, s.extent(d));
611 strides[k] = s.stride(d);
612 ++k;
613 }
614 });
615 return md_span<T, kept, layout_stride>(s.data_handle() + offset, extents, strides);
616 }
