110 template <typename E>
111 concept range_executor = std::copy_constructible<E> && requires(E& e, std::size_t n, detail::range_probe f) { e(n, f); };

115 struct inline_executor {
120 template <typename F>
121 constexpr void operator()(std::size_t n, F&& f) const

183 template <typename T, std::size_t RB, typename LA, typename LB>
184 constexpr bool batch_fits(const md_span<T, 3, LA>& a, const md_span<T, RB, LB>& b, std::size_t statuses) noexcept
185 {
186 return a.extent(1) == a.extent(2) && b.extent(0) == a.extent(0) && b.extent(1) == a.extent(1) && statuses == a.extent(0);

215 template <typename B, scalar T, typename LA, std::size_t RB, typename LB, range_executor E = inline_executor>
216 requires(RB == 2 || RB == 3)
217 constexpr result<std::size_t, error> lu_solve_batched(const B& b, const md_span<T, 3, LA>& a, std::span<pivot_index> pivots,
218 const md_span<T, RB, LB>& rhs, std::span<status> status, E ex = {})
219 {
220 static_assert(detail::has_lu_solve_batched<B, T>,
221 "snicholls::linalg::lu_solve_batched: the backend has no member result<std::size_t, error> lu_solve_batched(batch_view<T>, "
222 "std::span<pivot_index>, batch_view<T>, std::span<status>, E) const for every range_executor E");
223 if (!detail::batch_fits(a, rhs, status.size()) || pivots.size() != a.extent(0) * a.extent(1))
224 return err(error::dimension_mismatch);
225 if (detail::same_storage(a, rhs) || detail::same_storage(pivots, a) || detail::same_storage(pivots, rhs) ||
226 detail::same_storage(status, a) || detail::same_storage(status, rhs) || detail::same_storage(status, pivots))
227 return err(error::aliased);
228 return b.lu_solve_batched(batch_view<T>(a), pivots, detail::as_batch(rhs), status, ex);
229 }
