151 template <class... Args>
152 [[nodiscard]] T* create(Args&&... args)
153 {
154 slot* s = acquire();
155 if constexpr (std::is_nothrow_constructible_v<T, Args...>) {
156 T* p = ::new (static_cast<void*>(s->storage)) T(std::forward<Args>(args)...);
157 ++live_;
158 return p;
159 } else {
160 try {
161 T* p = ::new (static_cast<void*>(s->storage)) T(std::forward<Args>(args)...);
162 ++live_;
163 return p;
164 } catch (...) {
165 s->next_free = free_;
166 free_ = s;
167 throw;
168 }
169 }

184 void release_all() noexcept
185 {
186 free_ = nullptr;
187 next_ = chunks_.empty() ? nullptr : chunks_.front();
188 end_ = next_ ? next_ + ChunkSize : nullptr;
189 // Chained chunks after the first are reused through the free list.
190 for (std::size_t c = 1; c < chunks_.size(); ++c) {
191 slot* base = chunks_[c];
192 for (std::size_t i = ChunkSize; i-- > 0;) {
193 base[i].next_free = free_;
194 free_ = &base[i];
195 }
196 }
197 live_ = 0;
198 }
199
203 void reserve(std::size_t n)
204 {
205 while (capacity() < n) {
206 slot* base = new_chunk();
207 for (std::size_t i = ChunkSize; i-- > 0;) {
208 base[i].next_free = free_;
209 free_ = &base[i];
210 }
211 }
212 }

219 [[nodiscard]] std::size_t capacity() const noexcept { return chunks_.size() * ChunkSize; }

224private:
225 slot* acquire()
226 {
227 if (free_) {
228 slot* s = free_;
229 free_ = s->next_free;
230 return s;
231 }
232 if (next_ == end_) {
233 // Own the chunk first, then point at it: if recording it threw after next_ was set, next_ would point
234 // into a chunk already given back, and the next create() would hand out a dangling slot.
235 slot* base = new_chunk();
236 next_ = base;
237 end_ = base + ChunkSize;
238 }
239 return next_++;
240 }
241
242 static constexpr std::size_t chunk_bytes = ChunkSize * sizeof(slot);
243
244 // A chunk from the resource, recorded in chunks_. Strong: the list grows first, so once the chunk exists nothing
245 // can throw, and a failed allocation leaves only spare capacity in the list.
246 slot* new_chunk()
247 {
248 if (chunks_.size() == chunks_.capacity()) chunks_.reserve(chunks_.empty() ? 4 : 2 * chunks_.size());
249 void* raw = resource_->allocate(chunk_bytes, alignof(slot));
250 slot* base = static_cast<slot*>(raw);
251 for (std::size_t i = 0; i < ChunkSize; ++i) ::new (static_cast<void*>(base + i)) slot;
252 chunks_.push_back(base); // cannot throw: capacity reserved above
253 return base;
254 }
255
256 void free_chunks() noexcept
257 {
258 for (slot* c : chunks_) resource_->deallocate(c, chunk_bytes, alignof(slot));
259 chunks_.clear();
260 }
261
262 std::pmr::memory_resource* resource_ = std::pmr::new_delete_resource();
263 std::vector<slot*> chunks_;
264 slot* free_ = nullptr;
265 slot* next_ = nullptr;
266 slot* end_ = nullptr;
267 std::size_t live_ = 0;
268};
