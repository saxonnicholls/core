98template <typename S>
99struct dense_weights
100{
101 std::span<const S> data{};
102 std::size_t rows = 0;
103 std::size_t cols = 0;

108 [[nodiscard]] constexpr std::span<const S> row(std::size_t i) const { return data.subspan(i * cols, cols); }

183template <typename S, row X, mutable_row Y, row B = std::span<const row_value_t<X>>>
184 requires std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<B>>
185constexpr void linear(const dense_weights<S>& w, const X& x, Y&& y, const B& bias = B{})
186{
187 using T = row_value_t<X>;
188 const auto in = as_row(x);
189 const auto out = as_mutable_row(y);
190 const auto b = as_row(bias);
191 assert(in.size() == w.cols && out.size() == w.rows && w.data.size() == w.rows * w.cols && (b.empty() || b.size() == w.rows));
192 for (std::size_t i = 0; i < w.rows; ++i) {
193 const auto r = w.row(i);
194 T acc = b.empty() ? T(0) : b[i];
195 for (std::size_t j = 0; j < w.cols; ++j) {
196 const T p = detail::linear::widen<T>(r[j]) * in[j];
197 acc += p;
198 }
199 out[i] = acc;
200 }
201}

215template <typename S, row X, row DY, mutable_row DX, mutable_row DW, mutable_row DB = std::span<row_value_t<X>>>
216 requires std::same_as<row_value_t<X>, row_value_t<DY>> && std::same_as<row_value_t<X>, row_value_t<DX>>
217 && std::same_as<row_value_t<X>, row_value_t<DW>> && std::same_as<row_value_t<X>, row_value_t<DB>>
218constexpr void linear_backward(const dense_weights<S>& w, const X& x, const DY& dy, DX&& dx, DW&& dw, DB&& db = DB{})
219{
220 using T = row_value_t<X>;
221 const auto in = as_row(x);
222 const auto g = as_row(dy);
223 const auto gx = as_mutable_row(dx);
224 const auto gw = as_mutable_row(dw);
225 const auto gb = as_mutable_row(db);
226 assert(in.size() == w.cols && g.size() == w.rows && w.data.size() == w.rows * w.cols && gw.size() == w.rows * w.cols
227 && (gx.empty() || gx.size() == w.cols) && (gb.empty() || gb.size() == w.rows));
228 for (std::size_t i = 0; i < w.rows; ++i) {
229 const auto r = w.row(i);
230 const T d = g[i];
231 if (!gx.empty())
232 for (std::size_t j = 0; j < w.cols; ++j) {
233 const T p = detail::linear::widen<T>(r[j]) * d;
234 gx[j] += p;
235 }
236 for (std::size_t j = 0; j < w.cols; ++j) {
237 const T p = d * in[j];
238 gw[i * w.cols + j] += p;
239 }
240 if (!gb.empty())
241 gb[i] += d;
242 }
243}

250template <quant_block Block, std::size_t M, mutable_row Y, row B = std::span<const float>>
251 requires std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
252constexpr void linear(const quantized_weights<Block>& w, const std::array<block_q8_0, M>& x, Y&& y, const B& bias = B{})
253{
254 const auto out = as_mutable_row(y);
255 const auto b = as_row(bias);
256 assert(M * quant_block_size == w.cols && out.size() == w.rows && w.blocks.size() == w.rows * M && (b.empty() || b.size() == w.rows));
257 for (std::size_t i = 0; i < w.rows; ++i) {
258 const float d = dot(w.row(i), std::span<const block_q8_0>(x));
259 out[i] = b.empty() ? d : b[i] + d;
260 }
261}

277 // C = alpha op(A) op(B) + beta C. The shapes are right by construction and C never overlaps A or B, so the
278 // backend's product always succeeds
279 template <typename B, typename T, typename L, typename R, typename C>
280 constexpr void product(const B& backend, T alpha, const L& lhs, const R& rhs, T beta, const C& c)
281 {
282 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
283 assert(done.is_ok());
284 }

297template <math::math_number T, typename B>
298 requires linalg::blas_backend<B, T>
299constexpr void linear_sequence(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
300 std::type_identity_t<std::span<T>> ys, std::type_identity_t<std::span<const T>> bias = {})
301{
302 assert(w.cols > 0 && xs.size() % w.cols == 0);
303 const std::size_t n = xs.size() / w.cols;
304 assert(ys.size() == n * w.rows && w.data.size() == w.rows * w.cols && (bias.empty() || bias.size() == w.rows));
305 if (n == 0 || w.rows == 0)
306 return;
307 using detail::linear::view;
308 detail::linear::product(backend, T(1), view(xs, n, w.cols), linalg::transposed(view(w.data, w.rows, w.cols)), T(0),
309 view(ys, n, w.rows));
310 if (!bias.empty())
311 for (std::size_t t = 0; t < n; ++t)
312 for (std::size_t i = 0; i < w.rows; ++i)
313 ys[t * w.rows + i] += bias[i];
314}

325template <math::math_number T, typename B>
326 requires linalg::blas_backend<B, T>
327constexpr void linear_sequence_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
328 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
329 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db = {})
330{
331 assert(w.cols > 0 && xs.size() % w.cols == 0);
332 const std::size_t n = xs.size() / w.cols;
333 assert(dys.size() == n * w.rows && (dxs.empty() || dxs.size() == xs.size()) && dw.size() == w.rows * w.cols
334 && (db.empty() || db.size() == w.rows));
335 if (n == 0 || w.rows == 0)
336 return;
337 using detail::linear::view;
338 const auto dy = view(dys, n, w.rows);
339 if (!dxs.empty())
340 detail::linear::product(backend, T(1), dy, view(w.data, w.rows, w.cols), T(1), view(dxs, n, w.cols));
341 detail::linear::product(backend, T(1), linalg::transposed(dy), view(xs, n, w.cols), T(1), view(dw, w.rows, w.cols));
342 if (!db.empty())
343 for (std::size_t t = 0; t < n; ++t)
344 for (std::size_t i = 0; i < w.rows; ++i)
345 db[i] += dys[t * w.rows + i];
346}
