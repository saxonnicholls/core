119template <transformer_config C, row X, mutable_row Out>
120 requires std::same_as<row_value_t<X>, row_value_t<Out>>
121constexpr void block_norm(const X& x, std::span<const row_value_t<X>> weight, std::span<const row_value_t<X>> bias, Out&& out)
122{
123 using T = row_value_t<X>;
124 if constexpr (C.norm == norm_kind::layer)
125 layer_norm(x, weight, bias, out, norm_eps_v<C, T>);
126 else
127 rms_norm(x, weight, out, norm_eps_v<C, T>);
128}

138template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
139 requires std::same_as<row_value_t<X>, T>
140constexpr void block_forward(const block_weights<W, T>& w, X&& x, std::size_t pos, const S& cache, std::size_t layer,
141 const R& rope_src)
142{
143 const auto s = as_mutable_row(x);
144 assert(s.size() == C.d_model);
145 std::array<T, C.d_model> h{}, a{};
146 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
147 attention_forward<C>(w.attention, h, pos, cache, layer, rope_src, a);
148 for (std::size_t i = 0; i < C.d_model; ++i)
149 s[i] += a[i];
150 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
151 mlp_forward<C>(w.mlp, h, a);
152 for (std::size_t i = 0; i < C.d_model; ++i)
153 s[i] += a[i];
154}

164template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
165 requires std::same_as<row_value_t<X>, T>
166constexpr void block_prefill(const block_weights<W, T>& w, X&& xs, std::size_t first, const S& cache, std::size_t layer,
167 const R& rope_src)
168{
169 const auto s = as_mutable_row(xs);
170 assert(s.size() % C.d_model == 0);
171 for (std::size_t t = 0; t < s.size() / C.d_model; ++t)
172 block_forward<C>(w, s.subspan(t * C.d_model, C.d_model), first + t, cache, layer, rope_src);
173}

209 template <transformer_config C, typename T>
210 constexpr void norm_backward(std::span<const T> dh, std::span<const T> x, std::span<const T> weight,
211 std::span<const T> bias, std::span<T> dx, std::span<T> dweight, std::span<T> dbias)
212 {
213 std::array<T, C.d_model> scratch{};
214 if constexpr (C.norm == norm_kind::layer) {
215 const auto stats = layer_norm(x, weight, bias, scratch, norm_eps_v<C, T>);
216 layer_norm_backward(dh, x, weight, stats, dx, dweight, dbias);
217 } else {
218 const T rstd = rms_norm(x, weight, scratch, norm_eps_v<C, T>);
219 rms_norm_backward(dh, x, weight, rstd, dx, dweight);
220 }
221 }

231template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, typename R>
232 requires std::same_as<row_value_t<X>, T> && (C.rope == rope_kind::none || rope_source<R>)
233constexpr void block_forward_sequence(const block_weights<W, T>& w, X&& xs, const R& rope_src, const block_activations<T>& act)
234{
235 constexpr std::size_t d = C.d_model;
236 const auto s = as_mutable_row(xs);
237 const std::size_t n = s.size() / d;
238 assert(s.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d
239 && act.h2.size() == n * d && act.mlp_up.size() == n * C.d_ff
240 && (C.mlp != mlp_kind::swiglu || act.mlp_gate.size() == n * C.d_ff));
241 std::copy(s.begin(), s.end(), act.x0.begin());
242 for (std::size_t t = 0; t < n; ++t)
243 block_norm<C>(s.subspan(t * d, d), w.norm1_weight, w.norm1_bias, act.h1.subspan(t * d, d));
244 attention_forward_sequence<C>(w.attention, std::span<const T>(act.h1), rope_src, act.attention, s);
245 for (std::size_t i = 0; i < n * d; ++i)
246 s[i] = act.x0[i] + s[i];
247 std::copy(s.begin(), s.end(), act.x1.begin());
248 std::array<T, d> a{};
249 for (std::size_t t = 0; t < n; ++t) {
250 block_norm<C>(s.subspan(t * d, d), w.norm2_weight, w.norm2_bias, act.h2.subspan(t * d, d));
251 const mlp_activations<T> save{act.mlp_up.subspan(t * C.d_ff, C.d_ff),
252 act.mlp_gate.empty() ? std::span<T>{} : act.mlp_gate.subspan(t * C.d_ff, C.d_ff)};
253 mlp_forward<C>(w.mlp, std::span<const T>(act.h2).subspan(t * d, d), a, save);
254 for (std::size_t i = 0; i < d; ++i)
255 s[t * d + i] += a[i];
256 }
257}

270template <transformer_config C, typename S, math::math_number T, mutable_row DX, typename R>
271 requires std::same_as<row_value_t<DX>, T> && (C.rope == rope_kind::none || rope_source<R>)
272constexpr void block_backward(const block_weights<dense_weights<S>, T>& w, const R& rope_src,
273 const block_activations<T>& act, DX&& dxs, const block_gradients<T>& grads)
274{
275 constexpr std::size_t d = C.d_model;
276 const auto g = as_mutable_row(dxs);
277 const std::size_t n = g.size() / d;
278 assert(g.size() == n * d && act.x0.size() == n * d && grads.scratch.size() == n * d);
279 std::array<T, d> dh{}, dn{};
280 for (std::size_t t = 0; t < n; ++t) {
281 const auto gt = g.subspan(t * d, d);
282 dh.fill(T(0));
283 const mlp_activations<T> saved{act.mlp_up.subspan(t * C.d_ff, C.d_ff),
284 act.mlp_gate.empty() ? std::span<T>{} : act.mlp_gate.subspan(t * C.d_ff, C.d_ff)};
285 mlp_backward<C>(w.mlp, std::span<const T>(act.h2).subspan(t * d, d), saved, std::span<const T>(gt), dh, grads.mlp);
286 detail::block::norm_backward<C, T>(dh, std::span<const T>(act.x1).subspan(t * d, d), w.norm2_weight, w.norm2_bias, dn,
287 grads.norm2_weight, grads.norm2_bias);
288 for (std::size_t i = 0; i < d; ++i)
289 gt[i] += dn[i];
290 }
291 std::fill(grads.scratch.begin(), grads.scratch.end(), T(0));
292 attention_backward<C>(w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(g),
293 grads.scratch, grads.attention);
294 for (std::size_t t = 0; t < n; ++t) {
295 detail::block::norm_backward<C, T>(std::span<const T>(grads.scratch).subspan(t * d, d),
296 std::span<const T>(act.x0).subspan(t * d, d), w.norm1_weight, w.norm1_bias, dn,
297 grads.norm1_weight, grads.norm1_bias);
298 for (std::size_t i = 0; i < d; ++i)
299 g[t * d + i] += dn[i];
300 }
301}

316constexpr void block_forward_sequence(const B& backend, const block_weights<dense_weights<T>, T>& w, std::type_identity_t<std::span<T>> xs,
317 const R& rope_src, const block_activations<T>& act)
318{

320 const std::size_t n = xs.size() / d;
321 assert(xs.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d && act.h2.size() == n * d);
322 std::copy(xs.begin(), xs.end(), act.x0.begin());
323 for (std::size_t t = 0; t < n; ++t)
324 block_norm<C>(xs.subspan(t * d, d), w.norm1_weight, w.norm1_bias, act.h1.subspan(t * d, d));

326 for (std::size_t i = 0; i < n * d; ++i)
327 xs[i] = act.x0[i] + xs[i];
328 std::copy(xs.begin(), xs.end(), act.x1.begin());
329 for (std::size_t t = 0; t < n; ++t)
330 block_norm<C>(xs.subspan(t * d, d), w.norm2_weight, w.norm2_bias, act.h2.subspan(t * d, d));
331 mlp_forward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), xs, mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden});
332 for (std::size_t i = 0; i < n * d; ++i)
333 xs[i] = act.x1[i] + xs[i];

349template <transformer_config C, typename B, math::math_number T, typename R>
350 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
351constexpr void block_backward(const B& backend, const block_weights<dense_weights<T>, T>& w, const R& rope_src,
352 const block_activations<T>& act, std::type_identity_t<std::span<T>> dxs, const block_gradients<T>& grads)
353{
