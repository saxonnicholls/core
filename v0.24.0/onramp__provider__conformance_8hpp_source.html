67 template <typename Probe>
68 concept onramp_provider_probe = std::is_default_constructible_v<Probe> &&
69 onramp_provider<std::remove_reference_t<decltype(std::declval<Probe&>().venue())>> &&
70 requires(Probe& p, capability_errc c, std::string_view r) {
71 { p.unit() } -> std::same_as<asset>;
72 { p.quote_currency() } -> std::same_as<asset>;
73 { p.ticket() } -> std::same_as<money>;
74 { p.too_much() } -> std::same_as<money>;
75 { p.chain() } -> std::same_as<std::uint64_t>;
76 { p.address() } -> std::convertible_to<std::string>;
77 p.provider_fails(c, r);
78 };

99 detail::capability_check<Probe>(report, "a quote is exact in its assets", [](Probe& p, const auto& expect) {
100 const auto q = p.venue().quote({p.unit(), p.quote_currency(), p.ticket()});
101 expect(q.is_ok(), "a funded ticket is quoted");
102 if (!q.is_ok())
103 return;
104 expect(q->spend.asset_of() == p.quote_currency(), "spend is in the quote currency");
105 expect(q->receive.asset_of() == p.unit(), "receive is in the asset");
106 expect(!q->receive.is_zero(), "the ticket buys something");
107 expect(q->price.from() == p.unit() && q->price.to() == p.quote_currency(), "the price is quote per asset");
108 });

116 detail::capability_check<Probe>(report, "a buy is idempotent on its client order id", [&](Probe& p, const auto& expect) {
117 auto& v = p.venue();
118 const auto first = v.buy(kit_buy(p, "c-1", p.ticket()));
119 const auto again = v.buy(kit_buy(p, "c-1", p.ticket()));
120 expect(first.is_ok() && again.is_ok(), "both buys answer");
121 if (!first.is_ok() || !again.is_ok())
122 return;
123 expect(first->order_id == again->order_id, "a repeat returns the first order");
124 expect(code_is(v.buy(kit_buy(p, "c-1", p.too_much())), capability_errc::conflict), "the id with another buy is a conflict");
125 const auto s = v.order_status(first->order_id.view());
126 expect(s.is_ok() && s->client_order_id == std::string_view{"c-1"}, "the order keeps its client order id");
127 const std::string long_id(client_id::capacity() + 1, 'c');

129 "a client order id longer than 64 characters is invalid_argument");

141 detail::capability_check<Probe>(report, "a withdrawal checks chain, address and balance", [&](Probe& p, const auto& expect) {
142 auto& v = p.venue();
143 const auto bought = v.buy(kit_buy(p, "c-w", p.ticket()));
144 expect(bought.is_ok(), "the buy fills");
145 if (!bought.is_ok())
146 return;
147 const std::string address = p.address();

150 "a malformed address is invalid_argument");

154 expect(code_is(v.withdraw({"w-huge", huge, p.chain(), address, ""}), capability_errc::insufficient_funds),
155 "more than the balance is insufficient_funds");

163 const auto again = v.withdraw({"w-ok", one, p.chain(), address, ""});

165 });

184 detail::capability_check<Probe>(report, "the statement pages facts that validate", [](Probe& p, const auto& expect) {
185 auto& v = p.venue();
186 expect(v.buy(kit_buy(p, "c-s1", p.ticket())).is_ok() && v.buy(kit_buy(p, "c-s2", p.ticket())).is_ok(), "two buys fill");
187 const auto first = v.statement({onramp_statement_source::fills, {}, 1});
188 expect(first.is_ok() && first->facts.size() == 1 && !first->next_cursor.empty(), "the first page holds one fact and a cursor");
189 if (!first.is_ok() || first->facts.size() != 1)
190 return;
191 const auto second = v.statement({onramp_statement_source::fills, first->next_cursor.view(), 1});
192 expect(second.is_ok() && second->facts.size() == 1, "the second page holds one fact");
193 if (!second.is_ok() || second->facts.size() != 1)
194 return;
195 expect(first->facts[0].fact_id != second->facts[0].fact_id, "no fact twice");
196 for (const onramp_fact& f : {first->facts[0], second->facts[0]}) {
197 expect(f.kind == onramp_fact_kind::trade_fill, "a buy is a trade_fill");
198 expect(validate(f).is_ok(), "every fact validates");
199 }
200 const auto w = v.statement({onramp_statement_source::withdrawals, {}, 100});
201 expect(w.is_ok(), "the withdrawals statement answers");
202 if (w.is_ok())
203 for (const onramp_fact& f : w->facts)
204 expect(validate(f).is_ok(), "every fact validates");
205 });
