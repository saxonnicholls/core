125 // Runs one check on a fresh probe; an escaping exception is a failure, not a crash.
126 template <typename Probe, typename F>
127 void capability_check(capability_conformance_report& report, std::string_view name, F&& f)
128 {
129 ++report.checks;
130 const capability_expect expect{name, report};
131#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
132 try {
133 Probe probe{};
134 std::forward<F>(f)(probe, expect);
135 } catch (...) {
136 expect(false, "no exception escapes");
137 }
138#else
139 Probe probe{};
140 std::forward<F>(f)(probe, expect);
141#endif
142 }
