163 // Runs f(), turning an exception of the store into an error and letting the caller's visitor's through.
164 // visitor_threw is set by kv_visit when the visitor throws.
165 template <typename R, typename F>
166 R kv_guard(const bool& visitor_threw, F&& f)
167 {
168#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
169 try {
170 return f();
171 } catch (const std::bad_alloc&) {
172 if (visitor_threw)
173 throw;
174 return err(kv_errc::out_of_memory);
175 } catch (...) {
176 if (visitor_threw)
177 throw;
178 return err(kv_errc::backend);
179 }
180#else
181 static_cast<void>(visitor_threw);
182 return f();
183#endif
184 }
185
186 // Calls the caller's visitor, recording that an exception came from it.
187 template <typename F, typename... A>
188 decltype(auto) kv_visit(bool& visitor_threw, F& fn, A... args)
189 {
190#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
191 try {
192 return fn(args...);
193 } catch (...) {
194 visitor_threw = true;
195 throw;
196 }
197#else
198 static_cast<void>(visitor_threw);
199 return fn(args...);
200#endif
201 }
202 }

223 result<bool, kv_errc> read(kv_bytes key, kv_read_visitor fn) const override
224 {
225 bool threw = false;
226 return detail::kv_guard<result<bool, kv_errc>>(threw, [&] {
