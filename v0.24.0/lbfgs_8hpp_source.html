133 constexpr bool push(std::span<const T> s, std::span<const T> y) noexcept(linalg::detail::nothrow_scalar<T>)
134 {
135 T sy(0), yy(0);
136 for (std::size_t i = 0; i < n_; ++i) {
137 sy += s[i] * y[i];
138 yy += y[i] * y[i];
139 }
140 if (!(sy > std::numeric_limits<T>::epsilon() * yy) || !(yy > T(0)))
141 return false;
142 const std::size_t at = head_;
143 for (std::size_t i = 0; i < n_; ++i) {
144 s_[at * n_ + i] = s[i];
145 y_[at * n_ + i] = y[i];
146 }
147 rho_[at] = T(1) / sy;
148 gamma_ = sy / yy;
149 head_ = (head_ + 1) % M;
150 count_ = count_ < M ? count_ + 1 : M;
151 return true;
152 }
153
154 private:
155 // Pair k, oldest first, is in slot (head - count + k) mod M
156 [[nodiscard]] constexpr std::size_t slot(std::size_t k) const noexcept { return (head_ + M - count_ + k) % M; }
157
158 storage s_{};
159 storage y_{};
160 std::array<T, M> rho_{};
161 std::size_t n_ = N == std::dynamic_extent ? 0 : N;
162 std::size_t count_ = 0;
163 std::size_t head_ = 0;
164 T gamma_ = T(1);
165 };

175 template <linalg::scalar T, std::size_t M, std::size_t N>
176 constexpr void two_loop(const lbfgs_history<T, M, N>& history, std::type_identity_t<std::span<const T>> gradient, std::type_identity_t<std::span<T>> direction) noexcept(
177 linalg::detail::nothrow_scalar<T>)
178 {
179 const std::size_t n = history.dimension(), m = history.size();
180 std::array<T, M> alpha{};
181 for (std::size_t i = 0; i < n; ++i)
182 direction[i] = gradient[i];
183 for (std::size_t k = m; k-- > 0;) {
184 const auto s = history.s(k), y = history.y(k);
185 T a(0);
186 for (std::size_t i = 0; i < n; ++i)
187 a += s[i] * direction[i];
188 a *= history.rho(k);
189 alpha[k] = a;
190 for (std::size_t i = 0; i < n; ++i)
191 direction[i] -= a * y[i];
192 }
193 const T gamma = history.gamma();
194 for (std::size_t i = 0; i < n; ++i)
195 direction[i] *= gamma;
196 for (std::size_t k = 0; k < m; ++k) {
197 const auto s = history.s(k), y = history.y(k);
198 T b(0);
199 for (std::size_t i = 0; i < n; ++i)
200 b += y[i] * direction[i];
201 b *= history.rho(k);
202 for (std::size_t i = 0; i < n; ++i)
203 direction[i] += s[i] * (alpha[k] - b);
204 }
205 for (std::size_t i = 0; i < n; ++i)
206 direction[i] = -direction[i];
207 }

263 template <std::size_t M, typename F, linalg::scalar T>
264 [[nodiscard]] constexpr lbfgs_result<T> lbfgs_minimize(F&& fg, std::span<T> x, const std::type_identity_t<lbfgs_options<T>>& options, std::type_identity_t<std::span<T>> work) noexcept(
265 std::is_nothrow_invocable_v<F&, std::span<const T>, std::span<T>> && linalg::detail::nothrow_scalar<T>)
266 {
267 using cspan = std::span<const T>;
268 const std::size_t n = x.size();
269 if (work.size() < lbfgs_workspace(n, M))
270 return {};
271 const auto g = work.subspan(0, n), d = work.subspan(n, n), xt = work.subspan(2 * n, n), gt = work.subspan(3 * n, n);
272 lbfgs_history<T, M> history(work.subspan(4 * n, n * M), work.subspan((4 + M) * n, n * M), n);
273 const auto inf_norm = [&](std::span<const T> v) {
274 T m(0);
275 for (const T e : v)
276 m = detail::ls_abs(e) > m ? detail::ls_abs(e) : m;
277 return m;
278 };
279 T f = fg(cspan(x), g);
280 std::size_t evaluations = 1, k = 0;
281 T gnorm = inf_norm(cspan(g));
282 for (;;) {
283 if (gnorm <= options.gradient_tolerance)
284 return {lbfgs_status::gradient_converged, k, evaluations, f, gnorm};
285 if (k >= options.max_iterations)
286 return {lbfgs_status::iteration_limit, k, evaluations, f, gnorm};
287 wolfe_result<T> ls{};
288 for (int attempt = 0; attempt < 2; ++attempt) {
289 two_loop(history, cspan(g), d);
290 T gd(0), gg(0);
291 for (std::size_t i = 0; i < n; ++i) {
292 gd += g[i] * d[i];
293 gg += g[i] * g[i];
294 }
295 if (!(gd < T(0))) {
296 history.clear();
297 for (std::size_t i = 0; i < n; ++i)
298 d[i] = -g[i];
299 gd = -gg;
300 }
301 wolfe_options<T> lo = options.line_search;
302 if (history.empty()) {
303 const T inv = T(1) / static_cast<T>(math::sqrt(gg));
304 lo.initial_step = inv < T(1) ? inv : T(1);
305 } else
306 lo.initial_step = T(1);
307 ls = strong_wolfe(fg, cspan(x), cspan(d), f, gd, xt, gt, lo);
308 evaluations += ls.evaluations;
309 if (ls.step > T(0) && ls.value < f)
310 break;
311 if (history.empty())
312 return {lbfgs_status::line_search_failed, k, evaluations, f, gnorm};
313 history.clear();
314 }
315 if (!(ls.step > T(0) && ls.value < f))
316 return {lbfgs_status::line_search_failed, k, evaluations, f, gnorm};
317 // s = x_new - x into d, y = g_new - g into g, then move
318 for (std::size_t i = 0; i < n; ++i) {
319 d[i] = xt[i] - x[i];
320 g[i] = gt[i] - g[i];
321 }
322 history.push(cspan(d), cspan(g));
323 for (std::size_t i = 0; i < n; ++i) {
324 x[i] = xt[i];
325 g[i] = gt[i];
326 }
327 const T f_old = f;
328 f = ls.value;
329 gnorm = inf_norm(cspan(g));
330 ++k;
331 if (options.value_tolerance > T(0)) {
332 T scale = detail::ls_abs(f_old) > detail::ls_abs(f) ? detail::ls_abs(f_old) : detail::ls_abs(f);
333 scale = scale > T(1) ? scale : T(1);
334 if (detail::ls_abs(f_old - f) <= options.value_tolerance * scale && !(gnorm <= options.gradient_tolerance))
335 return {lbfgs_status::value_converged, k, evaluations, f, gnorm};
336 }
337 }
338 }
