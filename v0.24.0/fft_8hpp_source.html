126template <math::math_number T>
127constexpr T fft_scale(fft_norm norm, std::size_t n, bool inverse) noexcept
128{
129 if (n == 0)
130 return T(1);
131 const T nn = T(static_cast<long long>(n));
132 switch (norm) {
133 case fft_norm::backward: return inverse ? T(1) / nn : T(1);
134 case fft_norm::forward: return inverse ? T(1) : T(1) / nn;
135 case fft_norm::ortho: return T(1) / math::sqrt(nn);
136 case fft_norm::none: break;
137 }
138 return T(1);
139}

144constexpr bool is_power_of_two(std::size_t n) noexcept { return n != 0 && (n & (n - 1)) == 0; }

149constexpr std::size_t next_power_of_two(std::size_t n) noexcept
150{
151 std::size_t p = 1;
152 while (p < n)
153 p *= 2;
154 return p;

159struct fft_layout
160{
161 std::size_t n = 0;
162 std::size_t m = 0;
163 std::array<std::uint8_t, 64> factors{};
164 std::size_t factor_count = 0;
165
169 static constexpr fft_layout of(std::size_t n) noexcept
170 {
171 fft_layout l;
172 l.n = n;
173 std::size_t r = n;
174 const auto add = [&l](std::size_t p) { l.factors[l.factor_count++] = static_cast<std::uint8_t>(p); };
175 while (r % 4 == 0 && r > 1) { add(4); r /= 4; }
176 if (r % 2 == 0 && r > 1) { add(2); r /= 2; }
177 for (std::size_t p = 3; p <= fft_max_radix && r > 1; p += 2)
178 while (r % p == 0) { add(p); r /= p; }
179 if (r > 1) {
180 l = fft_layout{};
181 l.n = n;
182 l.m = next_power_of_two(2 * n - 1);
183 for (std::size_t q = l.m; q > 1;) {
184 const std::size_t p = q % 4 == 0 ? 4 : 2;
185 add(p);
186 q /= p;
187 }
188 }
189 return l;
190 }
191
194 [[nodiscard]] constexpr bool bluestein() const noexcept { return m != 0; }
197 [[nodiscard]] constexpr std::size_t storage_size() const noexcept { return bluestein() ? n + 2 * m : n; }
200 [[nodiscard]] constexpr std::size_t work_size() const noexcept { return bluestein() ? 2 * m : n; }
203 [[nodiscard]] constexpr std::span<const std::uint8_t> passes() const noexcept { return {factors.data(), factor_count}; }
204};

211 // W_n^k for k < n, from the first octant: bit-identical to twiddle<T>(k, n), with an eighth of the trigonometry.
212 template <typename T>
213 constexpr void fill_twiddles(cx<T>* tw, std::size_t n) noexcept
214 {
215 if (n % 4 != 0) {
216 for (std::size_t k = 0; k < n; ++k)
217 tw[k] = twiddle<T>(k, n);
218 return;
219 }
220 const std::size_t q = n / 4;
221 for (std::size_t k = 0; k < q; ++k)
222 tw[k] = 8 * k <= n ? twiddle<T>(k, n) : cx<T>{-tw[q - k].im, -tw[q - k].re};
223 for (std::size_t k = q; k < n; ++k)
224 tw[k] = cx<T>{tw[k - q].im, -tw[k - q].re}; // times -i
225 }

233 template <bool Inverse, typename T>
234 constexpr void radix2(const cx<T>* src, cx<T>* dst, const cx<T>* tw, std::size_t m, std::size_t s) noexcept
235 {
236 for (std::size_t j = 0; j < m; ++j) {
237 const cx<T> w = tw_at<Inverse>(tw, s * j);
238 const cx<T>* a = src + s * j;
239 const cx<T>* b = src + s * (j + m);
240 cx<T>* y0 = dst + s * (2 * j);
241 cx<T>* y1 = dst + s * (2 * j + 1);
242 for (std::size_t q = 0; q < s; ++q) {
243 const cx<T> u = a[q], v = b[q];
244 y0[q] = u + v;
245 y1[q] = (u - v) * w;
246 }
247 }
248 }

250 template <bool Inverse, typename T>
251 constexpr void radix4(const cx<T>* src, cx<T>* dst, const cx<T>* tw, std::size_t m, std::size_t s) noexcept
252 {
253 for (std::size_t j = 0; j < m; ++j) {
254 const cx<T> w1 = tw_at<Inverse>(tw, s * j), w2 = tw_at<Inverse>(tw, 2 * s * j), w3 = tw_at<Inverse>(tw, 3 * s * j);
255 const cx<T>* a0 = src + s * j;
256 const cx<T>* a1 = src + s * (j + m);
257 const cx<T>* a2 = src + s * (j + 2 * m);
258 const cx<T>* a3 = src + s * (j + 3 * m);
259 cx<T>* y = dst + s * (4 * j);
260 for (std::size_t q = 0; q < s; ++q) {
261 const cx<T> t0 = a0[q] + a2[q], t1 = a0[q] - a2[q], t2 = a1[q] + a3[q], d = a1[q] - a3[q];
262 const cx<T> t3 = Inverse ? cx<T>{-d.im, d.re} : cx<T>{d.im, -d.re}; // times +i or -i
263 y[q] = t0 + t2;
264 y[q + s] = (t1 + t3) * w1;
265 y[q + 2 * s] = (t0 - t2) * w2;
266 y[q + 3 * s] = (t1 - t3) * w3;
267 }
268 }

271 template <bool Inverse, typename T>
272 constexpr void radix_p(const cx<T>* src, cx<T>* dst, const cx<T>* tw, std::size_t n, std::size_t p, std::size_t m,
273 std::size_t s) noexcept
274 {
275 const std::size_t root = n / p; // omega_p^e = W_n^(e n / p)
276 for (std::size_t j = 0; j < m; ++j)
277 for (std::size_t q = 0; q < s; ++q) {
278 std::array<cx<T>, fft_max_radix> a{};
279 for (std::size_t r = 0; r < p; ++r)
280 a[r] = src[q + s * (j + r * m)];
281 for (std::size_t t = 0; t < p; ++t) {
282 cx<T> acc = a[0];
283 std::size_t e = 0;
284 for (std::size_t r = 1; r < p; ++r) {
285 e += t;
286 if (e >= p)
287 e -= p;
288 acc += a[r] * tw_at<Inverse>(tw, e * root);
289 }
290 dst[q + s * (p * j + t)] = acc * tw_at<Inverse>(tw, s * j * t);
291 }
292 }

295 // The Stockham autosort transform: one pass per factor, from x to y and back; the result ends in x.
296 template <bool Inverse, typename T>
297 constexpr void stockham(const cx<T>* tw, std::size_t n, std::span<const std::uint8_t> factors, cx<T>* x, cx<T>* y) noexcept
298 {
299 std::size_t len = n, s = 1;
300 cx<T>* src = x;
301 cx<T>* dst = y;
302 for (const std::uint8_t f : factors) {
303 const std::size_t p = f, m = len / p;
304 switch (p) {
305 case 2: radix2<Inverse>(src, dst, tw, m, s); break;
306 case 4: radix4<Inverse>(src, dst, tw, m, s); break;
307 default: radix_p<Inverse>(src, dst, tw, n, p, m, s); break;
308 }
309 cx<T>* t = src;
310 src = dst;
311 dst = t;
312 len = m;
313 s *= p;
314 }
315 if (src != x)
316 for (std::size_t i = 0; i < n; ++i)
317 x[i] = src[i];
318 }

320 constexpr std::size_t reverse_bits(std::size_t i, std::size_t bits) noexcept
321 {
322 std::size_t r = 0;
323 for (std::size_t b = 0; b < bits; ++b) {
324 r = (r << 1) | (i & 1);
325 i >>= 1;
326 }
327 return r;

338 // In-place radix-2 decimation in time over a power of two, with a bit-reversal permutation given or computed.
339 template <bool Inverse, typename T, typename Reverse>
340 constexpr void in_place_pow2(cx<T>* x, const cx<T>* tw, std::size_t n, Reverse reverse) noexcept
341 {
342 for (std::size_t i = 0; i < n; ++i) {
343 const std::size_t r = reverse(i);
344 if (i < r) {
345 const cx<T> t = x[i];
346 x[i] = x[r];
347 x[r] = t;
348 }
349 }
350 for (std::size_t len = 2; len <= n; len *= 2) {
351 const std::size_t half = len / 2, step = n / len;
352 for (std::size_t start = 0; start < n; start += len)
353 for (std::size_t k = 0; k < half; ++k) {
354 const cx<T> u = x[start + k], v = x[start + k + half] * tw_at<Inverse>(tw, k * step);
355 x[start + k] = u + v;
356 x[start + k + half] = u - v;
357 }
358 }
359 }

376 // Fills a plan's tables: W_n^k, or for Bluestein the chirp, W_m^k and the kernel's spectrum.
377 template <typename T>
378 constexpr void build(const fft_layout& l, cx<T>* st) noexcept
379 {
380 if (!l.bluestein()) {
381 fill_twiddles(st, l.n);
382 return;
383 }
384 const std::size_t n = l.n, m = l.m, two_n = 2 * n;
385 cx<T>* chirp = st;
386 cx<T>* tw = st + n;
387 cx<T>* kernel = st + n + m;
388 std::size_t sq = 0; // i^2 mod 2n, so e^(-i pi i^2 / n) is a twiddle of order 2n
389 for (std::size_t i = 0; i < n; ++i) {
390 chirp[i] = twiddle<T>(sq, two_n);
391 sq = (sq + 2 * i + 1) % two_n;
392 }
393 fill_twiddles(tw, m);
394 for (std::size_t i = 0; i < m; ++i)
395 kernel[i] = {};
396 kernel[0] = conj(chirp[0]);
397 for (std::size_t i = 1; i < n; ++i)
398 kernel[i] = kernel[m - i] = conj(chirp[i]);
399 const std::size_t bits = log2_of(m);
400 in_place_pow2<false>(kernel, tw, m, [bits](std::size_t i) { return reverse_bits(i, bits); });
401 }

403 // An unnormalised transform of x (length l.n) in place.
404 template <bool Inverse, typename T>
405 constexpr void execute(const fft_layout& l, const cx<T>* st, cx<T>* x, cx<T>* work) noexcept
406 {
407 if (!l.bluestein()) {
408 stockham<Inverse>(st, l.n, l.passes(), x, work);
409 return;
410 }
411 const std::size_t n = l.n, m = l.m;
412 const cx<T>* chirp = st;
413 const cx<T>* tw = st + n;
414 const cx<T>* kernel = st + n + m;
415 cx<T>* a = work;
416 cx<T>* b = work + m;
417 if (Inverse)
418 conj_all(x, n);
419 for (std::size_t i = 0; i < n; ++i)
420 a[i] = x[i] * chirp[i];
421 for (std::size_t i = n; i < m; ++i)
422 a[i] = {};
423 stockham<false>(tw, m, l.passes(), a, b);
424 for (std::size_t i = 0; i < m; ++i)
425 a[i] *= kernel[i];
426 stockham<true>(tw, m, l.passes(), a, b);
427 const T inv_m = T(1) / T(static_cast<long long>(m));
428 for (std::size_t k = 0; k < n; ++k)
429 x[k] = chirp[k] * a[k] * inv_m;
430 if (Inverse)
431 conj_all(x, n);
432 }

580template <math::math_number T>
581constexpr void dft(std::span<const complex<std::type_identity_t<T>>> in, std::span<complex<T>> out,
582 std::span<const complex<std::type_identity_t<T>>> table, bool inverse = false,
583 fft_norm norm = fft_norm::backward) noexcept
584{
585 const std::size_t n = in.size();
586 assert(out.size() == n && table.size() == n);
587 const T s = fft_scale<T>(norm, n, inverse);
588 for (std::size_t k = 0; k < n; ++k) {
589 complex<T> acc{};
590 std::size_t e = 0; // j k mod n
591 for (std::size_t j = 0; j < n; ++j) {
592 const complex<T> w = inverse ? conj(table[e]) : table[e];
593 acc += in[j] * w;
594 e += k;
595 if (e >= n)
596 e -= n;
597 }
598 out[k] = acc * s;
599 }
600}

608template <math::math_number T>
609constexpr void dft(std::span<const complex<std::type_identity_t<T>>> in, std::span<complex<T>> out, bool inverse = false,
610 fft_norm norm = fft_norm::backward) noexcept
611{
612 const std::size_t n = in.size();
613 assert(out.size() == n);
614 const T s = fft_scale<T>(norm, n, inverse);
615 for (std::size_t k = 0; k < n; ++k) {
616 complex<T> acc{};
617 std::size_t e = 0;
618 for (std::size_t j = 0; j < n; ++j) {
619 const complex<T> w = twiddle<T>(e, n);
620 acc += in[j] * (inverse ? conj(w) : w);
621 e += k;
622 if (e >= n)
623 e -= n;
624 }
625 out[k] = acc * s;
626 }

871 template <typename In>
872 constexpr void forward_from(In&& in, std::span<complex<T>> out, std::span<complex<T>> work) const noexcept
873 {
874 assert(out.size() == bins() && work.size() >= work_size());
875 const std::size_t n = layout_.n;
876 const fft_plan<T> inner = complex_plan();
877 if (!layout_.even()) {
878 const auto buf = work.first(n);
879 for (std::size_t j = 0; j < n; ++j)
880 buf[j] = {in(j), T(0)};
881 inner.template execute<false>(buf, work.subspan(n));
882 for (std::size_t k = 0; k < bins(); ++k)
883 out[k] = buf[k];
884 out[0].im = T(0); // exactly real, as the even path's
885 return;
886 }
887 const std::size_t h = n / 2;
888 for (std::size_t j = 0; j < h; ++j)
889 out[j] = {in(2 * j), in(2 * j + 1)};
890 inner.template execute<false>(out.first(h), work);
891 const complex<T>* w = tables_.data() + layout_.inner.storage_size();
892 const complex<T> z0 = out[0];
893 out[0] = {z0.re + z0.im, T(0)};
894 out[h] = {z0.re - z0.im, T(0)};
895 for (std::size_t k = 1; 2 * k <= h; ++k) {
896 const complex<T> zk = out[k], zh = out[h - k];
897 const complex<T> e = (zk + conj(zh)) * T(0.5);
898 const complex<T> d = zk - conj(zh);
899 const complex<T> o = complex<T>{d.im, -d.re} * T(0.5); // (zk - conj zh) / 2i
900 const complex<T> wo = w[k] * o;
901 out[k] = e + wo;
902 out[h - k] = conj(e - wo);
903 }
904 }

914 template <typename In, typename Out>
915 constexpr void inverse_to(In&& in, Out&& out, std::span<complex<T>> work) const noexcept
916 {
917 assert(work.size() >= work_size());
918 const std::size_t n = layout_.n;
919 const fft_plan<T> inner = complex_plan();
920 if (!layout_.even()) {
921 const auto buf = work.first(n);
922 buf[0] = {in(0).re, T(0)};
923 for (std::size_t k = 1; k <= n / 2; ++k) {
924 buf[k] = in(k);
925 buf[n - k] = conj(buf[k]);
926 }
927 inner.template execute<true>(buf, work.subspan(n));
928 for (std::size_t j = 0; j < n; ++j)
929 out(j, buf[j].re);
930 return;
931 }
932 const std::size_t h = n / 2;
933 const auto buf = work.first(h);
934 const complex<T>* w = tables_.data() + layout_.inner.storage_size();
935 for (std::size_t k = 0; k < h; ++k) {
936 const complex<T> xk = k == 0 ? complex<T>{in(0).re, T(0)} : in(k);
937 const complex<T> xh = k == 0 ? complex<T>{in(h).re, T(0)} : in(h - k);
938 const complex<T> e = xk + conj(xh);
939 const complex<T> o = (xk - conj(xh)) * conj(w[k]);
940 buf[k] = e + times_i(o);
941 }
942 inner.template execute<true>(buf, work.subspan(h));
943 for (std::size_t j = 0; j < h; ++j) {
944 out(2 * j, buf[j].re);
945 out(2 * j + 1, buf[j].im);
946 }
947 }
948
949private:
950 rfft_layout layout_;
951 std::span<const complex<T>> tables_;
952};

1156template <math::math_number T>
1157constexpr void fftn(std::span<const fft_plan<T>> plans, std::span<complex<std::type_identity_t<T>>> data,
1158 std::span<complex<std::type_identity_t<T>>> work, fft_norm norm = fft_norm::backward, bool inverse = false) noexcept
1159{
1160 std::size_t total = 1;
1161 for (const auto& p : plans)
1162 total *= p.size();
1163 assert(data.size() == total && work.size() >= fftn_work_size(plans));
1164 if (total == 0)
1165 return;
1166 std::size_t stride = total; // the product of the lengths of this axis and those after it, divided as we go
1167 for (const auto& p : plans) {
1168 const std::size_t n = p.size();
1169 stride /= n;
1170 const T s = fft_scale<T>(norm, n, inverse);
1171 const auto line = work.first(n);
1172 const auto rest = work.subspan(n);
1173 for (std::size_t outer = 0; outer < total; outer += n * stride)
1174 for (std::size_t inner = 0; inner < stride; ++inner) {
1175 const std::size_t base = outer + inner;
1176 for (std::size_t i = 0; i < n; ++i)
1177 line[i] = data[base + i * stride];
1178 if (inverse)
1179 p.template execute<true>(line, rest);
1180 else
1181 p.template execute<false>(line, rest);
1182 for (std::size_t i = 0; i < n; ++i)
1183 data[base + i * stride] = line[i] * s;
1184 }
1185 }
