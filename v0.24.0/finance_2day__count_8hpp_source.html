71 constexpr std::int64_t serial(std::chrono::sys_days d) noexcept
72 {
73 return static_cast<std::int64_t>(d.time_since_epoch().count());

75 // 360 * (y2 - y1) + 30 * (m2 - m1) + (d2 - d1), after a convention has adjusted d1 and d2
76 constexpr std::int64_t thirty(const civil_date& a, std::int64_t d1, const civil_date& b, std::int64_t d2) noexcept
77 {
78 return 360 * (b.year - a.year) + 30 * (static_cast<std::int64_t>(b.month) - static_cast<std::int64_t>(a.month)) + (d2 - d1);
79 }

202 static constexpr finance::year_fraction year_fraction(std::chrono::sys_days start, std::chrono::sys_days end) noexcept
203 {
204 std::int64_t a = detail::serial(start), b = detail::serial(end), sign = 1;
205 if (b < a) {
206 const std::int64_t t = a;
207 a = b;
208 b = t;
209 sign = -1;
210 }
211 // 365 * 366 = 133590: a common-year day is 366 / 133590 of a year, a leap-year day 365 / 133590
212 const auto weight = [](std::int64_t year) noexcept { return is_leap_year(year) ? std::int64_t{365} : std::int64_t{366}; };
213 const std::int64_t y1 = civil_from_days(a).year, y2 = civil_from_days(b).year;
214 if (y1 == y2)
215 return {sign * (b - a) * weight(y1), 133590};
216 const std::int64_t first = days_from_civil(y1 + 1, 1, 1) - a; // to the end of the first year
217 const std::int64_t last = b - days_from_civil(y2, 1, 1); // from the start of the last year
218 return {sign * ((y2 - y1 - 1) * 133590 + first * weight(y1) + last * weight(y2)), 133590};
219 }
