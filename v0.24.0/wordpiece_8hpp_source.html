126 // Greedy longest-match-first over one normalised word; appends the pieces, or the unknown token
127 constexpr bool cut(std::string_view word, const token_trie& vocab, std::uint32_t unknown, const wordpiece_options& o,
128 std::span<std::uint32_t> out, std::size_t& count) noexcept
129 {
130 if (o.max_word_chars != 0) {
131 std::size_t chars = 0;
132 for (std::size_t i = 0; i < word.size(); i += decode_utf8(word, i).length)
133 ++chars;
134 if (chars > o.max_word_chars) {
135 if (count == out.size())
136 return false;
137 out[count++] = unknown;
138 return true;
139 }
140 }
141 const std::size_t first = count;
142 for (std::size_t start = 0; start < word.size();) {
143 auto c = vocab.root();
144 bool alive = true;
145 for (const char b : start == 0 ? o.word_prefix : o.continuation_prefix)
146 if (!(alive = vocab.advance(c, b)))
147 break;
148 std::uint32_t best = no_token;
149 std::size_t end = start;
150 for (std::size_t i = start; alive && i < word.size();) {
151 const std::size_t len = decode_utf8(word, i).length;
152 for (std::size_t k = 0; k < len && alive; ++k)
153 alive = vocab.advance(c, word[i + k]);
154 i += len;
155 if (alive)
156 if (const std::uint32_t id = vocab.at(c); id != no_token) {
157 best = id;
158 end = i;
159 }
160 }
161 if (best == no_token) { // the word cannot be cut: it is the unknown token
162 count = first;
163 break;
164 }
165 if (count == out.size())
166 return false;
167 out[count++] = best;
168 start = end;
169 }
170 if (count == first) {
171 if (count == out.size())
172 return false;
173 out[count++] = unknown;
174 }
175 return true;
176 }

187constexpr result<std::size_t, wordpiece_error> wordpiece_encode(std::string_view text, const token_trie& vocab, std::uint32_t unknown,
188 const wordpiece_options& options, std::span<std::uint32_t> out,
189 std::span<char> scratch) noexcept
190{
191 std::size_t count = 0, len = 0;
192 const auto flush = [&]() {
193 const bool fits = len == 0 || detail::wordpiece::cut(std::string_view(scratch.data(), len), vocab, unknown, options, out, count);
194 len = 0;
195 return fits;
196 };
197 const auto append = [&](char32_t cp) {
198 if (scratch.size() - len < utf8_length(cp))
199 return false;
200 len += encode_utf8(cp, scratch.data() + len);
201 return true;
202 };
203 for (std::size_t i = 0; i < text.size();) {
204 const utf8_decoded d = decode_utf8(text, i);
205 i += d.length;
206 const char32_t cp = d.code_point;
207 const unicode_category cat = category(cp);
208 if (is_whitespace(cp)) {
209 if (!flush())
210 return err(wordpiece_error::output_too_small);
211 continue;
212 }
213 if (options.clean_text && (cp == 0 || cp == replacement_character || is_other(cat)))
214 continue;
215 const bool cjk = options.split_cjk && is_bert_cjk(cp);
216 std::array<char32_t, max_decomposition> parts{};
217 const std::size_t n = options.strip_accents ? decompose(cp, parts) : (parts[0] = cp, std::size_t{1});
218 for (std::size_t k = 0; k < n; ++k) {
219 const char32_t p = parts[k];
220 if (options.strip_accents && category(p) == unicode_category::nonspacing_mark)
221 continue;
222 const bool alone = cjk || is_bert_punctuation(p);
223 if (alone && !flush())
224 return err(wordpiece_error::output_too_small);
225 bool fits = true;
226 if (options.lowercase && p == U'İ')
227 fits = append(U'i') && append(U'̇'); // the one unconditional full lowercase mapping
228 else
229 fits = append(options.lowercase ? to_lower(p) : p);
230 if (!fits)
231 return err(wordpiece_error::scratch_too_small);
232 if (alone && !flush())
233 return err(wordpiece_error::output_too_small);
234 }
235 }
236 if (!flush())
237 return err(wordpiece_error::output_too_small);
238 return ok(count);
239}

250template <typename Piece, typename Skip>
251constexpr result<std::size_t, wordpiece_error> wordpiece_decode(std::span<const std::uint32_t> tokens, Piece&& piece, Skip&& skip,
252 const wordpiece_options& options, std::span<char> out)
253{
254 std::size_t at = 0;
255 bool first = true;
256 const auto put = [&](std::string_view s) {
257 if (out.size() - at < s.size())
258 return false;
259 for (const char c : s)
260 out[at++] = c;
261 return true;
262 };
263 for (const std::uint32_t id : tokens) {
264 const std::string_view p = piece(id);
265 if (p.empty())
266 return err(wordpiece_error::unknown_id);
267 if (skip(id))
268 continue;
269 bool starts_word = true;
270 std::string_view body = p;
271 if (!options.continuation_prefix.empty() && p.starts_with(options.continuation_prefix)) {
272 starts_word = false;
273 body = p.substr(options.continuation_prefix.size());
274 } else if (!options.word_prefix.empty()) {
275 starts_word = p.starts_with(options.word_prefix);
276 if (starts_word)
277 body = p.substr(options.word_prefix.size());
278 }
279 if (starts_word && !first && !put(" "))
280 return err(wordpiece_error::output_too_small);
281 if (!put(body))
282 return err(wordpiece_error::output_too_small);
283 first = false;
284 }
285 return ok(at);
286}
