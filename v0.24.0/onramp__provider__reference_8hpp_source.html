102 capability_result<order> buy(const buy_request& b)
103 {
104 if (b.client_order_id.empty())
105 return fail(capability_errc::invalid_argument, {}, "a buy needs a client order id");
106 auto client = make_id<client_id>(b.client_order_id);
107 if (!client)
108 return err(client.unwrap_err());
109 if (const auto it = buys_.find(b.client_order_id); it != buys_.end()) {
110 if (!it->second.matches(b))
111 return fail(capability_errc::conflict, {}, "client order id reused for another buy");
112 return ok(orders_.at(it->second.order_id));
113 }
114 if (auto e = take_injected())
115 return err(*e);
116 auto q = price(b.unit, b.quote_currency, b.amount);
117 if (!q)
118 return err(q.unwrap_err());
119 if (b.max_price && greater(q->price, *b.max_price))
120 return fail(capability_errc::rejected, {}, "the price is above max_price");
121 if (!has(q->spend))
122 return fail(capability_errc::insufficient_funds);
123 move_balance(q->spend, true);
124 move_balance(q->receive, false);
125 const std::string n = std::to_string(++next_id_);
126 order o{id<provider_id>("mem_ord_" + n), *client, order_status::filled, q->receive, q->spend, q->fee, clock_()};
127 orders_.emplace(std::string(o.order_id.view()), o);
128 onramp_fact f = fact("mem_trd_" + n, onramp_fact_kind::trade_fill);
129 f.debit = venue(*subtract(q->spend, q->fee));
130 f.credit = venue(q->receive);
131 f.fee = venue(q->fee);
132 f.price = q->price;
133 f.order_id = o.order_id;
134 f.client_order_id = o.client_order_id;
135 fill_facts_.push_back(f);
136 buys_.emplace(std::string(b.client_order_id),
137 kept_buy{b.unit, b.quote_currency, b.amount, b.max_price, std::string(o.order_id.view())});
138 return ok(o);
139 }

156 capability_result<withdrawal> withdraw(const withdraw_request& w)
157 {
158 if (w.client_reference.empty())
159 return fail(capability_errc::invalid_argument, {}, "a withdrawal needs a client reference");
160 auto client = make_id<client_id>(w.client_reference);
161 if (!client)
162 return err(client.unwrap_err());
163 if (const auto it = withdrawals_by_ref_.find(w.client_reference); it != withdrawals_by_ref_.end()) {
164 if (!it->second.matches(w))
165 return fail(capability_errc::conflict, {}, "client reference reused for another withdrawal");
166 return ok(withdrawals_.at(it->second.withdrawal_id));
167 }
168 if (auto e = take_injected())
169 return err(*e);
170 const auto a = std::find_if(assets_.begin(), assets_.end(),
171 [&](const onramp_asset& x) { return x.unit == w.amount.asset_of(); });
172 if (a == assets_.end())
173 return fail(capability_errc::invalid_argument, {}, "the venue does not deliver that asset");
174 if (std::find(a->chain_ids.begin(), a->chain_ids.end(), w.chain_id) == a->chain_ids.end())
175 return fail(capability_errc::invalid_argument, {}, "the asset is not withdrawn on that chain");
176 if (!is_evm_address(w.address))
177 return fail(capability_errc::invalid_argument, {}, "the address is not an EVM address");
178 if (w.amount.is_negative() || w.amount.is_zero())
179 return fail(capability_errc::invalid_argument, {}, "a withdrawal is positive");
180 const money fee = fee_for(w.amount.asset_of());
181 const auto gross = add(w.amount, fee);
182 if (!gross || !has(*gross))
183 return fail(capability_errc::insufficient_funds);
184 move_balance(*gross, true);
185 withdrawal out;
186 out.withdrawal_id = id<provider_id>("mem_wd_" + std::to_string(++next_id_));
187 out.client_reference = *client;
188 out.status = withdrawal_status::broadcast;
189 out.amount = w.amount;
190 out.fee = fee;
191 out.chain_id = w.chain_id;
192 out.address = id<short_text>(w.address);
193 out.transaction_hash = id<provider_id>("0x" + std::string(sha256::hash(out.withdrawal_id.view()).hex().view()));
194 out.updated_at = clock_();
195 withdrawals_.emplace(std::string(out.withdrawal_id.view()), out);
196 onramp_fact f = fact(std::string(out.withdrawal_id.view()), onramp_fact_kind::withdrawal);
197 f.debit = venue(w.amount);
198 f.credit = w.amount; // the 1:1 move to the chain asset
199 f.fee = venue(fee);
200 f.withdrawal_id = out.withdrawal_id;
201 f.client_reference = out.client_reference;
202 f.chain_id = w.chain_id;
203 f.address = out.address;
204 f.transaction_hash = out.transaction_hash;
205 withdrawal_facts_.push_back(f);
206 withdrawals_by_ref_.emplace(std::string(w.client_reference),
207 kept_withdrawal{w.amount, w.chain_id, std::string(w.address), std::string(w.memo),
208 std::string(out.withdrawal_id.view())});
209 return ok(out);
210 }

236 capability_result<onramp_statement_page> statement(const onramp_statement_query& q)
237 {
238 const auto& all = q.source == onramp_statement_source::fills ? fill_facts_ : withdrawal_facts_;
239 if (q.limit == 0 || q.limit > 100)
240 return fail(capability_errc::invalid_argument, {}, "a page holds 1 to 100 facts");
241 std::size_t skip = 0;
242 for (const char c : q.cursor) {
243 if (c < '0' || c > '9' || skip > all.size())
244 return fail(capability_errc::invalid_argument, {}, "not a cursor this venue gave");
245 skip = skip * 10 + static_cast<std::size_t>(c - '0');
246 }
247 onramp_statement_page page;
248 const std::size_t end = all.size() > skip ? all.size() - skip : 0;
249 const std::size_t begin = end > q.limit ? end - q.limit : 0;
250 page.facts.assign(all.begin() + static_cast<std::ptrdiff_t>(begin), all.begin() + static_cast<std::ptrdiff_t>(end));
251 if (begin > 0)
252 page.next_cursor = id<provider_id>(std::to_string(skip + (end - begin)));
253 return ok(std::move(page));
254 }

337 static bool greater(const rate& a, const rate& b) noexcept
338 {
339 const auto l = uint256::checked_mul(a.numerator(), b.denominator());
340 const auto r = uint256::checked_mul(b.numerator(), a.denominator());
341 return !l || (r && *l > *r); // an overflowing left side counts as greater: refuse rather than fill

344 capability_result<onramp_quote> price(const asset& unit, const asset& quote_currency, const money& amount) const
345 {
346 if (amount.is_negative() || amount.is_zero())
347 return fail(capability_errc::invalid_argument, {}, "an amount is positive");
348 if (amount.asset_of() != unit && amount.asset_of() != quote_currency)
349 return fail(capability_errc::invalid_argument, {}, "the amount is in neither asset");
350 const auto p = prices_.find(pair_key(unit, quote_currency));
351 if (p == prices_.end())
352 return fail(capability_errc::unsupported, {}, "no price for that pair");
353 money spend, receive, fee;
354 if (amount.asset_of() == quote_currency) {
355 spend = amount;
356 fee = fee_on(amount);
357 const auto per_asset = p->second.inverse();
358 const auto net = subtract(amount, fee);
359 const auto got = per_asset && net ? convert(*net, *per_asset, rounding::toward_zero)
360 : result<conversion, money_errc>(err(money_errc::overflow));
361 if (!got)
362 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
363 receive = got->converted;
364 } else {
365 receive = amount;
366 const auto cost = convert(amount, p->second, rounding::ceiling);
367 if (!cost)
368 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
369 fee = fee_on(cost->converted);
370 spend = *add(cost->converted, fee);
371 }
372 if (receive.is_zero())
373 return fail(capability_errc::invalid_argument, {}, "the ticket buys nothing");
374 const utc_s now = clock_();
375 return ok(onramp_quote{p->second, spend, receive, fee, now, now + quote_lifetime});
376 }

378 money fee_on(const money& m) const
379 {
380 const auto r = rate::make(m.asset_of(), m.asset_of(), fee_bps_, 10000);
381 const auto f = r ? convert(m, *r, rounding::ceiling) : result<conversion, money_errc>(err(money_errc::overflow));
382 return f ? f->converted : money::zero(m.asset_of());
383 }

385 money fee_for(const asset& a) const
386 {
387 const auto it = withdrawal_fees_.find(a.code());
388 return it == withdrawal_fees_.end() || it->second.asset_of() != a ? money::zero(a) : it->second;

391 bool has(const money& m) const
392 {
393 const auto it = balances_.find(m.asset_of().code());
394 return it != balances_.end() && it->second.asset_of() == m.asset_of() && it->second.magnitude() >= m.magnitude();

397 void move_balance(const money& m, bool out)
398 {
399 auto it = balances_.find(m.asset_of().code());
400 if (it == balances_.end())
401 it = balances_.emplace(std::string(m.asset_of().code()), money::zero(m.asset_of())).first;
402 it->second = out ? *subtract(it->second, m) : *add(it->second, m);
403 }
