114 [[nodiscard]] constexpr bool is_email_address(std::string_view s) noexcept
115 {
116 if (s.empty() || s.size() > max_email_length)
117 return false;
118 const auto at = s.find('@');
119 if (at == 0 || at == std::string_view::npos || s.find('@', at + 1) != std::string_view::npos)
120 return false;
121 const auto domain = s.substr(at + 1);
122 const auto dot = domain.rfind('.');
123 if (domain.empty() || dot == std::string_view::npos || dot == 0 || dot + 1 == domain.size() || domain.front() == '-')
124 return false;
125 for (const char c : s) {
126 const auto u = static_cast<unsigned char>(c);
127 if (u <= 0x20 || u == 0x7f || c == '<' || c == '>' || c == '"' || c == '(' || c == ')' || c == ',' ||
128 c == ';' || c == ':' || c == '\\' || c == '[' || c == ']')
129 return false;
130 }
131 return domain.find("..") == std::string_view::npos && s.substr(0, at).find("..") == std::string_view::npos;
132 }

158 [[nodiscard]] constexpr capability_result<void> validate(const message& m) noexcept
159 {
160 if (m.idempotency_key.size() > client_id::capacity())
161 return fail(capability_errc::invalid_argument, {}, "idempotency key longer than 64 characters");
162 switch (m.via) {
163 case channel::email:
164 if (!is_email_address(m.to))
165 return fail(capability_errc::invalid_argument, {}, "recipient is not an email address");
166 if (m.subject.empty() || m.subject.size() > max_subject_length || has_line_break(m.subject))
167 return fail(capability_errc::invalid_argument, {}, "subject is empty, too long or has a line break");
168 if (m.text.empty() && m.html.empty())
169 return fail(capability_errc::invalid_argument, {}, "email has no body");
170 return ok();
171 case channel::sms:
172 if (!is_canonical_e164(m.to))
173 return fail(capability_errc::invalid_argument, {}, "recipient is not a canonical E.164 number");
174 if (!m.subject.empty() || !m.html.empty())
175 return fail(capability_errc::invalid_argument, {}, "an SMS has no subject and no HTML");
176 if (m.text.empty() || m.text.size() > max_sms_length)
177 return fail(capability_errc::invalid_argument, {}, "SMS body is empty or too long");
178 return ok();
179 }
180 return fail(capability_errc::invalid_argument, {}, "unknown channel");
181 }
