125 // "event [guard] / action", each part only when there is one.
126 template <class R>
127 std::string row_label()
128 {
129 std::string out(event_info<typename R::event_type>::name);
130 const std::string guard = callable_label<typename R::guard_type>::text();
131 const std::string action = callable_label<typename R::action_type>::text();
132 if (!guard.empty())
133 out += (out.empty() ? "[" : " [") + guard + "]";
134 if (!action.empty()) {
135 out += out.empty() ? "/ " : " / ";
136 out += is_seq_v<typename R::action_type> ? "(" + action + ")" : action;
137 }
138 return out;
139 }

141 inline std::string diagram_id(std::string_view prefix, std::string_view name)
142 {
143 std::string out(prefix);
144 for (const char c : name)
145 out += ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9')) ? c : '_';
146 return out;
147 }

149 inline std::string escaped(std::string_view text)
150 {
151 std::string out;
152 for (const char c : text) {
153 if (c == '"' || c == '\\')
154 out += '\\';
155 out += c;
156 }
157 return out;
158 }

162 // One level of a machine, drawn: its rows in table order, its composites nested where they first appear.
163 template <diagram_language Language, class Def>
164 struct diagram_writer {
165 using info = table_info<Def>;
166 std::string& out;
167 std::string prefix;
168 std::string indent;

170 template <class S>
171 static constexpr bool composite = machine_definition<S>;
172
173 template <class S>
174 std::string id() const
175 {
176 if constexpr (std::is_same_v<S, terminate_state>)
177 return Language == diagram_language::dot ? prefix + "terminate" : "[*]";
178 else
179 return diagram_id(prefix, state_info<S>::name);
180 }

182 void declare_names()
183 {
184 [&]<class... Ss>(meta::type_list<Ss...>) { (declare<Ss>(), ...); }(typename info::states{});
185 }
186 template <class S>
187 void declare()
188 {
189 if constexpr (!std::is_same_v<S, terminate_state> && !composite<S>) {
190 const std::string_view name = state_info<S>::name;
191 if constexpr (Language == diagram_language::dot)
192 out += indent + id<S>() + " [label=\"" + escaped(name) + "\"];\n";
193 else if (diagram_id("", name) != name || !prefix.empty())
194 out += indent + "state \"" + escaped(name) + "\" as " + id<S>() + "\n";
195 }
196 }
197
198 template <class S>
199 void nested()
200 {
201 if constexpr (composite<S>) {
202 if (drawn_[info::template id_of<S>])
203 return;
204 drawn_[info::template id_of<S>] = true;
205 // PlantUML scopes a nested state's name, as sml's example relies on; Mermaid and DOT do not.
206 const std::string child = prefix + diagram_id("", state_info<S>::name) + "_";
207 const std::string inner_prefix = Language == diagram_language::plantuml ? prefix : child;
208 if constexpr (Language == diagram_language::dot) {
209 out += indent + "subgraph cluster_" + child + " {\n";
210 out += indent + " label=\"" + escaped(state_info<S>::name) + "\";\n";
211 } else
212 out += indent + "state " + id<S>() + " {\n";
213 diagram_writer<Language, S> inner{out, inner_prefix, indent + " "};
214 inner.draw();
215 out += indent + "}\n";
216 }
217 }

219 // The node an edge to or from a composite attaches to in DOT: the start point of its cluster.
220 template <class S>
221 std::string endpoint() const
222 {
223 if constexpr (Language == diagram_language::dot && composite<S>)
224 return prefix + diagram_id("", state_info<S>::name) + "_start";
225 else
226 return id<S>();
227 }

229 template <class R>
230 void row(bool& initial_seen)
231 {
232 using Src = src_of<R>;
233 using Dst = typename R::target;
234 if constexpr (R::source::initial) {
235 if (initial_seen && Language != diagram_language::dot)
236 out += indent + "--\n";
237 initial_seen = true;
238 start<Src>();
239 }
240 nested<Src>();
241 const std::string label = row_label<R>();
242 if constexpr (std::is_same_v<Dst, internal_target>) {
243 if constexpr (Language == diagram_language::dot)
244 out += indent + endpoint<Src>() + " -> " + endpoint<Src>() + " [label=\"" + escaped(label) +
245 "\", style=dashed];\n";
246 else
247 out += indent + id<Src>() + " : " + label + "\n";
248 } else {
249 nested<Dst>();
250 if constexpr (Language == diagram_language::dot) {
251 out += indent + endpoint<Src>() + " -> " + endpoint<Dst>() + " [";
252 if constexpr (composite<Src>)
253 out += "ltail=cluster_" + prefix + diagram_id("", state_info<Src>::name) + "_, ";
254 if constexpr (composite<Dst>)
255 out += "lhead=cluster_" + prefix + diagram_id("", state_info<Dst>::name) + "_, ";
256 out += "label=\"" + escaped(label) + "\"];\n";
257 } else {
258 out += indent + id<Src>() + " --> " + id<Dst>();
259 if (!label.empty())
260 out += " : " + label;
261 out += "\n";
262 }
263 }
264 }

266 template <class S>
267 void start()
268 {
269 if constexpr (Language == diagram_language::dot) {
270 const std::string point = prefix + "start" + std::to_string(starts_++);
271 out += indent + point + " [shape=point];\n";
272 out += indent + point + " -> " + endpoint<S>() + (composite<S> ? " [lhead=cluster_" + prefix +
273 diagram_id("", state_info<S>::name) + "_];\n"
274 : std::string(";\n"));
275 } else
276 out += indent + "[*] --> " + id<S>() + "\n";
277 }

312 {
313 using D = typename detail::definition_of<Def>::type;
314 std::string out = "stateDiagram-v2\n";
315 if constexpr (detail::table_info<D>::regions > 1) {
