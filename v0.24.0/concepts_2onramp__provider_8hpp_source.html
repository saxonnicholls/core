216 [[nodiscard]] constexpr bool is_evm_address(std::string_view s) noexcept
217 {
218 if (s.size() != 42 || s[0] != '0' || (s[1] != 'x' && s[1] != 'X'))
219 return false;
220 for (std::size_t i = 2; i < s.size(); ++i) {
221 const char c = s[i];
222 if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')))
223 return false;
224 }
225 return true;
226 }

269 [[nodiscard]] constexpr bool same_value(const money& a, const money& b) noexcept
270 {
271 if (a.is_negative() != b.is_negative())
272 return false;
273 const unsigned sa = a.asset_of().scale(), sb = b.asset_of().scale();
274 const auto up_a = uint256::checked_mul(a.magnitude(), uint256::pow10(sa < sb ? sb - sa : 0u));
275 const auto up_b = uint256::checked_mul(b.magnitude(), uint256::pow10(sb < sa ? sa - sb : 0u));
276 return up_a && up_b && *up_a == *up_b;
277 }

283 [[nodiscard]] constexpr capability_result<void> validate(const onramp_fact& f) noexcept
284 {
285 if (f.source.empty() || f.fact_id.empty() || f.account.empty())
286 return fail(capability_errc::invalid_argument, {}, "a fact needs its source, fact id and account");
287 if (f.debit.is_negative() || f.credit.is_negative() || f.fee.is_negative())
288 return fail(capability_errc::invalid_argument, {}, "a fact's amounts are not negative");
289 switch (f.kind) {
290 case onramp_fact_kind::trade_fill:
291 if (f.debit.is_zero() || f.credit.is_zero() || !f.price || f.order_id.empty())
292 return fail(capability_errc::invalid_argument, {}, "a fill needs both legs, a price and an order");
293 if (f.debit.asset_of() == f.credit.asset_of())
294 return fail(capability_errc::invalid_argument, {}, "a fill moves two assets");
295 break;
296 case onramp_fact_kind::withdrawal:
297 case onramp_fact_kind::withdrawal_reversed:
298 if (f.debit.is_zero() || f.chain_id == 0 || f.address.empty() || f.withdrawal_id.empty())
299 return fail(capability_errc::invalid_argument, {}, "a withdrawal needs an amount, chain, address and id");
300 if (f.credit.asset_of() == f.debit.asset_of() || !same_value(f.debit, f.credit))
301 return fail(capability_errc::invalid_argument, {}, "a withdrawal moves one value 1:1 to the chain asset");
302 break;
303 case onramp_fact_kind::internal_transfer:
304 if (f.debit.is_zero() || f.debit != f.credit || !f.fee.is_zero() || f.to_account.empty() || f.to_account == f.account)
305 return fail(capability_errc::invalid_argument, {}, "a transfer moves one amount between two accounts");
306 break;
307 }
308 return ok();
309 }
