134[[nodiscard]] constexpr std::size_t block_parameter_count(const transformer_config& c) noexcept
135{
136 const std::size_t d = c.d_model, norms = 2 * (c.norm == norm_kind::layer ? 2 * d : d);
137 const std::size_t attention = c.q_size() * d + 2 * c.kv_size() * d + d * c.q_size()
138 + (c.bias ? c.q_size() + 2 * c.kv_size() + d : 0);
139 const std::size_t mats = gated(c.mlp) ? 3 : 2;
140 const std::size_t mlp = mats * c.d_ff * d + (c.bias ? (mats - 1) * c.d_ff + d : 0);
141 return norms + attention + mlp;
142}

148[[nodiscard]] constexpr std::size_t jepa_predictor_parameters(const jepa_predictor_config& c) noexcept
149{
150 const std::size_t dp = c.block.d_model, de = c.encoder_width, b = c.block.bias ? 1 : 0;
151 return (dp * de + b * dp) + dp + c.depth * block_parameter_count(c.block) + (c.block.norm == norm_kind::layer ? 2 * dp : dp)
152 + (de * dp + b * de);
153}

170template <typename B, math::math_number T>
171constexpr void jepa_predictor_input_rows(const B&, std::type_identity_t<std::span<const T>> context, std::type_identity_t<std::span<const T>> mask_token,
172 std::type_identity_t<std::span<const T>> positions, std::span<const std::uint32_t> context_index,
173 std::span<const std::uint32_t> target_index, std::span<T> out, std::size_t width,
174 std::size_t sequences = 1) noexcept(std::is_floating_point_v<T>)
175{
176 assert(width > 0 && sequences > 0 && mask_token.size() == width && context_index.size() % sequences == 0
177 && target_index.size() % sequences == 0 && context.size() == context_index.size() * width
178 && out.size() == (context_index.size() + target_index.size()) * width && positions.size() % width == 0);
179 const std::size_t kc = context_index.size() / sequences, kt = target_index.size() / sequences, n = kc + kt;
180 for (std::size_t s = 0; s < sequences; ++s) {
181 for (std::size_t j = 0; j < kc; ++j)
182 for (std::size_t c = 0; c < width; ++c) {
183 const T x = context[(s * kc + j) * width + c];
184 out[(s * n + j) * width + c] = positions.empty() ? x : x + positions[context_index[s * kc + j] * width + c];
185 }
186 for (std::size_t j = 0; j < kt; ++j)
187 for (std::size_t c = 0; c < width; ++c)
188 out[(s * n + kc + j) * width + c] =
189 positions.empty() ? mask_token[c] : mask_token[c] + positions[target_index[s * kt + j] * width + c];
190 }
191}

204constexpr void jepa_predictor_input_rows_backward(const B&, std::type_identity_t<std::span<const T>> dout, std::size_t kc, std::size_t kt,
205 std::span<T> dcontext, std::type_identity_t<std::span<T>> dmask_token, std::size_t width,

207{

212 if (!dcontext.empty())
213 for (std::size_t i = 0; i < kc * width; ++i)

216 for (std::size_t c = 0; c < width; ++c)

249 template <typename T>
250 constexpr T sign(T x) noexcept
251 {
252 return x > T(0) ? T(1) : (x < T(0) ? T(-1) : T(0));
253 }
254
255 // One element's term and its derivative in the difference
256 template <typename T>
257 constexpr T term(jepa_loss k, T d) noexcept
258 {
259 const T a = d < T(0) ? -d : d;
260 switch (k) {
261 case jepa_loss::l1:
262 return a;
263 case jepa_loss::smooth_l1:
264 return a < T(1) ? T(0.5) * d * d : a - T(0.5);
265 default:
266 return d * d;
267 }
268 }
269 template <typename T>
270 constexpr T slope(jepa_loss k, T d) noexcept
271 {
272 switch (k) {
273 case jepa_loss::l1:
274 return sign(d);
275 case jepa_loss::smooth_l1:
276 return (d < T(0) ? -d : d) < T(1) ? d : sign(d);
277 default:
278 return T(2) * d;
279 }
280 }

300template <typename B, math::math_number T>
301constexpr T jepa_loss_rows(const B&, jepa_loss kind, std::type_identity_t<std::span<const T>> pred, std::type_identity_t<std::span<const T>> target,
302 std::size_t width, std::span<T> dpred, std::type_identity_t<std::span<T>> dtarget = {},
303 std::span<const std::size_t> lengths = {}, std::type_identity_t<T> weight = T(1))
304{
305 assert(width > 0 && pred.size() == target.size() && pred.size() % width == 0 && (dpred.empty() || dpred.size() == pred.size())
306 && (dtarget.empty() || dtarget.size() == pred.size()));
307 const std::size_t rows = pred.size() / width, sequences = lengths.empty() ? 1 : lengths.size(), per = rows / sequences;
308 assert(rows == per * sequences);
309 std::size_t count = 0;
310 for (std::size_t s = 0; s < sequences; ++s) {
311 assert(lengths.empty() || lengths[s] <= per);
312 count += lengths.empty() ? per : lengths[s];
313 }
314 if (count == 0)
315 return T(0);
316 const bool cosine = kind == jepa_loss::cosine;
317 const T g = cosine ? weight / T(count) : weight / T(count * width);
318 T total = T(0);
319 for (std::size_t s = 0; s < sequences; ++s)
320 for (std::size_t r = s * per; r < s * per + (lengths.empty() ? per : lengths[s]); ++r) {
321 const auto p = pred.subspan(r * width, width), t = target.subspan(r * width, width);
322 if (cosine) {
323 T pt = T(0), pp = T(0), tt = T(0);
324 for (std::size_t c = 0; c < width; ++c) {
325 pt += p[c] * t[c];
326 pp += p[c] * p[c];
327 tt += t[c] * t[c];
328 }
329 const T np = math::sqrt(pp), nt = math::sqrt(tt), eps = T(1e-8);
330 const T dp = np > eps ? np : eps, dt = nt > eps ? nt : eps, cs = pt / (dp * dt);
331 total += T(1) - cs;
332 for (std::size_t c = 0; c < width; ++c) {
333 // d(1 - cos)/dp = cos p / |p|² - t / (|p| |t|) when |p| > eps, else -t / (eps |t|)
334 if (!dpred.empty())
335 dpred[r * width + c] += g * ((np > eps ? cs * p[c] / (np * np) : T(0)) - t[c] / (dp * dt));
336 if (!dtarget.empty())
337 dtarget[r * width + c] += g * ((nt > eps ? cs * t[c] / (nt * nt) : T(0)) - p[c] / (dp * dt));
338 }
339 } else {
340 T row = T(0);
341 for (std::size_t c = 0; c < width; ++c) {
342 const T d = p[c] - t[c];
343 row += detail::jepa::term(kind, d);
344 const T dd = g * detail::jepa::slope(kind, d);
345 if (!dpred.empty())
346 dpred[r * width + c] += dd;
347 if (!dtarget.empty())
348 dtarget[r * width + c] -= dd;
349 }
350 total += row;
351 }
352 }
353 return cosine ? weight * (total / T(count)) : weight * (total / T(count * width));
354}

365template <typename B, math::math_number T>
366constexpr void ema_rows(const B&, std::span<T> target, std::type_identity_t<std::span<const T>> online, double momentum) noexcept(std::is_floating_point_v<T>)
367{
368 assert(target.size() == online.size() && momentum >= 0 && momentum <= 1);
369 const T m = static_cast<T>(momentum), r = static_cast<T>(1.0 - momentum);
370 for (std::size_t i = 0; i < target.size(); ++i) {
371 const T kept = target[i] * m;
372 const T added = r * online[i];
373 target[i] = kept + added;
374 }
375}

413[[nodiscard]] constexpr double ema_momentum(std::uint64_t step, std::uint64_t total, double m0 = 0.996, double m1 = 1.0,
414 momentum_schedule schedule = momentum_schedule::linear) noexcept
415{
416 assert(total > 0);
417 if (step >= total)
418 return m1;
419 const double f = double(step) / double(total);
420 if (schedule == momentum_schedule::cosine)
421 return m1 - (m1 - m0) * (math::cos(std::numbers::pi * f) + 1.0) / 2.0;
422 return m0 + double(step) * (m1 - m0) / double(total);
423}
