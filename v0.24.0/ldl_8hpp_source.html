137 template <typename T, typename I>
138 [[nodiscard]] constexpr result<void, sparse_error> etree(const csc_view<T, I>& a, std::span<std::remove_const_t<I>> parent, std::span<std::remove_const_t<I>> work) noexcept
139 {
140 using J = std::remove_const_t<I>;
141 const std::size_t n = a.cols;
142 if (!detail::ldl::square_valid<J>(a.rows, n))
143 return err(sparse_error::dimension_mismatch);
144 if (!is_valid(a))
145 return err(sparse_error::invalid_index);
146 if (parent.size() < n || work.size() < n)
147 return err(sparse_error::insufficient_capacity);
148 const auto ancestor = work.first(n);
149 for (std::size_t k = 0; k < n; ++k) {
150 parent[k] = ancestor[k] = -1;
151 for (std::size_t q = detail::at(a.col_ptr[k]); q < detail::at(a.col_ptr[k + 1]); ++q) {
152 // Climb from i to the root of its current subtree, pointing every node on the way at k
153 for (J i = a.row_idx[q]; i != -1 && detail::at(i) < k;) {
154 const J up = ancestor[i];
155 ancestor[i] = static_cast<J>(k);
156 if (up == -1)
157 parent[i] = static_cast<J>(k);
158 i = up;
159 }
160 }
161 }
162 return ok();
163 }

175 template <typename T, typename I>
176 [[nodiscard]] constexpr result<ldl_symbolic_result<std::remove_const_t<I>>, sparse_error>
177 ldl_symbolic(const csc_view<T, I>& a, std::span<std::remove_const_t<I>> parent, std::span<std::remove_const_t<I>> counts, std::span<std::remove_const_t<I>> col_ptr,
178 std::span<std::remove_const_t<I>> work) noexcept
179 {
180 using J = std::remove_const_t<I>;
181 const std::size_t n = a.cols;
182 if (!detail::ldl::square_valid<J>(a.rows, n))
183 return err(sparse_error::dimension_mismatch);
184 if (!is_valid(a))
185 return err(sparse_error::invalid_index);
186 if (parent.size() < n || counts.size() < n || col_ptr.size() < n + 1 || work.size() < n)
187 return err(sparse_error::insufficient_capacity);
188 const auto flag = work.first(n);
189 for (std::size_t k = 0; k < n; ++k) {
190 const J kk = static_cast<J>(k);
191 parent[k] = -1;
192 flag[k] = kk;
193 counts[k] = 0;
194 for (std::size_t q = detail::at(a.col_ptr[k]); q < detail::at(a.col_ptr[k + 1]); ++q) {
195 // Row k of L has an entry in every column on the tree path from i up to k
196 for (J i = a.row_idx[q]; detail::at(i) < k && flag[i] != kk; i = parent[i]) {
197 if (parent[i] == -1)
198 parent[i] = kk;
199 ++counts[i];
200 flag[i] = kk;
201 }
202 }
203 }
204 std::size_t total = 0;
205 col_ptr[0] = 0;
206 for (std::size_t k = 0; k < n; ++k) {
207 total += detail::at(counts[k]);
208 if (total > static_cast<std::size_t>(std::numeric_limits<J>::max()))
209 return err(sparse_error::insufficient_capacity);
210 col_ptr[k + 1] = static_cast<J>(total);
211 }
212 return ok(ldl_symbolic_result<J>{n, parent.first(n), counts.first(n), col_ptr.first(n + 1), total});
213 }

235 template <typename T, typename I, typename U>
236 [[nodiscard]] constexpr result<std::size_t, ldl_error> ldl_numeric(const csc_view<T, I>& a, const ldl_symbolic_result<std::remove_const_t<I>>& symbolic,
237 csc_view<U, std::remove_const_t<I>>& l, std::span<std::type_identity_t<U>> d,
238 std::span<std::type_identity_t<U>> dinv, std::span<std::remove_const_t<I>> iwork,
239 std::span<std::type_identity_t<U>> xwork, std::span<const std::int8_t> signs = {},
240 std::type_identity_t<U> delta = static_cast<U>(1e-8)) noexcept
241 {
242 using J = std::remove_const_t<I>;
243 const std::size_t n = symbolic.n;
244 if (a.rows != n || a.cols != n || symbolic.col_ptr.size() < n + 1 || symbolic.parent.size() < n)
245 return err(ldl_error::dimension_mismatch);
246 if (l.col_ptr.size() < n + 1 || l.row_idx.size() < symbolic.nnz || l.values.size() < symbolic.nnz || d.size() < n || dinv.size() < n || iwork.size() < 3 * n
247 || xwork.size() < n || (!signs.empty() && signs.size() < n))
248 return err(ldl_error::insufficient_capacity);
249 l.rows = l.cols = n;
250 if (l.col_ptr.data() != symbolic.col_ptr.data())
251 for (std::size_t k = 0; k <= n; ++k)
252 l.col_ptr[k] = symbolic.col_ptr[k];
253 const auto lp = symbolic.col_ptr;
254 const auto parent = symbolic.parent;
255 const auto pattern = iwork.first(n), flag = iwork.subspan(n, n), lnz = iwork.subspan(2 * n, n);
256 const auto y = xwork.first(n);
257 std::size_t regularised = 0;
258 for (std::size_t k = 0; k < n; ++k) {
259 const J kk = static_cast<J>(k);
260 // The pattern of row k (the reach of column k in the tree), topologically ordered in pattern[top, n)
261 y[k] = U(0);
262 std::size_t top = n;
263 flag[k] = kk;
264 lnz[k] = 0;
265 for (std::size_t q = detail::at(a.col_ptr[k]); q < detail::at(a.col_ptr[k + 1]); ++q) {
266 J i = a.row_idx[q];
267 if (detail::at(i) > k)
268 continue;
269 y[detail::at(i)] += static_cast<U>(a.values[q]);
270 std::size_t depth = 0;
271 for (; i != -1 && flag[i] != kk; i = parent[i]) {
272 pattern[depth++] = i;
273 flag[i] = kk;
274 }
275 while (depth > 0)
276 pattern[--top] = pattern[--depth];
277 }
278 // The sparse triangular solve for row k, and the pivot
279 U dk = y[k];
280 y[k] = U(0);
281 for (; top < n; ++top) {
282 const std::size_t i = detail::at(pattern[top]);
283 const U yi = y[i];
284 y[i] = U(0);
285 const std::size_t p0 = detail::at(lp[i]), p1 = p0 + detail::at(lnz[i]);
286 for (std::size_t p = p0; p < p1; ++p)
287 y[detail::at(l.row_idx[p])] -= l.values[p] * yi;
288 const U lki = yi * dinv[i];
289 dk -= lki * yi;
290 l.row_idx[p1] = kk;
291 l.values[p1] = lki;
292 ++lnz[i];
293 }
294 if (!(dk - dk == U(0)))
295 return err(ldl_error::zero_pivot); // not finite
296 const int s = signs.empty() ? 0 : signs[k] > 0 ? 1 : signs[k] < 0 ? -1 : 0;
297 if (s != 0 && !(static_cast<U>(s) * dk >= delta)) {
298 dk = static_cast<U>(s) * delta;
299 ++regularised;
300 }
301 if (dk == U(0))
302 return err(ldl_error::zero_pivot);
303 d[k] = dk;
304 dinv[k] = U(1) / dk;
305 }
306 return ok(regularised);
307 }

343 template <typename T, typename I, typename U>
344 constexpr void ldl_solve(const csc_view<T, I>& l, std::span<const std::type_identity_t<U>> dinv, std::span<U> x) noexcept
345 {
346 const std::size_t n = l.cols;
347 for (std::size_t j = 0; j < n; ++j) {
348 const U xj = x[j];
349 for (std::size_t p = detail::at(l.col_ptr[j]); p < detail::at(l.col_ptr[j + 1]); ++p)
350 x[detail::at(l.row_idx[p])] -= static_cast<U>(l.values[p]) * xj;
351 }
352 for (std::size_t j = 0; j < n; ++j)
353 x[j] *= dinv[j];
354 for (std::size_t j = n; j-- > 0;) {
355 U s = x[j];
356 for (std::size_t p = detail::at(l.col_ptr[j]); p < detail::at(l.col_ptr[j + 1]); ++p)
357 s -= static_cast<U>(l.values[p]) * x[detail::at(l.row_idx[p])];
358 x[j] = s;
359 }
360 }

380 template <typename T, typename I, typename TL, typename IL, typename U>
381 constexpr ldl_refine_result<U> ldl_refine(const csc_view<T, I>& a, const csc_view<TL, IL>& l, std::span<const std::type_identity_t<U>> dinv,
382 std::span<const std::type_identity_t<U>> b, std::span<U> x, std::span<std::type_identity_t<U>> residual,
383 std::size_t iterations, std::type_identity_t<U> tolerance) noexcept
384 {
385 const std::size_t n = a.cols;
386 for (std::size_t it = 0;; ++it) {
387 spmv_symmetric_upper(a, std::span<const U>(x.data(), n), residual.first(n));
388 U norm(0);
389 for (std::size_t i = 0; i < n; ++i) {
390 residual[i] = b[i] - residual[i];
391 const U m = residual[i] < U(0) ? -residual[i] : residual[i];
392 norm = m > norm ? m : norm;
393 }
394 if (norm <= tolerance || it == iterations)
395 return {it, norm};
396 ldl_solve(l, dinv, residual.first(n));
397 for (std::size_t i = 0; i < n; ++i)
398 x[i] += residual[i];
399 }
400 }
