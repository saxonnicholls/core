115template <typename S, math::math_number T>
116constexpr void linear_backward_input(const dense_weights<S>& w, std::span<const T> dy, std::type_identity_t<std::span<T>> dx) noexcept(std::is_floating_point_v<T>)
117{
118 assert(dy.size() == w.rows && dx.size() == w.cols);
119 for (std::size_t i = 0; i < w.rows; ++i) {
120 const auto r = w.row(i);
121 for (std::size_t j = 0; j < w.cols; ++j) {
122 const T p = detail::linear::widen<T>(r[j]) * dy[i];
123 dx[j] += p;
124 }
125 }
126}

132constexpr void linear_sequence_backward_input(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> dys,
133 std::type_identity_t<std::span<T>> dxs)

146template <typename W, math::math_number T>
147constexpr void lora_linear(const lora_weights<W, T>& w, std::type_identity_t<std::span<const T>> x, std::type_identity_t<std::span<T>> y,
148 std::type_identity_t<std::span<T>> u, std::type_identity_t<std::span<const T>> bias = {}) noexcept(std::is_floating_point_v<T>)
149{
150 const std::size_t rows = w.rows(), cols = w.cols(), r = w.rank;
151 assert(x.size() == cols && y.size() == rows && u.size() == r && w.a.size() == r * cols && w.b.size() == rows * r);
152 linear(w.base, x, y, bias);
153 for (std::size_t k = 0; k < r; ++k) {
154 T acc = T(0);
155 for (std::size_t j = 0; j < cols; ++j)
156 acc += w.a[k * cols + j] * x[j];
157 u[k] = acc;
158 }
159 for (std::size_t i = 0; i < rows; ++i) {
160 T acc = T(0);
161 for (std::size_t k = 0; k < r; ++k)
162 acc += w.b[i * r + k] * u[k];
163 y[i] += w.scaling * acc;
164 }
165}

171template <typename W, math::math_number T>
172constexpr void lora_linear_backward(const lora_weights<W, T>& w, std::type_identity_t<std::span<const T>> x, std::type_identity_t<std::span<const T>> u,
173 std::type_identity_t<std::span<const T>> dy, std::type_identity_t<std::span<T>> dx,
174 std::type_identity_t<std::span<T>> da, std::type_identity_t<std::span<T>> db,
175 std::type_identity_t<std::span<T>> du) noexcept(std::is_floating_point_v<T>)
176{
177 const std::size_t rows = w.rows(), cols = w.cols(), r = w.rank;
178 assert(x.size() == cols && u.size() == r && dy.size() == rows && (dx.empty() || dx.size() == cols) && da.size() == r * cols
179 && db.size() == rows * r && du.size() == r);
180 for (std::size_t k = 0; k < r; ++k)
181 du[k] = T(0);
182 for (std::size_t i = 0; i < rows; ++i) {
183 const T g = w.scaling * dy[i];
184 for (std::size_t k = 0; k < r; ++k) {
185 du[k] += w.b[i * r + k] * g;
186 db[i * r + k] += g * u[k];
187 }
188 }
189 for (std::size_t k = 0; k < r; ++k)
190 for (std::size_t j = 0; j < cols; ++j)
191 da[k * cols + j] += du[k] * x[j];
192 if (dx.empty())
193 return;
194 linear_backward_input(w.base, dy, dx);
195 for (std::size_t k = 0; k < r; ++k)
196 for (std::size_t j = 0; j < cols; ++j)
197 dx[j] += w.a[k * cols + j] * du[k];
198}

204template <typename W, math::math_number T, typename B>
205 requires linalg::blas_backend<B, T>
206constexpr void lora_sequence(const B& backend, const lora_weights<W, T>& w, std::type_identity_t<std::span<const T>> xs,
207 std::type_identity_t<std::span<T>> ys, std::type_identity_t<std::span<T>> us,
208 std::type_identity_t<std::span<const T>> bias = {})
209{
210 const std::size_t rows = w.rows(), cols = w.cols(), r = w.rank;
211 assert(cols > 0 && xs.size() % cols == 0);
212 const std::size_t n = xs.size() / cols;
213 assert(ys.size() == n * rows && us.size() == n * r && w.a.size() == r * cols && w.b.size() == rows * r);
214 linear_sequence(backend, w.base, xs, ys, bias);
215 if (n == 0 || r == 0)
216 return;
217 linear_sequence(backend, dense_weights<T>{w.a, r, cols}, xs, us);
218 using detail::linear::view;
219 detail::linear::product(backend, w.scaling, view(std::span<const T>(us), n, r), linalg::transposed(view(w.b, rows, r)), T(1),
220 view(ys, n, rows));
221}

262template <math::math_number T, typename B>
263 requires linalg::blas_backend<B, T>
264constexpr void lora_heads(const B& backend, const lora_weights<dense_weights<T>, T>& w, std::type_identity_t<std::span<const T>> xs,
265 std::type_identity_t<std::span<T>> out, std::type_identity_t<std::span<T>> us,
266 std::type_identity_t<std::span<const T>> bias, std::size_t heads, std::size_t sequences = 1)
267{
268 const std::size_t rows = w.rows(), cols = w.cols(), r = w.rank;
269 assert(heads > 0 && sequences > 0 && rows % heads == 0 && cols > 0 && xs.size() % (cols * sequences) == 0);
270 const std::size_t d = rows / heads, n = xs.size() / cols / sequences;
271 assert(us.size() == sequences * n * r && out.size() == sequences * n * rows);
272 linear_heads(backend, w.base, xs, out, bias, heads, sequences);
273 if (n == 0 || r == 0)
274 return;
275 linear_sequence(backend, dense_weights<T>{w.a, r, cols}, xs, us);
276 using detail::linear::view;
277 for (std::size_t s = 0; s < sequences; ++s)
278 for (std::size_t h = 0; h < heads; ++h)
279 detail::linear::product(backend, w.scaling, view(std::span<const T>(us).subspan(s * n * r, n * r), n, r),
280 linalg::transposed(view(w.b.subspan(h * d * r, d * r), d, r)), T(1),
281 view(out.subspan((s * heads + h) * n * d, n * d), n, d));
282}

339template <math::math_number T>
340constexpr void lora_initialise(std::span<T> a, std::type_identity_t<std::span<T>> b, std::size_t cols, stats::wyrand& g,
341 lora_init kind = lora_init::kaiming_uniform) noexcept(std::is_floating_point_v<T>)
342{
343 assert(cols > 0 && a.size() % cols == 0);
344 const std::size_t rank = a.size() / cols;
345 for (T& v : b)
346 v = T(0);
347 const double bound = 1.0 / math::sqrt(static_cast<double>(cols));
348 for (T& v : a) {
349 if (kind == lora_init::kaiming_uniform) {
350 v = T((2.0 * stats::uniform_real(g) - 1.0) * bound);
351 } else {
352 const double u = (static_cast<double>(g() >> 11) + 0.5) * 0x1.0p-53; // in (0, 1): a finite quantile
353 v = T(math::normal_quantile(u) / static_cast<double>(rank));
354 }
355 }

360 template <typename T>
361 constexpr T update(std::span<const T> a, std::span<const T> b, std::size_t rank, std::size_t cols, std::size_t i, std::size_t j)
362 {
363 T acc = T(0);
364 for (std::size_t k = 0; k < rank; ++k)
365 acc += b[i * rank + k] * a[k * cols + j];
366 return acc;
367 }

373template <math::math_number T>
374constexpr void lora_merge(std::span<T> w, std::size_t cols, std::type_identity_t<std::span<const T>> a,
375 std::type_identity_t<std::span<const T>> b, std::size_t rank, std::type_identity_t<T> scaling) noexcept(std::is_floating_point_v<T>)
376{
377 assert(cols > 0 && w.size() % cols == 0 && a.size() == rank * cols && b.size() == w.size() / cols * rank);
378 for (std::size_t i = 0; i < w.size() / cols; ++i)
379 for (std::size_t j = 0; j < cols; ++j)
380 w[i * cols + j] += scaling * detail::lora::update(a, b, rank, cols, i, j);
381}

385template <math::math_number T>
386constexpr void lora_unmerge(std::span<T> w, std::size_t cols, std::type_identity_t<std::span<const T>> a,
387 std::type_identity_t<std::span<const T>> b, std::size_t rank, std::type_identity_t<T> scaling) noexcept(std::is_floating_point_v<T>)
388{
389 assert(cols > 0 && w.size() % cols == 0 && a.size() == rank * cols && b.size() == w.size() / cols * rank);
390 for (std::size_t i = 0; i < w.size() / cols; ++i)
391 for (std::size_t j = 0; j < cols; ++j)
392 w[i * cols + j] -= scaling * detail::lora::update(a, b, rank, cols, i, j);
393}
