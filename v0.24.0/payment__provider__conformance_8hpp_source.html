75 template <typename Probe>
76 concept payment_provider_probe = std::is_default_constructible_v<Probe> &&
77 payment_provider<std::remove_reference_t<decltype(std::declval<Probe&>().provider())>> &&
78 requires(Probe& p, const collection& c, const webhook_event& e, capability_errc code, std::string_view r) {
79 { p.amount() } -> std::same_as<money>;
80 { p.unsupported_currency() } -> std::same_as<asset>;
81 p.payer_pays(c);
82 { p.deliver(e) } -> std::same_as<owned_delivery>;
83 { p.deliver_late(e) } -> std::same_as<owned_delivery>;
84 p.provider_fails(code, r);
85 };

87 template <typename Probe>
88 bool kit_supports(Probe& p, payment_method m)
89 {
90 const auto caps = p.provider().capabilities();
91 return std::find(caps.methods.begin(), caps.methods.end(), m) != caps.methods.end();
92 }

94 template <typename Probe>
95 payment_method kit_method(Probe& p)

97 return kit_supports(p, payment_method::card) ? payment_method::card : payment_method::bank_transfer;

297 detail::capability_check<Probe>(report, "a statement pages through every line once", [](Probe& p, const auto& expect) {
298 auto& pp = p.provider();
299 const payment_method m = kit_method(p);
300 for (const std::string_view key : {"k-s1", "k-s2", "k-s3"}) {
301 const auto c = pp.collect(kit_request(p, key, m));
302 expect(c.is_ok(), "a collection starts");
303 if (c.is_ok())
304 p.payer_pays(*c);
305 }
306 std::size_t others = 0;
307 if constexpr (requires { p.add_other_lines(); })
308 others = p.add_other_lines();
309 std::size_t not_payments = 0;
310 std::vector<std::string> seen;
311 statement_query q{p.amount().asset_of(), "", "", 2};
312 std::string cursor;
313 for (int guard = 0; guard < 50; ++guard) {
314 const auto page = pp.statement(q);
315 expect(page.is_ok(), "every page answers");
316 if (!page.is_ok())
317 return;
318 for (const payment_fact& f : page->facts) {
319 expect(validate(f).is_ok(), "every fact validates");
320 seen.emplace_back(f.fact_id.view());
321 if (f.kind != fact_kind::payment && f.kind != fact_kind::bank_credit)
322 ++not_payments;
323 }
324 if (page->next_cursor.empty())
325 break;
326 cursor = std::string(page->next_cursor.view());
327 q.cursor = cursor;
328 }
329 expect(seen.size() >= 3 + others, "every line is a fact");
330 expect(not_payments >= others, "lines that are not payments are facts too, never skipped");
331 std::sort(seen.begin(), seen.end());
332 expect(std::adjacent_find(seen.begin(), seen.end()) == seen.end(), "no line twice");
333 });
