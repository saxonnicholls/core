77 // Indices, not pointers: GCC cannot compare a pointer into a constexpr table with nullptr in a constant
78 // expression. A binary search over the rows in the order `at` gives (at(i) is the row at position i);
79 // iso3166_rows.size() when the key is not there.
80 template <typename Key, typename At, typename Get>
81 constexpr std::size_t iso3166_search(Key key, At at, Get get) noexcept
82 {
83 std::size_t lo = 0, hi = iso3166_rows.size();
84 while (lo < hi) {
85 const std::size_t mid = lo + (hi - lo) / 2;
86 const Key k = get(iso3166_rows[at(mid)]);
87 if (k == key)
88 return at(mid);
89 if (k < key)
90 lo = mid + 1;
91 else
92 hi = mid;
93 }
94 return iso3166_rows.size();
95 }

97 constexpr std::size_t iso3166_index(std::string_view code) noexcept
98 {
99 if (code.size() == 2)
100 return iso3166_search(code, [](std::size_t i) { return i; }, [](const iso3166_row& r) { return r.alpha2; });
101 if (code.size() == 3)
102 return iso3166_search(code, [](std::size_t i) -> std::size_t { return iso3166_by_alpha3[i]; },
103 [](const iso3166_row& r) { return r.alpha3; });
104 return iso3166_rows.size();
105 }
