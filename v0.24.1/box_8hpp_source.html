196 template <typename T, std::size_t D>
197 [[nodiscard]] constexpr auto volume(const box<T, D>& b) noexcept
198 {
199 using R = std::conditional_t<std::is_floating_point_v<T>, T, double>;
200 if (empty(b))
201 return R(0);
202 R v(1);
203 for (std::size_t i = 0; i < D; ++i)
204 v *= static_cast<R>(b.hi[i]) - static_cast<R>(b.lo[i]);
205 return v;

322 template <typename T, std::size_t D, metric::point P>
323 [[nodiscard]] constexpr box<T, D> bounding(std::span<const P> points) noexcept(noexcept(metric::coordinate(std::declval<const P&>(), 0)))
324 {
325 box<T, D> r{};
326 if (points.empty()) {
327 for (std::size_t i = 0; i < D; ++i) {
328 r.lo[i] = T(1);
329 r.hi[i] = T(0);
330 }
331 return r;
332 }
333 for (std::size_t i = 0; i < D; ++i)
334 r.lo[i] = r.hi[i] = static_cast<T>(metric::coordinate(points[0], i));
335 for (const auto& p : points)
336 for (std::size_t i = 0; i < D; ++i) {
337 const T x = static_cast<T>(metric::coordinate(p, i));
338 r.lo[i] = x < r.lo[i] ? x : r.lo[i];
339 r.hi[i] = x > r.hi[i] ? x : r.hi[i];
340 }
341 return r;
342 }

353 [[nodiscard]] constexpr double squared_distance(const box<T, D>& b, const P& p) noexcept(noexcept(metric::coordinate(p, 0)))
354 {
355 double s = 0;
356 for (std::size_t i = 0; i < D; ++i) {
357 const double x = static_cast<double>(metric::coordinate(p, i));

359 s += d * d;

368 template <typename T, std::size_t D>
369 [[nodiscard]] constexpr double iou(const box<T, D>& a, const box<T, D>& b) noexcept
370 {
371 const double i = static_cast<double>(volume(intersection(a, b)));
372 const double u = static_cast<double>(volume(a)) + static_cast<double>(volume(b)) - i;
373 return u > 0 ? i / u : 0.0;
374 }

381 template <typename T, std::size_t D>
382 [[nodiscard]] constexpr double giou(const box<T, D>& a, const box<T, D>& b) noexcept
383 {
384 if (empty(a) || empty(b))
385 return 0.0;
386 const double i = static_cast<double>(volume(intersection(a, b)));
387 const double u = static_cast<double>(volume(a)) + static_cast<double>(volume(b)) - i;
388 const double c = static_cast<double>(volume(hull(a, b)));
389 if (!(c > 0))
390 return u > 0 ? i / u : 1.0; // degenerate (zero-volume) boxes: equal points are a perfect match
391 // (c - u) / c is never negative in exact arithmetic; rounding of u can make it so, which would put GIoU
392 // above IoU, so it is clamped
393 const double penalty = (c - u) / c;
394 return (u > 0 ? i / u : 0.0) - (penalty > 0 ? penalty : 0.0);
395 }

409 template <typename T, std::size_t D, typename S>
410 [[nodiscard]] constexpr std::size_t non_max_suppression(std::span<const box<T, D>> boxes, std::span<const S> scores, double threshold, std::span<std::uint32_t> kept) noexcept
411 {
412 const std::size_t n = boxes.size();
413 if (scores.size() != n || kept.size() < n)
414 return 0;
415 for (std::size_t i = 0; i < n; ++i)
416 kept[i] = static_cast<std::uint32_t>(i);
417 // Descending score, ties by index so the result is deterministic
418 std::sort(kept.begin(), kept.begin() + static_cast<std::ptrdiff_t>(n), [&](std::uint32_t x, std::uint32_t y) {
419 return scores[x] > scores[y] || (!(scores[y] > scores[x]) && x < y);
420 });
421 std::size_t k = 0;
422 for (std::size_t c = 0; c < n; ++c) {
423 const std::uint32_t candidate = kept[c];
424 bool keep = true;
425 for (std::size_t j = 0; j < k && keep; ++j)
426 keep = !(iou(boxes[candidate], boxes[kept[j]]) > threshold);
427 if (keep)
428 kept[k++] = candidate;
429 }
430 return k;
431 }

439 struct geo_box {
440 T south{};
441 T west{};

447 [[nodiscard]] static constexpr geo_box from_degrees(T south, T west, T north, T east) noexcept
448 {
449 const T k = static_cast<T>(std::numbers::pi_v<long double> / 180.0L);
450 return {south * k, west * k, north * k, east * k};
451 }
452
453 [[nodiscard]] constexpr bool crosses_antimeridian() const noexcept { return west > east; }
455 [[nodiscard]] constexpr T longitude_span() const noexcept
456 {
457 const T two_pi = T(2) * static_cast<T>(std::numbers::pi_v<long double>);
458 return crosses_antimeridian() ? east - west + two_pi : east - west;
459 }
460
462 [[nodiscard]] constexpr bool contains(T lat, T lon) const noexcept
463 {
464 if (!(lat >= south && lat <= north))
465 return false;
466 return crosses_antimeridian() ? (lon >= west || lon <= east) : (lon >= west && lon <= east);
467 }
468
469 [[nodiscard]] constexpr std::array<T, 2> centre() const noexcept
470 {
471 const T pi = static_cast<T>(std::numbers::pi_v<long double>);
472 T lon = west + longitude_span() / T(2);
473 if (lon > pi)
474 lon -= T(2) * pi;
475 return {(south + north) / T(2), lon};
476 }
477
479 [[nodiscard]] constexpr geo_box extend(T lat, T lon) const noexcept
480 {
481 geo_box r = *this;
482 r.south = lat < south ? lat : south;
483 r.north = lat > north ? lat : north;
484 const bool lon_inside = crosses_antimeridian() ? (lon >= west || lon <= east) : (lon >= west && lon <= east);
485 if (lon_inside)
486 return r;
487 const T two_pi = T(2) * static_cast<T>(std::numbers::pi_v<long double>);
488 // Growing east to lon, or west to lon: whichever adds less span
489 T grow_east = lon - east;
490 if (grow_east < T(0))
491 grow_east += two_pi;
492 T grow_west = west - lon;
493 if (grow_west < T(0))
494 grow_west += two_pi;
495 if (grow_east <= grow_west)
496 r.east = lon;
497 else
498 r.west = lon;
499 return r;
500 }

504 template <stats::full_range_generator G>
505 [[nodiscard]] constexpr std::array<T, 2> sample(G& g) const noexcept(noexcept(g()))
506 {
507 const T pi = static_cast<T>(std::numbers::pi_v<long double>);
508 const T s0 = static_cast<T>(math::sin(south)), s1 = static_cast<T>(math::sin(north));
509 const T lat = static_cast<T>(math::asin(s0 + (s1 - s0) * static_cast<T>(stats::uniform_real(g))));
510 T lon = west + longitude_span() * static_cast<T>(stats::uniform_real(g));
511 if (lon > pi)
512 lon -= T(2) * pi;
513 return {lat, lon};
514 }
515
517 friend constexpr bool operator==(const geo_box&, const geo_box&) = default;
518 };
