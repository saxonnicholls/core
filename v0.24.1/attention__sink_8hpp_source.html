82 // One query over n keys: score(j) is the scaled score of key j, value(j) its value (out.size() channels). The
83 // sink, unless -infinity, is folded last as a key whose value is zero. Writes out, returns logsumexp.
84 template <typename T, typename Score, typename Value>
85 constexpr T forward(std::size_t n, Score&& score, Value&& value, std::span<T> out, T sink)
86 {
87 constexpr T none = -std::numeric_limits<T>::infinity();
88 std::fill(out.begin(), out.end(), T(0));
89 T max = none, sum = T(0);
90 for (std::size_t j = 0; j < n; ++j)
91 detail::attention::fold(max, sum, out, score(j), value(j));
92 if (sink != none) {
93 if (sink > max) {
94 const T s = math::exp(max - sink);
95 for (T& a : out)
96 a = a * s;
97 sum = sum * s + T(1);
98 max = sink;
99 } else {
100 sum += math::exp(sink - max);
101 }
102 }
103 for (T& v : out)
104 v = sum == T(0) ? T(0) : v / sum;
105 return sum == T(0) ? none : max + math::log(sum);
106 }

108 // The backward pass from the saved lse: grad(j, p, ds) is called for each key with its weight p and the gradient
109 // ds in its scaled score. Returns the gradient in the sink.
110 template <typename T, typename Score, typename Value, typename Grad>
111 constexpr T backward(std::size_t n, Score&& score, Value&& value, std::span<const T> out, std::span<const T> dout, T lse,
112 T sink, Grad&& grad)
113 {
114 constexpr T none = -std::numeric_limits<T>::infinity();
115 if (lse == none)
116 return T(0);
117 const T big_d = detail::attention::dot(dout, out);
118 for (std::size_t j = 0; j < n; ++j) {
119 const T p = math::exp(score(j) - lse);
120 const T ds = p * (detail::attention::dot(dout, value(j)) - big_d);
121 grad(j, p, ds);
122 }
123 return sink == none ? T(0) : -math::exp(sink - lse) * big_d;
124 }

132constexpr T attention_sink_weight(T lse, T sink)

134 return sink == -std::numeric_limits<T>::infinity() ? T(0) : math::exp(sink - lse);

145template <row Q, row K, row V, mutable_row Out>
146 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
147 && std::same_as<row_value_t<Q>, row_value_t<Out>>
148constexpr row_value_t<Q> attend_with_sink(const Q& q, const K& keys, const V& values, Out&& out,
149 std::type_identity_t<row_value_t<Q>> scale,
150 std::type_identity_t<row_value_t<Q>> sink = -std::numeric_limits<row_value_t<Q>>::infinity())
151{
152 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values);
153 const auto y = as_mutable_row(out);
154 const std::size_t dk = qs.size(), dv = y.size(), n = dk == 0 ? 0 : ks.size() / dk;
155 assert(dk > 0 && ks.size() == n * dk && vs.size() == n * dv);
156 return detail::attention_sink::forward(
157 n, [&](std::size_t j) { return scale * detail::attention::dot(qs, ks.subspan(j * dk, dk)); },
158 [&](std::size_t j) { return vs.subspan(j * dv, dv); }, y, sink);
159}

170template <row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK, mutable_row DV>
171 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
172 && std::same_as<row_value_t<Q>, row_value_t<O>> && std::same_as<row_value_t<Q>, row_value_t<DO>>
173 && std::same_as<row_value_t<Q>, row_value_t<DQ>> && std::same_as<row_value_t<Q>, row_value_t<DK>>
174 && std::same_as<row_value_t<Q>, row_value_t<DV>>
175constexpr row_value_t<Q> attend_with_sink_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
176 std::type_identity_t<row_value_t<Q>> lse,
177 std::type_identity_t<row_value_t<Q>> scale,
178 std::type_identity_t<row_value_t<Q>> sink, DQ&& dq, DK&& dkeys, DV&& dvalues)
179{
180 using T = row_value_t<Q>;
181 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values), o = as_row(out), g = as_row(dout);
182 const auto gq = as_mutable_row(dq), gk = as_mutable_row(dkeys), gv = as_mutable_row(dvalues);
183 const std::size_t dk = qs.size(), dv = o.size(), n = dk == 0 ? 0 : ks.size() / dk;
184 assert(dk > 0 && ks.size() == n * dk && vs.size() == n * dv && g.size() == dv && gq.size() == dk && gk.size() == ks.size()
185 && gv.size() == vs.size());
186 std::fill(gq.begin(), gq.end(), T(0));
187 return detail::attention_sink::backward(
188 n, [&](std::size_t j) { return scale * detail::attention::dot(qs, ks.subspan(j * dk, dk)); },
189 [&](std::size_t j) { return vs.subspan(j * dv, dv); }, o, g, lse, sink,
190 [&](std::size_t j, T p, T ds) {
191 for (std::size_t i = 0; i < dv; ++i)
192 gv[j * dv + i] += p * g[i];
193 for (std::size_t i = 0; i < dk; ++i) {
194 gq[i] += scale * ds * ks[j * dk + i];
195 gk[j * dk + i] += scale * ds * qs[i];
196 }
197 });
198}
