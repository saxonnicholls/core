66 struct weighted_gaussian {
67 T weight{};
68 gaussian<T, N> state{};
69 };

77 template <linalg::scalar T, std::size_t N>
78 [[nodiscard]] constexpr gaussian<T, N> reduce_to_single(std::span<const weighted_gaussian<T, N>> components) noexcept(linalg::detail::nothrow_scalar<T>)
79 {
80 T total(0);
81 for (const auto& c : components)
82 total += c.weight;
83 gaussian<T, N> g{};
84 if (!(total > T(0)) && !(total < T(0)))
85 return g;
86 for (const auto& c : components)
87 g.mean += c.state.mean * (c.weight / total);
88 for (const auto& c : components) {
89 const auto d = c.state.mean - g.mean;
90 g.covariance += (c.state.covariance + linalg::outer(d, d)) * (c.weight / total);
91 }
92 g.covariance = linalg::symmetrise(g.covariance);
93 return g;
94 }

100 template <linalg::scalar T, std::size_t N>
101 [[nodiscard]] constexpr weighted_gaussian<T, N> merge_pair(const weighted_gaussian<T, N>& a, const weighted_gaussian<T, N>& b) noexcept(linalg::detail::nothrow_scalar<T>)
102 {
103 const T w = a.weight + b.weight;
104 const T wa = a.weight / w, wb = b.weight / w;
105 const auto d = a.state.mean - b.state.mean;
106 weighted_gaussian<T, N> m{};
107 m.state.mean = a.state.mean * wa + b.state.mean * wb;
108 m.state.covariance = linalg::symmetrise(a.state.covariance * wa + b.state.covariance * wb + linalg::outer(d, d) * (wa * wb));
109 m.weight = w > T(1) ? T(1) : w;
110 return m;
111 }

117 template <linalg::scalar T, std::size_t N>
118 [[nodiscard]] constexpr result<T, linalg::error> runnalls_cost(const weighted_gaussian<T, N>& a, const weighted_gaussian<T, N>& b) noexcept(linalg::detail::nothrow_scalar<T>)
119 {
120 auto m = merge_pair(a, b);
121 m.weight = a.weight + b.weight;
122 const auto lm = linalg::cholesky(m.state.covariance);
123 const auto la = linalg::cholesky(a.state.covariance);
124 const auto lb = linalg::cholesky(b.state.covariance);
125 if (!lm || !la || !lb)
126 return err(linalg::error::not_positive_definite);
127 return ok((m.weight * linalg::cholesky_log_det(*lm) - a.weight * linalg::cholesky_log_det(*la) - b.weight * linalg::cholesky_log_det(*lb)) / T(2));
128 }

147 template <linalg::scalar T, std::size_t N>
148 constexpr std::size_t reduce_mixture(std::span<weighted_gaussian<T, N>> components, const mixture_reduction<T>& options = {}) noexcept(linalg::detail::nothrow_scalar<T>)
149 {
150 std::size_t n = components.size();
151 if (n == 0)
152 return 0;
153 // Prune, sharing the pruned weight equally among the rest
154 T pruned(0);
155 std::size_t kept = 0;
156 for (std::size_t i = 0; i < n; ++i) {
157 if (components[i].weight < options.prune_threshold)
158 pruned += components[i].weight;
159 else
160 components[kept++] = components[i];
161 }
162 n = kept;
163 for (std::size_t i = 0; i < n; ++i)
164 components[i].weight += pruned / static_cast<T>(n);
165 // Heaviest first (stable insertion sort: equal weights keep their order, as Stone Soup's sort does)
166 for (std::size_t i = 1; i < n; ++i)
167 for (std::size_t j = i; j > 0 && components[j].weight > components[j - 1].weight; --j) {
168 const auto t = components[j];
169 components[j] = components[j - 1];
170 components[j - 1] = t;
171 }
172 if (n > 1 && options.merge_threshold > T(0)) {
173 // components[0, done) are merged results; [done, n) remain, heaviest first. Stone Soup pops the
174 // heaviest and then tests the rest from the lightest up; the order matters because the merged
175 // component moves as it absorbs.
176 std::size_t done = 0;
177 while (done < n) {
178 auto best = components[done];
179 std::size_t rest = done + 1;
180 std::size_t write = n; // compact survivors toward the end, preserving their order
181 for (std::size_t i = n; i-- > rest;) {
182 const auto& c = components[i];
183 const auto d2 = squared_mahalanobis(best.state.mean - c.state.mean, c.state.covariance);
184 if (d2 && *d2 < options.merge_threshold)
185 best = merge_pair(best, c);
186 else
187 components[--write] = components[i];
188 }
189 components[done] = best;
190 // The survivors sit in [write, n); move them down to follow the merged component
191 std::size_t to = done + 1;
192 for (std::size_t i = write; i < n; ++i)
193 components[to++] = components[i];
194 n = to;
195 ++done;
196 }
197 // Heaviest first again
198 for (std::size_t i = 1; i < n; ++i)
199 for (std::size_t j = i; j > 0 && components[j].weight > components[j - 1].weight; --j) {
200 const auto t = components[j];
201 components[j] = components[j - 1];
202 components[j - 1] = t;
203 }
204 }
205 return n < options.max_components ? n : options.max_components;
206 }
