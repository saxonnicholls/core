89[[nodiscard]] constexpr std::size_t recurrence_checkpoints(std::size_t n, std::size_t chunk) noexcept
90{
91 return chunk == 0 ? 0 : (n + chunk - 1) / chunk;

111template <recurrence R>
112constexpr void recurrence_forward(const R& r, std::size_t n, std::size_t sequences, std::size_t chunk,
113 std::span<typename R::value_type> states, std::span<typename R::value_type> checkpoints = {})
114{
115 constexpr std::size_t S = R::state_size;
116 const std::size_t m = recurrence_checkpoints(n, chunk);
117 assert(chunk > 0 && states.size() == sequences * S && (checkpoints.empty() || checkpoints.size() == sequences * m * S));
118 for (std::size_t s = 0; s < sequences; ++s) {
119 const auto state = states.subspan(s * S, S);
120 for (std::size_t i = 0; i < n; ++i) {
121 if (!checkpoints.empty() && i % chunk == 0)
122 std::copy(state.begin(), state.end(), checkpoints.begin() + static_cast<std::ptrdiff_t>((s * m + i / chunk) * S));
123 r.step(s * n + i, state);
124 }
125 }
126}

136template <recurrence R>
137constexpr void recurrence_backward(const R& r, std::size_t n, std::size_t sequences, std::size_t chunk,
138 std::span<const typename R::value_type> checkpoints, std::span<typename R::value_type> dstates,
139 std::span<typename R::value_type> work)
140{
141 constexpr std::size_t S = R::state_size;
142 const std::size_t m = recurrence_checkpoints(n, chunk);
143 assert(chunk > 0 && checkpoints.size() == sequences * m * S && dstates.size() == sequences * S
144 && work.size() == recurrence_backward_work<R>(chunk));
145 const auto saved = work.first(chunk * S);
146 const auto scratch = work.subspan(chunk * S);
147 for (std::size_t s = 0; s < sequences; ++s) {
148 const auto dstate = dstates.subspan(s * S, S);
149 for (std::size_t c = m; c-- > 0;) {
150 const std::size_t first = c * chunk, len = std::min(chunk, n - first);
151 const auto from = checkpoints.subspan((s * m + c) * S, S);
152 std::copy(from.begin(), from.end(), saved.begin());
153 for (std::size_t j = 1; j < len; ++j) {
154 const auto prev = saved.subspan((j - 1) * S, S), next = saved.subspan(j * S, S);
155 std::copy(prev.begin(), prev.end(), next.begin());
156 r.advance(s * n + first + j - 1, next);
157 }
158 for (std::size_t j = len; j-- > 0;)
159 r.step_backward(s * n + first + j, std::span<const typename R::value_type>(saved.subspan(j * S, S)), dstate, scratch);
160 }
161 }
