129template <typename B, math::math_number T>
130constexpr void l2_normalise_rows(const B&, std::type_identity_t<std::span<const T>> x, std::span<T> out,
131 std::size_t width, std::type_identity_t<T> eps)
132{
133 assert(width > 0 && x.size() % width == 0 && out.size() == x.size());
134 for (std::size_t r = 0; r < x.size() / width; ++r) {
135 T ss = T(0);
136 for (std::size_t i = 0; i < width; ++i)
137 ss += x[r * width + i] * x[r * width + i];
138 const T inv = T(1) / math::sqrt(ss + eps);
139 for (std::size_t i = 0; i < width; ++i)
140 out[r * width + i] = x[r * width + i] * inv;
141 }
142}

147template <typename B, math::math_number T>
148constexpr void l2_normalise_rows_backward(const B&, std::type_identity_t<std::span<const T>> x, std::type_identity_t<std::span<const T>> dy,
149 std::span<T> dx, std::size_t width, std::type_identity_t<T> eps)
150{
151 assert(width > 0 && x.size() % width == 0 && dy.size() == x.size() && dx.size() == x.size());
152 for (std::size_t r = 0; r < x.size() / width; ++r) {
153 T ss = T(0), yd = T(0);
154 for (std::size_t i = 0; i < width; ++i)
155 ss += x[r * width + i] * x[r * width + i];
156 const T inv = T(1) / math::sqrt(ss + eps);
157 for (std::size_t i = 0; i < width; ++i)
158 yd += x[r * width + i] * inv * dy[r * width + i];
159 for (std::size_t i = 0; i < width; ++i)
160 dx[r * width + i] = (dy[r * width + i] - x[r * width + i] * inv * yd) * inv;
161 }

166 template <transformer_config C, typename T>
167 struct scan
168 {
169 using value_type = T;
170 static constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups;
171 static constexpr std::size_t state_size = H * N * P, work_size = N * P;
172 static constexpr T scale = mixer_scale_v<C, T>;
173
174 delta_rule_inputs<T> in{};
175 std::span<T> o{};
176 std::span<const T> dout{};
177 delta_rule_gradients<T> g{};
178 bool per_channel = false;
179
180 constexpr T gate(std::size_t row, std::size_t h, std::size_t n) const
181 {
182 return per_channel ? in.g[(row * H + h) * N + n] : in.g[row * H + h];
183 }
184
185 constexpr void update(std::size_t row, std::span<T> state, bool output) const
186 {
187 for (std::size_t h = 0; h < H; ++h) {
188 const std::size_t grp = h / (H / G);
189 const auto q = in.q.subspan((row * G + grp) * N, N), k = in.k.subspan((row * G + grp) * N, N);
190 const auto v = in.v.subspan((row * H + h) * P, P);
191 const T beta = in.beta[row * H + h];
192 const auto s = state.subspan(h * N * P, N * P);
193 std::array<T, P> delta{};
194 for (std::size_t n = 0; n < N; ++n) {
195 const T a = math::exp(gate(row, h, n));
196 for (std::size_t p = 0; p < P; ++p)
197 s[n * P + p] *= a;
198 }
199 for (std::size_t n = 0; n < N; ++n)
200 for (std::size_t p = 0; p < P; ++p)
201 delta[p] += s[n * P + p] * k[n];
202 for (std::size_t p = 0; p < P; ++p)
203 delta[p] = (v[p] - delta[p]) * beta;
204 for (std::size_t n = 0; n < N; ++n)
205 for (std::size_t p = 0; p < P; ++p)
206 s[n * P + p] += k[n] * delta[p];
207 if (output) {
208 std::array<T, P> out{};
209 for (std::size_t n = 0; n < N; ++n)
210 for (std::size_t p = 0; p < P; ++p)
211 out[p] += s[n * P + p] * q[n];
212 for (std::size_t p = 0; p < P; ++p)
213 o[(row * H + h) * P + p] = out[p] * scale;
214 }
215 }
216 }
217 constexpr void step(std::size_t row, std::span<T> state) const { update(row, state, true); }
218 constexpr void advance(std::size_t row, std::span<T> state) const { update(row, state, false); }

220 constexpr void step_backward(std::size_t row, std::span<const T> prev, std::span<T> dstate, std::span<T> work) const
221 {
222 for (std::size_t i = 0; i < G * N; ++i)
223 g.q[row * G * N + i] = g.k[row * G * N + i] = T(0);
224 for (std::size_t h = 0; h < H; ++h) {
225 const std::size_t grp = h / (H / G);
226 const auto q = in.q.subspan((row * G + grp) * N, N), k = in.k.subspan((row * G + grp) * N, N);
227 const auto v = in.v.subspan((row * H + h) * P, P);
228 const auto d = dout.subspan((row * H + h) * P, P);
229 const T beta = in.beta[row * H + h];
230 const auto s0 = prev.subspan(h * N * P, N * P);
231 const auto ds = dstate.subspan(h * N * P, N * P);
232 const auto sd = work.first(N * P); // the decayed state
233 const auto dq = g.q.subspan((row * G + grp) * N, N), dk = g.k.subspan((row * G + grp) * N, N);
234 std::array<T, P> u{}, delta{}, ddelta{}, du{};
235 for (std::size_t n = 0; n < N; ++n) {
236 const T a = math::exp(gate(row, h, n));
237 for (std::size_t p = 0; p < P; ++p) {
238 sd[n * P + p] = s0[n * P + p] * a;
239 u[p] += sd[n * P + p] * k[n];
240 }
241 }
242 for (std::size_t p = 0; p < P; ++p)
243 delta[p] = (v[p] - u[p]) * beta;
244 for (std::size_t n = 0; n < N; ++n) { // o = scale · Sᵀ q with S = sd + k δᵀ
245 T dqn = T(0), dkn = T(0);
246 for (std::size_t p = 0; p < P; ++p) {
247 dqn += d[p] * (sd[n * P + p] + k[n] * delta[p]);
248 ds[n * P + p] += q[n] * scale * d[p];
249 dkn += ds[n * P + p] * delta[p];
250 ddelta[p] += ds[n * P + p] * k[n];
251 }
252 dq[n] += dqn * scale;
253 dk[n] += dkn;
254 }
255 T dbeta = T(0);
256 for (std::size_t p = 0; p < P; ++p) {
257 dbeta += ddelta[p] * (v[p] - u[p]);
258 g.v[(row * H + h) * P + p] = ddelta[p] * beta;
259 du[p] = -ddelta[p] * beta;
260 }
261 g.beta[row * H + h] = dbeta;
262 T dg = T(0);
263 for (std::size_t n = 0; n < N; ++n) { // u = sdᵀ k, sd = e^g s0
264 const T a = math::exp(gate(row, h, n));
265 T dkn = T(0), dgn = T(0);
266 for (std::size_t p = 0; p < P; ++p) {
267 dkn += sd[n * P + p] * du[p];
268 const T dsd = ds[n * P + p] + k[n] * du[p];
269 dgn += dsd * sd[n * P + p];
270 ds[n * P + p] = dsd * a;
271 }
272 dk[n] += dkn;
273 if (per_channel)
274 g.g[(row * H + h) * N + n] = dgn;
275 else
276 dg += dgn;
277 }
278 if (!per_channel)
279 g.g[row * H + h] = dg;
280 }
281 }
282 };

284 template <transformer_config C, typename T>
285 constexpr std::size_t rows_of(const delta_rule_inputs<T>& in)
286 {
287 [[maybe_unused]] constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups;
288 static_assert(valid(C) && C.mixer_heads > 0, "snicholls::ml::delta_rule: the configuration has no mixer shape");
289 static_cast<void>(N), static_cast<void>(G); // read only by the assert (MSVC ignores [[maybe_unused]] there, C4189)
290 const std::size_t rows = in.v.size() / (H * P);
291 assert(in.v.size() == rows * H * P && in.q.size() == rows * G * N && in.k.size() == rows * G * N && in.beta.size() == rows * H
292 && (in.g.size() == rows * H || in.g.size() == rows * H * N));
293 return rows;
294 }

325template <transformer_config C, typename B, math::math_number T>
326 requires linalg::blas_backend<B, T>
327constexpr void delta_rule_forward_chunked(const B& backend, const delta_rule_inputs<T>& in, std::type_identity_t<std::span<T>> o,
328 std::size_t sequences, std::type_identity_t<std::span<T>> states,
329 std::type_identity_t<std::span<T>> checkpoints, std::type_identity_t<std::span<T>> work)
330{
331 constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups, L = C.mixer_chunk;
332 constexpr std::size_t S = H * N * P;
333 constexpr T scale = mixer_scale_v<C, T>;
334 const std::size_t rows = detail::delta_rule::rows_of<C>(in);
335 const auto sc = detail::delta_rule::make<C, T>(in, o, {}, {}, rows);
336 assert(sequences > 0 && rows % sequences == 0 && o.size() == in.v.size() && states.size() == sequences * S
337 && work.size() == delta_rule_chunk_work(C));
338 const std::size_t n = rows / sequences, m = recurrence_checkpoints(n, L);
339 assert(checkpoints.empty() || checkpoints.size() == sequences * m * S);
340 const auto cg = work.first(L * N), tri = work.subspan(L * N, L * L), uw = work.subspan(L * N + L * L, L * (P + N)),
341 dl = work.subspan(L * N + L * L + L * (P + N), L * P);
342 for (std::size_t sq = 0; sq < sequences; ++sq)
343 for (std::size_t ck = 0; ck < m; ++ck) {
344 const std::size_t first = sq * n + ck * L, len = n - ck * L < L ? n - ck * L : L;
345 const auto state = states.subspan(sq * S, S);
346 if (!checkpoints.empty())
347 for (std::size_t i = 0; i < S; ++i)
348 checkpoints[(sq * m + ck) * S + i] = state[i];
349 for (std::size_t h = 0; h < H; ++h) {
350 const std::size_t grp = h / (H / G);
351 const auto s = state.subspan(h * N * P, N * P);
352 const auto key = [&](std::size_t j) { return in.k.subspan(((first + j) * G + grp) * N, N); };
353 for (std::size_t j = 0; j < len; ++j)
354 for (std::size_t q = 0; q < N; ++q)
355 cg[j * N + q] = (j == 0 ? T(0) : cg[(j - 1) * N + q]) + sc.gate(first + j, h, q);
356 // (I + strictLower(diag(β) A)) [U | W] = diag(β) [V | e^G ⊙ K], unit lower triangular
357 for (std::size_t j = 0; j < len; ++j) {
358 const T beta = in.beta[(first + j) * H + h];
359 for (std::size_t i = 0; i < len; ++i) {
360 T a = T(0);
361 if (i < j)
362 for (std::size_t q = 0; q < N; ++q)
363 a += key(j)[q] * key(i)[q] * math::exp(cg[j * N + q] - cg[i * N + q]);
364 tri[j * len + i] = i < j ? beta * a : i == j ? T(1) : T(0);
365 }
366 for (std::size_t p = 0; p < P; ++p)
367 uw[j * (P + N) + p] = beta * in.v[((first + j) * H + h) * P + p];
368 for (std::size_t q = 0; q < N; ++q)
369 uw[j * (P + N) + P + q] = beta * key(j)[q] * math::exp(cg[j * N + q]);
370 }
371 [[maybe_unused]] const auto solved = linalg::triangular_solve(
372 backend, linalg::matrix_view<const T, linalg::layout_right>(tri.data(), {len, len}),
373 linalg::matrix_view<T, linalg::layout_right>(uw.data(), {len, P + N}), linalg::triangle::lower,
374 linalg::diagonal::implicit_unit);
375 assert(solved.is_ok());
376 for (std::size_t j = 0; j < len; ++j) // Δ = U - W S_0
377 for (std::size_t p = 0; p < P; ++p) {
378 T ws = T(0);
379 for (std::size_t q = 0; q < N; ++q)
380 ws += uw[j * (P + N) + P + q] * s[q * P + p];
381 dl[j * P + p] = uw[j * (P + N) + p] - ws;
382 }
383 for (std::size_t j = 0; j < len; ++j) {
384 const auto qj = in.q.subspan(((first + j) * G + grp) * N, N);
385 const auto oj = o.subspan(((first + j) * H + h) * P, P);
386 for (std::size_t p = 0; p < P; ++p)
387 oj[p] = T(0);
388 for (std::size_t q = 0; q < N; ++q) {
389 const T qd = qj[q] * math::exp(cg[j * N + q]);
390 for (std::size_t p = 0; p < P; ++p)
391 oj[p] += qd * s[q * P + p];
392 }
393 for (std::size_t i = 0; i <= j; ++i) {
394 T a = T(0);
395 for (std::size_t q = 0; q < N; ++q)
396 a += qj[q] * key(i)[q] * math::exp(cg[j * N + q] - cg[i * N + q]);
397 for (std::size_t p = 0; p < P; ++p)
398 oj[p] += a * dl[i * P + p];
399 }
400 for (std::size_t p = 0; p < P; ++p)
401 oj[p] *= scale;
402 }
403 for (std::size_t q = 0; q < N; ++q) {
404 const T last = cg[(len - 1) * N + q], total = math::exp(last);
405 for (std::size_t p = 0; p < P; ++p) {
406 T add = T(0);
407 for (std::size_t i = 0; i < len; ++i)
408 add += math::exp(last - cg[i * N + q]) * key(i)[q] * dl[i * P + p];
409 s[q * P + p] = total * s[q * P + p] + add;
410 }
411 }
412 }
413 }
