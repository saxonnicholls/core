211 // The widths of a configuration's layer
212 struct shape
213 {
214 std::size_t d, H, P, N, G, K, inner, keys, conv, in, S;
215 };
216 constexpr shape shape_of(const transformer_config& c) noexcept
217 {
218 const std::size_t inner = c.mixer_heads * c.mixer_head_dim, keys = c.mixer_groups * c.mixer_state;
219 const bool mamba = c.mixer == mixer_kind::mamba2;
220 const std::size_t conv = mamba ? inner + 2 * keys : 2 * keys + inner;
221 const std::size_t in = mamba ? 2 * inner + 2 * keys + c.mixer_heads : 2 * keys + 2 * inner + 2 * c.mixer_heads;
222 return {c.d_model, c.mixer_heads, c.mixer_head_dim, c.mixer_state, c.mixer_groups, c.conv_width, inner, keys, conv, in, mixer_state_size(c)};
223 }
224 constexpr bool layer_kind(const transformer_config& c) noexcept
225 {
226 return c.mixer == mixer_kind::mamba2 || c.mixer == mixer_kind::gated_deltanet;
227 }

229 // to[r · tw + to_at + i] = from[r · fw + from_at + i] for i < width
230 template <typename T>
231 constexpr void columns(std::span<const T> from, std::size_t fw, std::size_t from_at, std::span<T> to, std::size_t tw, std::size_t to_at,
232 std::size_t width, std::size_t rows)
233 {
234 for (std::size_t r = 0; r < rows; ++r)
235 for (std::size_t i = 0; i < width; ++i)
236 to[r * tw + to_at + i] = from[r * fw + from_at + i];
237 }

251[[nodiscard]] constexpr std::size_t mixer_parameter_count(const transformer_config& c) noexcept
252{
253 const auto s = detail::mixer_layer::shape_of(c);
254 const bool mamba = c.mixer == mixer_kind::mamba2;
255 return s.in * s.d + s.conv * s.K + (mamba ? s.conv : 0) + 2 * s.H + (mamba ? s.H : 0) + (mamba ? s.inner : s.P) + s.d * s.inner;
256}

262 const auto s = detail::mixer_layer::shape_of(c);
263 return short_conv_state_size(s.conv, s.K) + s.S;

269[[nodiscard]] constexpr std::size_t mixer_activation_values(const transformer_config& c, std::size_t rows, std::size_t sequences) noexcept
270{
271 const auto s = detail::mixer_layer::shape_of(c);
272 const bool mamba = c.mixer == mixer_kind::mamba2;
273 const std::size_t m = sequences == 0 ? 0 : recurrence_checkpoints(rows / sequences, c.mixer_chunk);
274 return rows * (s.in + 2 * s.conv + 2 * s.keys + s.inner + s.H + (mamba ? 0 : s.H) + s.inner + (mamba ? s.inner : 0) + (mamba ? s.G : s.H)
275 + s.inner)
276 + sequences * m * s.S + sequences * s.S + (mamba ? 0 : delta_rule_chunk_work(c));
277}

282template <transformer_config C, math::math_number T>
283[[nodiscard]] constexpr std::size_t mixer_gradient_work(std::size_t rows, std::size_t sequences) noexcept
284{
285 const auto s = detail::mixer_layer::shape_of(C);
286 const std::size_t op = C.mixer == mixer_kind::mamba2 ? ssd_backward_work<C, T>() : delta_rule_backward_work<C, T>();
287 return rows * (s.in + 3 * s.inner + 2 * s.keys + 2 * s.H + 2 * s.keys + 2 * s.conv) + s.H + sequences * s.S + op;
288}

292template <transformer_config C, math::math_number T>
293 requires(C.mixer == mixer_kind::mamba2)
294constexpr mamba2_weights<T> mixer_weights_in(std::span<const T> params)
295{
296 constexpr auto s = detail::mixer_layer::shape_of(C);
297 assert(params.size() == mixer_parameter_count(C));
298 detail::cutter<const T> cut{params};
299 mamba2_weights<T> w{};
300 w.in_proj = {cut.take(s.in * s.d), s.in, s.d};
301 w.conv = {cut.take(s.conv * s.K), s.conv, s.K, {}};
302 w.conv.bias = cut.take(s.conv);
303 w.dt_bias = cut.take(s.H);
304 w.a_log = cut.take(s.H);
305 w.d_skip = cut.take(s.H);
306 w.norm = cut.take(s.inner);
307 w.out_proj = {cut.take(s.d * s.inner), s.d, s.inner};
308 return w;
309}

333template <transformer_config C, math::math_number T>
334 requires(detail::mixer_layer::layer_kind(C))
335constexpr mixer_activations<T> mixer_activations_in(std::span<T> values, std::size_t rows, std::size_t sequences)
336{
337 constexpr auto s = detail::mixer_layer::shape_of(C);
338 constexpr bool mamba = C.mixer == mixer_kind::mamba2;
339 assert(values.size() == mixer_activation_values(C, rows, sequences));
340 const std::size_t m = recurrence_checkpoints(rows / sequences, C.mixer_chunk);
341 detail::cutter<T> cut{values};
342 mixer_activations<T> a{};
343 a.proj = cut.take(rows * s.in);
344 a.conv_in = cut.take(rows * s.conv);
345 a.conv_out = cut.take(rows * s.conv);
346 a.q = cut.take(rows * s.keys);
347 a.k = cut.take(rows * s.keys);
348 a.v = cut.take(rows * s.inner);
349 a.gate = cut.take(rows * s.H);
350 a.beta = cut.take(mamba ? 0 : rows * s.H);
351 a.y = cut.take(rows * s.inner);
352 a.gated = cut.take(mamba ? rows * s.inner : 0);
353 a.rstd = cut.take(rows * (mamba ? s.G : s.H));
354 a.normed = cut.take(rows * s.inner);
355 a.checkpoints = cut.take(sequences * m * s.S);
356 a.final_states = cut.take(sequences * s.S);
357 a.work = cut.take(mamba ? 0 : delta_rule_chunk_work(C));
358 return a;
359}

364template <transformer_config C, math::math_number T>
365 requires(detail::mixer_layer::layer_kind(C))
366constexpr mixer_gradients<T> mixer_gradients_in(std::span<T> grads, std::span<T> work)
367{
368 constexpr auto s = detail::mixer_layer::shape_of(C);
369 constexpr bool mamba = C.mixer == mixer_kind::mamba2;
370 assert(grads.size() == mixer_parameter_count(C));
371 detail::cutter<T> cut{grads};
372 mixer_gradients<T> g{};
373 g.in_proj = cut.take(s.in * s.d);
374 g.conv = cut.take(s.conv * s.K);
375 g.conv_bias = cut.take(mamba ? s.conv : 0);
376 g.dt_bias = cut.take(s.H);
377 g.a_log = cut.take(s.H);
378 g.d_skip = cut.take(mamba ? s.H : 0);
379 g.norm = cut.take(mamba ? s.inner : s.P);
380 g.out_proj = cut.take(s.d * s.inner);
381 g.work = work;
382 return g;
383}

389 // The backward pass's working space, cut once
390 template <typename T>
391 struct backward_work
392 {
393 std::span<T> dproj, dnormed, dgated, dv, dq, dk, dgate, dbeta, pre, dconv_out, dconv_in, da, dstates, op;
394 };
395 template <transformer_config C, typename T>
396 constexpr backward_work<T> backward_work_in(std::span<T> work, std::size_t rows, std::size_t sequences)
397 {
398 constexpr auto s = shape_of(C);
399 assert((work.size() == mixer_gradient_work<C, T>(rows, sequences)));
400 cutter<T> cut{work};
401 backward_work<T> b{};
402 b.dproj = cut.take(rows * s.in);
403 b.dnormed = cut.take(rows * s.inner);
404 b.dgated = cut.take(rows * s.inner);
405 b.dv = cut.take(rows * s.inner);
406 b.dq = cut.take(rows * s.keys);
407 b.dk = cut.take(rows * s.keys);
408 b.dgate = cut.take(rows * s.H);
409 b.dbeta = cut.take(rows * s.H);
410 b.pre = cut.take(rows * 2 * s.keys);
411 b.dconv_out = cut.take(rows * s.conv);
412 b.dconv_in = cut.take(rows * s.conv);
413 b.da = cut.take(s.H);
414 b.dstates = cut.take(sequences * s.S);
415 b.op = cut.rest;
416 return b;
417 }

430template <transformer_config C, typename B, math::math_number T>
431 requires(C.mixer == mixer_kind::mamba2 && linalg::blas_backend<B, T>)
432constexpr void mixer_forward_sequence(const B& backend, const mamba2_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
433 const mixer_activations<T>& act, std::type_identity_t<std::span<T>> out, std::size_t sequences = 1,
434 std::type_identity_t<std::span<T>> states = {})
435{
436 using namespace detail::mixer_layer;
437 constexpr auto s = shape_of(C);
438 const std::size_t rows = xs.size() / s.d, sk = short_conv_state_size(s.conv, s.K);
439 assert(xs.size() == rows * s.d && out.size() == xs.size() && sequences > 0 && rows % sequences == 0
440 && (states.empty() || states.size() == sequences * (sk + s.S)));
441 linear_sequence(backend, w.in_proj, xs, act.proj);
442 columns<T>(in(act.proj), s.in, s.inner, act.conv_in, s.conv, 0, s.conv, rows);
443 short_conv_forward(backend, w.conv, in(act.conv_in), act.conv_out, sequences, states.empty() ? std::span<T>{} : states.first(sequences * sk));
444 for (std::size_t r = 0; r < rows; ++r) {
445 for (std::size_t i = 0; i < s.inner; ++i)
446 act.v[r * s.inner + i] = silu(act.conv_out[r * s.conv + i]);
447 for (std::size_t i = 0; i < s.keys; ++i) {
448 act.k[r * s.keys + i] = silu(act.conv_out[r * s.conv + s.inner + i]);
449 act.q[r * s.keys + i] = silu(act.conv_out[r * s.conv + s.inner + s.keys + i]);
450 }
451 for (std::size_t h = 0; h < s.H; ++h)
452 act.gate[r * s.H + h] = softplus(act.proj[r * s.in + 2 * s.inner + 2 * s.keys + h] + w.dt_bias[h]);
453 }
454 std::array<T, s.H> a{};
455 for (std::size_t h = 0; h < s.H; ++h)
456 a[h] = -math::exp(w.a_log[h]);
457 const auto st = states.empty() ? act.final_states : states.subspan(sequences * sk);
458 if (states.empty())
459 fill_rows(backend, act.final_states, T(0));
460 ssd_forward_chunked<C>(backend, ssd_inputs<T>{in(act.v), in(act.gate), std::span<const T>(a), in(act.k), in(act.q)}, act.y, sequences, st,
461 act.checkpoints);
462 for (std::size_t r = 0; r < rows; ++r) {
463 for (std::size_t i = 0; i < s.inner; ++i) {
464 const std::size_t j = r * s.inner + i;
465 act.gated[j] = (act.y[j] + w.d_skip[i / s.P] * act.v[j]) * silu(act.proj[r * s.in + i]);
466 }
467 constexpr std::size_t width = s.inner / s.G;
468 for (std::size_t g = 0; g < s.G; ++g)
469 act.rstd[r * s.G + g] = rms_norm(in(act.gated).subspan(r * s.inner + g * width, width), w.norm.subspan(g * width, width),
470 act.normed.subspan(r * s.inner + g * width, width), norm_eps_v<C, T>);
471 }
472 linear_sequence(backend, w.out_proj, in(act.normed), out);
473}

481template <transformer_config C, typename B, math::math_number T>
482 requires(C.mixer == mixer_kind::mamba2 && linalg::blas_backend<B, T>)
483constexpr void mixer_backward(const B& backend, const mamba2_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
484 const mixer_activations<T>& act, std::type_identity_t<std::span<const T>> dout,
485 std::type_identity_t<std::span<T>> dxs, const mixer_gradients<T>& grads, std::size_t sequences = 1)
486{
487 using namespace detail::mixer_layer;
488 constexpr auto s = shape_of(C);
489 const std::size_t rows = xs.size() / s.d;
490 assert(dout.size() == xs.size() && dxs.size() == xs.size());
491 const auto b = backward_work_in<C, T>(grads.work, rows, sequences);
492 fill_rows(backend, b.dnormed, T(0));
493 linear_sequence_backward(backend, w.out_proj, in(act.normed), dout, b.dnormed, grads.out_proj);
494 fill_rows(backend, b.dproj, T(0));
495 constexpr std::size_t width = s.inner / s.G;
496 for (std::size_t r = 0; r < rows; ++r) {
497 for (std::size_t g = 0; g < s.G; ++g)
498 rms_norm_backward(in(b.dnormed).subspan(r * s.inner + g * width, width), in(act.gated).subspan(r * s.inner + g * width, width),
499 w.norm.subspan(g * width, width), act.rstd[r * s.G + g], b.dgated.subspan(r * s.inner + g * width, width),
500 grads.norm.subspan(g * width, width));
501 for (std::size_t i = 0; i < s.inner; ++i) { // gated = (y + D x) silu(z): dgated becomes dy
502 const std::size_t j = r * s.inner + i, h = i / s.P;
503 const T z = act.proj[r * s.in + i], yv = act.y[j] + w.d_skip[h] * act.v[j], dg = b.dgated[j];
504 b.dproj[r * s.in + i] = dg * yv * silu_derivative(z);
505 b.dgated[j] = dg * silu(z);
506 grads.d_skip[h] += b.dgated[j] * act.v[j];
507 }
508 }
509 std::array<T, s.H> a{};
510 for (std::size_t h = 0; h < s.H; ++h)
511 a[h] = -math::exp(w.a_log[h]);
512 fill_rows(backend, b.da, T(0));
513 fill_rows(backend, b.dstates, T(0));
514 ssd_backward<C>(backend, ssd_inputs<T>{in(act.v), in(act.gate), std::span<const T>(a), in(act.k), in(act.q)}, in(b.dgated),
515 ssd_gradients<T>{b.dv, b.dgate, b.da, b.dk, b.dq}, sequences, in(act.checkpoints), b.dstates, b.op);
516 for (std::size_t h = 0; h < s.H; ++h)
517 grads.a_log[h] += b.da[h] * a[h];
518 for (std::size_t r = 0; r < rows; ++r) {
519 for (std::size_t h = 0; h < s.H; ++h) { // dt = softplus(raw + dt_bias)
520 const std::size_t col = r * s.in + 2 * s.inner + 2 * s.keys + h;
521 const T d = b.dgate[r * s.H + h] * sigmoid(act.proj[col] + w.dt_bias[h]);
522 b.dproj[col] = d;
523 grads.dt_bias[h] += d;
524 }
525 for (std::size_t i = 0; i < s.conv; ++i) { // xBC = silu(conv_out)
526 const std::size_t j = r * s.conv + i;
527 const T dpost = i < s.inner ? b.dv[r * s.inner + i] + w.d_skip[i / s.P] * b.dgated[r * s.inner + i]
528 : i < s.inner + s.keys ? b.dk[r * s.keys + i - s.inner]
529 : b.dq[r * s.keys + i - s.inner - s.keys];
530 b.dconv_out[j] = dpost * silu_derivative(act.conv_out[j]);
531 }
532 }
533 short_conv_backward(backend, w.conv, in(act.conv_in), in(b.dconv_out), b.dconv_in, grads.conv, grads.conv_bias, sequences);
534 columns<T>(in(b.dconv_in), s.conv, 0, b.dproj, s.in, s.inner, s.conv, rows);
535 linear_sequence_backward(backend, w.in_proj, xs, in(b.dproj), dxs, grads.in_proj);
536}

546template <transformer_config C, typename B, math::math_number T>
547 requires(C.mixer == mixer_kind::gated_deltanet && linalg::blas_backend<B, T>)
548constexpr void mixer_forward_sequence(const B& backend, const gated_deltanet_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
549 const mixer_activations<T>& act, std::type_identity_t<std::span<T>> out, std::size_t sequences = 1,
550 std::type_identity_t<std::span<T>> states = {})
551{
552 using namespace detail::mixer_layer;
553 constexpr auto s = shape_of(C);
554 const std::size_t rows = xs.size() / s.d, sk = short_conv_state_size(s.conv, s.K);
555 assert(xs.size() == rows * s.d && out.size() == xs.size() && sequences > 0 && rows % sequences == 0
556 && (states.empty() || states.size() == sequences * (sk + s.S)));
557 linear_sequence(backend, w.in_proj, xs, act.proj);
558 columns<T>(in(act.proj), s.in, 0, act.conv_in, s.conv, 0, s.conv, rows);
559 short_conv_forward(backend, w.conv, in(act.conv_in), act.conv_out, sequences, states.empty() ? std::span<T>{} : states.first(sequences * sk));
560 for (std::size_t r = 0; r < rows; ++r) {
561 for (std::size_t i = 0; i < s.keys; ++i) {
562 act.q[r * s.keys + i] = silu(act.conv_out[r * s.conv + i]);
563 act.k[r * s.keys + i] = silu(act.conv_out[r * s.conv + s.keys + i]);
564 }
565 for (std::size_t i = 0; i < s.inner; ++i)
566 act.v[r * s.inner + i] = silu(act.conv_out[r * s.conv + 2 * s.keys + i]);
567 for (std::size_t h = 0; h < s.H; ++h) {
568 const std::size_t at = r * s.in + 2 * s.keys + 2 * s.inner;
569 act.beta[r * s.H + h] = sigmoid(act.proj[at + h]);
570 act.gate[r * s.H + h] = -math::exp(w.a_log[h]) * softplus(act.proj[at + s.H + h] + w.dt_bias[h]);
571 }
572 }
573 l2_normalise_rows(backend, in(act.q), act.q, s.N, norm_eps_v<C, T>);
574 l2_normalise_rows(backend, in(act.k), act.k, s.N, norm_eps_v<C, T>);
575 const auto st = states.empty() ? act.final_states : states.subspan(sequences * sk);
576 if (states.empty())
577 fill_rows(backend, act.final_states, T(0));
578 delta_rule_forward_chunked<C>(backend, delta_rule_inputs<T>{in(act.q), in(act.k), in(act.v), in(act.gate), in(act.beta)}, act.y, sequences,
579 st, act.checkpoints, act.work);
580 for (std::size_t r = 0; r < rows; ++r)
581 for (std::size_t h = 0; h < s.H; ++h) {
582 const std::size_t at = r * s.inner + h * s.P;
583 act.rstd[r * s.H + h] = rms_norm(in(act.y).subspan(at, s.P), w.norm, act.normed.subspan(at, s.P), norm_eps_v<C, T>);
584 for (std::size_t p = 0; p < s.P; ++p)
585 act.normed[at + p] *= silu(act.proj[r * s.in + 2 * s.keys + s.inner + h * s.P + p]);
586 }
587 linear_sequence(backend, w.out_proj, in(act.normed), out);
588}

596template <transformer_config C, typename B, math::math_number T>
597 requires(C.mixer == mixer_kind::gated_deltanet && linalg::blas_backend<B, T>)
598constexpr void mixer_backward(const B& backend, const gated_deltanet_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
599 const mixer_activations<T>& act, std::type_identity_t<std::span<const T>> dout,
600 std::type_identity_t<std::span<T>> dxs, const mixer_gradients<T>& grads, std::size_t sequences = 1)
601{
602 using namespace detail::mixer_layer;
603 constexpr auto s = shape_of(C);
604 const std::size_t rows = xs.size() / s.d;
605 assert(dout.size() == xs.size() && dxs.size() == xs.size());
606 const auto b = backward_work_in<C, T>(grads.work, rows, sequences);
607 fill_rows(backend, b.dnormed, T(0));
608 linear_sequence_backward(backend, w.out_proj, in(act.normed), dout, b.dnormed, grads.out_proj);
609 fill_rows(backend, b.dproj, T(0));
610 for (std::size_t r = 0; r < rows; ++r)
611 for (std::size_t h = 0; h < s.H; ++h) { // normed = rms_norm(y) silu(z)
612 const std::size_t at = r * s.inner + h * s.P, zat = r * s.in + 2 * s.keys + s.inner + h * s.P;
613 const T rstd = act.rstd[r * s.H + h];
614 for (std::size_t p = 0; p < s.P; ++p) {
615 const T z = act.proj[zat + p], n = act.y[at + p] * rstd * w.norm[p], d = b.dnormed[at + p];
616 b.dproj[zat + p] = d * n * silu_derivative(z);
617 b.dnormed[at + p] = d * silu(z);
618 }
619 rms_norm_backward(in(b.dnormed).subspan(at, s.P), in(act.y).subspan(at, s.P), w.norm, rstd, b.dgated.subspan(at, s.P), grads.norm);
620 }
621 fill_rows(backend, b.dstates, T(0));
622 delta_rule_backward<C>(backend, delta_rule_inputs<T>{in(act.q), in(act.k), in(act.v), in(act.gate), in(act.beta)}, in(b.dgated),
623 delta_rule_gradients<T>{b.dq, b.dk, b.dv, b.dgate, b.dbeta}, sequences, in(act.checkpoints), b.dstates, b.op);
624 for (std::size_t r = 0; r < rows; ++r) {
625 const std::size_t at = r * s.in + 2 * s.keys + 2 * s.inner;
626 for (std::size_t h = 0; h < s.H; ++h) {
627 const T beta = act.beta[r * s.H + h];
628 b.dproj[at + h] = b.dbeta[r * s.H + h] * beta * (T(1) - beta);
629 const T e = math::exp(w.a_log[h]), dg = b.dgate[r * s.H + h]; // g = -e softplus(a + dt_bias)
630 const T da = -e * sigmoid(act.proj[at + s.H + h] + w.dt_bias[h]) * dg;
631 b.dproj[at + s.H + h] = da;
632 grads.dt_bias[h] += da;
633 grads.a_log[h] += dg * act.gate[r * s.H + h];
634 }
635 for (std::size_t i = 0; i < s.keys; ++i) { // the normalisations' inputs, silu(conv_out)
636 b.pre[r * 2 * s.keys + i] = silu(act.conv_out[r * s.conv + i]);
637 b.pre[r * 2 * s.keys + s.keys + i] = silu(act.conv_out[r * s.conv + s.keys + i]);
638 }
639 }
640 for (std::size_t r = 0; r < rows; ++r) // through the L2 normalisation, in place in dq and dk
641 for (std::size_t g = 0; g < s.G; ++g) {
642 const auto qx = in(b.pre).subspan(r * 2 * s.keys + g * s.N, s.N), kx = in(b.pre).subspan(r * 2 * s.keys + s.keys + g * s.N, s.N);
643 l2_normalise_rows_backward(backend, qx, in(b.dq).subspan(r * s.keys + g * s.N, s.N), b.dq.subspan(r * s.keys + g * s.N, s.N), s.N,
644 norm_eps_v<C, T>);
645 l2_normalise_rows_backward(backend, kx, in(b.dk).subspan(r * s.keys + g * s.N, s.N), b.dk.subspan(r * s.keys + g * s.N, s.N), s.N,
646 norm_eps_v<C, T>);
647 }
648 for (std::size_t r = 0; r < rows; ++r)
649 for (std::size_t i = 0; i < s.conv; ++i) {
650 const std::size_t j = r * s.conv + i;
651 const T dpost = i < s.keys ? b.dq[r * s.keys + i] : i < 2 * s.keys ? b.dk[r * s.keys + i - s.keys] : b.dv[r * s.inner + i - 2 * s.keys];
652 b.dconv_out[j] = dpost * silu_derivative(act.conv_out[j]);
653 }
654 short_conv_backward(backend, w.conv, in(act.conv_in), in(b.dconv_out), b.dconv_in, grads.conv, std::span<T>{}, sequences);
655 columns<T>(in(b.dconv_in), s.conv, 0, b.dproj, s.in, 0, s.conv, rows);
656 linear_sequence_backward(backend, w.in_proj, xs, in(b.dproj), dxs, grads.in_proj);
657}

661 template <transformer_config C, typename W, typename T>
662 constexpr void decode(const W& w, std::span<const T> x, std::span<T> state, std::span<T> out)
663 {
664 constexpr std::size_t full = mixer_activation_values(C, 1, 1), states = 2 * shape_of(C).S;
665 std::array<T, full - states> values{};
666 mixer_activations<T> a{};
667 cutter<T> cut{std::span<T>(values)};
668 constexpr auto s = shape_of(C);
669 constexpr bool mamba = C.mixer == mixer_kind::mamba2;
670 a.proj = cut.take(s.in);
671 a.conv_in = cut.take(s.conv);
672 a.conv_out = cut.take(s.conv);
673 a.q = cut.take(s.keys);
674 a.k = cut.take(s.keys);
675 a.v = cut.take(s.inner);
676 a.gate = cut.take(s.H);
677 a.beta = cut.take(mamba ? 0 : s.H);
678 a.y = cut.take(s.inner);
679 a.gated = cut.take(mamba ? s.inner : 0);
680 a.rstd = cut.take(mamba ? s.G : s.H);
681 a.normed = cut.take(s.inner);
682 a.checkpoints = {};
683 a.final_states = {};
684 a.work = cut.rest;
685 mixer_forward_sequence<C>(linalg::reference, w, x, a, out, 1, state);
686 }
687} // namespace detail::mixer_layer
