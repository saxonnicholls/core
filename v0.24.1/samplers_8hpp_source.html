141template <math::math_number T>

143{

154 [[nodiscard]] constexpr token_id selected_token() const

165template <row R>
166constexpr token_candidates<row_value_t<R>> make_candidates(const R& logits, std::span<token_data<row_value_t<R>>> storage)
167{
168 using T = row_value_t<R>;
169 const auto x = as_row(logits);
170 assert(storage.size() == x.size());
171 for (std::size_t i = 0; i < x.size(); ++i)
172 storage[i] = {static_cast<token_id>(i), x[i], T(0)};
173 return {storage, false, -1};
174}

178 template <typename T>
179 inline constexpr T neg_inf = -std::numeric_limits<T>::infinity();
180
181 template <typename T>
182 constexpr T exp_or_zero(T x) { return x == neg_inf<T> ? T(0) : math::exp(x); }
183
184 template <typename T>
185 constexpr T log2(T x) { return math::log(x) / math::log(T(2)); }

187 // Descending logit, then ascending id: one order on every standard library
188 template <typename T>
189 constexpr void sort(token_candidates<T>& c)
190 {
191 if (c.sorted)
192 return;
193 std::sort(c.data.begin(), c.data.end(),
194 [](const token_data<T>& a, const token_data<T>& b) { return a.logit > b.logit || (a.logit == b.logit && a.id < b.id); });
195 c.sorted = true;
196 }

198 // llama_sampler_softmax_impl
199 template <typename T>
200 constexpr void softmax(token_candidates<T>& c, bool do_sort)
201 {
202 assert(c.size() > 0);
203 if (do_sort)
204 sort(c);
205 T max = c.data[0].logit;
206 if (!c.sorted)
207 for (const auto& d : c.data)
208 max = std::max(max, d.logit);
209 T sum = T(0);
210 for (auto& d : c.data) {
211 d.p = exp_or_zero(d.logit - max);
212 sum += d.p;
213 }
214 for (auto& d : c.data)
215 d.p /= sum;
216 }

218 // llama_sampler_temp_impl
219 template <typename T>
220 constexpr void temperature(token_candidates<T>& c, T temp)
221 {
222 if (c.size() == 0)
223 return;
224 if (temp <= T(0)) {
225 std::size_t max_i = 0;
226 T max_l = c.data[0].logit;
227 for (std::size_t i = 1; i < c.size(); ++i) {
228 if (c.data[i].logit > max_l) {
229 c.data[max_i].logit = neg_inf<T>;
230 max_i = i;
231 max_l = c.data[i].logit;
232 } else {
233 c.data[i].logit = neg_inf<T>;
234 }
235 }
236 return;
237 }
238 for (auto& d : c.data)
239 d.logit /= temp;
240 }

244 constexpr void top_k(token_candidates<T>& c, std::int32_t k)

252 // The inverse CDF over the candidates' p: the first whose running sum exceeds u times the total; rounding falls
253 // back to the last with a nonzero probability
254 template <typename T>
255 constexpr std::size_t inverse_cdf(const token_candidates<T>& c, double u)
256 {
257 assert(c.size() > 0 && u >= 0.0 && u < 1.0);
258 double total = 0.0;
259 for (const auto& d : c.data)
260 total += static_cast<double>(d.p);
261 const double target = u * total;
262 double run = 0.0;
263 std::size_t last = 0;
264 for (std::size_t i = 0; i < c.size(); ++i) {
265 run += static_cast<double>(c.data[i].p);
266 if (c.data[i].p > T(0))
267 last = i;
268 if (run > target)
269 return i;
270 }
271 return last;
272 }

274 // Finds token t among the candidates: at index t when the array is still in id order, else by a scan
275 template <typename T>
276 constexpr token_data<T>* find(token_candidates<T>& c, token_id t)
277 {
278 if (t >= 0 && static_cast<std::size_t>(t) < c.size() && c.data[static_cast<std::size_t>(t)].id == t)
279 return &c.data[static_cast<std::size_t>(t)];
280 for (auto& d : c.data)
281 if (d.id == t)
282 return &d;
283 return nullptr;
284 }

286 // The last `limit` accepted tokens, oldest overwritten: llama.cpp's ring_buffer
287 template <std::size_t N>
288 struct token_ring
289 {
290 std::array<token_id, N> buf{};
291 std::size_t first = 0, count = 0;
292
293 constexpr void push(token_id t, std::size_t limit)
294 {
295 if (limit == 0)
296 return;
297 if (count < limit) {
298 buf[(first + count) % N] = t;
299 ++count;
300 } else {
301 buf[(first + count) % N] = t; // over the oldest when count is N; any limit up to N stays contiguous mod N
302 first = (first + 1) % N;
303 }
304 }
305 // i-th from the end: rat(0) is the most recent
306 [[nodiscard]] constexpr token_id rat(std::size_t i) const { return buf[(first + count - 1 - i) % N]; }
307 constexpr void clear() { first = count = 0; }
308 };
309} // namespace detail::samplers

331struct dist_sampler
332{
336 template <math::math_number T>
337 constexpr void apply(token_candidates<T>& c, stats::wyrand& g) const
338 {
339 assert(c.size() > 0);
340 const double u = stats::uniform_real(g); // drawn even for one candidate, as llama.cpp keeps its stream aligned
341 T max = c.data[0].logit;
342 if (!c.sorted)
343 for (const auto& d : c.data)
344 max = std::max(max, d.logit);
345 double sum = 0.0; // llama.cpp: float exponentials, a double sum, one pass to select and normalise
346 for (auto& d : c.data) {
347 d.p = detail::samplers::exp_or_zero(d.logit - max);
348 sum += static_cast<double>(d.p);
349 }
350 const double target = sum * u;
351 double run = 0.0;
352 std::size_t last = 0;
353 c.selected = -1;
354 for (std::size_t i = 0; i < c.size(); ++i) {
355 if (c.selected < 0) {
356 run += static_cast<double>(c.data[i].p);
357 if (c.data[i].p > T(0))
358 last = i;
359 if (run > target)
360 c.selected = static_cast<std::ptrdiff_t>(i);
361 }
362 c.data[i].p = static_cast<T>(static_cast<double>(c.data[i].p) / sum);
363 }
364 if (c.selected < 0)
365 c.selected = static_cast<std::ptrdiff_t>(last);
366 }
367};

405 constexpr void apply(token_candidates<T>& c, stats::wyrand&) const
406 {

427template <math::math_number T>
428struct min_p_sampler
429{
430 T p = T(0.05);
431 std::size_t min_keep = 1;
432
435 constexpr void apply(token_candidates<T>& c, stats::wyrand&) const
436 {
437 if (p <= T(0) || c.size() == 0)
438 return;
439 if (!c.sorted) {
440 T max = c.data[0].logit;
441 for (const auto& d : c.data)
442 max = std::max(max, d.logit);
443 const T cut = max + math::log(p);
444 std::size_t kept = 0;
445 for (const auto& d : c.data)
446 kept += d.logit >= cut ? 1 : 0;
447 if (kept > 0 && kept >= min_keep) {
448 std::size_t at = 0;
449 for (std::size_t i = 0; i < c.size(); ++i)
450 if (c.data[i].logit >= cut)
451 c.data[at++] = c.data[i];
452 c.data = c.data.first(kept);
453 return;
454 }
455 }
456 detail::samplers::sort(c);
457 const T cut = c.data[0].logit + math::log(p);
458 std::size_t i = 1;
459 for (; i < c.size(); ++i)
460 if (c.data[i].logit < cut && i >= min_keep)
461 break;
462 c.data = c.data.first(i);
463 }
464};

470template <math::math_number T, std::size_t N = 64>
471struct penalties_sampler
472{
473 static_assert(N > 0, "snicholls::ml::penalties_sampler: the history must hold at least one token");
474 std::size_t last_n = N;
475 T repeat = T(1);
476 T frequency = T(0);
477 T presence = T(0);
478 detail::samplers::token_ring<N> history{};
479
482 constexpr void accept(token_id t)
483 {
484 assert(last_n <= N);
485 history.push(t, last_n);
486 }
487
488 constexpr void reset() { history.clear(); }
489
492 constexpr void apply(token_candidates<T>& c, stats::wyrand&) const
493 {
494 if (last_n == 0 || (repeat == T(1) && frequency == T(0) && presence == T(0)))
495 return;
496 assert(repeat > T(0));
497 for (std::size_t i = 0; i < history.count; ++i) {
498 const token_id t = history.rat(i);
499 bool first = true;
500 std::size_t count = 0;
501 for (std::size_t j = 0; j < history.count; ++j)
502 if (history.rat(j) == t) {
503 first = first && j >= i;
504 ++count;
505 }
506 if (!first)
507 continue;
508 auto* const d = detail::samplers::find(c, t);
509 if (d == nullptr)
510 continue;
511 d->logit = d->logit <= T(0) ? d->logit * repeat : d->logit / repeat;
512 d->logit -= T(count) * frequency + presence;
513 }
514 c.sorted = false;
515 }
516};

521template <math::math_number T>
522struct typical_sampler
523{
524 T p = T(1);
525 std::size_t min_keep = 1;
526
529 constexpr void apply(token_candidates<T>& c, stats::wyrand&) const
530 {
531 if (p >= T(1) || c.size() == 0)
532 return;
533 detail::samplers::softmax(c, true);
534 T entropy = T(0);
535 for (const auto& d : c.data)
536 if (d.p > T(0))
537 entropy += -d.p * math::log(d.p);
538 const auto score = [entropy](const token_data<T>& d) {
539 return d.p > T(0) ? math::abs(-math::log(d.p) - entropy) : std::numeric_limits<T>::infinity();
540 };
541 std::sort(c.data.begin(), c.data.end(), [&](const token_data<T>& a, const token_data<T>& b) {
542 const T sa = score(a), sb = score(b);
543 return sa < sb || (sa == sb && a.id < b.id);
544 });
545 T cum = T(0);
546 std::size_t last = c.size();
547 for (std::size_t i = 0; i < c.size(); ++i) {
548 cum += c.data[i].p;
549 if (cum > p && (min_keep == 0 || i + 1 >= min_keep)) {
550 last = i + 1;
551 break;
552 }
553 }
554 c.data = c.data.first(last);
555 c.sorted = false;
556 }
557};

562template <math::math_number T>
563struct temp_ext_sampler
564{
565 T temp = T(1);
566 T delta = T(0);
567 T exponent = T(1);
568
573 [[nodiscard]] constexpr T dynamic_temperature(T entropy, std::size_t n) const
574 {
575 const T lo = std::max(T(0), temp - delta), hi = temp + delta;
576 const T max_entropy = -math::log(T(1) / T(n));
577 return lo + (hi - lo) * math::pow(entropy / max_entropy, exponent);
578 }
579
582 constexpr void apply(token_candidates<T>& c, stats::wyrand&) const
583 {
584 if (delta <= T(0)) {
585 detail::samplers::temperature(c, temp);
586 return;
587 }
588 if (c.size() <= 1)
589 return;
590 detail::samplers::softmax(c, true);
591 T entropy = T(0);
592 for (const auto& d : c.data)
593 if (d.p > T(0))
594 entropy -= d.p * math::log(d.p);
595 detail::samplers::temperature(c, dynamic_temperature(entropy, c.size()));
596 const double max = static_cast<double>(c.data[0].logit);
597 double sum = 0.0;
598 for (auto& d : c.data) {
599 const double e = d.logit == detail::samplers::neg_inf<T> ? 0.0 : math::exp(static_cast<double>(d.logit) - max);
600 d.p = static_cast<T>(e);
601 sum += e;
602 }
603 for (auto& d : c.data)
604 d.p = static_cast<T>(static_cast<double>(d.p) / sum);
605 }
606};

611template <math::math_number T>
612struct xtc_sampler
613{
614 T probability = T(0);
615 T threshold = T(0.1);
616 std::size_t min_keep = 1;
617
621 constexpr void apply(token_candidates<T>& c, stats::wyrand& g) const
622 {
623 if (probability <= T(0) || threshold > T(0.5) || c.size() < 2)
624 return;
625 if (static_cast<T>(stats::uniform_real(g)) > probability)
626 return;
627 detail::samplers::softmax(c, true);
628 std::size_t pos_last = 0;
629 for (std::size_t i = 0; i < c.size(); ++i) {
630 if (c.data[i].p >= threshold)
631 pos_last = i;
632 else
633 break;
634 }
635 if (c.size() - pos_last >= min_keep && pos_last > 0)
636 c.data = c.data.subspan(pos_last);
637 }

643template <math::math_number T>
644struct top_n_sigma_sampler
645{
646 T n = T(1);
647
650 constexpr void apply(token_candidates<T>& c, stats::wyrand&) const
651 {
652 if (n <= T(0) || c.size() <= 1)
653 return;
654 using W = std::common_type_t<T, double>;
655 constexpr T ninf = detail::samplers::neg_inf<T>;
656 T max = c.data[0].logit, sum = T(0);
657 std::size_t valid = 0;
658 for (const auto& d : c.data)
659 if (d.logit != ninf) {
660 max = std::max(max, d.logit);
661 sum += d.logit;
662 ++valid;
663 }
664 const T mean = valid > 0 ? sum / T(valid) : T(0);
665 T acc = T(0);
666 for (const auto& d : c.data)
667 if (d.logit != ninf) {
668 const W dv = static_cast<W>(d.logit - mean);
669 acc = static_cast<T>(static_cast<W>(acc) + dv * dv); // pow(float, 2) is a double in llama.cpp
670 }
671 const T sd = valid > 0 ? math::sqrt(acc / T(valid)) : T(0);
672 for (auto& d : c.data)
673 if (d.logit < max - n * sd)
674 d.logit = ninf;
675 }
676};

681template <math::math_number T>
682struct mirostat_sampler
683{
684 std::int32_t n_vocab = 0;
685 T tau = T(5);
686 T eta = T(0.1);
687 std::int32_t m = 100;
688 T mu = T(10);
689
691 constexpr void reset() { mu = T(2) * tau; }
692
696 constexpr void apply(token_candidates<T>& c, stats::wyrand& g)
697 {
698 assert(c.size() > 0 && n_vocab > 0);
699 detail::samplers::softmax(c, true);
700 T sum_ti_bi = T(0), sum_ti_sq = T(0);
701 for (std::size_t i = 0; m > 0 && i < static_cast<std::size_t>(m - 1) && i + 1 < c.size(); ++i) {
702 const T t_i = math::log(T(i + 2) / T(i + 1));
703 const T b_i = math::log(c.data[i].p / c.data[i + 1].p);
704 sum_ti_bi += t_i * b_i;
705 sum_ti_sq += t_i * t_i;
706 }
707 std::size_t kk = 1;
708 if (sum_ti_sq > T(0)) {
709 const T s_hat = sum_ti_bi / sum_ti_sq, epsilon_hat = s_hat - T(1);
710 const T k = math::pow((epsilon_hat * math::pow(T(2), mu)) / (T(1) - math::pow(T(n_vocab), -epsilon_hat)), T(1) / s_hat);
711 kk = !(k >= T(1)) ? 1 : k >= T(c.size()) ? c.size() : static_cast<std::size_t>(k); // NaN to 1
712 }
713 detail::samplers::top_k(c, static_cast<std::int32_t>(kk));
714 detail::samplers::softmax(c, true);
715 const std::size_t idx = detail::samplers::inverse_cdf(c, stats::uniform_real(g));
716 c.selected = static_cast<std::ptrdiff_t>(idx);
717 mu = mu - eta * (-detail::samplers::log2(c.data[idx].p) - tau);
718 }
719};

756template <math::math_number T, std::size_t N = 64>
757struct dry_sampler
758{
759 static_assert(N > 1, "snicholls::ml::dry_sampler: the history must hold at least two tokens");
760 T multiplier = T(0);
761 T base = T(1.75);
762 std::int32_t allowed_length = 2;
763 std::size_t penalty_last_n = N;
764 std::span<const std::span<const token_id>> breakers{};
765 detail::samplers::token_ring<N> history{};
766 std::array<std::int32_t, N> repeat_count{};
767
770 [[nodiscard]] constexpr bool enabled() const noexcept { return multiplier != T(0) && base >= T(1) && penalty_last_n != 0; }
771
774 constexpr void accept(token_id t)
775 {
776 assert(penalty_last_n <= N);
777 if (enabled())
778 history.push(t, penalty_last_n);
779 }
780
781 constexpr void reset() { history.clear(); }
782
785 constexpr void apply(token_candidates<T>& c, stats::wyrand&)
786 {
787 if (!enabled())
788 return;
789 const int last_n_repeat = static_cast<int>(std::min(history.count, penalty_last_n));
790 if (last_n_repeat <= allowed_length)
791 return;
792 const auto rat = [this](int i) { return history.rat(static_cast<std::size_t>(i)); };
793 for (int i = 0; i < last_n_repeat; ++i)
794 repeat_count[static_cast<std::size_t>(i)] = 0;
795 // Step 1: the most recent restart sequence limits the repetition's length
796 int rep_limit = last_n_repeat;
797 for (int i = 0; i < last_n_repeat; ++i) {
798 const token_id token = rat(i);
799 int longest = -1;
800 for (const auto& b : breakers) {
801 if (b.empty() || b[0] != token)
802 continue;
803 const int seq_len = static_cast<int>(b.size()) - 1; // the tail, after the head
804 if (seq_len > longest && seq_len <= i) {
805 bool match = true;
806 for (int off = 0; off < seq_len && match; ++off)
807 match = b[static_cast<std::size_t>(off) + 1] == rat(i - off - 1);
808 if (match)
809 longest = seq_len;
810 }
811 }
812 if (longest >= 0) {
813 rep_limit = i - longest;
814 break;
815 }
816 }
817 if (rep_limit < allowed_length)
818 return;
819 // Step 2: the reverse Z-algorithm: repeat_count[last - k] is the length of the context's suffix that also ends k back
820 {
821 const int last = last_n_repeat - 1;
822 int rt = 0, lt = 0;
823 const auto at = [this](int i) -> std::int32_t& { return repeat_count[static_cast<std::size_t>(i)]; };
824 for (int k = 1; k < last_n_repeat; ++k) {
825 if (k > rt) {
826 int n = 0;
827 while (n + k < last_n_repeat && rat(n) == rat(n + k))
828 ++n;
829 at(last - k) = std::min(n, rep_limit);
830 if (n > 0) {
831 lt = k;
832 rt = k + n - 1;
833 }
834 } else {
835 const int p = k - lt, right_part_len = rt - k + 1;
836 if (at(last - p) < right_part_len) {
837 at(last - k) = std::min(at(last - p), rep_limit);
838 } else {
839 int i = rt + 1;
840 while (i < last_n_repeat && rat(i) == rat(i - k))
841 i += 1;
842 at(last - k) = std::min(i - k, rep_limit);
843 lt = k;
844 rt = i - 1;
845 }
846 }
847 }
848 }
849 // Steps 3 and 4: each token that would continue a repeat, penalised once by its longest
850 const double max_log = 88.7228391; // ln of float's largest value, as llama.cpp
851 const int max_exponent = base > T(1.000001) ? static_cast<int>(static_cast<float>(max_log) / static_cast<float>(math::log(base))) : 0;
852 for (int i = 0; i < last_n_repeat - 1; ++i) {
853 const int len = repeat_count[static_cast<std::size_t>(i)];
854 if (len < allowed_length)
855 continue;
856 const token_id token = rat(last_n_repeat - 2 - i);
857 bool longest = true; // only at the longest repeat this token continues (the first of equal ones)
858 for (int j = 0; j < last_n_repeat - 1 && longest; ++j)
859 if (j != i && rat(last_n_repeat - 2 - j) == token) {
860 const int other = repeat_count[static_cast<std::size_t>(j)];
861 longest = !(other > len || (other == len && j < i));
862 }
863 if (!longest)
864 continue;
865 bool single_breaker = false;
866 for (const auto& b : breakers)
867 single_breaker = single_breaker || (b.size() == 1 && b[0] == token);
868 auto* const d = detail::samplers::find(c, token);
869 if (d == nullptr || single_breaker)
870 continue;
871 int e = len - allowed_length;
872 if (max_exponent > 0 && e > max_exponent)
873 e = max_exponent;
874 d->logit -= static_cast<T>(static_cast<double>(multiplier) * math::pow(static_cast<double>(base), static_cast<double>(e)));
875 }
876 c.sorted = false;
877 }
878};
