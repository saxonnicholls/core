153class text_error_sink final : public error_sink
154{
155public:
159 explicit text_error_sink(std::ostream& out, error_level threshold = error_level::trace) noexcept
160 : out_(&out), threshold_(threshold)
161 {
162 }
163
168 void write(error_level level, const error_report& report) override
169 {
170 if (level < threshold_)
171 return;
172 std::string text = "[" + std::string(to_string(level)) + "] " + format_error_report(report) + "\n";
173 const std::lock_guard<std::mutex> lock(mutex_);
174 *out_ << text;
175 out_->flush();
176 }
177
178private:
179 std::ostream* out_;
180 error_level threshold_;
181 std::mutex mutex_;
182};
