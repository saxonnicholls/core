79 // SYMBOL ':' chain id, as characters, built at compile time.
80 template <basic_fixed_string Symbol, std::uint64_t Id>
81 inline constexpr auto token_code = [] {
82 std::array<char, Symbol.size() + 1 + token_digits(Id)> a{};
83 std::size_t i = 0;
84 for (; i < Symbol.size(); ++i)
85 a[i] = Symbol[i];
86 a[i] = ':';
87 std::uint64_t n = Id;
88 for (std::size_t j = a.size(); j-- > i + 1; n /= 10)
89 a[j] = static_cast<char>('0' + n % 10);
90 return a;
91 }();
92 } // namespace detail
