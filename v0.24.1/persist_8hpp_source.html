159 [[nodiscard]] std::string to_record(const SM& machine, std::uint64_t version)
160 {
161 std::string out(persist_format);
162 out += "\nmachine ";
163 out += detail::short_name(type_name<typename SM::definition>());
164 out += "\nversion ";
165 detail::append_number(out, version);
166 out += "\nregions ";
167 detail::append_number(out, static_cast<std::uint64_t>(SM::slot_count));
168 for (const auto& slot : machine.snapshot()) {
169 out += '\n';
170 detail::append_number(out, slot.value);
171 out += ' ';
172 out += slot.name;
173 }
174 out += '\n';
175 return out;

183 template <class SM>
184 result<std::uint64_t, fsm_errc> from_record(std::string_view record, SM& machine) noexcept
185 {
186 std::string_view rest = record;
187 std::string_view field;
188 std::uint64_t version = 0;
189 std::size_t regions = 0;
190 if (detail::next_line(rest) != persist_format || !detail::parse_field(detail::next_line(rest), "machine", field) ||
191 !detail::parse_field(detail::next_line(rest), "version", field) || !detail::parse_number(field, version) ||
192 !detail::parse_field(detail::next_line(rest), "regions", field) || !detail::parse_number(field, regions))
193 return err(fsm_errc::corrupt);
194 if (regions != SM::slot_count)
195 return err(fsm_errc::shape_mismatch);
196 std::array<wire_state, SM::slot_count> slots{};
197 for (auto& slot : slots) {
198 const std::string_view line = detail::next_line(rest);
199 const std::size_t space = line.find(' ');
200 if (space == std::string_view::npos || !detail::parse_number(line.substr(0, space), slot.value))
201 return err(fsm_errc::corrupt);
202 slot.name = line.substr(space + 1);
203 }
204 if (!rest.empty())
205 return err(fsm_errc::corrupt);
206 const auto restored = machine.restore_snapshot(slots);
207 if (restored.is_err())
208 return err(restored.unwrap_err_unchecked());
209 return ok(version);
210 }

221 template <kv_writable Store, kv_byte_source K, class SM>
222 result<std::uint64_t, fsm_errc> save(Store& store, const K& key, const SM& machine, std::uint64_t expected_version)
223 {
224 const std::uint64_t version = expected_version + 1;
225 const kv_bytes key_bytes = kv_bytes_of(key);
226 const std::string_view key_text(reinterpret_cast<const char*>(key_bytes.data()), key_bytes.size());
227 const std::string record = to_record(machine, version);
228 kv_write_batch batch;
229 batch.insert(version_key(key_text, version), record);
230 batch.put(key_text, record);
231 const auto applied = store.apply(batch);
232 if (applied.is_err())
233 return err(applied.unwrap_err_unchecked() == kv_errc::conflict ? fsm_errc::stale
234 : detail::from_kv(applied.unwrap_err_unchecked()));
235 return ok(version);
236 }
