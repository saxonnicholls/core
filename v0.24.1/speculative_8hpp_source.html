104template <std::integral Token>
105constexpr std::size_t ngram_draft(std::span<const Token> history, std::size_t n_min, std::size_t n_max,
106 std::type_identity_t<std::span<Token>> draft)
107{
108 assert(n_min >= 1 && n_max >= n_min);
109 const std::size_t h = history.size();
110 for (std::size_t n = std::min(n_max, h == 0 ? 0 : h - 1); n >= n_min; --n) {
111 const auto pattern = history.subspan(h - n);
112 // the most recent start j < h - n whose n tokens are the pattern
113 for (std::size_t j = h - n; j-- > 0;) {
114 if (!std::equal(pattern.begin(), pattern.end(), history.begin() + static_cast<std::ptrdiff_t>(j)))
115 continue;
116 const std::size_t from = j + n, count = std::min(draft.size(), h - from);
117 std::copy_n(history.begin() + static_cast<std::ptrdiff_t>(from), count, draft.begin());
118 return count;
119 }
120 if (n == n_min)
121 break;
122 }
123 return 0;
124}

141template <std::integral Token, row R>
142constexpr verify_result<Token> verify_greedy(std::span<const Token> draft, const R& target_logits, std::size_t vocab)
143{
144 const auto x = as_row(target_logits);
145 assert(vocab > 0 && x.size() == (draft.size() + 1) * vocab);
146 std::size_t i = 0;
147 for (; i <= draft.size(); ++i) {
148 const auto best = static_cast<Token>(argmax(x.subspan(i * vocab, vocab)));
149 if (i == draft.size() || best != draft[i])
150 return {i, best};
151 }
152 return {draft.size(), Token(0)}; // not reached
153}

157 // Inverse CDF over weights w (any nonnegative scale), u in [0, 1): the first whose running sum exceeds u · total
158 template <typename W>
159 constexpr std::size_t pick(W&& weight, std::size_t n, double u)
160 {
161 double total = 0.0;
162 for (std::size_t i = 0; i < n; ++i)
163 total += static_cast<double>(weight(i));
164 const double target = u * total;
165 double run = 0.0;
166 std::size_t last = 0;
167 for (std::size_t i = 0; i < n; ++i) {
168 const double v = static_cast<double>(weight(i));
169 run += v;
170 if (v > 0.0)
171 last = i;
172 if (run > target)
173 return i;
174 }
175 return last;
176 }

178 // One speculative-sampling acceptance test for draft token x: one draw unless p(x) >= q(x), as llama.cpp
179 template <typename T>
180 constexpr bool accept(std::span<const T> p, std::span<const T> q, std::size_t x, stats::wyrand& g)
181 {
182 assert(q[x] > T(0));
183 const T px = p[x], qx = q[x];
184 return px >= qx || static_cast<T>(stats::uniform_real(g)) < px / qx; // the draw only when needed
185 }
186} // namespace detail::speculative

195template <std::integral Token, row P, row Q>
196 requires std::same_as<row_value_t<P>, row_value_t<Q>>
197constexpr verify_result<Token> verify_sampled(std::span<const Token> draft, const P& target, const Q& proposal, std::size_t vocab,
198 stats::wyrand& g)
199{
200 using T = row_value_t<P>;
201 const auto p = as_row(target);
202 const auto q = as_row(proposal);
203 const std::size_t k = draft.size();
204 assert(vocab > 0 && p.size() == (k + 1) * vocab && q.size() == k * vocab);
205 for (std::size_t i = 0; i < k; ++i) {
206 const auto pi = p.subspan(i * vocab, vocab), qi = q.subspan(i * vocab, vocab);
207 const auto x = static_cast<std::size_t>(draft[i]);
208 assert(x < vocab);
209 if (detail::speculative::accept(pi, qi, x, g))
210 continue;
211 const auto residual = [&](std::size_t t) { return pi[t] > qi[t] ? pi[t] - qi[t] : T(0); };
212 const double u = stats::uniform_real(g);
213 bool any = false;
214 for (std::size_t t = 0; t < vocab && !any; ++t)
215 any = residual(t) > T(0);
216 const std::size_t next = any ? detail::speculative::pick(residual, vocab, u) : detail::speculative::pick([&](std::size_t t) { return pi[t]; }, vocab, u);
217 return {i, static_cast<Token>(next)};
218 }
219 const auto last = p.subspan(k * vocab, vocab);
220 return {k, static_cast<Token>(detail::speculative::pick([&](std::size_t t) { return last[t]; }, vocab, stats::uniform_real(g)))};
221}

227constexpr std::size_t tree_depth(std::span<const std::int32_t> parent, std::size_t i)
228{
229 std::size_t d = 0;
230 for (std::int32_t at = static_cast<std::int32_t>(i); at >= 0; at = parent[static_cast<std::size_t>(at)]) {
231 assert(static_cast<std::size_t>(at) < parent.size() && (parent[static_cast<std::size_t>(at)] < at));
232 ++d;
233 }
234 return d;
235}

240template <std::integral M, std::size_t E>
241constexpr void tree_attention_mask(std::span<const std::int32_t> parent, std::span<M, E> mask)
242{
243 const std::size_t n = parent.size();
244 assert(mask.size() == n * n);
245 std::fill(mask.begin(), mask.end(), M(0));
246 for (std::size_t i = 0; i < n; ++i)
247 for (std::int32_t at = static_cast<std::int32_t>(i); at >= 0; at = parent[static_cast<std::size_t>(at)]) {
248 assert(parent[static_cast<std::size_t>(at)] < at);
249 mask[i * n + static_cast<std::size_t>(at)] = M(1);
250 }
251}

270template <std::integral Token, row R>
271constexpr tree_verify_result<Token> verify_tree_greedy(std::span<const Token> tokens, std::span<const std::int32_t> parent, const R& target_logits,
272 std::size_t vocab, std::span<std::int32_t> path)
273{
274 const auto x = as_row(target_logits);
275 const std::size_t n = tokens.size();
276 assert(parent.size() == n && x.size() == (n + 1) * vocab);
277 std::int32_t at = -1;
278 std::size_t len = 0;
279 for (;;) {
280 const auto best = static_cast<Token>(argmax(x.subspan(static_cast<std::size_t>(at + 1) * vocab, vocab)));
281 std::int32_t child = -1;
282 for (std::size_t c = 0; c < n && child < 0; ++c)
283 if (parent[c] == at && tokens[c] == best)
284 child = static_cast<std::int32_t>(c);
285 if (child < 0)
286 return {len, best};
287 assert(len < path.size());
288 path[len++] = child;
289 at = child;
290 }
291}

303template <std::integral Token, row P, row Q>
304 requires std::same_as<row_value_t<P>, row_value_t<Q>>
305constexpr tree_verify_result<Token> verify_tree_sampled(std::span<const Token> tokens, std::span<const std::int32_t> parent, const P& target,
306 const Q& proposal, std::size_t vocab, std::span<std::int32_t> path,
307 std::span<row_value_t<P>> scratch, stats::wyrand& g)
308{
309 using T = row_value_t<P>;
310 const auto p = as_row(target);
311 const auto q = as_row(proposal);
312 const std::size_t n = tokens.size();
313 assert(parent.size() == n && p.size() == (n + 1) * vocab && q.size() == p.size() && scratch.size() == vocab);
314 std::int32_t at = -1;
315 std::size_t len = 0;
316 for (;;) {
317 const std::size_t row = static_cast<std::size_t>(at + 1) * vocab;
318 const auto qi = q.subspan(row, vocab);
319 std::copy_n(p.begin() + static_cast<std::ptrdiff_t>(row), vocab, scratch.begin());
320 std::int32_t accepted = -1;
321 for (std::size_t c = 0; c < n && accepted < 0; ++c) {
322 if (parent[c] != at)
323 continue;
324 const auto x = static_cast<std::size_t>(tokens[c]);
325 assert(x < vocab);
326 if (detail::speculative::accept(std::span<const T>(scratch), qi, x, g)) {
327 accepted = static_cast<std::int32_t>(c);
328 break;
329 }
330 T sum = T(0); // p' <- normalize(max(0, p' - q)); left as it was if that is empty
331 for (std::size_t t = 0; t < vocab; ++t)
332 sum += scratch[t] > qi[t] ? scratch[t] - qi[t] : T(0);
333 if (sum > T(0))
334 for (std::size_t t = 0; t < vocab; ++t)
335 scratch[t] = (scratch[t] > qi[t] ? scratch[t] - qi[t] : T(0)) / sum;
336 }
337 if (accepted < 0)
338 return {len, static_cast<Token>(detail::speculative::pick([&](std::size_t t) { return scratch[t]; }, vocab, stats::uniform_real(g)))};
339 assert(len < path.size());
340 path[len++] = accepted;
341 at = accepted;
342 }
343}

365template <row R, std::integral Token, std::size_t E>
366constexpr void mtp_draft_sample(const R& logits, std::size_t vocab, std::type_identity_t<row_value_t<R>> temperature, std::span<Token, E> draft,
367 std::span<row_value_t<R>> q, stats::wyrand& g)
368{
369 using T = row_value_t<R>;
370 const auto x = as_row(logits);
371 assert(vocab > 0 && temperature > T(0) && x.size() == draft.size() * vocab && q.size() == x.size());
372 for (std::size_t k = 0; k < draft.size(); ++k) {
373 const auto row = x.subspan(k * vocab, vocab);
374 const auto out = q.subspan(k * vocab, vocab);
375 const T max = row[argmax(row)];
376 T sum = T(0);
377 for (std::size_t t = 0; t < vocab; ++t) {
378 out[t] = math::exp((row[t] - max) / temperature);
379 sum += out[t];
380 }
381 for (T& v : out)
382 v /= sum;
383 draft[k] = static_cast<Token>(detail::speculative::pick([&](std::size_t t) { return out[t]; }, vocab, stats::uniform_real(g)));
384 }
385}
