101constexpr std::size_t stft_frames(std::size_t length, const stft_options& o) noexcept
102{
103 const std::size_t padded = length + 2 * o.padding();
104 return padded < o.n_fft || o.hop == 0 ? 0 : 1 + (padded - o.n_fft) / o.hop;
105}

111constexpr std::size_t istft_length(std::size_t frames, const stft_options& o) noexcept
112{
113 if (frames == 0)
114 return 0;
115 const std::size_t full = o.n_fft + o.hop * (frames - 1), pad = 2 * o.padding();
116 return full > pad ? full - pad : 0;
117}

121 // The index in x of sample p of the padded signal, or -1 for a zero of constant padding.
122 constexpr long long source_index(std::size_t p, std::size_t pad, std::size_t n, pad_mode mode) noexcept
123 {
124 const long long q = static_cast<long long>(p) - static_cast<long long>(pad), nn = static_cast<long long>(n);
125 if (q >= 0 && q < nn)
126 return q;
127 if (mode == pad_mode::constant)
128 return -1;
129 const long long r = q < 0 ? -q : 2 * (nn - 1) - q;
130 assert(r >= 0 && r < nn); // reflection needs a signal longer than the padding
131 return r;
132 }

134 template <typename T>
135 constexpr T sample(std::span<const T> x, std::size_t p, std::size_t pad, pad_mode mode) noexcept
136 {
137 const long long i = source_index(p, pad, x.size(), mode);
138 return i < 0 ? T(0) : x[static_cast<std::size_t>(i)];

147 // sum_t w[p - t hop]^2 over the frames covering padded sample p: istft's divisor.
148 template <typename T>
149 constexpr T envelope(std::span<const T> w, std::size_t p, std::size_t hop, std::size_t frames) noexcept
150 {
151 const std::size_t n = w.size();
152 const std::size_t first = p + 1 > n ? (p + 1 - n + hop - 1) / hop : 0;
153 const std::size_t last = p / hop < frames ? p / hop : frames - 1;
154 T e = T(0);
155 for (std::size_t t = first; t <= last && t < frames; ++t)
156 e += w[p - t * hop] * w[p - t * hop];
157 return e;
158 }

178template <math::math_number T>
179constexpr void stft(const rfft_plan<T>& plan, std::span<const std::type_identity_t<T>> x, std::span<const std::type_identity_t<T>> window,
180 std::span<complex<std::type_identity_t<T>>> out, std::span<complex<std::type_identity_t<T>>> work,
181 const stft_options& o) noexcept
182{
183 detail::stft::check(plan.size(), window, o);
184 const std::size_t frames = stft_frames(x.size(), o), bins = plan.bins(), pad = o.padding();
185 assert(out.size() == frames * bins);
186 const T s = detail::stft::spectrum_scale<T>(o);
187 for (std::size_t t = 0; t < frames; ++t) {
188 const auto row = out.subspan(t * bins, bins);
189 plan.forward_from([&](std::size_t j) { return window[j] * detail::stft::sample(x, t * o.hop + j, pad, o.pad); }, row, work);
190 detail::fft::scale_all(row.data(), bins, s);
191 }
192}

202template <math::math_number T>
203constexpr void stft(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> x, std::span<const std::type_identity_t<T>> window,
204 std::span<complex<std::type_identity_t<T>>> out, std::span<complex<std::type_identity_t<T>>> work,
205 const stft_options& o) noexcept
206{
207 detail::stft::check(plan.size(), window, o);
208 const std::size_t frames = stft_frames(x.size(), o), n = o.n_fft, pad = o.padding();
209 assert(out.size() == frames * n);
210 const T s = detail::stft::spectrum_scale<T>(o);
211 for (std::size_t t = 0; t < frames; ++t) {
212 const auto row = out.subspan(t * n, n);
213 for (std::size_t j = 0; j < n; ++j)
214 row[j] = {window[j] * detail::stft::sample(x, t * o.hop + j, pad, o.pad), T(0)};
215 plan.template execute<false>(row, work);
216 detail::fft::scale_all(row.data(), n, s);
217 }

222 // Overlap-adds frame t's samples (already inverse-transformed and scaled) into out, then (once all are in) divides.
223 template <typename T>
224 constexpr void normalise(std::span<const T> window, std::span<T> out, std::size_t frames, const stft_options& o) noexcept
225 {
226 const std::size_t pad = o.padding(), covered = o.n_fft + o.hop * (frames == 0 ? 0 : frames - 1);
227 for (std::size_t i = 0; i < out.size(); ++i) {
228 const std::size_t p = i + pad;
229 if (frames == 0 || p >= covered) {
230 out[i] = T(0);
231 continue;
232 }
233 const T e = envelope(window, p, o.hop, frames);
234 assert(e > T(1e-11)); // NOLA: torch.istft raises here
235 out[i] /= e;
236 }
237 }
238} // namespace detail::stft

248template <math::math_number T>
249constexpr void istft(const rfft_plan<T>& plan, std::span<const complex<std::type_identity_t<T>>> spectrum,
250 std::span<const std::type_identity_t<T>> window, std::span<std::type_identity_t<T>> out,
251 std::span<complex<std::type_identity_t<T>>> work, const stft_options& o) noexcept
252{
253 detail::stft::check(plan.size(), window, o);
254 const std::size_t bins = plan.bins(), frames = spectrum.size() / bins, pad = o.padding();
255 assert(spectrum.size() == frames * bins);
256 const T s = T(1) / T(static_cast<long long>(o.n_fft)) / detail::stft::spectrum_scale<T>(o);
257 for (auto& v : out)
258 v = T(0);
259 for (std::size_t t = 0; t < frames; ++t)
260 plan.inverse_to([&](std::size_t k) { return spectrum[t * bins + k]; },
261 [&](std::size_t j, T v) {
262 const std::size_t p = t * o.hop + j;
263 if (p >= pad && p - pad < out.size())
264 out[p - pad] += window[j] * v * s;
265 },
266 work);
267 detail::stft::normalise(window, out, frames, o);
268}

278template <math::math_number T>
279constexpr void istft(const fft_plan<T>& plan, std::span<const complex<std::type_identity_t<T>>> spectrum,
280 std::span<const std::type_identity_t<T>> window, std::span<std::type_identity_t<T>> out,
281 std::span<complex<std::type_identity_t<T>>> work, const stft_options& o) noexcept
282{
283 detail::stft::check(plan.size(), window, o);
284 const std::size_t n = o.n_fft, frames = n == 0 ? 0 : spectrum.size() / n, pad = o.padding();
285 assert(spectrum.size() == frames * n && work.size() >= n + plan.work_size());
286 const T s = T(1) / T(static_cast<long long>(n)) / detail::stft::spectrum_scale<T>(o);
287 for (auto& v : out)
288 v = T(0);
289 const auto buf = work.first(n);
290 for (std::size_t t = 0; t < frames; ++t) {
291 for (std::size_t k = 0; k < n; ++k)
292 buf[k] = spectrum[t * n + k];
293 plan.template execute<true>(buf, work.subspan(n));
294 for (std::size_t j = 0; j < n; ++j) {
295 const std::size_t p = t * o.hop + j;
296 if (p >= pad && p - pad < out.size())
297 out[p - pad] += window[j] * buf[j].re * s;
298 }
299 }
300 detail::stft::normalise(window, out, frames, o);
301}

311template <math::math_number T>
312constexpr void stft_backward(const rfft_plan<T>& plan, std::span<const complex<std::type_identity_t<T>>> grad_spectrum,
313 std::span<const std::type_identity_t<T>> window, std::span<std::type_identity_t<T>> grad_x,
314 std::span<complex<std::type_identity_t<T>>> work, const stft_options& o) noexcept
315{
316 detail::stft::check(plan.size(), window, o);
317 const std::size_t bins = plan.bins(), frames = stft_frames(grad_x.size(), o), pad = o.padding(), n = o.n_fft;
318 assert(grad_spectrum.size() == frames * bins);
319 const T s = detail::stft::spectrum_scale<T>(o);
320 for (auto& v : grad_x)
321 v = T(0);
322 for (std::size_t t = 0; t < frames; ++t)
323 plan.inverse_to(
324 [&](std::size_t k) {
325 const complex<T> g = grad_spectrum[t * bins + k];
326 return k == 0 || 2 * k == n ? g : g * T(0.5);
327 },
328 [&](std::size_t j, T v) {
329 const long long i = detail::stft::source_index(t * o.hop + j, pad, grad_x.size(), o.pad);
330 if (i >= 0)
331 grad_x[static_cast<std::size_t>(i)] += window[j] * v * s;
332 },
333 work);
334}

344template <math::math_number T>
345constexpr void istft_backward(const rfft_plan<T>& plan, std::span<const std::type_identity_t<T>> grad_y,
346 std::span<const std::type_identity_t<T>> window, std::span<complex<std::type_identity_t<T>>> grad_spectrum,
347 std::span<complex<std::type_identity_t<T>>> work, const stft_options& o) noexcept
348{
349 detail::stft::check(plan.size(), window, o);
350 const std::size_t bins = plan.bins(), frames = grad_spectrum.size() / bins, pad = o.padding(), n = o.n_fft;
351 const std::size_t covered = n + o.hop * (frames == 0 ? 0 : frames - 1);
352 const T s = T(1) / T(static_cast<long long>(n)) / detail::stft::spectrum_scale<T>(o);
353 for (std::size_t t = 0; t < frames; ++t) {
354 const auto row = grad_spectrum.subspan(t * bins, bins);
355 plan.forward_from(
356 [&](std::size_t j) {
357 const std::size_t p = t * o.hop + j;
358 if (p < pad || p - pad >= grad_y.size() || p >= covered)
359 return T(0);
360 return window[j] * grad_y[p - pad] / detail::stft::envelope(window, p, o.hop, frames);
361 },
362 row, work);
363 for (std::size_t k = 0; k < bins; ++k)
364 row[k] *= k == 0 || 2 * k == n ? s : T(2) * s;
365 }

395template <math::math_number T>
396constexpr void spectrogram(const rfft_plan<T>& plan, std::span<const std::type_identity_t<T>> x, std::span<const std::type_identity_t<T>> window,
397 std::span<std::type_identity_t<T>> out, std::span<complex<std::type_identity_t<T>>> work, const stft_options& o,
398 spectrum_kind kind = spectrum_kind::power) noexcept
399{
400 detail::stft::check(plan.size(), window, o);
401 const std::size_t frames = stft_frames(x.size(), o), bins = plan.bins(), pad = o.padding();
402 assert(out.size() == frames * bins && work.size() >= spectrogram_work_size(plan));
403 const T s = detail::stft::spectrum_scale<T>(o);
404 const auto row = work.first(bins);
405 for (std::size_t t = 0; t < frames; ++t) {
406 plan.forward_from([&](std::size_t j) { return window[j] * detail::stft::sample(x, t * o.hop + j, pad, o.pad); }, row,
407 work.subspan(bins));
408 for (std::size_t k = 0; k < bins; ++k) {
409 const T p = norm(row[k]) * (s * s);
410 out[t * bins + k] = kind == spectrum_kind::power ? p : math::sqrt(p);
411 }
412 }
413}
