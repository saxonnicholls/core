74 // Slicing-by-8: table[0] is the byte-at-a-time table; table[k][b] is table[0][b] advanced by k zero bytes.
75 template <typename T, T Poly>
76 inline constexpr std::array<std::array<T, 256>, 8> crc_tables = [] {
77 std::array<std::array<T, 256>, 8> t{};
78 for (std::size_t b = 0; b < 256; ++b) {
79 T c = static_cast<T>(b);
80 for (int k = 0; k < 8; ++k)
81 c = static_cast<T>((c & 1) ? (c >> 1) ^ Poly : c >> 1);
82 t[0][b] = c;
83 }
84 for (std::size_t k = 1; k < 8; ++k)
85 for (std::size_t b = 0; b < 256; ++b)
86 t[k][b] = static_cast<T>((t[k - 1][b] >> 8) ^ t[0][t[k - 1][b] & 0xff]);
87 return t;
88 }();

98template <typename T, T Poly, T Init, T XorOut>
99class reflected_crc : public detail::one_shot_hash<reflected_crc<T, Poly, Init, XorOut>>
100{
101public:
103 using value_type = T;
104
106 constexpr reflected_crc() noexcept = default;
107
111 [[nodiscard]] static constexpr reflected_crc resume(T crc) noexcept
112 {
113 reflected_crc c;
114 c.reg_ = static_cast<T>(crc ^ XorOut);
115 return c;
116 }
117
120 [[nodiscard]] constexpr T finish() const noexcept { return static_cast<T>(reg_ ^ XorOut); }
121
122private:
123 friend detail::hash_updates<reflected_crc>;
124
125 template <typename U>
126 constexpr void absorb(const U* p, std::size_t n) noexcept
127 {
128 constexpr const auto& t = detail::crc_tables<T, Poly>;
129 T c = reg_;
130 for (; n >= 8; p += 8, n -= 8) {
131 const std::uint64_t w = static_cast<std::uint64_t>(c) ^ detail::load_le_word<std::uint64_t>(p);
132 c = static_cast<T>(t[7][w & 0xff] ^ t[6][(w >> 8) & 0xff] ^ t[5][(w >> 16) & 0xff] ^ t[4][(w >> 24) & 0xff] ^
133 t[3][(w >> 32) & 0xff] ^ t[2][(w >> 40) & 0xff] ^ t[1][(w >> 48) & 0xff] ^ t[0][w >> 56]);
134 }
135 for (std::size_t i = 0; i < n; ++i)
136 c = static_cast<T>((c >> 8) ^ t[0][(c ^ detail::octet_of(p[i])) & 0xff]);
137 reg_ = c;
138 }
139
140 T reg_ = Init;
141};
