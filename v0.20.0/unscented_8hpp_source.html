96 template <linalg::scalar T, std::size_t N>
97 [[nodiscard]] constexpr result<point_set<T, N, 2 * N + 1>, linalg::error> make_sigma_points(const gaussian<T, N>& g, const unscented_parameters<T>& params = {}) noexcept(linalg::detail::nothrow_scalar<T>)
98 {
99 const auto l = linalg::cholesky(g.covariance);
100 if (!l)
101 return err(l.unwrap_err_unchecked());
102 const T n = static_cast<T>(N);
103 const T kappa = params.kappa ? *params.kappa : T(3) - n;
104 const T a2 = params.alpha * params.alpha;
105 const T lambda = a2 * (n + kappa) - n;
106 const T c = n + lambda;
107 const T root = static_cast<T>(math::sqrt(c));
108 point_set<T, N, 2 * N + 1> s{};
109 s.points[0] = g.mean;
110 for (std::size_t i = 0; i < N; ++i) {
111 linalg::fixed_vector<T, N> li{};
112 for (std::size_t r = 0; r < N; ++r)
113 li[r] = (*l)(r, i) * root;
114 s.points[1 + i] = g.mean + li;
115 s.points[1 + N + i] = g.mean - li;
116 }
117 for (std::size_t p = 0; p < 2 * N + 1; ++p)
118 s.mean_weights[p] = s.covariance_weights[p] = T(0.5) / c;
119 s.mean_weights[0] = lambda / c;
120 s.covariance_weights[0] = lambda / c + (T(1) - a2 + params.beta);
121 return ok(s);
122 }

131 template <linalg::scalar T, std::size_t N>
132 [[nodiscard]] constexpr result<point_set<T, N, 2 * N>, linalg::error> make_cubature_points(const gaussian<T, N>& g, T alpha = T(1)) noexcept(linalg::detail::nothrow_scalar<T>)
133 {
134 const auto l = linalg::cholesky(g.covariance);
135 if (!l)
136 return err(l.unwrap_err_unchecked());
137 const T root = static_cast<T>(math::sqrt(alpha * static_cast<T>(N)));
138 point_set<T, N, 2 * N> s{};
139 for (std::size_t i = 0; i < N; ++i) {
140 linalg::fixed_vector<T, N> li{};
141 for (std::size_t r = 0; r < N; ++r)
142 li[r] = (*l)(r, i) * root;
143 s.points[i] = g.mean + li;
144 s.points[N + i] = g.mean - li;
145 }
146 for (std::size_t p = 0; p < 2 * N; ++p) {
147 s.mean_weights[p] = T(1) / static_cast<T>(2 * N);
148 s.covariance_weights[p] = T(1) / (static_cast<T>(2 * N) * alpha);
149 }
150 return ok(s);
151 }

178 template <auto Kinds, std::size_t M, typename F, linalg::scalar T, std::size_t N, std::size_t P>
179 [[nodiscard]] constexpr transformed<T, N, M> point_transform(const point_set<T, N, P>& s, F&& f, const linalg::fixed_matrix<T, M, M>& noise)
180 {
181 std::array<linalg::fixed_vector<T, M>, P> y{};
182 for (std::size_t p = 0; p < P; ++p)
183 y[p] = f(s.points[p]);
184 transformed<T, N, M> t{};
185 t.mean = geometry::weighted_mean<Kinds>(y, s.mean_weights);
186 linalg::fixed_vector<T, N> xm{};
187 for (std::size_t p = 0; p < P; ++p)
188 xm += s.points[p] * s.mean_weights[p];
189 for (std::size_t p = 0; p < P; ++p) {
190 const auto dy = geometry::residual<Kinds>(y[p], t.mean);
191 t.covariance += linalg::outer(dy, dy) * s.covariance_weights[p];
192 t.cross_covariance += linalg::outer(s.points[p] - xm, dy) * s.covariance_weights[p];
193 }
194 t.covariance = linalg::symmetrise(t.covariance + noise);
195 return t;
196 }
