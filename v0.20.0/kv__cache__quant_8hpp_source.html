87{

112 constexpr quantized_kv_cache_view(std::span<KBlock> keys, std::span<VBlock> values, std::size_t layers, std::size_t capacity) noexcept

124 template <row K, row V>
125 requires std::same_as<row_value_t<K>, float> && std::same_as<row_value_t<V>, float>
126 constexpr void append(std::size_t layer, std::size_t pos, const K& k, const V& v) const
127 {
128 const auto ks = as_row(k);
129 const auto vs = as_row(v);
130 assert(layer < layers_ && capacity_ > 0 && ks.size() == C.kv_size() && vs.size() == C.kv_size());
131 for (std::size_t h = 0; h < C.n_kv_heads; ++h) {
132 const std::size_t at = offset(layer, h, pos % capacity_);
133 quantize(ks.subspan(h * C.head_dim, C.head_dim), keys_.subspan(at, blocks_per_row));
134 quantize(vs.subspan(h * C.head_dim, C.head_dim), values_.subspan(at, blocks_per_row));
135 }
136 }

222template <transformer_config C, quant_block KBlock, quant_block VBlock, row Q, mutable_row Out>
223 requires std::same_as<row_value_t<Q>, float> && std::same_as<row_value_t<Out>, float>
224constexpr float attend_quantized(const quantized_kv_cache_view<C, KBlock, VBlock>& cache, std::size_t layer, std::size_t kv_head, const Q& q,
225 key_range keys, Out&& out)
226{
227 constexpr std::size_t d = C.head_dim, blocks = d / quant_block_size;
228 const auto qs = as_row(q);
229 const auto y = as_mutable_row(out);
230 assert(qs.size() == d && y.size() == d && keys.size() <= cache.capacity());
231 std::array<block_q8_0, blocks> qq{};
232 quantize(qs, std::span<block_q8_0, blocks>(qq));
233 const float scale = 1.0f / math::sqrt(float(d));
234 std::array<float, d> v{};
235 std::fill(y.begin(), y.end(), 0.0f);
236 float max = -std::numeric_limits<float>::infinity(), sum = 0.0f;
237 for (std::size_t p = keys.first; p < keys.last; ++p) {
238 const std::size_t slot = p % cache.capacity();
239 float s = scale * dot(cache.key_blocks(layer, kv_head, slot), std::span<const block_q8_0>(qq));
240 if constexpr (C.attention_softcap > 0)
241 s = softcap(s, attention_softcap_v<C, float>);
242 dequantize(cache.value_blocks(layer, kv_head, slot), v);
243 detail::attention::fold(max, sum, y, s, std::span<const float>(v));
244 }
245 for (float& x : y)
246 x = sum == 0.0f ? 0.0f : x / sum;
247 return sum == 0.0f ? -std::numeric_limits<float>::infinity() : max + math::log(sum);
248}

259template <transformer_config C, quant_block KBlock, quant_block VBlock, std::size_t Capacity>
260constexpr void kv_shift_quantized_keys(const quantized_kv_cache_view<C, KBlock, VBlock>& cache, kv_cells<Capacity>& cells)
261{
262 static_assert(C.rope != rope_kind::none && !C.latent(),
263 "snicholls::ml::kv_shift_quantized_keys: the configuration rotates no keys, or its keys are latent");
264 assert(cache.capacity() == Capacity);
265 constexpr std::size_t d = C.head_dim, h = d / 2;
266 if (!cells.has_shift())
267 return;
268 std::array<float, d> row{};
269 for (std::size_t i = 0; i < Capacity; ++i) {
270 const kv_pos delta = cells.shift(i);
271 if (cells.is_empty(i) || delta == 0)
272 continue;
273 std::array<float, h> c{}, s{};
274 rope_angles(static_cast<std::size_t>(delta < 0 ? -delta : delta), c, s, rope_base_v<C, float>, rope_scaling_v<C, float>);
275 for (std::size_t layer = 0; layer < cache.layers(); ++layer)
276 for (std::size_t kv = 0; kv < C.n_kv_heads; ++kv) {
277 const auto blocks = cache.mutable_key_blocks(layer, kv, i);
278 dequantize(std::span<const KBlock>(blocks), row);
279 if (delta > 0)
280 rope(row, c, s, C.layout());
281 else
282 rope_inverse(row, c, s, C.layout());
283 quantize(row, blocks);
284 }
285 }
286 cells.reset_shift();
287}
