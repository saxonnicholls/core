212template <typename S, row X, mutable_row Y, row B = std::span<const row_value_t<X>>>
213 requires std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<B>>
214constexpr void linear(const dense_weights<S>& w, const X& x, Y&& y, const B& bias = B{})
215{
216 using T = row_value_t<X>;
217 const auto in = as_row(x);
218 const auto out = as_mutable_row(y);
219 const auto b = as_row(bias);
220 assert(in.size() == w.cols && out.size() == w.rows && w.data.size() == w.rows * w.cols && (b.empty() || b.size() == w.rows));
221 for (std::size_t i = 0; i < w.rows; ++i) {
222 const auto r = w.row(i);
223 T acc = b.empty() ? T(0) : b[i];
224 for (std::size_t j = 0; j < w.cols; ++j) {
225 const T p = detail::linear::widen<T>(r[j]) * in[j];
226 acc += p;
227 }
228 out[i] = acc;
229 }
230}

244template <typename S, row X, row DY, mutable_row DX, mutable_row DW, mutable_row DB = std::span<row_value_t<X>>>
245 requires std::same_as<row_value_t<X>, row_value_t<DY>> && std::same_as<row_value_t<X>, row_value_t<DX>>
246 && std::same_as<row_value_t<X>, row_value_t<DW>> && std::same_as<row_value_t<X>, row_value_t<DB>>
247constexpr void linear_backward(const dense_weights<S>& w, const X& x, const DY& dy, DX&& dx, DW&& dw, DB&& db = DB{})
248{
249 using T = row_value_t<X>;
250 const auto in = as_row(x);
251 const auto g = as_row(dy);
252 const auto gx = as_mutable_row(dx);
253 const auto gw = as_mutable_row(dw);
254 const auto gb = as_mutable_row(db);
255 assert(in.size() == w.cols && g.size() == w.rows && w.data.size() == w.rows * w.cols && gw.size() == w.rows * w.cols
256 && (gx.empty() || gx.size() == w.cols) && (gb.empty() || gb.size() == w.rows));
257 for (std::size_t i = 0; i < w.rows; ++i) {
258 const auto r = w.row(i);
259 const T d = g[i];
260 if (!gx.empty())
261 for (std::size_t j = 0; j < w.cols; ++j) {
262 const T p = detail::linear::widen<T>(r[j]) * d;
263 gx[j] += p;
264 }
265 for (std::size_t j = 0; j < w.cols; ++j) {
266 const T p = d * in[j];
267 gw[i * w.cols + j] += p;
268 }
269 if (!gb.empty())
270 gb[i] += d;
271 }
272}

279template <quant_block Block, std::size_t M, mutable_row Y, row B = std::span<const float>>
280 requires std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
281constexpr void linear(const quantized_weights<Block>& w, const std::array<block_q8_0, M>& x, Y&& y, const B& bias = B{})
282{
283 const auto out = as_mutable_row(y);
284 const auto b = as_row(bias);
285 assert(M * quant_block_size == w.cols && out.size() == w.rows && w.blocks.size() == w.rows * M && (b.empty() || b.size() == w.rows));
286 for (std::size_t i = 0; i < w.rows; ++i) {
287 const float d = dot(w.row(i), std::span<const block_q8_0>(x));
288 out[i] = b.empty() ? d : b[i] + d;
289 }
290}

306 // C = alpha op(A) op(B) + beta C. The shapes are right by construction and C never overlaps A or B, so the
307 // backend's product always succeeds
308 template <typename B, typename T, typename L, typename R, typename C>
309 constexpr void product(const B& backend, T alpha, const L& lhs, const R& rhs, T beta, const C& c)
310 {
311 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
312 assert(done.is_ok());
313 }

411template <math::math_number T, typename B>
412 requires linalg::blas_backend<B, T>
413constexpr void linear_sequence(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
414 std::type_identity_t<std::span<T>> ys, std::type_identity_t<std::span<const T>> bias = {})
415{
416 assert(w.cols > 0 && xs.size() % w.cols == 0);
417 const std::size_t n = xs.size() / w.cols;
418 assert(ys.size() == n * w.rows && w.data.size() == w.rows * w.cols && (bias.empty() || bias.size() == w.rows));
419 if (n == 0 || w.rows == 0)
420 return;
421 using detail::linear::view;
422 detail::linear::product(backend, T(1), view(xs, n, w.cols), linalg::transposed(view(w.data, w.rows, w.cols)), T(0),
423 view(ys, n, w.rows));
424 if (!bias.empty())
425 add_bias_rows(backend, ys, bias);
426}

437template <math::math_number T, typename B>
438 requires linalg::blas_backend<B, T>
439constexpr void linear_sequence_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
440 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
441 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db = {})
442{
443 assert(w.cols > 0 && xs.size() % w.cols == 0);
444 const std::size_t n = xs.size() / w.cols;
445 assert(dys.size() == n * w.rows && (dxs.empty() || dxs.size() == xs.size()) && dw.size() == w.rows * w.cols
446 && (db.empty() || db.size() == w.rows));
447 if (n == 0 || w.rows == 0)
448 return;
449 using detail::linear::view;
450 const auto dy = view(dys, n, w.rows);
451 if (!dxs.empty())
452 detail::linear::product(backend, T(1), dy, view(w.data, w.rows, w.cols), T(1), view(dxs, n, w.cols));
453 detail::linear::product(backend, T(1), linalg::transposed(dy), view(xs, n, w.cols), T(1), view(dw, w.rows, w.cols));
454 if (!db.empty())
455 sum_rows(backend, dys, db);
456}

468template <math::math_number T, typename B>
469 requires linalg::blas_backend<B, T>
470constexpr void linear_heads(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
471 std::type_identity_t<std::span<T>> out, std::type_identity_t<std::span<const T>> bias, std::size_t heads,
472 std::size_t sequences = 1)
473{
474 assert(heads > 0 && sequences > 0 && w.rows % heads == 0 && w.cols > 0 && xs.size() % (w.cols * sequences) == 0);
475 const std::size_t d = w.rows / heads, n = xs.size() / w.cols / sequences;
476 assert(out.size() == sequences * n * w.rows);
477 for (std::size_t s = 0; s < sequences; ++s)
478 for (std::size_t h = 0; h < heads; ++h)
479 linear_sequence(backend, dense_weights<T>{w.data.subspan(h * d * w.cols, d * w.cols), d, w.cols}, xs.subspan(s * n * w.cols, n * w.cols),
480 out.subspan((s * heads + h) * n * d, n * d), bias.empty() ? bias : bias.subspan(h * d, d));
481}

488template <math::math_number T, typename B>
489 requires linalg::blas_backend<B, T>
490constexpr void linear_heads_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
491 std::type_identity_t<std::span<const T>> douts, std::type_identity_t<std::span<T>> dxs,
492 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db, std::size_t heads,

497 assert(douts.size() == sequences * n * w.rows && dw.size() == w.rows * w.cols);
