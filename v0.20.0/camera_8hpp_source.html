129 template <linalg::scalar T>
130 [[nodiscard]] constexpr result<linalg::fixed_vector<T, 3>, camera_error> ray_plane(const linalg::fixed_vector<T, 3>& c, const linalg::fixed_vector<T, 3>& d,
131 const linalg::fixed_vector<T, 3>& n, T h) noexcept(linalg::detail::nothrow_scalar<T>)
132 {
133 const T nd = linalg::dot(n, d);
134 const T lim = T(1e-12) * linalg::norm(n) * linalg::norm(d);
135 if (!(nd > lim || nd < -lim))
136 return err(camera_error::parallel_to_plane);
137 const T t = (h - linalg::dot(n, c)) / nd;
138 if (!(t > T(0)))
139 return err(camera_error::behind_camera);
140 return ok(c + d * t);
141 }

146 template <linalg::scalar T>
147 [[nodiscard]] constexpr T grazing_angle(const linalg::fixed_vector<T, 3>& d, const linalg::fixed_vector<T, 3>& n) noexcept(linalg::detail::nothrow_scalar<T>)
148 {
149 T s = linalg::dot(n, d) / (linalg::norm(n) * linalg::norm(d));
150 if (s < T(0))
151 s = -s;
152 return static_cast<T>(math::asin(s > T(1) ? T(1) : s));
153 }

178 template <linalg::scalar T>
179 [[nodiscard]] constexpr result<plane_fix_result<T>, camera_error> plane_fix(const pinhole<T>& k, const linalg::fixed_matrix<T, 3, 3>& r, const linalg::fixed_vector<T, 3>& c,
180 const linalg::fixed_vector<T, 2>& u, const linalg::fixed_matrix<T, 2, 2>& pixel_covariance,
181 const linalg::fixed_vector<T, 3>& n, T h, T height_variance = T(0)) noexcept(linalg::detail::nothrow_scalar<T>)
182 {
183 const auto d = pixel_ray(k, r, u);
184 const auto x = ray_plane(c, d, n, h);
185 if (!x)
186 return err(x.unwrap_err_unchecked());
187 const T nd = linalg::dot(n, d);
188 const T t = (h - linalg::dot(n, c)) / nd;
189 // ∂d/∂u = R e_x / fx, ∂d/∂v = R e_y / fy; ∂x/∂u_k = t (∂d_k - (n·∂d_k / n·d) d)
190 linalg::fixed_matrix<T, 3, 2> j{};
191 for (std::size_t q = 0; q < 2; ++q) {
192 const T f = q == 0 ? k.fx : k.fy;
193 linalg::fixed_vector<T, 3> dd{{r(0, q) / f, r(1, q) / f, r(2, q) / f}};
194 const auto col = (dd - d * (linalg::dot(n, dd) / nd)) * t;
195 for (std::size_t i = 0; i < 3; ++i)
196 j(i, q) = col[i];
197 }
198 auto cov = linalg::sandwich(j, pixel_covariance);
199 if (height_variance > T(0)) {
200 const auto dh = d / nd; // ∂x/∂h
201 cov += linalg::outer(dh, dh) * height_variance;
202 }
203 return ok(plane_fix_result<T>{*x, cov, linalg::norm(*x - c), grazing_angle(d, n)});
204 }
