97 // Crouse's shortest augmenting path over the n x m problem (n <= m) read through at(i, j). Working arrays
98 // are spans of at least n (u, col4row, sr) or m (the rest) elements.
99 template <typename T, typename At>
100 constexpr result<T, assignment_error> solve(std::size_t n, std::size_t m, At at, std::span<T> u, std::span<T> v, std::span<T> spc, std::span<std::int32_t> path,
101 std::span<std::int32_t> col4row, std::span<std::int32_t> row4col, std::span<std::int32_t> remaining,
102 std::span<std::uint8_t> sr, std::span<std::uint8_t> sc) noexcept(linalg::detail::nothrow_scalar<T>)
103 {
104 const T inf = std::numeric_limits<T>::infinity();
105 for (std::size_t i = 0; i < n; ++i) {
106 u[i] = T(0);
107 col4row[i] = -1;
108 }
109 for (std::size_t j = 0; j < m; ++j) {
110 v[j] = T(0);
111 row4col[j] = -1;
112 path[j] = -1;
113 }
114 for (std::size_t i = 0; i < n; ++i)
115 for (std::size_t j = 0; j < m; ++j) {
116 const T c = at(i, j);
117 if (c != c || c == -inf)
118 return err(assignment_error::invalid_cost);
119 }
120 for (std::size_t cur = 0; cur < n; ++cur) {
121 T min_val(0);
122 auto i = static_cast<std::int32_t>(cur);
123 std::size_t left = m;
124 for (std::size_t j = 0; j < m; ++j) {
125 remaining[j] = static_cast<std::int32_t>(m - j - 1);
126 spc[j] = inf;
127 sc[j] = 0;
128 }
129 for (std::size_t r = 0; r < n; ++r)
130 sr[r] = 0;
131 std::int32_t sink = -1;
132 while (sink == -1) {
133 std::size_t index = m;
134 T lowest = inf;
135 sr[static_cast<std::size_t>(i)] = 1;
136 for (std::size_t it = 0; it < left; ++it) {
137 const auto j = static_cast<std::size_t>(remaining[it]);
138 const T r = min_val + at(static_cast<std::size_t>(i), j) - u[static_cast<std::size_t>(i)] - v[j];
139 if (r < spc[j]) {
140 path[j] = i;
141 spc[j] = r;
142 }
143 if (spc[j] < lowest || (spc[j] == lowest && row4col[j] == -1)) {
144 lowest = spc[j];
145 index = it;
146 }
147 }
148 min_val = lowest;
149 if (min_val == inf || index == m)
150 return err(assignment_error::infeasible);
151 const auto j = static_cast<std::size_t>(remaining[index]);
152 if (row4col[j] == -1)
153 sink = static_cast<std::int32_t>(j);
154 else
155 i = row4col[j];
156 sc[j] = 1;
157 remaining[index] = remaining[--left];
158 }
159 u[cur] += min_val;
160 for (std::size_t r = 0; r < n; ++r)
161 if (sr[r] && r != cur)
162 u[r] += min_val - spc[static_cast<std::size_t>(col4row[r])];
163 for (std::size_t j = 0; j < m; ++j)
164 if (sc[j])
165 v[j] -= min_val - spc[j];
166 auto j = sink;
167 for (;;) {
168 const std::int32_t r = path[static_cast<std::size_t>(j)];
169 row4col[static_cast<std::size_t>(j)] = r;
170 const std::int32_t t = col4row[static_cast<std::size_t>(r)];
171 col4row[static_cast<std::size_t>(r)] = j;
172 j = t;
173 if (r == static_cast<std::int32_t>(cur))
174 break;
175 }
176 }
177 T total(0);
178 for (std::size_t r = 0; r < n; ++r)
179 total += at(r, static_cast<std::size_t>(col4row[r]));
180 return ok(total);
181 }

189 template <linalg::scalar T, std::size_t MaxRows, std::size_t MaxCols>
190 class assignment_solver {
191 static constexpr std::size_t lo = MaxRows < MaxCols ? MaxRows : MaxCols;
192 static constexpr std::size_t hi = MaxRows < MaxCols ? MaxCols : MaxRows;
193
194 public:
202 [[nodiscard]] constexpr result<T, assignment_error> solve(std::span<const T> cost, std::size_t rows, std::size_t cols, std::span<std::int32_t> row_to_col,
203 objective goal = objective::minimise) noexcept(linalg::detail::nothrow_scalar<T>)
204 {
205 if (rows > MaxRows || cols > MaxCols || cost.size() < rows * cols || row_to_col.size() < rows)
206 return err(assignment_error::too_large);
207 for (std::size_t r = 0; r < rows; ++r)
208 row_to_col[r] = -1;
209 if (rows == 0 || cols == 0)
210 return ok(T(0));
211 const T sign = goal == objective::maximise ? T(-1) : T(1);
212 const bool transposed = rows > cols;
213 const std::size_t n = transposed ? cols : rows, m = transposed ? rows : cols;
214 auto at = [&](std::size_t i, std::size_t j) { return sign * (transposed ? cost[j * cols + i] : cost[i * cols + j]); };
215 const auto s = detail::assignment::solve<T>(n, m, at, std::span<T>(u_).first(n), std::span<T>(v_).first(m), std::span<T>(spc_).first(m),
216 std::span<std::int32_t>(path_).first(m), std::span<std::int32_t>(col4row_).first(n),
217 std::span<std::int32_t>(row4col_).first(m), std::span<std::int32_t>(remaining_).first(m),
218 std::span<std::uint8_t>(sr_).first(n), std::span<std::uint8_t>(sc_).first(m));
219 if (!s)
220 return err(s.unwrap_err_unchecked());
221 if (transposed) {
222 for (std::size_t j = 0; j < m; ++j)
223 if (row4col_[j] >= 0)
224 row_to_col[j] = row4col_[j];
225 // row4col_ maps the transposed problem's columns (our rows) to its rows (our columns)
226 } else {
227 for (std::size_t r = 0; r < n; ++r)
228 row_to_col[r] = col4row_[r];
229 }
230 return ok(sign * *s);
231 }
232
233 private:
234 std::array<T, lo> u_{};
235 std::array<T, hi> v_{}, spc_{};
236 std::array<std::int32_t, hi> path_{}, row4col_{}, remaining_{};
237 std::array<std::int32_t, lo> col4row_{};
238 std::array<std::uint8_t, lo> sr_{};
239 std::array<std::uint8_t, hi> sc_{};
240 };

250 template <linalg::scalar T, std::size_t R, std::size_t C>
251 [[nodiscard]] constexpr result<assignment<T, R>, assignment_error> linear_assignment(const linalg::fixed_matrix<T, R, C>& cost, objective goal = objective::minimise) noexcept(linalg::detail::nothrow_scalar<T>)
252 {
253 assignment_solver<T, R, C> solver;
254 assignment<T, R> a{};
255 const auto s = solver.solve(std::span<const T>(cost.elements), R, C, std::span<std::int32_t>(a.row_to_col), goal);
256 if (!s)
257 return err(s.unwrap_err_unchecked());
258 a.total = *s;
259 return ok(a);
260 }
