69template <hash_function H>
70class hmac : public detail::hash_updates<hmac<H>>
71{
72public:
74 using hash_type = H;
76 static constexpr std::size_t block_size = H::block_size;
78 static constexpr std::size_t digest_size = H::digest_size;
80 using digest_type = digest<digest_size>;
81
84 template <hash_input K>
85 constexpr explicit hmac(const K& key) noexcept
86 {
87 std::array<std::uint8_t, block_size> pad{};
88 const std::size_t n = detail::input_size(key);
89 if (n > block_size) {
90 digest_type hashed = H::hash(key);
91 for (std::size_t i = 0; i < digest_size; ++i)
92 pad[i] = static_cast<std::uint8_t>(hashed.bytes[i]);
93 secure_wipe(hashed);
94 }
95 else {
96 const auto* k = detail::input_data(key);
97 for (std::size_t i = 0; i < n; ++i)
98 pad[i] = detail::octet_of(k[i]);
99 }
100 for (auto& b : pad)
101 b ^= 0x36;
102 inner_.update(pad);
103 for (auto& b : pad)
104 b ^= 0x36 ^ 0x5c;
105 outer_.update(pad);
106 secure_wipe(std::span<std::uint8_t, block_size>(pad));
107 }
108
111 [[nodiscard]] constexpr digest_type finish() noexcept
112 {
113 digest_type inner = inner_.finish();
114 outer_.update(inner);
115 return outer_.finish();
116 }

122 template <hash_input K, hash_input S>
123 [[nodiscard]] static constexpr digest_type mac(const K& key, const S& message) noexcept
124 {
125 hmac m{key};
126 m.update(message);

128 }

135 template <hash_input K, hash_input S, hash_input T>
136 [[nodiscard]] static constexpr bool verify(const K& key, const S& message, const T& tag) noexcept
137 {
138 return constant_time_equal(mac(key, message), tag);
139 }

145 {
146 inner_.update(std::span<const U>(p, n));
147 }
