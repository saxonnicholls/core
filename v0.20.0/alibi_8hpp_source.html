80template <math::math_number T>
81[[nodiscard]] constexpr T alibi_slope(std::size_t h, std::size_t heads, T max_bias = T(8))
82{
83 assert(heads > 0 && h < heads && max_bias >= T(0));
84 if (max_bias == T(0))
85 return T(1);
86 std::size_t n = 1;
87 while (2 * n <= heads)
88 n *= 2; // 2^floor(log2 heads)
89 const T m0 = math::pow(T(2), -max_bias / T(n)), m1 = math::pow(T(2), -(max_bias / T(2)) / T(n));
90 return h < n ? math::pow(m0, T(h + 1)) : math::pow(m1, T(2 * (h - n) + 1));
91}

98template <math::math_number T, std::size_t Heads>
99 requires(Heads > 0)
100[[nodiscard]] constexpr std::array<T, Heads> alibi_slopes(T max_bias = T(8))
101{
102 std::array<T, Heads> out{};
103 for (std::size_t h = 0; h < Heads; ++h)
104 out[h] = alibi_slope<T>(h, Heads, max_bias);
105 return out;
106}

114template <math::math_number T>
115[[nodiscard]] constexpr T alibi_bias(T slope, std::int64_t query_pos, std::int64_t key_pos) noexcept
116{
117 const std::int64_t d = query_pos - key_pos;
118 return slope * -T(d < 0 ? -d : d);
119}

132template <row Q, row K, row V, mutable_row Out>
133 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
134 && std::same_as<row_value_t<Q>, row_value_t<Out>>
135constexpr row_value_t<Q> attend_alibi(const Q& q, const K& keys, const V& values, Out&& out, std::type_identity_t<row_value_t<Q>> scale,
136 std::type_identity_t<row_value_t<Q>> slope, std::int64_t query_pos, std::int64_t first_key_pos)
137{
138 using T = row_value_t<Q>;
139 const auto qs = as_row(q);
140 const auto ks = as_row(keys);
141 const auto vs = as_row(values);
142 const auto y = as_mutable_row(out);
143 const std::size_t d = qs.size();
144 assert(y.size() == d && ks.size() == vs.size() && (d == 0 || ks.size() % d == 0));
145 std::fill(y.begin(), y.end(), T(0));
146 T max = -std::numeric_limits<T>::infinity(), sum = T(0);
147 for (std::size_t j = 0; d != 0 && j < ks.size() / d; ++j) {
148 const T s = scale * detail::attention::dot(qs, ks.subspan(j * d, d));
149 detail::attention::fold(max, sum, y, s + alibi_bias(slope, query_pos, first_key_pos + std::int64_t(j)), vs.subspan(j * d, d));
150 }
151 for (T& v : y)
152 v = sum == T(0) ? T(0) : v / sum;
153 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
154}

171template <row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK, mutable_row DV>
172 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
173 && std::same_as<row_value_t<Q>, row_value_t<O>> && std::same_as<row_value_t<Q>, row_value_t<DO>>
174 && std::same_as<row_value_t<Q>, row_value_t<DQ>> && std::same_as<row_value_t<Q>, row_value_t<DK>>
175 && std::same_as<row_value_t<Q>, row_value_t<DV>>
176constexpr void attend_alibi_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
177 std::type_identity_t<row_value_t<Q>> lse, std::type_identity_t<row_value_t<Q>> scale,
178 std::type_identity_t<row_value_t<Q>> slope, std::int64_t query_pos, std::int64_t first_key_pos,
179 DQ&& dq, DK&& dkeys, DV&& dvalues)
180{
181 using T = row_value_t<Q>;
182 const auto qs = as_row(q);
183 const auto ks = as_row(keys);
184 const auto vs = as_row(values);
185 const auto o = as_row(out);
186 const auto g = as_row(dout);
187 const auto gq = as_mutable_row(dq);
188 const auto gk = as_mutable_row(dkeys);
189 const auto gv = as_mutable_row(dvalues);
190 const std::size_t d = qs.size();
191 assert(o.size() == d && g.size() == d && gq.size() == d && ks.size() == vs.size() && gk.size() == ks.size()
192 && gv.size() == vs.size() && (d == 0 || ks.size() % d == 0));
193 std::fill(gq.begin(), gq.end(), T(0));
194 if (d == 0 || lse == -std::numeric_limits<T>::infinity())
195 return;
196 const T big_d = detail::attention::dot(g, o);
197 for (std::size_t j = 0; j < ks.size() / d; ++j) {
198 const auto k = ks.subspan(j * d, d);
199 const auto v = vs.subspan(j * d, d);
200 const T s = scale * detail::attention::dot(qs, k) + alibi_bias(slope, query_pos, first_key_pos + std::int64_t(j));
201 const T p = math::exp(s - lse);
202 const T ds = p * (detail::attention::dot(g, v) - big_d);
203 for (std::size_t i = 0; i < d; ++i) {
204 gv[j * d + i] += p * g[i];
205 gq[i] += scale * ds * k[i];
206 gk[j * d + i] += scale * ds * qs[i];
207 }
208 }
209}
