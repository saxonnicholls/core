145 template <full_range_generator G>
146 [[nodiscard]] constexpr std::uint64_t uniform_index(G& g, std::uint64_t n) noexcept(noexcept(g()))
147 {
148 assert(n > 0);
149 snicholls::detail::wide_product m = snicholls::detail::umul128(random_u64(g), n);
150 if (m.low < n) {
151 const std::uint64_t threshold = (std::uint64_t{0} - n) % n;
152 while (m.low < threshold)
153 m = snicholls::detail::umul128(random_u64(g), n);
154 }
155 return m.high;
156 }

168 template <typename Refill>
169 [[nodiscard]] constexpr std::uint32_t uniform_index32(std::uint32_t bits, std::uint32_t n, Refill&& refill) noexcept(noexcept(refill()))
170 {
171 assert(n > 0);
172 std::uint64_t m = static_cast<std::uint64_t>(bits) * n;
173 auto low = static_cast<std::uint32_t>(m);
174 if (low < n) {
175 const std::uint32_t threshold = (std::uint32_t{0} - n) % n;
176 while (low < threshold) {
177 m = static_cast<std::uint64_t>(static_cast<std::uint32_t>(refill())) * n;
178 low = static_cast<std::uint32_t>(m);
179 }
180 }
181 return static_cast<std::uint32_t>(m >> 32);
182 }
