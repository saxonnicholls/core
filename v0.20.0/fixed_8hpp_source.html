89 template <scalar T, std::size_t R, std::size_t C>
90 struct fixed_matrix {
91 static_assert(R > 0 && C > 0, "snicholls::linalg::fixed_matrix: a matrix has at least one row and one column");
92
93 using value_type = T;
94 static constexpr std::size_t rows = R;
95 static constexpr std::size_t cols = C;
96 static constexpr std::size_t size = R * C;
97
98 std::array<T, R * C> elements{};
99
101 [[nodiscard]] constexpr T& operator()(std::size_t i, std::size_t j) noexcept { return elements[i * C + j]; }
103 [[nodiscard]] constexpr const T& operator()(std::size_t i, std::size_t j) const noexcept { return elements[i * C + j]; }
105 [[nodiscard]] constexpr T& operator[](std::size_t i) noexcept
106 requires(R == 1 || C == 1)
107 {
108 return elements[i];
109 }
110
111 [[nodiscard]] constexpr const T& operator[](std::size_t i) const noexcept
112 requires(R == 1 || C == 1)
113 {
114 return elements[i];
115 }
116
118 [[nodiscard]] static constexpr fixed_matrix zero() noexcept(detail::nothrow_scalar<T>) { return fixed_matrix{}; }
120 [[nodiscard]] static constexpr fixed_matrix identity() noexcept(detail::nothrow_scalar<T>)
121 requires(R == C)
122 {
123 fixed_matrix m{};

458 template <scalar T, std::size_t N>
459 [[nodiscard]] constexpr result<fixed_matrix<T, N, N>, error> cholesky(const fixed_matrix<T, N, N>& a) noexcept(detail::fixed::nt<T>)
460 {
461 T scale(0);
462 for (std::size_t i = 0; i < N; ++i)
463 if (detail::fixed::abs(a(i, i)) > scale)
464 scale = detail::fixed::abs(a(i, i));
465 const T tol = pivot_tolerance<T>(N, scale);
466 // The factor is built inside the result and the result is the only object returned, so it is never copied
467 result<fixed_matrix<T, N, N>, error> out{std::in_place};
468 fixed_matrix<T, N, N>& l = *out;
469 for (std::size_t j = 0; j < N; ++j) {
470 T d = a(j, j);
471 for (std::size_t k = 0; k < j; ++k)
472 d -= l(j, k) * l(j, k);
473 if (!(d > tol)) {
474 out = err(error::not_positive_definite);
475 return out;
476 }
477 const T ljj = static_cast<T>(math::sqrt(d));
478 l(j, j) = ljj;
479 for (std::size_t i = j + 1; i < N; ++i) {
480 T s = a(i, j);
481 for (std::size_t k = 0; k < j; ++k)
482 s -= l(i, k) * l(j, k);
483 l(i, j) = s / ljj;
484 }
485 }
486 return out;
487 }

493 template <scalar T, std::size_t N, std::size_t C>
494 [[nodiscard]] constexpr fixed_matrix<T, N, C> lower_solve(const fixed_matrix<T, N, N>& l, const fixed_matrix<T, N, C>& b) noexcept(detail::fixed::nt<T>)
495 {

560 template <typename T, std::size_t N>
561 struct lu {
562 fixed_matrix<T, N, N> f{};
563 std::array<std::size_t, N> perm{};
564 bool odd = false;
565 bool singular = false;
566 };
567
568 // Doolittle with partial pivoting: f holds L (unit, below) and U (on and above); perm[i] is the row of A in row i
569 template <typename T, std::size_t N>
570 constexpr lu<T, N> lu_factor(const fixed_matrix<T, N, N>& a) noexcept(nt<T>)
571 {
572 lu<T, N> r{a, {}, false, false};
573 T scale(0);
574 for (const auto& e : a.elements)
575 if (abs(e) > scale)
576 scale = abs(e);
577 const T tol = pivot_tolerance<T>(N, scale);
578 for (std::size_t i = 0; i < N; ++i)
579 r.perm[i] = i;
580 for (std::size_t k = 0; k < N; ++k) {
581 std::size_t p = k;
582 for (std::size_t i = k + 1; i < N; ++i)
583 if (abs(r.f(i, k)) > abs(r.f(p, k)))
584 p = i;
585 if (!(abs(r.f(p, k)) > tol)) {
586 r.singular = true;
587 return r;
588 }
589 if (p != k) {
590 for (std::size_t j = 0; j < N; ++j) {
591 const T t = r.f(k, j);
592 r.f(k, j) = r.f(p, j);
593 r.f(p, j) = t;
594 }
595 const std::size_t t = r.perm[k];
596 r.perm[k] = r.perm[p];
597 r.perm[p] = t;
598 r.odd = !r.odd;
599 }
600 for (std::size_t i = k + 1; i < N; ++i) {
601 const T m = r.f(i, k) / r.f(k, k);
602 r.f(i, k) = m;
603 for (std::size_t j = k + 1; j < N; ++j)
604 r.f(i, j) -= m * r.f(k, j);
605 }
606 }
607 return r;
608 }

615 template <scalar T, std::size_t N, std::size_t C>
616 [[nodiscard]] constexpr result<fixed_matrix<T, N, C>, error> solve(const fixed_matrix<T, N, N>& a, const fixed_matrix<T, N, C>& b) noexcept(detail::fixed::nt<T>)
617 {
618 const auto lu = detail::fixed::lu_factor(a);
619 if (lu.singular)
620 return err(error::singular);
621 fixed_matrix<T, N, C> x{};
622 for (std::size_t j = 0; j < C; ++j) {
623 for (std::size_t i = 0; i < N; ++i) {
624 T s = b(lu.perm[i], j);
625 for (std::size_t k = 0; k < i; ++k)
626 s -= lu.f(i, k) * x(k, j);
627 x(i, j) = s;
628 }
629 for (std::size_t ii = N; ii-- > 0;) {
630 T s = x(ii, j);
631 for (std::size_t k = ii + 1; k < N; ++k)
632 s -= lu.f(ii, k) * x(k, j);
633 x(ii, j) = s / lu.f(ii, ii);
634 }
635 }
636 return ok(x);
637 }

651 template <scalar T, std::size_t N>
652 [[nodiscard]] constexpr T determinant(const fixed_matrix<T, N, N>& a) noexcept(detail::fixed::nt<T>)
653 {
654 const auto lu = detail::fixed::lu_factor(a);
655 if (lu.singular)
656 return T(0);

679 template <scalar T, std::size_t N>
680 [[nodiscard]] constexpr result<eigen_decomposition<T, N>, error> symmetric_eigen(const fixed_matrix<T, N, N>& a) noexcept(detail::fixed::nt<T>)
681 {
682 fixed_matrix<T, N, N> m = a;
683 for (std::size_t i = 0; i < N; ++i)
684 for (std::size_t j = 0; j < i; ++j)
685 m(i, j) = m(j, i);
686 auto v = fixed_matrix<T, N, N>::identity();
687 const T eps = std::numeric_limits<T>::epsilon();
688 bool converged = false;
689 for (int sweep = 0; sweep < 64 && !converged; ++sweep) {
690 T off(0), diag(0);
691 for (std::size_t i = 0; i < N; ++i) {
692 diag += m(i, i) * m(i, i);
693 for (std::size_t j = i + 1; j < N; ++j)
694 off += m(i, j) * m(i, j);
695 }
696 if (off <= eps * eps * diag || off == T(0)) {
697 converged = true;
698 break;
699 }
700 for (std::size_t p = 0; p < N; ++p)
701 for (std::size_t q = p + 1; q < N; ++q) {
702 const T apq = m(p, q);
703 if (apq == T(0))
704 continue;
705 const T theta = (m(q, q) - m(p, p)) / (T(2) * apq);
706 const T at = detail::fixed::abs(theta);
707 // t = tan of the rotation, the smaller root; for a huge theta, 1 / (2 theta) without squaring it
708 T t = at > T(1e150) ? T(1) / (T(2) * at) : T(1) / (at + static_cast<T>(math::sqrt(theta * theta + T(1))));
709 if (theta < T(0))
710 t = -t;
711 const T c = T(1) / static_cast<T>(math::sqrt(t * t + T(1)));
712 const T s = t * c;
713 for (std::size_t k = 0; k < N; ++k) {
714 const T mkp = m(k, p), mkq = m(k, q);
715 m(k, p) = c * mkp - s * mkq;
716 m(k, q) = s * mkp + c * mkq;
717 }
718 for (std::size_t k = 0; k < N; ++k) {
719 const T mpk = m(p, k), mqk = m(q, k);
720 m(p, k) = c * mpk - s * mqk;
721 m(q, k) = s * mpk + c * mqk;
722 }
723 for (std::size_t k = 0; k < N; ++k) {
724 const T vkp = v(k, p), vkq = v(k, q);
725 v(k, p) = c * vkp - s * vkq;
726 v(k, q) = s * vkp + c * vkq;
727 }
728 }
729 }
730 if (!converged)
731 return err(error::did_not_converge);
732 eigen_decomposition<T, N> r{diagonal_of(m), v};
733 // Insertion sort, ascending, moving each eigenvector with its value
734 for (std::size_t i = 1; i < N; ++i)
735 for (std::size_t j = i; j > 0 && r.values[j] < r.values[j - 1]; --j) {
736 const T t = r.values[j];
737 r.values[j] = r.values[j - 1];
738 r.values[j - 1] = t;
739 for (std::size_t k = 0; k < N; ++k) {
740 const T u = r.vectors(k, j);
741 r.vectors(k, j) = r.vectors(k, j - 1);
742 r.vectors(k, j - 1) = u;
743 }
744 }
745 return ok(r);
746 }

756 template <scalar T, std::size_t N, std::size_t K>
757 [[nodiscard]] constexpr fixed_matrix<T, N, N> tria(const fixed_matrix<T, N, K>& a) noexcept(detail::fixed::nt<T>)
758 {
759 static_assert(K >= N, "snicholls::linalg::tria: A needs at least as many columns as rows");
760 fixed_matrix<T, K, N> m = transpose(a);
761 for (std::size_t j = 0; j < N; ++j) {
762 T sigma(0);
763 for (std::size_t i = j; i < K; ++i)
764 sigma += m(i, j) * m(i, j);
765 if (sigma == T(0))
766 continue;
767 const T nrm = static_cast<T>(math::sqrt(sigma));
768 const T alpha = m(j, j) > T(0) ? -nrm : nrm;
769 // v = x - alpha e1, stored in a copy; H = I - 2 v vᵀ / (vᵀ v)
770 std::array<T, K> v{};
771 for (std::size_t i = j; i < K; ++i)
772 v[i] = m(i, j);
773 v[j] -= alpha;
774 T vv(0);
775 for (std::size_t i = j; i < K; ++i)
776 vv += v[i] * v[i];
777 if (vv == T(0))
778 continue;
779 for (std::size_t c = j; c < N; ++c) {
780 T s(0);
781 for (std::size_t i = j; i < K; ++i)
782 s += v[i] * m(i, c);
783 s = T(2) * s / vv;
784 for (std::size_t i = j; i < K; ++i)
785 m(i, c) -= s * v[i];
786 }
787 }
788 fixed_matrix<T, N, N> s{};
789 for (std::size_t i = 0; i < N; ++i)
790 for (std::size_t j = 0; j <= i; ++j)
791 s(i, j) = m(j, i);
792 for (std::size_t j = 0; j < N; ++j)
793 if (s(j, j) < T(0))
794 for (std::size_t i = j; i < N; ++i)
795 s(i, j) = -s(i, j);
796 return s;
797 }
