85template <typename B, math::math_number T>
86constexpr void mean_pool_rows(const B&, std::type_identity_t<std::span<const T>> rows, std::size_t width, std::span<T> out,
87 std::size_t sequences = 1, std::span<const std::size_t> lengths = {})
88{
89 assert(width > 0 && sequences > 0 && rows.size() % (width * sequences) == 0 && out.size() == sequences * width
90 && (lengths.empty() || lengths.size() == sequences));
91 const std::size_t n = rows.size() / width / sequences;
92 for (std::size_t s = 0; s < sequences; ++s) {
93 const std::size_t len = detail::probe::valid_rows(lengths, s, n);
94 assert(len <= n);
95 for (std::size_t c = 0; c < width; ++c)
96 out[s * width + c] = T(0);
97 for (std::size_t r = 0; r < len; ++r)
98 for (std::size_t c = 0; c < width; ++c)
99 out[s * width + c] += rows[(s * n + r) * width + c];
100 for (std::size_t c = 0; c < width; ++c)
101 out[s * width + c] = len == 0 ? T(0) : out[s * width + c] / T(len);
102 }
103}

187template <typename B, math::math_number T>
188 requires linalg::blas_backend<B, T>
189constexpr void attentive_probe_forward(const B& backend, const attentive_probe_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
190 const attentive_probe_activations<T>& act, std::type_identity_t<std::span<T>> out, std::size_t heads,
191 std::size_t sequences = 1, std::span<const std::size_t> lengths = {})
192{
193 const std::size_t d = w.query.size(), n = d == 0 || sequences == 0 ? 0 : xs.size() / d / sequences;
194 assert(d > 0 && heads > 0 && d % heads == 0 && xs.size() == sequences * n * d && out.size() == sequences * d
195 && act.h.size() == xs.size() && act.k.size() == xs.size() && act.v.size() == xs.size() && act.q.size() == d
196 && act.y.size() == sequences * d && act.lse.size() == sequences * heads && (lengths.empty() || lengths.size() == sequences));
197 for (std::size_t r = 0; r < sequences * n; ++r)
198 layer_norm(xs.subspan(r * d, d), w.norm_weight, w.norm_bias, act.h.subspan(r * d, d), w.norm_eps);
199 linear_sequence(backend, w.q, w.query, act.q, w.q_bias);
200 linear_sequence(backend, w.k, std::span<const T>(act.h), act.k, w.k_bias);
201 linear_sequence(backend, w.v, std::span<const T>(act.h), act.v, w.v_bias);
202 const std::size_t dh = d / heads;
203 const T scale = T(1) / math::sqrt(T(dh));
204 for (std::size_t s = 0; s < sequences; ++s) {
205 const std::size_t len = detail::probe::valid_rows(lengths, s, n);
206 for (std::size_t h = 0; h < heads; ++h) {
207 const auto qh = std::span<const T>(act.q).subspan(h * dh, dh);
208 const auto yh = act.y.subspan(s * d + h * dh, dh);
209 T max = -std::numeric_limits<T>::infinity();
210 for (std::size_t j = 0; j < len; ++j) {
211 T sc = T(0);
212 for (std::size_t i = 0; i < dh; ++i)
213 sc += qh[i] * act.k[(s * n + j) * d + h * dh + i];
214 max = scale * sc > max ? scale * sc : max;
215 }
216 for (T& v : yh)
217 v = T(0);
218 T sum = T(0);
219 for (std::size_t j = 0; j < len; ++j) {
220 T sc = T(0);
221 for (std::size_t i = 0; i < dh; ++i)
222 sc += qh[i] * act.k[(s * n + j) * d + h * dh + i];
223 const T e = math::exp(scale * sc - max);
224 sum += e;
225 for (std::size_t i = 0; i < dh; ++i)
226 yh[i] += e * act.v[(s * n + j) * d + h * dh + i];
227 }
228 for (T& v : yh)
229 v = sum == T(0) ? T(0) : v / sum;
230 act.lse[s * heads + h] = sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
231 }
232 for (std::size_t c = 0; c < d; ++c)
233 out[s * d + c] = w.query[c] + act.y[s * d + c];
234 }
235}

249template <typename B, math::math_number T>
250 requires linalg::blas_backend<B, T>
251constexpr void attentive_probe_backward(const B& backend, const attentive_probe_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
252 const attentive_probe_activations<T>& act, std::type_identity_t<std::span<const T>> dout,
253 std::type_identity_t<std::span<T>> dxs, const attentive_probe_gradients<T>& grads, std::size_t heads,
254 std::size_t sequences = 1, std::span<const std::size_t> lengths = {})
255{
256 const std::size_t d = w.query.size(), n = d == 0 || sequences == 0 ? 0 : xs.size() / d / sequences;
257 assert(dout.size() == sequences * d && (dxs.empty() || dxs.size() == xs.size()) && grads.dq.size() == d
258 && grads.dk.size() == xs.size() && grads.dv.size() == xs.size() && grads.dh.size() == xs.size() && grads.query.size() == d);
259 const std::size_t dh = d / heads;
260 const T scale = T(1) / math::sqrt(T(dh));
261 for (T& v : grads.dq)
262 v = T(0);
263 for (T& v : grads.dk)
264 v = T(0);
265 for (T& v : grads.dv)
266 v = T(0);
267 for (T& v : grads.dh)
268 v = T(0);
269 for (std::size_t s = 0; s < sequences; ++s) {
270 for (std::size_t c = 0; c < d; ++c)
271 grads.query[c] += dout[s * d + c]; // the residual
272 const std::size_t len = detail::probe::valid_rows(lengths, s, n);
273 for (std::size_t h = 0; h < heads; ++h) {
274 const T lse = act.lse[s * heads + h];
275 if (lse == -std::numeric_limits<T>::infinity())
276 continue;
277 const auto qh = std::span<const T>(act.q).subspan(h * dh, dh);
278 const auto g = dout.subspan(s * d + h * dh, dh);
279 T big = T(0);
280 for (std::size_t i = 0; i < dh; ++i)
281 big += g[i] * act.y[s * d + h * dh + i];
282 for (std::size_t j = 0; j < len; ++j) {
283 const std::size_t at = (s * n + j) * d + h * dh;
284 T sc = T(0), gv = T(0);
285 for (std::size_t i = 0; i < dh; ++i) {
286 sc += qh[i] * act.k[at + i];
287 gv += g[i] * act.v[at + i];
288 }
289 const T p = math::exp(scale * sc - lse);
290 const T ds = p * (gv - big);
291 for (std::size_t i = 0; i < dh; ++i) {
292 grads.dv[at + i] += p * g[i];
293 grads.dq[h * dh + i] += scale * ds * act.k[at + i];
294 grads.dk[at + i] += scale * ds * qh[i];
295 }
296 }
297 }
298 }
299 linear_sequence_backward(backend, w.q, w.query, std::span<const T>(grads.dq), grads.query, grads.wq, grads.q_bias);
300 linear_sequence_backward(backend, w.k, std::span<const T>(act.h), std::span<const T>(grads.dk), grads.dh, grads.wk, grads.k_bias);
301 linear_sequence_backward(backend, w.v, std::span<const T>(act.h), std::span<const T>(grads.dv), grads.dh, grads.wv, grads.v_bias);
302 if (dxs.empty() && grads.norm_weight.empty() && grads.norm_bias.empty())
303 return;
304 for (std::size_t r = 0; r < sequences * n; ++r) {
305 const auto x = xs.subspan(r * d, d);
306 const auto stats = layer_norm(x, w.norm_weight, w.norm_bias, grads.dk.subspan(r * d, d), w.norm_eps); // dk is spent
307 const auto dx = grads.dv.subspan(r * d, d); // so is dv
308 layer_norm_backward(std::span<const T>(grads.dh).subspan(r * d, d), x, w.norm_weight, stats, dx, grads.norm_weight, grads.norm_bias);
309 if (!dxs.empty())
310 for (std::size_t c = 0; c < d; ++c)
311 dxs[r * d + c] += dx[c];
312 }
313}
