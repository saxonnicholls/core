85template <math::math_number T>
86struct online_softmax
87{
88 T max = -std::numeric_limits<T>::infinity();
89 T sum = T(0);
90
93 constexpr void add(T x)
94 {
95 if (x == -std::numeric_limits<T>::infinity())
96 return;
97 if (x > max) {
98 sum = sum * math::exp(max - x) + T(1);
99 max = x;
100 } else {
101 sum += math::exp(x - max);
102 }
103 }
104
107 constexpr void merge(const online_softmax& other)
108 {
109 if (other.sum == T(0))
110 return;
111 if (sum == T(0)) {
112 *this = other;
113 return;
114 }
115 if (other.max > max) {
116 sum = sum * math::exp(max - other.max) + other.sum;
117 max = other.max;
118 } else {
119 sum += other.sum * math::exp(other.max - max);
120 }
121 }
122
125 [[nodiscard]] constexpr T log_sum_exp() const
126 {
127 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
128 }
129
133 [[nodiscard]] constexpr T probability(T x) const { return math::exp(x - max) / sum; }
134};

138 template <typename T>
139 constexpr T max_of(std::span<const T> in)
140 {
141 T m = -std::numeric_limits<T>::infinity();
142 for (const T x : in) {
143 if (x != x)
144 return x; // NaN
145 if (x > m)
146 m = x;
147 }
148 return m;
149 }

151 // The maximum and sum e^(x - max), kept apart: x_i - lse is computed as (x_i - max) - log(sum), so it is not
152 // rounded at the magnitude of the largest logit.
153 template <typename T>
154 struct shifted_sum
155 {
156 T max;
157 T sum;
158 };
159
160 template <typename T>
161 constexpr shifted_sum<T> shifted_sum_of(std::span<const T> in)
162 {
163 const T m = max_of(in);
164 if (m == -std::numeric_limits<T>::infinity() || m != m)
165 return {m, T(1)};
166 T s = T(0);
167 for (const T x : in)
168 s += math::exp(x - m);
169 return {m, s};
170 }
171
172 template <typename T>
173 constexpr T logsumexp(std::span<const T> in)
174 {
175 const auto [m, s] = shifted_sum_of(in);
176 return m + math::log(s);
177 }
178} // namespace detail::softmax

195template <row In, mutable_row Out>
196 requires std::same_as<row_value_t<In>, row_value_t<Out>>
197constexpr void softmax(const In& in, Out&& out)
198{
199 using T = row_value_t<In>;
200 const auto x = as_row(in);
201 const auto y = as_mutable_row(out);
202 assert(x.size() == y.size());
203 const T m = detail::softmax::max_of(x);
204 T s = T(0);
205 for (std::size_t i = 0; i < x.size(); ++i) {
206 y[i] = math::exp(x[i] - m);
207 s += y[i];
208 }
209 const T inv = T(1) / s;
210 for (T& p : y)
211 p *= inv;
212}

222template <row In, mutable_row Out>
223 requires std::same_as<row_value_t<In>, row_value_t<Out>>
224constexpr void log_softmax(const In& in, Out&& out)
225{
226 const auto x = as_row(in);
227 const auto y = as_mutable_row(out);
228 assert(x.size() == y.size());
229 const auto [m, s] = detail::softmax::shifted_sum_of(x);
230 const auto log_s = math::log(s);
231 for (std::size_t i = 0; i < x.size(); ++i)
232 y[i] = (x[i] - m) - log_s;
233}

242template <row L, row D, mutable_row Out>
243 requires std::same_as<row_value_t<L>, row_value_t<D>> && std::same_as<row_value_t<L>, row_value_t<Out>>
244constexpr void log_softmax_backward(const L& logprobs, const D& dlogprobs, Out&& dlogits)
245{
246 using T = row_value_t<L>;
247 const auto lp = as_row(logprobs);
248 const auto d = as_row(dlogprobs);
249 const auto g = as_mutable_row(dlogits);
250 assert(lp.size() == d.size() && lp.size() == g.size());
251 T sum = T(0);
252 for (std::size_t i = 0; i < d.size(); ++i)
253 sum += d[i];
254 for (std::size_t i = 0; i < lp.size(); ++i)
255 g[i] = d[i] - math::exp(lp[i]) * sum;
256}

265template <row P, row D, mutable_row Out>
266 requires std::same_as<row_value_t<P>, row_value_t<D>> && std::same_as<row_value_t<P>, row_value_t<Out>>
267constexpr void softmax_backward(const P& probs, const D& dprobs, Out&& dlogits)
268{
269 using T = row_value_t<P>;
270 const auto p = as_row(probs);
271 const auto d = as_row(dprobs);
272 const auto g = as_mutable_row(dlogits);
273 assert(p.size() == d.size() && p.size() == g.size());
274 T dot = T(0);
275 for (std::size_t i = 0; i < p.size(); ++i)
276 dot += p[i] * d[i];
277 for (std::size_t i = 0; i < p.size(); ++i)
278 g[i] = p[i] * (d[i] - dot);
279}

289template <row In>
290constexpr row_value_t<In> cross_entropy(const In& logits, std::size_t target)
291{
292 const auto x = as_row(logits);
293 assert(target < x.size());
294 const auto [m, s] = detail::softmax::shifted_sum_of(x);
295 return (m - x[target]) + math::log(s);
296}

309template <row In, mutable_row Out>
310 requires std::same_as<row_value_t<In>, row_value_t<Out>>
311constexpr row_value_t<In> cross_entropy_with_gradient(const In& logits, std::size_t target, Out&& dlogits,
312 std::type_identity_t<row_value_t<In>> scale = 1)
313{
314 using T = row_value_t<In>;
315 const auto x = as_row(logits);
316 const auto g = as_mutable_row(dlogits);
317 assert(target < x.size() && x.size() == g.size());
318 const auto [m, s] = detail::softmax::shifted_sum_of(x);
319 const T loss = (m - x[target]) + math::log(s);
320 const T inv = T(1) / s;
321 for (std::size_t i = 0; i < x.size(); ++i)
322 g[i] = scale * (math::exp(x[i] - m) * inv - (i == target ? T(1) : T(0)));
323 return loss;
324}

344template <typename B, math::math_number T>
345constexpr void cross_entropy_rows(const B&, std::span<T> logits, std::type_identity_t<std::span<const std::uint32_t>> targets,
346 std::span<T> losses, std::type_identity_t<T> scale)
347{
348 const std::size_t n = targets.size();
349 assert(n > 0 && logits.size() % n == 0 && losses.size() == n);
350 const std::size_t v = logits.size() / n;
351 for (std::size_t t = 0; t < n; ++t) {
352 const auto row = logits.subspan(t * v, v);
353 losses[t] = cross_entropy_with_gradient(row, targets[t], row, scale);
354 }
355}

364constexpr void cross_entropy_rows(const B&, std::span<T> logits, std::type_identity_t<std::span<const std::uint32_t>> targets,
365 std::span<T> losses, std::type_identity_t<T> scale, std::size_t classes)
366{
367 const std::size_t n = targets.size();
368 assert(n > 0 && logits.size() % n == 0 && losses.size() == n && classes > 0 && classes <= logits.size() / n);
369 const std::size_t v = logits.size() / n;
370 for (std::size_t t = 0; t < n; ++t) {
371 const auto row = logits.subspan(t * v, classes);
372 losses[t] = cross_entropy_with_gradient(row, targets[t], row, scale);

374 }
