156template <typename Token>
157constexpr byte_fallback make_byte_level_fallback(Token&& token)
158{
159 byte_fallback f;
160 for (std::size_t b = 0; b < 256; ++b) {
161 char s[4] = {};
162 const std::size_t n = encode_utf8(byte_to_unicode[b], s);
163 f.ids[b] = token(std::string_view(s, n));
164 }
165 return f;
166}

180struct tokenizer_scratch_need
181{
182 std::size_t text = 0;
183 std::size_t ranked = 0;
184 std::size_t scored = 0;
185 std::size_t lattice = 0;
186
190 constexpr tokenizer_scratch_need(const tokenizer_vocab& vocab, std::size_t n) noexcept
191 {
192 switch (vocab.model) {
193 case tokenizer_model::sentencepiece_bpe:
194 text = sentencepiece_escaped_size(n);
195 scored = merge_scratch_size(text);
196 break;
197 case tokenizer_model::bpe: // invalid UTF-8 repaired (3n), then escaped or mapped
198 text = 3 * n + (vocab.escape_whitespaces ? sentencepiece_escaped_size(3 * n) : 6 * n);
199 ranked = merge_scratch_size(vocab.escape_whitespaces ? sentencepiece_escaped_size(3 * n) : 6 * n);
200 break;
201 case tokenizer_model::unigram:
202 text = unigram_normalized_size(n, vocab.charsmap);
203 lattice = text + 1;
204 break;
205 case tokenizer_model::wordpiece:
206 text = wordpiece_scratch_size(n);
207 break;
208 case tokenizer_model::rwkv:
209 break;
210 }
211 }
212};

217 {
218 return e == bpe_error::output_too_small ? tokenizer_error::output_too_small

251 // One run of ordinary text (no added tokens) through the model; first_fragment: whether it opens the text or
252 // follows an added token
253 constexpr result<std::size_t, tokenizer_error> fragment(const tokenizer_vocab& v, std::string_view text, bool first_fragment,
254 std::span<std::uint32_t> out, const tokenizer_scratch& s)
255 {
256 const tokenizer_scratch_need need(v, text.size());
257 if (s.text.size() < need.text || s.ranked.size() < need.ranked || s.scored.size() < need.scored ||
258 s.lattice.size() < need.lattice)
259 return err(tokenizer_error::scratch_too_small);
260 switch (v.model) {
261 case tokenizer_model::sentencepiece_bpe: {
262 const auto e = sentencepiece_escape(text, s.text, v.add_space_prefix && first_fragment);
263 if (e.is_err())
264 return err(from(e.unwrap_err()));
265 const auto r = sentencepiece_bpe_encode(std::string_view(s.text.data(), e.unwrap()), finder{&v.vocab}, scorer{&v}, v.bytes, out, s.scored);
266 if (r.is_err())
267 return err(from(r.unwrap_err()));
268 return ok(r.unwrap());
269 }
270 case tokenizer_model::bpe: {
271 // As llama.cpp's split does, an invalid byte becomes U+FFFD; then spaces are escaped (Gemma 4) or
272 // each piece is written in GPT-2's alphabet, after the repaired text in the scratch
273 std::string_view source = text;
274 std::span<char> rest = s.text;
275 if (!is_valid_utf8(text)) {
276 const std::size_t n = replace_invalid_utf8(text, s.text);
277 source = std::string_view(s.text.data(), n);
278 rest = s.text.subspan(n);
279 }
280 if (v.escape_whitespaces) {
281 const auto e = sentencepiece_escape(source, rest, false);
282 if (e.is_err())
283 return err(from(e.unwrap_err()));
284 source = std::string_view(rest.data(), e.unwrap());
285 rest = rest.subspan(e.unwrap());
286 }
287 std::size_t count = 0;
288 bpe_error failure{};
289 bool failed = false;
290 snicholls::ml::pretokenize(v.pre, source, [&](std::string_view piece) {
291 if (failed)
292 return;
293 merge_options o{merge_units::code_points, v.ignore_merges, v.byte_level};
294 std::string_view units = piece;
295 if (v.byte_level) {
296 const auto m = byte_level_encode(piece, rest);
297 if (m.is_err()) {
298 failed = true;
299 failure = m.unwrap_err();
300 return;
301 }
302 units = std::string_view(rest.data(), m.unwrap());
303 } else if (piece.find_first_not_of('\n') == std::string_view::npos) {
304 o.ignore_merges = true; // Gemma 4: a run of newlines is one token if there is one
305 }
306 const auto r = merge_encode(units, v.merges, finder{&v.vocab}, v.bytes, out.subspan(count), s.ranked, o);
307 if (r.is_err()) {
308 failed = true;
309 failure = r.unwrap_err();
310 return;
311 }
312 count += r.unwrap();
313 });
314 if (failed)
315 return err(from(failure));
316 return ok(count);
317 }
318 case tokenizer_model::unigram: {
319 const auto n = unigram_normalize(text, v.charsmap, v.added, v.normalizer, s.text);
320 if (n.is_err())
321 return err(from(n.unwrap_err()));
322 const auto r = unigram_encode(std::string_view(s.text.data(), n.unwrap()), v.vocab, scorer{&v}, kinder{&v},
323 unigram_unknown{v.unknown, v.unknown_score}, out, s.lattice);
324 if (r.is_err())
325 return err(from(r.unwrap_err()));
326 return ok(r.unwrap());
327 }
328 case tokenizer_model::wordpiece: {
329 const auto r = wordpiece_encode(text, v.vocab, v.unknown, v.wordpiece, out, s.text);
330 if (r.is_err())
331 return err(from(r.unwrap_err()));
332 return ok(r.unwrap());
333 }
334 case tokenizer_model::rwkv: {
335 const auto r = rwkv_encode(text, v.vocab, v.unknown, out);
336 if (r.is_err())
337 return err(from(r.unwrap_err()));
338 return ok(r.unwrap());
339 }
340 }
341 return ok(std::size_t{0});
342 }

351constexpr result<std::size_t, tokenizer_error> tokenize(const tokenizer_vocab& vocab, std::string_view text, std::span<std::uint32_t> out,
352 const tokenizer_scratch& scratch)
353{
354 std::size_t count = 0;
355 bool first = true, failed = false;
356 tokenizer_error failure{};
357 split_added_tokens(
358 text, vocab.added,
359 [&](std::string_view run) {
360 if (failed)
361 return;
362 const auto r = detail::tokenizer::fragment(vocab, run, first, out.subspan(count), scratch);
363 if (r.is_err()) {
364 failed = true;
365 failure = r.unwrap_err();
366 return;
367 }
368 count += r.unwrap();
369 first = false;
370 },
371 [&](std::uint32_t id) {
372 if (failed)
373 return;
374 if (count == out.size()) {
375 failed = true;
376 failure = tokenizer_error::output_too_small;
377 return;
378 }
379 out[count++] = id;
380 first = true;
381 });
382 if (failed)
383 return err(failure);
384 return ok(count);
385}

393constexpr result<std::size_t, tokenizer_error> detokenize(const tokenizer_vocab& vocab, std::span<const std::uint32_t> tokens, std::span<char> out,
394 bool render_special = false)
395{
396 const auto piece = [&](std::uint32_t id) { return vocab.piece(id); };
397 const auto kind = [&](std::uint32_t id) { return vocab.kind(id); };
398 const auto special = [&](std::uint32_t id) {
399 const token_kind k = vocab.kind(id);
400 return !render_special && (k == token_kind::control || k == token_kind::unknown);
401 };
402 switch (vocab.model) {
403 case tokenizer_model::sentencepiece_bpe:
404 case tokenizer_model::unigram: {
405 const auto r = sentencepiece_decode(tokens, piece, kind, out,
406 {vocab.model == tokenizer_model::unigram ? vocab.normalizer.add_dummy_prefix : vocab.add_space_prefix,
407 render_special});
408 if (r.is_err())
409 return err(detail::tokenizer::from(r.unwrap_err()));
410 return ok(r.unwrap());
411 }
412 case tokenizer_model::bpe: {
413 if (!vocab.byte_level) {
414 const auto r = sentencepiece_decode(tokens, piece, kind, out, {false, render_special});
415 if (r.is_err())
416 return err(detail::tokenizer::from(r.unwrap_err()));
417 return ok(r.unwrap());
418 }
419 std::size_t at = 0;
420 for (const std::uint32_t id : tokens) {
421 const std::string_view p = vocab.piece(id);
422 if (p.empty())
423 return err(tokenizer_error::unknown_bytes);
424 if (special(id))
425 continue;
426 const token_kind k = vocab.kind(id);
427 if (k == token_kind::user_defined || k == token_kind::control || k == token_kind::unknown) {
428 if (out.size() - at < p.size())
429 return err(tokenizer_error::output_too_small);
430 for (const char c : p)
431 out[at++] = c;
432 continue;
433 }
434 const auto r = byte_level_decode(p, out.subspan(at));
435 if (r.is_err())
436 return err(detail::tokenizer::from(r.unwrap_err()));
437 at += r.unwrap();
438 }
439 return ok(at);
440 }
441 case tokenizer_model::wordpiece: {
442 const auto r = wordpiece_decode(tokens, piece, special, vocab.wordpiece, out);
443 if (r.is_err())
444 return err(r.unwrap_err() == wordpiece_error::unknown_id ? tokenizer_error::unknown_bytes : tokenizer_error::output_too_small);
445 return ok(r.unwrap());
446 }
447 case tokenizer_model::rwkv: {
448 const auto r = bpe_decode(tokens, piece, out);
449 if (r.is_err())
450 return err(detail::tokenizer::from(r.unwrap_err()));
451 return ok(r.unwrap());
452 }
453 }
454 return ok(std::size_t{0});
455}
