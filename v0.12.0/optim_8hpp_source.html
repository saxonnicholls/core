101template <mutable_row P, row G, mutable_row M, mutable_row V>
102 requires std::same_as<row_value_t<P>, row_value_t<G>> && std::same_as<row_value_t<P>, row_value_t<M>>
103 && std::same_as<row_value_t<P>, row_value_t<V>>
104constexpr void adamw_step(P&& params, const G& grads, M&& m, V&& v, std::uint64_t step, const adamw_config<row_value_t<P>>& config)
105{
106 using T = row_value_t<P>;
107 const auto p = as_mutable_row(params);
108 const auto g = as_row(grads);
109 const auto mm = as_mutable_row(m);
110 const auto vv = as_mutable_row(v);
111 assert(step >= 1 && g.size() == p.size() && mm.size() == p.size() && vv.size() == p.size());
112 const T c1 = T(1) - math::pow(config.beta1, T(step));
113 const T c2 = T(1) - math::pow(config.beta2, T(step));
114 for (std::size_t i = 0; i < p.size(); ++i) {
115 mm[i] = config.beta1 * mm[i] + (T(1) - config.beta1) * g[i];
116 vv[i] = config.beta2 * vv[i] + (T(1) - config.beta2) * g[i] * g[i];
117 const T m_hat = mm[i] / c1;
118 const T v_hat = vv[i] / c2;
119 p[i] -= config.learning_rate * (m_hat / (math::sqrt(v_hat) + config.eps) + config.weight_decay * p[i]);
120 }
121}

130template <math::math_number T>
131constexpr T warmup_cosine(std::uint64_t step, std::uint64_t warmup, std::uint64_t total, T max_lr, std::type_identity_t<T> min_lr)
132{
133 assert(warmup <= total);
134 if (step < warmup)
135 return max_lr * T(step + 1) / T(warmup);
136 if (step >= total)
137 return min_lr;
138 const T ratio = T(step - warmup) / T(total - warmup);
139 const T coefficient = T(0.5) * (T(1) + math::cos(std::numbers::pi_v<T> * ratio));
140 return min_lr + coefficient * (max_lr - min_lr);
141}

150template <math::math_number T>
151constexpr T warmup_warmdown(std::uint64_t step, std::uint64_t total, std::uint64_t warmup, std::uint64_t warmdown, T final_fraction)
152{
153 assert(warmup + warmdown <= total);
154 if (step < warmup)
155 return T(step + 1) / T(warmup);
156 if (step + warmdown <= total)
157 return T(1);
158 const T progress = step >= total ? T(0) : T(total - step) / T(warmdown);
159 return progress + (T(1) - progress) * final_fraction;
160}
