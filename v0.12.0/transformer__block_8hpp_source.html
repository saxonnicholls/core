127template <transformer_config C, row X, mutable_row Out>
128 requires std::same_as<row_value_t<X>, row_value_t<Out>>
129constexpr void block_norm(const X& x, std::span<const row_value_t<X>> weight, std::span<const row_value_t<X>> bias, Out&& out)
130{
131 using T = row_value_t<X>;
132 if constexpr (C.norm == norm_kind::layer)
133 layer_norm(x, weight, bias, out, norm_eps_v<C, T>);
134 else
135 rms_norm(x, weight, out, norm_eps_v<C, T>);
136}

146template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
147 requires std::same_as<row_value_t<X>, T>
148constexpr void block_forward(const block_weights<W, T>& w, X&& x, std::size_t pos, const S& cache, std::size_t layer,
149 const R& rope_src)
150{
151 const auto s = as_mutable_row(x);
152 assert(s.size() == C.d_model);
153 std::array<T, C.d_model> h{}, a{};
154 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
155 attention_forward<C>(w.attention, h, pos, cache, layer, rope_src, a);
156 for (std::size_t i = 0; i < C.d_model; ++i)
157 s[i] += a[i];
158 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
159 mlp_forward<C>(w.mlp, h, a);
160 for (std::size_t i = 0; i < C.d_model; ++i)
161 s[i] += a[i];
162}

172template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
173 requires std::same_as<row_value_t<X>, T>
174constexpr void block_prefill(const block_weights<W, T>& w, X&& xs, std::size_t first, const S& cache, std::size_t layer,
175 const R& rope_src)
176{
177 const auto s = as_mutable_row(xs);
178 assert(s.size() % C.d_model == 0);
179 for (std::size_t t = 0; t < s.size() / C.d_model; ++t)
180 block_forward<C>(w, s.subspan(t * C.d_model, C.d_model), first + t, cache, layer, rope_src);
181}

216 // The backward of block_norm for one row: dx written, the weight's and bias's gradients added into
217 template <transformer_config C, typename T>
218 constexpr void norm_backward(std::span<const T> dh, std::span<const T> x, std::span<const T> weight,
219 std::span<const T> bias, std::span<T> dx, std::span<T> dweight, std::span<T> dbias)
220 {
221 std::array<T, C.d_model> scratch{};
222 if constexpr (C.norm == norm_kind::layer) {
223 const auto stats = layer_norm(x, weight, bias, scratch, norm_eps_v<C, T>);
224 layer_norm_backward(dh, x, weight, stats, dx, dweight, dbias);
225 } else {
226 const T rstd = rms_norm(x, weight, scratch, norm_eps_v<C, T>);
227 rms_norm_backward(dh, x, weight, rstd, dx, dweight);
228 }
229 }

239template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, typename R>
240 requires std::same_as<row_value_t<X>, T> && (C.rope == rope_kind::none || rope_source<R>)
241constexpr void block_forward_sequence(const block_weights<W, T>& w, X&& xs, const R& rope_src, const block_activations<T>& act)
242{
243 constexpr std::size_t d = C.d_model;
244 const auto s = as_mutable_row(xs);
245 const std::size_t n = s.size() / d;
246 assert(s.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d
247 && act.h2.size() == n * d && act.mlp_up.size() == n * C.d_ff
248 && (C.mlp != mlp_kind::swiglu || act.mlp_gate.size() == n * C.d_ff));
249 std::copy(s.begin(), s.end(), act.x0.begin());
250 for (std::size_t t = 0; t < n; ++t)
251 block_norm<C>(s.subspan(t * d, d), w.norm1_weight, w.norm1_bias, act.h1.subspan(t * d, d));
252 attention_forward_sequence<C>(w.attention, std::span<const T>(act.h1), rope_src, act.attention, s);
253 for (std::size_t i = 0; i < n * d; ++i)
254 s[i] = act.x0[i] + s[i];
255 std::copy(s.begin(), s.end(), act.x1.begin());
256 std::array<T, d> a{};
257 for (std::size_t t = 0; t < n; ++t) {
258 block_norm<C>(s.subspan(t * d, d), w.norm2_weight, w.norm2_bias, act.h2.subspan(t * d, d));
259 const mlp_activations<T> save{act.mlp_up.subspan(t * C.d_ff, C.d_ff),
260 act.mlp_gate.empty() ? std::span<T>{} : act.mlp_gate.subspan(t * C.d_ff, C.d_ff)};
261 mlp_forward<C>(w.mlp, std::span<const T>(act.h2).subspan(t * d, d), a, save);
262 for (std::size_t i = 0; i < d; ++i)
263 s[t * d + i] += a[i];
264 }
265}

278template <transformer_config C, typename S, math::math_number T, mutable_row DX, typename R>
279 requires std::same_as<row_value_t<DX>, T> && (C.rope == rope_kind::none || rope_source<R>)
280constexpr void block_backward(const block_weights<dense_weights<S>, T>& w, const R& rope_src,
281 const block_activations<T>& act, DX&& dxs, const block_gradients<T>& grads)
282{
283 constexpr std::size_t d = C.d_model;
284 const auto g = as_mutable_row(dxs);
285 const std::size_t n = g.size() / d;
286 assert(g.size() == n * d && act.x0.size() == n * d && grads.scratch.size() == n * d);
287 std::array<T, d> dh{}, dn{};
288 for (std::size_t t = 0; t < n; ++t) {
289 const auto gt = g.subspan(t * d, d);
290 dh.fill(T(0));
291 const mlp_activations<T> saved{act.mlp_up.subspan(t * C.d_ff, C.d_ff),
292 act.mlp_gate.empty() ? std::span<T>{} : act.mlp_gate.subspan(t * C.d_ff, C.d_ff)};
293 mlp_backward<C>(w.mlp, std::span<const T>(act.h2).subspan(t * d, d), saved, std::span<const T>(gt), dh, grads.mlp);
294 detail::block::norm_backward<C, T>(dh, std::span<const T>(act.x1).subspan(t * d, d), w.norm2_weight, w.norm2_bias, dn,
295 grads.norm2_weight, grads.norm2_bias);
296 for (std::size_t i = 0; i < d; ++i)
297 gt[i] += dn[i];
298 }
299 std::fill(grads.scratch.begin(), grads.scratch.end(), T(0));
300 attention_backward<C>(w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(g),
301 grads.scratch, grads.attention);
302 for (std::size_t t = 0; t < n; ++t) {
303 detail::block::norm_backward<C, T>(std::span<const T>(grads.scratch).subspan(t * d, d),
304 std::span<const T>(act.x0).subspan(t * d, d), w.norm1_weight, w.norm1_bias, dn,
305 grads.norm1_weight, grads.norm1_bias);
306 for (std::size_t i = 0; i < d; ++i)
307 g[t * d + i] += dn[i];
308 }
309}

359template <transformer_config C, typename B, math::math_number T, typename R>
360 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
361constexpr void block_forward_sequence(const B& backend, const block_weights<dense_weights<T>, T>& w, std::type_identity_t<std::span<T>> xs,
362 const R& rope_src, const block_activations<T>& act)
363{
364 constexpr std::size_t d = C.d_model;
365 [[maybe_unused]] const std::size_t n = xs.size() / d;
366 assert(xs.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d && act.h2.size() == n * d);
367 copy_rows(backend, std::span<const T>(xs), act.x0);
368 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm1_weight, w.norm1_bias, act.h1);
369 attention_forward_sequence<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, xs);
370 add_rows(backend, std::span<const T>(act.x0), xs);
371 copy_rows(backend, std::span<const T>(xs), act.x1);
372 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm2_weight, w.norm2_bias, act.h2);
373 mlp_forward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), xs, mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden});
374 add_rows(backend, std::span<const T>(act.x1), xs);
375}

390template <transformer_config C, typename B, math::math_number T, typename R>
391 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
392constexpr void block_backward(const B& backend, const block_weights<dense_weights<T>, T>& w, const R& rope_src,
393 const block_activations<T>& act, std::type_identity_t<std::span<T>> dxs, const block_gradients<T>& grads)
394{
395 constexpr std::size_t d = C.d_model;
396 [[maybe_unused]] const std::size_t n = dxs.size() / d;
397 assert(dxs.size() == n * d && act.x0.size() == n * d && grads.scratch.size() == n * d);
398 fill_rows(backend, grads.scratch, T(0));
399 mlp_backward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden},
400 std::span<const T>(dxs), grads.scratch, grads.mlp);
401 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x1), w.norm2_weight, w.norm2_bias,
402 dxs, grads.norm2_weight, grads.norm2_bias);
403 fill_rows(backend, grads.scratch, T(0));
404 attention_backward<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(dxs),
405 grads.scratch, grads.attention);
406 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x0), w.norm1_weight, w.norm1_bias,
407 dxs, grads.norm1_weight, grads.norm1_bias);
408}
