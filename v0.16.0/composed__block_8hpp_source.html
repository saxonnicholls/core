253template <transformer_config C, typename B, math::math_number T, typename Attention, typename Mlp, typename AttentionActivations,
254 typename MlpActivations, typename R>
255 requires linalg::blas_backend<B, T>
256constexpr moe_losses<T> block_forward_sequence(const B& backend, const composed_block_weights<Attention, Mlp, T>& w,
257 std::type_identity_t<std::span<T>> xs, const R& rope_src,
258 const composed_block_activations<AttentionActivations, MlpActivations, T>& act,
259 std::size_t sequences = 1)
260{
261 assert(xs.size() % C.d_model == 0 && act.x0.size() == xs.size() && act.h1.size() == xs.size() && act.x1.size() == xs.size()
262 && act.h2.size() == xs.size());
263 copy_rows(backend, std::span<const T>(xs), act.x0);
264 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm1_weight, w.norm1_bias, act.h1);
265 detail::composed_block::attention_pass<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, xs, sequences);
266 add_rows(backend, std::span<const T>(act.x0), xs);
267 copy_rows(backend, std::span<const T>(xs), act.x1);
268 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm2_weight, w.norm2_bias, act.h2);
269 const auto losses = detail::composed_block::mlp_pass<C>(backend, w.mlp, std::span<const T>(act.h2), xs, act.mlp, sequences);
270 add_rows(backend, std::span<const T>(act.x1), xs);
271 return losses;
272}

285template <transformer_config C, typename B, math::math_number T, typename Attention, typename Mlp, typename AttentionActivations,
286 typename MlpActivations, typename AttentionGradients, typename MlpGradients, typename R>
287 requires linalg::blas_backend<B, T>
288constexpr void block_backward(const B& backend, const composed_block_weights<Attention, Mlp, T>& w, const R& rope_src,
289 const composed_block_activations<AttentionActivations, MlpActivations, T>& act,
290 std::type_identity_t<std::span<T>> dxs, const composed_block_gradients<AttentionGradients, MlpGradients, T>& grads,
291 std::size_t sequences = 1)
292{
293 assert(dxs.size() == act.x0.size() && grads.scratch.size() == dxs.size());
294 fill_rows(backend, grads.scratch, T(0));
295 detail::composed_block::mlp_pass_backward<C>(backend, w.mlp, std::span<const T>(act.h2), act.mlp, std::span<const T>(dxs), grads.scratch,
296 grads.mlp, sequences);
297 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x1), w.norm2_weight, w.norm2_bias,
298 dxs, grads.norm2_weight, grads.norm2_bias);
299 fill_rows(backend, grads.scratch, T(0));
300 detail::composed_block::attention_pass_backward<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention,
301 std::span<const T>(dxs), grads.scratch, grads.attention, sequences);
302 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x0), w.norm1_weight, w.norm1_bias,
303 dxs, grads.norm1_weight, grads.norm1_bias);
304}

315template <transformer_config C, typename Attention, typename Mlp, math::math_number T, mutable_row X, typename Cache, typename R>
316 requires std::same_as<row_value_t<X>, T>
317constexpr void block_forward(const composed_block_weights<Attention, Mlp, T>& w, X&& x, std::size_t pos, const Cache& cache, std::size_t layer,
318 const R& rope_src)
319{
320 const auto s = as_mutable_row(x);
321 assert(s.size() == C.d_model);
322 std::array<T, C.d_model> h{}, a{};
323 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
324 detail::composed_block::attention_token<C>(w.attention, std::span<const T>(h), pos, cache, layer, rope_src, std::span<T>(a));
325 for (std::size_t i = 0; i < C.d_model; ++i)
326 s[i] += a[i];
327 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
328 mlp_forward<C>(w.mlp, h, a);
329 for (std::size_t i = 0; i < C.d_model; ++i)
330 s[i] += a[i];
331}

381template <transformer_config C, math::math_number T>
382 requires(C.latent() && C.moe())
383constexpr latent_moe_block_weights<T> latent_moe_block_weights_in(std::span<const T> params, std::span<const T> selection_bias = {})
384{
385 assert(params.size() == latent_moe_block_parameter_count(C));
386 constexpr std::size_t d = C.d_model, b = C.norm == norm_kind::layer ? d : 0;
387 detail::cutter<const T> c{params};
388 latent_moe_block_weights<T> w{};
389 w.norm1_weight = c.take(d);
390 w.norm1_bias = c.take(b);
391 w.attention = latent_attention_weights_in<C>(c.take(latent_attention_parameter_count(C)));
392 w.norm2_weight = c.take(d);
393 w.norm2_bias = c.take(b);
394 w.mlp = moe_weights_in<C>(c.take(moe_parameter_count(C)), selection_bias);
395 return w;
396}

405template <transformer_config C, math::math_number T>
406 requires(C.latent() && C.moe())
407constexpr latent_moe_block_activations<T> latent_moe_block_activations_in(std::span<T> values, std::span<std::uint32_t> indices,
408 std::size_t rows, std::span<const std::uint32_t> selected = {})
409{
410 assert(values.size() == latent_moe_block_activation_values(C, rows) && indices.size() == latent_moe_block_activation_indices(C, rows));
411 detail::cutter<T> v{values};
412 latent_moe_block_activations<T> a{};
413 a.x0 = v.take(rows * C.d_model);
414 a.h1 = v.take(rows * C.d_model);
415 a.x1 = v.take(rows * C.d_model);
416 a.h2 = v.take(rows * C.d_model);
417 a.attention = latent_attention_activations_in<C>(v.take(latent_attention_activation_values(C, rows)), rows, selected);
418 a.mlp = moe_activations_in<C>(v.take(moe_activation_values(C, rows)), indices, rows);
419 return a;
420}

428template <transformer_config C, math::math_number T>
429 requires(C.latent() && C.moe())
430constexpr latent_moe_block_gradients<T> latent_moe_block_gradients_in(std::span<T> grads, std::span<T> work, std::size_t rows)
431{
432 assert(grads.size() == latent_moe_block_parameter_count(C) && work.size() == latent_moe_block_gradient_work(C, rows));
433 constexpr std::size_t d = C.d_model, b = C.norm == norm_kind::layer ? d : 0;
434 detail::cutter<T> c{grads}, w{work};
435 latent_moe_block_gradients<T> g{};
436 g.norm1_weight = c.take(d);
437 g.norm1_bias = c.take(b);
438 g.attention = latent_attention_gradients_in<C>(c.take(latent_attention_parameter_count(C)), w.take(latent_attention_gradient_work(C, rows)),
439 rows);
440 g.norm2_weight = c.take(d);
441 g.norm2_bias = c.take(b);
442 g.mlp = moe_gradients_in<C>(c.take(moe_parameter_count(C)), w.take(moe_gradient_work(C, rows)), rows);
443 g.scratch = w.take(rows * d);
444 return g;
445}
