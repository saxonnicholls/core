207[[nodiscard]] constexpr bool valid_long_context(const transformer_config& c) noexcept
208{
209 const bool standard = c.q_lora_rank == 0 && c.qk_rope_dim == 0 && c.v_head_dim == 0;
210 const bool latent = c.qk_rope_dim > 0 && c.qk_rope_dim % 2 == 0 && c.qk_rope_dim < c.head_dim && c.v_head_dim > 0
211 && c.rope != rope_kind::none && c.n_kv_heads == c.n_heads && c.mask == attention_mask::causal;
212 const bool yarn = c.rope_scaling == rope_scaling_kind::yarn || c.rope_scaling == rope_scaling_kind::yarn_index_ramp;
213 const bool scaling = c.rope_factor >= 1 && c.yarn_mscale >= 0
214 && (!yarn || (c.rope_original_context > 0 && c.yarn_beta_fast > c.yarn_beta_slow && c.yarn_beta_slow > 0));
215 const bool mtp = c.mtp_loss_weight >= 0 && (c.mtp_depth > 0 || c.mtp_loss_weight == 0);
216 const bool options = c.kv_lora_rank > 0 || (!c.attention_sinks && c.sparse_top_k == 0); // the latent attention's options
217 return (c.kv_lora_rank == 0 ? standard : latent) && scaling && mtp && options;
218}

227[[nodiscard]] constexpr bool valid(const transformer_config& c) noexcept
228{
229 const bool dense = c.experts == 0 && c.top_k == 0 && c.shared_experts == 0 && c.expert_d_ff == 0 && c.expert_groups == 0 && c.group_top_k == 0;
230 const bool groups = (c.expert_groups == 0 && c.group_top_k == 0)
231 || (c.expert_groups > 0 && c.experts % c.expert_groups == 0 && c.group_top_k > 0 && c.group_top_k <= c.expert_groups
232 && c.group_top_k * (c.experts / c.expert_groups) >= c.top_k);
233 const bool moe = c.experts > 0 && c.experts <= 0xffffffffu && c.top_k > 0 && c.top_k <= c.experts && c.expert_d_ff > 0
234 && c.route_scale > 0 && c.bias_update_rate >= 0 && c.balance_loss_weight >= 0 && c.z_loss_weight >= 0 && groups;
235 return c.d_model > 0 && c.n_heads > 0 && c.n_kv_heads > 0 && c.head_dim > 0 && c.d_ff > 0 && c.n_heads % c.n_kv_heads == 0
236 && (c.rope == rope_kind::none || c.head_dim % 2 == 0) && c.norm_eps > 0 && c.attention_softcap >= 0
237 && (c.rope == rope_kind::none || c.rope_base > 1) && (dense || moe) && (c.mask == attention_mask::causal || c.window == 0)
238 && ((c.mask == attention_mask::block_causal) == (c.mask_block > 0)) && valid_long_context(c);
239}
