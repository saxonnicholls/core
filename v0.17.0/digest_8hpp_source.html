138 [[nodiscard]] constexpr digest_hex<N> hex() const noexcept
139 {
140 constexpr char digits[] = "0123456789abcdef";
141 digest_hex<N> out{};
142 for (std::size_t i = 0; i < N; ++i) {
143 const auto b = static_cast<unsigned>(bytes[i]);
144 out.chars[2 * i] = digits[b >> 4];
145 out.chars[2 * i + 1] = digits[b & 0xFu];

166 [[nodiscard]] static constexpr std::optional<digest> from_hex(std::string_view text) noexcept
167 {
168 if (text.size() >= 2 && text[0] == '0' && (text[1] == 'x' || text[1] == 'X'))
169 text.remove_prefix(2);
170 if (text.size() != 2 * N)
171 return std::nullopt;
172 digest out{};
173 for (std::size_t i = 0; i < N; ++i) {
174 const int hi = nibble(text[2 * i]), lo = nibble(text[2 * i + 1]);
175 if (hi < 0 || lo < 0)
176 return std::nullopt;
177 out.bytes[i] = static_cast<std::byte>((hi << 4) | lo);
178 }
179 return out;
180 }

252 // update(input) for every hasher, forwarding the units to Derived::absorb(const U*, std::size_t).
253 template <typename Derived>
254 class hash_updates
255 {
256 public:
257 template <hash_input S>
258 constexpr Derived& update(const S& input) noexcept
259 {
260 Derived& self = static_cast<Derived&>(*this);
261 self.absorb(input_data(input), input_size(input));
262 return self;
263 }
