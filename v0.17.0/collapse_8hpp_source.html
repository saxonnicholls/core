100template <math::math_number T>
101struct embedding_moments
102{
103 std::span<T> mean{};
104 std::span<T> comoment{};
105 std::size_t count = 0;
106
108 [[nodiscard]] constexpr bool full() const noexcept { return comoment.size() != mean.size(); }
109
112 constexpr void add(std::span<const T> x)
113 {
114 const std::size_t d = mean.size();
115 assert(x.size() == d && (comoment.size() == d || comoment.size() == d * d));
116 ++count;
117 const T n = T(count);
118 if (full()) {
119 // M_ij += δ_i · (x_j - mean_j) with the new mean; δ is kept in the row's own pass
120 for (std::size_t i = 0; i < d; ++i) {
121 const T di = x[i] - mean[i];
122 for (std::size_t j = 0; j < d; ++j) {
123 const T newj = mean[j] + (x[j] - mean[j]) / n;
124 comoment[i * d + j] += di * (x[j] - newj);
125 }
126 }
127 for (std::size_t j = 0; j < d; ++j)
128 mean[j] += (x[j] - mean[j]) / n;
129 } else {
130 for (std::size_t j = 0; j < d; ++j) {
131 const T dj = x[j] - mean[j];
132 mean[j] += dj / n;
133 comoment[j] += dj * (x[j] - mean[j]);
134 }
135 }
136 }
137
139 constexpr void add_rows(std::span<const T> rows)
140 {
141 const std::size_t d = mean.size();
142 assert(d > 0 && rows.size() % d == 0);
143 for (std::size_t r = 0; r < rows.size() / d; ++r)
144 add(rows.subspan(r * d, d));
145 }
146
149 constexpr void merge(const embedding_moments& o)
150 {
151 const std::size_t d = mean.size();
152 assert(o.mean.size() == d && o.comoment.size() == comoment.size());
153 if (o.count == 0)
154 return;
155 const T na = T(count), nb = T(o.count), n = na + nb, f = na * nb / n;
156 for (std::size_t i = 0; i < d; ++i) {
157 const T di = o.mean[i] - mean[i];
158 if (full())
159 for (std::size_t j = 0; j < d; ++j)
160 comoment[i * d + j] += o.comoment[i * d + j] + di * (o.mean[j] - mean[j]) * f;
161 else
162 comoment[i] += o.comoment[i] + di * di * f;
163 }
164 for (std::size_t i = 0; i < d; ++i)
165 mean[i] += (o.mean[i] - mean[i]) * nb / n;
166 count += o.count;
167 }
168
170 constexpr void variance(std::span<T> out) const
171 {
172 const std::size_t d = mean.size();
173 assert(out.size() == d && count >= 2);
174 for (std::size_t j = 0; j < d; ++j)
175 out[j] = comoment[full() ? j * d + j : j] / T(count - 1);
176 }
177
179 constexpr void standard_deviation(std::span<T> out) const
180 {
181 variance(out);
182 for (T& v : out)
183 v = math::sqrt(v);
184 }
185
187 constexpr void covariance(std::span<T> out) const
188 {
189 assert(full() && out.size() == comoment.size() && count >= 2);
190 for (std::size_t i = 0; i < out.size(); ++i)
191 out[i] = comoment[i] / T(count - 1);
192 }
193
199 template <typename B>
200 constexpr result<T, linalg::error> effective_rank(const B& backend, std::span<T> work) const
201 {
202 const std::size_t d = mean.size();
203 assert(work.size() == 2 * d * d + d);
204 covariance(work.first(d * d));
205 using view = linalg::matrix_view<T, linalg::layout_right>;
206 const auto done = linalg::symmetric_eigen(backend, view(work.data(), {d, d}), work.subspan(2 * d * d, d),
207 view(work.data() + d * d, {d, d}));
208 if (!done)
209 return err(done.unwrap_err_unchecked());
210 const auto w = work.subspan(2 * d * d, d);
211 T total = T(0);
212 for (T& l : w) {
213 l = l > T(0) ? math::sqrt(l) : T(0);
214 total += l;
215 }
216 if (total == T(0))
217 return ok(T(0));
218 T h = T(0);
219 for (const T s : w)
220 if (s > T(0)) {
221 const T p = s / total;
222 h -= p * math::log(p);
223 }
224 return ok(math::exp(h));
225 }
226};

230 // Each column's mean of n rows of d, the rows summed in order then divided
231 template <typename T>
232 constexpr void column_means(std::span<const T> x, std::size_t d, std::span<T> mean)
233 {
234 const std::size_t n = x.size() / d;
235 for (std::size_t j = 0; j < d; ++j)
236 mean[j] = T(0);
237 for (std::size_t r = 0; r < n; ++r)
238 for (std::size_t j = 0; j < d; ++j)
239 mean[j] += x[r * d + j];
240 for (std::size_t j = 0; j < d; ++j)
241 mean[j] /= T(n);
242 }
243} // namespace detail::collapse

257template <typename B, math::math_number T>
258constexpr T vicreg_variance_rows(const B&, std::type_identity_t<std::span<const T>> x, std::size_t width, std::span<T> dx,
259 std::type_identity_t<std::span<T>> work, std::type_identity_t<T> weight = T(1),
260 std::type_identity_t<T> gamma = T(1), std::type_identity_t<T> eps = T(1e-4))
261{
262 const std::size_t d = width, n = width == 0 ? 0 : x.size() / width;
263 assert(d > 0 && x.size() == n * d && n >= 2 && (dx.empty() || dx.size() == x.size()) && work.size() == 2 * d);
264 const auto mean = work.first(d), sd = work.last(d);
265 detail::collapse::column_means(x, d, mean);
266 for (std::size_t j = 0; j < d; ++j)
267 sd[j] = T(0);
268 for (std::size_t r = 0; r < n; ++r)
269 for (std::size_t j = 0; j < d; ++j) {
270 const T c = x[r * d + j] - mean[j];
271 sd[j] += c * c;
272 }
273 T total = T(0);
274 for (std::size_t j = 0; j < d; ++j) {
275 sd[j] = math::sqrt(sd[j] / T(n - 1) + eps);
276 total += gamma - sd[j] > T(0) ? gamma - sd[j] : T(0);
277 }
278 if (!dx.empty())
279 for (std::size_t r = 0; r < n; ++r)
280 for (std::size_t j = 0; j < d; ++j)
281 if (gamma - sd[j] > T(0))
282 dx[r * d + j] -= weight / T(d) * (x[r * d + j] - mean[j]) / (T(n - 1) * sd[j]);
283 return weight * (total / T(d));
284}

296template <typename B, math::math_number T>
297constexpr T vicreg_covariance_rows(const B&, std::type_identity_t<std::span<const T>> x, std::size_t width, std::span<T> dx,
298 std::type_identity_t<std::span<T>> work, std::type_identity_t<T> weight = T(1))
299{
300 const std::size_t d = width, n = width == 0 ? 0 : x.size() / width;
301 assert(d > 0 && x.size() == n * d && n >= 2 && (dx.empty() || dx.size() == x.size()) && work.size() == d + d * d);
302 const auto mean = work.first(d), c = work.subspan(d, d * d);
303 detail::collapse::column_means(x, d, mean);
304 for (T& v : c)
305 v = T(0);
306 for (std::size_t r = 0; r < n; ++r)
307 for (std::size_t i = 0; i < d; ++i) {
308 const T xi = x[r * d + i] - mean[i];
309 for (std::size_t j = 0; j < d; ++j)
310 c[i * d + j] += xi * (x[r * d + j] - mean[j]);
311 }
312 T total = T(0);
313 for (std::size_t i = 0; i < d; ++i)
314 for (std::size_t j = 0; j < d; ++j) {
315 c[i * d + j] /= T(n - 1);
316 if (i != j)
317 total += c[i * d + j] * c[i * d + j];
318 }
319 if (!dx.empty()) {
320 const T g = weight * T(4) / (T(d) * T(n - 1));
321 for (std::size_t r = 0; r < n; ++r)
322 for (std::size_t b = 0; b < d; ++b) {
323 T s = T(0);
324 for (std::size_t j = 0; j < d; ++j)
325 if (j != b)
326 s += (x[r * d + j] - mean[j]) * c[j * d + b];
327 dx[r * d + b] += g * s;
328 }
329 }
330 return weight * (total / T(d));
331}

338template <math::math_number T, std::size_t K = 17>
339 requires(K >= 3 && K % 2 == 1)
340struct sigreg_quadrature
341{
342 std::array<T, K> t{};
343 std::array<T, K> phi{};
344 std::array<T, K> weight{};
345
347 constexpr explicit sigreg_quadrature(T t_max = T(3))
348 {
349 const T dt = t_max / T(K - 1);
350 for (std::size_t k = 0; k < K; ++k) {
351 t[k] = T(k) * dt;
352 phi[k] = math::exp(-(t[k] * t[k]) * T(0.5));
353 weight[k] = (k == 0 || k == K - 1 ? dt : T(2) * dt) * phi[k];
354 }
355 }
356};

365template <math::math_number T>
366constexpr void sigreg_directions(std::uint64_t seed, std::uint64_t step, std::size_t width, std::span<T> out)
367{
368 assert(width > 0 && out.size() % width == 0);
369 stats::wyrand g = stats::wyrand::stream(seed, step);
370 const stats::normal_inverse_cdf normal{};
371 for (std::size_t a = 0; a < out.size() / width; ++a) {
372 const auto v = out.subspan(a * width, width);
373 double s = 0;
374 for (T& x : v) {
375 const double z = normal(g);
376 x = static_cast<T>(z);
377 s += z * z;
378 }
379 const double len = math::sqrt(s);
380 for (T& x : v)
381 x = static_cast<T>(static_cast<double>(x) / len);
382 }
383}

392template <typename B, math::math_number T, std::size_t K>
393constexpr void sigreg_moments_rows(const B&, std::type_identity_t<std::span<const T>> x, std::size_t width,
394 std::type_identity_t<std::span<const T>> directions, const sigreg_quadrature<T, K>& q,
395 std::span<T> sums)
396{
397 assert(width > 0 && x.size() % width == 0 && directions.size() % width == 0 && sums.size() == directions.size() / width * K * 2);
398 const std::size_t m = directions.size() / width;
399 for (std::size_t r = 0; r < x.size() / width; ++r)
400 for (std::size_t a = 0; a < m; ++a) {
401 T z = T(0);
402 for (std::size_t c = 0; c < width; ++c)
403 z += x[r * width + c] * directions[a * width + c];
404 for (std::size_t k = 0; k < K; ++k) {
405 // One argument for both, so that a compiler pairing them into one sincos call (Apple's libm, whose
406 // sincos can differ from sin and cos in the last bit) pairs them wherever this loop is compiled
407 const T arg = q.t[k] * z;
408 sums[(a * K + k) * 2] += math::cos(arg);
409 sums[(a * K + k) * 2 + 1] += math::sin(arg);
410 }
411 }
412}

419template <math::math_number T, std::size_t K>
420[[nodiscard]] constexpr T sigreg_value(std::span<const T> sums, std::size_t count, const sigreg_quadrature<T, K>& q)
421{
422 assert(count > 0 && sums.size() % (K * 2) == 0);
423 const std::size_t m = sums.size() / (K * 2);
424 const T n = T(count);
425 T total = T(0);
426 for (std::size_t a = 0; a < m; ++a) {
427 T ep = T(0);
428 for (std::size_t k = 0; k < K; ++k) {
429 const T c = sums[(a * K + k) * 2] / n - q.phi[k], s = sums[(a * K + k) * 2 + 1] / n;
430 ep += q.weight[k] * (c * c + s * s);
431 }
432 total += n * ep;
433 }
434 return total / T(m);
435}

447template <typename B, math::math_number T, std::size_t K>
448constexpr void sigreg_backward_rows(const B&, std::type_identity_t<std::span<const T>> x, std::size_t width,
449 std::type_identity_t<std::span<const T>> directions, const sigreg_quadrature<T, K>& q,
450 std::type_identity_t<std::span<const T>> sums, std::size_t count, std::span<T> dx,
451 std::type_identity_t<T> weight = T(1))
452{
453 assert(width > 0 && x.size() % width == 0 && dx.size() == x.size() && count > 0);
454 const std::size_t m = directions.size() / width;
455 assert(sums.size() == m * K * 2);
456 const T n = T(count), g = weight / T(m);
457 for (std::size_t r = 0; r < x.size() / width; ++r)
458 for (std::size_t a = 0; a < m; ++a) {
459 T z = T(0);
460 for (std::size_t c = 0; c < width; ++c)
461 z += x[r * width + c] * directions[a * width + c];
462 T dz = T(0);
463 for (std::size_t k = 0; k < K; ++k) {
464 const T cb = sums[(a * K + k) * 2] / n - q.phi[k], sb = sums[(a * K + k) * 2 + 1] / n;
465 dz += T(2) * q.weight[k] * q.t[k] * (sb * math::cos(q.t[k] * z) - cb * math::sin(q.t[k] * z));
466 }
467 for (std::size_t c = 0; c < width; ++c)
468 dx[r * width + c] += g * dz * directions[a * width + c];
469 }
470}

483template <typename B, math::math_number T, std::size_t K>
484constexpr T sigreg_rows(const B& backend, std::type_identity_t<std::span<const T>> x, std::size_t width,
485 std::type_identity_t<std::span<const T>> directions, const sigreg_quadrature<T, K>& q, std::span<T> sums,
486 std::type_identity_t<std::span<T>> dx, std::type_identity_t<T> weight = T(1))
487{
488 for (T& v : sums)
489 v = T(0);
490 sigreg_moments_rows(backend, x, width, directions, q, sums);
491 const std::size_t n = x.size() / width;
492 if (!dx.empty())
493 sigreg_backward_rows(backend, x, width, directions, q, std::span<const T>(sums), n, dx, weight);
494 return weight * sigreg_value(std::span<const T>(sums), n, q);
495}
