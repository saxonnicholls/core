67struct control_vector_view

77 [[nodiscard]] constexpr std::span<const T> direction(std::int32_t il) const noexcept
78 {
79 if (il < 1 || layer_start < 0 || il < layer_start || il > layer_end)
80 return {};
81 const std::size_t at = width * std::size_t(il - 1);
82 return at + width <= data.size() ? data.subspan(at, width) : std::span<const T>{};
83 }
84};

91template <math::math_number T>
92constexpr void apply_control_vector(const control_vector_view<T>& cv, std::int32_t il, std::span<T> rows)
93{
94 assert(cv.width > 0 && rows.size() % cv.width == 0);
95 const auto d = cv.direction(il);
96 if (d.empty())
97 return;
98 for (std::size_t r = 0; r < rows.size(); r += cv.width)
99 for (std::size_t c = 0; c < cv.width; ++c)
100 rows[r + c] += d[c];
101}

108template <math::math_number T>
109constexpr void add_control_vector(std::span<T> sum, std::type_identity_t<std::span<const T>> data, std::type_identity_t<T> strength)
110{
111 assert(sum.size() == data.size());
112 for (std::size_t i = 0; i < sum.size(); ++i)
113 sum[i] += strength * data[i];
114}

124template <math::math_number T>
125constexpr void control_vector_backward(const control_vector_view<T>& cv, std::int32_t il, std::type_identity_t<std::span<const T>> drows,
126 std::span<T> ddirection)
127{
128 assert(cv.width > 0 && drows.size() % cv.width == 0 && ddirection.size() == cv.width);
129 if (cv.direction(il).empty())
130 return;
131 for (std::size_t r = 0; r < drows.size(); r += cv.width)
132 for (std::size_t c = 0; c < cv.width; ++c)
133 ddirection[c] += drows[r + c];
134}
