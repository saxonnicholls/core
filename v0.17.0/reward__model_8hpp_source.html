83template <math::math_number T>
84constexpr pair_loss<T> preference_loss(const bradley_terry& c, T r_chosen, std::type_identity_t<T> r_rejected,
85 std::type_identity_t<T> = T(0), std::type_identity_t<T> = T(0)) noexcept(std::is_floating_point_v<T>)
86{
87 const T z = (r_chosen - r_rejected) - T(c.margin);
88 const T s = sigmoid(-z);
89 return {softplus(-z), -s, s, r_chosen, r_rejected};
90}

97template <row W, row H>
98 requires std::same_as<row_value_t<W>, row_value_t<H>>
99constexpr row_value_t<W> value_head(const W& w, std::type_identity_t<row_value_t<W>> bias, const H& h) noexcept(std::is_floating_point_v<row_value_t<W>>)
100{
101 const auto a = as_row(w);
102 const auto x = as_row(h);
103 assert(a.size() == x.size());
104 auto r = bias;
105 for (std::size_t j = 0; j < a.size(); ++j)
106 r += a[j] * x[j];
107 return r;
108}

113template <math::math_number T>
114constexpr void value_head_backward(std::span<const T> w, std::type_identity_t<std::span<const T>> h, std::type_identity_t<T> dr,
115 std::type_identity_t<std::span<T>> dh, std::type_identity_t<std::span<T>> dw, T& db) noexcept(std::is_floating_point_v<T>)
116{
117 assert(h.size() == w.size() && dw.size() == w.size() && (dh.empty() || dh.size() == w.size()));
118 for (std::size_t j = 0; j < w.size(); ++j) {
119 if (!dh.empty())
120 dh[j] += dr * w[j];
121 dw[j] += dr * h[j];
122 }
123 db += dr;
124}

128template <math::math_number T, typename B>
129constexpr void value_head_rows(const B&, std::span<const T> w, std::type_identity_t<T> bias, std::type_identity_t<std::span<const T>> hs,
130 std::type_identity_t<std::span<T>> values) noexcept(std::is_floating_point_v<T>)
131{
132 const std::size_t d = w.size();

157constexpr void last_response_positions(std::span<const std::uint8_t> mask, std::span<std::uint32_t> positions) noexcept
158{
159 const std::size_t rows = positions.size();
160 assert(rows == 0 || mask.size() % rows == 0);
161 if (rows == 0)
162 return;
163 const std::size_t len = mask.size() / rows;
164 for (std::size_t r = 0; r < rows; ++r) {
165 std::size_t last = len;
166 for (std::size_t t = 0; t < len; ++t)
167 if (mask[r * len + t] != 0)
168 last = t;
169 assert(last < len);
170 positions[r] = static_cast<std::uint32_t>(last);
171 }
172}
