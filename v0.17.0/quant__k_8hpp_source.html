169 // ggml's make_q3_quants: a symmetric scale for n numbers, codes in [0, 2 nmax), refined for the x^2-weighted
170 // error when do_rmse
171 constexpr float make_q3_quants(int n, int nmax, const float* x, std::int8_t* L, bool do_rmse)
172 {
173 float max = 0, amax = 0;
174 for (int i = 0; i < n; ++i) {
175 const float ax = abs(x[i]);
176 if (ax > amax) {
177 amax = ax;
178 max = x[i];
179 }
180 }
181 if (amax < group_max_eps) {
182 for (int i = 0; i < n; ++i)
183 L[i] = 0;
184 return 0.0f;
185 }
186 const float iscale = float(-nmax) / max;
187 if (do_rmse) {
188 float sumlx = 0, suml2 = 0;
189 for (int i = 0; i < n; ++i) {
190 const float v = iscale * x[i];
191 const int l = imax(-nmax, imin(nmax - 1, nearest_int(v)));
192 L[i] = static_cast<std::int8_t>(l);
193 const float w = x[i] * x[i];
194 const float wx = w * x[i];
195 const float wxl = wx * float(l);
196 sumlx += wxl;
197 const float wl = w * float(l);
198 const float wll = wl * float(l);
199 suml2 += wll;
200 }
201 for (int itry = 0; itry < 5; ++itry) {
202 int n_changed = 0;
203 for (int i = 0; i < n; ++i) {
204 const float w = x[i] * x[i];
205 const float wx = w * x[i];
206 const float wxl = wx * float(L[i]);
207 float slx = sumlx - wxl;
208 if (slx > 0) {
209 const float wl = w * float(L[i]);
210 const float wll = wl * float(L[i]);
211 float sl2 = suml2 - wll;
212 const float xs = x[i] * sl2;
213 const float v = xs / slx;
214 const int new_l = imax(-nmax, imin(nmax - 1, nearest_int(v)));
215 if (new_l != L[i]) {
216 const float wxn = wx * float(new_l);
217 slx += wxn;
218 const float wn = w * float(new_l);
219 const float wnn = wn * float(new_l);
220 sl2 += wnn;
221 const float s1 = slx * slx;
222 const float lhs = s1 * suml2;
223 const float s2 = sumlx * sumlx;
224 const float rhs = s2 * sl2;
225 if (sl2 > 0 && lhs > rhs) {
226 L[i] = static_cast<std::int8_t>(new_l);
227 sumlx = slx;
228 suml2 = sl2;
229 ++n_changed;
230 }
231 }
232 }
233 }
234 if (n_changed == 0)
235 break;
236 }
237 for (int i = 0; i < n; ++i)
238 L[i] = static_cast<std::int8_t>(L[i] + nmax);
239 return suml2 > 0.0f ? sumlx / suml2 : 0.0f;
240 }
241 for (int i = 0; i < n; ++i) {
242 const float v = iscale * x[i];
243 L[i] = static_cast<std::int8_t>(imax(-nmax, imin(nmax - 1, nearest_int(v))) + nmax);
244 }
245 return 1 / iscale;
246 }

248 // ggml's get_scale_min_k4: the 6-bit scale and minimum of sub-block j of Q4_K and Q5_K
249 constexpr void scale_min_k4(int j, const std::array<std::uint8_t, 12>& q, std::uint8_t& d, std::uint8_t& m)
250 {
251 if (j < 4) {
252 d = static_cast<std::uint8_t>(q[j] & 63);
253 m = static_cast<std::uint8_t>(q[j + 4] & 63);
254 } else {
255 d = static_cast<std::uint8_t>((q[j + 4] & 0xf) | ((q[j - 4] >> 6) << 4));
256 m = static_cast<std::uint8_t>((q[j + 4] >> 4) | ((q[j] >> 6) << 4));
257 }
258 }

260 // Q4_K's and Q5_K's packing of eight 6-bit scales and minimums into 12 bytes
261 constexpr void pack_scale_min_k4(const std::uint8_t* ls, const std::uint8_t* lm, std::array<std::uint8_t, 12>& s)
262 {
263 s = {};
264 for (int j = 0; j < 8; ++j) {
265 if (j < 4) {
266 s[j] = ls[j];
267 s[j + 4] = lm[j];
268 } else {
269 s[j + 4] = static_cast<std::uint8_t>((ls[j] & 0xf) | ((lm[j] & 0xf) << 4));
270 s[j - 4] = static_cast<std::uint8_t>(s[j - 4] | ((ls[j] >> 4) << 6));
271 s[j] = static_cast<std::uint8_t>(s[j] | ((lm[j] >> 4) << 6));
272 }
273 }
274 }

276 // Q3_K's 6-bit scale of sub-block j, minus 32
277 constexpr int q3_scale(const std::array<std::uint8_t, 12>& s, int j)
278 {
279 const int low = j < 8 ? s[j] & 0xf : s[j - 8] >> 4;
280 const int high = (s[8 + j % 4] >> (2 * (j / 4))) & 3;
281 return static_cast<std::int8_t>(low | (high << 4)) - 32;
282 }

284 // Q3_K's packing of sixteen 6-bit scales (0 to 63)
285 constexpr void pack_q3_scales(const int* l6, std::array<std::uint8_t, 12>& s)
286 {
287 s = {};
288 for (int j = 0; j < 16; ++j) {
289 int l = l6[j];
290 if (j < 8)
291 s[j] = static_cast<std::uint8_t>(l & 0xf);
292 else
293 s[j - 8] = static_cast<std::uint8_t>(s[j - 8] | ((l & 0xf) << 4));
294 l >>= 4;
295 s[j % 4 + 8] = static_cast<std::uint8_t>(s[j % 4 + 8] | (l << (2 * (j / 4))));
296 }
297 }

299 // The 2-bit packing of Q2_K and Q3_K: element 128 h + 32 g + l in bits 2g of byte 32 h + l
300 template <typename L>
301 constexpr void pack2(const L* codes, std::array<std::uint8_t, 64>& qs)
302 {
303 for (int j = 0; j < 256; j += 128)
304 for (int l = 0; l < 32; ++l)
305 qs[j / 4 + l] = static_cast<std::uint8_t>(codes[j + l] | (codes[j + l + 32] << 2) | (codes[j + l + 64] << 4) | (codes[j + l + 96] << 6));
306 }

308 // A block's codes, element by element, unpacked the way ggml's scalar dot products unpack them
309 constexpr void unpack(const block_q2_k& b, std::array<std::int8_t, 256>& a)
310 {
311 for (std::size_t h = 0; h < 2; ++h)
312 for (std::size_t g = 0; g < 4; ++g)
313 for (std::size_t l = 0; l < 32; ++l)
314 a[128 * h + 32 * g + l] = static_cast<std::int8_t>((b.qs[32 * h + l] >> (2 * g)) & 3);
315 }

317 constexpr void unpack(const block_q3_k& b, std::array<std::int8_t, 256>& a)
318 {
319 for (std::size_t h = 0; h < 2; ++h)
320 for (std::size_t g = 0; g < 4; ++g) {
321 const auto m = static_cast<std::uint8_t>(1u << (4 * h + g));
322 for (std::size_t l = 0; l < 32; ++l)
323 a[128 * h + 32 * g + l] = static_cast<std::int8_t>(((b.qs[32 * h + l] >> (2 * g)) & 3) - (b.hmask[l] & m ? 0 : 4));
324 }

336 constexpr void unpack(const block_q5_k& b, std::array<std::int8_t, 256>& a)
337 {
338 for (std::size_t h = 0; h < 4; ++h) {
339 const auto m1 = static_cast<std::uint8_t>(1u << (2 * h)), m2 = static_cast<std::uint8_t>(2u << (2 * h));
340 for (std::size_t l = 0; l < 32; ++l) {
341 a[64 * h + l] = static_cast<std::int8_t>((b.qs[32 * h + l] & 0xf) + (b.qh[l] & m1 ? 16 : 0));
342 a[64 * h + 32 + l] = static_cast<std::int8_t>((b.qs[32 * h + l] >> 4) + (b.qh[l] & m2 ? 16 : 0));
343 }
344 }

347 constexpr void unpack(const block_q6_k& b, std::array<std::int8_t, 256>& a)
348 {
349 for (std::size_t n = 0; n < 2; ++n)
350 for (std::size_t l = 0; l < 32; ++l) {
351 const std::uint8_t lo = b.ql[64 * n + l], hi = b.ql[64 * n + 32 + l], h = b.qh[32 * n + l];
352 a[128 * n + l] = static_cast<std::int8_t>(((lo & 0xf) | ((h & 3) << 4)) - 32);
353 a[128 * n + 32 + l] = static_cast<std::int8_t>(((hi & 0xf) | (((h >> 2) & 3) << 4)) - 32);
354 a[128 * n + 64 + l] = static_cast<std::int8_t>(((lo >> 4) | (((h >> 4) & 3) << 4)) - 32);
355 a[128 * n + 96 + l] = static_cast<std::int8_t>(((hi >> 4) | (((h >> 6) & 3) << 4)) - 32);
356 }

359 // The sum over a block's 256 products scale[e / 16] · q8[e] · a[e] in ggml's eight lanes (lane e % 8), then
360 // sums[lane] += d · lane: the order of ggml_vec_dot_q{3,4,5,6}_K_q8_K_generic (integer sums, so exact)
361 constexpr void lanes(const block_q8_k& y, float d, const std::array<std::int8_t, 256>& a, const std::array<int, 16>& scale, std::array<float, 8>& sums)
362 {
363 std::array<int, 8> aux32{};
364 for (std::size_t j = 0; j < 32; ++j) {
365 const int s = scale[j / 2];
366 for (std::size_t l = 0; l < 8; ++l)
367 aux32[l] += s * (int(y.qs[8 * j + l]) * int(a[8 * j + l]));
368 }
369 for (std::size_t l = 0; l < 8; ++l) {
370 const float term = d * float(aux32[l]);
371 sums[l] += term;
372 }
373 }
374
375 constexpr float sum_lanes(float sumf, const std::array<float, 8>& sums)
376 {
377 for (const float s : sums)
378 sumf += s;
379 return sumf;
380 }

382 // Q4_K's and Q5_K's reference: an affine fit per 32 (make_qkx2_quants), weighted av_x + |x|
383 template <int Nmax>
384 constexpr void reference_k4(const float* x, std::array<std::uint8_t, 256>& L, float16& d16, float16& dmin16, std::array<std::uint8_t, 12>& packed,
385 float rmin, float rdelta, int nstep)
386 {
387 std::array<float, 8> scales{}, mins{};
388 std::array<float, 32> weights{};
389 std::array<std::uint8_t, 32> Laux{};
390 float max_scale = 0, max_min = 0;
391 for (int j = 0; j < 8; ++j) {
392 float sum_x2 = 0;
393 for (int l = 0; l < 32; ++l) {
394 const float v2 = x[32 * j + l] * x[32 * j + l];
395 sum_x2 += v2;
396 }
397 const float av_x = sqrt(sum_x2 / 32);
398 for (int l = 0; l < 32; ++l)
399 weights[std::size_t(l)] = av_x + abs(x[32 * j + l]);
400 scales[std::size_t(j)] = make_qkx_quants<false>(32, Nmax, x + 32 * j, weights.data(), L.data() + 32 * j, &mins[std::size_t(j)], Laux.data(), rmin, rdelta, nstep, false);
401 if (scales[std::size_t(j)] > max_scale)
402 max_scale = scales[std::size_t(j)];
403 if (mins[std::size_t(j)] > max_min)
404 max_min = mins[std::size_t(j)];
405 }
406 const float inv_scale = max_scale > 0 ? 63.0f / max_scale : 0.0f;
407 const float inv_min = max_min > 0 ? 63.0f / max_min : 0.0f;
408 std::array<std::uint8_t, 8> ls{}, lm{};
409 for (std::size_t j = 0; j < 8; ++j) {
410 const float vs = inv_scale * scales[j];
411 const float vm = inv_min * mins[j];
412 const auto a = static_cast<std::uint8_t>(nearest_int(vs));
413 const auto b = static_cast<std::uint8_t>(nearest_int(vm));
414 ls[j] = a < 63 ? a : std::uint8_t{63};
415 lm[j] = b < 63 ? b : std::uint8_t{63};
416 }
417 pack_scale_min_k4(ls.data(), lm.data(), packed);
418 d16 = fp16(max_scale / 63.0f);
419 dmin16 = fp16(max_min / 63.0f);
420 }

422 // Q4_K's and Q5_K's importance-weighted path: make_qkx3_quants per 32, make_qp_quants for the scales
423 template <int Nmax>
424 constexpr void weighted_k4(const float* x, const float* qw, std::array<std::uint8_t, 256>& L, float16& d16, float16& dmin16,
425 std::array<std::uint8_t, 12>& packed, bool clamp)
426 {
427 std::array<float, 8> scales{}, mins{}, sw{};
428 std::array<float, 32> weights{};
429 std::array<std::uint8_t, 32> Laux{};
430 std::array<std::uint8_t, 8> Ls{}, Lm{};
431 float sum_x2 = 0;
432 for (int l = 0; l < 256; ++l) {
433 const float v2 = x[l] * x[l];
434 sum_x2 += v2;
435 }
436 const float sigma2 = 2 * sum_x2 / 256;
437 for (int j = 0; j < 8; ++j) {
438 for (int l = 0; l < 32; ++l)
439 weights[std::size_t(l)] = importance_weight(qw[32 * j + l], sigma2, x[32 * j + l]);
440 float sumw = 0;
441 for (const float w : weights)
442 sumw += w;
443 sw[std::size_t(j)] = sumw;
444 scales[std::size_t(j)] = make_qkx_quants<true>(32, Nmax, x + 32 * j, weights.data(), L.data() + 32 * j, &mins[std::size_t(j)], Laux.data(), -0.9f, 0.05f, 36, false);
445 }
446 const float d_block = make_qp_quants(8, 63, scales.data(), Ls.data(), sw.data());
447 const float m_block = make_qp_quants(8, 63, mins.data(), Lm.data(), sw.data());
448 if (clamp)
449 for (std::size_t j = 0; j < 8; ++j) {
450 Ls[j] = Ls[j] < 63 ? Ls[j] : std::uint8_t{63};
451 Lm[j] = Lm[j] < 63 ? Lm[j] : std::uint8_t{63};
452 }
453 pack_scale_min_k4(Ls.data(), Lm.data(), packed);
454 d16 = fp16(d_block);
455 dmin16 = fp16(m_block);
456 }

458 // The codes of Q4_K and Q5_K from the packed scales: round((x + dmin·m) / (d·s)) clamped to [0, top]
459 constexpr void requantize_k4(const float* x, const float16& d16, const float16& dmin16, const std::array<std::uint8_t, 12>& packed, int top,
460 std::array<std::uint8_t, 256>& L)
461 {
462 for (int j = 0; j < 8; ++j) {
463 std::uint8_t sc = 0, m = 0;
464 scale_min_k4(j, packed, sc, m);
465 const float d = f32(d16) * float(sc);
466 if (d == 0.0f)
467 continue;
468 const float dm = f32(dmin16) * float(m);
469 for (int ii = 0; ii < 32; ++ii) {
470 const float v = (x[32 * j + ii] + dm) / d;
471 L[std::size_t(32 * j + ii)] = static_cast<std::uint8_t>(imax(0, imin(top, nearest_int(v))));
472 }
473 }
474 }

481constexpr void quantize_row(std::span<const float> x, std::span<const float> importance, std::span<block_q2_k> y)
482{
483 using namespace detail::quant_k;
484 check(256, y.size(), x.size(), importance.size());
485 for (std::size_t i = 0; i < y.size(); ++i) {
486 const float* xb = x.data() + 256 * i;
487 std::array<std::uint8_t, 256> L{};
488 std::array<std::uint8_t, 16> Laux{};
489 std::array<float, 16> weight{}, mins{}, scales{};
490 block_q2_k& b = y[i];
491 if (importance.empty()) {
492 float max_scale = 0, max_min = 0;
493 for (int j = 0; j < 16; ++j) {
494 for (int l = 0; l < 16; ++l)
495 weight[std::size_t(l)] = abs(xb[16 * j + l]);
496 scales[std::size_t(j)] = make_qkx_quants<false>(16, 3, xb + 16 * j, weight.data(), L.data() + 16 * j, &mins[std::size_t(j)], Laux.data(), -0.5f, 0.1f, 15, true);
497 if (scales[std::size_t(j)] > max_scale)
498 max_scale = scales[std::size_t(j)];
499 if (mins[std::size_t(j)] > max_min)
500 max_min = mins[std::size_t(j)];
501 }
502 b.scales = {};
503 if (max_scale > 0) {
504 const float iscale = 15.0f / max_scale;
505 for (std::size_t j = 0; j < 16; ++j) {
506 const float v = iscale * scales[j];
507 b.scales[j] = static_cast<std::uint8_t>(nearest_int(v));
508 }
509 b.d = fp16(max_scale / 15.0f);
510 } else {
511 b.d = fp16(0.0f);
512 }
513 if (max_min > 0) {
514 const float iscale = 15.0f / max_min;
515 for (std::size_t j = 0; j < 16; ++j) {
516 const float v = iscale * mins[j];
517 b.scales[j] = static_cast<std::uint8_t>(b.scales[j] | (nearest_int(v) << 4));
518 }
519 b.dmin = fp16(max_min / 15.0f);
520 } else {
521 b.dmin = fp16(0.0f);
522 }
523 } else {
524 const float* qw = importance.data() + 256 * i;
525 std::array<float, 16> sw{};
526 std::array<std::uint8_t, 16> Ls{}, Lm{};
527 float sumx2 = 0;
528 for (int j = 0; j < 256; ++j) {
529 const float v2 = xb[j] * xb[j];
530 sumx2 += v2;
531 }
532 const float sigma2 = sumx2 / 256;
533 for (int j = 0; j < 16; ++j) {
534 for (int l = 0; l < 16; ++l)
535 weight[std::size_t(l)] = importance_weight(qw[16 * j + l], sigma2, xb[16 * j + l]);
536 for (const float w : weight)
537 sw[std::size_t(j)] += w;
538 scales[std::size_t(j)] = make_qkx_quants<true>(16, 3, xb + 16 * j, weight.data(), L.data() + 16 * j, &mins[std::size_t(j)], Laux.data(), -0.9f, 0.05f, 36, false);
539 }
540 b.d = fp16(make_qp_quants(16, 15, scales.data(), Ls.data(), sw.data()));
541 b.dmin = fp16(make_qp_quants(16, 15, mins.data(), Lm.data(), sw.data()));
542 for (std::size_t j = 0; j < 16; ++j)
543 b.scales[j] = static_cast<std::uint8_t>(Ls[j] | (Lm[j] << 4));
544 }
545 for (int j = 0; j < 16; ++j) {
546 const float d = f32(b.d) * float(b.scales[std::size_t(j)] & 0xf);
547 if (d == 0.0f)
548 continue;
549 const float dm = f32(b.dmin) * float(b.scales[std::size_t(j)] >> 4);
550 for (int ii = 0; ii < 16; ++ii) {
551 const float v = (xb[16 * j + ii] + dm) / d;
552 L[std::size_t(16 * j + ii)] = static_cast<std::uint8_t>(imax(0, imin(3, nearest_int(v))));
553 }
554 }
555 pack2(L.data(), b.qs);
556 }
557}

572 block_q3_k& b = y[i];
573 if (importance.empty()) {
574 float max_scale = 0, amax = 0;
575 for (int j = 0; j < 16; ++j) {
576 scales[std::size_t(j)] = make_q3_quants(16, 4, xb + 16 * j, L.data() + 16 * j, true);
577 const float scale = abs(scales[std::size_t(j)]);
578 if (scale > amax) {
579 amax = scale;
580 max_scale = scales[std::size_t(j)];
581 }
582 }
583 if (max_scale != 0.0f) {
584 const float iscale = -32.0f / max_scale;
585 for (std::size_t j = 0; j < 16; ++j) {
586 const float v = iscale * scales[j];
587 const int l = static_cast<std::int8_t>(nearest_int(v));
588 l6[j] = imax(-32, imin(31, l)) + 32;
589 }
590 pack_q3_scales(l6.data(), b.scales);
591 b.d = fp16(1 / iscale);
592 } else {
593 b.scales = {};
594 b.d = fp16(0.0f);
595 }
596 } else {
597 const float* qw = importance.data() + 256 * i;
598 std::array<float, 16> weight{}, sw{};
599 std::array<std::int8_t, 16> Ls{};
600 float sumx2 = 0;
601 for (int j = 0; j < 256; ++j) {
602 const float v2 = xb[j] * xb[j];
603 sumx2 += v2;
604 }
605 const float sigma2 = 2 * sumx2 / 256;
606 for (int j = 0; j < 16; ++j) {
607 for (int l = 0; l < 16; ++l)
608 weight[std::size_t(l)] = importance_weight(qw[16 * j + l], sigma2, xb[16 * j + l]);
609 float sumw = 0;
610 for (const float w : weight)
611 sumw += w;
612 sw[std::size_t(j)] = sumw;
613 scales[std::size_t(j)] = make_qx_quants(16, 4, xb + 16 * j, L.data() + 16 * j, 1, weight.data());
614 }
615 const float d_block = make_qx_quants(16, 32, scales.data(), Ls.data(), 1, sw.data());
616 for (std::size_t j = 0; j < 16; ++j)
617 l6[j] = Ls[j];
618 pack_q3_scales(l6.data(), b.scales);
619 b.d = fp16(d_block);
620 }
621 for (int j = 0; j < 16; ++j) {
622 const float d = f32(b.d) * float(q3_scale(b.scales, j));
623 if (d == 0.0f)
624 continue;
625 for (int ii = 0; ii < 16; ++ii) {
626 const float v = xb[16 * j + ii] / d;
627 L[std::size_t(16 * j + ii)] = static_cast<std::int8_t>(imax(-4, imin(3, nearest_int(v))) + 4);
628 }
629 }
630 b.hmask = {};
631 for (std::size_t j = 0; j < 256; ++j)
632 if (L[j] > 3) {
633 b.hmask[j % 32] = static_cast<std::uint8_t>(b.hmask[j % 32] | (1u << (j / 32)));
634 L[j] = static_cast<std::int8_t>(L[j] - 4);
635 }
636 pack2(L.data(), b.qs);
637 }
638}

712 const float* qw = importance.empty() ? nullptr : importance.data() + 256 * i + 16 * std::size_t(ib);
713 const float scale = make_qx_quants(16, 32, xb + 16 * ib, L.data() + 16 * ib, 1, qw);

738 }

741 for (std::size_t l = 0; l < 32; ++l) {

793 template <typename B>
794 constexpr void dequantize_k4(std::span<const B> x, std::span<float> y)
795 {

809 }
810 }
811 }

814 // Q4_K's and Q5_K's dot product: the lanes with a scale per 32, minus dmin · (mins · bsums) per block
815 template <typename B>
816 constexpr float dot_k4(std::span<const B> x, std::span<const block_q8_k> y)
817 {
818 assert(x.size() == y.size());
819 std::array<float, 8> sums{};
820 std::array<std::int8_t, 256> a{};
821 float sumf = 0;
822 for (std::size_t i = 0; i < x.size(); ++i) {
823 const B& b = x[i];
824 unpack(b, a);
825 std::array<int, 16> sc{};
826 int sumi = 0;
827 for (std::size_t j = 0; j < 8; ++j) {
828 std::uint8_t s = 0, m = 0;
829 scale_min_k4(int(j), b.scales, s, m);
830 sc[2 * j] = sc[2 * j + 1] = s;
831 sumi += y[i].bsums[2 * j] * m;
832 sumi += y[i].bsums[2 * j + 1] * m;
833 }
834 const float d = to_float(b.d) * y[i].d;
835 lanes(y[i], d, a, sc, sums);
836 const float dmin = to_float(b.dmin) * y[i].d;
837 const float term = dmin * float(sumi);
838 sumf -= term;
839 }
840 return sum_lanes(sumf, sums);
841 }

876constexpr float dot(std::span<const block_q2_k> x, std::span<const block_q8_k> y)
877{
878 assert(x.size() == y.size());
879 std::array<std::int8_t, 256> a{};
880 float sumf = 0;
881 for (std::size_t i = 0; i < x.size(); ++i) {
882 detail::quant_k::unpack(x[i], a);
883 int summs = 0;
884 for (std::size_t j = 0; j < 16; ++j)
885 summs += y[i].bsums[j] * (x[i].scales[j] >> 4);
886 const float dall = y[i].d * to_float(x[i].d);
887 const float dmin = y[i].d * to_float(x[i].dmin);
888 int isum = 0;
889 for (std::size_t j = 0; j < 16; ++j) {
890 int isuml = 0;
891 for (std::size_t l = 0; l < 16; ++l)
892 isuml += int(y[i].qs[16 * j + l]) * int(a[16 * j + l]);
893 isum += (x[i].scales[j] & 0xf) * isuml;
894 }
895 const float p = dall * float(isum);
896 const float q = dmin * float(summs);
897 const float term = p - q;
898 sumf += term;
899 }
900 return sumf;
901}
