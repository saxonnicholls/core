106 template <std::size_t Block>
107 constexpr void set_i2(std::span<std::uint8_t> out, std::size_t j, unsigned code)
108 {
109 constexpr std::size_t g = Block / 4;
110 const std::size_t byte = j / Block * g + j % g;
111 const unsigned shift = unsigned(6 - 2 * ((j % Block) / g));
112 out[byte] = static_cast<std::uint8_t>(out[byte] | (code << shift));
113 }

130template <std::size_t Block = i2_s_block_x86>
131 requires(Block == i2_s_block_x86 || Block == i2_s_block_arm)
132constexpr float quantize_i2_s(std::span<const float> w, std::span<std::uint8_t> out) noexcept
133{
134 using namespace detail::bitnet;
135 assert(w.size() % Block == 0 && out.size() == i2_s_bytes(w.size()));
136 double max = 0;
137 for (const float v : w)
138 max = dmax(max, dabs(double(v)));
139 for (std::size_t i = 0; i < w.size() / 4; ++i)
140 out[i] = 0;
141 for (std::size_t j = 0; j < w.size(); ++j) {
142 const double v = double(w[j]);
143 const double p = v * max;
144 set_i2<Block>(out, j, dabs(v) < 1e-6 ? 1u : p > 0 ? 2u : 0u);
145 }
146 put_scale(out, w.size() / 4, float(max));
147 return float(max);
148}

155template <std::size_t Block = i2_s_block_x86>
156 requires(Block == i2_s_block_x86 || Block == i2_s_block_arm)
157constexpr void pack_i2_s(std::span<const std::int8_t> t, float scale, std::span<std::uint8_t> out) noexcept
158{
159 assert(t.size() % Block == 0 && out.size() == i2_s_bytes(t.size()));
160 for (std::size_t i = 0; i < t.size() / 4; ++i)
161 out[i] = 0;
162 for (std::size_t j = 0; j < t.size(); ++j)

164 detail::bitnet::put_scale(out, t.size() / 4, scale);
165}

196template <std::size_t Block = i2_s_block_x86>
197[[nodiscard]] constexpr std::int32_t i2_s_dot_raw(std::span<const std::uint8_t> packed, std::span<const std::int8_t> a) noexcept
198{
199 assert(a.size() % Block == 0 && packed.size() >= a.size() / 4);
200 constexpr std::size_t g = Block / 4;
201 std::int32_t s = 0;
202 for (std::size_t b = 0; b < a.size() / Block; ++b)
203 for (std::size_t k = 0; k < 4; ++k)
204 for (std::size_t j = 0; j < g; ++j)
205 s += std::int32_t((packed[b * g + j] >> (6 - 2 * k)) & 3) * std::int32_t(a[b * Block + k * g + j]);
206 return s;
207}

214template <std::size_t Block = i2_s_block_x86>
215[[nodiscard]] constexpr std::int32_t i2_s_dot(std::span<const std::uint8_t> packed, std::span<const std::int8_t> a) noexcept
216{
217 std::int32_t sum = 0;

302template <row X>
303 requires std::same_as<row_value_t<X>, float>
304constexpr float lut_activation_quantize(const X& x, std::span<std::int8_t> q) noexcept
305{
306 const auto in = as_row(x);
307 assert(q.size() == in.size());
308 float amax = 0.0f;
309 for (const float v : in)
310 amax = detail::quantize::fmax(detail::quantize::abs(v), amax);
311 const float s = amax > 0.0f ? 127.0f / amax : std::numeric_limits<float>::infinity();
312 for (std::size_t i = 0; i < in.size(); ++i) {
313 const float v = in[i] * s;
314 q[i] = static_cast<std::int8_t>(detail::bitnet::round_clamp(v, -127, 127));
315 }
316 return s;
317}

395 // One numpy step, a reshape to `shape` then a transpose by `perm`; source_of maps a flat index of the result of
396 // the whole chain back to the flat index it came from
397 struct step
398 {
399 std::size_t n = 0;
400 std::array<std::size_t, 8> shape{};
401 std::array<std::size_t, 8> perm{};
402 };
403
404 template <std::size_t S>
405 constexpr std::size_t source_of(std::size_t o, const std::array<step, S>& steps)
406 {
407 for (std::size_t s = S; s-- > 0;) {
408 const step& st = steps[s];
409 std::array<std::size_t, 8> idx_out{}, idx_in{};
410 for (std::size_t k = st.n; k-- > 0;) { // decompose o by the transposed shape
411 const std::size_t extent = st.shape[st.perm[k]];
412 idx_out[k] = o % extent;
413 o /= extent;
414 }
415 for (std::size_t k = 0; k < st.n; ++k)
416 idx_in[st.perm[k]] = idx_out[k];
417 o = 0;
418 for (std::size_t k = 0; k < st.n; ++k)
419 o = o * st.shape[k] + idx_in[k];
420 }
421 return o;
422 }

424 // The four tiling steps the TL converters share, for groups of g weights (by the kernel's group of columns)
425 constexpr std::array<step, 4> tl_tiling(std::size_t M, std::size_t K, std::size_t g, std::size_t BM, std::size_t BY, std::size_t bm,
426 std::size_t by)
427 {
428 return {step{3, {M / BM, BM, K / g}, {0, 2, 1}}, step{4, {M / BM, K / BY, BY / g, BM}, {0, 1, 3, 2}},
429 step{5, {M / BM, K / BY, BM / bm, bm, BY / g}, {0, 1, 2, 4, 3}},
430 step{6, {M / BM, K / BY, BM / bm, BY / by, by / g, bm}, {0, 1, 2, 3, 5, 4}}};
431 }

433 // TL2's byte rows of bm: groups of 8 bytes 1 and 2 swapped (weight[:, [0, 1, 2, 3], :] = weight[:, [0, 2, 1, 3], :])
434 constexpr std::size_t swap_groups(std::size_t o)
435 {
436 constexpr std::array<std::size_t, 4> perm{0, 2, 1, 3};
437 return o / 32 * 32 + perm[o % 32 / 8] * 8 + o % 8;
438 }

440 // The two-weight (TL1-coded) part of a TL2 matrix: BY = 32, bm = 32, by = 4, as preprocess_two_weights_tl2
441 constexpr void tl2_two(std::span<const std::uint8_t> codes, std::size_t M, std::size_t K, std::size_t BM, std::span<std::uint8_t> out)
442 {
443 const auto steps = tl_tiling(M, K, 2, BM, 32, 32, 4);
444 for (std::size_t o = 0; o < out.size(); ++o) {
445 const std::size_t b = swap_groups(o);
446 out[o] = static_cast<std::uint8_t>((codes[source_of(2 * b, steps)] << 4) + codes[source_of(2 * b + 1, steps)]);
447 }
448 }

463constexpr void tl1_pack(std::span<const std::int8_t> t, std::size_t M, std::size_t K, lut_tiles tiles, std::span<std::uint8_t> out) noexcept
464{
465 using namespace detail::bitnet;
466 const std::size_t BM = tiles.BM, BY = tiles.BK, bm = tiles.bm, by = 256 / bm;
467 assert(t.size() == M * K && out.size() == tl1_packed_bytes(M, K) && M % BM == 0 && K % BY == 0 && BM % bm == 0 && BY % by == 0
468 && bm % 16 == 0 && by % 4 == 0);
469 const auto tiling = tl_tiling(M, K, 2, BM, BY, bm, by);
470 const std::array<detail::bitnet::step, 6> steps{tiling[0], tiling[1], tiling[2], tiling[3],
471 detail::bitnet::step{7, {M / BM, K / BY, BM / bm, BY / by, bm / 16, 16, by / 2}, {0, 1, 2, 3, 4, 6, 5}},
472 detail::bitnet::step{8, {M / BM, K / BY, BM / bm, BY / by, bm / 16, by / 4, 2, 16}, {0, 1, 2, 3, 4, 5, 7, 6}}};
473 const auto code = [&](std::size_t pair) { return tl1_code(t[2 * pair], t[2 * pair + 1]); };
474 for (std::size_t o = 0; o < out.size(); ++o)
475 out[o] = static_cast<std::uint8_t>((code(source_of(2 * o, steps)) << 4) + code(source_of(2 * o + 1, steps)));
476}

497constexpr void tl2_pack(std::span<const std::int8_t> t, std::size_t M, std::size_t K, lut_tiles tiles, std::span<std::uint8_t> out,
498 std::span<std::uint8_t> scratch = {}) noexcept
499{
500 using namespace detail::bitnet;
501 const std::size_t BM = tiles.BM, BY = tiles.BK, bm = tiles.bm, by = 192 / bm;
502 const std::size_t K3 = K - K % BY, K2 = K % BY;
503 assert(t.size() == M * K && out.size() == tl2_packed_bytes(M, K) && bm == 32 && M % BM == 0 && BM % bm == 0 && BY % (4 * by) == 0
504 && K2 % 32 == 0 && scratch.size() == M * K2 / 2);
505 const auto triple = [&](std::size_t i) { // triple i of the first K3 columns, row-major
506 const std::size_t r = i / (K3 / 3), c = 3 * (i % (K3 / 3));
507 const std::int8_t* w = &t[r * K + c];
508 return tl2_code(w[0], w[1], w[2]);
509 };
510 // magnitudes: two triples a byte, rows of bm bytes with groups swapped
511 const auto mags = tl_tiling(M, K3, 3, BM, BY, bm, by);
512 const std::size_t n_mag = M * K3 / 6;
513 for (std::size_t o = 0; o < n_mag; ++o) {
514 const std::size_t b = swap_groups(o);
515 out[o] = static_cast<std::uint8_t>((triple(source_of(2 * b, mags)).magnitude << 4) + triple(source_of(2 * b + 1, mags)).magnitude);
516 }
517 // signs: 16 bits a uint16, written little-endian
518 const std::array<detail::bitnet::step, 5> signs{mags[0], mags[1], mags[2],
519 detail::bitnet::step{6, {M / BM, K3 / BY, BM / bm, BY / (by * 4), by / 3 * 4, bm}, {0, 1, 2, 3, 5, 4}},
520 detail::bitnet::step{6, {M / BM, K3 / BY, BM / bm, BY / (by * 4), bm, by / 3 * 4}, {0, 1, 2, 3, 5, 4}}};
521 const std::size_t n_sign = M * K3 / 3 / 8;
522 for (std::size_t u = 0; u < n_sign / 2; ++u) {
523 const std::size_t outer = u / (bm / 2), c = u % (bm / 2);
524 unsigned word = 0;
525 for (std::size_t i = 0; i < 16; ++i)
526 word |= unsigned(triple(source_of(outer * 16 * (bm / 2) + i * (bm / 2) + c, signs)).negative) << (15 - i);
527 out[n_mag + 2 * u] = static_cast<std::uint8_t>(word & 0xff);
528 out[n_mag + 2 * u + 1] = static_cast<std::uint8_t>(word >> 8);
529 }
530 std::size_t end = n_mag + n_sign;
531 if (K2 != 0) {
532 for (std::size_t r = 0; r < M; ++r)
533 for (std::size_t p = 0; p < K2 / 2; ++p)
534 scratch[r * K2 / 2 + p] = tl1_code(t[r * K + K3 + 2 * p], t[r * K + K3 + 2 * p + 1]);
535 tl2_two(scratch, M, K2, BM, out.subspan(end, M * K2 / 4));
536 end += M * K2 / 4;
537 }
538 assert(end <= out.size());
539 for (std::size_t o = end; o < out.size(); ++o)
540 out[o] = 0;
541}
