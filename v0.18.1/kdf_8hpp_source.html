104template <hash_function H, hash_input P, hash_input I>
105constexpr result<void, kdf_error> hkdf_expand(const P& prk, const I& info, std::span<std::byte> out) noexcept
106{
107 if (out.empty())
108 return err(kdf_error::empty_output);
109 if (out.size() > 255 * H::digest_size)
110 return err(kdf_error::output_too_long);
111 const hmac<H> keyed{prk};
112 digest_of_t<H> t{};
113 std::size_t done = 0;
114 for (std::uint8_t counter = 1; done < out.size(); ++counter) {
115 hmac<H> m = keyed;
116 if (counter > 1)
117 m.update(t);
118 m.update(info);
119 m.update(std::array<std::uint8_t, 1>{counter});
120 t = m.finish();
121 for (std::size_t i = 0; i < H::digest_size && done < out.size(); ++i)
122 out[done++] = t.bytes[i];
123 }
124 secure_wipe(t);
125 return ok();
126}

180 // PBKDF2's blocks (RFC 8018 5.2): T_i for i = 1, 2, ... until `length` bytes; `salt(m)` feeds the salt into an
181 // HMAC, `emit(offset, T_i, count)` takes the first `count` bytes of T_i for output offset `offset`.
182 template <hash_function H, hash_input P, typename Salt, typename Emit>
183 constexpr void pbkdf2_blocks(const P& password, Salt&& salt, std::uint32_t iterations, std::size_t length, Emit&& emit) noexcept
184 {
185 const hmac<H> keyed{password};
186 digest_of_t<H> u{}, t{};
187 std::size_t done = 0;
188 for (std::uint32_t block = 1; done < length; ++block) {
189 hmac<H> m = keyed;
190 salt(m);
191 const std::array<std::uint8_t, 4> index{static_cast<std::uint8_t>(block >> 24), static_cast<std::uint8_t>(block >> 16),
192 static_cast<std::uint8_t>(block >> 8), static_cast<std::uint8_t>(block)};
193 m.update(index);
194 u = m.finish();
195 t = u;
196 for (std::uint32_t i = 1; i < iterations; ++i) {
197 hmac<H> next = keyed;
198 next.update(u);
199 u = next.finish();
200 for (std::size_t j = 0; j < H::digest_size; ++j)
201 t.bytes[j] ^= u.bytes[j];
202 }
203 const std::size_t count = length - done < H::digest_size ? length - done : H::digest_size;
204 emit(done, t, count);
205 done += count;
206 }
207 secure_wipe(u);
208 secure_wipe(t);
209 }

222template <hash_function H, hash_input P, hash_input S>
223constexpr result<void, kdf_error> pbkdf2_hmac(const P& password, const S& salt, std::uint32_t iterations,
224 std::span<std::byte> out) noexcept
225{
226 if (iterations == 0)
227 return err(kdf_error::zero_iterations);
228 if (out.empty())
229 return err(kdf_error::empty_output);
230 if (out.size() / H::digest_size >= 0xffffffffull)
231 return err(kdf_error::output_too_long);
232 detail::pbkdf2_blocks<H>(password, [&salt](hmac<H>& m) { m.update(salt); }, iterations, out.size(),
233 [&out](std::size_t offset, const digest_of_t<H>& t, std::size_t count) {
234 for (std::size_t j = 0; j < count; ++j)
235 out[offset + j] = t.bytes[j];
236 });
237 return ok();
238}

248template <hash_function H, std::size_t L, hash_input P, hash_input S>
249[[nodiscard]] constexpr result<digest<L>, kdf_error> pbkdf2_hmac(const P& password, const S& salt, std::uint32_t iterations) noexcept
250{
251 static_assert(L >= 1, "snicholls::pbkdf2_hmac: L is at least 1 byte");
252 digest<L> out{};
