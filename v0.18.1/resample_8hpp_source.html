71 static constexpr resample_ratio of(std::size_t up, std::size_t down) noexcept
72 {
73 assert(up >= 1 && down >= 1);
74 const std::size_t g = std::gcd(up, down);

109template <math::math_number T, typename W = kaiser>
110constexpr void resample_kernel(std::size_t up, std::size_t down, std::span<T> h, const W& window = W{5.0}) noexcept
111{
112 const resample_ratio r = resample_ratio::of(up, down);
113 assert(h.size() == r.taps());
114 if (r.half_length() == 0) {
115 h[0] = T(1);
116 return;
117 }
118 const std::size_t m = r.up > r.down ? r.up : r.down;
119 fir_design(h, fir_band::lowpass, T(1) / T(static_cast<long long>(m)), T(0), window);
120 for (auto& v : h)
121 v *= T(static_cast<long long>(r.up));
122}

131template <math::math_number T>
132constexpr void resample_poly(std::span<const T> x, std::size_t up, std::size_t down, std::span<const std::type_identity_t<T>> h,
133 std::span<std::type_identity_t<T>> y) noexcept
134{
135 const resample_ratio r = resample_ratio::of(up, down);
136 assert(h.size() == r.taps() && y.size() == resample_size(x.size(), up, down));
137 const std::size_t taps = h.size(), half = r.half_length(), n = x.size();
138 for (std::size_t o = 0; o < y.size(); ++o) {
139 // y[o] = sum_i x[i] h[t - i up], t = o down + half: the centre tap on output o's position.
140 const std::size_t t = o * r.down + half;
141 const std::size_t lo = t + 1 > taps ? (t + 1 - taps + r.up - 1) / r.up : 0;
142 const std::size_t hi = t / r.up < n ? t / r.up : n - 1;
143 T acc = T(0);
144 for (std::size_t i = lo; i <= hi && n > 0; ++i)
145 acc += x[i] * h[t - i * r.up];
146 y[o] = acc;
147 }
148}
