79constexpr void keccak_f1600(std::array<std::uint64_t, 25>& a) noexcept
80{
81 constexpr std::uint64_t rc[24] = {
82 0x0000000000000001, 0x0000000000008082, 0x800000000000808a, 0x8000000080008000, 0x000000000000808b,
83 0x0000000080000001, 0x8000000080008081, 0x8000000000008009, 0x000000000000008a, 0x0000000000000088,
84 0x0000000080008009, 0x000000008000000a, 0x000000008000808b, 0x800000000000008b, 0x8000000000008089,
85 0x8000000000008003, 0x8000000000008002, 0x8000000000000080, 0x000000000000800a, 0x800000008000000a,
86 0x8000000080008081, 0x8000000000008080, 0x0000000080000001, 0x8000000080008008};
87 // rho and pi together: lane pi_lane[i] takes the previous lane, rotated by rho[i]
88 constexpr unsigned rho[24] = {1, 3, 6, 10, 15, 21, 28, 36, 45, 55, 2, 14, 27, 41, 56, 8, 25, 43, 62, 18, 39, 61, 20, 44};
89 constexpr std::size_t pi_lane[24] = {10, 7, 11, 17, 18, 3, 5, 16, 8, 21, 24, 4, 15, 23, 19, 13, 12, 2, 20, 14, 22, 9, 6, 1};
90 for (std::size_t round = 0; round < 24; ++round) {
91 std::uint64_t c[5]{};
92 for (std::size_t x = 0; x < 5; ++x)
93 c[x] = a[x] ^ a[x + 5] ^ a[x + 10] ^ a[x + 15] ^ a[x + 20];
94 for (std::size_t x = 0; x < 5; ++x) {
95 const std::uint64_t d = c[(x + 4) % 5] ^ detail::rotl<std::uint64_t>(c[(x + 1) % 5], 1);
96 for (std::size_t y = 0; y < 25; y += 5)
97 a[y + x] ^= d;
98 }
99 std::uint64_t t = a[1];
100 for (std::size_t i = 0; i < 24; ++i) {
101 const std::uint64_t next = a[pi_lane[i]];
102 a[pi_lane[i]] = detail::rotl<std::uint64_t>(t, rho[i]);
103 t = next;
104 }
105 for (std::size_t y = 0; y < 25; y += 5) {
106 const std::uint64_t r0 = a[y], r1 = a[y + 1], r2 = a[y + 2], r3 = a[y + 3], r4 = a[y + 4];
107 a[y] = r0 ^ (~r1 & r2);
108 a[y + 1] = r1 ^ (~r2 & r3);
109 a[y + 2] = r2 ^ (~r3 & r4);
110 a[y + 3] = r3 ^ (~r4 & r0);
111 a[y + 4] = r4 ^ (~r0 & r1);
112 }
113 a[0] ^= rc[round];
114 }
115}

119 // The sponge's state, absorption and squeezing, shared by the fixed-length hashes and the XOFs.
120 template <std::size_t Rate, std::uint8_t Pad>
121 class keccak_core
122 {
123 static_assert(Rate % 8 == 0 && Rate > 0 && Rate < 200, "snicholls::keccak: the rate is a whole number of lanes");
124
125 public:
126 template <typename U>
127 constexpr void absorb(const U* p, std::size_t n) noexcept
128 {
129 in_.feed(p, n, [this](const auto* b) { this->absorb_block(b); });
130 }
131
132 // Pads the last block and permutes; the first squeezed byte is lane 0's lowest.
133 constexpr void pad() noexcept
134 {
135 for (std::size_t i = in_.fill; i < Rate; ++i)
136 in_.buf[i] = 0;
137 in_.buf[in_.fill] ^= Pad;
138 in_.buf[Rate - 1] ^= 0x80;
139 absorb_block(in_.buf.data());
140 position_ = 0;
141 }
142
143 template <typename Out>
144 constexpr void squeeze(Out* out, std::size_t n) noexcept
145 {
146 for (std::size_t i = 0; i < n; ++i) {
147 if (position_ == Rate) {
148 keccak_f1600(a_);
149 position_ = 0;
150 }
151 out[i] = static_cast<Out>(static_cast<std::uint8_t>(a_[position_ / 8] >> (8 * (position_ % 8))));
152 ++position_;
153 }
154 }
155
156 private:
157 template <typename U>
158 constexpr void absorb_block(const U* b) noexcept
159 {
160 for (std::size_t i = 0; i < Rate / 8; ++i)
161 a_[i] ^= load_le_word<std::uint64_t>(b + 8 * i);
162 keccak_f1600(a_);
163 }
164
165 std::array<std::uint64_t, 25> a_{};
166 block_feeder<Rate> in_{};
167 std::size_t position_ = 0;
168 };

170} // namespace detail

177template <std::size_t Rate, std::uint8_t Pad, std::size_t DigestSize>
178class keccak_sponge : public detail::one_shot_hash<keccak_sponge<Rate, Pad, DigestSize>>
179{
180 static_assert(DigestSize <= Rate, "snicholls::keccak_sponge: the digest fits in one squeeze");
181
182public:
184 static constexpr std::size_t block_size = Rate;
186 static constexpr std::size_t digest_size = DigestSize;
188 using digest_type = digest<DigestSize>;

191 constexpr keccak_sponge() noexcept = default;
192
195 [[nodiscard]] constexpr digest_type finish() noexcept

198 core_.pad();
199 core_.squeeze(out.bytes.data(), DigestSize);

212};

220template <std::size_t Rate>

233 {

236 squeezing_ = true;
