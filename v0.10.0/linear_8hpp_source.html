189template <typename S, row X, mutable_row Y, row B = std::span<const row_value_t<X>>>
190 requires std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<B>>
191constexpr void linear(const dense_weights<S>& w, const X& x, Y&& y, const B& bias = B{})
192{
193 using T = row_value_t<X>;
194 const auto in = as_row(x);
195 const auto out = as_mutable_row(y);
196 const auto b = as_row(bias);
197 assert(in.size() == w.cols && out.size() == w.rows && w.data.size() == w.rows * w.cols && (b.empty() || b.size() == w.rows));
198 for (std::size_t i = 0; i < w.rows; ++i) {
199 const auto r = w.row(i);
200 T acc = b.empty() ? T(0) : b[i];
201 for (std::size_t j = 0; j < w.cols; ++j) {
202 const T p = detail::linear::widen<T>(r[j]) * in[j];
203 acc += p;
204 }
205 out[i] = acc;
206 }
207}

221template <typename S, row X, row DY, mutable_row DX, mutable_row DW, mutable_row DB = std::span<row_value_t<X>>>
222 requires std::same_as<row_value_t<X>, row_value_t<DY>> && std::same_as<row_value_t<X>, row_value_t<DX>>
223 && std::same_as<row_value_t<X>, row_value_t<DW>> && std::same_as<row_value_t<X>, row_value_t<DB>>
224constexpr void linear_backward(const dense_weights<S>& w, const X& x, const DY& dy, DX&& dx, DW&& dw, DB&& db = DB{})
225{
226 using T = row_value_t<X>;
227 const auto in = as_row(x);
228 const auto g = as_row(dy);
229 const auto gx = as_mutable_row(dx);
230 const auto gw = as_mutable_row(dw);
231 const auto gb = as_mutable_row(db);
232 assert(in.size() == w.cols && g.size() == w.rows && w.data.size() == w.rows * w.cols && gw.size() == w.rows * w.cols
233 && (gx.empty() || gx.size() == w.cols) && (gb.empty() || gb.size() == w.rows));
234 for (std::size_t i = 0; i < w.rows; ++i) {
235 const auto r = w.row(i);
236 const T d = g[i];
237 if (!gx.empty())
238 for (std::size_t j = 0; j < w.cols; ++j) {
239 const T p = detail::linear::widen<T>(r[j]) * d;
240 gx[j] += p;
241 }
242 for (std::size_t j = 0; j < w.cols; ++j) {
243 const T p = d * in[j];
244 gw[i * w.cols + j] += p;
245 }
246 if (!gb.empty())
247 gb[i] += d;
248 }
249}

256template <quant_block Block, std::size_t M, mutable_row Y, row B = std::span<const float>>
257 requires std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
258constexpr void linear(const quantized_weights<Block>& w, const std::array<block_q8_0, M>& x, Y&& y, const B& bias = B{})
259{
260 const auto out = as_mutable_row(y);
261 const auto b = as_row(bias);
262 assert(M * quant_block_size == w.cols && out.size() == w.rows && w.blocks.size() == w.rows * M && (b.empty() || b.size() == w.rows));
263 for (std::size_t i = 0; i < w.rows; ++i) {
264 const float d = dot(w.row(i), std::span<const block_q8_0>(x));
265 out[i] = b.empty() ? d : b[i] + d;
266 }
267}

283 // C = alpha op(A) op(B) + beta C. The shapes are right by construction and C never overlaps A or B, so the
284 // backend's product always succeeds
285 template <typename B, typename T, typename L, typename R, typename C>
286 constexpr void product(const B& backend, T alpha, const L& lhs, const R& rhs, T beta, const C& c)
287 {
288 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
289 assert(done.is_ok());
290 }

325template <typename B, math::math_number T>
326constexpr void add_bias_rows(const B&, std::span<T> ys, std::type_identity_t<std::span<const T>> bias)
327{
328 const std::size_t r = bias.size();
329 assert(r > 0 && ys.size() % r == 0);
330 for (std::size_t t = 0; t < ys.size() / r; ++t)
331 for (std::size_t i = 0; i < r; ++i)
332 ys[t * r + i] += bias[i];
333}

388template <math::math_number T, typename B>
389 requires linalg::blas_backend<B, T>
390constexpr void linear_sequence(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
391 std::type_identity_t<std::span<T>> ys, std::type_identity_t<std::span<const T>> bias = {})
392{
393 assert(w.cols > 0 && xs.size() % w.cols == 0);
394 const std::size_t n = xs.size() / w.cols;
395 assert(ys.size() == n * w.rows && w.data.size() == w.rows * w.cols && (bias.empty() || bias.size() == w.rows));
396 if (n == 0 || w.rows == 0)
397 return;
398 using detail::linear::view;
399 detail::linear::product(backend, T(1), view(xs, n, w.cols), linalg::transposed(view(w.data, w.rows, w.cols)), T(0),
400 view(ys, n, w.rows));
401 if (!bias.empty())
402 add_bias_rows(backend, ys, bias);
403}

414template <math::math_number T, typename B>
415 requires linalg::blas_backend<B, T>
416constexpr void linear_sequence_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
417 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
418 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db = {})
419{
420 assert(w.cols > 0 && xs.size() % w.cols == 0);
421 const std::size_t n = xs.size() / w.cols;
422 assert(dys.size() == n * w.rows && (dxs.empty() || dxs.size() == xs.size()) && dw.size() == w.rows * w.cols
423 && (db.empty() || db.size() == w.rows));
424 if (n == 0 || w.rows == 0)
425 return;
426 using detail::linear::view;
427 const auto dy = view(dys, n, w.rows);
428 if (!dxs.empty())
429 detail::linear::product(backend, T(1), dy, view(w.data, w.rows, w.cols), T(1), view(dxs, n, w.cols));
430 detail::linear::product(backend, T(1), linalg::transposed(dy), view(xs, n, w.cols), T(1), view(dw, w.rows, w.cols));
431 if (!db.empty())
432 sum_rows(backend, dys, db);
433}
