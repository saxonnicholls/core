96template <std::size_t Rank>
97 requires(Rank >= 1 && Rank <= 3)
98struct pool_shape
99{
100 std::size_t channels = 0;
101 std::array<std::size_t, Rank> input{};
102 std::array<std::size_t, Rank> kernel{};
103 std::array<std::size_t, Rank> stride = detail::conv::ones<Rank>();
104 std::array<std::size_t, Rank> padding{};
105 bool count_padding = false;
106
109 [[nodiscard]] constexpr bool valid() const noexcept
110 {
111 if (channels == 0)
112 return false;
113 for (std::size_t d = 0; d < Rank; ++d)
114 if (kernel[d] == 0 || stride[d] == 0 || input[d] + 2 * padding[d] < kernel[d])
115 return false;
116 return true;
117 }
118
120 [[nodiscard]] constexpr std::array<std::size_t, Rank> output() const noexcept
121 {
122 std::array<std::size_t, Rank> o{};
123 for (std::size_t d = 0; d < Rank; ++d)
124 o[d] = (input[d] + 2 * padding[d] - kernel[d]) / stride[d] + 1;
125 return o;
126 }
127
129 [[nodiscard]] constexpr std::size_t x_size() const noexcept { return channels * detail::conv::product(input); }
132 [[nodiscard]] constexpr std::size_t y_size() const noexcept { return channels * detail::conv::product(output()); }
133
136 friend constexpr bool operator==(const pool_shape&, const pool_shape&) = default;
137};

180 // The input position tap k of output o reads, or `none` in the padding
181 template <std::size_t Rank>
182 constexpr std::size_t source(const pool_shape<Rank>& s, const std::array<std::size_t, Rank>& o, const std::array<std::size_t, Rank>& k,
183 std::size_t none) noexcept
184 {
185 std::size_t at = 0;
186 for (std::size_t d = 0; d < Rank; ++d) {
187 const std::size_t padded = o[d] * s.stride[d] + k[d];
188 if (padded < s.padding[d] || padded - s.padding[d] >= s.input[d])
189 return none;
190 at = at * s.input[d] + (padded - s.padding[d]);
191 }
192 return at;
193 }

195 // One output of a pool: the value, the first maximal tap (max) and the divisor (avg)
196 template <typename T>
197 struct pooled
198 {
199 T value;
200 std::size_t argmax;
201 std::size_t count;
202 };

204 template <std::size_t Rank, typename T>
205 constexpr pooled<T> pool_one(const pool_shape<Rank>& s, pool_kind kind, std::span<const T> plane, const std::array<std::size_t, Rank>& o)
206 {
207 const std::size_t is = conv::product(s.input), ks = conv::product(s.kernel);
208 pooled<T> r{kind == pool_kind::max ? std::numeric_limits<T>::lowest() : T(0), is, 0};
209 for (std::size_t t = 0; t < ks; ++t) {
210 const std::size_t at = source(s, o, conv::unflatten<Rank>(t, s.kernel), is);
211 if (at == is)
212 continue;
213 ++r.count;
214 if (kind == pool_kind::avg)
215 r.value += plane[at];
216 else if (r.argmax == is || plane[at] > r.value) {
217 r.value = plane[at];
218 r.argmax = at;
219 }
220 }
221 if (kind == pool_kind::avg)
222 r.count = s.count_padding ? ks : r.count;
223 return r;
224 }

231 // Keys' cubic convolution kernel with a = -0.75, as ggml writes it
232 template <typename T>
233 constexpr std::array<T, 4> cubic_weights(T x) noexcept
234 {
235 const T a = T(-0.75);
236 const auto w1 = [a](T v) { return ((a + T(2)) * v - (a + T(3))) * v * v + T(1); };
237 const auto w2 = [a](T v) { return ((a * v - T(5) * a) * v + T(8) * a) * v - T(4) * a; };
238 return {w2(x + T(1)), w1(x), w1(T(1) - x), w2(T(2) - x)};
239 }

241 // Visit every (output index, input index, weight) of an upscale of one plane
242 template <typename T, typename F>
243 constexpr void upscale_taps(upscale_mode mode, bool align_corners, std::size_t in_h, std::size_t in_w, std::size_t out_h, std::size_t out_w, F&& f)
244 {
245 T sf0 = T(out_w) / T(in_w), sf1 = T(out_h) / T(in_h), offset = T(0.5);
246 if (align_corners) {
247 offset = T(0);
248 sf0 = out_w > 1 && in_w > 1 ? T(out_w - 1) / T(in_w - 1) : sf0;
249 sf1 = out_h > 1 && in_h > 1 ? T(out_h - 1) / T(in_h - 1) : sf1;
250 }
251 for (std::size_t i1 = 0; i1 < out_h; ++i1)
252 for (std::size_t i0 = 0; i0 < out_w; ++i0) {
253 const std::size_t o = i1 * out_w + i0;
254 if (mode == upscale_mode::nearest) {
255 const auto y0 = static_cast<std::size_t>(T(i1) / sf1), x0 = static_cast<std::size_t>(T(i0) / sf0);
256 f(o, (y0 < in_h ? y0 : in_h - 1) * in_w + (x0 < in_w ? x0 : in_w - 1), T(1));
257 continue;
258 }
259 const T y = (T(i1) + offset) / sf1 - offset, x = (T(i0) + offset) / sf0 - offset;
260 const auto fy = static_cast<std::int64_t>(math::floor(y)), fx = static_cast<std::int64_t>(math::floor(x));
261 if (mode == upscale_mode::bilinear) { // the corners clamped, the fractions from the clamped corner
262 const std::size_t y0 = clamp_index(fy, in_h), y1 = clamp_index(fy + 1, in_h);
263 const std::size_t x0 = clamp_index(fx, in_w), x1 = clamp_index(fx + 1, in_w);
264 T dy = y - T(y0), dx = x - T(x0);
265 dy = dy < T(0) ? T(0) : dy > T(1) ? T(1) : dy;
266 dx = dx < T(0) ? T(0) : dx > T(1) ? T(1) : dx;
267 f(o, y0 * in_w + x0, (T(1) - dx) * (T(1) - dy));
268 f(o, y0 * in_w + x1, dx * (T(1) - dy));
269 f(o, y1 * in_w + x0, (T(1) - dx) * dy);
270 f(o, y1 * in_w + x1, dx * dy);
271 continue;
272 }
273 const auto wy = cubic_weights(y - T(fy)), wx = cubic_weights(x - T(fx)); // bicubic
274 for (std::int64_t j = 0; j < 4; ++j)
275 for (std::int64_t i = 0; i < 4; ++i)
276 f(o, clamp_index(fy + j - 1, in_h) * in_w + clamp_index(fx + i - 1, in_w),
277 wy[static_cast<std::size_t>(j)] * wx[static_cast<std::size_t>(i)]);
278 }
279 }

281 // One plane of upscale, each output by ggml's own expression (ggml_compute_forward_upscale_f32)
282 template <typename T>
283 constexpr void upscale_plane(upscale_mode mode, bool align_corners, std::span<const T> src, std::size_t in_h, std::size_t in_w,
284 std::span<T> dst, std::size_t out_h, std::size_t out_w)
285 {
286 T sf0 = T(out_w) / T(in_w), sf1 = T(out_h) / T(in_h), offset = T(0.5);
287 if (align_corners) {
288 offset = T(0);
289 sf0 = out_w > 1 && in_w > 1 ? T(out_w - 1) / T(in_w - 1) : sf0;
290 sf1 = out_h > 1 && in_h > 1 ? T(out_h - 1) / T(in_h - 1) : sf1;
291 }
292 const auto at = [&](std::int64_t y, std::int64_t x) { return src[clamp_index(y, in_h) * in_w + clamp_index(x, in_w)]; };
293 for (std::size_t i1 = 0; i1 < out_h; ++i1)
294 for (std::size_t i0 = 0; i0 < out_w; ++i0) {
295 T& out = dst[i1 * out_w + i0];
296 if (mode == upscale_mode::nearest) {
297 out = at(static_cast<std::int64_t>(T(i1) / sf1), static_cast<std::int64_t>(T(i0) / sf0));
298 continue;
299 }
300 const T y = (T(i1) + offset) / sf1 - offset, x = (T(i0) + offset) / sf0 - offset;
301 const auto fy = static_cast<std::int64_t>(math::floor(y)), fx = static_cast<std::int64_t>(math::floor(x));
302 if (mode == upscale_mode::bilinear) {
303 const std::size_t y0 = clamp_index(fy, in_h), x0 = clamp_index(fx, in_w);
304 T dy = y - T(y0), dx = x - T(x0);
305 dy = dy < T(0) ? T(0) : dy > T(1) ? T(1) : dy;
306 dx = dx < T(0) ? T(0) : dx > T(1) ? T(1) : dx;
307 const T a = at(fy, fx), b = at(fy, fx + 1), c = at(fy + 1, fx), d = at(fy + 1, fx + 1);
308 const T ta = a * (T(1) - dx) * (T(1) - dy), tb = b * dx * (T(1) - dy), tc = c * (T(1) - dx) * dy, td = d * dx * dy;
309 out = ta + tb + tc + td;
310 continue;
311 }
312 const auto wx = cubic_weights(x - T(fx)), wy = cubic_weights(y - T(fy));
313 std::array<T, 4> rows{};
314 for (std::int64_t j = 0; j < 4; ++j) {
315 T r = T(0);
316 for (std::int64_t i = 0; i < 4; ++i) {
317 const T term = at(fy + j - 1, fx + i - 1) * wx[static_cast<std::size_t>(i)];
318 r = i == 0 ? term : r + term;
319 }
320 rows[static_cast<std::size_t>(j)] = r;
321 }
322 T v = T(0);
323 for (std::size_t j = 0; j < 4; ++j) {
324 const T term = rows[j] * wy[j];
325 v = j == 0 ? term : v + term;
326 }
327 out = v;
328 }
329 }

331 template <std::size_t Rank>
332 constexpr std::size_t flat(const std::array<std::size_t, Rank>& c, const std::array<std::size_t, Rank>& e) noexcept
333 {
334 std::size_t at = 0;
335 for (std::size_t d = 0; d < Rank; ++d)
336 at = at * e[d] + c[d];

347template <std::size_t Rank, row X, mutable_row Y>
348 requires std::same_as<row_value_t<X>, row_value_t<Y>>
349constexpr void pool_forward(const pool_shape<Rank>& s, pool_kind kind, const X& x, Y&& y)
350{
351 using T = row_value_t<X>;
352 const auto in = as_row(x);
353 const auto out = as_mutable_row(y);
354 assert(s.valid() && in.size() == s.x_size() && out.size() == s.y_size());
355 const std::size_t is = detail::conv::product(s.input), os = detail::conv::product(s.output());
356 for (std::size_t c = 0; c < s.channels; ++c)
357 for (std::size_t p = 0; p < os; ++p) {
358 const auto r = detail::spatial::pool_one(s, kind, in.subspan(c * is, is), detail::conv::unflatten<Rank>(p, s.output()));
359 out[c * os + p] = kind == pool_kind::max ? r.value : r.count == 0 ? T(0) : r.value / T(r.count);
360 }
361}

371template <std::size_t Rank, row X, row DY, mutable_row DX>
372 requires std::same_as<row_value_t<X>, row_value_t<DY>> && std::same_as<row_value_t<X>, row_value_t<DX>>
373constexpr void pool_backward(const pool_shape<Rank>& s, pool_kind kind, const X& x, const DY& dy, DX&& dx)
374{
375 using T = row_value_t<X>;
376 const auto in = as_row(x);
377 const auto g = as_row(dy);
378 const auto gx = as_mutable_row(dx);
379 assert(s.valid() && in.size() == s.x_size() && g.size() == s.y_size() && gx.size() == s.x_size());
380 const std::size_t is = detail::conv::product(s.input), os = detail::conv::product(s.output()), ks = detail::conv::product(s.kernel);
381 for (T& v : gx)
382 v = T(0);
383 for (std::size_t c = 0; c < s.channels; ++c)
384 for (std::size_t p = 0; p < os; ++p) {
385 const auto o = detail::conv::unflatten<Rank>(p, s.output());
386 const auto r = detail::spatial::pool_one(s, kind, in.subspan(c * is, is), o);
387 if (kind == pool_kind::max) {
388 if (r.argmax != is)
389 gx[c * is + r.argmax] += g[c * os + p];
390 continue;
391 }
392 if (r.count == 0)
393 continue;
394 const T share = g[c * os + p] / T(r.count);
395 for (std::size_t t = 0; t < ks; ++t) {
396 const std::size_t at = detail::spatial::source(s, o, detail::conv::unflatten<Rank>(t, s.kernel), is);
397 if (at != is)
398 gx[c * is + at] += share;
399 }
400 }

412template <row X, mutable_row Y>
413 requires std::same_as<row_value_t<X>, row_value_t<Y>>
414constexpr void upscale(upscale_mode mode, bool align_corners, const X& x, std::size_t in_h, std::size_t in_w, Y&& y, std::size_t out_h,
415 std::size_t out_w)
416{
417 const auto in = as_row(x);
418 const auto out = as_mutable_row(y);
419 assert(in_h > 0 && in_w > 0 && in.size() % (in_h * in_w) == 0);
420 const std::size_t planes = in.size() / (in_h * in_w), ps = out_h * out_w;
421 assert(out.size() == planes * ps);
422 for (std::size_t c = 0; c < planes; ++c)
423 detail::spatial::upscale_plane(mode, align_corners, in.subspan(c * in_h * in_w, in_h * in_w), in_h, in_w,
424 out.subspan(c * ps, ps), out_h, out_w);
425}

467template <std::size_t Rank, row X, mutable_row Y>
468 requires std::same_as<row_value_t<X>, row_value_t<Y>>
469constexpr void pad(const std::array<std::size_t, Rank>& extents, const std::array<std::size_t, Rank>& left,
470 const std::array<std::size_t, Rank>& right, const X& x, Y&& y)
471{
472 using T = row_value_t<X>;
473 const auto in = as_row(x);
474 const auto out = as_mutable_row(y);
475 std::array<std::size_t, Rank> padded{};
476 for (std::size_t d = 0; d < Rank; ++d)
477 padded[d] = extents[d] + left[d] + right[d];
478 assert(in.size() == detail::conv::product(extents) && out.size() == detail::conv::product(padded));
479 for (std::size_t i = 0; i < out.size(); ++i) {
480 auto c = detail::conv::unflatten<Rank>(i, padded);
481 bool inside = true;
482 for (std::size_t d = 0; d < Rank; ++d) {
483 inside = inside && c[d] >= left[d] && c[d] - left[d] < extents[d];
484 c[d] -= inside ? left[d] : 0;
485 }
486 out[i] = inside ? in[detail::spatial::flat(c, extents)] : T(0);
487 }
488}

527{

529 const auto out = as_mutable_row(y);

532 for (std::size_t r = 0; r < rows; ++r)

535 out[r * w + i] = in[r * width + j];
536 }

571template <std::size_t Rank, row X, mutable_row Y>
572 requires std::same_as<row_value_t<X>, row_value_t<Y>>
573constexpr void roll(const std::array<std::size_t, Rank>& extents, const std::array<std::ptrdiff_t, Rank>& shifts, const X& x, Y&& y)
574{
575 const auto in = as_row(x);
576 const auto out = as_mutable_row(y);
577 assert(in.size() == detail::conv::product(extents) && out.size() == in.size());
578 for (std::size_t i = 0; i < out.size(); ++i) {
579 auto c = detail::conv::unflatten<Rank>(i, extents);
580 for (std::size_t d = 0; d < Rank; ++d) {
581 const auto n = static_cast<std::ptrdiff_t>(extents[d]);
582 const std::ptrdiff_t s = ((shifts[d] % n) + n) % n;
583 c[d] = static_cast<std::size_t>((static_cast<std::ptrdiff_t>(c[d]) - s + n) % n);
584 }
585 out[i] = in[detail::spatial::flat(c, extents)];
586 }

607template <row X, mutable_row W>
608 requires std::same_as<row_value_t<X>, row_value_t<W>>
609constexpr void window_partition(std::size_t height, std::size_t width, std::size_t channels, std::size_t window, const X& x, W&& windows)
610{
611 using T = row_value_t<X>;
612 const auto in = as_row(x);
613 const auto out = as_mutable_row(windows);
614 assert(window > 0 && in.size() == height * width * channels && out.size() == window_count(height, width, window) * window * window * channels);
615 const std::size_t across = (width + window - 1) / window;
616 for (std::size_t i = 0; i < out.size(); ++i) {
617 const std::size_t ch = i % channels, wx = (i / channels) % window, wy = (i / (channels * window)) % window,
618 win = i / (channels * window * window);
619 const std::size_t gy = (win / across) * window + wy, gx = (win % across) * window + wx;
620 out[i] = gy < height && gx < width ? in[(gy * width + gx) * channels + ch] : T(0);
621 }
622}

632template <row W, mutable_row X>
633 requires std::same_as<row_value_t<X>, row_value_t<W>>
634constexpr void window_unpartition([[maybe_unused]] std::size_t height, std::size_t width, std::size_t channels, std::size_t window, const W& windows, X&& x)
635{
636 const auto in = as_row(windows);
637 const auto out = as_mutable_row(x);
638 assert(window > 0 && out.size() == height * width * channels && in.size() == window_count(height, width, window) * window * window * channels);
639 const std::size_t across = (width + window - 1) / window;
640 for (std::size_t i = 0; i < out.size(); ++i) {
641 const std::size_t ch = i % channels, gx = (i / channels) % width, gy = i / (channels * width);
642 const std::size_t win = (gy / window) * across + gx / window;
643 out[i] = in[((win * window + gy % window) * window + gx % window) * channels + ch];
644 }
645}

652 requires std::same_as<row_value_t<R>, row_value_t<Out>>
653constexpr void rel_pos_gather(const R& rel, std::size_t size, Out&& out)
654{
655 const auto r = as_row(rel);
656 const auto o = as_mutable_row(out);
657 assert(size > 0 && r.size() % (2 * size - 1) == 0);
658 const std::size_t c = r.size() / (2 * size - 1);
659 assert(o.size() == size * size * c);

663 o[(q * size + k) * c + i] = r[(q + size - 1 - k) * c + i];

676template <row Q, row RH, row RW, mutable_row OH, mutable_row OW>
677 requires std::same_as<row_value_t<Q>, row_value_t<RH>> && std::same_as<row_value_t<Q>, row_value_t<RW>>
678 && std::same_as<row_value_t<Q>, row_value_t<OH>> && std::same_as<row_value_t<Q>, row_value_t<OW>>
679constexpr void rel_pos_scores(const Q& q, const RH& rh, const RW& rw, std::size_t h, std::size_t w, std::size_t c, OH&& rel_h, OW&& rel_w)
680{
681 using T = row_value_t<Q>;
682 const auto qs = as_row(q);
683 const auto a = as_row(rh);
684 const auto b = as_row(rw);
685 const auto oh = as_mutable_row(rel_h);
686 const auto ow = as_mutable_row(rel_w);
687 assert(qs.size() == h * w * c && a.size() == h * h * c && b.size() == w * w * c && oh.size() == h * w * h && ow.size() == h * w * w);
688 for (std::size_t y = 0; y < h; ++y)
689 for (std::size_t x = 0; x < w; ++x) {
690 const auto qv = qs.subspan((y * w + x) * c, c);
691 for (std::size_t k = 0; k < h; ++k) {
692 T s = T(0);
693 for (std::size_t i = 0; i < c; ++i)
694 s += qv[i] * a[(y * h + k) * c + i];
695 oh[(y * w + x) * h + k] = s;
696 }
697 for (std::size_t k = 0; k < w; ++k) {
698 T s = T(0);
699 for (std::size_t i = 0; i < c; ++i)
700 s += qv[i] * b[(x * w + k) * c + i];
701 ow[(y * w + x) * w + k] = s;
702 }
703 }
704}

713template <mutable_row A, row RH, row RW>
714 requires std::same_as<row_value_t<A>, row_value_t<RH>> && std::same_as<row_value_t<A>, row_value_t<RW>>
715constexpr void add_rel_pos(A&& attn, const RH& rel_h, const RW& rel_w, std::size_t h, std::size_t w)
716{
717 const auto s = as_mutable_row(attn);
718 const auto a = as_row(rel_h);
719 const auto b = as_row(rel_w);
720 const std::size_t n = h * w;
721 assert(s.size() == n * n && a.size() == n * h && b.size() == n * w);
722 for (std::size_t q = 0; q < n; ++q)
723 for (std::size_t k = 0; k < n; ++k) {
724 const auto t = s[q * n + k] + a[q * h + k / w];
725 s[q * n + k] = t + b[q * w + k % w];
726 }
727}

738constexpr void add_rel_pos_backward(const DA& dattn, std::size_t h, std::size_t w, DH&& drel_h, DW&& drel_w)
739{
740 const auto g = as_row(dattn);
741 const auto a = as_mutable_row(drel_h);
742 const auto b = as_mutable_row(drel_w);
743 const std::size_t n = h * w;
744 assert(g.size() == n * n && a.size() == n * h && b.size() == n * w);
745 for (std::size_t q = 0; q < n; ++q)
746 for (std::size_t k = 0; k < n; ++k) {
747 a[q * h + k / w] += g[q * n + k];
748 b[q * w + k % w] += g[q * n + k];
749 }
750}
