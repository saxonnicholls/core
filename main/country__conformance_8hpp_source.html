75 template <country_like T>
76 [[nodiscard]] constexpr std::optional<std::string_view> check_country(const T& t) noexcept
77 {
78 if (const auto failed = check_named(t))
79 return failed;
80 const std::string_view alpha2 = t.alpha2(), alpha3 = t.alpha3();
81 if (!detail::country_capitals(alpha2, 2))
82 return "country: alpha2() is not two capitals";
83 if (!detail::country_capitals(alpha3, 3))
84 return "country: alpha3() is not three capitals";
85 const auto numeric = static_cast<std::uint16_t>(t.numeric());
86 if (numeric < 1 || numeric > 999)
87 return "country: numeric() is not 1 to 999";
88 if (const std::size_t i = detail::iso3166_index(alpha2); i < detail::iso3166_rows.size()) {
89 if (detail::iso3166_rows[i].alpha3 != alpha3 || detail::iso3166_rows[i].numeric != numeric)
90 return "country: alpha3() or numeric() disagrees with ISO 3166-1";
91 }
92 for (const std::string_view c : t.currencies())
93 if (!detail::country_capitals(c, 3))
94 return "country: a currency is not three capitals";
95 if (static_cast<std::uint16_t>(t.calling_code()) > 999)
96 return "country: calling_code() is above 999";
97 return std::nullopt;
98 }
