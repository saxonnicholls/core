90 // The row's maximum, log of the shifted sum, and 1 / sum: everything the forward and backward need
91 template <typename T>
92 struct row_stats
93 {
94 T max;
95 T log_sum;
96 T inv_sum;
97 };
98
99 template <typename T>
100 constexpr row_stats<T> stats_of(std::span<const T> x)
101 {
102 const auto [m, s] = detail::softmax::shifted_sum_of(x);
103 return {m, math::log(s), T(1) / s};
104 }
105
106 template <typename T>
107 constexpr T entropy(std::span<const T> x, const row_stats<T>& st)
108 {
109 T h = T(0);
110 for (const T v : x) {
111 const T lp = (v - st.max) - st.log_sum;
112 h -= math::exp(v - st.max) * st.inv_sum * lp;
113 }
114 return h;
115 }
116} // namespace detail::logprobs

123template <row In>
124constexpr row_value_t<In> token_logprob(const In& logits, std::size_t target) noexcept(std::is_floating_point_v<row_value_t<In>>)
125{
126 const auto x = as_row(logits);
127 assert(target < x.size());
128 const auto [m, s] = detail::softmax::shifted_sum_of(x);
129 return (x[target] - m) - math::log(s);
130}

140template <row In, mutable_row Out>
141 requires std::same_as<row_value_t<In>, row_value_t<Out>>
142constexpr row_value_t<In> token_logprob_with_gradient(const In& logits, std::size_t target, Out&& dlogits,
143 std::type_identity_t<row_value_t<In>> dlogp) noexcept(std::is_floating_point_v<row_value_t<In>>)
144{
145 using T = row_value_t<In>;
146 const auto x = as_row(logits);
147 const auto g = as_mutable_row(dlogits);
148 assert(target < x.size() && g.size() == x.size());
149 const auto st = detail::logprobs::stats_of(x);
150 const T lp = (x[target] - st.max) - st.log_sum;
151 for (std::size_t i = 0; i < x.size(); ++i)
152 g[i] = dlogp * ((i == target ? T(1) : T(0)) - math::exp(x[i] - st.max) * st.inv_sum);
153 return lp;
154}

172template <math::math_number T, typename B>
173constexpr void token_logprobs_rows(const B&, std::type_identity_t<std::span<const T>> logits, std::span<const std::uint32_t> targets,
174 std::span<T> logps, std::type_identity_t<std::span<T>> entropies = {}) noexcept(std::is_floating_point_v<T>)
175{
176 const std::size_t n = targets.size();
177 assert(logps.size() == n && (entropies.empty() || entropies.size() == n) && (n == 0 || logits.size() % n == 0));
178 if (n == 0)
179 return;
180 const std::size_t v = logits.size() / n;
181 for (std::size_t t = 0; t < n; ++t) {
182 const auto x = logits.subspan(t * v, v);
183 assert(targets[t] < v);
184 const auto st = detail::logprobs::stats_of(x);
185 logps[t] = (x[targets[t]] - st.max) - st.log_sum;
186 if (!entropies.empty())
187 entropies[t] = detail::logprobs::entropy(x, st);
188 }
189}

197template <math::math_number T, typename B>
198constexpr void token_logprobs_rows_backward(const B&, std::span<T> logits, std::span<const std::uint32_t> targets,
199 std::type_identity_t<std::span<const T>> dlogps,
200 std::type_identity_t<std::span<const T>> dentropies = {}) noexcept(std::is_floating_point_v<T>)
201{
202 const std::size_t n = targets.size();
203 assert(dlogps.size() == n && (dentropies.empty() || dentropies.size() == n) && (n == 0 || logits.size() % n == 0));
204 if (n == 0)
205 return;
206 const std::size_t v = logits.size() / n;
207 for (std::size_t t = 0; t < n; ++t) {
208 const auto x = logits.subspan(t * v, v);
209 assert(targets[t] < v);
210 const auto st = detail::logprobs::stats_of(std::span<const T>(x));
211 const T dh = dentropies.empty() ? T(0) : dentropies[t];
212 const T h = dh == T(0) ? T(0) : detail::logprobs::entropy(std::span<const T>(x), st);
213 const T dl = dlogps[t];
214 for (std::size_t i = 0; i < v; ++i) {
215 const T lp = (x[i] - st.max) - st.log_sum;
216 const T p = math::exp(x[i] - st.max) * st.inv_sum;
217 const T g = dl * ((i == targets[t] ? T(1) : T(0)) - p);
218 x[i] = dh == T(0) ? g : g - dh * p * (lp + h);
219 }
220 }
221}

237template <math::math_number T, typename B>
238constexpr void sequence_logprobs(const B&, std::type_identity_t<std::span<const T>> token_logps, std::span<const std::uint8_t> mask,
239 std::span<T> out, logprob_reduction reduction) noexcept(std::is_floating_point_v<T>)
240{
241 const std::size_t rows = out.size();
242 assert(mask.size() == token_logps.size() && (rows == 0 || token_logps.size() % rows == 0));
243 if (rows == 0)
244 return;
245 const std::size_t len = token_logps.size() / rows;
246 for (std::size_t r = 0; r < rows; ++r) {
247 T s = T(0);
248 std::size_t c = 0;
249 for (std::size_t t = r * len; t < (r + 1) * len; ++t)
250 if (mask[t] != 0) {
251 s += token_logps[t];
252 ++c;
253 }
254 out[r] = reduction == logprob_reduction::sum ? s : (c == 0 ? T(0) : s / T(c));
255 }
256}

263template <math::math_number T, typename B>
264constexpr void sequence_logprobs_backward(const B&, std::span<const std::uint8_t> mask, std::type_identity_t<std::span<const T>> dseq,
265 std::span<T> dtoken, logprob_reduction reduction) noexcept(std::is_floating_point_v<T>)
266{
267 const std::size_t rows = dseq.size();
268 assert(mask.size() == dtoken.size() && (rows == 0 || dtoken.size() % rows == 0));
269 if (rows == 0)
270 return;
271 const std::size_t len = dtoken.size() / rows;
272 for (std::size_t r = 0; r < rows; ++r) {
273 const auto m = mask.subspan(r * len, len);
274 const std::size_t c = response_length(m);
275 const T g = reduction == logprob_reduction::sum ? dseq[r] : (c == 0 ? T(0) : dseq[r] / T(c));
276 for (std::size_t t = 0; t < len; ++t)
277 dtoken[r * len + t] = m[t] != 0 ? g : T(0);
278 }
279}
