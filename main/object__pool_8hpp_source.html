128 template <class... Args>
129 [[nodiscard]] T* create(Args&&... args)
130 {
131 slot* s = acquire();
132 if constexpr (std::is_nothrow_constructible_v<T, Args...>) {
133 T* p = ::new (static_cast<void*>(s->storage)) T(std::forward<Args>(args)...);
134 ++live_;
135 return p;
136 } else {
137 try {
138 T* p = ::new (static_cast<void*>(s->storage)) T(std::forward<Args>(args)...);
139 ++live_;
140 return p;
141 } catch (...) {
142 s->next_free = free_;
143 free_ = s;
144 throw;
145 }
146 }
147 }

151 void destroy(T* p) noexcept(std::is_nothrow_destructible_v<T>)
152 {
153 p->~T();
154 slot* s = reinterpret_cast<slot*>(p);
155 s->next_free = free_;

157 --live_;
158 }

161 void release_all() noexcept
162 {
163 free_ = nullptr;
164 next_ = chunks_.empty() ? nullptr : chunks_.front().get();
165 end_ = next_ ? next_ + ChunkSize : nullptr;
166 // Chained chunks after the first are reused through the free list.
167 for (std::size_t c = 1; c < chunks_.size(); ++c) {
168 slot* base = chunks_[c].get();
169 for (std::size_t i = ChunkSize; i-- > 0;) {
170 base[i].next_free = free_;
171 free_ = &base[i];
172 }
173 }
174 live_ = 0;
175 }
176
180 void reserve(std::size_t n)
181 {
182 while (capacity() < n) {
183 auto chunk = std::make_unique<slot[]>(ChunkSize);
184 slot* base = chunk.get();
185 chunks_.push_back(std::move(chunk));
186 for (std::size_t i = ChunkSize; i-- > 0;) {
187 base[i].next_free = free_;
188 free_ = &base[i];
189 }
190 }
191 }
192
195 [[nodiscard]] std::size_t live() const noexcept { return live_; }
198 [[nodiscard]] std::size_t capacity() const noexcept { return chunks_.size() * ChunkSize; }
199
200private:
201 slot* acquire()
202 {
203 if (free_) {
204 slot* s = free_;
205 free_ = s->next_free;
206 return s;
207 }
208 if (next_ == end_) {
209 // Own the chunk first, then point at it: if push_back threw after next_ was set, next_ would point into
210 // the chunk the unique_ptr just freed, and the next create() would hand out a dangling slot.
211 auto chunk = std::make_unique<slot[]>(ChunkSize);
212 slot* base = chunk.get();
213 chunks_.push_back(std::move(chunk));
214 next_ = base;
215 end_ = base + ChunkSize;
216 }
217 return next_++;
218 }
219
220 std::vector<std::unique_ptr<slot[]>> chunks_;
221 slot* free_ = nullptr;
222 slot* next_ = nullptr;
223 slot* end_ = nullptr;
224 std::size_t live_ = 0;
225};
