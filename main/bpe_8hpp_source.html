119 // The rank of parts i and i+1 merged with i+2: the bytes from part i to part i+3. A function, not a lambda
120 // capturing the lookup: MSVC cannot read a captured reference to a function during constant evaluation.
121 template <typename Rank>
122 constexpr std::uint32_t rank_after_merge(std::string_view piece, Rank& rank, const bpe_part* parts, std::size_t len, std::size_t i)
123 {
124 if (i + 3 < len)
125 return rank(piece.substr(parts[i].start, parts[i + 3].start - parts[i].start));
126 return bpe_no_rank;
127 }

142template <bpe_vocabulary Rank>
143constexpr result<std::size_t, bpe_error> bpe_encode_piece(std::string_view piece, Rank&& rank, std::span<std::uint32_t> out,
144 std::span<bpe_part> scratch)
145{
146 const std::size_t n = piece.size();
147 if (n == 0)
148 return ok(std::size_t{0});
149 if (out.empty())
150 return err(bpe_error::output_too_small);
151 if (const std::uint32_t whole = rank(piece); whole != bpe_no_rank) { // the common case: one token, no scratch
152 out[0] = whole;
153 return ok(std::size_t{1});
154 }
155 if (out.size() < n)
156 return err(bpe_error::output_too_small);
157 if (scratch.size() < bpe_scratch_size(n))
158 return err(bpe_error::scratch_too_small);
159
160 // parts[i] = (start of part i, rank of parts i and i+1 merged); the last two are sentinels
161 std::size_t len = n + 1;
162 bpe_part* parts = scratch.data();
163 std::uint32_t min_rank = bpe_no_rank;
164 std::size_t min_at = 0;
165 for (std::size_t i = 0; i + 1 < n; ++i) {
166 const std::uint32_t r = rank(piece.substr(i, 2));
167 if (r < min_rank) {
168 min_rank = r;
169 min_at = i;
170 }
171 parts[i] = {static_cast<std::uint32_t>(i), r};
172 }
173 parts[n - 1] = {static_cast<std::uint32_t>(n - 1), bpe_no_rank};
174 parts[n] = {static_cast<std::uint32_t>(n), bpe_no_rank};
175
176 while (min_rank != bpe_no_rank) {
177 const std::size_t i = min_at;
178 if (i > 0)
179 parts[i - 1].rank = detail::bpe::rank_after_merge(piece, rank, parts, len, i - 1);
180 parts[i].rank = detail::bpe::rank_after_merge(piece, rank, parts, len, i);
181 for (std::size_t k = i + 1; k + 1 < len; ++k) // remove part i+1
182 parts[k] = parts[k + 1];
183 --len;
184 min_rank = bpe_no_rank;
185 for (std::size_t k = 0; k + 1 < len; ++k)
186 if (parts[k].rank < min_rank) {
187 min_rank = parts[k].rank;
188 min_at = k;
189 }
190 }
191
192 // The tokens are the ranks of the parts. Only a part of one byte can be missing (a merged part is a token by
193 // construction), and then the vocabulary is not byte-level.
194 const std::size_t count = len - 1;
195 for (std::size_t k = 0; k < count; ++k) {
196 out[k] = rank(piece.substr(parts[k].start, parts[k + 1].start - parts[k].start));
197 if (out[k] == bpe_no_rank)
198 return err(bpe_error::unknown_bytes);
199 }
200 return ok(count);
201}

208 // Heap order: the lower rank, then the leftmost start
209 constexpr bool long_before(const bpe_part& a, const bpe_part& b) noexcept
210 {
211 return a.rank < b.rank || (a.rank == b.rank && a.start < b.start);
212 }
213
214 constexpr void long_push(bpe_part* heap, std::size_t& size, bpe_part e) noexcept
215 {
216 std::size_t i = size++;
217 while (i > 0) {
218 const std::size_t p = (i - 1) / 2;
219 if (!long_before(e, heap[p]))
220 break;
221 heap[i] = heap[p];
222 i = p;
223 }
224 heap[i] = e;
225 }
226
227 constexpr bpe_part long_pop(bpe_part* heap, std::size_t& size) noexcept
228 {
229 const bpe_part top = heap[0];
230 const bpe_part e = heap[--size];
231 std::size_t i = 0;
232 for (;;) {
233 std::size_t c = 2 * i + 1;
234 if (c >= size)
235 break;
236 if (c + 1 < size && long_before(heap[c + 1], heap[c]))
237 ++c;
238 if (!long_before(heap[c], e))
239 break;
240 heap[i] = heap[c];
241 i = c;
242 }
243 if (size > 0)
244 heap[i] = e;
245 return top;
246 }

248 // The rank of the part at i merged with the next: the bytes from i to the end of the next part
249 template <typename Rank>
250 constexpr std::uint32_t long_pair_rank(std::string_view piece, Rank& rank, const bpe_part* link, std::uint32_t i)
251 {
252 const std::uint32_t j = link[i].start;
253 if (j >= piece.size())
254 return bpe_no_rank;
255 return rank(piece.substr(i, link[j].start - i));

282template <bpe_vocabulary Rank>
283constexpr result<std::size_t, bpe_error> bpe_encode_long_piece(std::string_view piece, Rank&& rank, std::span<std::uint32_t> out,
284 std::span<bpe_part> scratch)
285{
286 const std::size_t n = piece.size();
287 if (n == 0)
288 return ok(std::size_t{0});
289 if (out.empty())
290 return err(bpe_error::output_too_small);
291 if (const std::uint32_t whole = rank(piece); whole != bpe_no_rank) { // the common case: one token, no scratch
292 out[0] = whole;
293 return ok(std::size_t{1});
294 }
295 if (out.size() < n)
296 return err(bpe_error::output_too_small);
297 if (scratch.size() < bpe_long_scratch_size(n))
298 return err(bpe_error::scratch_too_small);
299
300 // link[i] = {start of the next part, rank of part i merged with it}, for a part starting at i;
301 // back[i] = {start of the previous part (n for none), 1 while the part is alive, 0 once merged away}
302 bpe_part* link = scratch.data();
303 bpe_part* back = link + n;
304 bpe_part* heap = back + n; // {start, rank}, a min-heap by (rank, start)
305 std::size_t size = 0;
306
307 for (std::uint32_t i = 0; i < n; ++i) {
308 link[i] = {i + 1, bpe_no_rank};
309 back[i] = {i == 0 ? static_cast<std::uint32_t>(n) : i - 1, 1};
310 }
311 for (std::uint32_t i = 0; i + 1 < n; ++i)
312 if ((link[i].rank = rank(piece.substr(i, 2))) != bpe_no_rank)
313 detail::bpe::long_push(heap, size, {i, link[i].rank});
314
315 while (size > 0) {
316 const bpe_part e = detail::bpe::long_pop(heap, size);
317 const std::uint32_t i = e.start;
318 if (back[i].rank == 0 || link[i].rank != e.rank) // merged away, or its pair has grown since
319 continue;
320 const std::uint32_t j = link[i].start; // part j joins part i
321 const std::uint32_t k = link[j].start;
322 back[j].rank = 0;
323 link[i].start = k;
324 if (k < n)
325 back[k].start = i;
326 if ((link[i].rank = detail::bpe::long_pair_rank(piece, rank, link, i)) != bpe_no_rank)
327 detail::bpe::long_push(heap, size, {i, link[i].rank});
328 if (const std::uint32_t p = back[i].start; p < n)
329 if ((link[p].rank = detail::bpe::long_pair_rank(piece, rank, link, p)) != bpe_no_rank)
330 detail::bpe::long_push(heap, size, {p, link[p].rank});
331 }
332
333 // The tokens are the ranks of the parts; as in bpe_encode_piece, only a single byte can be missing
334 std::size_t count = 0;
335 for (std::uint32_t i = 0; i < n; i = link[i].start) {
336 out[count] = rank(piece.substr(i, link[i].start - i));
337 if (out[count] == bpe_no_rank)
338 return err(bpe_error::unknown_bytes);
339 ++count;
340 }
341 return ok(count);
342}

354constexpr result<std::size_t, bpe_error> bpe_decode(std::span<const std::uint32_t> tokens, Bytes&& bytes, std::span<char> out)
355{
356 std::size_t at = 0;
357 for (const std::uint32_t id : tokens) {
358 const std::string_view b = bytes(id);
359 if (b.empty())
360 return err(bpe_error::unknown_bytes);
361 if (out.size() - at < b.size())
362 return err(bpe_error::output_too_small);
363 for (const char c : b)
364 out[at++] = c;
365 }
366 return ok(at);
367}
