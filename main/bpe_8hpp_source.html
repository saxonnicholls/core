115template <bpe_vocabulary Rank>
116constexpr result<std::size_t, bpe_error> bpe_encode_piece(std::string_view piece, Rank&& rank, std::span<std::uint32_t> out,
117 std::span<bpe_part> scratch)
118{
119 const std::size_t n = piece.size();
120 if (n == 0)
121 return ok(std::size_t{0});
122 if (out.empty())
123 return err(bpe_error::output_too_small);
124 if (const std::uint32_t whole = rank(piece); whole != bpe_no_rank) { // the common case: one token, no scratch
125 out[0] = whole;
126 return ok(std::size_t{1});
127 }
128 if (out.size() < n)
129 return err(bpe_error::output_too_small);
130 if (scratch.size() < bpe_scratch_size(n))
131 return err(bpe_error::scratch_too_small);
132
133 // parts[i] = (start of part i, rank of parts i and i+1 merged); the last two are sentinels
134 std::size_t len = n + 1;
135 bpe_part* parts = scratch.data();
136 std::uint32_t min_rank = bpe_no_rank;
137 std::size_t min_at = 0;
138 for (std::size_t i = 0; i + 1 < n; ++i) {
139 const std::uint32_t r = rank(piece.substr(i, 2));
140 if (r < min_rank) {
141 min_rank = r;
142 min_at = i;
143 }
144 parts[i] = {static_cast<std::uint32_t>(i), r};
145 }
146 parts[n - 1] = {static_cast<std::uint32_t>(n - 1), bpe_no_rank};
147 parts[n] = {static_cast<std::uint32_t>(n), bpe_no_rank};
148
149 // The rank of parts i and i+1 merged with i+2: the bytes from part i to part i+3
150 const auto rank_after_merge = [&](std::size_t i) -> std::uint32_t {
151 if (i + 3 < len)
152 return rank(piece.substr(parts[i].start, parts[i + 3].start - parts[i].start));
153 return bpe_no_rank;
154 };
155 while (min_rank != bpe_no_rank) {
156 const std::size_t i = min_at;
157 if (i > 0)
158 parts[i - 1].rank = rank_after_merge(i - 1);
159 parts[i].rank = rank_after_merge(i);
160 for (std::size_t k = i + 1; k + 1 < len; ++k) // remove part i+1
161 parts[k] = parts[k + 1];
162 --len;
163 min_rank = bpe_no_rank;
164 for (std::size_t k = 0; k + 1 < len; ++k)
165 if (parts[k].rank < min_rank) {
166 min_rank = parts[k].rank;
167 min_at = k;
168 }
169 }
170
171 // The tokens are the ranks of the parts. Only a part of one byte can be missing (a merged part is a token by
172 // construction), and then the vocabulary is not byte-level.
173 const std::size_t count = len - 1;
174 for (std::size_t k = 0; k < count; ++k) {
175 out[k] = rank(piece.substr(parts[k].start, parts[k + 1].start - parts[k].start));
176 if (out[k] == bpe_no_rank)
177 return err(bpe_error::unknown_bytes);
178 }
179 return ok(count);
180}

188template <typename Bytes>
189 requires requires(Bytes& b, std::uint32_t id) {

192constexpr result<std::size_t, bpe_error> bpe_decode(std::span<const std::uint32_t> tokens, Bytes&& bytes, std::span<char> out)
193{
194 std::size_t at = 0;
195 for (const std::uint32_t id : tokens) {
196 const std::string_view b = bytes(id);
197 if (b.empty())
198 return err(bpe_error::unknown_bytes);
199 if (out.size() - at < b.size())
200 return err(bpe_error::output_too_small);
201 for (const char c : b)
202 out[at++] = c;
203 }
204 return ok(at);
205}
