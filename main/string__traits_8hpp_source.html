230// Checked only when no standard carrier matched, so naming e.g. std::basic_string<std::byte> (invalid on libc++)
231// never instantiates it.
232template <typename B>
233struct shape_of : shape<B> {};

235template <typename B>
236 requires(shape<B>::kind == carrier::none && derived_from_basic_fixed_string<B>)
237struct shape_of<B>
238{
239 using base = decltype(detail::fixed_string_base(static_cast<const B*>(nullptr)));
240 static constexpr carrier kind = carrier::fixed_string;
241 using element = const typename base::value_type; // read-only API: data() is const
242 static constexpr bool owning = true;
243 static constexpr std::size_t extent = base::size();

246template <typename S>
247using bare_t = std::remove_cv_t<std::remove_reference_t<S>>;

249template <typename Shape>
250concept has_traits = requires { typename Shape::traits; };

252// Every property of S, or valid == false (the defaults keep the boolean traits well-defined for any type).
253template <typename S, typename Shape = shape_of<bare_t<S>>>
254struct classify
255{
256 static constexpr carrier kind = carrier::none;
257 static constexpr bool valid = false;
258 static constexpr bool text = false;
259 static constexpr bool bytes = false;
260 static constexpr bool owning = false;
261 static constexpr bool mutable_units = false;
262};
263
264template <typename S, typename Shape>
265 requires(Shape::kind != carrier::none)
266struct classify<S, Shape>
267{
268 using element = typename Shape::element;
269 using unit = std::remove_const_t<element>;
270
271 static constexpr carrier kind = Shape::kind;
272 static constexpr bool text = is_text_code_unit_v<unit>;
273 static constexpr bool bytes = is_byte_code_unit_v<unit>;
274 static constexpr bool text_only = kind == carrier::pointer || kind == carrier::string
275 || kind == carrier::string_view || kind == carrier::fixed_string;
276
277 static constexpr bool valid = (text || (bytes && !text_only)) && !std::is_volatile_v<element>
278 && !std::is_volatile_v<std::remove_reference_t<S>>;
279
280 static constexpr bool owning = Shape::owning && !std::is_reference_v<S>;
281 static constexpr bool mutable_units =
282 !std::is_const_v<element> && !(Shape::owning && std::is_const_v<std::remove_reference_t<S>>);
283
284 static constexpr null_termination termination =
285 (kind == carrier::string || kind == carrier::fixed_string) ? null_termination::guaranteed
286 : kind == carrier::pointer || (kind == carrier::array && text) ? null_termination::by_contract
287 : null_termination::none;
288
289 static constexpr std::size_t static_length = kind == carrier::pointer ? std::dynamic_extent
290 : kind == carrier::array && text ? Shape::extent - 1
291 : Shape::extent;
292
293 static constexpr length_kind length = kind == carrier::pointer ? length_kind::linear_time
294 : static_length == std::dynamic_extent ? length_kind::constant_time
295 : length_kind::compile_time;
296
297 static constexpr auto traits_tag() noexcept
298 {
299 if constexpr (has_traits<Shape>)
300 return std::type_identity<typename Shape::traits>{};
301 else if constexpr (is_text_code_unit_v<unit>)
302 return std::type_identity<std::char_traits<unit>>{};
303 else
304 return std::type_identity<void>{}; // never name char_traits of a byte unit (deprecated in libc++)
305 }
306};
