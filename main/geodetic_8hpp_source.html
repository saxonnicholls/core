83 [[nodiscard]] constexpr linalg::fixed_vector<T, 3> lla_to_ecef(const linalg::fixed_vector<T, 3>& lla, const ellipsoid<T>& e = wgs84<T>) noexcept(linalg::detail::nothrow_scalar<T>)

97 template <linalg::scalar T>
98 [[nodiscard]] constexpr linalg::fixed_vector<T, 3> ecef_to_lla(const linalg::fixed_vector<T, 3>& ecef, const ellipsoid<T>& e = wgs84<T>) noexcept(linalg::detail::nothrow_scalar<T>)
99 {
100 const T x = ecef[0], y = ecef[1], z = ecef[2];
101 const T e2 = e.e2();
102 const T p = static_cast<T>(math::sqrt(x * x + y * y));
103 if (p == T(0) && z == T(0))
104 return {{T(0), T(0), -e.b()}};
105 T lat = static_cast<T>(math::atan2(z, p * (T(1) - e2)));
106 for (int i = 0; i < 64; ++i) {
107 const T s = static_cast<T>(math::sin(lat));
108 const T n = e.a / static_cast<T>(math::sqrt(T(1) - e2 * s * s));
109 const T next = static_cast<T>(math::atan2(z + e2 * n * s, p));
110 if (next == lat)
111 break;
112 lat = next;
113 }
114 const T s = static_cast<T>(math::sin(lat)), c = static_cast<T>(math::cos(lat));
115 const T h = p * c + z * s - e.a * static_cast<T>(math::sqrt(T(1) - e2 * s * s));
116 const T lon = p == T(0) ? T(0) : static_cast<T>(math::atan2(y, x));
117 return {{lat, lon, h}};
118 }

126 template <linalg::scalar T>
127 [[nodiscard]] constexpr linalg::fixed_matrix<T, 3, 3> enu_rotation(T lat, T lon) noexcept(linalg::detail::nothrow_scalar<T>)
128 {
129 const T sl = static_cast<T>(math::sin(lat)), cl = static_cast<T>(math::cos(lat));
130 const T so = static_cast<T>(math::sin(lon)), co = static_cast<T>(math::cos(lon));
131 return {{-so, co, T(0), -sl * co, -sl * so, cl, cl * co, cl * so, sl}};
