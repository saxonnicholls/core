93template <row X, row W, row B, mutable_row Out>
94 requires std::same_as<row_value_t<X>, row_value_t<W>> && std::same_as<row_value_t<X>, row_value_t<B>>
95 && std::same_as<row_value_t<X>, row_value_t<Out>>
96constexpr norm_stats<row_value_t<X>> layer_norm(const X& x, const W& weight, const B& bias, Out&& out,
97 std::type_identity_t<row_value_t<X>> eps = 1e-5)
98{
99 using T = row_value_t<X>;
100 const auto in = as_row(x);
101 const auto w = as_row(weight);
102 const auto b = as_row(bias);
103 const auto y = as_mutable_row(out);
104 const std::size_t n = in.size();
105 assert(y.size() == n && detail::norm::fits<T>(w.size(), n) && detail::norm::fits<T>(b.size(), n));
106 T mean = T(0);
107 for (const T v : in)
108 mean += v;
109 mean = n == 0 ? T(0) : mean / T(n);
110 T var = T(0);
111 for (const T v : in)
112 var += (v - mean) * (v - mean);
113 var = n == 0 ? T(0) : var / T(n);
114 const T rstd = T(1) / math::sqrt(var + eps);
115 for (std::size_t i = 0; i < n; ++i)
116 y[i] = (in[i] - mean) * rstd * detail::norm::param(w, i, T(1)) + detail::norm::param(b, i, T(0));
117 return {mean, rstd};
118}

131template <row D, row X, row W, mutable_row DX, mutable_row DW, mutable_row DB>
132 requires std::same_as<row_value_t<X>, row_value_t<D>> && std::same_as<row_value_t<X>, row_value_t<W>>
133 && std::same_as<row_value_t<X>, row_value_t<DX>> && std::same_as<row_value_t<X>, row_value_t<DW>>
134 && std::same_as<row_value_t<X>, row_value_t<DB>>
135constexpr void layer_norm_backward(const D& dout, const X& x, const W& weight, norm_stats<row_value_t<X>> stats, DX&& dx,
136 DW&& dweight, DB&& dbias)
137{
138 using T = row_value_t<X>;
139 const auto d = as_row(dout);
140 const auto in = as_row(x);
141 const auto w = as_row(weight);
142 const auto gx = as_mutable_row(dx);
143 const auto gw = as_mutable_row(dweight);
144 const auto gb = as_mutable_row(dbias);
145 const std::size_t n = in.size();
146 assert(d.size() == n && gx.size() == n && detail::norm::fits<T>(w.size(), n) && detail::norm::fits<T>(gw.size(), n)
147 && detail::norm::fits<T>(gb.size(), n));
148 if (n == 0)
149 return;
150 T mean_g = T(0), mean_gn = T(0);
151 for (std::size_t i = 0; i < n; ++i) {
152 const T norm = (in[i] - stats.mean) * stats.rstd;
153 const T g = detail::norm::param(w, i, T(1)) * d[i];
154 mean_g += g;
155 mean_gn += g * norm;
156 }
157 mean_g /= T(n);
158 mean_gn /= T(n);
159 for (std::size_t i = 0; i < n; ++i) {
160 const T norm = (in[i] - stats.mean) * stats.rstd;
161 const T g = detail::norm::param(w, i, T(1)) * d[i];
162 if (!gw.empty())
163 gw[i] += d[i] * norm;
164 if (!gb.empty())
165 gb[i] += d[i];
166 gx[i] = stats.rstd * (g - mean_g - norm * mean_gn);
167 }
168}

179template <row X, row W, mutable_row Out>
180 requires std::same_as<row_value_t<X>, row_value_t<W>> && std::same_as<row_value_t<X>, row_value_t<Out>>
181constexpr row_value_t<X> rms_norm(const X& x, const W& weight, Out&& out,
182 std::type_identity_t<row_value_t<X>> eps = std::numeric_limits<row_value_t<X>>::epsilon())
183{
184 using T = row_value_t<X>;
185 const auto in = as_row(x);
186 const auto w = as_row(weight);
187 const auto y = as_mutable_row(out);
188 const std::size_t n = in.size();
189 assert(y.size() == n && detail::norm::fits<T>(w.size(), n));
190 T ms = T(0);
191 for (const T v : in)
192 ms += v * v;
193 ms = n == 0 ? T(0) : ms / T(n);
194 const T rstd = T(1) / math::sqrt(ms + eps);
195 for (std::size_t i = 0; i < n; ++i)
196 y[i] = in[i] * rstd * detail::norm::param(w, i, T(1));
197 return rstd;
198}

209template <row D, row X, row W, mutable_row DX, mutable_row DW>
210 requires std::same_as<row_value_t<X>, row_value_t<D>> && std::same_as<row_value_t<X>, row_value_t<W>>
211 && std::same_as<row_value_t<X>, row_value_t<DX>> && std::same_as<row_value_t<X>, row_value_t<DW>>
212constexpr void rms_norm_backward(const D& dout, const X& x, const W& weight, std::type_identity_t<row_value_t<X>> rstd, DX&& dx,
213 DW&& dweight)
214{
215 using T = row_value_t<X>;
216 const auto d = as_row(dout);
217 const auto in = as_row(x);
218 const auto w = as_row(weight);
219 const auto gx = as_mutable_row(dx);
220 const auto gw = as_mutable_row(dweight);
221 const std::size_t n = in.size();
222 assert(d.size() == n && gx.size() == n && detail::norm::fits<T>(w.size(), n) && detail::norm::fits<T>(gw.size(), n));
223 if (n == 0)
224 return;
225 T mean_gn = T(0);
226 for (std::size_t i = 0; i < n; ++i)
227 mean_gn += detail::norm::param(w, i, T(1)) * d[i] * in[i] * rstd;
228 mean_gn /= T(n);
229 for (std::size_t i = 0; i < n; ++i) {
230 const T norm = in[i] * rstd;
231 const T g = detail::norm::param(w, i, T(1)) * d[i];
232 if (!gw.empty())
233 gw[i] += d[i] * norm;
234 gx[i] = rstd * (g - norm * mean_gn);
235 }
236}
