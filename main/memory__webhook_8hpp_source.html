39 // The signing scheme of the reference providers.
40 struct memory_webhook {
41 static constexpr std::string_view header = "Memory-Signature";
42 static constexpr std::chrono::seconds tolerance{300};
43
44 static digest_hex<32> tag(const secret<std::string>& key, std::string_view t, std::string_view body)
45 {
46 hmac<sha256> m{key.reveal()};
47 m.update(t);
48 m.update(std::string_view{"."});
49 m.update(body);
50 return m.finish().hex();
51 }

53 static bool parse_int(std::string_view s, std::int64_t& out) noexcept
54 {
55 const auto [end, ec] = std::from_chars(s.data(), s.data() + s.size(), out);
56 return ec == std::errc{} && end == s.data() + s.size();
57 }

59 // A delivery of `e`, signed at `signed_at` and received at `received_at`.
60 static owned_delivery sign(const secret<std::string>& key, const webhook_event& e, utc_s signed_at, utc_s received_at)
61 {
62 owned_delivery d;
63 d.body = format(e);
64 const std::string t = std::to_string(signed_at.unix_seconds());
65 d.headers.push_back({std::string(header), "t=" + t + ",v1=" + std::string(tag(key, t, d.body).view())});
66 d.received_at = received_at;
67 return d;
68 }

70 // Authenticates (constant time) and parses; fails closed.
71 static capability_result<webhook_event> verify(const secret<std::string>& key, const webhook_delivery& w)
72 {
73 const auto sig = find_header(w.headers, header);
74 if (!sig)
75 return fail(capability_errc::signature_invalid, {}, "no signature header");
76 std::string_view t;
77 std::vector<std::string_view> tags;
78 for (std::string_view rest = *sig; !rest.empty();) {
79 const auto comma = rest.find(',');
80 const std::string_view part = rest.substr(0, comma);
81 rest = comma == std::string_view::npos ? std::string_view{} : rest.substr(comma + 1);
82 if (part.starts_with("t="))
83 t = part.substr(2);
84 else if (part.starts_with("v1="))
85 tags.push_back(part.substr(3));
86 }
87 std::int64_t seconds = 0;
88 if (t.empty() || tags.empty() || !parse_int(t, seconds))
89 return fail(capability_errc::signature_invalid, {}, "a malformed signature header");
90 const auto expected = tag(key, t, w.body);
91 bool matched = false;
92 for (const std::string_view candidate : tags)
93 matched = constant_time_equal(expected.view(), candidate) || matched;
94 if (!matched)
95 return fail(capability_errc::signature_invalid, {}, "no signature matches");
96 const auto sent = utc_s::from_unix_seconds(seconds);
97 if (w.received_at - sent > tolerance || sent - w.received_at > tolerance)
98 return fail(capability_errc::stale, {}, "outside the 300-second window");
99 return parse(w.body);
100 }

102 static std::string format(const webhook_event& e)
103 {
104 std::string b;
105 const auto line = [&b](std::string_view k, std::string_view v) {
106 b += k;
107 b += '=';
108 b += v;
109 b += '\n';
110 };
111 line("source", e.source.view());
112 line("provider_account", e.provider_account.view());
113 line("id", e.event_id.view());
114 line("fact_id", e.fact_id.view());
115 line("kind", enum_name(e.kind));
116 line("type", e.type.view());
117 line("payment_id", e.payment_id.view());
118 line("our_reference", e.our_reference.view());
119 line("reference", e.reference.view());
120 line("occurred_at", std::to_string(e.occurred_at.unix_seconds()));
121 line("needs_lookup", e.needs_lookup ? "1" : "0");
122 line("is_test", e.is_test ? "1" : "0");
123 return b;
124 }

126 static capability_result<webhook_event> parse(std::string_view body)
127 {
128 webhook_event e;
129 bool have_id = false, have_kind = false, fits = true;
130 const auto set = [&fits](auto& field, std::string_view v) {
131 auto r = std::remove_reference_t<decltype(field)>::make(v);
132 if (r)
133 field = *r;
134 else
135 fits = false;
136 };
137 for (std::string_view rest = body; !rest.empty();) {
138 const auto nl = rest.find('\n');
139 const std::string_view line = rest.substr(0, nl);
140 rest = nl == std::string_view::npos ? std::string_view{} : rest.substr(nl + 1);
141 const auto eq = line.find('=');
142 if (eq == std::string_view::npos)
143 return fail(capability_errc::invalid_argument, {}, "a webhook line without =");
144 const std::string_view k = line.substr(0, eq), v = line.substr(eq + 1);
145 if (k == "id") {
146 set(e.event_id, v);
147 have_id = !v.empty();
148 } else if (k == "kind") {
149 const auto kind = enum_cast<event_kind>(v);
150 if (!kind)
151 return fail(capability_errc::invalid_argument, {}, "an unknown event kind");
152 e.kind = *kind;
153 have_kind = true;
154 } else if (k == "occurred_at") {
155 std::int64_t s = 0;
156 if (!parse_int(v, s))
157 return fail(capability_errc::invalid_argument, {}, "occurred_at is not unix seconds");
158 e.occurred_at = utc_s::from_unix_seconds(s);
159 } else if (k == "needs_lookup") {
160 e.needs_lookup = v == "1";
161 } else if (k == "is_test") {
162 e.is_test = v == "1";
163 } else if (k == "source") {
164 set(e.source, v);
165 } else if (k == "provider_account") {
166 set(e.provider_account, v);
167 } else if (k == "fact_id") {
168 set(e.fact_id, v);
169 } else if (k == "type") {
170 set(e.type, v);
171 } else if (k == "payment_id") {
172 set(e.payment_id, v);
173 } else if (k == "our_reference") {
174 set(e.our_reference, v);
175 } else if (k == "reference") {
176 set(e.reference, v);
177 } else {
178 return fail(capability_errc::invalid_argument, {}, "an unknown webhook field");
179 }
180 }
181 if (!have_id || !have_kind || !fits)
182 return fail(capability_errc::invalid_argument, {}, "a webhook without id or kind, or a field too long");
183 return ok(e);
184 }
