107 capability_result<send_receipt> send(const message& m)
108 {
109 if (!supports(m.via))
110 return fail(capability_errc::unsupported);
111 if (auto valid = validate(m); !valid)
112 return err(valid.unwrap_err());
113 if (!m.idempotency_key.empty()) {
114 if (const auto it = by_key_.find(m.idempotency_key); it != by_key_.end()) {
115 const sent& earlier = outbox_[it->second];
116 if (earlier.msg.view() != m)
117 return fail(capability_errc::conflict, {}, "idempotency key reused for another message");
118 return ok(earlier.receipt);
119 }
120 }
121 if (injected_) {
122 const capability_error e = *injected_;
123 injected_.reset();
124 return err(e);
125 }
126 const std::string id = std::string(m.via == channel::email ? "memory-email-" : "memory-sms-") +
127 std::to_string(outbox_.size() + 1);
128 const send_receipt receipt{m.via, *provider_id::make(id)};
129 outbox_.push_back({kept_message::of(m), receipt});
130 if (!m.idempotency_key.empty()) {
131#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
132 try {
133 by_key_.emplace(std::string(m.idempotency_key), outbox_.size() - 1);
134 } catch (...) {
135 outbox_.pop_back();
136 throw;
137 }
138#else
139 by_key_.emplace(std::string(m.idempotency_key), outbox_.size() - 1);
140#endif
141 }
142 return ok(receipt);
143 }
