125 [[nodiscard]] constexpr std::optional<asset> make_asset(std::string_view code, unsigned scale) noexcept
126 {
127 if (!is_asset_code(code) || scale > max_scale)
128 return std::nullopt;
129 asset a;
130 for (std::size_t i = 0; i < code.size(); ++i)
131 a.chars[i] = code[i];
132 a.length = static_cast<std::uint8_t>(code.size());
133 a.places = static_cast<std::uint8_t>(scale);
134 return a;
