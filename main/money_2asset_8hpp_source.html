91 [[nodiscard]] constexpr bool is_asset_code(std::string_view code) noexcept
92 {

131 [[nodiscard]] constexpr std::optional<asset> make_asset(std::string_view code, unsigned scale) noexcept
132 {
133 if (!is_asset_code(code) || scale > max_scale)
134 return std::nullopt;
135 asset a;
136 for (std::size_t i = 0; i < code.size(); ++i)
137 a.chars[i] = code[i];
138 a.length = static_cast<std::uint8_t>(code.size());
139 a.places = static_cast<std::uint8_t>(scale);
140 return a;
141 }
