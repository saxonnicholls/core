59 template <typename Key, typename Value, std::size_t N, typename Less = std::less<>>
60 class static_map {
61 public:
62 using value_type = std::pair<Key, Value>;
63 using const_iterator = typename std::array<value_type, N>::const_iterator;
64
68 consteval explicit static_map(std::array<value_type, N> items) : entries_(std::move(items))
69 {
70 std::sort(entries_.begin(), entries_.end(), [](const value_type& a, const value_type& b) { return Less{}(a.first, b.first); });
71 for (std::size_t i = 1; i < N; ++i)
72 if (!Less{}(entries_[i - 1].first, entries_[i].first))
73 throw std::invalid_argument("snicholls::containers::static_map: a key appears twice");
74 }
