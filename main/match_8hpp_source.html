242 template <class X, class... Cs>
243 constexpr decltype(auto) match_one(X& x, const Cs&... cs)
244 {
245 using info = match_result<X, Cs...>;
246 static_assert(info::exhaustive,
247 "snicholls::fn::match: not exhaustive: a value or an alternative has no case without a guard");
248 if constexpr (info::exhaustive) {
249 using C = meta::at_t<meta::type_list<Cs...>, info::index>;
250 using R = std::remove_cvref_t<decltype(std::declval<const C&>()(x))>;
251 return detail::dispatch<R>(x, cs...);
252 }

255 // What is matched: a variant's alternative, a result's side, or the value.
256 struct match_fn {
257 template <class V, class... C>
258 requires specialisation_of<V, std::variant>
259 constexpr decltype(auto) operator()(V&& v, C&&... c) const
260 {
261 return [&](const auto&... cs) -> decltype(auto) {
262 // The first alternative's result type, which every other alternative's converts to (as documented)
263 using R = decltype(detail::match_one(std::get<0>(v), cs...));
264 return std::visit([&](auto& alt) -> R { return detail::match_one(alt, cs...); }, v);
265 }(detail::to_case(std::forward<C>(c))...);
266 }
267
268 template <class V, class... C>
269 requires specialisation_of<V, snicholls::result>
270 constexpr decltype(auto) operator()(V&& r, C&&... c) const
271 {
272 return [&](const auto&... cs) -> decltype(auto) {
273 using T = std::remove_reference_t<decltype(r.unwrap_unchecked())>;
274 using E = std::remove_reference_t<decltype(r.unwrap_err_unchecked())>;
275 using R = std::common_type_t<decltype(detail::match_one(std::declval<ok_side<T>&>(), cs...)),
276 decltype(detail::match_one(std::declval<err_side<E>&>(), cs...))>;
277 if (r.is_ok()) {
278 ok_side<T> side{r.unwrap_unchecked()};
279 return static_cast<R>(detail::match_one(side, cs...));
280 }
281 err_side<E> side{r.unwrap_err_unchecked()};
282 return static_cast<R>(detail::match_one(side, cs...));
283 }(detail::to_case(std::forward<C>(c))...);
284 }
