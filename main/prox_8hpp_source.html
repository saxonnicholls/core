87 // W0(e^a): the w > 0 with w + log w = a, by Halley's iteration on f(w) = w + log w - a
88 // (f' = 1 + 1/w, f'' = -1/w^2).
89 template <typename T>
90 constexpr T w0_exp(T a)
91 {
92 if (a != a)
93 return a;
94 if (a == std::numeric_limits<T>::infinity())
95 return a;
96 if (a < T(1)) {
97 const T e = math::exp(a);
98 if (e < std::numeric_limits<T>::epsilon())
99 return e * (T(1) - e); // W(x) = x - x^2 + ...
100 }
101 T w = a >= T(1) ? a - math::log(a) : math::exp(a) / (T(1) + math::exp(a));
102 for (int i = 0; i < 50; ++i) {
103 const T f = w + math::log(w) - a, d1 = T(1) + T(1) / w, d2 = -T(1) / (w * w);
104 const T step = T(2) * f * d1 / (T(2) * d1 * d1 - f * d2);
105 T next = w - step;
106 if (!(next > T(0)))
107 next = w / T(2);
108 if (math::abs(next - w) <= T(4) * std::numeric_limits<T>::epsilon() * next)
109 return next;
110 w = next;
111 }
112 return w;
113 }

115 template <typename T>
116 constexpr T huber(T x, T m) { return math::abs(x) <= m ? x * x : T(2) * m * math::abs(x) - m * m; }

118 template <typename T>
119 constexpr T huber_prox(T v, T lambda, T m)
120 {
121 if (math::abs(v) <= m * (T(1) + T(2) * lambda))
122 return v / (T(1) + T(2) * lambda);
123 return v > T(0) ? v - T(2) * lambda * m : v + T(2) * lambda * m;
124 }

146 {
147 const T n = detail::cones::norm(std::span<const T>(v));
148 const T k = n > lambda ? T(1) - lambda / n : T(0);

160 template <linalg::scalar T>
161 constexpr result<void, error> prox_group_l2(std::span<T> v, std::span<const std::size_t> sizes, std::type_identity_t<T> lambda) noexcept(detail::prox::nt<T>)
162 {
163 std::size_t total = 0;
164 for (const std::size_t s : sizes)
165 total += s;
166 if (total != v.size())
167 return err(error::dimension_mismatch);
168 std::size_t offset = 0;
169 for (const std::size_t s : sizes) {
170 prox_l2(v.subspan(offset, s), lambda);
171 offset += s;
172 }
173 return ok();
174 }

183 template <linalg::scalar T>
184 constexpr result<void, error> prox_linf(std::span<T> v, std::type_identity_t<T> lambda, std::span<T> workspace) noexcept(detail::prox::nt<T>)
185 {
186 if (workspace.size() < v.size())
187 return err(error::workspace_too_small);
188 T s(0);
189 for (const T& x : v)
190 s += math::abs(x);
191 if (s <= lambda) {
192 project_zero(v);
193 return ok();
194 }
195 const T tau = detail::cones::condat_tau(v.size(), [&](std::size_t i) { return math::abs(v[i]); }, lambda, workspace);
196 project_box(v, -tau, tau);
197 return ok();
198 }

302 constexpr void prox_log_barrier(std::span<T> v, std::type_identity_t<T> lambda) noexcept(detail::prox::nt<T>)
303 {
304 for (T& x : v) {
305 const T root = math::sqrt(x * x + T(4) * lambda);
306 x = x >= T(0) ? (x + root) / T(2) : T(2) * lambda / (root - x);

436 template <linalg::scalar T, typename Prox>
437 constexpr result<void, error> prox_conjugate(Prox&& prox_f, std::span<T> v, std::type_identity_t<T> lambda, std::span<T> workspace) noexcept(
438 noexcept(prox_f(std::declval<std::span<T>>(), std::declval<T>())) && detail::prox::nt<T>)
439 {
440 if (workspace.size() < v.size())
441 return err(error::workspace_too_small);
442 const std::span<T> y = workspace.first(v.size());
443 for (std::size_t i = 0; i < v.size(); ++i)
444 y[i] = v[i] / lambda;
445 prox_f(y, T(1) / lambda);
446 for (std::size_t i = 0; i < v.size(); ++i)
447 v[i] -= lambda * y[i];
448 return ok();
449 }
