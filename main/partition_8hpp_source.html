155 consteval std::uint64_t bell(std::size_t n) noexcept
156 {
157 // The Bell triangle: each row starts with the previous row's last entry.
158 std::array<std::uint64_t, 27> row{};
159 std::array<std::uint64_t, 27> next{};
160 row[0] = 1;
161 for (std::size_t r = 1; r <= n; ++r) {
162 next[0] = row[r - 1];
163 for (std::size_t c = 1; c <= r; ++c)
164 next[c] = next[c - 1] + row[c - 1];
165 row = next;
166 }
167 return row[0];
168 }

190 template <std::size_t N, typename F>
191 constexpr void for_each_set_partition(F&& f) noexcept(noexcept(f(std::declval<const std::array<std::size_t, N>&>(), std::size_t{})))
192 {
193 std::array<std::size_t, N> block{};
194 if constexpr (N == 0) {
195 f(std::as_const(block), std::size_t{0});
196 } else {
197 std::array<std::size_t, N> largest{}; // largest[i]: the largest block among 0 .. i
198 while (true) {
199 f(std::as_const(block), largest[N - 1] + 1);
200 std::size_t i = N - 1;
201 while (i > 0 && block[i] > largest[i - 1])
202 --i;
203 if (i == 0)
204 return;
205 ++block[i];
206 largest[i] = block[i] > largest[i - 1] ? block[i] : largest[i - 1];
207 for (std::size_t j = i + 1; j < N; ++j) {
208 block[j] = 0;
209 largest[j] = largest[i];
210 }
211 }
212 }

222 [[nodiscard]] constexpr std::array<std::uint64_t, N> blocks_as_masks(const std::array<std::size_t, N>& block_of, std::size_t blocks) noexcept
223 {
224 static_assert(N <= 64, "snicholls::blocks_as_masks: at most 64 elements");
225 std::array<std::uint64_t, N> masks{};
226 for (std::size_t i = 0; i < N; ++i)

228 masks[block_of[i]] |= std::uint64_t{1} << i;
