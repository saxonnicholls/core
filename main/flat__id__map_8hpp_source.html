148 bool insert(key_type key, Value value)
149 {
150 if (key == empty_key) [[unlikely]]
151 return false; // it marks an empty slot: storing it would count an entry that no lookup can find
152 if ((size_ + 1) * 2 > capacity()) grow();

154 slot& s = slots_[i];
155 if (s.key == empty_key) {

168 bool erase(key_type key) noexcept
169 {
170 if (key == empty_key) [[unlikely]]
171 return false;
172 std::size_t i = home(key);
173 for (;; i = (i + 1) & mask_) {
174 if (slots_[i].key == key) break;
175 if (slots_[i].key == empty_key) return false;
176 }
177 // Backward-shift deletion: pull later members of the probe run into the hole.
178 for (std::size_t j = (i + 1) & mask_;; j = (j + 1) & mask_) {
179 if (slots_[j].key == empty_key) break;
180 const std::size_t k = home(slots_[j].key);
181 // Move j into i unless j's home lies cyclically in (i, j].
182 const bool in_range = (i <= j) ? (i < k && k <= j) : (i < k || k <= j);
183 if (!in_range) {
184 slots_[i] = slots_[j];
185 i = j;
186 }
187 }
188 slots_[i].key = empty_key;
189 --size_;
190 return true;
191 }

225 static std::size_t capacity_for(std::size_t n)
226 {
227 // Twice n, rounded up to a power of two: beyond max/4 that overflows (and the loop never ended).
228 if (n > (std::numeric_limits<std::size_t>::max() >> 2)) [[unlikely]]
229 throw std::length_error("snicholls::containers::flat_id_map: too many entries");
230 std::size_t c = 16;
231 while (c < n * 2) c *= 2;
232 return c;
233 }

240 void allocate(std::size_t cap)
241 {
242 slots_ = std::make_unique_for_overwrite<slot[]>(cap);
243 for (std::size_t i = 0; i < cap; ++i) slots_[i].key = empty_key;
244 mask_ = cap - 1;
245 shift_ = 64 - static_cast<unsigned>(std::countr_zero(cap));
246 }

248 void grow() { rehash(capacity() * 2); }

250 void rehash(std::size_t cap)
251 {
252 auto old = std::move(slots_);
253 const std::size_t old_cap = mask_ + 1;
254 const auto old_mask = mask_;
255 const auto old_shift = shift_;
256 try {
257 allocate(cap);
258 } catch (...) {
259 slots_ = std::move(old);
260 mask_ = old_mask;
261 shift_ = old_shift;
262 throw;
263 }
264 for (std::size_t i = 0; i < old_cap; ++i) {
265 if (old[i].key == empty_key) continue;
266 for (std::size_t j = home(old[i].key);; j = (j + 1) & mask_) {
267 if (slots_[j].key == empty_key) {
268 slots_[j] = old[i];
269 break;
270 }
271 }
272 }
273 }
