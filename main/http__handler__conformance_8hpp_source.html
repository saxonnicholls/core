61 concept http_handler_probe = std::is_default_constructible_v<Probe> &&
62 http_handler<std::remove_reference_t<decltype(std::declval<Probe&>().handler())>> && requires(Probe& p) {

73 template <typename Probe>
74 requires detail::http_handler_probe<Probe>
75 [[nodiscard]] capability_conformance_report run_http_handler_conformance()
76 {
77 capability_conformance_report report;
78
79 detail::capability_check<Probe>(report, "a known route answers 2xx", [](Probe& p, const auto& expect) {
80 const http_response r = p.handler().handle(p.known());
81 expect(r.status >= 200 && r.status < 300, "the status is 2xx");
82 expect(!r.body.empty(), "there is a body");
83 });
84
85 detail::capability_check<Probe>(report, "an unknown path is 404", [](Probe& p, const auto& expect) {
86 expect(p.handler().handle(p.unknown()).status == 404, "the status is 404");
87 });
88
89 detail::capability_check<Probe>(report, "HEAD is GET without a body", [](Probe& p, const auto& expect) {
90 http_request head = p.known();
91 head.method = http_method::head;
92 const http_response got = p.handler().handle(p.known());
93 const http_response r = p.handler().handle(head);
94 expect(r.status == got.status, "HEAD answers GET's status");
95 expect(r.body.empty(), "HEAD has no body");
96 });
97
98 detail::capability_check<Probe>(report, "any request gets a status in range", [](Probe& p, const auto& expect) {
99 http_request odd;
100 odd.method = http_method::patch;
101 odd.path = "";
102 odd.body = std::string("\0\xff", 2);
103 const int status = p.handler().handle(odd).status;
104 expect(status >= 100 && status <= 599, "the status is from 100 to 599");
105 });
106
107 detail::capability_check<Probe>(report, "a throwing handler answers 500 without detail", [](Probe& p, const auto& expect) {
108 const http_response r = p.handler().handle(p.throwing());
109 expect(r.status == 500, "the status is 500");
110 expect(r.body.find(capability_redaction_marker) == std::string::npos, "the body carries nothing of the exception");
111 for (const http_header& h : r.headers)
112 expect(h.value.find(capability_redaction_marker) == std::string::npos, "no header carries the exception");
113 });
114
115 return report;
116 }
