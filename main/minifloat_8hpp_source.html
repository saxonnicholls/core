103template <int E, int M, minifloat_kind Kind = minifloat_kind::ieee>
104struct minifloat
105{
106 static_assert(E >= 1 && E <= 8 && M >= 0 && E + M <= 15, "snicholls::ml::minifloat: 1 to 8 exponent bits, at most 15 bits besides the sign");
107 static_assert(Kind != minifloat_kind::ieee || M >= 1, "snicholls::ml::minifloat: an IEEE format needs a mantissa bit to tell NaN from infinity");
108
110 using storage_type = std::conditional_t<(E + M + 1 <= 8), std::uint8_t, std::uint16_t>;
111
112 static constexpr int exponent_bits = E;
113 static constexpr int mantissa_bits = M;
114 static constexpr int total_bits = E + M + 1;
115 static constexpr int bias = (1 << (E - 1)) - 1;
116 static constexpr minifloat_kind kind = Kind;
118 static constexpr overflow_policy default_overflow = Kind == minifloat_kind::ieee ? overflow_policy::infinity : overflow_policy::saturate;
119
120 storage_type bits = 0;
121
125 [[nodiscard]] static constexpr minifloat from_bits(std::uint32_t b) noexcept
126 {
127 return minifloat{static_cast<storage_type>(b & ((1u << total_bits) - 1))};
128 }
129
131 static constexpr bool has_infinity = Kind == minifloat_kind::ieee;
133 static constexpr bool has_nan = Kind != minifloat_kind::finite;
134};

167 template <typename F>
168 constexpr std::uint32_t max_code() noexcept
169 {
170 constexpr std::uint32_t all = (1u << (F::exponent_bits + F::mantissa_bits)) - 1;
171 if constexpr (F::kind == minifloat_kind::ieee)
172 return (((1u << F::exponent_bits) - 1) << F::mantissa_bits) - 1;
173 else if constexpr (F::kind == minifloat_kind::finite_nan)
174 return all - 1;
175 else
176 return all;
177 }
178
179 template <typename F>
180 constexpr std::uint32_t nan_code() noexcept
181 {
182 if constexpr (F::kind == minifloat_kind::ieee)
183 return (((1u << F::exponent_bits) - 1) << F::mantissa_bits) | (1u << (F::mantissa_bits - 1));
184 else
185 return (1u << (F::exponent_bits + F::mantissa_bits)) - 1;
186 }
187
188 template <typename F>
189 constexpr std::uint32_t overflowed(overflow_policy p) noexcept
190 {
191 if (p == overflow_policy::saturate || F::kind == minifloat_kind::finite)
192 return max_code<F>();
193 if constexpr (F::kind == minifloat_kind::ieee)
194 return ((1u << F::exponent_bits) - 1) << F::mantissa_bits; // infinity
195 else
196 return nan_code<F>();
197 }

199 // The code of |x| (finite, at least 0) rounded to F: to nearest even, or up with probability = the fraction
200 // dropped when `stochastic`, against `random` / 2^32.
201 template <typename F>
202 constexpr std::uint32_t encode(double x, bool stochastic, std::uint32_t random, overflow_policy p) noexcept
203 {
204 const std::uint64_t u = std::bit_cast<std::uint64_t>(x) & ~(std::uint64_t{1} << 63);
205 const std::uint32_t sign = static_cast<std::uint32_t>(std::bit_cast<std::uint64_t>(x) >> 63) << (F::exponent_bits + F::mantissa_bits);
206 const int field = static_cast<int>(u >> 52);
207 if (field == 0x7ff) {
208 if ((u & ((std::uint64_t{1} << 52) - 1)) != 0) // NaN
209 return F::has_nan ? (sign | nan_code<F>()) : (sign | max_code<F>());
210 return sign | overflowed<F>(p); // infinity
211 }
212 if (field == 0) // zero, or a double subnormal: far below every format's smallest value
213 return sign;
214 constexpr int e_min = 1 - F::bias;
215 const int e = field - 1023;
216 const int q_exp = (e < e_min ? e_min : e) - F::mantissa_bits; // the exponent of the format's last place here
217 if (q_exp > 1000)
218 return sign | overflowed<F>(p);
219 const double q = std::bit_cast<double>(u) * pow2(-q_exp); // exact: a power-of-two scaling, below 2^(M+1)
220 const auto whole = static_cast<std::uint64_t>(q);
221 const double frac = q - static_cast<double>(whole); // exact
222 std::uint64_t n = whole;
223 if (stochastic) {
224 if (frac * 4294967296.0 > static_cast<double>(random))
225 ++n;
226 } else if (frac > 0.5 || (frac == 0.5 && (whole & 1) != 0)) {
227 ++n;
228 }
229 // Continuous codes: a mantissa that rounds up to 2^M carries into the exponent, a subnormal into the first normal
230 const std::uint64_t code = e < e_min ? n : (static_cast<std::uint64_t>(e + F::bias - 1) << F::mantissa_bits) + n;
231 if (code > max_code<F>())
232 return sign | overflowed<F>(p);
233 return sign | static_cast<std::uint32_t>(code);
234 }
235} // namespace detail::minifloat_

268[[nodiscard]] constexpr bfloat16 to_bfloat16(float x) noexcept
269{
270 const std::uint32_t u = std::bit_cast<std::uint32_t>(x);
271 if ((u & 0x7fffffffu) > 0x7f800000u) // NaN: keep the sign, quiet
272 return bfloat16::from_bits((u >> 16) | 0x0040u);
273 return bfloat16::from_bits((u + 0x7fffu + ((u >> 16) & 1u)) >> 16);
274}

280template <typename F>
281 requires is_minifloat_v<F>
282[[nodiscard]] constexpr double to_double(F f) noexcept
283{
284 constexpr int E = F::exponent_bits, M = F::mantissa_bits;
285 const std::uint32_t b = f.bits;
286 const bool negative = ((b >> (E + M)) & 1u) != 0;
287 const std::uint32_t ef = (b >> M) & ((1u << E) - 1);
288 const std::uint32_t mf = b & ((1u << M) - 1);
289 double v = 0.0;
290 if (F::kind == minifloat_kind::ieee && ef == (1u << E) - 1)
291 v = mf == 0 ? std::numeric_limits<double>::infinity() : std::numeric_limits<double>::quiet_NaN();
292 else if (F::kind == minifloat_kind::finite_nan && ef == (1u << E) - 1 && mf == (1u << M) - 1)
293 v = std::numeric_limits<double>::quiet_NaN();
294 else if (ef == 0)
295 v = static_cast<double>(mf) * detail::minifloat_::pow2(1 - F::bias - M);
296 else
297 v = static_cast<double>((1u << M) + mf) * detail::minifloat_::pow2(static_cast<int>(ef) - F::bias - M);
298 return negative ? -v : v;
299}
