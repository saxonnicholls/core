98template <int E, int M, minifloat_kind Kind = minifloat_kind::ieee>
99struct minifloat
100{
101 static_assert(E >= 1 && E <= 8 && M >= 0 && E + M <= 15, "snicholls::ml::minifloat: 1 to 8 exponent bits, at most 15 bits besides the sign");
102 static_assert(Kind != minifloat_kind::ieee || M >= 1, "snicholls::ml::minifloat: an IEEE format needs a mantissa bit to tell NaN from infinity");
103
105 using storage_type = std::conditional_t<(E + M + 1 <= 8), std::uint8_t, std::uint16_t>;
106
107 static constexpr int exponent_bits = E;
108 static constexpr int mantissa_bits = M;
109 static constexpr int total_bits = E + M + 1;
110 static constexpr int bias = (1 << (E - 1)) - 1;
111 static constexpr minifloat_kind kind = Kind;
113 static constexpr overflow_policy default_overflow = Kind == minifloat_kind::ieee ? overflow_policy::infinity : overflow_policy::saturate;
114
115 storage_type bits = 0;
116
120 [[nodiscard]] static constexpr minifloat from_bits(std::uint32_t b) noexcept
121 {
122 return minifloat{static_cast<storage_type>(b & ((1u << total_bits) - 1))};
123 }
124
126 static constexpr bool has_infinity = Kind == minifloat_kind::ieee;
128 static constexpr bool has_nan = Kind != minifloat_kind::finite;
129};

162 template <typename F>
163 constexpr std::uint32_t max_code() noexcept
164 {
165 constexpr std::uint32_t all = (1u << (F::exponent_bits + F::mantissa_bits)) - 1;
166 if constexpr (F::kind == minifloat_kind::ieee)
167 return (((1u << F::exponent_bits) - 1) << F::mantissa_bits) - 1;
168 else if constexpr (F::kind == minifloat_kind::finite_nan)
169 return all - 1;
170 else
171 return all;
172 }
173
174 template <typename F>
175 constexpr std::uint32_t nan_code() noexcept
176 {
177 if constexpr (F::kind == minifloat_kind::ieee)
178 return (((1u << F::exponent_bits) - 1) << F::mantissa_bits) | (1u << (F::mantissa_bits - 1));
179 else
180 return (1u << (F::exponent_bits + F::mantissa_bits)) - 1;
181 }
182
183 template <typename F>
184 constexpr std::uint32_t overflowed(overflow_policy p) noexcept
185 {
186 if (p == overflow_policy::saturate || F::kind == minifloat_kind::finite)
187 return max_code<F>();
188 if constexpr (F::kind == minifloat_kind::ieee)
189 return ((1u << F::exponent_bits) - 1) << F::mantissa_bits; // infinity
190 else
191 return nan_code<F>();
192 }

194 // The code of |x| (finite, at least 0) rounded to F: to nearest even, or up with probability = the fraction
195 // dropped when `stochastic`, against `random` / 2^32.
196 template <typename F>
197 constexpr std::uint32_t encode(double x, bool stochastic, std::uint32_t random, overflow_policy p) noexcept
198 {
199 const std::uint64_t u = std::bit_cast<std::uint64_t>(x) & ~(std::uint64_t{1} << 63);
200 const std::uint32_t sign = static_cast<std::uint32_t>(std::bit_cast<std::uint64_t>(x) >> 63) << (F::exponent_bits + F::mantissa_bits);
201 const int field = static_cast<int>(u >> 52);
202 if (field == 0x7ff) {
203 if ((u & ((std::uint64_t{1} << 52) - 1)) != 0) // NaN
204 return F::has_nan ? (sign | nan_code<F>()) : (sign | max_code<F>());
205 return sign | overflowed<F>(p); // infinity
206 }
207 if (field == 0) // zero, or a double subnormal: far below every format's smallest value
208 return sign;
209 constexpr int e_min = 1 - F::bias;
210 const int e = field - 1023;
211 const int q_exp = (e < e_min ? e_min : e) - F::mantissa_bits; // the exponent of the format's last place here
212 if (q_exp > 1000)
213 return sign | overflowed<F>(p);
214 const double q = std::bit_cast<double>(u) * pow2(-q_exp); // exact: a power-of-two scaling, below 2^(M+1)
215 const auto whole = static_cast<std::uint64_t>(q);
216 const double frac = q - static_cast<double>(whole); // exact
217 std::uint64_t n = whole;
218 if (stochastic) {
219 if (frac * 4294967296.0 > static_cast<double>(random))
220 ++n;
221 } else if (frac > 0.5 || (frac == 0.5 && (whole & 1) != 0)) {
222 ++n;
223 }
224 // Continuous codes: a mantissa that rounds up to 2^M carries into the exponent, a subnormal into the first normal
225 const std::uint64_t code = e < e_min ? n : (static_cast<std::uint64_t>(e + F::bias - 1) << F::mantissa_bits) + n;
226 if (code > max_code<F>())
227 return sign | overflowed<F>(p);
228 return sign | static_cast<std::uint32_t>(code);
229 }
230} // namespace detail::minifloat_

263[[nodiscard]] constexpr bfloat16 to_bfloat16(float x) noexcept
264{
265 const std::uint32_t u = std::bit_cast<std::uint32_t>(x);
266 if ((u & 0x7fffffffu) > 0x7f800000u) // NaN: keep the sign, quiet
267 return bfloat16::from_bits((u >> 16) | 0x0040u);
268 return bfloat16::from_bits((u + 0x7fffu + ((u >> 16) & 1u)) >> 16);
269}

275template <typename F>
276 requires is_minifloat_v<F>
277[[nodiscard]] constexpr double to_double(F f) noexcept
278{
279 constexpr int E = F::exponent_bits, M = F::mantissa_bits;
280 const std::uint32_t b = f.bits;
281 const bool negative = ((b >> (E + M)) & 1u) != 0;
282 const std::uint32_t ef = (b >> M) & ((1u << E) - 1);
283 const std::uint32_t mf = b & ((1u << M) - 1);
284 double v = 0.0;
285 if (F::kind == minifloat_kind::ieee && ef == (1u << E) - 1)
286 v = mf == 0 ? std::numeric_limits<double>::infinity() : std::numeric_limits<double>::quiet_NaN();
287 else if (F::kind == minifloat_kind::finite_nan && ef == (1u << E) - 1 && mf == (1u << M) - 1)
288 v = std::numeric_limits<double>::quiet_NaN();
289 else if (ef == 0)
290 v = static_cast<double>(mf) * detail::minifloat_::pow2(1 - F::bias - M);
291 else
292 v = static_cast<double>((1u << M) + mf) * detail::minifloat_::pow2(static_cast<int>(ef) - F::bias - M);
293 return negative ? -v : v;
294}
