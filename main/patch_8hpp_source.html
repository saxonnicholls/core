101struct patch_grid
102{
103 std::size_t frames = 1;
104 std::size_t height = 0;
105 std::size_t width = 0;
106 std::size_t channels = 1;
107 std::size_t tubelet = 1;
108 std::size_t patch_height = 0;
109 std::size_t patch_width = 0;
110
112 [[nodiscard]] constexpr std::size_t grid_frames() const noexcept { return frames / tubelet; }
114 [[nodiscard]] constexpr std::size_t grid_height() const noexcept { return height / patch_height; }
116 [[nodiscard]] constexpr std::size_t grid_width() const noexcept { return width / patch_width; }
118 [[nodiscard]] constexpr std::size_t tokens() const noexcept { return grid_frames() * grid_height() * grid_width(); }
120 [[nodiscard]] constexpr std::size_t token_size() const noexcept { return tubelet * patch_height * patch_width * channels; }
122 [[nodiscard]] constexpr std::size_t values() const noexcept { return frames * height * width * channels; }
123};
124
128[[nodiscard]] constexpr bool valid(const patch_grid& g) noexcept
129{
130 return g.frames > 0 && g.height > 0 && g.width > 0 && g.channels > 0 && g.tubelet > 0 && g.patch_height > 0
131 && g.patch_width > 0 && g.frames % g.tubelet == 0 && g.height % g.patch_height == 0 && g.width % g.patch_width == 0;
132}
133
139[[nodiscard]] constexpr std::size_t patch_source(const patch_grid& g, std::size_t token, std::size_t element) noexcept
140{
141 const std::size_t gw = g.grid_width(), gh = g.grid_height();
142 const std::size_t gx = token % gw, gy = token / gw % gh, gt = token / gw / gh;
143 const std::size_t c = element % g.channels, dx = element / g.channels % g.patch_width,
144 dy = element / g.channels / g.patch_width % g.patch_height,
145 dt = element / g.channels / g.patch_width / g.patch_height;
146 const std::size_t f = gt * g.tubelet + dt, y = gy * g.patch_height + dy, x = gx * g.patch_width + dx;
147 return ((f * g.height + y) * g.width + x) * g.channels + c;
148}

152 // Visits every run of a patch row (patch_width · channels contiguous values in the volume) in token order, with
153 // its offset in the volume and in the tokens: no index arithmetic beyond additions per run
154 template <typename F>
155 constexpr void runs(const patch_grid& g, std::size_t samples, F&& f)
156 {
157 const std::size_t run = g.patch_width * g.channels, row = g.width * g.channels, frame = g.height * row;
158 std::size_t at = 0;
159 for (std::size_t s = 0; s < samples; ++s)
160 for (std::size_t gt = 0; gt < g.grid_frames(); ++gt)
161 for (std::size_t gy = 0; gy < g.grid_height(); ++gy)
162 for (std::size_t gx = 0; gx < g.grid_width(); ++gx)
163 for (std::size_t dt = 0; dt < g.tubelet; ++dt) {
164 std::size_t from = s * g.values() + (gt * g.tubelet + dt) * frame + gy * g.patch_height * row + gx * run;
165 for (std::size_t dy = 0; dy < g.patch_height; ++dy, from += row, at += run)
166 f(from, at, run);
167 }
168 }
169} // namespace detail::patch

176template <typename B, math::math_number T>
177constexpr void patchify_rows(const B&, const patch_grid& g, std::type_identity_t<std::span<const T>> volumes, std::span<T> tokens,
178 std::size_t samples = 1) noexcept(std::is_floating_point_v<T>)
179{
180 assert(valid(g) && volumes.size() == samples * g.values() && tokens.size() == volumes.size());
181 const T* v = volumes.data();
182 T* t = tokens.data();
183 detail::patch::runs(g, samples, [&](std::size_t from, std::size_t at, std::size_t run) { std::copy_n(v + from, run, t + at); });
184}

210 // One axis of a sine-cosine embedding: width channels (even) of position p, [sin(p w_i) ..., cos(p w_i) ...] with
211 // w_i = 1 / 10000^(i / (width/2)), in double, rounded once to T; the first out.size() of them written
212 template <typename T>
213 constexpr void sincos_axis(std::span<T> out, std::size_t width, std::size_t p)
214 {
215 const std::size_t h = width / 2;
216 for (std::size_t i = 0; i < h; ++i) {
217 const double omega = 1.0 / math::pow(10000.0, double(i) / double(h));
218 const double a = double(p) * omega;
219 if (i < out.size())
220 out[i] = static_cast<T>(math::sin(a));
221 if (h + i < out.size())
222 out[h + i] = static_cast<T>(math::cos(a));
223 }
224 }
225
226 constexpr std::size_t uniform_axis(std::size_t dim) noexcept { return (dim + 5) / 6 * 2; }
227} // namespace detail::patch

235template <math::math_number T>
236constexpr void sincos_position_2d(std::size_t dim, std::size_t gh, std::size_t gw, std::span<T> out,
237 sincos_layout layout = sincos_layout::row_first)
238{
239 assert(dim % 4 == 0 && out.size() == gh * gw * dim);
240 const std::size_t a = dim / 2;
241 for (std::size_t y = 0; y < gh; ++y)
242 for (std::size_t x = 0; x < gw; ++x) {
243 const auto row = out.subspan((y * gw + x) * dim, dim);
244 const bool rows = layout == sincos_layout::row_first;
245 detail::patch::sincos_axis(row.first(a), a, rows ? y : x);
246 detail::patch::sincos_axis(row.last(a), a, rows ? x : y);
247 }
248}

258template <math::math_number T>
259constexpr void sincos_position_3d(std::size_t dim, std::size_t gt, std::size_t gh, std::size_t gw, std::span<T> out,
260 bool uniform_power = false)
261{
262 assert(out.size() == gt * gh * gw * dim && (uniform_power ? dim % 2 == 0 && dim >= 6 : dim % 8 == 0));
263 const std::size_t u = detail::patch::uniform_axis(dim);
264 const std::size_t dd = uniform_power ? u : dim / 2, dh = uniform_power ? u : dim / 4, dw = dh;
265 for (std::size_t t = 0; t < gt; ++t)
266 for (std::size_t y = 0; y < gh; ++y)
267 for (std::size_t x = 0; x < gw; ++x) {
268 const auto row = out.subspan(((t * gh + y) * gw + x) * dim, dim);
269 const std::size_t widths[3] = {dd, dh, dw}, coords[3] = {t, y, x};
270 std::size_t at = 0; // the three axes side by side, the last cut at dim
271 for (std::size_t k = 0; k < 3 && at < dim; at += widths[k], ++k)
272 detail::patch::sincos_axis(row.subspan(at, std::min(widths[k], dim - at)), widths[k], coords[k]);
273 }
274}

315template <math::math_number T, std::size_t HeadDim, std::size_t Gt, std::size_t Gh, std::size_t Gw>
316 requires(HeadDim % 2 == 0 && Gt > 0 && Gh > 0 && Gw > 0 && HeadDim / 2 >= (Gt > 1 ? 3 : 2))
317struct axial_rope_table
318{
319 static constexpr std::size_t axes = Gt > 1 ? 3 : 2;
320 static constexpr std::size_t share = HeadDim / 2 / axes;
321 std::array<std::array<T, HeadDim / 2>, Gt * Gh * Gw> cos{};
322 std::array<std::array<T, HeadDim / 2>, Gt * Gh * Gw> sin{};
323
325 constexpr explicit axial_rope_table(T base = T(10000))
326 {
327 for (std::size_t token = 0; token < Gt * Gh * Gw; ++token) {
328 const std::size_t x = token % Gw, y = token / Gw % Gh, t = token / Gw / Gh;
329 const std::size_t coords[3] = {axes == 3 ? t : y, axes == 3 ? y : x, x};
330 for (std::size_t p = 0; p < HeadDim / 2; ++p) {
331 const std::size_t a = p / share;
332 if (a >= axes) {
333 cos[token][p] = T(1);
334 sin[token][p] = T(0);
335 continue;
336 }
337 const T angle = T(coords[a]) * rope_frequency<T>(p % share, 2 * share, base);
338 cos[token][p] = math::cos(angle);
339 sin[token][p] = math::sin(angle);
340 }
341 }
342 }
343
344 [[nodiscard]] constexpr std::span<const T, HeadDim / 2> cos_at(std::size_t token) const { return cos[token]; }
346 [[nodiscard]] constexpr std::span<const T, HeadDim / 2> sin_at(std::size_t token) const { return sin[token]; }
347};
