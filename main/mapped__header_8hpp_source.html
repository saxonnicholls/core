148 template <class U>
149 constexpr void put_le(std::span<std::byte> out, std::size_t at, U v) noexcept
150 {
151 for (std::size_t i = 0; i < sizeof(U); ++i) out[at + i] = static_cast<std::byte>((v >> (8 * i)) & 0xff);
152 }
153 template <class U>
154 constexpr U get_le(std::span<const std::byte> in, std::size_t at) noexcept
155 {
156 U v = 0;
157 for (std::size_t i = 0; i < sizeof(U); ++i) v |= static_cast<U>(static_cast<U>(in[at + i]) << (8 * i));
158 return v;
159 }

171[[nodiscard]] constexpr result<void, header_error> encode_header(const mapped_header& h, std::span<std::byte> out) noexcept
172{
173 if (out.size() < mapped_header_size) return err(header_error::too_short);
174 if (h.count > h.capacity) return err(header_error::count_over_capacity);
175 detail::put_le(out, 0, mapped_header_magic);
176 detail::put_le(out, 8, mapped_header_version);
177 detail::put_le(out, 12, static_cast<std::uint32_t>(mapped_header_size));
178 detail::put_le(out, 16, h.type_tag);
179 detail::put_le(out, 24, h.element_size);
180 detail::put_le(out, 28, h.element_align);
181 detail::put_le(out, 32, h.count);
182 detail::put_le(out, 40, h.capacity);
183 detail::put_le(out, 48, h.user);
184 detail::put_le(out, 56, detail::header_checksum(out));
185 return ok();
186}

192[[nodiscard]] constexpr result<mapped_header, header_error> decode_header(std::span<const std::byte> in) noexcept
193{
194 if (in.size() < mapped_header_size) return err(header_error::too_short);
195 if (detail::get_le<std::uint64_t>(in, 0) != mapped_header_magic) return err(header_error::bad_magic);
196 if (detail::get_le<std::uint32_t>(in, 8) != mapped_header_version) return err(header_error::bad_version);
197 if (detail::get_le<std::uint32_t>(in, 12) != mapped_header_size) return err(header_error::bad_size);
198 if (detail::get_le<std::uint64_t>(in, 56) != detail::header_checksum(in)) return err(header_error::bad_checksum);
199 mapped_header h{detail::get_le<std::uint64_t>(in, 16), detail::get_le<std::uint32_t>(in, 24),
200 detail::get_le<std::uint32_t>(in, 28), detail::get_le<std::uint64_t>(in, 32),
201 detail::get_le<std::uint64_t>(in, 40), detail::get_le<std::uint64_t>(in, 48)};
202 if (h.count > h.capacity) return err(header_error::count_over_capacity);
203 return ok(h);
204}

211[[nodiscard]] constexpr result<void, header_error> validate_header(std::span<const std::byte> in,
212 const mapped_header& expected) noexcept
213{
214 auto h = decode_header(in);
215 if (h.is_err()) return err(h.unwrap_err_unchecked());
216 if (h->element_size != expected.element_size || h->element_align != expected.element_align)
217 return err(header_error::element_mismatch);
218 if (h->type_tag != expected.type_tag) return err(header_error::type_mismatch);
219 return ok();
220}
