115 template <transformer_config C, typename T>
116 struct scan
117 {
118 using value_type = T;
119 static constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups;
120 static constexpr std::size_t state_size = H * N * P, work_size = 0;
121 static constexpr T scale = mixer_scale_v<C, T>;
122
123 gla_inputs<T> in{};
124 std::span<T> o{};
125 std::span<const T> dout{};
126 gla_gradients<T> g{};
127
128 constexpr void update(std::size_t row, std::span<T> state, bool output) const
129 {
130 for (std::size_t h = 0; h < H; ++h) {
131 const std::size_t grp = h / (H / G);
132 const auto q = in.q.subspan((row * G + grp) * N, N), k = in.k.subspan((row * G + grp) * N, N);
133 const auto v = in.v.subspan((row * H + h) * P, P);
134 const auto s = state.subspan(h * N * P, N * P);
135 std::array<T, P> out{};
136 for (std::size_t n = 0; n < N; ++n) {
137 const T qn = q[n] * scale, gate = math::exp(in.gk[(row * H + h) * N + n]);
138 for (std::size_t p = 0; p < P; ++p) {
139 s[n * P + p] = s[n * P + p] * gate + v[p] * k[n];
140 out[p] += s[n * P + p] * qn;
141 }
142 }
143 if (output)
144 for (std::size_t p = 0; p < P; ++p)
145 o[(row * H + h) * P + p] = out[p];
146 }
147 }
148 constexpr void step(std::size_t row, std::span<T> state) const { update(row, state, true); }
149 constexpr void advance(std::size_t row, std::span<T> state) const { update(row, state, false); }

151 constexpr void step_backward(std::size_t row, std::span<const T> prev, std::span<T> dstate, std::span<T>) const
152 {
153 for (std::size_t i = 0; i < G * N; ++i)
154 g.q[row * G * N + i] = g.k[row * G * N + i] = T(0);
155 for (std::size_t h = 0; h < H; ++h) {
156 const std::size_t grp = h / (H / G);
157 const auto q = in.q.subspan((row * G + grp) * N, N), k = in.k.subspan((row * G + grp) * N, N);
158 const auto v = in.v.subspan((row * H + h) * P, P);
159 const auto d = dout.subspan((row * H + h) * P, P);
160 const auto s0 = prev.subspan(h * N * P, N * P);
161 const auto ds = dstate.subspan(h * N * P, N * P);
162 std::array<T, P> dv{};
163 for (std::size_t n = 0; n < N; ++n) {
164 const T gate = math::exp(in.gk[(row * H + h) * N + n]);
165 T dq = T(0), dk = T(0), dg = T(0);
166 for (std::size_t p = 0; p < P; ++p) {
167 const T st = s0[n * P + p] * gate + v[p] * k[n];
168 dq += d[p] * st;
169 const T dsp = ds[n * P + p] + q[n] * scale * d[p];
170 dk += dsp * v[p];
171 dv[p] += dsp * k[n];
172 dg += dsp * s0[n * P + p];
173 ds[n * P + p] = dsp * gate;
174 }
175 g.q[(row * G + grp) * N + n] += dq * scale;
176 g.k[(row * G + grp) * N + n] += dk;
177 g.gk[(row * H + h) * N + n] = dg * gate;
178 }
179 for (std::size_t p = 0; p < P; ++p)
180 g.v[(row * H + h) * P + p] = dv[p];
181 }
182 }
183 };

185 template <transformer_config C, typename T>
186 constexpr std::size_t rows_of(const gla_inputs<T>& in)
187 {
188 [[maybe_unused]] constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups;
189 static_assert(valid(C) && C.mixer_heads > 0, "snicholls::ml::gla: the configuration has no mixer shape");
190 static_cast<void>(N), static_cast<void>(G); // read only by the assert (MSVC ignores [[maybe_unused]] there, C4189)
191 const std::size_t rows = in.v.size() / (H * P);
192 assert(in.v.size() == rows * H * P && in.q.size() == rows * G * N && in.k.size() == rows * G * N && in.gk.size() == rows * H * N);
193 return rows;
194 }

217template <transformer_config C, typename B, math::math_number T>
218constexpr void gla_forward_chunked(const B&, const gla_inputs<T>& in, std::type_identity_t<std::span<T>> o, std::size_t sequences,
219 std::type_identity_t<std::span<T>> states, std::type_identity_t<std::span<T>> checkpoints,
220 std::type_identity_t<std::span<T>> work)
221{
222 constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups, L = C.mixer_chunk;
223 constexpr std::size_t S = H * N * P;
224 constexpr T scale = mixer_scale_v<C, T>;
225 const std::size_t rows = detail::gla::rows_of<C>(in);
226 assert(sequences > 0 && rows % sequences == 0 && o.size() == in.v.size() && states.size() == sequences * S
227 && work.size() == gla_chunk_work(C));
228 const std::size_t n = rows / sequences, m = recurrence_checkpoints(n, L);
229 assert(checkpoints.empty() || checkpoints.size() == sequences * m * S);
230 for (std::size_t sq = 0; sq < sequences; ++sq)
231 for (std::size_t ck = 0; ck < m; ++ck) {
232 const std::size_t first = sq * n + ck * L, len = n - ck * L < L ? n - ck * L : L;
233 const auto state = states.subspan(sq * S, S);
234 if (!checkpoints.empty())
235 for (std::size_t i = 0; i < S; ++i)
236 checkpoints[(sq * m + ck) * S + i] = state[i];
237 for (std::size_t h = 0; h < H; ++h) {
238 const std::size_t grp = h / (H / G);
239 const auto s = state.subspan(h * N * P, N * P);
240 const auto cg = work; // cg[j · N + n]: the cumulative log-gate
241 for (std::size_t j = 0; j < len; ++j)
242 for (std::size_t q = 0; q < N; ++q)
243 cg[j * N + q] = (j == 0 ? T(0) : cg[(j - 1) * N + q]) + in.gk[((first + j) * H + h) * N + q];
244 for (std::size_t j = 0; j < len; ++j) {
245 const auto qj = in.q.subspan(((first + j) * G + grp) * N, N);
246 const auto oj = o.subspan(((first + j) * H + h) * P, P);
247 for (std::size_t p = 0; p < P; ++p)
248 oj[p] = T(0);
249 for (std::size_t i = 0; i <= j; ++i) {
250 const auto ki = in.k.subspan(((first + i) * G + grp) * N, N);
251 T a = T(0);
252 for (std::size_t q = 0; q < N; ++q)
253 a += qj[q] * ki[q] * math::exp(cg[j * N + q] - cg[i * N + q]);
254 const auto vi = in.v.subspan(((first + i) * H + h) * P, P);
255 for (std::size_t p = 0; p < P; ++p)
256 oj[p] += a * vi[p];
257 }
258 for (std::size_t q = 0; q < N; ++q) {
259 const T qd = qj[q] * math::exp(cg[j * N + q]);
260 for (std::size_t p = 0; p < P; ++p)
261 oj[p] += qd * s[q * P + p];
262 }
263 for (std::size_t p = 0; p < P; ++p)
264 oj[p] *= scale;
265 }
266 for (std::size_t q = 0; q < N; ++q) {
267 const T last = cg[(len - 1) * N + q], total = math::exp(last);
268 for (std::size_t p = 0; p < P; ++p) {
269 T add = T(0);
270 for (std::size_t i = 0; i < len; ++i)
271 add += math::exp(last - cg[i * N + q]) * in.k[((first + i) * G + grp) * N + q] * in.v[((first + i) * H + h) * P + p];
272 s[q * P + p] = total * s[q * P + p] + add;
273 }
274 }
275 }
276 }
