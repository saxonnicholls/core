265 struct trade_lifecycle {
267 using state_enum = trade_state;
268
271 constexpr auto operator()() const noexcept
272 {
273 using namespace trade_event;
274 constexpr auto placed = enum_state<trade_state::placed>;
275 constexpr auto filled = enum_state<trade_state::filled>;
276 constexpr auto settling = enum_state<trade_state::settling>;
277 constexpr auto settled = enum_state<trade_state::settled>;
278 constexpr auto failed = enum_state<trade_state::failed>;
279 constexpr auto cancelled = enum_state<trade_state::cancelled>;
280 return make_transition_table(
281 *placed + event<fill> = filled,
282 placed + event<cancel> = cancelled,
283 filled + event<instruct> = settling,
284 settling + event<settle> = settled,
285 settling + event<fail> = failed);
286 }
287 };
