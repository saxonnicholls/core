105 template <linalg::scalar T, std::size_t MaxTruths, std::size_t MaxEstimates>
106 class set_metric_solver {
107 static_assert(MaxTruths > 0 && MaxEstimates > 0, "snicholls::metric::set_metric_solver: hold at least one point on each side");
108
109 // min(d, c)^p for every pair, and the optimal assignment of the smaller set into the larger
110 template <typename X, typename Y, typename M>
111 constexpr result<T, set_metric_error> assign(const X& x, const Y& y, T c, T p, const M& m)
112 {
113 const std::size_t nx = std::ranges::size(x), ny = std::ranges::size(y);
114 const T cp = static_cast<T>(math::pow(c, p));
115 for (std::size_t i = 0; i < nx; ++i)
116 for (std::size_t j = 0; j < ny; ++j) {
117 const T d = static_cast<T>(m(x[i], y[j]));
118 cost_[i * ny + j] = d < c ? static_cast<T>(math::pow(d, p)) : cp;
119 }
120 const auto s = solver_.solve(std::span<const T>(cost_).first(nx * ny), nx, ny, std::span<std::int32_t>(row_to_col_).first(nx));
121 if (!s)
122 return err(set_metric_error::too_large);
123 return ok(*s);
124 }

137 template <point_range X, point_range Y, typename M = euclidean>
138 requires distance_for<M, std::ranges::range_value_t<const X>, std::ranges::range_value_t<const Y>>
139 [[nodiscard]] constexpr result<T, set_metric_error> ospa(const X& truths, const Y& estimates, T c, T p = T(1), const M& m = M{}) noexcept(linalg::detail::nothrow_scalar<T> && noexcept(std::declval<const M&>()(std::declval<std::ranges::range_reference_t<const X>>(), std::declval<std::ranges::range_reference_t<const Y>>())))
140 {
141 if (!(c > T(0)) || !(p >= T(1)))
142 return err(set_metric_error::invalid_parameters);
143 const std::size_t nx = std::ranges::size(truths), ny = std::ranges::size(estimates);
144 if (nx > MaxTruths || ny > MaxEstimates)
145 return err(set_metric_error::too_large);
146 if (nx == 0 && ny == 0)
147 return ok(T(0));
148 if (nx == 0 || ny == 0)
149 return ok(c);
150 const auto s = assign(truths, estimates, c, p, m);
151 if (!s)
152 return s;
153 const std::size_t big = nx > ny ? nx : ny, small = nx > ny ? ny : nx;
154 const T total = *s + static_cast<T>(math::pow(c, p)) * static_cast<T>(big - small);
155 return ok(static_cast<T>(math::pow(total / static_cast<T>(big), T(1) / p)));
156 }

169 template <point_range X, point_range Y, typename M = euclidean>
170 requires distance_for<M, std::ranges::range_value_t<const X>, std::ranges::range_value_t<const Y>>
171 [[nodiscard]] constexpr result<gospa_result<T>, set_metric_error> gospa(const X& truths, const Y& estimates, T c, T p = T(1), T alpha = T(2), const M& m = M{}) noexcept(linalg::detail::nothrow_scalar<T> && noexcept(std::declval<const M&>()(std::declval<std::ranges::range_reference_t<const X>>(), std::declval<std::ranges::range_reference_t<const Y>>())))
172 {
173 if (!(c > T(0)) || !(p >= T(1)) || !(alpha > T(0)) || alpha > T(2))
174 return err(set_metric_error::invalid_parameters);
175 const std::size_t nx = std::ranges::size(truths), ny = std::ranges::size(estimates);
176 if (nx > MaxTruths || ny > MaxEstimates)
177 return err(set_metric_error::too_large);
178 gospa_result<T> g{};
179 const T cp = static_cast<T>(math::pow(c, p));
180 T assigned(0);
181 std::size_t paired = 0;
182 if (nx > 0 && ny > 0) {
183 const auto s = assign(truths, estimates, c, p, m);
184 if (!s)
185 return err(s.unwrap_err_unchecked());
186 assigned = *s;
187 for (std::size_t i = 0; i < nx; ++i) {
188 const auto j = row_to_col_[i];
189 if (j < 0)
190 continue;
191 const T d = static_cast<T>(m(truths[i], estimates[static_cast<std::size_t>(j)]));
192 if (d < c) {
193 g.localisation += static_cast<T>(math::pow(d, p));
194 ++paired;
195 }
196 }
197 }
198 g.missed = nx - paired;
199 g.false_tracks = ny - paired;
200 const std::size_t mismatch = nx > ny ? nx - ny : ny - nx;
201 g.distance = static_cast<T>(math::pow(assigned + cp / alpha * static_cast<T>(mismatch), T(1) / p));
202 return ok(g);
203 }
204
205 private:
206 std::array<T, MaxTruths * MaxEstimates> cost_{};
207 std::array<std::int32_t, MaxTruths> row_to_col_{};
208 combinatorics::assignment_solver<T, MaxTruths, MaxEstimates> solver_{};
209 };

220 template <point_range X, point_range Y, typename M = euclidean>
221 requires distance_for<M, std::ranges::range_value_t<const X>, std::ranges::range_value_t<const Y>>
222 [[nodiscard]] constexpr double hausdorff(const X& x, const Y& y, const M& m = M{})
223 {
224 const std::size_t nx = std::ranges::size(x), ny = std::ranges::size(y);
225 if (nx == 0 && ny == 0)
226 return 0.0;
227 if (nx == 0 || ny == 0)
228 return std::numeric_limits<double>::infinity();
229 double worst = 0;
230 for (std::size_t i = 0; i < nx; ++i) {
231 double best = std::numeric_limits<double>::infinity();
232 for (std::size_t j = 0; j < ny; ++j) {
233 const double d = static_cast<double>(m(x[i], y[j]));
234 best = d < best ? d : best;
235 }
236 worst = best > worst ? best : worst;
237 }
238 for (std::size_t j = 0; j < ny; ++j) {
239 double best = std::numeric_limits<double>::infinity();
240 for (std::size_t i = 0; i < nx; ++i) {
241 const double d = static_cast<double>(m(x[i], y[j]));
242 best = d < best ? d : best;
243 }
244 worst = best > worst ? best : worst;
245 }
246 return worst;
247 }
