101 // RFC 7693 3.2, F: the compression of one block at byte offset (t_low, t_high), `last` for the final block.
102 template <typename T, typename U>
103 constexpr void blake2_compress(std::array<typename T::word, 8>& h, const U* block, typename T::word t_low,
104 typename T::word t_high, bool last) noexcept
105 {
106 using W = typename T::word;
107 W m[16]{};
108 for (std::size_t i = 0; i < 16; ++i)
109 m[i] = load_le_word<W>(block + i * sizeof(W));
110 W v[16]{};
111 for (std::size_t i = 0; i < 8; ++i) {
112 v[i] = h[i];
113 v[i + 8] = T::iv[i];
114 }
115 v[12] ^= t_low;
116 v[13] ^= t_high;
117 if (last)
118 v[14] = static_cast<W>(~v[14]);
119 const auto g = [&v](std::size_t a, std::size_t b, std::size_t c, std::size_t d, W x, W y) {
120 v[a] = static_cast<W>(v[a] + v[b] + x);
121 v[d] = rotr<W>(v[d] ^ v[a], T::r1);
122 v[c] = static_cast<W>(v[c] + v[d]);
123 v[b] = rotr<W>(v[b] ^ v[c], T::r2);
124 v[a] = static_cast<W>(v[a] + v[b] + y);
125 v[d] = rotr<W>(v[d] ^ v[a], T::r3);
126 v[c] = static_cast<W>(v[c] + v[d]);
127 v[b] = rotr<W>(v[b] ^ v[c], T::r4);
128 };
129 for (std::size_t r = 0; r < T::rounds; ++r) {
130 const std::uint8_t* s = blake2_sigma[r];
131 g(0, 4, 8, 12, m[s[0]], m[s[1]]);
132 g(1, 5, 9, 13, m[s[2]], m[s[3]]);
133 g(2, 6, 10, 14, m[s[4]], m[s[5]]);
134 g(3, 7, 11, 15, m[s[6]], m[s[7]]);
135 g(0, 5, 10, 15, m[s[8]], m[s[9]]);
136 g(1, 6, 11, 12, m[s[10]], m[s[11]]);
137 g(2, 7, 8, 13, m[s[12]], m[s[13]]);
138 g(3, 4, 9, 14, m[s[14]], m[s[15]]);
139 }
140 for (std::size_t i = 0; i < 8; ++i)
141 h[i] ^= v[i] ^ v[i + 8];
142 }

150template <typename Traits, std::size_t N>
151class blake2_hash : public detail::one_shot_hash<blake2_hash<Traits, N>>
152{
153 static_assert(N >= 1 && N <= Traits::max_size, "snicholls::blake2: the output length is 1 to 64 bytes (BLAKE2b) or 32 (BLAKE2s)");
154 using word = typename Traits::word;
155
156public:
158 static constexpr std::size_t block_size = Traits::block_size;
160 static constexpr std::size_t digest_size = N;
162 static constexpr std::size_t max_key_size = Traits::max_size;
164 using digest_type = digest<N>;
165
167 constexpr blake2_hash() noexcept { init(0); }
168
172 template <hash_input K>
173 constexpr explicit blake2_hash(const K& key)
174 {
175 const std::size_t n = detail::input_size(key);
176 if (n > max_key_size)
177 throw std::length_error("snicholls::blake2: the key is longer than the variant allows");
178 init(n);
179 if (n != 0) {
180 const auto* k = detail::input_data(key);
181 for (std::size_t i = 0; i < n; ++i)
182 buf_[i] = detail::octet_of(k[i]);
183 fill_ = block_size; // the key block is the first block, held back like any other
184 }
185 }
186
192 template <hash_input K, hash_input S>
193 [[nodiscard]] static constexpr digest_type keyed_hash(const K& key, const S& input)
194 {
195 blake2_hash h{key};
196 h.update(input);
197 return h.finish();
198 }
199
202 [[nodiscard]] constexpr digest_type finish() noexcept
203 {
204 count(fill_);
205 for (std::size_t i = fill_; i < block_size; ++i)
206 buf_[i] = 0;
207 detail::blake2_compress<Traits>(h_, buf_.data(), t_low_, t_high_, true);
208 std::array<std::uint8_t, 8 * sizeof(word)> bytes{};
209 for (std::size_t i = 0; i < 8; ++i)
210 detail::store_le_word(bytes.data() + i * sizeof(word), h_[i]);
211 digest_type out{};
212 for (std::size_t i = 0; i < N; ++i)
213 out.bytes[i] = static_cast<std::byte>(bytes[i]);
214 return out;
215 }
216
217private:
218 friend detail::hash_updates<blake2_hash>;
219
220 constexpr void init(std::size_t key_size) noexcept
221 {
222 for (std::size_t i = 0; i < 8; ++i)
223 h_[i] = Traits::iv[i];
224 h_[0] ^= static_cast<word>(0x01010000u ^ (key_size << 8) ^ N);
225 }
226
227 constexpr void count(std::size_t n) noexcept
228 {
229 t_low_ = static_cast<word>(t_low_ + n);
230 if (t_low_ < n)
231 ++t_high_;
232 }
233
234 template <typename U>
235 constexpr void absorb(const U* p, std::size_t n) noexcept
236 {
237 while (n != 0) {
238 if (fill_ == block_size) { // more input follows, so the held block is not the last
239 count(block_size);
240 detail::blake2_compress<Traits>(h_, buf_.data(), t_low_, t_high_, false);
241 fill_ = 0;
242 }
243 if (fill_ == 0 && n > block_size) { // a whole block that is not the last: straight from the input
244 count(block_size);
245 detail::blake2_compress<Traits>(h_, p, t_low_, t_high_, false);
246 p += block_size;
247 n -= block_size;
248 continue;
249 }
250 const std::size_t take = n < block_size - fill_ ? n : block_size - fill_;
251 for (std::size_t i = 0; i < take; ++i)
252 buf_[fill_ + i] = detail::octet_of(p[i]);
253 fill_ += take;
254 p += take;
255 n -= take;
256 }
257 }
258
259 std::array<word, 8> h_{};
260 std::array<std::uint8_t, Traits::block_size> buf_{};
261 std::size_t fill_ = 0;
262 word t_low_ = 0, t_high_ = 0;
263};
