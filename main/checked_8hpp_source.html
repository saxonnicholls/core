70 [[nodiscard]] constexpr std::optional<T> checked_add(T a, T b) noexcept
71 {
72#if defined(__GNUC__) || defined(__clang__)
73 T r{};
74 if (!__builtin_add_overflow(a, b, &r))

76 return std::nullopt;

111 template <std::integral T>
112 [[nodiscard]] constexpr std::optional<T> checked_mul(T a, T b) noexcept
113 {
114#if defined(__GNUC__) || defined(__clang__)
115 T r{};
116 if (!__builtin_mul_overflow(a, b, &r))
117 return r;
118 return std::nullopt;
119#else
120 if (a == 0 || b == 0)
121 return T{0};
122 // min * -1 first: the division below would itself overflow on it
123 if (std::is_signed_v<T> && ((a == T(-1) && b == std::numeric_limits<T>::min()) || (b == T(-1) && a == std::numeric_limits<T>::min())))
124 return std::nullopt;
125 const T r = static_cast<T>(static_cast<std::make_unsigned_t<T>>(a) * static_cast<std::make_unsigned_t<T>>(b));
126 if (r / b != a)
127 return std::nullopt;
128 return r;
129#endif
130 }

174 template <typename To, typename From>
175 requires std::is_arithmetic_v<To> && std::is_arithmetic_v<From>
176 [[nodiscard]] constexpr std::optional<To> numeric_cast(From v) noexcept
177 {
178 if constexpr (std::integral<To> && std::integral<From>) {
179 return std::in_range<To>(v) ? std::optional<To>(static_cast<To>(v)) : std::nullopt;
180 } else {
181 if (v != v) // NaN survives only into a floating type
182 return std::floating_point<To> ? std::optional<To>(static_cast<To>(v)) : std::nullopt;
183 if (std::integral<To> && (v < static_cast<From>(std::numeric_limits<To>::lowest()) ||
184 v >= static_cast<From>(std::numeric_limits<To>::max()) + static_cast<From>(1)))
185 return std::nullopt;
186 const To t = static_cast<To>(v);
187 return static_cast<From>(t) == v ? std::optional<To>(t) : std::nullopt;
188 }

192 // Whether the standard library converts T with std::from_chars: every one does integers; Xcode 16's libc++
193 // and older have no floating-point overloads (a double would reach the deleted bool overload).
194 template <typename T>
195 concept has_from_chars = requires(const char* p, T& v) {
196 { std::from_chars(p, p, v) } -> std::same_as<std::from_chars_result>;
197 };

199 // The text std::from_chars accepts for a floating-point number in its general format: an optional minus, then
200 // digits with an optional point and exponent, or inf, infinity, nan, nan(chars), in any case. No leading
201 // space, no plus, no hexadecimal.
202 constexpr bool float_syntax(std::string_view t) noexcept
203 {
204 const auto lower = [](char c) { return static_cast<char>(c >= 'A' && c <= 'Z' ? c - 'A' + 'a' : c); };
205 const auto starts = [&](std::string_view w) {
206 if (t.size() < w.size())
207 return false;
208 for (std::size_t i = 0; i < w.size(); ++i)
209 if (lower(t[i]) != w[i])
210 return false;
211 return true;
212 };
213 const auto digit = [](char c) { return c >= '0' && c <= '9'; };
214 if (!t.empty() && t.front() == '-')
215 t.remove_prefix(1);
216 if (starts("infinity"))
217 return t.size() == 8;
218 if (starts("inf"))
219 return t.size() == 3;
220 if (starts("nan")) {
221 t.remove_prefix(3);
222 if (t.empty())
223 return true;
224 if (t.front() != '(' || t.back() != ')')
225 return false;
226 for (const char c : t.substr(1, t.size() - 2))
227 if (!(digit(c) || (lower(c) >= 'a' && lower(c) <= 'z') || c == '_'))
228 return false;
229 return true;
230 }
231 std::size_t i = 0, mantissa = 0;
232 while (i < t.size() && digit(t[i]))
233 ++i, ++mantissa;
234 if (i < t.size() && t[i] == '.') {
235 ++i;
236 while (i < t.size() && digit(t[i]))
237 ++i, ++mantissa;
238 }
239 if (mantissa == 0)
240 return false;
241 if (i < t.size() && (t[i] == 'e' || t[i] == 'E')) {
242 ++i;
243 if (i < t.size() && (t[i] == '+' || t[i] == '-'))
244 ++i;
245 const std::size_t exponent = i;
246 while (i < t.size() && digit(t[i]))
247 ++i;
248 if (i == exponent)
249 return false;
250 }
251 return i == t.size();
252 }

258 // The fallback where std::from_chars has no floating-point overload: the same syntax, checked first, then the
259 // C library's correctly rounded conversion. The C library reads the decimal point from LC_NUMERIC, which is
260 // "." unless the program calls setlocale; up to 255 characters.
261 template <std::floating_point T>
262 result<T, std::errc> parse_float(std::string_view text) noexcept
263 {
264 if (!float_syntax(text) || text.size() > 255)
265 return err(std::errc::invalid_argument);
266 std::array<char, 256> buffer{};
267 std::copy(text.begin(), text.end(), buffer.begin());
268 char* end = nullptr;
269 errno = 0;
270 const T value = c_parse(buffer.data(), &end, static_cast<T*>(nullptr));
271 if (end != buffer.data() + text.size())
272 return err(std::errc::invalid_argument);
273 if (errno == ERANGE && (value > T(1) || value < T(-1))) // overflow; an underflow is the nearest value
274 return err(std::errc::result_out_of_range);
275 return ok(value);
276 }
