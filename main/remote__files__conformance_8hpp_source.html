72 template <typename Probe>
73 concept remote_files_probe = std::is_default_constructible_v<Probe> &&
74 remote_files<std::remove_reference_t<decltype(std::declval<Probe&>().files())>> &&
75 requires(Probe& p, std::string_view n, std::span<const std::byte> b, utc_s t, capability_errc c) {
76 p.seed(n, b, t);
77 p.provider_fails(c, n);
78 };

80 // n bytes of a fixed pattern, different for each seed
81 inline std::vector<std::byte> kit_bytes(std::size_t n, unsigned seed)
82 {
83 std::vector<std::byte> out(n);
84 std::uint32_t x = 0x9e3779b9u ^ seed;
85 for (auto& b : out) {
86 x ^= x << 13;
87 x ^= x >> 17;
88 x ^= x << 5;
89 b = static_cast<std::byte>(x);
90 }
91 return out;
92 }
