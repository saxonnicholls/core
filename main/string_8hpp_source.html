74 // long along the outer loop, short along the row
75 template <typename L, typename S>
76 static constexpr std::size_t run(const L& lng, std::size_t nl, const S& sht, std::size_t ns) noexcept(noexcept(coordinate(lng, 0)) && noexcept(coordinate(sht, 0)))
77 {
78 if (ns > MaxLength)
79 return std::numeric_limits<std::size_t>::max();
80 std::array<std::size_t, MaxLength + 1> row{};
81 for (std::size_t j = 0; j <= ns; ++j)
82 row[j] = j;
83 for (std::size_t i = 1; i <= nl; ++i) {
84 std::size_t diagonal = row[0];
85 row[0] = i;
86 for (std::size_t j = 1; j <= ns; ++j) {
87 const std::size_t above = row[j];
88 const std::size_t substitute = diagonal + (coordinate(lng, i - 1) == coordinate(sht, j - 1) ? 0u : 1u);
89 const std::size_t remove = above + 1, insert = row[j - 1] + 1;
90 std::size_t best = substitute < remove ? substitute : remove;
91 best = insert < best ? insert : best;
92 row[j] = best;
93 diagonal = above;
94 }
95 }
96 return row[ns];
97 }
