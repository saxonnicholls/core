51// Implementation notes, as written in ts-moveables:
52//
53// constexpr_for<0, 4> ([&](auto I) { use<I.value>(); }); // fully unrolled
54// constexpr_for<0, 10, 2> ([&](auto I) { even(I); }); // with a step
55// constexpr_for<10, 0, -1> ([&](auto I) { down(I); }); // and backwards
56// constexpr_for<0, 3> (f, a, b); // f(I, a, b)
57// unrolled_for<4>(0, n, [&](std::size_t i) { out[i] = g(in[i]); });
58//
59// constexpr_for<Start, End, Inc> expands into exactly the calls it needs and
60// leaves no loop at all - the bounds are template parameters, so there is no
61// counter, no compare and no branch. unrolled_for takes a RUN-TIME range and
62// emits the body Unroll times per iteration, with a tail for the remainder.
63//
64// ------------------------------------------------------ the index is a type
65//
66// The body is handed std::integral_constant<T, I>, not a bare T, and that is
67// the whole reason this is worth having over a plain loop.
68//
69// A function parameter is never a constant expression, however constant the
70// value passed in was - so a body taking `std::size_t i` cannot write
71// std::get<i>(tuple) or use i as a template argument, which is usually the
72// entire point of unrolling by hand. Handing it a type carrying the index
73// fixes that: `I.value` is a constant expression inside the body, while `I`
74// still converts implicitly to its underlying type, so the ordinary
75// arithmetic spelling keeps working:
76//
77// constexpr_for<0, N>([&](auto I) { sum += v[I]; }); // fine
78// constexpr_for<0, N>([&](auto I) { use(std::get<I.value>(tup)); }); // also fine
79//
80// The index type is decltype(Start), so constexpr_for<0, 4> yields int and
81// constexpr_for<std::size_t{0}, std::size_t{4}> yields std::size_t. Say which
82// you want when it matters.
83//
84// ------------------------------------------------------------ expansion, not
85// recursion
86// The obvious implementation recurses - body, then constexpr_for<Start+Inc,
87// End, Inc> - and it is shorter. This expands an index_sequence through a fold
88// instead, because the recursive form instantiates one template per iteration:
89// template depth grows with the count, a few hundred iterations meets the
90// compiler's instantiation-depth limit, and compile time grows superlinearly
91// well before that. The fold is depth 1 whatever the count. Same unrolled
92// output, no cliff to fall off.
93//
94// ------------------------------------------------------------ extra arguments
95//
96// Trailing arguments are passed through to every call as LVALUES, never
97// forwarded. Forwarding would move the same object once per iteration, which
98// is a use-after-move that compiles quietly; if a body needs to consume
99// something it should capture it instead.
100//
101// ------------------------------------------------------------- when to reach
102//
103// Be sceptical of unrolled_for as a speed tool. At -O2 the compiler already
104// unrolls simple counted loops, and doing it by hand can lose: more code in
105// the instruction cache, more live registers, and a tail that confuses the
106// vectoriser. `make bench-parallel` prints unrolled against not so the claim
107// stays falsifiable rather than folkloric.
108//
109// Where it does earn its place: when the body must be instantiated per index -
110// a tuple element, a template argument, a fixed-size matrix row - which no
111// compiler unroll can do for you, because that is a language requirement
112// rather than an optimisation. Reach for constexpr_for for that; reach for
113// unrolled_for only with a benchmark in hand.
114//

127 // How many iterations [Start, End) by Inc actually performs. Computed
128 // in long long so a negative Inc against unsigned bounds cannot wrap -
129 // the arithmetic that makes a hand-rolled version of this silently
130 // produce 18 quintillion iterations.
131 constexpr std::size_t constexpr_for_count(long long start, long long end, long long inc)
132 {
133 const long long span = (inc > 0) ? (end - start) : (start - end);
134 const long long step = (inc > 0) ? inc : -inc;
135 return span > 0 ? static_cast<std::size_t>((span + step - 1) / step) : 0u;
136 }

138 template <auto Start, auto Inc, typename F, std::size_t... I, typename... Args>
139 constexpr void constexpr_for_impl(F&& f, std::index_sequence<I...>, Args&... args)
140 {
141 using T = decltype(Start);
142 // Fold over the comma operator: one call per index, in order, and
143 // nothing at all generated for an empty sequence.
144 (static_cast<void>(f(std::integral_constant<
145 T, static_cast<T>(static_cast<long long>(Start) +
146 static_cast<long long>(I) *
147 static_cast<long long>(Inc))>{},
148 args...)),
149 ...);
150 }
151 } // namespace detail

161 // Invoke f(I, args...) for each I in [Start, End) stepping by Inc, fully
162 // unrolled at compile time. Inc may be negative to count down. I is an
163 // integral_constant<decltype(Start), ...>.
164 template <auto Start, auto End, auto Inc = 1, typename F, typename... Args>
165 constexpr void constexpr_for(F&& f, Args&&... args)
166 {
167 static_assert(Inc != 0, "constexpr_for with a step of 0 would never terminate");
168 constexpr std::size_t n = detail::constexpr_for_count(
169 static_cast<long long>(Start), static_cast<long long>(End),
170 static_cast<long long>(Inc));
171 detail::constexpr_for_impl<Start, Inc>(std::forward<F>(f),
172 std::make_index_sequence<n>{}, args...);
173 }

181 // Invoke f(I, args...) for each I in [0, Count) - the common case, shorter.
182 // Disjoint from the overload above: that one cannot deduce End, this one
183 // cannot take a second non-type argument.
184 template <auto Count, typename F, typename... Args>
185 constexpr void constexpr_for(F&& f, Args&&... args)
186 {
187 constexpr_for<static_cast<decltype(Count)>(0), Count, 1>(std::forward<F>(f), args...);
188 }

192 // N-deep nest, peeled one extent at a time. The recursion here is over
193 // DEPTH (2, 3, 4...), not over iteration count, so it is three or four
194 // instantiations deep rather than thousands - the cliff the flat
195 // constexpr_for avoids by folding does not exist on this axis.
196 template <std::size_t... Extents>
197 struct nest;
198
199 template <>
200 struct nest<> {
201 template <typename F, typename... Acc>
202 static constexpr void run(F& f, Acc... acc) { f(acc...); }
203 };
204
205 template <std::size_t E0, std::size_t... Rest>
206 struct nest<E0, Rest...> {
207 template <typename F, typename... Acc>
208 static constexpr void run(F& f, Acc... acc)
209 {
210 constexpr_for<std::size_t{0}, E0>(
211 [&](auto I) { nest<Rest...>::run(f, acc..., I); });
212 }
213 };
214 } // namespace detail

220 // Nested loops, every level unrolled, to any depth:
221 //
222 // constexpr_nest<2, 3> ([&](auto I, auto J) { m[I][J] += a[I]*b[J]; });
223 // constexpr_nest<2, 3, 4>([&](auto I, auto J, auto K) { t[I][J][K] = 0; });
224 //
225 // The body is called once per point of the cartesian product, in row-major
226 // order (last extent varies fastest), with one integral_constant per level
227 // - so every index is a constant expression and a fixed-size kernel can be
228 // written with no loop and no run-time indexing at all.
229 //
230 // Mind the multiplication: <8,8,8> is 512 expansions of the body, and the
231 // object code grows with it. This is for small fixed extents - a 4x4
232 // transform, a tile of a matrix kernel - not for whole arrays.
233 template <std::size_t... Extents, typename F>
234 constexpr void constexpr_nest(F&& f)
235 {
236 detail::nest<Extents...>::run(f);
237 }

248 // Walk a RUN-TIME range [first, last), calling body(i, args...) for each i,
249 // with the body emitted Unroll times per iteration and a tail loop for the
250 // remainder.
251 //
252 // body receives a plain Index here, not an integral_constant: the value is
253 // genuinely a run-time one, and pretending otherwise would be a lie that
254 // only compiles when the bounds happen to be constant.
255 template <std::size_t Unroll = 4, typename Index, typename Body, typename... Args>
256 constexpr void unrolled_for(Index first, Index last, Body body, Args&... args)
257 {
258 static_assert(std::is_integral_v<Index>, "unrolled_for needs an integral index");
259 static_assert(Unroll >= 1, "unrolled_for needs an unroll factor of at least 1");
260
261 if (!(first < last))
262 return;
263
264 const auto count = static_cast<std::size_t>(last - first);
265 const auto whole = count - (count % Unroll);
266 const Index main_end = first + static_cast<Index>(whole);
267
268 Index i = first;
269 for (; i < main_end; i += static_cast<Index>(Unroll))
270 constexpr_for<std::size_t{0}, Unroll>(
271 [&](auto K) { body(i + static_cast<Index>(K.value), args...); });
272 for (; i < last; ++i) // the tail Unroll did not divide
273 body(i, args...);
274 }
275} // namespace snicholls
