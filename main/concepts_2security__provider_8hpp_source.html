199 [[nodiscard]] constexpr bool is_instrument_code(std::string_view code) noexcept
200 {
201 // A loop rather than find(): GCC 13 refuses string_view::find over an asset's own characters in a constant
202 // expression.
203 std::size_t colon = 0, colons = 0;
204 for (std::size_t i = 0; i < code.size(); ++i)
205 if (code[i] == ':') {
206 colon = i;
207 ++colons;
208 }
209 if (!is_asset_code(code) || colons != 1 || colon == 0 || colon + 1 == code.size())
210 return false;
211 for (std::size_t i = colon + 1; i < code.size(); ++i)
212 if (!((code[i] >= 'A' && code[i] <= 'Z') || (code[i] >= '0' && code[i] <= '9')))
213 return false;
214 return true;
215 }

297 [[nodiscard]] constexpr std::optional<money> cash_movement(const security_fact& f) noexcept
298 {
299 auto net = subtract(f.consideration, f.fees);
300 for (const auto& tax : {f.taxes.duty, f.taxes.withheld})
301 if (tax && net)
302 net = subtract(*net, *tax);
303 if (!net)
304 return std::nullopt;
305 return *net;
306 }

323 [[nodiscard]] constexpr capability_result<void> validate(const security_fact& f) noexcept
324 {
325 using enum security_fact_kind;
326 const auto bad = [](capability_message why) noexcept { return fail(capability_errc::invalid_argument, {}, why); };
327 if (f.source.empty() || f.fact_id.empty())
328 return bad("a fact needs its source and fact id");
329 if (const auto k = static_cast<std::uint8_t>(f.kind); k < 1 || k > static_cast<std::uint8_t>(distribution_reinvested))
330 return bad("not a security fact kind");
331 if (auto dates = validate(f.dates); !dates)
332 return dates;
333 if (!is_instrument_code(f.instrument.code()))
334 return bad("an instrument is <SYMBOL>:<VENUE>");
335 if (f.quantity.asset_of() != f.instrument)
336 return bad("the quantity is in the instrument");
337 if (f.isin && !validate_isin(f.isin->view()))
338 return bad("not a valid ISIN");
339 if (needs_other_instrument(f.kind) && !f.other_instrument)
340 return bad("this action names its other instrument");
341 if (f.other_instrument && (!is_instrument_code(f.other_instrument->code()) || *f.other_instrument == f.instrument))
342 return bad("the other instrument is another <SYMBOL>:<VENUE>");
343 if (f.other_quantity && (!f.other_instrument || f.other_quantity->asset_of() != *f.other_instrument))
344 return bad("the other quantity is in the other instrument");
345 if (f.cost_apportionment && !f.other_instrument)
346 return bad("a cost apportionment goes to the other instrument");
347 if (!detail::unit_interval(f.cost_apportionment))
348 return bad("a cost apportionment is from 0 to 1");
349 if (f.kind == demerger && !f.cost_apportionment)
350 return bad("a demerger apportions the cost base");
351 if (needs_ratio(f.kind) && !f.new_for_old)
352 return bad("this action has a new-for-old ratio");
353 if (f.new_for_old) {
354 const finance::ratio one{1, 1}, r = *f.new_for_old;
355 if (r <= finance::ratio{} || (f.kind == split && r <= one) || (f.kind == consolidation && r >= one))
356 return bad("a ratio is positive, a split's above 1, a consolidation's below");
357 }
358 const asset& currency = f.consideration.asset_of();
359 if (f.fees.asset_of() != currency || f.fees.is_negative() || !detail::cash_of(f.taxes.withheld, currency) ||
360 !detail::cash_of(f.taxes.duty, currency) || !detail::cash_of(f.taxes.franking_credit, currency))
361 return bad("fees and taxes are not negative, in the cash currency");
362 if (f.price && (f.price->is_negative() || f.price->asset_of().code() != currency.code() ||
363 f.price->asset_of().scale() < currency.scale()))
364 return bad("a price is not negative, in the cash currency");
365 if ((f.kind == buy && (f.quantity.signum() <= 0 || f.consideration.signum() > 0)) ||
366 (f.kind == sell && (f.quantity.signum() >= 0 || f.consideration.is_negative())) ||
367 (f.kind == application && (f.quantity.is_negative() || f.consideration.signum() > 0)) ||
368 (f.kind == redemption && (f.quantity.signum() > 0 || f.consideration.is_negative())))
369 return bad("a purchase adds units and pays; a sale removes and receives");
370 if ((f.taxes.franking_credit || f.taxes.franked) && f.kind != dividend_entitlement && f.kind != dividend_paid &&
371 f.kind != dividend_reinvested && f.kind != distribution_reinvested)
372 return bad("franking is a dividend's");
373 if (!detail::unit_interval(f.taxes.franked))
374 return bad("the franked part is from 0 to 1");
375 if ((f.ex_date || f.record_date || f.pay_date) && !is_corporate_action(f.kind) && f.kind != distribution_reinvested)
376 return bad("ex, record and pay dates are a corporate action's");
377 if ((f.ex_date && f.record_date && *f.ex_date > *f.record_date) ||
378 (f.record_date && f.pay_date && *f.record_date > *f.pay_date) || (f.ex_date && f.pay_date && *f.ex_date > *f.pay_date))
379 return bad("ex, then record, then pay");
380 if ((f.kind == dividend_entitlement || f.kind == capital_return || needs_ratio(f.kind) || f.kind == demerger) &&
381 !f.record_date)
382 return bad("an entitlement has its record date");
383 if (is_repo(f.kind) && !f.repo)
384 return bad("a repo fact carries its terms");
385 if (f.repo) {
386 if (f.repo->counterparty.empty() || f.repo->agreement_id.empty())
387 return bad("a repo names its counterparty and agreement");
388 const auto basis = static_cast<std::uint8_t>(f.repo->day_count);
389 if (basis < 1 || basis > static_cast<std::uint8_t>(finance::day_count_basis::act_act_isda))
390 return bad("not a day-count basis");
391 if (f.repo->maturity && *f.repo->maturity < f.dates.trade_at.days())
392 return bad("a repo matures after it opens");
393 }
394 return ok();
395 }
