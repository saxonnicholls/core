110 // libstdc++'s uniform_int_distribution::operator() (bits/uniform_int_dist.h, GCC 11 and later), with
111 // __int128 available: the same types, the same branches, the same arithmetic.
112 template <>
113 struct uniform_int_algorithm<stats::draw::libstdcxx> {
114 template <typename W, typename Up, typename G>
115 static constexpr Up nearly_divisionless(G& g, Up range) noexcept(noexcept(g()))
116 {
117 W product = static_cast<W>(static_cast<W>(g()) * static_cast<W>(range));
118 auto low = static_cast<Up>(product);
119 if (low < range) {
120 const auto threshold = static_cast<Up>(static_cast<Up>(0 - range) % range);
121 while (low < threshold) {
122 product = static_cast<W>(static_cast<W>(g()) * static_cast<W>(range));
123 low = static_cast<Up>(product);
124 }
125 }
126 return static_cast<Up>(product >> std::numeric_limits<Up>::digits);
127 }

129 // The 64-bit case multiplies into 128 bits: core's umul128 is __int128's product, high and low.
130 template <typename G>
131 static constexpr std::uint64_t nearly_divisionless64(G& g, std::uint64_t range) noexcept(noexcept(g()))
132 {
133 auto product = snicholls::detail::umul128(static_cast<std::uint64_t>(g()), range);
134 if (product.low < range) {
135 const std::uint64_t threshold = (0 - range) % range;
136 while (product.low < threshold)
137 product = snicholls::detail::umul128(static_cast<std::uint64_t>(g()), range);
138 }
139 return product.high;
140 }

142 template <typename T, typename G>
143 static constexpr T run(G& g, T a, T b) noexcept(noexcept(g()))
144 {
145 using U = std::common_type_t<typename G::result_type, std::make_unsigned_t<T>>;
146 constexpr U gmin = static_cast<U>(G::min());
147 constexpr U grange = static_cast<U>(static_cast<U>(G::max()) - gmin);
148 const auto range = static_cast<U>(static_cast<U>(b) - static_cast<U>(a));
149 U ret{};
150 if (grange > range) {
151 const auto erange = static_cast<U>(range + 1);
152 if constexpr (grange == std::numeric_limits<std::uint64_t>::max())
153 ret = static_cast<U>(nearly_divisionless64(g, static_cast<std::uint64_t>(erange)));
154 else if constexpr (grange == std::numeric_limits<std::uint32_t>::max())
155 ret = static_cast<U>(nearly_divisionless<std::uint64_t>(g, static_cast<std::uint32_t>(erange)));
156 else {
157 const auto scaling = static_cast<U>(grange / erange);
158 const auto past = static_cast<U>(erange * scaling);
159 do
160 ret = static_cast<U>(static_cast<U>(g()) - gmin);
161 while (ret >= past);
162 ret = static_cast<U>(ret / scaling);
163 }
164 }
165 else if (grange < range) {
166 // Every value in [0, range] is (grange + 1) * high + low, one way: draw high, then low. (Only a
167 // generator narrower than U gets here; the guard keeps grange + 1 from being 0 in the others.)
168 if constexpr (grange < std::numeric_limits<U>::max()) {
169 constexpr auto egrange = static_cast<U>(grange + 1);
170 U tmp{};
171 do {
172 tmp = static_cast<U>(egrange * static_cast<U>(run<T>(g, T{0}, static_cast<T>(range / egrange))));
173 ret = static_cast<U>(tmp + static_cast<U>(static_cast<U>(g()) - gmin));
174 } while (ret > range || ret < tmp);
175 }
176 }
177 else
178 ret = static_cast<U>(static_cast<U>(g()) - gmin);
179 return static_cast<T>(static_cast<U>(ret + static_cast<U>(a)));
180 }
181 };

183 // libc++'s uniform_int_distribution::operator() and __independent_bits_engine (__random/), member for member.
184 template <>
185 struct uniform_int_algorithm<stats::draw::libcxx> {
186 template <typename E, typename Ut>
187 struct bits_engine {
188 using engine_type = typename E::result_type;
189 using working = std::conditional_t<sizeof(engine_type) <= sizeof(Ut), Ut, engine_type>;
190 static constexpr working rp = static_cast<working>(static_cast<working>(E::max() - E::min()) + working{1});
191 static constexpr std::size_t wdt = std::numeric_limits<working>::digits;
192 static constexpr std::size_t edt = std::numeric_limits<engine_type>::digits;
193 // __log2<working, rp>: floor(log2 rp), or every bit of `working` when rp wrapped to 0.
194 static constexpr std::size_t m = rp == 0 ? wdt : static_cast<std::size_t>(std::bit_width(rp) - 1);
195
196 E& e;
197 std::size_t w, w0 = 0, n = 0, n0 = 0;
198 working y0 = 0, y1 = 0;
199 engine_type mask0 = 0, mask1 = 0;
200
201 constexpr bits_engine(E& engine, std::size_t bits) noexcept : e(engine), w(bits)
202 {
203 n = w / m + (w % m != 0);
204 w0 = w / n;
205 if constexpr (rp == 0)
206 y0 = rp;
207 else if (w0 < wdt)
208 y0 = static_cast<working>((rp >> w0) << w0);
209 else
210 y0 = 0;
211 if (static_cast<working>(rp - y0) > static_cast<working>(y0 / n)) {
212 ++n;
213 w0 = w / n;
214 y0 = w0 < wdt ? static_cast<working>((rp >> w0) << w0) : working{0};
215 }
216 n0 = n - w % n;
217 y1 = w0 < wdt - 1 ? static_cast<working>((rp >> (w0 + 1)) << (w0 + 1)) : working{0};
218 mask0 = w0 > 0 ? static_cast<engine_type>(static_cast<engine_type>(~engine_type{0}) >> (edt - w0)) : engine_type{0};
219 mask1 = w0 < edt - 1 ? static_cast<engine_type>(static_cast<engine_type>(~engine_type{0}) >> (edt - (w0 + 1)))
220 : static_cast<engine_type>(~engine_type{0});
221 }
222
223 constexpr Ut operator()() noexcept(noexcept(e()))
224 {
225 if constexpr (rp == 0)
226 return static_cast<Ut>(e() & mask0);
227 else {
228 constexpr std::size_t wrt = std::numeric_limits<Ut>::digits;
229 Ut sp = 0;
230 for (std::size_t k = 0; k < n0; ++k) {
231 engine_type u{};
232 do
233 u = static_cast<engine_type>(e() - E::min());
234 while (u >= y0);
235 sp = w0 < wrt ? static_cast<Ut>(sp << w0) : Ut{0};
236 sp = static_cast<Ut>(sp + (u & mask0));
237 }
238 for (std::size_t k = n0; k < n; ++k) {
239 engine_type u{};
240 do
241 u = static_cast<engine_type>(e() - E::min());
242 while (u >= y1);
243 sp = w0 < wrt - 1 ? static_cast<Ut>(sp << (w0 + 1)) : Ut{0};
244 sp = static_cast<Ut>(sp + (u & mask1));
245 }
246 return sp;
247 }
248 }
249 };

251 template <typename T, typename G>
252 static constexpr T run(G& g, T a, T b) noexcept(noexcept(g()))
253 {
254 using Ut = std::conditional_t<sizeof(T) <= sizeof(std::uint32_t), std::uint32_t, std::make_unsigned_t<T>>;
255 const auto rp = static_cast<Ut>(static_cast<Ut>(static_cast<Ut>(b) - static_cast<Ut>(a)) + Ut{1});
256 if (rp == 1)
257 return a;
258 constexpr std::size_t dt = std::numeric_limits<Ut>::digits;
259 if (rp == 0)
260 return static_cast<T>(bits_engine<G, Ut>(g, dt)());
261 auto w = static_cast<std::size_t>(dt - static_cast<std::size_t>(std::countl_zero(rp)) - 1);
262 if ((rp & (std::numeric_limits<Ut>::max() >> (dt - w))) != 0)
263 ++w;
264 bits_engine<G, Ut> e(g, w);
265 Ut u{};
266 do
267 u = e();
268 while (u >= rp);
269 return static_cast<T>(static_cast<Ut>(u + static_cast<Ut>(a)));
270 }
271 };
