188 // A callable F reached through a set S, with S's constness: `F&` or `const F&`.
189 template <typename S, typename F>
190 using element_ref_t = std::conditional_t<std::is_const_v<S>, const F&, F&>;

192 // A non-const lvalue reference can bind neither a borrowed const object nor a value about to go.
193 template <typename A>
194 inline constexpr bool bindable_from_source_v =
195 !(std::is_lvalue_reference_v<A> && !std::is_const_v<std::remove_reference_t<A>>);

197 // What get<T>(i) hands back: contextually bool, and *h a T.
198 template <typename H, typename T>
199 concept holds = requires(H& h) {
200 static_cast<bool>(h);
201 { *h } -> std::convertible_to<const T&>;
202 };

204 // A borrowed argument (a pointer into the source) reaching a parameter of type A: by const
205 // reference when A binds one, as a copy when A is an rvalue reference.
206 template <typename A>
207 using borrowed_t =
208 std::conditional_t<std::is_rvalue_reference_v<A>, std::remove_cvref_t<A>, const std::remove_cvref_t<A>&>;
209
210 template <typename A, typename H>
211 requires std::is_pointer_v<H>
212 constexpr borrowed_t<A> take(H h)
213 {
214 return *h;
215 }
216
217 // An owned argument (a value the source produced) reaching a parameter of type A: moved unless A
218 // is a const lvalue reference.
219 template <typename A>
220 using owned_t =
221 std::conditional_t<std::is_lvalue_reference_v<A>, const std::remove_cvref_t<A>&, std::remove_cvref_t<A>&&>;
222
223 template <typename A, typename H>
224 requires(!std::is_pointer_v<H>)
225 constexpr owned_t<A> take(H& h)
226 {
227 return static_cast<owned_t<A>>(*h);
228 }

230 // Call f and keep its result in out as an R: R{} for void, otherwise the value converted.
231 template <typename R, typename F, typename... X>
232 requires std::is_void_v<std::invoke_result_t<F, X...>>
233 void call_into(R& out, F&& f, X&&... x)
234 {
235 std::forward<F>(f)(std::forward<X>(x)...);
236 out = R{};
237 }

239 template <typename R, typename F, typename... X>
240 requires(!std::is_void_v<std::invoke_result_t<F, X...>>)
241 void call_into(R& out, F&& f, X&&... x)
242 {
243 static_assert(std::is_constructible_v<R, std::invoke_result_t<F, X...>>,
244 "snicholls::invoke_with: a callable's result must convert to R (for std::any, a copyable "
245 "type); return void or a convertible type");
246 out = R(std::forward<F>(f)(std::forward<X>(x)...));
247 }

254 template <typename R, typename F, argument_source S, typename... A, std::size_t... I>
255 bool try_call(F&& f, [[maybe_unused]] const S& source, R& out, meta::type_list<A...>,
256 std::index_sequence<I...>)
257 {
258 static_assert((bindable_from_source_v<A> && ...),
259 "snicholls::invoke_with: a parameter cannot be a non-const lvalue reference, because a source "
260 "lends a const object or gives a value; take it by value, by const& or by &&");
261 static_assert(std::is_invocable_v<F, borrowed_t<A>...>,
262 "snicholls::invoke_with: the callable cannot be called through the set's constness (a "
263 "mutable lambda needs a non-const set)");
264 // One get per parameter: a refusal (null, or an empty optional) is a parameter the source cannot fill.
265 return [&](auto... held) {
266 static_assert((holds<decltype(held), std::remove_cvref_t<A>> && ...),
267 "snicholls::invoke_with: a source's get<T>(i) must return a const T* or a std::optional<T>");
268 if (!(held && ...))
269 return false;
270 call_into(out, std::forward<F>(f), take<A>(held)...);
271 return true;
272 }(source.template get<std::remove_cvref_t<A>>(I)...);
273 }
274#if defined(_MSC_VER) && !defined(__clang__)
275#pragma warning(pop)
276#endif

278 // One candidate: read its parameter list, check the arity, then the parameters.
279 template <typename F, typename R, typename Set, argument_source S>
280 bool try_one(Set& set, const S& source, R& out, bool& arity_seen)
281 {
282 static_assert(single_signature_callable<F>,
283 "snicholls::invoke_with: every callable needs exactly one non-template operator() or to be "
284 "a function pointer, so that its parameter types can be read; a generic lambda has none");
285 using params = typename callable_traits<F>::arguments;
286 if (source.size() != meta::size_v<params>)
287 return false;
288 arity_seen = true;
289 return try_call(static_cast<element_ref_t<Set, F>>(set), source, out, params{},
290 std::make_index_sequence<meta::size_v<params>>{});
291 }

293 // The candidates in order; the first that is called ends the fold.
294 template <typename R, typename Set, argument_source S, typename... Fs>
295 result<R, overload_mismatch> dispatch_with([[maybe_unused]] Set& set, [[maybe_unused]] const S& source,
296 meta::type_list<Fs...>)
297 {
298 static_assert(std::is_default_constructible_v<R>,
299 "snicholls::invoke_with: R must be default constructible; it is what a void callable returns");
300 R out{};
301 bool arity_seen = false;
302 if ((try_one<Fs>(set, source, out, arity_seen) || ...))
303 return result<R, overload_mismatch>(std::in_place, std::move(out));
304 return err(arity_seen ? overload_mismatch::types : overload_mismatch::arity);
305 }
