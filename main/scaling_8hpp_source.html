125 template <typename T>
126 constexpr T ruiz_limit(T v, const ruiz_options<T>& o) noexcept
127 {
128 return v < o.min_scaling ? T(1) : (v > o.max_scaling ? o.max_scaling : v);
129 }
130 // The power of two nearest s > 0 in the geometric sense: 2ᵏ with 2ᵏ ≤ s < 2ᵏ⁺¹, or 2ᵏ⁺¹ if s / 2ᵏ ≥ √2
131 template <typename T>
132 constexpr T ruiz_round(T s, const ruiz_options<T>& o) noexcept
133 {
134 if (!o.power_of_two)
135 return s;
136 T p(1);
137 if (s >= T(1))
138 while (p * T(2) <= s)
139 p *= T(2);
140 else
141 while (p > s)
142 p /= T(2);
143 return s / p < T(1.41421356237309504880L) ? p : p * T(2);
144 }
145 template <typename T>
146 constexpr T ruiz_factor(T norm, const ruiz_options<T>& o) noexcept
147 {
148 return ruiz_round(T(1) / static_cast<T>(math::sqrt(ruiz_limit(norm, o))), o);
149 }

167 template <linalg::scalar T, typename I>
168 [[nodiscard]] constexpr result<void, sparse::sparse_error> ruiz_equilibrate(const sparse::csc_view<T, I>& p, const sparse::csc_view<T, I>& a, std::type_identity_t<std::span<T>> q,
169 std::type_identity_t<std::span<T>> d, std::type_identity_t<std::span<T>> e, T& c,
170 std::size_t iterations, std::type_identity_t<std::span<T>> work,
171 const std::type_identity_t<ruiz_options<T>>& options = {}) noexcept
172 {
173 const std::size_t n = p.cols, m = a.rows;
174 if (p.rows != n || a.cols != n || q.size() != n || d.size() != n || e.size() != m)
175 return err(sparse::sparse_error::dimension_mismatch);
176 if (work.size() < ruiz_workspace(n, m))
177 return err(sparse::sparse_error::insufficient_capacity);
178 const auto dt = work.subspan(0, n), tmp = work.subspan(n, n), et = work.subspan(2 * n, m);
179 for (std::size_t j = 0; j < n; ++j)
180 d[j] = T(1);
181 for (std::size_t i = 0; i < m; ++i)
182 e[i] = T(1);
183 c = T(1);
184 for (std::size_t it = 0; it < iterations; ++it) {
185 sparse::symmetric_upper_inf_norms(p, dt);
186 sparse::column_inf_norms(a, tmp);
187 sparse::row_inf_norms(a, et);
188 for (std::size_t j = 0; j < n; ++j)
189 dt[j] = detail::ruiz_factor(dt[j] > tmp[j] ? dt[j] : tmp[j], options);
190 for (std::size_t i = 0; i < m; ++i)
191 et[i] = detail::ruiz_factor(et[i], options);
192 sparse::scale(p, std::span<const T>(dt), std::span<const T>(dt));
193 sparse::scale(a, std::span<const T>(et), std::span<const T>(dt));
194 for (std::size_t j = 0; j < n; ++j) {
195 q[j] *= dt[j];
196 d[j] *= dt[j];
197 }
198 for (std::size_t i = 0; i < m; ++i)
199 e[i] *= et[i];
200 if (options.scale_cost) {
201 sparse::symmetric_upper_inf_norms(p, tmp);
202 T mean(0), qn(0);
203 for (std::size_t j = 0; j < n; ++j) {
204 mean += tmp[j];
205 const T v = q[j] < T(0) ? -q[j] : q[j];
206 qn = v > qn ? v : qn;
207 }
208 mean = n > 0 ? mean / static_cast<T>(n) : T(0);
209 qn = detail::ruiz_limit(qn, options);
210 const T measure = detail::ruiz_limit(mean > qn ? mean : qn, options);
211 const T gamma = detail::ruiz_round(T(1) / measure, options);
212 for (std::size_t k = 0; k < p.nnz(); ++k)
213 p.values[k] *= gamma;
214 for (std::size_t j = 0; j < n; ++j)
215 q[j] *= gamma;
216 c *= gamma;
217 }
218 }
219 return ok();
220 }

233 template <linalg::scalar T, typename I>
234 [[nodiscard]] constexpr result<void, sparse::sparse_error> ruiz_equilibrate_rectangular(const sparse::csc_view<T, I>& a, std::type_identity_t<std::span<T>> r,
235 std::type_identity_t<std::span<T>> s, std::size_t iterations,
236 std::type_identity_t<std::span<T>> work, const std::type_identity_t<ruiz_options<T>>& options = {}) noexcept
237 {
238 const std::size_t m = a.rows, n = a.cols;
239 if (r.size() != m || s.size() != n)
240 return err(sparse::sparse_error::dimension_mismatch);
241 if (work.size() < ruiz_rectangular_workspace(m, n))
242 return err(sparse::sparse_error::insufficient_capacity);
243 const auto rt = work.subspan(0, m), st = work.subspan(m, n);
244 for (std::size_t i = 0; i < m; ++i)
245 r[i] = T(1);
246 for (std::size_t j = 0; j < n; ++j)
247 s[j] = T(1);
248 for (std::size_t it = 0; it < iterations; ++it) {
249 sparse::row_inf_norms(a, rt);
250 sparse::column_inf_norms(a, st);
251 for (std::size_t i = 0; i < m; ++i) {
252 rt[i] = detail::ruiz_factor(rt[i], options);
253 r[i] *= rt[i];
254 }
255 for (std::size_t j = 0; j < n; ++j) {
256 st[j] = detail::ruiz_factor(st[j], options);
257 s[j] *= st[j];
258 }
259 sparse::scale(a, std::span<const T>(rt), std::span<const T>(st));
260 }
261 return ok();
262 }

379 template <linalg::scalar T, typename I>
380 constexpr void unscale_data(const sparse::csc_view<T, I>& p, const sparse::csc_view<T, I>& a, std::type_identity_t<std::span<T>> q,
381 std::type_identity_t<std::span<const T>> d, std::type_identity_t<std::span<const T>> e, std::type_identity_t<T> c) noexcept
382 {
383 for (std::size_t j = 0; j < p.cols; ++j)
384 for (std::size_t k = static_cast<std::size_t>(p.col_ptr[j]); k < static_cast<std::size_t>(p.col_ptr[j + 1]); ++k)
385 p.values[k] = p.values[k] / d[static_cast<std::size_t>(p.row_idx[k])] / d[j] / c;
386 for (std::size_t j = 0; j < a.cols; ++j)
387 for (std::size_t k = static_cast<std::size_t>(a.col_ptr[j]); k < static_cast<std::size_t>(a.col_ptr[j + 1]); ++k)
388 a.values[k] = a.values[k] / e[static_cast<std::size_t>(a.row_idx[k])] / d[j];
389 for (std::size_t j = 0; j < q.size(); ++j)
390 q[j] = q[j] / d[j] / c;
391 }
