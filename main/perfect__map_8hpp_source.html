81 template <typename Key, typename Value, std::size_t N>
82 class perfect_map {
83 static constexpr std::size_t buckets = std::max<std::size_t>(1, std::bit_ceil(N) / 2);
84 static constexpr std::size_t slots = std::bit_ceil(N + N / 2 + 1);
85
86 public:
90 consteval explicit perfect_map(const std::array<std::pair<Key, Value>, N>& items)
91 {
92 std::array<std::size_t, N> order{}, home{}; // items grouped by bucket, the fullest first; each key hashed once
93 std::array<std::size_t, buckets> fill{};
94 for (std::size_t i = 0; i < N; ++i) {
95 order[i] = i;
96 home[i] = bucket(items[i].first);
97 ++fill[home[i]];
98 }
99 std::sort(order.begin(), order.end(), [&](std::size_t a, std::size_t b) {
100 return fill[home[a]] != fill[home[b]] ? fill[home[a]] > fill[home[b]] : home[a] < home[b];
101 });
102 std::array<std::size_t, N> at{}; // the trial slots of the bucket being placed
103 for (std::size_t start = 0; start < N;) {
104 const std::size_t b = home[order[start]], end = start + fill[b];
105 for (std::size_t i = start; i < end; ++i) // equal keys always share a bucket
106 for (std::size_t j = start; j < i; ++j)
107 if (items[order[i]].first == items[order[j]].first)
108 throw std::invalid_argument("snicholls::containers::perfect_map: a key appears twice");
109 for (std::uint32_t d = 1;; ++d) { // the first displacement that puts the whole bucket in free slots
110 bool fits = true;
111 for (std::size_t i = start; i < end && fits; ++i) {
112 at[i] = slot(items[order[i]].first, d);
113 fits = !used_[at[i]];
114 for (std::size_t j = start; j < i && fits; ++j)
115 fits = at[j] != at[i];
116 }
117 if (!fits)
118 continue;
119 displacement_[b] = d;
120 for (std::size_t i = start; i < end; ++i) {
121 used_[at[i]] = true;
122 table_[at[i]] = items[order[i]];
123 }
124 break;
125 }
126 start = end;
127 }
128 }

134 template <typename K>
135 [[nodiscard]] constexpr const Value* find(const K& key) const noexcept
136 {
137 const std::size_t s = slot(key, displacement_[bucket(key)]);
138 return used_[s] && table_[s].first == key ? &table_[s].second : nullptr;
139 }

145 template <typename K>
146 [[nodiscard]] constexpr bool contains(const K& key) const noexcept
147 {
148 return find(key) != nullptr;
149 }

153 [[nodiscard]] static constexpr std::size_t size() noexcept { return N; }

156 template <typename K>
157 static constexpr std::size_t bucket(const K& k) noexcept
158 {
159 return static_cast<std::size_t>(detail::perfect_hash(k, 0) & (buckets - 1));
160 }
161 template <typename K>
162 static constexpr std::size_t slot(const K& k, std::uint32_t d) noexcept
163 {
164 return static_cast<std::size_t>(detail::perfect_hash(k, d) & (slots - 1));
165 }
166
167 std::array<std::uint32_t, buckets> displacement_{};
168 std::array<std::pair<Key, Value>, slots> table_{};

170 };
