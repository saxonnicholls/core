138template <transformer_config C, row X, mutable_row Out>
139 requires std::same_as<row_value_t<X>, row_value_t<Out>>
140constexpr void block_norm(const X& x, std::span<const row_value_t<X>> weight, std::span<const row_value_t<X>> bias, Out&& out)
141{
142 using T = row_value_t<X>;
143 if constexpr (C.norm == norm_kind::layer)
144 layer_norm(x, weight, bias, out, norm_eps_v<C, T>);
145 else
146 rms_norm(x, weight, out, norm_eps_v<C, T>);
147}

157template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
158 requires std::same_as<row_value_t<X>, T>
159constexpr void block_forward(const block_weights<W, T>& w, X&& x, std::size_t pos, const S& cache, std::size_t layer,
160 const R& rope_src)
161{
162 const auto s = as_mutable_row(x);
163 assert(s.size() == C.d_model);
164 std::array<T, C.d_model> h{}, a{};
165 block_norm<C>(s, w.norm1_weight, w.norm1_bias, h);
166 attention_forward<C>(w.attention, h, pos, cache, layer, rope_src, a);
167 for (std::size_t i = 0; i < C.d_model; ++i)
168 s[i] += a[i];
169 block_norm<C>(s, w.norm2_weight, w.norm2_bias, h);
170 mlp_forward<C>(w.mlp, h, a);
171 for (std::size_t i = 0; i < C.d_model; ++i)
172 s[i] += a[i];
173}

183template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, kv_store<T> S, typename R>
184 requires std::same_as<row_value_t<X>, T>
185constexpr void block_prefill(const block_weights<W, T>& w, X&& xs, std::size_t first, const S& cache, std::size_t layer,
186 const R& rope_src)
187{
188 const auto s = as_mutable_row(xs);
189 assert(s.size() % C.d_model == 0);
190 for (std::size_t t = 0; t < s.size() / C.d_model; ++t)
191 block_forward<C>(w, s.subspan(t * C.d_model, C.d_model), first + t, cache, layer, rope_src);
192}

228 template <transformer_config C, typename T>
229 constexpr void norm_backward(std::span<const T> dh, std::span<const T> x, std::span<const T> weight,
230 std::span<const T> bias, std::span<T> dx, std::span<T> dweight, std::span<T> dbias)
231 {
232 std::array<T, C.d_model> scratch{};
233 if constexpr (C.norm == norm_kind::layer) {
234 const auto stats = layer_norm(x, weight, bias, scratch, norm_eps_v<C, T>);
235 layer_norm_backward(dh, x, weight, stats, dx, dweight, dbias);
236 } else {
237 const T rstd = rms_norm(x, weight, scratch, norm_eps_v<C, T>);
238 rms_norm_backward(dh, x, weight, rstd, dx, dweight);
239 }
240 }

250template <transformer_config C, weight_matrix W, math::math_number T, mutable_row X, typename R>
251 requires std::same_as<row_value_t<X>, T> && (C.rope == rope_kind::none || rope_source<R>)
252constexpr void block_forward_sequence(const block_weights<W, T>& w, X&& xs, const R& rope_src, const block_activations<T>& act)
253{
254 constexpr std::size_t d = C.d_model;
255 const auto s = as_mutable_row(xs);
256 const std::size_t n = s.size() / d;
257 assert(s.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d
258 && act.h2.size() == n * d && act.mlp_up.size() == n * C.d_ff
259 && (!gated(C.mlp) || act.mlp_gate.size() == n * C.d_ff));
260 std::copy(s.begin(), s.end(), act.x0.begin());
261 for (std::size_t t = 0; t < n; ++t)
262 block_norm<C>(s.subspan(t * d, d), w.norm1_weight, w.norm1_bias, act.h1.subspan(t * d, d));
263 attention_forward_sequence<C>(w.attention, std::span<const T>(act.h1), rope_src, act.attention, s);
264 for (std::size_t i = 0; i < n * d; ++i)
265 s[i] = act.x0[i] + s[i];
266 std::copy(s.begin(), s.end(), act.x1.begin());
267 std::array<T, d> a{};
268 for (std::size_t t = 0; t < n; ++t) {
269 block_norm<C>(s.subspan(t * d, d), w.norm2_weight, w.norm2_bias, act.h2.subspan(t * d, d));
270 const mlp_activations<T> save{act.mlp_up.subspan(t * C.d_ff, C.d_ff),
271 act.mlp_gate.empty() ? std::span<T>{} : act.mlp_gate.subspan(t * C.d_ff, C.d_ff)};
272 mlp_forward<C>(w.mlp, std::span<const T>(act.h2).subspan(t * d, d), a, save);
273 for (std::size_t i = 0; i < d; ++i)
274 s[t * d + i] += a[i];
275 }
276}

289template <transformer_config C, typename S, math::math_number T, mutable_row DX, typename R>
290 requires std::same_as<row_value_t<DX>, T> && (C.rope == rope_kind::none || rope_source<R>)
291constexpr void block_backward(const block_weights<dense_weights<S>, T>& w, const R& rope_src,
292 const block_activations<T>& act, DX&& dxs, const block_gradients<T>& grads)
293{
294 constexpr std::size_t d = C.d_model;
295 const auto g = as_mutable_row(dxs);
296 const std::size_t n = g.size() / d;
297 assert(g.size() == n * d && act.x0.size() == n * d && grads.scratch.size() == n * d);
298 std::array<T, d> dh{}, dn{};
299 for (std::size_t t = 0; t < n; ++t) {
300 const auto gt = g.subspan(t * d, d);
301 dh.fill(T(0));
302 const mlp_activations<T> saved{act.mlp_up.subspan(t * C.d_ff, C.d_ff),
303 act.mlp_gate.empty() ? std::span<T>{} : act.mlp_gate.subspan(t * C.d_ff, C.d_ff)};
304 mlp_backward<C>(w.mlp, std::span<const T>(act.h2).subspan(t * d, d), saved, std::span<const T>(gt), dh, grads.mlp);
305 detail::block::norm_backward<C, T>(dh, std::span<const T>(act.x1).subspan(t * d, d), w.norm2_weight, w.norm2_bias, dn,
306 grads.norm2_weight, grads.norm2_bias);
307 for (std::size_t i = 0; i < d; ++i)
308 gt[i] += dn[i];
309 }
310 std::fill(grads.scratch.begin(), grads.scratch.end(), T(0));
311 attention_backward<C>(w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(g),
312 grads.scratch, grads.attention);
313 for (std::size_t t = 0; t < n; ++t) {
314 detail::block::norm_backward<C, T>(std::span<const T>(grads.scratch).subspan(t * d, d),
315 std::span<const T>(act.x0).subspan(t * d, d), w.norm1_weight, w.norm1_bias, dn,
316 grads.norm1_weight, grads.norm1_bias);
317 for (std::size_t i = 0; i < d; ++i)
318 g[t * d + i] += dn[i];
319 }
320}

371template <transformer_config C, typename B, math::math_number T, typename R>
372 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
373constexpr void block_forward_sequence(const B& backend, const block_weights<dense_weights<T>, T>& w, std::type_identity_t<std::span<T>> xs,
374 const R& rope_src, const block_activations<T>& act, std::size_t sequences = 1)
375{
376 constexpr std::size_t d = C.d_model;
377 [[maybe_unused]] const std::size_t n = xs.size() / d;
378 assert(xs.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d && act.h2.size() == n * d);
379 copy_rows(backend, std::span<const T>(xs), act.x0);
380 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm1_weight, w.norm1_bias, act.h1);
381 attention_forward_sequence<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, xs, sequences);
382 add_rows(backend, std::span<const T>(act.x0), xs);
383 copy_rows(backend, std::span<const T>(xs), act.x1);
384 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm2_weight, w.norm2_bias, act.h2);
385 mlp_forward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), xs, mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden});
386 add_rows(backend, std::span<const T>(act.x1), xs);
387}

403template <transformer_config C, typename B, math::math_number T, typename R>
404 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
405constexpr void block_backward(const B& backend, const block_weights<dense_weights<T>, T>& w, const R& rope_src,
406 const block_activations<T>& act, std::type_identity_t<std::span<T>> dxs, const block_gradients<T>& grads,
407 std::size_t sequences = 1)
408{
409 constexpr std::size_t d = C.d_model;
410 [[maybe_unused]] const std::size_t n = dxs.size() / d;
411 assert(dxs.size() == n * d && act.x0.size() == n * d && grads.scratch.size() == n * d);
412 fill_rows(backend, grads.scratch, T(0));
413 mlp_backward_sequence<C>(backend, w.mlp, std::span<const T>(act.h2), mlp_activations<T>{act.mlp_up, act.mlp_gate, act.mlp_hidden},
414 std::span<const T>(dxs), grads.scratch, grads.mlp);
415 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x1), w.norm2_weight, w.norm2_bias,
416 dxs, grads.norm2_weight, grads.norm2_bias);
417 fill_rows(backend, grads.scratch, T(0));
418 attention_backward<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(dxs),
419 grads.scratch, grads.attention, sequences);
420 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x0), w.norm1_weight, w.norm1_bias,
421 dxs, grads.norm1_weight, grads.norm1_bias);
422}
