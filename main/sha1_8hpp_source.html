8// SHA-1 (FIPS 180-4), constexpr, for name-based (version 5) UUIDs only.
9//
10// snicholls::detail::sha1 h; h.update(bytes); h.update(more); std::array<std::uint8_t, 20> d = h.finish();
11// constexpr auto d = snicholls::detail::sha1_digest("abc");
12//
13// SHA-1 is broken for collision resistance: never use it for signatures, integrity or passwords. RFC 9562 fixes it
14// for UUIDv5, where it only spreads a name into 122 bits. Not public API.
15//
16// ts-moveables has a private, run-time-only SHA-1 in http/websocket.hpp; this one is constexpr.

29class sha1

31public:
32 using digest_type = std::array<std::uint8_t, 20>;
33
34 constexpr sha1() noexcept = default;
35
36 constexpr sha1& update(std::span<const std::uint8_t> data) noexcept
37 {
38 for (const std::uint8_t b : data)
39 push(b);
40 return *this;
41 }
42
43 constexpr sha1& update(std::span<const std::byte> data) noexcept
44 {
45 for (const std::byte b : data)
46 push(static_cast<std::uint8_t>(b));
47 return *this;
48 }
49
50 constexpr sha1& update(std::string_view text) noexcept
51 {
52 for (const char c : text)
53 push(static_cast<std::uint8_t>(c));
54 return *this;
55 }
56
57 // Pads, processes the last block(s) and returns the digest. The object must not be used afterwards.
58 constexpr digest_type finish() noexcept
59 {
60 const std::uint64_t bit_length = total_ * 8U;
61 push(0x80);
62 while (used_ != 56)
63 push(0x00);
64 for (int shift = 56; shift >= 0; shift -= 8)
65 push(static_cast<std::uint8_t>(bit_length >> shift));
66 digest_type out{};
67 for (std::size_t i = 0; i < 5; ++i) {
68 out[4 * i + 0] = static_cast<std::uint8_t>(h_[i] >> 24);
69 out[4 * i + 1] = static_cast<std::uint8_t>(h_[i] >> 16);
70 out[4 * i + 2] = static_cast<std::uint8_t>(h_[i] >> 8);
71 out[4 * i + 3] = static_cast<std::uint8_t>(h_[i]);
72 }
73 return out;
74 }
75
76private:
77 static constexpr std::uint32_t rotl(std::uint32_t x, int n) noexcept { return (x << n) | (x >> (32 - n)); }
78
79 constexpr void push(std::uint8_t b) noexcept
80 {
81 block_[used_++] = b;
82 ++total_;
83 if (used_ == 64) {
84 compress();
85 used_ = 0;
86 }
87 }
88
89 constexpr void compress() noexcept
90 {
91 std::uint32_t w[80]{};
92 for (std::size_t i = 0; i < 16; ++i)
93 w[i] = (std::uint32_t{block_[4 * i]} << 24) | (std::uint32_t{block_[4 * i + 1]} << 16) |
94 (std::uint32_t{block_[4 * i + 2]} << 8) | std::uint32_t{block_[4 * i + 3]};
95 for (std::size_t i = 16; i < 80; ++i)
96 w[i] = rotl(w[i - 3] ^ w[i - 8] ^ w[i - 14] ^ w[i - 16], 1);
97
98 std::uint32_t a = h_[0], b = h_[1], c = h_[2], d = h_[3], e = h_[4];
99 for (std::size_t i = 0; i < 80; ++i) {
100 std::uint32_t f = 0, k = 0;
101 if (i < 20) {
102 f = (b & c) | (~b & d);
103 k = 0x5A827999U;
104 } else if (i < 40) {
105 f = b ^ c ^ d;
106 k = 0x6ED9EBA1U;
107 } else if (i < 60) {
108 f = (b & c) | (b & d) | (c & d);
109 k = 0x8F1BBCDCU;
110 } else {
111 f = b ^ c ^ d;
112 k = 0xCA62C1D6U;
113 }
114 const std::uint32_t t = rotl(a, 5) + f + e + k + w[i];
115 e = d;
116 d = c;
117 c = rotl(b, 30);
118 b = a;
119 a = t;
120 }
121 h_[0] += a;
122 h_[1] += b;
123 h_[2] += c;
124 h_[3] += d;
125 h_[4] += e;
126 }
127
128 std::uint32_t h_[5]{0x67452301U, 0xEFCDAB89U, 0x98BADCFEU, 0x10325476U, 0xC3D2E1F0U};
129 std::uint8_t block_[64]{};
130 std::size_t used_ = 0;
131 std::uint64_t total_ = 0;
132};
133
134constexpr sha1::digest_type sha1_digest(std::string_view text) noexcept
135{
136 sha1 h;
137 h.update(text);
138 return h.finish();
139}
