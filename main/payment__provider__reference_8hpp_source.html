114 capability_result<collection> collect(const collection_request& c)
115 {
116 if (auto bad = check_request(c.idempotency_key, c.our_reference, c.amount))
117 return err(*bad);
118 if (const auto it = collect_keys_.find(c.idempotency_key); it != collect_keys_.end()) {
119 if (it->second.first != fingerprint(c))
120 return fail(capability_errc::conflict, {}, "idempotency key reused for another request");
121 return ok(it->second.second);
122 }
123 if (c.method != payment_method::card && c.method != payment_method::bank_transfer)
124 return fail(capability_errc::unsupported, {}, "card and bank transfer only");
125 if (auto e = take_injected())
126 return err(*e);
127 collection out;
128 out.pay = new_payment("mem_pay_", movement::collection, c.our_reference, c.method, c.amount);
129 if (c.method == payment_method::card) {
130 out.pay.status = payment_status::requires_action;
131 out.next.kind = next_step_kind::redirect;
132 out.next.url = "memory://checkout/" + std::string(out.pay.payment_id.view());
133 } else {
134 auto reference = c.payer_reference.empty() ? make_id<short_text>("MEM-" + std::to_string(++next_reference_))
135 : make_id<short_text>(c.payer_reference);
136 if (!reference)
137 return err(reference.unwrap_err());
138 out.next.kind = next_step_kind::bank_transfer;
139 out.next.bank = bank_details{"Memory Payments", "Memory Bank", "000000001", "000-000", "MEMOXXXX", {}, *reference};
140 awaiting_credit_.emplace(std::string(reference->view()), std::string(out.pay.payment_id.view()));
141 }
142 payments_.insert_or_assign(std::string(out.pay.payment_id.view()), out.pay);
143 collect_keys_.emplace(std::string(c.idempotency_key), std::make_pair(fingerprint(c), out));
144 return ok(std::move(out));
145 }

150 capability_result<payment> payout(const payout_request& o)
151 {
152 if (auto bad = check_request(o.idempotency_key, o.our_reference, o.amount))
153 return err(*bad);
154 if (const auto it = payout_keys_.find(o.idempotency_key); it != payout_keys_.end()) {
155 if (it->second.first != fingerprint(o))
156 return fail(capability_errc::conflict, {}, "idempotency key reused for another request");
157 return ok(it->second.second);
158 }
159 if (auto e = take_injected())
160 return err(*e);
161 if (!beneficiaries_.contains(o.beneficiary_id))
162 return fail(capability_errc::not_found, {}, "no such beneficiary");
163 if (o.quote_id) {
164 const auto q = quotes_.find(*o.quote_id);
165 if (q == quotes_.end())
166 return fail(capability_errc::not_found, {}, "no such quote");
167 if (clock_() > q->second.expires_at)
168 return fail(capability_errc::stale, {}, "the quote expired");
169 }
170 if (auto taken = take(o.amount); !taken)
171 return err(taken.unwrap_err());
172 payment p = new_payment("mem_out_", movement::payout, o.our_reference, payment_method::bank_transfer, o.amount);
173 p.status = payment_status::succeeded;
174 p.received = o.amount;
175 payments_.insert_or_assign(std::string(p.payment_id.view()), p);
176 record(std::string(p.payment_id.view()), fact_kind::payout, -o.amount, money::zero(o.amount.asset_of()),
177 fact_refs{{}, {}, {}, {}, p.payment_id});
178 payout_keys_.emplace(std::string(o.idempotency_key), std::make_pair(fingerprint(o), p));
179 return ok(std::move(p));
180 }

185 capability_result<payment> refund(const refund_request& r)
186 {
187 if (r.idempotency_key.empty() || r.idempotency_key.size() > client_id::capacity())
188 return fail(capability_errc::invalid_argument, {}, "an idempotency key of 1 to 64 characters");
189 if (const auto it = refund_keys_.find(r.idempotency_key); it != refund_keys_.end()) {
190 if (it->second.first != fingerprint(r))
191 return fail(capability_errc::conflict, {}, "idempotency key reused for another request");
192 return ok(it->second.second);
193 }
194 if (auto e = take_injected())
195 return err(*e);
196 const auto found = payments_.find(r.payment_id);
197 if (found == payments_.end() || found->second.kind != movement::collection)
198 return fail(capability_errc::not_found);
199 payment& original = found->second;
200 if (original.status != payment_status::succeeded && original.status != payment_status::partially_refunded)
201 return fail(capability_errc::conflict, {}, "only a succeeded collection is refunded");
202 const auto remaining = subtract(original.received, original.refunded);
203 if (!remaining)
204 return fail(capability_errc::internal);
205 const money amount = r.amount ? *r.amount : *remaining;
206 if (amount.asset_of() != original.received.asset_of())
207 return fail(capability_errc::invalid_argument, {}, "a refund is in the collection's currency");
208 if (amount.is_negative() || amount.is_zero())
209 return fail(capability_errc::invalid_argument, {}, "an amount is positive");
210 if (amount.magnitude() > remaining->magnitude())
211 return fail(capability_errc::conflict, {}, "more than remains to refund");
212 if (auto taken = take(amount); !taken)
213 return err(taken.unwrap_err());
214 original.refunded = *add(original.refunded, amount);
215 original.status = original.refunded == original.received ? payment_status::refunded : payment_status::partially_refunded;
216 original.updated_at = clock_();
217 payment p = new_payment("mem_ref_", movement::refund, original.our_reference.view(), original.method, amount);
218 p.status = payment_status::succeeded;
219 p.received = amount;
220 payments_.insert_or_assign(std::string(p.payment_id.view()), p);
221 record(std::string(p.payment_id.view()), fact_kind::refund, -amount, money::zero(amount.asset_of()),
222 fact_refs{original.payment_id, {}, p.payment_id, {}, {}});
223 refund_keys_.emplace(std::string(r.idempotency_key), std::make_pair(fingerprint(r), p));
224 return ok(std::move(p));
225 }

230 capability_result<fx_quote> quote(const quote_request& q)
231 {
232 if (q.amount.is_negative() || q.amount.is_zero())
233 return fail(capability_errc::invalid_argument, {}, "an amount is positive");
234 const asset source_unit = q.amount_is_source ? q.amount.asset_of() : q.other;
235 const asset target_unit = q.amount_is_source ? q.other : q.amount.asset_of();
236 const auto r = rates_.find(key(source_unit, target_unit));
237 if (r == rates_.end())
238 return fail(capability_errc::unsupported, {}, "no rate for that pair");
239 money source, target, fee;
240 if (q.amount_is_source) {
241 source = q.amount;
242 fee = fee_on(q.amount);
243 const auto net = subtract(q.amount, fee);
244 const auto converted = net ? convert(*net, r->second, rounding::toward_zero)
245 : result<conversion, money_errc>(err(money_errc::overflow));
246 if (!converted)
247 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
248 target = converted->converted;
249 } else {
250 target = q.amount;
251 const auto back = r->second.inverse();
252 const auto net = back ? convert(q.amount, *back, rounding::ceiling)
253 : result<conversion, money_errc>(err(money_errc::overflow));
254 if (!net)
255 return fail(capability_errc::invalid_argument, {}, "the amount cannot be converted");
256 fee = fee_on(net->converted);
257 source = *add(net->converted, fee);
258 }
259 const fx_quote out{id<provider_id>("mem_q_" + std::to_string(++next_id_)), source, target, fee, r->second,
260 clock_() + quote_lifetime};
261 quotes_.insert_or_assign(std::string(out.quote_id.view()), out);
262 return ok(out);
263 }

301 [[nodiscard]] capability_result<webhook_event> verify_webhook(const webhook_delivery& w) const
302 {
303 const auto sig = find_header(w.headers, signature_header);
304 if (!sig)
305 return fail(capability_errc::signature_invalid, {}, "no signature header");
306 std::string_view t;
307 std::vector<std::string_view> tags;
308 for (std::string_view rest = *sig; !rest.empty();) {
309 const auto comma = rest.find(',');
310 const std::string_view part = rest.substr(0, comma);
311 rest = comma == std::string_view::npos ? std::string_view{} : rest.substr(comma + 1);
312 if (part.starts_with("t="))
313 t = part.substr(2);
314 else if (part.starts_with("v1="))
315 tags.push_back(part.substr(3));
316 }
317 std::int64_t seconds = 0;
318 if (t.empty() || tags.empty() || !parse_int(t, seconds))
319 return fail(capability_errc::signature_invalid, {}, "a malformed signature header");
320 const auto expected = tag(t, w.body);
321 bool matched = false;
322 for (const std::string_view candidate : tags)
323 matched = constant_time_equal(expected.view(), candidate) || matched;
324 if (!matched)
325 return fail(capability_errc::signature_invalid, {}, "no signature matches");
326 const auto sent = utc_s::from_unix_seconds(seconds);
327 if (w.received_at - sent > webhook_tolerance || sent - w.received_at > webhook_tolerance)
328 return fail(capability_errc::stale, {}, "outside the 300-second window");
329 return parse_event(w.body);
330 }

354 capability_result<statement_page> statement(const statement_query& q)
355 {
356 std::size_t from = 0;
357 if (!q.cursor.empty()) {
358 std::int64_t at = 0;
359 if (!parse_int(q.cursor, at) || at < 0)
360 return fail(capability_errc::invalid_argument, {}, "not a cursor this provider gave");
361 from = static_cast<std::size_t>(at);
362 }
363 statement_page page;
364 std::size_t i = from;
365 for (; i < ledger_.size() && page.facts.size() < q.limit; ++i) {
366 const payment_fact& f = ledger_[i].second;
367 if (f.settled.gross.asset_of() == q.currency && (q.payout_id.empty() || f.refs.payout == q.payout_id))
368 page.facts.push_back(f);
369 }
370 if (i < ledger_.size())
371 page.next_cursor = id<provider_id>(std::to_string(i));
372 return ok(std::move(page));
373 }

411 {
412 if (net.is_negative() || net.is_zero())
413 return fail(capability_errc::invalid_argument, {}, "a credit is positive");
414 auto ref = make_id<short_text>(reference);
415 auto from = make_id<short_text>(sender);

427 p.received = net;

431 }
432 }
433 record(related, fact_kind::bank_credit, net, money::zero(net.asset_of()),
434 fact_refs{id<provider_id>(related), {}, {}, {}, {}}, std::string(c.credit_id.view()));
435 return ok(c);

489 static void field(std::string& out, std::string_view v)
490 {
491 out += std::to_string(v.size());
492 out += ':';
493 out += v;
494 out += '\x1f';

497 {
498 field(out, to_string(m));
499 field(out, m.asset_of().code());
500 field(out, std::to_string(m.asset_of().scale()));

502 static std::string fingerprint(const collection_request& c)
503 {
504 std::string out;
505 for (const std::string_view v : {c.idempotency_key, c.our_reference, c.payer_reference, c.return_url, c.cancel_url})
506 field(out, v);
507 field(out, c.amount);
508 field(out, std::to_string(static_cast<int>(c.method)));
509 for (const auto& [k, v] : c.metadata) {
510 field(out, k);
511 field(out, v);
512 }
513 return out;
514 }
515 static std::string fingerprint(const payout_request& o)
516 {
517 std::string out;
518 for (const std::string_view v : {o.idempotency_key, o.our_reference, o.beneficiary_id, o.reference})
519 field(out, v);
520 field(out, o.amount);
521 field(out, o.quote_id ? "1" : "0");
522 field(out, o.quote_id.value_or(std::string_view{}));
523 return out;
524 }
525 static std::string fingerprint(const refund_request& r)
526 {
527 std::string out;
528 for (const std::string_view v : {r.idempotency_key, r.payment_id, r.reason})
529 field(out, v);
530 if (r.amount)
531 field(out, *r.amount);
532 else
533 field(out, "-");
534 return out;
535 }

543 std::optional<capability_error> check_request(std::string_view idempotency_key, std::string_view our_reference,
544 const money& amount) const
545 {
546 if (idempotency_key.empty() || idempotency_key.size() > client_id::capacity())
547 return capability_error{capability_errc::invalid_argument, {}, "an idempotency key of 1 to 64 characters"};
548 if (our_reference.size() > client_id::capacity())
549 return capability_error{capability_errc::invalid_argument, {}, "our reference is at most 64 characters"};
550 if (amount.is_negative() || amount.is_zero())
551 return capability_error{capability_errc::invalid_argument, {}, "an amount is positive"};
552 if (std::find(currencies_.begin(), currencies_.end(), amount.asset_of()) == currencies_.end())
553 return capability_error{capability_errc::unsupported, {}, "the currency is not held"};
554 return std::nullopt;
555 }

589 // One balance transaction: gross, the fee, and net = gross - fee, so balances(settled) holds by construction.
590 void record(std::string related, fact_kind kind, const money& gross, const money& fee, const fact_refs& refs,
591 const std::string& line_id = {})
592 {
593 payment_fact f;
594 f.source = "memory";
595 f.provider_account = provider_account_;
596 f.fact_id = id<fact_id>(line_id.empty() ? "mem_txn_" + std::to_string(++next_id_) : line_id);
597 f.kind = kind;
598 f.occurred_at = clock_();
599 f.settled = settlement{gross, fee, *subtract(gross, fee)};
600 f.refs = refs;
601 ledger_.emplace_back(std::move(related), f);

613 capability_result<void> take(const money& m)
614 {
615 const auto it = balances_.find(m.asset_of().code());
616 if (it == balances_.end() || it->second.asset_of() != m.asset_of() || it->second.magnitude() < m.magnitude())

660 static capability_result<webhook_event> parse_event(std::string_view body)
661 {
662 webhook_event e;
663 bool have_id = false, have_kind = false, fits = true;
664 const auto set = [&fits](auto& field, std::string_view v) {
665 auto r = std::remove_reference_t<decltype(field)>::make(v);
666 if (r)
667 field = *r;
668 else
669 fits = false;
670 };
671 for (std::string_view rest = body; !rest.empty();) {
672 const auto nl = rest.find('\n');
673 const std::string_view line = rest.substr(0, nl);
674 rest = nl == std::string_view::npos ? std::string_view{} : rest.substr(nl + 1);
675 const auto eq = line.find('=');
676 if (eq == std::string_view::npos)
677 return fail(capability_errc::invalid_argument, {}, "a webhook line without =");
678 const std::string_view k = line.substr(0, eq), v = line.substr(eq + 1);
679 if (k == "id") {
680 set(e.event_id, v);
681 have_id = !v.empty();
682 } else if (k == "kind") {
683 const auto kind = enum_cast<event_kind>(v);
684 if (!kind)
685 return fail(capability_errc::invalid_argument, {}, "an unknown event kind");
686 e.kind = *kind;
687 have_kind = true;
688 } else if (k == "occurred_at") {
689 std::int64_t s = 0;
690 if (!parse_int(v, s))
691 return fail(capability_errc::invalid_argument, {}, "occurred_at is not unix seconds");
692 e.occurred_at = utc_s::from_unix_seconds(s);
693 } else if (k == "needs_lookup") {
694 e.needs_lookup = v == "1";
695 } else if (k == "source") {
696 set(e.source, v);
697 } else if (k == "provider_account") {
698 set(e.provider_account, v);
699 } else if (k == "fact_id") {
700 set(e.fact_id, v);
701 } else if (k == "type") {
702 set(e.type, v);
703 } else if (k == "payment_id") {
704 set(e.payment_id, v);
705 } else if (k == "our_reference") {
706 set(e.our_reference, v);
707 } else if (k == "reference") {
708 set(e.reference, v);
709 } else {
710 return fail(capability_errc::invalid_argument, {}, "an unknown webhook field");
711 }
712 }
713 if (!have_id || !have_kind || !fits)
714 return fail(capability_errc::invalid_argument, {}, "a webhook without id or kind, or a field too long");
715 return ok(e);
716 }
