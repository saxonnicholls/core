86constexpr std::size_t convolve_size(std::size_t na, std::size_t nb, convolve_mode mode) noexcept
87{
88 if (na == 0 || nb == 0)
89 return 0;
90 switch (mode) {
91 case convolve_mode::full: return na + nb - 1;
92 case convolve_mode::same: return na;
93 case convolve_mode::valid: break;
94 }
95 return (na > nb ? na - nb : nb - na) + 1;
96}

100 // The index of the full output at which the mode's output starts.
101 constexpr std::size_t start(std::size_t na, std::size_t nb, convolve_mode mode) noexcept
102 {
103 switch (mode) {
104 case convolve_mode::full: return 0;
105 case convolve_mode::same: return (nb - 1) / 2;
106 case convolve_mode::valid: break;
107 }
108 return (na < nb ? na : nb) - 1;
109 }

111 // Direct: full-output index k = start + i of sum_j a_j b_(k-j), b read through get_b.
112 template <typename T, typename B>
113 constexpr void direct(std::span<const T> a, std::size_t nb, B get_b, std::span<T> out, convolve_mode mode) noexcept
114 {
115 const std::size_t na = a.size();
116 assert(out.size() == convolve_size(na, nb, mode));
117 if (out.empty())
118 return;
119 const std::size_t s = start(na, nb, mode);
120 for (std::size_t i = 0; i < out.size(); ++i) {
121 const std::size_t k = s + i;
122 const std::size_t lo = k + 1 > nb ? k + 1 - nb : 0, hi = k < na - 1 ? k : na - 1;
123 T acc = T(0);
124 for (std::size_t j = lo; j <= hi; ++j)
125 acc += a[j] * get_b(k - j);
126 out[i] = acc;
127 }
128 }

130 // Through one transform of a + i b and one inverse.
131 template <typename T, typename B>
132 constexpr void by_fft(const fft_plan<T>& plan, std::span<const T> a, std::size_t nb, B get_b, std::span<T> out,
133 std::span<complex<T>> work, convolve_mode mode) noexcept
134 {
135 const std::size_t na = a.size(), n = plan.size();
136 assert(out.size() == convolve_size(na, nb, mode));
137 if (out.empty())
138 return;
139 assert(n >= na + nb - 1 && work.size() >= n + plan.work_size());
140 const auto z = work.first(n);
141 for (std::size_t i = 0; i < n; ++i)
142 z[i] = {i < na ? a[i] : T(0), i < nb ? get_b(i) : T(0)};
143 plan.template execute<false>(z, work.subspan(n));
144 // A_k = (Z_k + conj Z_(n-k)) / 2, B_k = (Z_k - conj Z_(n-k)) / 2i, so A_k B_k = (Z_k^2 - conj(Z_(n-k))^2) / 4i.
145 for (std::size_t k = 0; 2 * k <= n; ++k) {
146 const std::size_t r = (n - k) % n;
147 const complex<T> zk = z[k], zr = z[r];
148 const complex<T> pk = zk * zk - conj(zr) * conj(zr), pr = zr * zr - conj(zk) * conj(zk);
149 z[k] = complex<T>{pk.im, -pk.re} * T(0.25); // divided by 4i
150 z[r] = complex<T>{pr.im, -pr.re} * T(0.25);
151 }
152 plan.template execute<true>(z, work.subspan(n));
153 const T inv = T(1) / T(static_cast<long long>(n));
154 const std::size_t s = start(na, nb, mode);
155 for (std::size_t i = 0; i < out.size(); ++i)
156 out[i] = z[s + i].re * inv;
157 }

260template <math::math_number T>
261constexpr void overlap_add(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> h, std::span<const std::type_identity_t<T>> x,
262 std::span<std::type_identity_t<T>> out, std::span<complex<std::type_identity_t<T>>> work) noexcept
263{
264 const std::size_t n = plan.size(), taps = h.size();
265 assert(out.size() == convolve_size(x.size(), taps, convolve_mode::full));
266 if (out.empty())
267 return;
268 assert(n >= taps && work.size() >= block_filter_work_size(plan));
269 const std::size_t step = n - taps + 1;
270 const auto hs = work.first(n), block = work.subspan(n, n), rest = work.subspan(2 * n);
271 detail::convolve::spectrum_of(plan, std::span<const T>(h), hs, rest);
272 const T inv = T(1) / T(static_cast<long long>(n));
273 for (auto& v : out)
274 v = T(0);
275 for (std::size_t s = 0; s < x.size(); s += step) {
276 for (std::size_t i = 0; i < n; ++i)
277 block[i] = {i < step && s + i < x.size() ? x[s + i] : T(0), T(0)};
278 plan.template execute<false>(block, rest);
279 for (std::size_t i = 0; i < n; ++i)
280 block[i] *= hs[i];
281 plan.template execute<true>(block, rest);
282 for (std::size_t i = 0; i < n && s + i < out.size(); ++i)
283 out[s + i] += block[i].re * inv;
284 }
285}

294template <math::math_number T>
295constexpr void overlap_save(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> h, std::span<const std::type_identity_t<T>> x,
296 std::span<std::type_identity_t<T>> out, std::span<complex<std::type_identity_t<T>>> work) noexcept
297{
298 const std::size_t n = plan.size(), taps = h.size();
299 assert(out.size() == x.size());
300 if (out.empty() || taps == 0) {
301 for (auto& v : out)
302 v = T(0);
303 return;
304 }
305 assert(n >= taps && work.size() >= block_filter_work_size(plan));
306 const std::size_t step = n - taps + 1, history = taps - 1;
307 const auto hs = work.first(n), block = work.subspan(n, n), rest = work.subspan(2 * n);
308 detail::convolve::spectrum_of(plan, std::span<const T>(h), hs, rest);
309 const T inv = T(1) / T(static_cast<long long>(n));
310 for (std::size_t s = 0; s < x.size(); s += step) {
311 // the block is x[s - history, s + step), zero outside the signal
312 for (std::size_t i = 0; i < n; ++i) {
313 const std::size_t p = s + i;
314 block[i] = {p >= history && p - history < x.size() ? x[p - history] : T(0), T(0)};
315 }
316 plan.template execute<false>(block, rest);
317 for (std::size_t i = 0; i < n; ++i)
318 block[i] *= hs[i];
319 plan.template execute<true>(block, rest);
320 for (std::size_t i = 0; i < step && s + i < out.size(); ++i)
321 out[s + i] = block[history + i].re * inv;
322 }
323}
