91template <mutable_row C, mutable_row S>
92 requires std::same_as<row_value_t<C>, row_value_t<S>>
93constexpr void rope_angles(std::size_t pos, C&& cos, S&& sin, std::type_identity_t<row_value_t<C>> base = 10000)
94{
95 using T = row_value_t<C>;
96 const auto c = as_mutable_row(cos);
97 const auto s = as_mutable_row(sin);
98 assert(c.size() == s.size());
99 for (std::size_t i = 0; i < c.size(); ++i) {
100 const T angle = T(pos) * rope_frequency<T>(i, 2 * c.size(), base);
101 c[i] = math::cos(angle);
102 s[i] = math::sin(angle);
103 }
104}

138 template <typename T>
139 constexpr void rotate(std::span<T> x, std::span<const T> c, std::span<const T> s, rope_layout layout, T sign)
140 {
141 const std::size_t h = c.size();
142 assert(x.size() == 2 * h && s.size() == h);
143 for (std::size_t i = 0; i < h; ++i) {
144 T& a = layout == rope_layout::half ? x[i] : x[2 * i];
145 T& b = layout == rope_layout::half ? x[i + h] : x[2 * i + 1];
146 const T x1 = a, x2 = b, sn = sign * s[i];
147 a = x1 * c[i] - x2 * sn;
148 b = x1 * sn + x2 * c[i];
149 }
150 }
151} // namespace detail::rope
