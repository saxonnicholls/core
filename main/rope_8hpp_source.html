112template <math::math_number T>
113constexpr T rope_frequency(std::size_t i, std::size_t head_dim, T base = T(10000))
114{
115 return T(1) / math::pow(base, T(2 * i) / T(head_dim));
116}

123template <mutable_row C, mutable_row S>
124 requires std::same_as<row_value_t<C>, row_value_t<S>>
125constexpr void rope_angles(std::size_t pos, C&& cos, S&& sin, std::type_identity_t<row_value_t<C>> base = 10000)
126{
127 using T = row_value_t<C>;
128 const auto c = as_mutable_row(cos);
129 const auto s = as_mutable_row(sin);
130 assert(c.size() == s.size());
131 for (std::size_t i = 0; i < c.size(); ++i) {
132 const T angle = T(pos) * rope_frequency<T>(i, 2 * c.size(), base);
133 c[i] = math::cos(angle);
134 s[i] = math::sin(angle);
135 }
136}

171template <math::math_number T>
172constexpr T yarn_ramp(T r, T alpha, T beta) noexcept
173{
174 return r < alpha ? T(0) : r > beta ? T(1) : (r - alpha) / (beta - alpha);
175}

185template <math::math_number T>
186constexpr T yarn_correction_dim(T rotations, std::size_t head_dim, T base, std::size_t context)
187{
188 return T(head_dim) * math::log(T(context) / (rotations * T(2) * T(math::pi))) / (T(2) * math::log(base));

202constexpr T yarn_attention_factor(T factor, T mscale = T(1))
203{

215template <math::math_number T>
216constexpr T rope_scaled_frequency(std::size_t i, std::size_t head_dim, T base, const rope_scaling<T>& scaling)
217{
218 const T f = rope_frequency<T>(i, head_dim, base), s = scaling.factor;
219 if (scaling.kind == rope_scaling_kind::none || s == T(1))
220 return f;
221 switch (scaling.kind) {
222 case rope_scaling_kind::linear:
223 return f / s;
224 case rope_scaling_kind::ntk:
225 assert(head_dim > 2);
226 return rope_frequency<T>(i, head_dim, base * math::pow(s, T(head_dim) / T(head_dim - 2)));
227 case rope_scaling_kind::yarn: {
228 assert(scaling.original_context > 0 && scaling.beta_fast > scaling.beta_slow);
229 const T r = T(scaling.original_context) * f / (T(2) * T(math::pi)); // L / λ, λ = 2π / θ
230 const T g = yarn_ramp(r, scaling.beta_slow, scaling.beta_fast);
231 return (T(1) - g) * f / s + g * f;
232 }
233 default: { // yarn_index_ramp
234 assert(scaling.original_context > 0 && scaling.beta_fast > scaling.beta_slow);
235 const T lo_dim = math::floor(yarn_correction_dim(scaling.beta_fast, head_dim, base, scaling.original_context));
236 const T hi_dim = -math::floor(-yarn_correction_dim(scaling.beta_slow, head_dim, base, scaling.original_context));
237 const T lo = lo_dim < T(0) ? T(0) : lo_dim;
238 T hi = hi_dim > T(head_dim - 1) ? T(head_dim - 1) : hi_dim;
239 if (hi == lo)
240 hi += T(0.001);
241 const T ramp = yarn_ramp(T(i), lo, hi); // clamp((i - lo) / (hi - lo), 0, 1)
242 const T smooth = T(1) - ramp;
243 return f / s * (T(1) - smooth) + f * smooth;
244 }
245 }
246}

255template <mutable_row C, mutable_row S>
256 requires std::same_as<row_value_t<C>, row_value_t<S>>
257constexpr void rope_angles(std::size_t pos, C&& cos, S&& sin, std::type_identity_t<row_value_t<C>> base,
258 const rope_scaling<row_value_t<C>>& scaling)
259{
260 using T = row_value_t<C>;
261 const auto c = as_mutable_row(cos);
262 const auto s = as_mutable_row(sin);
263 assert(c.size() == s.size());
264 if (scaling.amplitude == T(1) && (scaling.kind == rope_scaling_kind::none || scaling.factor == T(1))) {
265 // The plain angles through the plain function, so that they are its bits however each call is optimised (a
266 // compiler may turn pow(base, 0.5) into sqrt(base) where it sees the exponent, and a platform's pow differs)
267 rope_angles(pos, c, s, base);
268 return;
269 }
270 for (std::size_t i = 0; i < c.size(); ++i) {
271 const T angle = T(pos) * rope_scaled_frequency<T>(i, 2 * c.size(), base, scaling);
272 c[i] = scaling.amplitude * math::cos(angle);
273 s[i] = scaling.amplitude * math::sin(angle);
274 }
275}

282template <math::math_number T, std::size_t Positions, std::size_t HeadDim>
283 requires(HeadDim % 2 == 0 && HeadDim > 0)
284struct rope_table
285{
286 std::array<std::array<T, HeadDim / 2>, Positions> cos{};
287 std::array<std::array<T, HeadDim / 2>, Positions> sin{};
288
291 constexpr explicit rope_table(T base = T(10000))
292 {
293 for (std::size_t p = 0; p < Positions; ++p)
294 rope_angles(p, cos[p], sin[p], base);
295 }
296
300 constexpr rope_table(T base, const rope_scaling<T>& scaling)
301 {
302 for (std::size_t p = 0; p < Positions; ++p)
303 rope_angles(p, cos[p], sin[p], base, scaling);
304 }

318 template <typename T>
319 constexpr void rotate(std::span<T> x, std::span<const T> c, std::span<const T> s, rope_layout layout, T sign)
320 {
321 const std::size_t h = c.size();
322 assert(x.size() == 2 * h && s.size() == h);
323 for (std::size_t i = 0; i < h; ++i) {
324 T& a = layout == rope_layout::half ? x[i] : x[2 * i];
325 T& b = layout == rope_layout::half ? x[i + h] : x[2 * i + 1];
326 const T x1 = a, x2 = b, sn = sign * s[i];
327 a = x1 * c[i] - x2 * sn;
328 b = x1 * sn + x2 * c[i];
329 }
330 }
331} // namespace detail::rope
