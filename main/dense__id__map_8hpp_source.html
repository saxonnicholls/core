142 bool insert(key_type key, Value value)
143 {
144 if (key == empty_key) [[unlikely]]
145 return false; // reserved, as in flat_id_map
146 if (slots_.empty() && fallback_.empty()) {
147 base_ = key;
148 slots_.assign(initial_, nullptr);
149 }
150 key_type i = key - base_;
151 if (i >= slots_.size() && key >= base_ && i < max_window_ && !fallback_.contains(key)) {
152 grow(static_cast<std::size_t>(i) + 1);
153 i = key - base_;
154 }
155 if (i < slots_.size()) {
156 if (slots_[i]) return false;
157 slots_[i] = value;
158 ++in_window_;
159 if (i < low_) low_ = static_cast<std::size_t>(i); // a re-used id below the lowest live one
160 return true;
161 }
162 return fallback_.insert(key, value);
163 }

179 }

200 // Lowest possibly-occupied slot, advanced lazily as the front empties.
201 void advance_low() noexcept
202 {
203 while (low_ < slots_.size() && !slots_[low_]) ++low_;
204 }

206 void grow(std::size_t needed)
207 {
208 // Strong guarantee: everything that can throw (the new window, the list of fallback keys it will cover) is
209 // allocated first; the map changes only after that, in steps that cannot throw.
210 // Slide the window up first if its front half is empty (ids below the lowest live one are never reused).
211 advance_low(); // moves a lazy cursor over empty slots only: no observable change
212 const std::size_t drop = low_;
213 std::size_t size = std::max<std::size_t>(slots_.size(), 64);
214 while (size < needed - drop) size *= 2;
215 const bool slide = size == slots_.size() && drop > 0;
216 std::vector<Value> next;
217 if (!slide) next.assign(std::max(size, needed - drop), nullptr);
218 const key_type new_base = base_ + drop;
219 const std::size_t new_size = slide ? slots_.size() : next.size();
220 // Keep the invariant "no fallback key lies inside the window": find the keys the new window covers.
221 std::vector<std::pair<key_type, Value>> covered;
222 fallback_.for_each([&](key_type k, Value v) {
223 if (k - new_base < new_size) covered.emplace_back(k, v);
224 });
225 // Commit: nothing below allocates or throws.
226 if (slide) {
227 std::move(slots_.begin() + static_cast<std::ptrdiff_t>(drop), slots_.end(), slots_.begin());
228 std::fill(slots_.end() - static_cast<std::ptrdiff_t>(drop), slots_.end(), nullptr);
229 } else {
230 std::copy(slots_.begin() + static_cast<std::ptrdiff_t>(drop), slots_.end(), next.begin());
231 slots_.swap(next);
232 }
233 base_ = new_base;
234 low_ = 0;
235 for (const auto& [k, v] : covered) {
236 fallback_.erase(k);
237 slots_[k - base_] = v;
238 ++in_window_;
239 }
240 }
241
242 std::vector<Value> slots_;
243 flat_id_map<Value> fallback_;
244 key_type base_ = 0;
245 std::size_t in_window_ = 0;
246 std::size_t low_ = 0;
247 std::size_t initial_;
248 std::size_t max_window_;
249};
