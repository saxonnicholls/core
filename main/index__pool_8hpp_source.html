119template <class T, class Index = std::uint32_t>
120class index_pool
121{
122 static_assert(std::is_trivially_copyable_v<T>, "snicholls: index_pool needs a trivially copyable T (it may be "
123 "copied, mapped or reopened byte for byte)");
124 static_assert(std::is_same_v<Index, std::uint32_t> || std::is_same_v<Index, std::uint64_t>,
125 "snicholls: index_pool indices are std::uint32_t or std::uint64_t");

127 struct state
128 {
129 std::uint64_t magic;
130 std::uint64_t slot_size;
131 std::uint64_t index_size;
132 std::uint64_t high_water;
133 std::uint64_t free_head;
134 std::uint64_t live;
135 std::uint64_t reserved[2];
136 };
137 static_assert(sizeof(state) == 64 && std::is_trivially_copyable_v<state>); // read from a region with memcpy
138
139 static constexpr std::size_t round_up(std::size_t n, std::size_t a) noexcept { return (n + a - 1) / a * a; }

141public:
142 using value_type = T;
143 using index_type = Index;

146 static constexpr Index npos = std::numeric_limits<Index>::max();
148 static constexpr std::uint64_t magic = 0x314c505844494e53ull;
150 static constexpr std::size_t region_alignment = std::max({alignof(T), alignof(Index), alignof(state)});
152 static constexpr std::size_t slot_size = round_up(std::max(sizeof(T), sizeof(Index)),
153 std::max(alignof(T), alignof(Index)));
155 static constexpr std::size_t header_size = round_up(sizeof(state), region_alignment);
156
160 [[nodiscard]] static constexpr std::size_t bytes_for(std::size_t slots) noexcept
161 {
162 return header_size + slots * slot_size;
163 }

171 [[nodiscard]] static result<index_pool, index_pool_error> format(std::span<std::byte> region) noexcept
172 {
173 if (reinterpret_cast<std::uintptr_t>(region.data()) % region_alignment != 0) return err(index_pool_error::misaligned);
174 if (region.size() < header_size) return err(index_pool_error::too_small);
175 index_pool p;
176 p.base_ = region.data();
177 p.capacity_ = capacity_of(region.size());
178 p.st_ = ::new (static_cast<void*>(region.data())) state{magic, slot_size, sizeof(Index), 0, npos, 0, {0, 0}};
179 return ok(p);
180 }

185 [[nodiscard]] static result<index_pool, index_pool_error> attach(std::span<std::byte> region) noexcept
186 {
187 index_pool p;
188 if (auto r = p.rebase(region); r.is_err()) return err(r.unwrap_err_unchecked());
189 return ok(p);
190 }

195 [[nodiscard]] result<void, index_pool_error> rebase(std::span<std::byte> region) noexcept
196 {
197 if (reinterpret_cast<std::uintptr_t>(region.data()) % region_alignment != 0) return err(index_pool_error::misaligned);
198 if (region.size() < header_size) return err(index_pool_error::too_small);
199 // The bytes may hold another pool's state (another T or Index), or no object at all (a file mapped in): read
200 // them as bytes, never through this pool's state type, which would be undefined (GCC at -O3 then reads stale
201 // bytes). Only a valid header becomes this pool's state, constructed in place with the same value.
202 state seen;
203 std::memcpy(&seen, region.data(), sizeof(state));
204 if (seen.magic != magic) return err(index_pool_error::not_formatted);
205 if (seen.slot_size != slot_size || seen.index_size != sizeof(Index)) return err(index_pool_error::layout_mismatch);
206 const std::size_t cap = capacity_of(region.size());
207 if (seen.live > seen.high_water || (seen.free_head != npos && seen.free_head >= seen.high_water)
208 || (seen.free_head == npos && seen.live != seen.high_water))
209 return err(index_pool_error::corrupt);
210 if (seen.high_water > cap) return err(index_pool_error::too_small);
211 base_ = region.data();
212 capacity_ = cap;
213 st_ = ::new (static_cast<void*>(region.data())) state(seen);
214 return ok();
215 }
216
219 [[nodiscard]] result<Index, index_pool_error> allocate() noexcept
220 {
221 if (st_->free_head != npos) {
222 const auto i = static_cast<Index>(st_->free_head);
223 Index next;
224 std::memcpy(&next, slot_bytes(i), sizeof next);
225 st_->free_head = next;
226 ++st_->live;
227 return ok(i);
228 }
229 if (st_->high_water == capacity_) return err(index_pool_error::full);
230 const auto i = static_cast<Index>(st_->high_water++);
231 ++st_->live;
232 return ok(i);
233 }
234
238 [[nodiscard]] result<Index, index_pool_error> create(const T& value) noexcept
239 {
240 auto r = allocate();
241 if (r.is_ok()) ::new (static_cast<void*>(slot_bytes(*r))) T(value);
242 return r;
243 }
244
247 void destroy(Index i) noexcept
248 {
249 const auto next = static_cast<Index>(st_->free_head);
250 std::memcpy(slot_bytes(i), &next, sizeof next);
251 st_->free_head = i;
252 --st_->live;
253 }

258 [[nodiscard]] T& operator[](Index i) noexcept { return *std::launder(reinterpret_cast<T*>(slot_bytes(i))); }
262 [[nodiscard]] const T& operator[](Index i) const noexcept
263 {
264 return *std::launder(reinterpret_cast<const T*>(slot_bytes(i)));
265 }

269 [[nodiscard]] std::size_t live() const noexcept { return static_cast<std::size_t>(st_->live); }
272 [[nodiscard]] std::size_t high_water() const noexcept { return static_cast<std::size_t>(st_->high_water); }
275 [[nodiscard]] std::size_t capacity() const noexcept { return capacity_; }
278 [[nodiscard]] bool attached() const noexcept { return st_ != nullptr; }
281 [[nodiscard]] std::byte* data() const noexcept { return base_; }
282
283private:
284 static constexpr std::size_t capacity_of(std::size_t bytes) noexcept
285 {
286 return std::min<std::size_t>((bytes - header_size) / slot_size, static_cast<std::size_t>(npos));
287 }
288 std::byte* slot_bytes(Index i) const noexcept { return base_ + header_size + static_cast<std::size_t>(i) * slot_size; }
289
290 std::byte* base_ = nullptr;
291 state* st_ = nullptr;
292 std::size_t capacity_ = 0;
293};
