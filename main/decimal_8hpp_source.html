103 [[nodiscard]] static constexpr std::optional<decimal> from_parts(std::int64_t mantissa, int exponent) noexcept
104 {
105 if (mantissa == 0) return decimal{};
106 if (mantissa == INT64_MIN) return std::nullopt;
107 while (mantissa % 10 == 0) {
108 mantissa /= 10;
109 ++exponent;
110 }
111 const std::int64_t magnitude = mantissa < 0 ? -mantissa : mantissa;
112 if (magnitude >= limit) return std::nullopt;
113 if (exponent < min_exponent || exponent > max_exponent) return std::nullopt;
114 decimal d;
115 d.mantissa_ = mantissa;
116 d.exponent_ = static_cast<std::int8_t>(exponent);
117 return d;
118 }

129 [[nodiscard]] static constexpr std::optional<decimal> parse(std::string_view text,
130 decimal_syntax syntax = decimal_syntax::canonical) noexcept
131 {
132 bool negative = false;
133 if (!text.empty() && (text.front() == '-' || (text.front() == '+' && syntax == decimal_syntax::plain))) {
134 negative = text.front() == '-';
135 text.remove_prefix(1);
136 }
137 const std::size_t point = text.find('.');
138 const std::string_view whole = text.substr(0, point);
139 const std::string_view frac = point == std::string_view::npos ? std::string_view{} : text.substr(point + 1);
140 if (whole.empty() || (point != std::string_view::npos && frac.empty())) return std::nullopt;
141 if (!all_digits(whole) || !all_digits(frac)) return std::nullopt;
142 if (whole.size() > 1 && whole.front() == '0') return std::nullopt; // no leading zeros
143 if (syntax == decimal_syntax::canonical && !frac.empty() && frac.back() == '0') return std::nullopt;
144 // Significant digits: strip leading zeros of whole+frac and trailing zeros of frac.
145 std::int64_t mantissa = 0;
146 int digits = 0;
147 int exponent = 0;
148 int pending_zeros = 0; // zeros of the whole part not yet multiplied in (trailing zeros of an integer)
149 for (const char c : whole) {
150 if (c == '0' && digits == 0) continue;
151 if (c == '0') {
152 ++pending_zeros;
153 continue;
154 }
155 for (; pending_zeros > 0; --pending_zeros) {
156 if (++digits > max_digits) return std::nullopt;
157 mantissa *= 10;
158 }
159 if (++digits > max_digits) return std::nullopt;
160 mantissa = mantissa * 10 + (c - '0');
161 }
162 std::size_t frac_len = frac.size();
163 while (frac_len > 0 && frac[frac_len - 1] == '0') --frac_len; // plain: trailing fractional zeros
164 if (frac_len > 0) {
165 for (; pending_zeros > 0; --pending_zeros) {
166 if (++digits > max_digits) return std::nullopt;
167 mantissa *= 10;
168 }
169 for (std::size_t i = 0; i < frac_len; ++i) {
170 const char c = frac[i];
171 --exponent;
172 if (c == '0' && digits == 0) continue;
173 if (++digits > max_digits) return std::nullopt;
174 mantissa = mantissa * 10 + (c - '0');
175 }
176 } else {
177 exponent += pending_zeros;
178 }
179 if (mantissa == 0) {
180 if (negative && syntax == decimal_syntax::canonical) return std::nullopt; // "-0"
181 return decimal{};
182 }
183 return from_parts(negative ? -mantissa : mantissa, exponent);
184 }

240 friend constexpr std::strong_ordering operator<=>(const decimal& a, const decimal& b) noexcept
241 {
242 if (a.mantissa_ == b.mantissa_ && a.exponent_ == b.exponent_) return std::strong_ordering::equal;
243 const int sa = sign(a.mantissa_), sb = sign(b.mantissa_);
244 if (sa != sb) return sa <=> sb;
245 const std::strong_ordering magnitude = compare_magnitude(a, b);
246 return sa >= 0 ? magnitude : 0 <=> magnitude;
247 }

252 static constexpr bool all_digits(std::string_view s) noexcept
253 {
254 for (const char c : s)
255 if (c < '0' || c > '9') return false;
256 return true;

258 static constexpr int sign(std::int64_t m) noexcept { return m < 0 ? -1 : (m > 0 ? 1 : 0); }
259 static constexpr int magnitude_digits(std::int64_t m) noexcept
260 {
261 std::uint64_t u = m < 0 ? static_cast<std::uint64_t>(-m) : static_cast<std::uint64_t>(m);
262 int n = 1;
263 while (u >= 10) {
264 u /= 10;
265 ++n;
266 }
267 return n;
268 }
269 static constexpr std::strong_ordering compare_magnitude(const decimal& a, const decimal& b) noexcept
270 {
271 std::int64_t ma = a.mantissa_ < 0 ? -a.mantissa_ : a.mantissa_;
272 std::int64_t mb = b.mantissa_ < 0 ? -b.mantissa_ : b.mantissa_;
273 const int da = magnitude_digits(ma), db = magnitude_digits(mb);
274 const int ta = a.exponent_ + da, tb = b.exponent_ + db; // position of the leading digit
275 if (ta != tb) return ta <=> tb;
276 for (int i = da; i < db; ++i) ma *= 10; // same leading position: align digit counts (both <= 18)
277 for (int i = db; i < da; ++i) mb *= 10;
278 return ma <=> mb;
279 }

281 constexpr std::size_t write(std::array<char, 40>& out) const noexcept
282 {
283 std::size_t n = 0;
284 if (mantissa_ == 0) {
285 out[n++] = '0';
286 return n;
287 }
288 if (mantissa_ < 0) out[n++] = '-';
289 std::array<char, 20> digits{};
290 int count = 0;
291 for (std::uint64_t u = mantissa_ < 0 ? static_cast<std::uint64_t>(-mantissa_) : static_cast<std::uint64_t>(mantissa_);
292 u > 0; u /= 10)
293 digits[static_cast<std::size_t>(count++)] = static_cast<char>('0' + u % 10);
294 // digits[] is reversed. value = digits x 10^exponent_.
295 if (exponent_ >= 0) {
296 for (int i = count - 1; i >= 0; --i) out[n++] = digits[static_cast<std::size_t>(i)];
297 for (int i = 0; i < exponent_; ++i) out[n++] = '0';
298 return n;
299 }
300 const int frac = -exponent_;
301 if (count > frac) {
302 for (int i = count - 1; i >= frac; --i) out[n++] = digits[static_cast<std::size_t>(i)];
303 out[n++] = '.';
304 for (int i = frac - 1; i >= 0; --i) out[n++] = digits[static_cast<std::size_t>(i)];
305 } else {
306 out[n++] = '0';
307 out[n++] = '.';
308 for (int i = 0; i < frac - count; ++i) out[n++] = '0';
309 for (int i = count - 1; i >= 0; --i) out[n++] = digits[static_cast<std::size_t>(i)];
310 }
311 return n;
312 }

315 std::int8_t exponent_ = 0;
