97 struct kv_conformance_report {
98 std::size_t checks = 0;
99 std::vector<kv_conformance_failure> failures;
100
103 [[nodiscard]] bool ok() const noexcept { return failures.empty(); }
104
108 [[nodiscard]] bool failed(std::string_view name) const noexcept
109 {
110 for (const auto& f : failures)
111 if (f.check == name)
112 return true;
113 return false;
114 }
115 };

170 // Runs one check on a fresh store from make(), recording an escaping exception as a failure.
171 template <typename Make, typename Check>
172 void kv_run(kv_conformance_report& report, std::string_view name, Make& make, Check&& check)
173 {
174 ++report.checks;
175 const kv_expect expect{name, report};
176#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
177 try {
178 decltype(auto) store = make();
179 check(kv_target(store), expect);
180 } catch (...) {
181 expect(false, "no exception crosses the interface");
182 }
183#else
184 decltype(auto) store = make();
185 check(kv_target(store), expect);
186#endif
187 }
188 }
