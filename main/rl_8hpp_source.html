141template <math::math_number T, typename B>
142constexpr void group_advantages(const B&, std::span<const T> rewards, std::size_t group_size, std::type_identity_t<std::span<T>> advantages,
143 advantage_scale scale = advantage_scale::group_std, double epsilon = 1e-4) noexcept(std::is_floating_point_v<T>)
144{
145 assert(group_size > 0 && rewards.size() % group_size == 0 && advantages.size() == rewards.size());
146 const T g = T(group_size);
147 for (std::size_t b = 0; b < rewards.size(); b += group_size) {
148 T mean = T(0);
149 for (std::size_t i = b; i < b + group_size; ++i)
150 mean += rewards[i];
151 mean = mean / g;
152 T div = T(1);
153 if (scale == advantage_scale::group_std) {
154 T ss = T(0);
155 for (std::size_t i = b; i < b + group_size; ++i)
156 ss += (rewards[i] - mean) * (rewards[i] - mean);
157 const T sd = group_size > 1 ? math::sqrt(ss / T(group_size - 1)) : T(0);
158 div = sd + T(epsilon);
159 }
160 for (std::size_t i = b; i < b + group_size; ++i)
161 advantages[i] = (rewards[i] - mean) / div;
162 }
163}

194template <math::math_number T>
195constexpr T kl_k3(T logp, std::type_identity_t<T> ref_logp) noexcept(std::is_floating_point_v<T>)
196{
197 const T u = ref_logp - logp;
198 return math::expm1(u) - u;
199}

203template <math::math_number T>
204constexpr T kl_k3_derivative(T logp, std::type_identity_t<T> ref_logp) noexcept(std::is_floating_point_v<T>)
205{
206 return -math::expm1(ref_logp - logp);

225template <math::math_number T>
226constexpr policy_token<T> policy_token_loss(T logp, std::type_identity_t<T> old_logp, std::type_identity_t<T> ref_logp,
227 std::type_identity_t<T> advantage, const policy_loss_config& c) noexcept(std::is_floating_point_v<T>)
228{
229 policy_token<T> p;
230 p.ratio = math::exp(logp - old_logp);
231 const T lo = T(1) - T(c.clip_low), hi = T(1) + T(c.clip_high);
232 const T clipped_ratio = p.ratio < lo ? lo : (p.ratio > hi ? hi : p.ratio);
233 const T unclipped = p.ratio * advantage, clipped = clipped_ratio * advantage;
234 p.clipped = clipped < unclipped;
235 p.loss = p.clipped ? -clipped : -unclipped;
236 p.d_logp = p.clipped ? T(0) : -unclipped; // d(r Â)/d logp = r Â
237 if (c.kl_beta != 0.0) {
238 const T beta = T(c.kl_beta);
239 p.kl = kl_k3(logp, ref_logp);
240 p.loss += beta * p.kl;
241 p.d_logp += beta * kl_k3_derivative(logp, ref_logp);
242 }
243 return p;
244}

248template <math::math_number T>
249struct policy_loss_metrics
250{
251 std::uint64_t tokens = 0;
252 std::uint64_t clipped = 0;
253 T kl = T(0);
254 T ratio = T(0);
255 T entropy = T(0);
256
258 constexpr void merge(const policy_loss_metrics& o) noexcept(std::is_floating_point_v<T>)
259 {
260 tokens += o.tokens;
261 clipped += o.clipped;
262 kl += o.kl;
263 ratio += o.ratio;
264 entropy += o.entropy;
265 }
266
267 [[nodiscard]] constexpr double clip_fraction() const noexcept { return tokens == 0 ? 0.0 : static_cast<double>(clipped) / static_cast<double>(tokens); }
269 [[nodiscard]] constexpr T mean_kl() const noexcept(std::is_floating_point_v<T>) { return tokens == 0 ? T(0) : kl / T(tokens); }
271 [[nodiscard]] constexpr T mean_ratio() const noexcept(std::is_floating_point_v<T>) { return tokens == 0 ? T(0) : ratio / T(tokens); }
273 [[nodiscard]] constexpr T mean_entropy() const noexcept(std::is_floating_point_v<T>) { return tokens == 0 ? T(0) : entropy / T(tokens); }
274};

279constexpr double token_weight(std::span<const std::uint8_t> mask, std::size_t rows, const policy_loss_config& c, std::size_t row,
280 std::size_t total) noexcept
281{
282 const std::size_t len = mask.size() / rows;
283 std::size_t n = 0;
284 for (std::size_t t = row * len; t < (row + 1) * len; ++t)
285 n += mask[t] != 0 ? 1 : 0;
286 if (n == 0)
287 return 0.0;
288 switch (c.aggregation) {
289 case token_aggregation::sequence_mean: return 1.0 / (static_cast<double>(rows) * static_cast<double>(n));
290 case token_aggregation::token_mean: return 1.0 / static_cast<double>(total);
291 case token_aggregation::constant: return 1.0 / (static_cast<double>(rows) * static_cast<double>(c.max_tokens == 0 ? len : c.max_tokens));
292 }
293 return 0.0;
294}

310template <math::math_number T, typename B>
311constexpr T policy_loss_rows(const B&, std::span<const std::uint8_t> mask, std::span<const T> logps, std::type_identity_t<std::span<const T>> old_logps,
312 std::type_identity_t<std::span<const T>> ref_logps, std::type_identity_t<std::span<const T>> advantages,
313 std::type_identity_t<std::span<const T>> entropies, std::type_identity_t<std::span<T>> dlogps,
314 std::type_identity_t<std::span<T>> dentropies, std::size_t rows, const policy_loss_config& c,
315 policy_loss_metrics<T>& metrics) noexcept(std::is_floating_point_v<T>)
316{
317 const std::size_t n = mask.size();
318 assert(logps.size() == n && old_logps.size() == n && advantages.size() == n && dlogps.size() == n
319 && (c.kl_beta == 0.0 || ref_logps.size() == n) && (entropies.empty() || entropies.size() == n)
320 && dentropies.size() == entropies.size() && (rows == 0 ? n == 0 : n % rows == 0));
321 if (rows == 0)
322 return T(0);
323 const std::size_t len = n / rows;
324 std::size_t total = 0;
325 for (const auto m : mask)
326 total += m != 0 ? 1 : 0;
327 const T coef = T(c.entropy_coef);
328 T loss = T(0);
329 for (std::size_t r = 0; r < rows; ++r) {
330 const T w = T(token_weight(mask, rows, c, r, total));
331 for (std::size_t t = r * len; t < (r + 1) * len; ++t) {
332 if (mask[t] == 0) {
333 dlogps[t] = T(0);
334 if (!dentropies.empty())
335 dentropies[t] = T(0);
336 continue;
337 }
338 const auto p = policy_token_loss(logps[t], old_logps[t], ref_logps.empty() ? T(0) : ref_logps[t], advantages[t], c);
339 T l = p.loss;
340 ++metrics.tokens;
341 metrics.clipped += p.clipped ? 1 : 0;
342 metrics.kl += p.kl;
343 metrics.ratio += p.ratio;
344 if (!entropies.empty()) {
345 l -= coef * entropies[t];
346 metrics.entropy += entropies[t];
347 dentropies[t] = -coef * w;
348 }
349 loss += w * l;
350 dlogps[t] = w * p.d_logp;
351 }
352 }
353 return loss;
354}

364template <math::math_number T, typename B>
365constexpr void gae_rows(const B&, std::span<const std::uint8_t> mask, std::span<const T> rewards, std::type_identity_t<std::span<const T>> values,
366 double gamma, double lambda, std::type_identity_t<std::span<T>> advantages, std::type_identity_t<std::span<T>> returns,
367 std::size_t rows) noexcept(std::is_floating_point_v<T>)
368{
369 const std::size_t n = mask.size();
370 assert(rewards.size() == n && values.size() == n && advantages.size() == n && returns.size() == n && (rows == 0 ? n == 0 : n % rows == 0));
371 if (rows == 0)
372 return;
373 const std::size_t len = n / rows;
374 const T g = T(gamma), gl = T(gamma) * T(lambda);
375 for (std::size_t r = 0; r < rows; ++r) {
376 T next_value = T(0), next_adv = T(0);
377 for (std::size_t k = len; k-- > 0;) {
378 const std::size_t t = r * len + k;
379 if (mask[t] == 0) {
380 advantages[t] = T(0);
381 returns[t] = T(0);
382 continue;
383 }
384 const T delta = rewards[t] + g * next_value - values[t];
385 next_adv = delta + gl * next_adv;
386 next_value = values[t];
387 advantages[t] = next_adv;
388 returns[t] = next_adv + values[t];
389 }
390 }

395template <math::math_number T, typename B>
396constexpr void masked_whiten(const B&, std::span<const std::uint8_t> mask, std::span<const T> values, std::type_identity_t<std::span<T>> out) noexcept(std::is_floating_point_v<T>)
397{
398 assert(values.size() == mask.size() && out.size() == mask.size());
399 std::size_t n = 0;
400 T mean = T(0);
401 for (std::size_t i = 0; i < mask.size(); ++i)
402 if (mask[i] != 0) {
403 mean += values[i];
404 ++n;
405 }
406 if (n == 0) {
407 for (T& v : out)
408 v = T(0);
409 return;
410 }
411 mean = mean / T(n);
412 T ss = T(0);
413 for (std::size_t i = 0; i < mask.size(); ++i)
414 if (mask[i] != 0)
415 ss += (values[i] - mean) * (values[i] - mean);
416 const T inv = T(1) / math::sqrt((n > 1 ? ss / T(n - 1) : T(0)) + T(1e-8));
417 for (std::size_t i = 0; i < mask.size(); ++i)
418 out[i] = mask[i] != 0 ? (values[i] - mean) * inv : T(0);
419}

426template <math::math_number T, typename B>
427constexpr T value_loss_rows(const B&, std::span<const std::uint8_t> mask, std::span<const T> values, std::type_identity_t<std::span<const T>> old_values,
428 std::type_identity_t<std::span<const T>> returns, double clip, std::type_identity_t<std::span<T>> dvalues) noexcept(std::is_floating_point_v<T>)
429{
430 const std::size_t n = mask.size();
431 assert(values.size() == n && old_values.size() == n && returns.size() == n && dvalues.size() == n);
432 std::size_t count = 0;
433 for (const auto m : mask)
434 count += m != 0 ? 1 : 0;
435 const T w = count == 0 ? T(0) : T(1) / T(count);
436 const T eps = T(clip);
437 T loss = T(0);
438 for (std::size_t t = 0; t < n; ++t) {
439 if (mask[t] == 0) {
440 dvalues[t] = T(0);
441 continue;
442 }
443 const T e = values[t] - returns[t];
444 T l = T(0.5) * e * e, d = e;
445 if (clip > 0.0) {
446 const T lo = old_values[t] - eps, hi = old_values[t] + eps;
447 const T vc = values[t] < lo ? lo : (values[t] > hi ? hi : values[t]);
448 const T ec = vc - returns[t];
449 const T lc = T(0.5) * ec * ec;
450 if (lc > l) {
451 l = lc;
452 d = T(0); // the clipped value does not move with V where it binds
453 }
454 }
455 loss += w * l;
456 dvalues[t] = w * d;
457 }
458 return loss;
459}
