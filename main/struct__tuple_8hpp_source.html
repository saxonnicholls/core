74// get<I> for any tuple-like: std::get for std types, the type's own get (via ADL) otherwise.
75template <std::size_t I, typename Tup>
76constexpr decltype(auto) element(Tup&& tup) noexcept
77{
78 using std::get;
79 return get<I>(static_cast<Tup&&>(tup));

106// tie for lvalues, tuple of rvalue references for rvalues (moves members out).
107template <typename S>
108constexpr auto forward_tie_struct(S&& s) noexcept
109{
110 if constexpr (std::is_lvalue_reference_v<S>) {
111 return tie_struct(s);
112 } else {
113 return std::apply([](auto&... m) noexcept { return std::forward_as_tuple(std::move(m)...); }, tie_struct(s));
114 }
115}
116
117template <typename S>
118using struct_tie_t = decltype(forward_tie_struct(std::declval<S>()));

129// Whether S{get<I>(tup)...} is well-formed: every element converts to its member without narrowing. Asked in a
130// requires-expression, where narrowing is ill-formed on every compiler; GCC otherwise only warns (-Wnarrowing)
131// about a narrowing brace-initialisation inside a template, which would let a narrowing from_tuple compile there.
132template <typename S, typename Tup, std::size_t... I>
133constexpr bool brace_initialisable_from(std::index_sequence<I...>) noexcept
134{
135 return requires(Tup&& tup) { S{element<I>(static_cast<Tup&&>(tup))...}; };
136}
137
138template <typename S, typename Tup>
139constexpr bool converts_without_narrowing() noexcept
140{
141 if constexpr (std::tuple_size_v<std::remove_cvref_t<Tup>> != member_count_v<S>)
142 return true; // the size static_assert reports this one
143 else
144 return brace_initialisable_from<S, Tup>(std::make_index_sequence<member_count_v<S>>{});
145}
146
147// The noexcept-specification of from_tuple. Ill-formed input (wrong size, narrowing) gives false here instead of an
148// error, so the error comes from from_tuple's static_asserts with their snicholls:: message: MSVC stops at an
149// ill-formed noexcept-specification and never reaches the body. Well-formed input gives exactly the noexcept of the
150// brace-initialisation, as before.
151template <typename S, typename Tup>
152constexpr bool nothrow_from_tuple() noexcept
153{
154 if constexpr (std::tuple_size_v<std::remove_cvref_t<Tup>> != member_count_v<S>)
155 return false;
156 else if constexpr (!brace_initialisable_from<S, Tup>(std::make_index_sequence<member_count_v<S>>{}))
157 return false;
158 else
159 return noexcept(struct_from_tuple<S>(std::declval<Tup>(), std::make_index_sequence<member_count_v<S>>{}));
160}

251template <typename S, typename Tup>
252 requires plain_struct<S> && tuple_like<Tup>
253constexpr S from_tuple(Tup&& tup) noexcept(detail::nothrow_from_tuple<S, Tup>())
254{
255 static_assert(std::tuple_size_v<std::remove_cvref_t<Tup>> == member_count_v<S>,
256 "snicholls::from_tuple: tuple size must equal the struct's number of members");
257 static_assert(detail::converts_without_narrowing<S, Tup>(),
258 "snicholls::from_tuple: every element must convert to its member without narrowing");
259 return detail::struct_from_tuple<S>(static_cast<Tup&&>(tup), std::make_index_sequence<member_count_v<S>>{});
260}
