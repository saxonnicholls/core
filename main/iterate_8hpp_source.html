144 template <typename F, typename T, typename Same = std::equal_to<>>
145 [[nodiscard]] constexpr fixed_point_result<T> fixed_point(F&& f, T x, std::size_t limit, Same same = {})
146 {
147 fixed_point_result<T> out{x, 0, false};
148 while (out.iterations < limit) {
149 T next = std::invoke(f, std::as_const(out.value));
150 ++out.iterations;
151 if (std::invoke(same, std::as_const(out.value), std::as_const(next))) {
152 out.value = std::move(next);
153 out.converged = true;
154 return out;
155 }
156 out.value = std::move(next);
157 }
158 return out;
159 }
