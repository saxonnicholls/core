74 template <typename Probe>
75 concept https_client_probe = std::is_default_constructible_v<Probe> &&
76 https_client<std::remove_reference_t<decltype(std::declval<Probe&>().client())>> &&
77 requires(Probe& p, std::string_view path, https_response a, capability_errc c, std::string_view r) {
78 { p.url(path) } -> std::convertible_to<std::string>;
79 { p.unreachable_url() } -> std::convertible_to<std::string>;
80 p.answer(a);
81 { p.seen() } -> std::convertible_to<https_observed_request>;
82 p.provider_fails(c, r);
83 };

95 template <typename Probe>

97 [[nodiscard]] capability_conformance_report run_https_client_conformance()
98 {
99 capability_conformance_report report;
100 using detail::header_or_empty;
101
102 detail::capability_check<Probe>(report, "a GET returns status, headers and body", [](Probe& p, const auto& expect) {
103 p.answer({200, {{"Content-Type", "application/json"}, {"Request-Id", "req_1"}}, R"({"ok":true})"});
104 https_request r;
105 r.url = p.url("/v1/thing?expand[]=x");
106 const auto got = p.client().send(r);
107 expect(got.is_ok(), "the request is answered");
108 if (!got.is_ok())
109 return;
110 expect(got->status == 200 && got->ok() && got->body == R"({"ok":true})", "the status and the body");
111 expect(header_or_empty(got->headers, "request-id") == "req_1", "the headers, found ignoring case");
112 const https_observed_request seen = p.seen();
113 expect(seen.method == "GET" && seen.target == "/v1/thing?expand[]=x", "the server saw the method and the target");
114 });
115
116 detail::capability_check<Probe>(report, "an error status is a response, not an error", [](Probe& p, const auto& expect) {
117 p.answer({503, {{"Retry-After", "2"}}, "busy"});
118 https_request r;
119 r.url = p.url("/busy");
120 const auto got = p.client().send(r);
121 expect(got.is_ok() && got->status == 503 && !got->ok(), "a 503 is a response");
122 if (got.is_ok())
123 expect(header_or_empty(got->headers, "Retry-After") == "2", "with its headers");
124 });
125
126 detail::capability_check<Probe>(report, "a POST carries its body, idempotency key and authorization", [](Probe& p, const auto& expect) {
127 p.answer({200, {}, "{}"});
128 https_request r;
129 r.method = http_method::post;
130 r.url = p.url("/v1/refunds");
131 r.headers = {{"Content-Type", "application/x-www-form-urlencoded"}};
132 r.authorization = secret<std::string>{"Bearer kit_test_only"};
133 r.body = "payment_intent=pi_1&amount=100";
134 r.idempotency_key = "kit-key-1";
135 expect(p.client().send(r).is_ok(), "the request is answered");
136 const https_observed_request seen = p.seen();
137 expect(seen.method == "POST" && seen.body == r.body, "the method and the body arrive");
138 expect(header_or_empty(seen.headers, "Idempotency-Key") == "kit-key-1", "the idempotency key arrives as a header");
139 expect(header_or_empty(seen.headers, "Authorization") == "Bearer kit_test_only", "the credential arrives as Authorization");
140 expect(header_or_empty(seen.headers, "Content-Type") == "application/x-www-form-urlencoded", "the headers arrive");
141 });

143 detail::capability_check<Probe>(report, "only https URLs and clean headers are sent", [](Probe& p, const auto& expect) {
144 const auto refused = [&](https_request r) {
145 const auto got = p.client().send(r);
146 return got.is_err() && got.unwrap_err().code == capability_errc::invalid_argument;
147 };
148 https_request plain;
149 plain.url = "http://example.test/";
150 expect(refused(plain), "http:// is invalid_argument");
151 https_request other;
152 other.url = "ftp://example.test/";
153 expect(refused(other), "ftp:// is invalid_argument");
154 https_request user;
155 user.url = "https://user:pass@example.test/";
156 expect(refused(user), "user information in the URL is invalid_argument");
157 https_request split;
158 split.url = p.url("/x");
159 split.headers = {{"X-Evil", "a\r\nInjected: yes"}};
160 expect(refused(split), "a value with CR LF is invalid_argument");
161 https_request bad_name;
162 bad_name.url = p.url("/x");
163 bad_name.headers = {{"Bad:Name", "v"}};
164 expect(refused(bad_name), "a name with a colon is invalid_argument");
165 https_request smuggled;
166 smuggled.url = p.url("/x");
167 smuggled.authorization = secret<std::string>{"Bearer a\r\nX-Injected: 1"};
168 expect(refused(smuggled), "a credential with CR LF is invalid_argument");
169 });

171 detail::capability_check<Probe>(report, "an unreachable host is unavailable", [](Probe& p, const auto& expect) {
172 https_request r;
173 r.url = p.unreachable_url();
174 const auto got = p.client().send(r);
175 expect(got.is_err() && (got.unwrap_err().code == capability_errc::unavailable ||
176 got.unwrap_err().code == capability_errc::timeout),
177 "unavailable (or timeout)");
178 });
179
180 detail::capability_check<Probe>(report, "the credential never reaches an error", [](Probe& p, const auto& expect) {
181 https_request r;
182 r.url = p.unreachable_url();
183 r.authorization = secret<std::string>{"Bearer " + std::string(capability_redaction_marker)};
184 r.body = std::string(capability_redaction_marker);
185 r.sensitive_body = true;
186 const auto got = p.client().send(r);
187 expect(got.is_err(), "the unreachable host fails");
188 if (got.is_err())
189 expect(!capability_error_leaks(got.unwrap_err(), capability_redaction_marker),
190 "neither the credential nor the sensitive body reaches the error");
191 });
192
193 detail::capability_check<Probe>(report, "a provider failure never leaks its response", [](Probe& p, const auto& expect) {
194 https_request r;
195 r.url = p.url("/v1/thing");
196 p.provider_fails(capability_errc::unavailable, capability_provider_response);
197 const auto got = p.client().send(r);
198 expect(got.is_err() && got.unwrap_err().code == capability_errc::unavailable, "the provider's failure is returned");
199 if (got.is_err())
200 expect(!capability_error_leaks(got.unwrap_err(), capability_redaction_marker),
201 "neither provider_code nor message carries the provider's response");
202 });

204 return report;
