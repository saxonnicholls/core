160 // the rules shared by a name and a prefix; a prefix may end in '/' or with an empty input
161 constexpr bool remote_path_ok(std::string_view s, bool prefix) noexcept
162 {
163 if (s.size() > max_remote_name_length || (!prefix && s.empty()))
164 return false;
165 if (s.empty())
166 return true;
167 if (s.front() == '/' || (!prefix && s.back() == '/'))
168 return false;
169 for (const char c : s) {
170 const auto u = static_cast<unsigned char>(c);
171 if (u < 0x20 || u == 0x7f || c == '\\')
172 return false;
173 }
174 std::size_t start = 0;
175 while (start <= s.size()) {
176 const std::size_t slash = s.find('/', start);
177 const std::size_t end = slash == std::string_view::npos ? s.size() : slash;
178 const std::string_view segment = s.substr(start, end - start);
179 const bool last = slash == std::string_view::npos;
180 if (segment.empty() && !(prefix && last))
181 return false; // "a//b"; a prefix's trailing '/' leaves an empty last segment
182 if (segment == "..")
183 return false;
184 if (segment == "." && !(prefix && last))
185 return false; // a prefix's last segment is the start of a name: ".git" begins with "."
186 if (last)
187 break;
188 start = slash + 1;
189 }
190 return true;
191 }
192 } // namespace detail
