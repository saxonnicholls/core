64 [[nodiscard]] constexpr bool is_partition_of(std::size_t n, std::span<const coalition> groups) noexcept
65 {
66 coalition seen = 0;
67 for (const coalition g : groups) {
68 if (g == 0 || (g & seen) != 0)
69 return false;
70 seen |= g;
71 }
72 return seen == grand_coalition(n);
73 }

77 template <game V, typename Values>
78 constexpr void owen_into(std::size_t n, const V& v, std::span<const coalition> groups, attribution<game_value_t<V>, Values>& out)
79 {
80 using T = game_value_t<V>;
81 using scalar = game_scalar_t<T>;
82 assert(is_partition_of(n, groups) && "snicholls::game::owen: the groups do not partition the players");
83 out.baseline = std::invoke(v, coalition{0});
84 out.total = std::invoke(v, grand_coalition(n));
85 const std::size_t m = groups.size();
86 std::vector<T> inner;
87 std::vector<std::size_t> members;
88 for (std::size_t k = 0; k < m; ++k) {
89 members.clear();
90 for (coalition g = groups[k]; g != 0; g &= g - 1)
91 members.push_back(static_cast<std::size_t>(std::countr_zero(g)));
92 const std::size_t b = members.size();
93 inner.assign(table_size(b), T{});
94 const coalition other_groups = grand_coalition(m) & ~singleton(k);
95 for_each_subset_of(other_groups, [&](coalition r) {
96 coalition union_of_r = 0;
97 for (coalition rs = r; rs != 0; rs &= rs - 1)
98 union_of_r |= groups[static_cast<std::size_t>(std::countr_zero(rs))];
99 // Every subset of the group, on top of the other groups in r, once
100 for (coalition t = 0; t < table_size(b); ++t) {
101 coalition s = union_of_r;
102 for (coalition ts = t; ts != 0; ts &= ts - 1)
103 s = with(s, members[static_cast<std::size_t>(std::countr_zero(ts))]);
104 inner[t] = std::invoke(v, s);
105 }
106 const scalar outer_weight = shapley_weight<T>(m, coalition_size(r));
107 for (std::size_t j = 0; j < b; ++j)
108 for (coalition t = 0; t < table_size(b); ++t) {
109 if (contains(t, j))
110 continue;
111 const scalar w = outer_weight * shapley_weight<T>(b, coalition_size(t));
112 out.values[members[j]] = out.values[members[j]] + (inner[with(t, j)] - inner[t]) * w;
113 }
114 });
115 }
116 }

129 [[nodiscard]] constexpr fixed_attribution<game_value_t<V>, N> owen(const V& v, std::span<const coalition> groups)
130 {
131 static_assert(N <= 64, "snicholls::game::owen: at most 64 players");
132 fixed_attribution<game_value_t<V>, N> out{};
133 detail::owen_into(N, v, groups, out);
