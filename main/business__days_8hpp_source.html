96constexpr std::chrono::sys_days add_business_days(std::chrono::sys_days d, std::int64_t n) noexcept
97{
98 using std::chrono::days;
99 if (n == 0) return d;
100 const unsigned iso = std::chrono::weekday(d).iso_encoding(); // Monday 1 .. Sunday 7
101 if (n > 0) {
102 if (iso >= 6) d -= days(iso - 5); // weekend: count from the Friday before
103 const auto w = static_cast<std::int64_t>(std::chrono::weekday(d).iso_encoding() - 1); // 0..4
104 const std::int64_t rem = n % 5;
105 return d + days(n / 5 * 7 + rem + (w + rem > 4 ? 2 : 0));
106 }
107 if (iso >= 6) d += days(8 - iso); // weekend: count from the Monday after
108 const auto w = static_cast<std::int64_t>(std::chrono::weekday(d).iso_encoding() - 1);
109 const std::int64_t m = -n;
110 const std::int64_t rem = m % 5;
111 return d - days(m / 5 * 7 + rem + (w - rem < 0 ? 2 : 0));
112}

121constexpr std::int64_t business_days_between(std::chrono::sys_days a, std::chrono::sys_days b) noexcept
122{
123 // f(x) = business days in [M, x) for a fixed Monday M (1970-01-05): 5 per whole week + min(rest, 5).
124 constexpr std::int64_t monday = 4;
125 const auto f = [](std::int64_t x) noexcept {
126 const std::int64_t k = x - monday;
127 const std::int64_t q = k >= 0 ? k / 7 : -((-k + 6) / 7);
128 const std::int64_t r = k - q * 7;
129 return q * 5 + (r < 5 ? r : 5);
130 };
131 return f(b.time_since_epoch().count()) - f(a.time_since_epoch().count());
132}
