723template <typename Remaining, template <typename, typename> class Depends>
724struct topo_impl;
725template <template <typename, typename> class Depends>
726struct topo_impl<type_list<>, Depends>
727{
728 using type = type_list<>;
729};
730template <typename... Ts, template <typename, typename> class Depends>
731struct topo_impl<type_list<Ts...>, Depends>
732{
733 template <typename Node>
734 static constexpr bool blocked = (false || ... || (!std::is_same_v<Node, Ts> && static_cast<bool>(Depends<Node, Ts>::value)));
735
736 static consteval std::size_t first_free() noexcept
737 {
738 constexpr bool free_nodes[]{!blocked<Ts>...};
739 for (std::size_t i = 0; i < sizeof...(Ts); ++i) {
740 if (free_nodes[i]) {
741 return i;
742 }
743 }
744 return npos;
745 }
746
747 static constexpr std::size_t pick = first_free();
748 static_assert(pick != npos, "meta::topological_sort_t: the dependency graph has a cycle");
749 using list = type_list<Ts...>;
750 using type = push_front_t<typename topo_impl<erase_at_t<list, pick>, Depends>::type,
751 at_t<list, (pick == npos ? 0 : pick)>>;
752};

896template <int M, int N, int S = 1>
897struct thread_range
898{
899 static_assert(S != 0, "snicholls::meta::thread_range: the step must not be zero");
900
904 static constexpr bool valid(int total) noexcept
905 {
906 return M != 0 && N != 0 && (M < 0 ? -M : M) <= total && (N < 0 ? -N : N) <= total;
907 }
908
912 static constexpr bool selects(int position, int total) noexcept
913 {
914 const int first = M < 0 ? total + 1 + M : M;
915 const int last = N < 0 ? total + 1 + N : N;
916 const int distance = S > 0 ? position - first : first - position;
917 const int length = S > 0 ? last - first : first - last;
918 return distance >= 0 && distance <= length && distance % (S > 0 ? S : -S) == 0;
919 }
