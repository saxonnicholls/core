304 static T* next(T* n) noexcept
305 {
306 if (T* r = hk(n).right) {
307 while (T* l = hk(r).left) r = l;
308 return r;
309 }
310 T* p = hk(n).parent;
311 while (p && hk(p).right == n) {
312 n = p;
313 p = hk(p).parent;
314 }
315 return p;
316 }

322 {

388 std::pair<T*, bool> insert(T& node) noexcept
389 {
390 const auto& k = Traits::key(node);
391 T* p = root_;
392 bool go_left = false;
393 while (p) {
394 const auto& pk = Traits::key(*p);
395 if (Traits::less(k, pk)) {
396 go_left = true;
397 if (!hk(p).left) break;
398 p = hk(p).left;
399 } else if (Traits::less(pk, k)) {
400 go_left = false;
401 if (!hk(p).right) break;
402 p = hk(p).right;
403 } else {
404 return {p, false}; // equal key: nothing touched (not even node's hook)
405 }
406 }
407 auto& nh = hk(&node);
408 nh.left = nh.right = nullptr;
409 nh.parent = p;
410 nh.height = 1;
411 ++size_;
412 if (!p) {
413 root_ = first_ = last_ = &node;
414 return {&node, true};
415 }
416 (go_left ? hk(p).left : hk(p).right) = &node;
417 if (go_left && p == first_) first_ = &node;
418 if (!go_left && p == last_) last_ = &node;
419 retrace_insert(p);
420 return {&node, true};
421 }

426 void erase(T& node) noexcept
427 {
428 T* n = &node;
429 if (n == first_) first_ = next(n);
430 if (n == last_) last_ = prev(n);
431 auto& nh = hk(n);
432 T* retrace_from;
433 if (nh.left && nh.right) {
434 // Replace n by its successor s (leftmost of the right subtree), structurally.
435 T* s = nh.right;
436 while (hk(s).left) s = hk(s).left;
437 auto& sh = hk(s);
438 T* s_right = sh.right;
439 if (sh.parent == n) {
440 retrace_from = s;
441 } else {
442 retrace_from = sh.parent;
443 hk(sh.parent).left = s_right;
444 if (s_right) hk(s_right).parent = sh.parent;
445 sh.right = nh.right;
446 hk(nh.right).parent = s;
447 }
448 sh.left = nh.left;
449 hk(nh.left).parent = s;
450 sh.parent = nh.parent;
451 sh.height = nh.height;
452 replace_child(nh.parent, n, s);
453 } else {
454 T* child = nh.left ? nh.left : nh.right;
455 if (child) hk(child).parent = nh.parent;
456 replace_child(nh.parent, n, child);
457 retrace_from = nh.parent;
458 }
459 nh.left = nh.right = nh.parent = nullptr;
460 nh.height = 0;
461 --size_;
462 retrace_erase(retrace_from);
463 }

473 template <class Dispose>
474 void clear_and_dispose(Dispose&& dispose) noexcept
475 {
476 T* n = root_;
477 while (n) {
478 auto& nh = hk(n);
479 if (nh.left) n = nh.left;
480 else if (nh.right) n = nh.right;
481 else {
482 T* p = nh.parent;
483 if (p) (hk(p).left == n ? hk(p).left : hk(p).right) = nullptr;
484 nh.parent = nullptr;
485 nh.height = 0;
486 dispose(*n);
487 n = p;
488 }

496 [[nodiscard]] bool validate() const noexcept
497 {
498 if (!root_) return size_ == 0 && !first_ && !last_;
499 if (hk(root_).parent) return false;
500 size_type count = 0;
501 if (check(root_, count) < 0) return false;
502 if (count != size_) return false;
503 T* lo = root_;
504 while (hk(lo).left) lo = hk(lo).left;
505 T* hi = root_;
506 while (hk(hi).right) hi = hk(hi).right;
507 if (lo != first_ || hi != last_) return false;
508 // In-order traversal is strictly increasing.
509 for (T* a = first_; a; ) {
510 T* b = next(a);
511 if (b && !Traits::less(Traits::key(*a), Traits::key(*b))) return false;
512 a = b;
513 }
514 return true;
515 }

528 void replace_child(T* parent, T* old_child, T* new_child) noexcept
529 {

535 // Rotations return the new subtree root; heights of both moved nodes are recomputed.
536 T* rotate_left(T* x) noexcept
537 {
538 T* y = hk(x).right;
539 T* b = hk(y).left;
540 hk(x).right = b;
541 if (b) hk(b).parent = x;
542 hk(y).parent = hk(x).parent;
543 replace_child(hk(x).parent, x, y);
544 hk(y).left = x;
545 hk(x).parent = y;
546 update_height(x);
547 update_height(y);
548 return y;
549 }
550
551 T* rotate_right(T* x) noexcept
552 {
553 T* y = hk(x).left;
554 T* b = hk(y).right;
555 hk(x).left = b;
556 if (b) hk(b).parent = x;
557 hk(y).parent = hk(x).parent;
558 replace_child(hk(x).parent, x, y);
559 hk(y).right = x;
560 hk(x).parent = y;
561 update_height(x);
562 update_height(y);
563 return y;
564 }

566 // Restore balance at n (whose children are balanced); returns the subtree's new root.
567 T* rebalance(T* n) noexcept
568 {
569 const int bf = h(hk(n).left) - h(hk(n).right);
570 if (bf > 1) {
571 T* l = hk(n).left;
572 if (h(hk(l).left) < h(hk(l).right)) rotate_left(l);
573 return rotate_right(n);
574 }
575 if (bf < -1) {
576 T* r = hk(n).right;
577 if (h(hk(r).right) < h(hk(r).left)) rotate_right(r);
578 return rotate_left(n);
579 }
580 update_height(n);
581 return n;
582 }

584 void retrace_insert(T* n) noexcept
585 {
586 while (n) {
587 const int old = hk(n).height;
588 T* top = rebalance(n);
589 if (hk(top).height == old) return; // subtree height unchanged: ancestors unaffected
590 n = hk(top).parent;
591 }
592 }

594 void retrace_erase(T* n) noexcept
595 {
596 while (n) {
597 const int old = hk(n).height;
598 T* top = rebalance(n);
599 if (hk(top).height == old) return; // subtree height unchanged: ancestors unaffected
600 n = hk(top).parent;
601 }
602 }

604 // Returns the subtree height, or -1 on any violation.
605 int check(const T* n, size_type& count) const noexcept
606 {
607 if (!n) return 0;
608 ++count;
609 const auto& nh = hk(n);
610 if (nh.left && (hk(nh.left).parent != n || !Traits::less(Traits::key(*nh.left), Traits::key(*n)))) return -1;
611 if (nh.right && (hk(nh.right).parent != n || !Traits::less(Traits::key(*n), Traits::key(*nh.right)))) return -1;
612 const int l = check(nh.left, count);
613 if (l < 0) return -1;
614 const int r = check(nh.right, count);
615 if (r < 0) return -1;
616 if (l - r > 1 || r - l > 1) return -1;
617 const int height = 1 + (l > r ? l : r);
618 return height == nh.height ? height : -1;
619 }
