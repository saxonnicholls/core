139 template <typename T>
140 constexpr T ls_abs(T v) noexcept
141 {
142 return v < T(0) ? -v : v;
143 }
144 // A point on the line: the step, φ and φ'
145 template <typename T>
146 struct ls_point {
147 T a, f, d;
148 };
149 // The minimiser of the cubic interpolating φ, φ' at p and q (Nocedal and Wright 3.59), else that of the
150 // quadratic through φ(p), φ'(p), φ(q), else the midpoint; kept a tenth of the interval from either end
151 template <typename T>
152 constexpr T ls_interpolate(const ls_point<T>& p, const ls_point<T>& q) noexcept
153 {
154 const T width = ls_abs(q.a - p.a);
155 const T left = (p.a < q.a ? p.a : q.a) + width / T(10), right = (p.a < q.a ? q.a : p.a) - width / T(10);
156 const auto inside = [&](T a) { return a >= left && a <= right; };
157 const T d1 = p.d + q.d - T(3) * (p.f - q.f) / (p.a - q.a);
158 const T disc = d1 * d1 - p.d * q.d;
159 if (disc >= T(0)) {
160 const T root = static_cast<T>(math::sqrt(disc));
161 const T d2 = q.a > p.a ? root : -root;
162 const T den = q.d - p.d + T(2) * d2;
163 if (den != T(0)) {
164 const T a = q.a - (q.a - p.a) * (q.d + d2 - d1) / den;
165 if (inside(a))
166 return a;
167 }
168 }
169 const T h = q.a - p.a;
170 const T curv = T(2) * (q.f - p.f - p.d * h);
171 if (curv > T(0)) {
172 const T a = p.a - p.d * h * h / curv;
173 if (inside(a))
174 return a;
175 }
176 return (p.a + q.a) / T(2);
177 }
178 } // namespace detail

192 template <typename F, linalg::scalar T>
193 [[nodiscard]] constexpr armijo_result<T> backtracking_armijo(F&& f, std::type_identity_t<std::span<const T>> x, std::type_identity_t<std::span<const T>> direction, T value,
194 T gradient_dot_direction, std::type_identity_t<std::span<T>> trial,
195 const std::type_identity_t<armijo_options<T>>& options = {}) noexcept(std::is_nothrow_invocable_v<F&, std::span<const T>> &&
196 linalg::detail::nothrow_scalar<T>)
197 {
198 if (!(gradient_dot_direction < T(0)))
199 return {line_search_status::not_descent_direction, T(0), value, 0};
200 T alpha = options.initial_step, phi = value;
201 for (std::size_t ev = 1; ev <= options.max_evaluations; ++ev) {
202 for (std::size_t i = 0; i < x.size(); ++i)
203 trial[i] = x[i] + alpha * direction[i];
204 phi = f(std::span<const T>(trial));
205 if (phi <= value + options.c1 * alpha * gradient_dot_direction)
206 return {line_search_status::satisfied, alpha, phi, ev};
207 if (ev < options.max_evaluations)
208 alpha *= options.shrink;
209 }
210 return {line_search_status::evaluation_limit, options.max_evaluations > 0 ? alpha : T(0), phi, options.max_evaluations};
211 }

227 template <typename F, linalg::scalar T>
228 [[nodiscard]] constexpr wolfe_result<T> strong_wolfe(F&& fg, std::type_identity_t<std::span<const T>> x, std::type_identity_t<std::span<const T>> direction, T value,
229 T gradient_dot_direction, std::type_identity_t<std::span<T>> trial, std::type_identity_t<std::span<T>> trial_gradient,
230 const std::type_identity_t<wolfe_options<T>>& options = {}) noexcept(std::is_nothrow_invocable_v<F&, std::span<const T>, std::span<T>> &&
231 linalg::detail::nothrow_scalar<T>)
232 {
233 using point = detail::ls_point<T>;
234 const T d0 = gradient_dot_direction;
235 if (!(d0 < T(0)))
236 return {line_search_status::not_descent_direction, T(0), value, d0, 0};
237 std::size_t evaluations = 0;
238 T last = T(-1); // the step whose point trial holds
239 const auto eval = [&](T a) -> point {
240 for (std::size_t i = 0; i < x.size(); ++i)
241 trial[i] = x[i] + a * direction[i];
242 const T phi = fg(std::span<const T>(trial), trial_gradient);
243 T dphi(0);
244 for (std::size_t i = 0; i < x.size(); ++i)
245 dphi += trial_gradient[i] * direction[i];
246 ++evaluations;
247 last = a;
248 return {a, phi, dphi};
249 };
250 const auto decreases = [&](const point& p) { return p.f <= value + options.c1 * p.a * d0; };
251 const auto curved = [&](const point& p) { return detail::ls_abs(p.d) <= -options.c2 * d0; };
252 // A failure returns the best point with sufficient decrease, re-evaluated if trial holds another
253 const auto fail = [&](line_search_status s, const point& lo) -> wolfe_result<T> {
254 if (last != lo.a)
255 eval(lo.a);
256 return {s, lo.a, lo.f, lo.d, evaluations};
257 };
258 // Algorithm 3.6: lo has sufficient decrease and the lowest φ seen; φ'(lo) (hi - lo) < 0
259 const auto zoom = [&](point lo, point hi) -> wolfe_result<T> {
260 for (;;) {
261 if (evaluations >= options.max_evaluations)
262 return fail(line_search_status::evaluation_limit, lo);
263 const T aj = detail::ls_interpolate(lo, hi);
264 if (!(aj != lo.a && aj != hi.a))
265 return fail(line_search_status::interval_collapsed, lo);
266 const point pj = eval(aj);
267 if (!decreases(pj) || pj.f >= lo.f)
268 hi = pj;
269 else {
270 if (curved(pj))
271 return {line_search_status::satisfied, pj.a, pj.f, pj.d, evaluations};
272 if (pj.d * (hi.a - lo.a) >= T(0))
273 hi = lo;
274 lo = pj;
275 }
276 }
277 };
278 // Algorithm 3.5: expand until a bracket is found or the conditions hold
279 point prev{T(0), value, d0};
280 T a = options.initial_step < options.max_step ? options.initial_step : options.max_step;
281 for (std::size_t i = 1;; ++i) {
282 if (evaluations >= options.max_evaluations)
283 return fail(line_search_status::evaluation_limit, prev);
284 const point p = eval(a);
285 if (!decreases(p) || (i > 1 && p.f >= prev.f))
286 return zoom(prev, p);
287 if (curved(p))
288 return {line_search_status::satisfied, p.a, p.f, p.d, evaluations};
289 if (p.d >= T(0))
290 return zoom(p, prev);
291 if (a >= options.max_step)
292 return {line_search_status::step_at_maximum, p.a, p.f, p.d, evaluations};
293 prev = p;
294 a = a * options.expansion < options.max_step ? a * options.expansion : options.max_step;
295 }
