90 [[nodiscard]] constexpr bool is_key_id(std::string_view id) noexcept
91 {
92 if (id.empty() || id.size() > max_key_id_length)
93 return false;
94 for (const char c : id)
95 if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_' || c == '-' ||
96 c == '.' || c == ':'))
97 return false;
98 return true;
99 }

105 [[nodiscard]] inline capability_result<std::string_view> sealed_key_id(std::span<const std::byte> sealed) noexcept
106 {
107 if (sealed.size() < 2 || sealed[0] != sealed_format_v1)
108 return fail(capability_errc::invalid_argument, {}, "not a sealed envelope");
109 const auto n = static_cast<std::size_t>(sealed[1]);
110 if (n == 0 || n > max_key_id_length || sealed.size() < 2 + n)
111 return fail(capability_errc::invalid_argument, {}, "not a sealed envelope");
112 char text[max_key_id_length]{};
113 for (std::size_t i = 0; i < n; ++i)
114 text[i] = static_cast<char>(sealed[2 + i]);
115 if (!is_key_id({text, n}))
116 return fail(capability_errc::invalid_argument, {}, "the envelope's key id is not one");
117 // the id is ASCII, so the envelope's bytes are its characters
118 return ok(std::string_view(reinterpret_cast<const char*>(sealed.data() + 2), n));
119 }

126 [[nodiscard]] inline std::vector<std::byte> sealed_header(std::string_view key_id)
127 {
128 std::vector<std::byte> out(2 + key_id.size()); // sized once: GCC 15 misreads push_back after reserve
129 out[0] = sealed_format_v1;
130 out[1] = static_cast<std::byte>(key_id.size());
131 for (std::size_t i = 0; i < key_id.size(); ++i)
132 out[2 + i] = static_cast<std::byte>(key_id[i]);
133 return out;
134 }
