358 [[nodiscard]] constexpr inplace_string<101> dedup_key() const noexcept
359 {
360 std::array<char, 101> text{};
361 std::size_t n = 0;
362 for (const std::string_view part : {source.view(), std::string_view{":evt:"}, event_id.view()})
363 for (const char c : part)
364 text[n++] = c;
365 return *inplace_string<101>::make({text.data(), n});

453 [[nodiscard]] constexpr capability_result<void> validate(const payment_fact& f) noexcept
454 {
455 if (f.source.empty() || f.fact_id.empty())
456 return fail(capability_errc::invalid_argument, {}, "a fact needs its source and fact id");
457 if (!balances(f.settled))
458 return fail(capability_errc::invalid_argument, {}, "gross is not net plus fee in one asset");
459 if (f.presentment && *f.presentment == f.settled.gross)
460 return fail(capability_errc::invalid_argument, {}, "a presentment only when it differs");
461 if (f.presentment.has_value() != f.exchange_rate.has_value())
462 return fail(capability_errc::invalid_argument, {}, "a presentment comes with its exchange rate");
463 return ok();
464 }

507 concept payment_provider = requires(P& p, const P& cp, const collection_request& c, const payout_request& o,
508 const refund_request& r, const quote_request& q, std::string_view id,
509 const credit_query& cq, const webhook_delivery& w, const webhook_event& e,
