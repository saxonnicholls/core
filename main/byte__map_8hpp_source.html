136 // Byte i of p as an unsigned value; constexpr, unlike a load through memcpy or reinterpret_cast
137 constexpr std::uint64_t byte_map_byte(const char* p, std::size_t i) noexcept
138 {
139 return static_cast<unsigned char>(p[i]);
140 }
141
142 // Little-endian loads assembled with shifts: the same word on every machine, and folded to one load by GCC
143 // and Clang on a little-endian target
144 constexpr std::uint64_t byte_map_load4(const char* p) noexcept
145 {
146 return byte_map_byte(p, 0) | byte_map_byte(p, 1) << 8 | byte_map_byte(p, 2) << 16 | byte_map_byte(p, 3) << 24;
147 }
148 constexpr std::uint64_t byte_map_load8(const char* p) noexcept
149 {
150 return byte_map_load4(p) | byte_map_load4(p + 4) << 32;
151 }
152
153 // The first min(n, 8) bytes as a word, zero above them, reading only [p, p + n): from 4 to 7 bytes two
154 // overlapping loads agree on the bytes they share, so OR is exact; below 4, the first, middle and last byte
155 constexpr std::uint64_t byte_map_head(const char* p, std::size_t n) noexcept
156 {
157 if (n >= 8)
158 return byte_map_load8(p);
159 if (n >= 4)
160 return byte_map_load4(p) | byte_map_load4(p + n - 4) << (8 * (n - 4));
161 if (n == 0)
162 return 0;
163 return byte_map_byte(p, 0) | byte_map_byte(p, n >> 1) << (8 * (n >> 1)) | byte_map_byte(p, n - 1) << (8 * (n - 1));
164 }
165
166 // Keys of up to 16 bytes hash with one or two wymix rounds over words already loaded; longer keys by wyhash
167 constexpr std::uint64_t byte_map_hash(const char* p, std::size_t n, std::uint64_t head) noexcept
168 {
169 if (n <= 8)
170 return wymix(head ^ 0xa0761d6478bd642full, n ^ 0xe7037ed1a0b428dbull);
171 if (n <= 16)
172 return wymix(head ^ 0x8ebc6af09c88c6e3ull, byte_map_load8(p + n - 8) ^ (n * 0x589965cc75374cc3ull));
173 return wyhash(std::string_view(p, n));
174 }

189 // The three spans of a table, as plain pointers: the view and the owning map share every operation
190 struct byte_map_parts {
191 byte_map_slot* slots;
192 std::size_t capacity; // a power of two, or 0
193 char* arena;
194 std::size_t arena_capacity;
195 byte_map_key_ref* by_value;
196 std::size_t values;
197 };
198
199 constexpr void byte_map_clear(const byte_map_parts& t) noexcept
200 {
201 for (std::size_t i = 0; i < t.capacity; ++i)
202 t.slots[i] = byte_map_slot{};
203 for (std::size_t i = 0; i < t.values; ++i)
204 t.by_value[i] = byte_map_key_ref{0, byte_map_no_value};
205 }
206
207 constexpr byte_map_probe byte_map_probe_of(std::string_view key) noexcept
208 {
209 const std::uint64_t head = byte_map_head(key.data(), key.size());
210 return {head, byte_map_hash(key.data(), key.size(), head)};
211 }
212
213 // The index of key's slot if it is there, else of the empty slot that ends its probe; capacity > 0
214 constexpr std::size_t byte_map_slot_of(const byte_map_slot* slots, std::size_t capacity, const char* arena,
215 std::string_view key, byte_map_probe pr) noexcept
216 {
217 const std::size_t mask = capacity - 1, n = key.size();
218 for (std::size_t i = pr.hash & mask;; i = (i + 1) & mask) {
219 const byte_map_slot& e = slots[i];
220 if (e.value == byte_map_no_value)
221 return i;
222 if (e.head == pr.head && (e.off_len & 0xffu) == n &&
223 (n <= 8 || std::string_view(arena + (e.off_len >> 8) + 8, n - 8) == std::string_view(key.data() + 8, n - 8)))
224 return i;
225 }
226 }

228 constexpr result<void, byte_map_error> byte_map_insert(const byte_map_parts& t, std::size_t& size, std::size_t& arena_used,
229 std::string_view key, std::uint32_t value) noexcept
230 {
231 if (key.size() > 255)
232 return err(byte_map_error::key_too_long);
233 if (value == byte_map_no_value || (t.values != 0 && value >= t.values))
234 return err(byte_map_error::bad_value);
235 if (size + 1 > t.capacity / 2)
236 return err(byte_map_error::too_full);
237 const byte_map_probe pr = byte_map_probe_of(key);
238 const std::size_t i = byte_map_slot_of(t.slots, t.capacity, t.arena, key, pr);
239 if (t.slots[i].value != byte_map_no_value || (t.values != 0 && t.by_value[value].second != byte_map_no_value))
240 return err(byte_map_error::duplicate);
241 if (key.size() > t.arena_capacity - arena_used || arena_used + key.size() > (std::size_t{1} << 24))
242 return err(byte_map_error::arena_full);
243 const auto off = static_cast<std::uint32_t>(arena_used);
244 for (std::size_t k = 0; k < key.size(); ++k)
245 t.arena[arena_used + k] = key[k];
246 arena_used += key.size();
247 t.slots[i] = byte_map_slot{pr.head, value, off << 8 | static_cast<std::uint32_t>(key.size())};
248 if (t.values != 0)
249 t.by_value[value] = byte_map_key_ref{off, static_cast<std::uint32_t>(key.size())};
250 ++size;
251 return ok();
252 }

254 // The lookups, written once for both kinds of map: Derived gives parts() and the counts
255 template <typename Derived>
256 class byte_map_lookup {
257 public:
261 [[nodiscard]] constexpr byte_map_probe probe(std::string_view key) const noexcept
262 {
263 const byte_map_probe pr = byte_map_probe_of(key);
264#if defined(__GNUC__) || defined(__clang__)
265 const detail::byte_map_parts t = self().parts();
266 if (!std::is_constant_evaluated() && t.capacity != 0)
267 __builtin_prefetch(t.slots + (pr.hash & (t.capacity - 1)));
268#endif
269 return pr;
270 }
271
276 [[nodiscard]] constexpr std::uint32_t find(std::string_view key, byte_map_probe pr) const noexcept
277 {
278 const detail::byte_map_parts t = self().parts();
279 if (t.capacity == 0)
280 return byte_map_no_value;
281 return t.slots[byte_map_slot_of(t.slots, t.capacity, t.arena, key, pr)].value;
282 }
283
288 [[nodiscard]] constexpr std::uint32_t operator()(std::string_view key) const noexcept
289 {
290 return find(key, byte_map_probe_of(key));
291 }

296 [[nodiscard]] constexpr bool contains(std::string_view key) const noexcept { return (*this)(key) != byte_map_no_value; }
297
301 [[nodiscard]] constexpr std::string_view key_of(std::uint32_t value) const noexcept
302 {
303 const detail::byte_map_parts t = self().parts();
304 if (value >= t.values || t.by_value[value].second == byte_map_no_value)
305 return {};
306 return std::string_view(t.arena + t.by_value[value].first, t.by_value[value].second);
307 }

311 [[nodiscard]] constexpr std::size_t size() const noexcept { return self().size_; }
