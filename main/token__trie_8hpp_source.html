91 // Heap sort written out, O(n log n) and constexpr: GCC's -Warray-bounds misreads both std::sort's insertion
92 // pass and std::make_heap over a fixed array of a few elements
93 template <typename Less>
94 constexpr void sift_down(std::uint32_t* v, std::size_t i, std::size_t n, Less& less)
95 {
96 for (std::size_t c = 2 * i + 1; c < n; i = c, c = 2 * i + 1) {
97 if (c + 1 < n && less(v[c], v[c + 1]))
98 ++c;
99 if (!less(v[i], v[c]))
100 return;
101 const std::uint32_t t = v[i];
102 v[i] = v[c];
103 v[c] = t;
104 }
105 }
106
107 template <typename Less>
108 constexpr void sort(std::span<std::uint32_t> span, Less& less)
109 {
110 const std::size_t n = span.size();
111 if (n < 2)
112 return;
113 std::uint32_t* v = span.data();
114 for (std::size_t i = n / 2; i-- > 0;)
115 sift_down(v, i, n, less);
116 for (std::size_t end = n - 1; end > 0; --end) {
117 const std::uint32_t t = v[0];
118 v[0] = v[end];
119 v[end] = t;
120 sift_down(v, 0, end, less);
121 }
122 }

132template <typename Keep>
133constexpr std::size_t build_token_index(std::span<const std::string_view> pieces, std::span<std::uint32_t> order, Keep&& keep)
134{
135 assert(order.size() >= pieces.size());
136 std::size_t n = 0;
137 for (std::size_t id = 0; id < pieces.size(); ++id)
138 if (keep(static_cast<std::uint32_t>(id)))
139 order[n++] = static_cast<std::uint32_t>(id);
140 const auto before = [&](std::uint32_t a, std::uint32_t b) { return pieces[a] < pieces[b] || (pieces[a] == pieces[b] && a > b); };
141 detail::vocab_index::sort(order.first(n), before);
142 return n;
143}

162 struct cursor
163 {
164 std::uint32_t lo;
165 std::uint32_t hi;
166 std::uint32_t depth;
167
170 constexpr bool empty() const noexcept { return lo >= hi; }
171 };
172
175 constexpr cursor root() const noexcept { return {0, static_cast<std::uint32_t>(order.size()), 0}; }
176
181 constexpr bool advance(cursor& c, char byte) const noexcept
182 {
183 const auto b = static_cast<unsigned char>(byte);
184 // key(p) = p[depth], or -1 when p ends at depth; non-decreasing over the range
185 const auto key = [&](std::uint32_t i) -> int {
186 const std::string_view p = pieces[order[i]];
187 return p.size() > c.depth ? static_cast<int>(static_cast<unsigned char>(p[c.depth])) : -1;
188 };
189 std::uint32_t lo = c.lo, hi = c.hi;
190 while (lo < hi) { // first with key >= b
191 const std::uint32_t mid = lo + (hi - lo) / 2;
192 if (key(mid) < static_cast<int>(b))
193 lo = mid + 1;
194 else
195 hi = mid;
196 }
197 std::uint32_t end = lo, top = c.hi;
198 while (end < top) { // first with key > b
199 const std::uint32_t mid = end + (top - end) / 2;
200 if (key(mid) <= static_cast<int>(b))
201 end = mid + 1;
202 else
203 top = mid;
204 }
205 c = {lo, end, c.depth + 1};
206 return lo < end;
207 }
208
212 constexpr std::uint32_t at(const cursor& c) const noexcept
213 {
214 if (c.lo < c.hi && pieces[order[c.lo]].size() == c.depth)
215 return order[c.lo];
216 return no_token;
217 }
218
222 constexpr std::uint32_t find(std::string_view bytes) const noexcept
223 {
224 std::size_t lo = 0, hi = order.size();
225 while (lo < hi) {
226 const std::size_t mid = lo + (hi - lo) / 2;
227 if (pieces[order[mid]] < bytes)
228 lo = mid + 1;
229 else
230 hi = mid;
231 }
232 return lo < order.size() && pieces[order[lo]] == bytes ? order[lo] : no_token;
233 }
234
238 constexpr std::uint32_t operator()(std::string_view bytes) const noexcept { return find(bytes); }
239
244 constexpr std::uint32_t longest_prefix(std::string_view text, std::size_t& length) const noexcept
245 {
246 std::uint32_t best = no_token;
247 length = 0;
248 cursor c = root();
249 for (std::size_t i = 0; i < text.size() && advance(c, text[i]); ++i)
250 if (const std::uint32_t id = at(c); id != no_token) {
251 best = id;
252 length = i + 1;
253 }
254 return best;
255 }
256};
