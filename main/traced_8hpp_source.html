185template <class T>
186[[nodiscard]] std::string error_field_text(const T& value)
187{
188 if constexpr (traced_detail::masked_value<T>)
189 return std::string(masked_field_text);
190 else if constexpr (std::is_same_v<T, bool>)
191 return value ? "true" : "false";
192 else if constexpr (std::is_convertible_v<const T&, std::string_view>)
193 {
194 if constexpr (std::is_pointer_v<T>)
195 if (value == nullptr)
196 return std::string("(null)"); // std::string_view(nullptr) is undefined behaviour
197 return std::string(std::string_view(value));
198 }
199 else if constexpr (result_detail::std_formattable<T>)
200 return std::format("{}", value);
201 else
202 static_assert(snicholls::always_false_v<T>,
203 "snicholls::traced: a field value must be a secret, string-like or std::format-able");
204}

591[[nodiscard]] error_report make_error_report(const E& e, source_location log_site = source_location::current())
592{
593 error_report r;
594 if constexpr (is_traced_v<E>)
595 {
596 r.message = e.message();
597 r.error = error_text(e.error());
598 r.where = e.where();
599 r.located = true;
600 r.frames.assign(e.frames().begin(), e.frames().end());
601 r.fields.assign(e.fields().begin(), e.fields().end());
602 r.stack = std::string(e.stack());
603 }
604 else
605 {
606 r.message = error_text(e);
607 r.error = r.message;
608 r.where = log_site;
609 }
610 return r;
611}

639[[nodiscard]] inline std::string format_error_report(const error_report& r)
640{
641 // Every line of text is one line: a newline in a message, key or value is written as \n (and \r, \\ alike), so
642 // a value cannot forge a line of its own (" fields:", "[ERROR] ..."). Only the stack keeps its own lines.
643 const auto one_line = [](std::string_view s) {
644 std::string escaped;
645 escaped.reserve(s.size());
646 for (const char c : s)
647 {
648 if (c == '\n')
649 escaped += "\\n";
650 else if (c == '\r')
651 escaped += "\\r";
652 else if (c == '\\')
653 escaped += "\\\\";
654 else
655 escaped += c;
656 }
657 return escaped;
658 };
659 std::string out = "error: " + one_line(r.message);
660 out += r.located ? "\n at " : "\n logged at ";
661 out += source_site(r.where);
662 if (!r.frames.empty())
663 {
664 out += "\n context (innermost first):";
665 for (std::size_t i = 0; i < r.frames.size(); ++i)
666 {
667 const auto& f = r.frames[i];
668 out += "\n " + std::to_string(i) + ": ";
669 if (f.kind == frame_kind::propagated)
670 out += "propagated `" + one_line(f.message) + "`";
671 else
672 out += one_line(f.message);
673 out += " at " + source_site(f.where);
674 }
675 }
676 if (!r.fields.empty())
677 {
678 out += "\n fields:";
679 for (const auto& f : r.fields)
680 out += "\n " + one_line(f.key) + " = " + one_line(f.value);
681 }
682 if (!r.stack.empty())
683 {
684 out += "\n stack:";
685 std::string_view s = r.stack;
686 while (!s.empty())
687 {
688 const auto nl = s.find('\n');
689 out += "\n ";
690 out += s.substr(0, nl);
691 s = nl == std::string_view::npos ? std::string_view() : s.substr(nl + 1);
692 }
693 }
694 return out;
695}

718[[nodiscard]] inline std::vector<error_field> error_report_fields(const error_report& r)
719{
720 std::vector<error_field> out;
721 out.reserve(3 + r.frames.size() + r.fields.size());
722 out.push_back({"error", r.error});
723 out.push_back({"src", source_site(r.where)});
724 for (std::size_t i = 0; i < r.frames.size(); ++i)
725 {
726 const auto& f = r.frames[i];
727 std::string v = f.kind == frame_kind::propagated ? "propagated " + f.message : f.message;
728 v += " @ " + source_site(f.where);
729 out.push_back({"context." + std::to_string(i), std::move(v)});
730 }
731 for (const auto& f : r.fields)
732 out.push_back({is_reserved_error_field(f.key) ? "field." + f.key : f.key, f.value});
733 if (!r.stack.empty())
734 out.push_back({"stack", r.stack});
735 return out;
736}
