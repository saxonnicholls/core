183template <class T>
184[[nodiscard]] std::string error_field_text(const T& value)
185{
186 if constexpr (traced_detail::masked_value<T>)
187 return std::string(masked_field_text);
188 else if constexpr (std::is_same_v<T, bool>)
189 return value ? "true" : "false";
190 else if constexpr (std::is_convertible_v<const T&, std::string_view>)
191 {
192 if constexpr (std::is_pointer_v<T>)
193 if (value == nullptr)
194 return std::string("(null)"); // std::string_view(nullptr) is undefined behaviour
195 return std::string(std::string_view(value));
196 }
197 else if constexpr (result_detail::std_formattable<T>)
198 return std::format("{}", value);
199 else
200 static_assert(snicholls::always_false_v<T>,
201 "snicholls::traced: a field value must be a secret, string-like or std::format-able");
202}

590[[nodiscard]] error_report make_error_report(const E& e, source_location log_site = source_location::current())
591{
592 error_report r;
593 if constexpr (is_traced_v<E>)
594 {
595 r.message = e.message();
596 r.error = error_text(e.error());
597 r.where = e.where();
598 r.located = true;
599 r.frames.assign(e.frames().begin(), e.frames().end());
600 r.fields.assign(e.fields().begin(), e.fields().end());
601 r.stack = std::string(e.stack());
602 }
603 else
604 {
605 r.message = error_text(e);
606 r.error = r.message;
607 r.where = log_site;
608 }
609 return r;
610}

638[[nodiscard]] inline std::string format_error_report(const error_report& r)
639{
640 // Every line of text is one line: a newline in a message, key or value is written as \n (and \r, \\ alike), so
641 // a value cannot forge a line of its own (" fields:", "[ERROR] ..."). Only the stack keeps its own lines.
642 const auto one_line = [](std::string_view s) {
643 std::string escaped;
644 escaped.reserve(s.size());
645 for (const char c : s)
646 {
647 if (c == '\n')
648 escaped += "\\n";
649 else if (c == '\r')
650 escaped += "\\r";
651 else if (c == '\\')
652 escaped += "\\\\";
653 else
654 escaped += c;
655 }
656 return escaped;
657 };
658 std::string out = "error: " + one_line(r.message);
659 out += r.located ? "\n at " : "\n logged at ";
660 out += source_site(r.where);
661 if (!r.frames.empty())
662 {
663 out += "\n context (innermost first):";
664 for (std::size_t i = 0; i < r.frames.size(); ++i)
665 {
666 const auto& f = r.frames[i];
667 out += "\n " + std::to_string(i) + ": ";
668 if (f.kind == frame_kind::propagated)
669 out += "propagated `" + one_line(f.message) + "`";
670 else
671 out += one_line(f.message);
672 out += " at " + source_site(f.where);
673 }
674 }
675 if (!r.fields.empty())
676 {
677 out += "\n fields:";
678 for (const auto& f : r.fields)
679 out += "\n " + one_line(f.key) + " = " + one_line(f.value);
680 }
681 if (!r.stack.empty())
682 {
683 out += "\n stack:";
684 std::string_view s = r.stack;
685 while (!s.empty())
686 {
687 const auto nl = s.find('\n');
688 out += "\n ";
689 out += s.substr(0, nl);
690 s = nl == std::string_view::npos ? std::string_view() : s.substr(nl + 1);
691 }
692 }
693 return out;
694}

701[[nodiscard]] constexpr bool is_reserved_error_field(std::string_view key) noexcept
702{
703 return key == "error" || key == "src" || key == "stack" || key == "where" || key.starts_with("context.") ||
704 key.starts_with("field.");

717[[nodiscard]] inline std::vector<error_field> error_report_fields(const error_report& r)
718{
719 std::vector<error_field> out;
720 out.reserve(3 + r.frames.size() + r.fields.size());
721 out.push_back({"error", r.error});
722 out.push_back({"src", source_site(r.where)});
723 for (std::size_t i = 0; i < r.frames.size(); ++i)
724 {
725 const auto& f = r.frames[i];
726 std::string v = f.kind == frame_kind::propagated ? "propagated " + f.message : f.message;
727 v += " @ " + source_site(f.where);
728 out.push_back({"context." + std::to_string(i), std::move(v)});
729 }
730 for (const auto& f : r.fields)
731 out.push_back({is_reserved_error_field(f.key) ? "field." + f.key : f.key, f.value});
732 if (!r.stack.empty())
733 out.push_back({"stack", r.stack});
734 return out;
735}
