70 template <typename Probe>
71 concept signer_probe = std::is_default_constructible_v<Probe> &&
72 signer<std::remove_reference_t<decltype(std::declval<Probe&>().signer())>> &&
73 requires(Probe& p, const signature& s, std::span<const std::byte> b, capability_errc c, std::string_view r) {
74 { p.active_key() } -> std::convertible_to<std::string>;
75 { p.retired_key() } -> std::convertible_to<std::string>;
76 { p.verify(s, b) } -> std::same_as<bool>;
77 { p.deterministic() } -> std::same_as<bool>;
78 p.provider_fails(c, r);
79 };

88 [[nodiscard]] capability_conformance_report run_signer_conformance()
89 {
90 capability_conformance_report report;
91 using detail::kit_payload;
92
93 detail::capability_check<Probe>(report, "a signature verifies and names its key and version", [](Probe& p, const auto& expect) {
94 const std::string id = p.active_key();
95 const auto r = p.signer().sign({id, kit_payload(), payload_kind::message});
96 expect(r.is_ok(), "the active key signs");
97 if (r.is_ok()) {
98 expect(r->key_id == std::string_view{id}, "the signature names its key");
99 expect(r->key_version >= 1, "the signature names its version");
100 expect(!r->bytes.empty(), "the signature has bytes");
101 expect(p.verify(*r, kit_payload()), "the signature verifies");
102 }
103 });
104
105 detail::capability_check<Probe>(report, "an unknown key is not found", [](Probe& p, const auto& expect) {
106 const auto s = p.signer().sign({"no-such-key", kit_payload(), payload_kind::message});
107 expect(s.is_err() && s.unwrap_err().code == capability_errc::not_found, "sign with an unknown key is not_found");
108 const auto k = p.signer().public_key_of({"no-such-key", 0});
109 expect(k.is_err() && k.unwrap_err().code == capability_errc::not_found, "public_key_of an unknown key is not_found");
110 });
111
112 detail::capability_check<Probe>(report, "a retired key no longer signs", [](Probe& p, const auto& expect) {
113 const std::string id = p.retired_key();
114 const auto r = p.signer().sign({id, kit_payload(), payload_kind::message});
115 expect(r.is_err() && r.unwrap_err().code == capability_errc::conflict, "a retired key is a conflict");
116 });
117
118 detail::capability_check<Probe>(report, "a digest that is not 32 bytes is refused", [](Probe& p, const auto& expect) {
119 const std::string id = p.active_key();
120 const std::array<std::byte, 31> short_digest{};
121 const auto r = p.signer().sign({id, short_digest, payload_kind::digest});
122 expect(r.is_err() && (r.unwrap_err().code == capability_errc::invalid_argument ||
123 r.unwrap_err().code == capability_errc::unsupported),
124 "a 31-byte digest is invalid_argument (or unsupported, where digests are)");
125 });

127 detail::capability_check<Probe>(report, "a deterministic signer repeats itself", [](Probe& p, const auto& expect) {
128 if (!p.deterministic())
129 return;
130 const std::string id = p.active_key();
131 const auto a = p.signer().sign({id, kit_payload(), payload_kind::message});
132 const auto b = p.signer().sign({id, kit_payload(), payload_kind::message});
133 expect(a.is_ok() && b.is_ok() && *a == *b, "two signatures of one payload are equal");
134 });

136 detail::capability_check<Probe>(report, "public keys exist exactly for asymmetric algorithms", [](Probe& p, const auto& expect) {
137 const std::string id = p.active_key();
138 const auto s = p.signer().sign({id, kit_payload(), payload_kind::message});
139 expect(s.is_ok(), "the active key signs");
140 if (!s.is_ok())
141 return;
142 const auto k = p.signer().public_key_of({id, 0});
143 if (is_asymmetric(s->algorithm)) {
144 expect(k.is_ok(), "an asymmetric key publishes its public key");
145 if (k.is_ok())
146 expect(k->algorithm == s->algorithm && !k->bytes.empty(), "the public key matches the algorithm");
147 } else {
148 expect(k.is_err() && k.unwrap_err().code == capability_errc::unsupported,
149 "a symmetric key has no public key: unsupported");
150 }
151 });
152
153 detail::capability_check<Probe>(report, "a provider failure never leaks its response", [](Probe& p, const auto& expect) {
154 const std::string id = p.active_key();
155 p.provider_fails(capability_errc::unavailable, capability_provider_response);
156 const auto r = p.signer().sign({id, kit_payload(), payload_kind::message});
157 expect(r.is_err() && r.unwrap_err().code == capability_errc::unavailable, "the provider's failure is returned");
158 if (r.is_err())
159 expect(!capability_error_leaks(r.unwrap_err(), capability_redaction_marker),
160 "neither provider_code nor message carries the provider's response");
161 });
