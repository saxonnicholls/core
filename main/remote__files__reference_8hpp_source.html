112 capability_result<file_digest> fetch(std::string_view name, remote_file_sink sink)
113 {
114 if (auto valid = validate_remote_name(name); !valid)
115 return err(valid.unwrap_err());
116 if (auto injected = take_injected())
117 return err(*injected);
118 const auto it = files_.find(name);
119 if (it == files_.end())
120 return fail(capability_errc::not_found, {}, "no such file in the drop");
121 const std::span<const std::byte> all{it->second.bytes};
122 file_digester digest;
123 for (std::size_t at = 0; at < all.size(); at += chunk_) {
124 const auto chunk = all.subspan(at, std::min(chunk_, all.size() - at));
125 if (auto taken = sink(chunk); !taken)
126 return err(taken.unwrap_err());
127 digest.update(chunk);
128 }
129 return ok(digest.finish());
130 }

136 capability_result<void> move(std::string_view from, std::string_view to)
137 {
138 if (!movable_)
139 return fail(capability_errc::unsupported);
140 if (auto valid = validate_remote_name(from); !valid)
141 return valid;
142 if (auto valid = validate_remote_name(to); !valid)
143 return valid;
144 if (auto injected = take_injected())
145 return err(*injected);
146 const auto it = files_.find(from);
147 if (it == files_.end())
148 return fail(capability_errc::not_found, {}, "no such file in the drop");
149 if (files_.find(to) != files_.end())
150 return fail(capability_errc::conflict, {}, "the new name is taken");
151 std::string new_name(to); // allocate before anything changes
152 auto node = files_.extract(it);
153 node.key() = std::move(new_name);
154 files_.insert(std::move(node));
155 return ok();
156 }
