166[[nodiscard]] inline bool is_secret_key(std::string_view key)
167{
168 std::string upper(key);
169 std::transform(upper.begin(), upper.end(), upper.begin(), detail::ascii_upper);
170 constexpr std::string_view markers[] = {"PASSWORD", "PASSWD", "PASSPHRASE", "SECRET", "TOKEN", "CREDENTIAL",
171 "PRIVATE", "AUTH", "API_KEY", "APIKEY", "ACCESS_KEY", "COOKIE",
172 "SESSION", "SIGNATURE"};
173 for (const std::string_view marker : markers)
174 if (upper.find(marker) != std::string::npos)
175 return true;
176 return upper == "KEY" || upper.ends_with("_KEY") || upper.ends_with("_PWD");
177}

185[[nodiscard]] inline std::string mask_url_credentials(std::string_view value)
186{
187 const std::size_t scheme = value.find("://");
188 if (scheme == std::string_view::npos)
189 return std::string(value);
190 const std::size_t authority = scheme + 3;
191 const std::size_t end = value.find_first_of("/?#", authority);
192 const std::string_view host_part = value.substr(authority, end == std::string_view::npos ? value.npos : end - authority);
193 const std::size_t at = host_part.rfind('@');
194 if (at == std::string_view::npos)
195 return std::string(value);
196 const std::size_t colon = host_part.find(':');
197 if (colon == std::string_view::npos || colon > at)
198 return std::string(value);
199 std::string out(value.substr(0, authority + colon + 1));
200 out += masked_text;
201 out += value.substr(authority + at);
202 return out;
203}
