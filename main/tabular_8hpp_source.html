94 constexpr value operator()(coalition s) const
95 {
96 value sum{};
97 std::size_t count = 0;
98 for (const auto& background_row : *background_) {
99 assert(static_cast<std::size_t>(std::ranges::size(background_row)) == row_.size());
100 for (std::size_t j = 0; j < row_.size(); ++j)
101 row_[j] = contains(s, j) ? static_cast<feature>((*instance_)[j]) : static_cast<feature>(background_row[j]);
102 sum = sum + std::invoke(*model_, std::span<const feature>(row_));
103 ++count;
104 }
105 assert(count > 0 && "snicholls::game::interventional_game: the background is empty");
106 return sum * (game_scalar_t<value>(1) / static_cast<game_scalar_t<value>>(count));
107 }

111 const Instance* instance_;
112 const Background* background_;
