154 constexpr void append_utf8(std::string& out, std::uint32_t c)
155 {
156 if (c < 0x80) {
157 out += static_cast<char>(c);
158 } else if (c < 0x800) {
159 out += static_cast<char>(0xc0 | (c >> 6));
160 out += static_cast<char>(0x80 | (c & 0x3f));
161 } else if (c < 0x10000) {
162 out += static_cast<char>(0xe0 | (c >> 12));
163 out += static_cast<char>(0x80 | ((c >> 6) & 0x3f));
164 out += static_cast<char>(0x80 | (c & 0x3f));
165 } else {
166 out += static_cast<char>(0xf0 | (c >> 18));
167 out += static_cast<char>(0x80 | ((c >> 12) & 0x3f));

201 constexpr int hex(char c) const
202 {
203 return ('0' <= c && c <= '9') ? c - '0' : ('a' <= c && c <= 'f') ? c - 'a' + 10 : ('A' <= c && c <= 'F') ? c - 'A' + 10 : -1;

205 constexpr bool hex4(std::uint32_t& out)
206 {
207 out = 0;
208 for (int i = 0; i < 4; ++i, ++pos) {
209 const int h = hex(peek());
210 if (h < 0)
211 return fail("expecting four hexadecimal digits");
212 out = out * 16 + static_cast<std::uint32_t>(h);
213 }
214 return true;

216 constexpr bool string(std::string& out)
217 {
218 ++pos; // the opening quote
219 while (true) {
220 if (pos >= s.size())
221 return fail("unterminated string");
222 const char c = s[pos];
223 if (c == '"') {
224 ++pos;
225 return true;
226 }
227 if (static_cast<unsigned char>(c) < 0x20)
228 return fail("control character in a string");
229 if (c != '\\') {
230 out += c;
231 ++pos;
232 continue;
233 }
234 ++pos;
235 const char e = peek();
236 ++pos;
237 switch (e) {
238 case '"': out += '"'; break;
239 case '\\': out += '\\'; break;
240 case '/': out += '/'; break;
241 case 'b': out += '\b'; break;
242 case 'f': out += '\f'; break;
243 case 'n': out += '\n'; break;
244 case 'r': out += '\r'; break;
245 case 't': out += '\t'; break;
246 case 'u': {
247 std::uint32_t c1 = 0;
248 if (!hex4(c1))
249 return false;
250 if (c1 >= 0xd800 && c1 < 0xdc00) { // a surrogate pair
251 std::uint32_t c2 = 0;
252 if (peek() != '\\' || (pos + 1 < s.size() && s[pos + 1] != 'u'))
253 return fail("unpaired surrogate");
254 pos += 2;
255 if (!hex4(c2))
256 return false;
257 if (c2 < 0xdc00 || c2 > 0xdfff)
258 return fail("unpaired surrogate");
259 c1 = 0x10000 + ((c1 - 0xd800) << 10) + (c2 - 0xdc00);
260 } else if (c1 >= 0xdc00 && c1 <= 0xdfff) {
261 return fail("unpaired surrogate");
262 }
263 append_utf8(out, c1);
264 break;
265 }
266 default: --pos; return fail("invalid escape");
267 }
268 }

270 constexpr bool digits()
271 {
272 const std::size_t start = pos;
273 while ('0' <= peek() && peek() <= '9')
274 ++pos;
275 return pos != start || fail("expecting a digit");

277 constexpr bool number(value& v)
278 {
279 const std::size_t start = pos;
280 if (peek() == '-')
281 ++pos;
282 if (peek() == '0')
283 ++pos;
284 else if (!digits())
285 return false;
286 if (peek() == '.') {
287 ++pos;
288 if (!digits())
289 return false;
290 }
291 if (peek() == 'e' || peek() == 'E') {
292 ++pos;
293 if (peek() == '+' || peek() == '-')
294 ++pos;
295 if (!digits())
296 return false;
297 }
298 v.kind = json_kind::number;
299 v.text = std::string(s.substr(start, pos - start));
300 return true;
301 }
302 constexpr bool parse(value& v, int depth)
303 {
304 if (depth > 512)
305 return fail("JSON nested more than 512 deep");
306 space();
307 const char c = peek();
308 if (c == '{') {
309 v.kind = json_kind::object;
310 ++pos;
311 space();
312 if (peek() == '}') {
313 ++pos;
314 return true;
315 }
316 while (true) {
317 space();
318 if (peek() != '"')
319 return fail("expecting a key");
320 std::string key;
321 if (!string(key))
322 return false;
323 space();
324 if (peek() != ':')
325 return fail("expecting ':'");
326 ++pos;
327 value item;
328 if (!parse(item, depth + 1))
329 return false;
330 bool replaced = false; // a repeated key keeps its place and takes the last value
331 for (std::size_t i = 0; i < v.keys.size(); ++i)
332 if (v.keys[i] == key) {
333 v.items[i] = std::move(item);
334 replaced = true;
335 break;
336 }
337 if (!replaced) {
338 v.keys.push_back(std::move(key));
339 v.items.push_back(std::move(item));
340 }
341 space();
342 if (peek() == ',') {
343 ++pos;
344 continue;
345 }
346 if (peek() == '}') {
347 ++pos;
348 return true;
349 }
350 return fail("expecting ',' or '}'");
351 }
352 }
353 if (c == '[') {
354 v.kind = json_kind::array;
355 ++pos;
356 space();
357 if (peek() == ']') {
358 ++pos;
359 return true;
360 }
361 while (true) {
362 value item;
363 if (!parse(item, depth + 1))
364 return false;
365 v.items.push_back(std::move(item));
366 space();
367 if (peek() == ',') {
368 ++pos;
369 continue;
370 }
371 if (peek() == ']') {
372 ++pos;
373 return true;
374 }
375 return fail("expecting ',' or ']'");
376 }
377 }
378 if (c == '"') {
379 v.kind = json_kind::string;
380 return string(v.text);
381 }
382 if (c == 't') {
383 v.kind = json_kind::boolean;
384 v.boolean = true;
385 return word("true");
386 }
387 if (c == 'f') {
388 v.kind = json_kind::boolean;
389 return word("false");
390 }
391 if (c == 'n') {
392 v.kind = json_kind::null;
393 return word("null");
394 }
395 if (c == '-' || ('0' <= c && c <= '9'))
396 return number(v);
397 return fail("expecting a JSON value");
398 }

402 constexpr void dump_string(std::string& out, std::string_view s)
403 {
404 constexpr char digits[] = "0123456789abcdef";
405 out += '"';
406 for (const char c : s) {
407 switch (c) {
408 case '"': out += "\\\""; break;
409 case '\\': out += "\\\\"; break;
410 case '\b': out += "\\b"; break;
411 case '\f': out += "\\f"; break;
412 case '\n': out += "\\n"; break;
413 case '\r': out += "\\r"; break;
414 case '\t': out += "\\t"; break;
415 default:
416 if (static_cast<unsigned char>(c) < 0x20) {
417 out += "\\u00";
418 out += digits[(static_cast<unsigned char>(c) >> 4) & 0xf];
419 out += digits[static_cast<unsigned char>(c) & 0xf];
420 } else {
421 out += c;
422 }
423 }
424 }
425 out += '"';
426 }

428 constexpr void dump(std::string& out, const value& v)
429 {
430 switch (v.kind) {
431 case json_kind::null: out += "null"; break;
432 case json_kind::boolean: out += v.boolean ? "true" : "false"; break;
433 case json_kind::number: out += v.text; break;
434 case json_kind::string: dump_string(out, v.text); break;
435 case json_kind::array:
436 out += '[';
437 for (std::size_t i = 0; i < v.items.size(); ++i) {
438 if (i > 0)
439 out += ',';
440 dump(out, v.items[i]);
441 }
442 out += ']';
443 break;
444 case json_kind::object:
445 out += '{';
446 for (std::size_t i = 0; i < v.items.size(); ++i) {
447 if (i > 0)
448 out += ',';
449 dump_string(out, v.keys[i]);
450 out += ':';
451 dump(out, v.items[i]);
452 }
453 out += '}';
454 break;
455 }

460 constexpr std::string decimal(std::int64_t v)
461 {
462 if (v == std::numeric_limits<std::int64_t>::min())
463 return "-9223372036854775808";
464 const bool negative = v < 0;
465 std::uint64_t u = negative ? static_cast<std::uint64_t>(-v) : static_cast<std::uint64_t>(v);
466 std::string s;
467 do {
468 s.insert(s.begin(), static_cast<char>('0' + u % 10));
469 u /= 10;
470 } while (u != 0);
471 if (negative)
472 s.insert(s.begin(), '-');
473 return s;
474 }

505 constexpr std::string build_repetition(const std::string& item, int min_items, int max_items, const std::string& separator = "")
506 {
507 const bool has_max = max_items != no_max_items;
508 if (max_items == 0)
509 return "";
510 if (min_items == 0 && max_items == 1)
511 return item + "?";
512 if (separator.empty()) {
513 if (min_items == 1 && !has_max)
514 return item + "+";
515 if (min_items == 0 && !has_max)
516 return item + "*";
517 return item + "{" + decimal(min_items) + "," + (has_max ? decimal(max_items) : std::string()) + "}";
518 }
519 std::string result = item + " "
520 + build_repetition("(" + separator + " " + item + ")", min_items == 0 ? 0 : min_items - 1,
521 has_max ? max_items - 1 : max_items);
522 if (min_items == 0)
523 result = "(" + result + ")?";
524 return result;
525 }

556 constexpr void uniform_range(std::string& out, std::string_view from, std::string_view to)
557 {
558 std::size_t i = 0;
559 while (i < from.size() && i < to.size() && from[i] == to[i])
560 ++i;
561 if (i > 0) {
562 out += '"';
563 out += from.substr(0, i);
564 out += '"';
565 }
566 if (i < from.size() && i < to.size()) {
567 if (i > 0)
568 out += ' ';
569 const std::size_t sub_len = from.size() - i - 1;
570 if (sub_len > 0) {
571 const std::string from_sub(from.substr(i + 1)), to_sub(to.substr(i + 1));
572 const std::string zeros = repeat("0", sub_len), nines = repeat("9", sub_len);
573 bool to_reached = false;
574 out += '(';
575 if (from_sub == zeros) {
576 digit_range(out, from[i], static_cast<char>(to[i] - 1));
577 out += ' ';
578 more_digits(out, static_cast<int>(sub_len), static_cast<int>(sub_len));
579 } else {
580 out += '[';
581 out += from[i];
582 out += "] (";
583 uniform_range(out, from_sub, nines);
584 out += ')';
585 if (from[i] < to[i] - 1) {
586 out += " | ";
587 if (to_sub == nines) {
588 digit_range(out, static_cast<char>(from[i] + 1), to[i]);
589 to_reached = true;
590 } else {
591 digit_range(out, static_cast<char>(from[i] + 1), static_cast<char>(to[i] - 1));
592 }
593 out += ' ';
594 more_digits(out, static_cast<int>(sub_len), static_cast<int>(sub_len));
595 }
596 }
597 if (!to_reached) {
598 out += " | ";
599 digit_range(out, to[i], to[i]);
600 out += ' ';
601 uniform_range(out, zeros, to_sub);
602 }
603 out += ')';
604 } else {
605 out += '[';
606 out += from[i];
607 out += '-';
608 out += to[i];
609 out += ']';
610 }
611 }

614 // llama.cpp's build_min_max_int: the integers in [min_value, max_value] digit by digit, without leading zeros
615 constexpr void build_min_max_int(std::string& out, std::int64_t min_value, std::int64_t max_value, int decimals_left = 16,
616 bool top_level = true)
617 {
618 const bool has_min = min_value != no_min, has_max = max_value != no_max;
619 if (has_min && has_max) {
620 if (min_value < 0 && max_value < 0) {
621 out += "\"-\" (";
622 build_min_max_int(out, -max_value, -min_value, decimals_left, true);
623 out += ')';
624 return;
625 }
626 if (min_value < 0) {
627 out += "\"-\" (";
628 build_min_max_int(out, 0, -min_value, decimals_left, true);
629 out += ") | ";
630 min_value = 0;
631 }
632 std::string min_s = decimal(min_value);
633 const std::string max_s = decimal(max_value);
634 for (std::size_t digits = min_s.size(); digits < max_s.size(); ++digits) {
635 uniform_range(out, min_s, repeat("9", digits));
636 min_s = "1" + repeat("0", digits);
637 out += " | ";
638 }
639 uniform_range(out, min_s, max_s);
640 return;
641 }
642 const int less_decimals = std::max(decimals_left - 1, 1);
643 if (has_min) {
644 if (min_value < 0) {
645 out += "\"-\" (";
646 build_min_max_int(out, no_min, -min_value, decimals_left, false);
647 out += ") | [0] | [1-9] ";
648 more_digits(out, 0, decimals_left - 1);
649 } else if (min_value == 0) {
650 if (top_level) {
651 out += "[0] | [1-9] ";
652 more_digits(out, 0, less_decimals);
653 } else {
654 more_digits(out, 1, decimals_left);
655 }
656 } else if (min_value <= 9) {
657 const char c = static_cast<char>('0' + min_value);
658 const char range_start = top_level ? '1' : '0';
659 if (c > range_start) {
660 digit_range(out, range_start, static_cast<char>(c - 1));
661 out += ' ';
662 more_digits(out, 1, less_decimals);
663 out += " | ";
664 }
665 digit_range(out, c, '9');
666 out += ' ';
667 more_digits(out, 0, less_decimals);
668 } else {
669 const std::string min_s = decimal(min_value);
670 const int len = static_cast<int>(min_s.size());
671 const char c = min_s[0];
672 if (c > '1') {
673 digit_range(out, top_level ? '1' : '0', static_cast<char>(c - 1));
674 out += ' ';
675 more_digits(out, len, less_decimals);
676 out += " | ";
677 }
678 digit_range(out, c, c);
679 out += " (";
680 build_min_max_int(out, to_int(std::string_view(min_s).substr(1)), no_max, less_decimals, false);
681 out += ')';
682 if (c < '9') {
683 out += " | ";
684 digit_range(out, static_cast<char>(c + 1), '9');
685 out += ' ';
686 more_digits(out, len - 1, less_decimals);
687 }
688 }
689 return;
690 }
691 if (has_max) {
692 if (max_value >= 0) {
693 if (top_level) {
694 out += "\"-\" [1-9] ";
695 more_digits(out, 0, less_decimals);
696 out += " | ";
697 }
698 build_min_max_int(out, 0, max_value, decimals_left, true);
699 } else {
700 out += "\"-\" (";
701 build_min_max_int(out, -max_value, no_max, decimals_left, false);
702 out += ')';
703 }
704 }

791 // A bound as an integer: a fraction rounded inwards (common_chat_schema_builder::get_bound)
792 constexpr bool integer_bound(const value& v, bool round_up, std::int64_t& out)
793 {
794 if (v.kind != json_kind::number)
795 return false;
796 std::string_view t = v.text;
797 const bool negative = !t.empty() && t[0] == '-';
798 if (negative)
799 t.remove_prefix(1);
800 std::string digits;
801 std::int64_t point = 0, exponent = 0;
802 bool seen_point = false, nonzero_fraction = false;
803 std::size_t i = 0;
804 for (; i < t.size() && t[i] != 'e' && t[i] != 'E'; ++i) {
805 if (t[i] == '.') {
806 seen_point = true;
807 continue;
808 }
809 digits += t[i];
810 if (seen_point)
811 ++point;
812 }
813 if (i < t.size()) {
814 ++i;
815 const bool eneg = t[i] == '-';
816 if (t[i] == '+' || t[i] == '-')
817 ++i;
818 for (; i < t.size(); ++i) {
819 exponent = exponent * 10 + (t[i] - '0');
820 if (exponent > 40)
821 return false;
822 }
823 if (eneg)
824 exponent = -exponent;
825 }
826 const std::int64_t shift = exponent - point; // value = digits · 10^shift
827 std::uint64_t whole = 0;
828 const std::int64_t n = static_cast<std::int64_t>(digits.size());
829 const std::int64_t keep = shift >= 0 ? n : n + shift; // the digits left of the point
830 for (std::int64_t k = 0; k < n; ++k) {
831 if (k < keep) {
832 if (whole > (std::numeric_limits<std::uint64_t>::max() - 9) / 10)
833 return false;
834 whole = whole * 10 + static_cast<std::uint64_t>(digits[static_cast<std::size_t>(k)] - '0');
835 } else if (digits[static_cast<std::size_t>(k)] != '0') {
836 nonzero_fraction = true;
837 }
838 }
839 for (std::int64_t k = 0; k < shift; ++k) {
840 if (whole > std::numeric_limits<std::uint64_t>::max() / 10)
841 return false;
842 whole *= 10;
843 }
844 if (whole > static_cast<std::uint64_t>(no_max))
845 return false;
846 std::int64_t r = negative ? -static_cast<std::int64_t>(whole) : static_cast<std::int64_t>(whole);
847 if (nonzero_fraction && (round_up != negative)) // ceil of a positive or floor of a negative moves away from zero
848 r += negative ? -1 : 1;
849 out = r;
850 return true;
851 }

1000 // a JSON pointer into the document; nullptr when it does not resolve
1001 constexpr const value* resolve(std::string_view ref) const
1002 {
1003 const value* target = &root;
1004 std::string_view rest = ref.substr(1); // past '#'
1005 if (!rest.empty())
1006 rest.remove_prefix(1); // past the first '/'
1007 while (true) {
1008 const std::size_t slash = rest.find('/');
1009 const std::string_view sel = rest.substr(0, slash);
1010 if (target->kind == json_kind::object && target->has(sel)) {
1011 target = target->get(sel);
1012 } else if (target->kind == json_kind::array) {
1013 std::size_t idx = 0;
1014 bool number = !sel.empty();
1015 for (const char c : sel) {
1016 number = number && '0' <= c && c <= '9';
1017 idx = idx * 10 + static_cast<std::size_t>(c - '0');
1018 }
1019 if (!number || idx >= target->items.size())
1020 return nullptr;
1021 target = &target->items[idx];
1022 } else {
1023 return nullptr;
1024 }
1025 if (slash == std::string_view::npos)
1026 return target;
1027 rest.remove_prefix(slash + 1);
1028 }

1137 // _not_strings: any JSON string but the given ones, through a trie of their code points
1138 constexpr std::string not_strings(const std::vector<std::string>& strings)
1139 {
1140 struct node
1141 {
1142 std::vector<std::pair<std::uint32_t, std::size_t>> children; // sorted by code point (llama.cpp's std::map)
1143 bool pattern = false;
1144 };
1145 std::vector<node> trie(1);
1146 for (const auto& s : strings) {
1147 const auto cps = decode_utf8_partial(s, {}).first;
1148 std::size_t cur = 0;
1149 for (const std::uint32_t c : cps) {
1150 auto& ch = trie[cur].children;
1151 auto it = std::lower_bound(ch.begin(), ch.end(), c, [](const auto& p, std::uint32_t v) { return p.first < v; });
1152 if (it == ch.end() || it->first != c) {
1153 const std::size_t child = trie.size();
1154 it = ch.insert(it, {c, child});
1155 trie.emplace_back(); // invalidates `ch` and `it`: not used after
1156 cur = child;
1157 } else {
1158 cur = it->second;
1159 }
1160 }
1161 trie[cur].pattern = true;
1162 }
1163 const std::string char_rule = primitive("char");
1164 std::string out = "[\"] ( ";
1165 bool ok = true;
1166 const auto walk = [&](const auto& self, std::size_t idx, int level) -> void {
1167 if (level >= recursion_limit) {
1168 ok = false;
1169 return;
1170 }
1171 std::string rejects;
1172 bool first = true;
1173 for (const auto& [cpt, child] : trie[idx].children) {
1174 std::string c;
1175 append_utf8(c, cpt);
1176 rejects += c;
1177 if (!first)
1178 out += " | ";
1179 first = false;
1180 out += "[" + c + "]";
1181 if (!trie[child].children.empty()) {
1182 out += " (";
1183 self(self, child, level + 1);
1184 out += ")";
1185 } else {
1186 out += " " + char_rule + "+";
1187 }
1188 }
1189 if (!trie[idx].children.empty())
1190 out += " | [^\"" + rejects + "] " + char_rule + "*";
1191 };
1192 walk(walk, 0, 0);
1193 if (!ok)
1194 return fail("#", "a property name is longer than 128 characters");
1195 out += " )";
1196 if (!trie[0].pattern)
1197 out += "?";
1198 out += " [\"]";
1199 return out;
1200 }

1211 constexpr std::string build_object_rule(const std::vector<property>& properties, const std::vector<std::string>& required,
1212 const std::string& name, extra additional, const value* additional_schema,
1213 const std::string& path)
1214 {
1215 std::vector<std::string> required_props, optional_props, prop_names;
1216 std::vector<std::pair<std::string, std::string>> kv; // property -> its kv rule
1217 const auto kv_rule = [&](const std::string& p) -> std::string {
1218 for (const auto& [k, v] : kv)
1219 if (k == p)
1220 return v;
1221 return {};
1222 };
1223 const std::string prefix = name + (name.empty() ? "" : "-");
1224 for (const auto& p : properties) {
1225 const std::string prop_rule = visit(p.schema, prefix + p.name, p.path);
1226 if (failed)
1227 return {};
1228 std::string key;
1229 dump_string(key, p.name);
1230 kv.emplace_back(p.name, add_rule(prefix + p.name + "-kv", format_literal(key) + " space \":\" space " + prop_rule));
1231 if (std::find(required.begin(), required.end(), p.name) != required.end())
1232 required_props.push_back(p.name);
1233 else
1234 optional_props.push_back(p.name);
1235 prop_names.push_back(p.name);
1236 }
1237 if (additional != extra::none) {
1238 const std::string sub_name = prefix + "additional";
1239 const std::string value_rule = additional == extra::schema && classify(*additional_schema).k != kind::any
1240 ? visit(*additional_schema, sub_name + "-value", path + "/additionalProperties")
1241 : primitive("value");
1242 if (failed)
1243 return {};
1244 const std::string key_rule = prop_names.empty() ? primitive("string") : add_rule(sub_name + "-k", not_strings(prop_names));
1245 kv.emplace_back("*", add_rule(sub_name + "-kv", key_rule + " \":\" space " + value_rule));
1246 optional_props.emplace_back("*");
1247 }
1248 if (required_props.empty() && optional_props.empty())
1249 return "\"{\" space \"}\"";
1250 std::string out = "\"{\" space ";
1251 for (std::size_t i = 0; i < required_props.size(); ++i) {
1252 if (i > 0)
1253 out += " \",\" space ";
1254 out += kv_rule(required_props[i]);
1255 }
1256 if (!optional_props.empty()) {
1257 out += " (";
1258 if (!required_props.empty())
1259 out += " \",\" space ( ";
1260 const auto rest = [&](const auto& self, std::size_t from, bool first_is_optional) -> std::string {
1261 if (from >= optional_props.size())
1262 return {};
1263 const std::string& k = optional_props[from];
1264 const std::string kv_name = kv_rule(k);
1265 const std::string comma_ref = "( \",\" space " + kv_name + " )";
1266 std::string res = first_is_optional ? comma_ref + (k == "*" ? "*" : "?") : kv_name + (k == "*" ? " " + comma_ref + "*" : "");
1267 if (from + 1 < optional_props.size())
1268 res += " " + add_rule(prefix + k + "-rest", self(self, from + 1, true));
1269 return res;
1270 };
1271 for (std::size_t i = 0; i < optional_props.size(); ++i) {
1272 if (i > 0)
1273 out += " | ";
1274 out += rest(rest, i, false);
1275 }
1276 if (!required_props.empty())
1277 out += " )";
1278 out += " )?";
1279 }
1280 out += " space \"}\"";
1281 return out;
1282 }

1317 constexpr std::string visit_all_of(const value& s, const std::string& name, const std::string& rule_name, const std::string& path)
1318 {
1319 const value* children = s.get("allOf");
1320 if (children->kind != json_kind::array)
1321 return fail(path + "/allOf", "allOf must be an array of schemas");
1322 if (children->items.empty())
1323 return fail(path + "/allOf", "allOf must not be empty");
1324 std::vector<std::string> required;
1325 std::vector<property> properties;
1326 std::vector<std::pair<std::string, std::size_t>> enum_values; // sorted by literal (llama.cpp's std::map)
1327 std::vector<std::pair<value, std::string>> leaves;
1328 const auto add_component = [&](const auto& self, const value& comp, bool is_required, int level, const std::string& at) -> void {
1329 if (failed)
1330 return;
1331 if (level >= recursion_limit) {
1332 fail(at, "the schema is nested more than 128 deep");
1333 return;
1334 }
1335 const auto c = classify(comp);
1336 if (c.k == kind::ref) {
1337 const value* r = comp.get("$ref");
1338 const value* target = r->kind == json_kind::string ? resolve(r->text) : nullptr;
1339 if (target != nullptr)
1340 self(self, *target, is_required, level + 1, r->text);
1341 } else if (c.k == kind::object) {
1342 if (const value* props = comp.get("properties"); props != nullptr && props->kind == json_kind::object)
1343 for (std::size_t i = 0; i < props->keys.size(); ++i) {
1344 properties.push_back({props->keys[i], props->items[i], at + "/properties/" + props->keys[i]});
1345 if (is_required)
1346 required.push_back(props->keys[i]);
1347 }
1348 } else if (c.k == kind::enumeration) {
1349 const value* e = comp.get("enum");
1350 if (e->kind == json_kind::array)
1351 for (const value& v : e->items) {
1352 const std::string lit = constant(v);
1353 auto it = std::lower_bound(enum_values.begin(), enum_values.end(), lit,
1354 [](const auto& p, const std::string& l) { return p.first < l; });
1355 if (it != enum_values.end() && it->first == lit)
1356 ++it->second;
1357 else
1358 enum_values.insert(it, {lit, 1});
1359 }
1360 } else if (is_required && c.k != kind::any) {
1361 leaves.emplace_back(comp, at);
1362 }
1363 };
1364 for (std::size_t i = 0; i < children->items.size(); ++i) {
1365 const value& child = children->items[i];
1366 const std::string at = path + "/allOf/" + decimal(static_cast<std::int64_t>(i));
1367 if (classify(child).k == kind::any_of) {
1368 std::string alt_path;
1369 for (const value& alt : alternatives(child, at, alt_path))
1370 add_component(add_component, alt, false, 0, alt_path);
1371 } else {
1372 add_component(add_component, child, true, 0, at);
1373 }
1374 }
1375 if (failed)
1376 return {};
1377 if (!enum_values.empty()) {
1378 std::vector<std::string> intersection;
1379 for (const auto& [lit, n] : enum_values)
1380 if (n == children->items.size())
1381 intersection.push_back(lit);
1382 if (!intersection.empty())
1383 return add_rule(rule_name, "(" + join(intersection, " | ") + ")");
1384 }
1385 if (properties.empty() && !leaves.empty()) {
1386 if (leaves.size() == 1)
1387 return visit(leaves[0].first, name, leaves[0].second);
1388 bool all_numeric = true, any_integer = false;
1389 std::int64_t lo = no_min, hi = no_max;
1390 for (const auto& [leaf, at] : leaves) {
1391 const auto k = classify(leaf).k;
1392 if (k == kind::integer) {
1393 std::int64_t a = 0, b = 0;
1394 if (!integer_range(leaf, at, a, b))
1395 return {};
1396 any_integer = true;
1397 lo = std::max(lo, a);
1398 hi = std::min(hi, b);
1399 } else if (k != kind::number) {
1400 all_numeric = false;
1401 }
1402 }
1403 if (all_numeric) {
1404 if (lo > hi)
1405 return fail(path + "/allOf", "allOf has an empty numeric range");
1406 if (any_integer)
1407 return integer_rule(lo, hi, is_reserved(name) ? name + "-" : name.empty() ? "root" : name);
1408 return visit(leaves[0].first, name, leaves[0].second);
1409 }
1410 }
1411 const std::string body = build_object_rule(properties, required, name, extra::none, nullptr, path);
1412 return failed ? std::string() : add_rule(rule_name, body);
1413 }

1416 {
1417 if (failed)
1418 return {};
1419 if (++depth > recursion_limit)

1423 return out;

1426 constexpr std::string visit_impl(const value& s, const std::string& name, const std::string& path)
1427 {
1428 const std::string rule_name = is_reserved(name) ? name + "-" : name.empty() ? "root" : name;
1429 const std::string sub_name = name + (name.empty() ? "" : "-");
1430 const auto c = classify(s);
1431 if (c.k == kind::invalid)
1432 return fail(path, c.message);
1433 for (const std::string_view k : unsupported_keywords)
1434 if (strict && s.has(k))
1435 return fail(path, "unsupported keyword " + std::string(k));
1436 switch (c.k) {
1437 case kind::ref: {
1438 const std::string target = resolve_ref(s, path);
1439 return failed ? std::string() : add_rule(rule_name, target);
1440 }
1441 case kind::any_of: {
1442 std::string alt_path;
1443 const auto alts = alternatives(s, path, alt_path);
1444 std::vector<std::string> rules;
1445 for (std::size_t i = 0; i < alts.size(); ++i) {
1446 const std::string index = decimal(static_cast<std::int64_t>(i));
1447 rules.push_back(visit(alts[i], name + (name.empty() ? "alternative-" : "-") + index, alt_path + "/" + index));
1448 }
1449 return failed ? std::string() : add_rule(rule_name, join(rules, " | "));
1450 }
1451 case kind::all_of: return visit_all_of(s, name, rule_name, path);
1452 case kind::constant: return add_rule(rule_name, constant(*s.get("const")));
1453 case kind::enumeration: {
1454 const value* e = s.get("enum");
1455 if (e->kind != json_kind::array || e->items.empty())
1456 return fail(path, "enum must be a non-empty array");
1457 std::vector<std::string> values;
1458 for (const value& v : e->items)
1459 values.push_back(constant(v));
1460 return add_rule(rule_name, "(" + join(values, " | ") + ")");
1461 }
1462 case kind::object: return visit_object(s, name, rule_name, path);
1463 case kind::array: {
1464 if (const value* items = s.get(s.has("items") ? "items" : "prefixItems"); items != nullptr && items->kind == json_kind::array) {
1465 const std::string key = s.has("items") ? "items" : "prefixItems";
1466 std::string rule = "\"[\" space ";
1467 for (std::size_t i = 0; i < items->items.size(); ++i) {
1468 if (i > 0)
1469 rule += " \",\" space ";
1470 const std::string index = decimal(static_cast<std::int64_t>(i));
1471 rule += visit(items->items[i], sub_name + "tuple-" + index, path + "/" + key + "/" + index);
1472 }
1473 rule += " space \"]\"";
1474 return failed ? std::string() : add_rule(rule_name, rule);
1475 }
1476 const value* items = s.get(s.has("items") ? "items" : "prefixItems");
1477 int min_items = 0, max_items = -1;
1478 if (!count(s, "minItems", path, 0, min_items) || !count(s, "maxItems", path, -1, max_items))
1479 return {};
1480 if ((items == nullptr || classify(*items).k == kind::any) && min_items == 0 && max_items < 0)
1481 return visit_primitive(rule_name, "array");
1482 const value any_item{json_kind::object, false, {}, {}, {}}; // no items: any value, as {}
1483 const std::string item_rule = visit(items == nullptr ? any_item : *items, sub_name + "item",
1484 path + (s.has("items") ? "/items" : "/prefixItems"));
1485 if (failed)
1486 return {};
1487 return add_rule(rule_name, "\"[\" space " + build_repetition(item_rule, min_items, max_items < 0 ? no_max_items : max_items, "\",\" space")
1488 + " space \"]\"");
1489 }
1490 case kind::string: {
1491 if (s.has("pattern")) // only when not strict: llama.cpp's fallback for a pattern it cannot convert
1492 return add_rule(rule_name, primitive("string"));
1493 const value* f = s.get("format");
1494 const std::string_view format = f != nullptr && f->kind == json_kind::string && has_format(s) ? std::string_view(f->text) : "";
1495 if (!format.empty() && format.substr(0, 4) == "uuid")
1496 return visit_primitive(rule_name, "uuid");
1497 if (!format.empty())
1498 return add_rule(rule_name, primitive(std::string(format) + "-string"));
1499 int min_length = 0, max_length = -1;
1500 if (!count(s, "minLength", path, 0, min_length) || !count(s, "maxLength", path, -1, max_length))
1501 return {};
1502 if (min_length > 0 || max_length >= 0) {
1503 const std::string char_rule = primitive("char");
1504 return add_rule(rule_name, "\"\\\"\" " + build_repetition(char_rule, min_length, max_length < 0 ? no_max_items : max_length)
1505 + " \"\\\"\"");
1506 }
1507 return visit_primitive(rule_name, "string");
1508 }
1509 case kind::integer: {
1510 std::int64_t lo = 0, hi = 0;
1511 if (!integer_range(s, path, lo, hi))
1512 return {};
1513 return integer_rule(lo, hi, rule_name);
1514 }
1515 case kind::number:
1516 for (const std::string_view k : {"minimum", "maximum", "exclusiveMinimum", "exclusiveMaximum"})
1517 if (strict && s.has(k))
1518 return fail(path, "unsupported keyword " + std::string(k) + " on a number");
1519 return visit_primitive(rule_name, "number");
1520 case kind::boolean: return visit_primitive(rule_name, "boolean");
1521 case kind::null: return visit_primitive(rule_name, "null");
1522 case kind::any: return add_rule(rule_name, primitive("value"));
1523 case kind::invalid: break;
1524 }
1525 return {};
1526 }
