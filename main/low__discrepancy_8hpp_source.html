71 // Joe and Kuo's recurrence for dimension D (0-based): v_k = m_k << (32 - k) for k <= s, then
72 // v_k = v_{k-s} ^ (v_{k-s} >> s) ^ (a_1 v_{k-1}) ^ ... ^ (a_{s-1} v_{k-s+1}).
73 template <std::size_t D>
74 consteval std::array<std::uint32_t, 32> sobol_dimension() noexcept
75 {
76 std::array<std::uint32_t, 32> v{};
77 if constexpr (D == 0) {
78 for (std::size_t k = 0; k < 32; ++k)
79 v[k] = std::uint32_t{1} << (31 - k);
80 } else {
81 const std::size_t s = snicholls::detail::sobol_degree[D - 1];
82 const std::uint32_t a = snicholls::detail::sobol_coefficients[D - 1];
83 const std::size_t offset = snicholls::detail::sobol_m_offset[D - 1];
84 for (std::size_t k = 0; k < 32; ++k) {
85 if (k < s) {
86 v[k] = static_cast<std::uint32_t>(snicholls::detail::sobol_m[offset + k]) << (31 - k);
87 } else {
88 std::uint32_t x = v[k - s] ^ (v[k - s] >> s);
89 for (std::size_t j = 1; j < s; ++j)
90 if (((a >> (s - 1 - j)) & 1u) != 0)
91 x ^= v[k - j];
92 v[k] = x;
93 }
94 }
95 }
96 return v;
97 }

101 inline constexpr std::array<std::uint32_t, 32> sobol_dimension_v = sobol_dimension<D>();

112 template <std::size_t N>
113 consteval std::array<std::uint32_t, N> first_primes() noexcept
114 {
115 std::array<std::uint32_t, N> primes{};
116 std::size_t found = 0;
117 for (std::uint32_t candidate = 2; found < N; ++candidate) {
118 bool prime = true;
119 for (std::size_t i = 0; i < found && primes[i] * primes[i] <= candidate; ++i)
120 if (candidate % primes[i] == 0) {
121 prime = false;
122 break;
123 }
124 if (prime)
125 primes[found++] = candidate;
126 }
127 return primes;
128 }

137 [[nodiscard]] constexpr double radical_inverse(std::uint64_t base, std::uint64_t i) noexcept
138 {
139 assert(base >= 2);
140 double result = 0;
141 double scale = 1.0 / static_cast<double>(base);
142 while (i != 0) {
143 result += scale * static_cast<double>(i % base);
144 i /= base;
145 scale /= static_cast<double>(base);
146 }
147 return result;
148 }

210 constexpr std::array<double, Dims> next() noexcept
211 {
212 assert(index_ < (std::uint64_t{1} << 32) && "snicholls::stats::sobol: more than 2^32 points");
213 std::array<double, Dims> p{};
214 for (std::size_t d = 0; d < Dims; ++d)
215 p[d] = static_cast<double>(x_[d] ^ shift_[d]) * 0x1.0p-32;
216 const auto c = static_cast<std::size_t>(std::countr_zero(~index_)); // the rightmost zero bit
217 for (std::size_t d = 0; d < Dims; ++d)
218 x_[d] ^= detail::sobol_v<Dims>[d][c];
219 ++index_;
220 return p;
221 }

226 [[nodiscard]] static constexpr std::array<double, Dims> point(std::uint64_t i) noexcept
227 {
228 std::array<double, Dims> p{};
229 const std::uint64_t gray = i ^ (i >> 1);
230 for (std::size_t d = 0; d < Dims; ++d) {
231 std::uint32_t x = 0;
232 for (std::uint64_t bits = gray; bits != 0; bits &= bits - 1)
233 x ^= detail::sobol_v<Dims>[d][static_cast<std::size_t>(std::countr_zero(bits))];
234 p[d] = static_cast<double>(x) * 0x1.0p-32;
235 }
236 return p;
237 }
