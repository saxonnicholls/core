106 struct slaney_constants
107 {
108 static constexpr C f_sp = C(200) / C(3);
109 static constexpr C min_log_hz = C(1000);
110 static constexpr C min_log_mel = min_log_hz / f_sp; // 15

125template <math::math_number T>
126constexpr T hz_to_mel(T hz, mel_scale scale = mel_scale::slaney) noexcept
127{
128 using C = compute_t<T>;
129 using K = detail::mel::slaney_constants<C>;
130 const C f = C(hz);
131 if (scale == mel_scale::htk)
132 return static_cast<T>(C(2595) * math::log10(C(1) + f / C(700)));
133 if (f >= K::min_log_hz)
134 return static_cast<T>(K::min_log_mel + math::log(f / K::min_log_hz) / detail::mel::logstep<C>());
135 return static_cast<T>(f / K::f_sp);
136}

143template <math::math_number T>
144constexpr T mel_to_hz(T mel, mel_scale scale = mel_scale::slaney) noexcept
145{
146 using C = compute_t<T>;
147 using K = detail::mel::slaney_constants<C>;
148 const C m = C(mel);
149 if (scale == mel_scale::htk)
150 return static_cast<T>(C(700) * (math::pow(C(10), m / C(2595)) - C(1)));
151 if (m >= K::min_log_mel)
152 return static_cast<T>(K::min_log_hz * math::exp(detail::mel::logstep<C>() * (m - K::min_log_mel)));
153 return static_cast<T>(K::f_sp * m);
154}

161{
162 T sample_rate = T(16000);
163 std::size_t n_fft = 400;
164 std::size_t n_mels = 80;
165 T f_min = T(0);
166 T f_max = T(0);
167 mel_scale scale = mel_scale::slaney;
168 mel_norm norm = mel_norm::slaney;

172 [[nodiscard]] constexpr std::size_t bins() const noexcept { return n_fft / 2 + 1; }
173};

179template <math::math_number T>
180constexpr void mel_filterbank(const mel_filterbank_options<T>& o, std::span<std::type_identity_t<T>> out) noexcept
181{
182 using C = compute_t<T>;
183 const std::size_t bins = o.bins(), mels = o.n_mels;
184 assert(out.size() == mels * bins);
185 const C f_max = o.f_max > T(0) ? C(o.f_max) : C(o.sample_rate) / C(2);
186 const C lo = C(hz_to_mel(C(o.f_min), o.scale)), hi = C(hz_to_mel(f_max, o.scale));
187 // np.linspace(lo, hi, mels + 2): i * step + lo, the last exactly hi; then back to hertz.
188 const auto mel_f = [&](std::size_t i) {
189 const C step = (hi - lo) / C(static_cast<long long>(mels + 1));
190 const C m = i == mels + 1 ? hi : C(static_cast<long long>(i)) * step + lo;
191 return mel_to_hz(m, o.scale);
192 };
193 // np.fft.rfftfreq(n_fft, 1 / sr): k * (1 / (n_fft * (1 / sr))).
194 const C val = C(1) / (C(static_cast<long long>(o.n_fft)) * (C(1) / C(o.sample_rate)));
195 for (std::size_t m = 0; m < mels; ++m) {
196 const C f0 = mel_f(m), f1 = mel_f(m + 1), f2 = mel_f(m + 2);
197 const C enorm = C(2) / (f2 - f0);
198 for (std::size_t k = 0; k < bins; ++k) {
199 const C fk = C(static_cast<long long>(k)) * val;
200 const C lower = -(f0 - fk) / (f1 - f0), upper = (f2 - fk) / (f2 - f1);
201 const C w = lower < upper ? lower : upper;
202 const T wt = static_cast<T>(w > C(0) ? w : C(0)); // librosa stores the triangle, then scales it
203 out[m * bins + k] = o.norm == mel_norm::slaney ? static_cast<T>(C(wt) * enorm) : wt;
204 }
205 }
206}

234constexpr void mel_spectrogram(std::span<const T> filters, std::size_t n_mels, std::span<const std::type_identity_t<T>> spectrogram,
235 std::span<std::type_identity_t<T>> out) noexcept
236{
237 const std::size_t bins = n_mels == 0 ? 0 : filters.size() / n_mels, frames = bins == 0 ? 0 : spectrogram.size() / bins;
238 assert(filters.size() == n_mels * bins && spectrogram.size() == frames * bins && out.size() == frames * n_mels);
239 for (std::size_t t = 0; t < frames; ++t)
240 for (std::size_t m = 0; m < n_mels; ++m) {
241 T s = T(0);
242 for (std::size_t k = 0; k < bins; ++k)
243 s += filters[m * bins + k] * spectrogram[t * bins + k];
244 out[t * n_mels + m] = s;
245 }

306template <math::math_number T>
307constexpr void log_compress(std::span<const T> in, std::span<std::type_identity_t<T>> out, const log_compression<std::type_identity_t<T>>& c) noexcept
308{
309 assert(out.size() == in.size());
310 if (in.empty())
311 return;
312 T top = T(0);
313 for (std::size_t i = 0; i < in.size(); ++i) {
314 const T y = c.multiplier * math::log10(in[i] > c.floor ? in[i] : c.floor);
315 out[i] = y;
316 top = i == 0 || y > top ? y : top;
317 }
318 const T lowest = top - c.range;
319 for (auto& y : out)
320 y = ((y > lowest ? y : lowest) + c.offset) * c.scale;
321}

325 // What log_compress's backward pass needs beyond each element: the maximum, the floor below it, and each tied
326 // maximum's share of the gradients of the values floored to it.
327 template <typename T>
328 struct compression_backward
329 {
330 log_compression<T> c;
331 T top = T(0), lowest = T(0), share = T(0);
332
333 constexpr T log_of(T x) const noexcept { return c.multiplier * math::log10(x > c.floor ? x : c.floor); }
334
335 template <typename In, typename Grad>
336 constexpr compression_backward(const log_compression<T>& comp, std::size_t n, In in, Grad grad) noexcept : c(comp)
337 {
338 top = n == 0 ? T(0) : log_of(in(0));
339 for (std::size_t i = 1; i < n; ++i)
340 top = log_of(in(i)) > top ? log_of(in(i)) : top;
341 lowest = top - c.range;
342 T to_top = T(0);
343 std::size_t ties = 0;
344 for (std::size_t i = 0; i < n; ++i) {
345 const T y = log_of(in(i));
346 ties += y == top ? 1 : 0;
347 to_top += y < lowest ? grad(i) * c.scale : T(0);
348 }
349 share = ties == 0 ? T(0) : to_top / T(static_cast<long long>(ties));
350 }
351
352 // dL/dx for one element.
353 constexpr T operator()(T x, T g) const noexcept
354 {
355 const T y = log_of(x);
356 const T dy = (y < lowest ? T(0) : g * c.scale) + (y == top ? share : T(0));
357 return x >= c.floor ? dy * c.multiplier / (x * math::log(T(10))) : T(0);
358 }
359 };
360} // namespace detail::mel

372constexpr void log_compress_backward(std::span<const T> in, std::span<const std::type_identity_t<T>> grad_out,

374{

376 const detail::mel::compression_backward<T> b(c, in.size(), [in](std::size_t i) { return in[i]; },

446template <math::math_number T>
447constexpr void whisper_log_mel(const rfft_plan<T>& plan, std::span<const std::type_identity_t<T>> window, std::span<const std::type_identity_t<T>> filters,
448 std::size_t n_mels, std::span<const std::type_identity_t<T>> audio, std::size_t padding,
449 std::span<std::type_identity_t<T>> out, std::span<complex<std::type_identity_t<T>>> work) noexcept
450{
451 const stft_options o = whisper_stft_options();
452 const std::size_t bins = plan.bins(), length = audio.size() + padding, frames = whisper_frames(audio.size(), padding);
453 assert(plan.size() == whisper_n_fft && window.size() == whisper_n_fft && filters.size() == n_mels * bins);
454 assert(out.size() == n_mels * frames && work.size() >= whisper_work_size(plan));
455 const auto row = work.first(bins);
456 const auto sample = [&](std::size_t p) {
457 const long long i = detail::stft::source_index(p, o.padding(), length, o.pad);
458 return static_cast<std::size_t>(i) < audio.size() ? audio[static_cast<std::size_t>(i)] : T(0);
459 };
460 for (std::size_t t = 0; t < frames; ++t) {
461 plan.forward_from([&](std::size_t j) { return window[j] * sample(t * o.hop + j); }, row, work.subspan(bins));
462 for (std::size_t k = 0; k < bins; ++k)
463 row[k].re = norm(row[k]);
464 for (std::size_t m = 0; m < n_mels; ++m) {
465 T s = T(0);
466 for (std::size_t k = 0; k < bins; ++k)
467 s += filters[m * bins + k] * row[k].re;
468 out[m * frames + t] = s;
469 }
470 }
471 log_compress(std::span<const T>(out), out, whisper_compression<T>());
472}

486template <math::math_number T>
487constexpr void whisper_log_mel_backward(const rfft_plan<T>& plan, std::span<const std::type_identity_t<T>> window,
488 std::span<const std::type_identity_t<T>> filters, std::size_t n_mels,
489 std::span<const std::type_identity_t<T>> audio, std::size_t padding,
490 std::span<const std::type_identity_t<T>> grad_out, std::span<std::type_identity_t<T>> grad_audio,
491 std::span<std::type_identity_t<T>> scratch, std::span<complex<std::type_identity_t<T>>> work) noexcept
492{
493 const stft_options o = whisper_stft_options();
494 const std::size_t bins = plan.bins(), length = audio.size() + padding, frames = whisper_frames(audio.size(), padding), n = o.n_fft;
495 assert(grad_out.size() == n_mels * frames && grad_audio.size() == audio.size() && scratch.size() == n_mels * frames);
496 assert(work.size() >= whisper_work_size(plan));
497 const auto row = work.first(bins);
498 const auto sample = [&](std::size_t p) {
499 const long long i = detail::stft::source_index(p, o.padding(), length, o.pad);
500 return static_cast<std::size_t>(i) < audio.size() ? audio[static_cast<std::size_t>(i)] : T(0);
501 };
502 const auto spectrum_of = [&](std::size_t t) {
503 plan.forward_from([&](std::size_t j) { return window[j] * sample(t * o.hop + j); }, row, work.subspan(bins));
504 };
505 for (std::size_t t = 0; t < frames; ++t) { // the mel spectrogram, before compression
506 spectrum_of(t);
507 for (std::size_t m = 0; m < n_mels; ++m) {
508 T s = T(0);
509 for (std::size_t k = 0; k < bins; ++k)
510 s += filters[m * bins + k] * norm(row[k]);
511 scratch[m * frames + t] = s;
512 }
513 }
514 const detail::mel::compression_backward<T> back(whisper_compression<T>(), scratch.size(), [scratch](std::size_t i) { return scratch[i]; },
515 [grad_out](std::size_t i) { return grad_out[i]; });
516 for (std::size_t i = 0; i < scratch.size(); ++i)
517 scratch[i] = back(scratch[i], grad_out[i]); // now dL/d mel
518 for (auto& g : grad_audio)
519 g = T(0);
520 for (std::size_t t = 0; t < frames; ++t) {
521 spectrum_of(t);
522 for (std::size_t k = 0; k < bins; ++k) { // dL/dX_k = 2 X_k sum_m F[m][k] dL/dmel[m][t]
523 T dp = T(0);
524 for (std::size_t m = 0; m < n_mels; ++m)
525 dp += filters[m * bins + k] * scratch[m * frames + t];
526 row[k] *= T(2) * dp;
527 }
528 plan.inverse_to([&](std::size_t k) { return k == 0 || 2 * k == n ? row[k] : row[k] * T(0.5); },
529 [&](std::size_t j, T v) {
530 const long long i = detail::stft::source_index(t * o.hop + j, o.padding(), length, o.pad);
531 if (static_cast<std::size_t>(i) < audio.size())
532 grad_audio[static_cast<std::size_t>(i)] += window[j] * v;
533 },
534 work.subspan(bins));
535 }
536}
