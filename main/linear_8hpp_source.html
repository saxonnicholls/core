196template <typename S, row X, mutable_row Y, row B = std::span<const row_value_t<X>>>
197 requires std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<B>>
198constexpr void linear(const dense_weights<S>& w, const X& x, Y&& y, const B& bias = B{})
199{
200 using T = row_value_t<X>;
201 const auto in = as_row(x);
202 const auto out = as_mutable_row(y);
203 const auto b = as_row(bias);
204 assert(in.size() == w.cols && out.size() == w.rows && w.data.size() == w.rows * w.cols && (b.empty() || b.size() == w.rows));
205 for (std::size_t i = 0; i < w.rows; ++i) {
206 const auto r = w.row(i);
207 T acc = b.empty() ? T(0) : b[i];
208 for (std::size_t j = 0; j < w.cols; ++j) {
209 const T p = detail::linear::widen<T>(r[j]) * in[j];
210 acc += p;
211 }
212 out[i] = acc;
213 }
214}

228template <typename S, row X, row DY, mutable_row DX, mutable_row DW, mutable_row DB = std::span<row_value_t<X>>>
229 requires std::same_as<row_value_t<X>, row_value_t<DY>> && std::same_as<row_value_t<X>, row_value_t<DX>>
230 && std::same_as<row_value_t<X>, row_value_t<DW>> && std::same_as<row_value_t<X>, row_value_t<DB>>
231constexpr void linear_backward(const dense_weights<S>& w, const X& x, const DY& dy, DX&& dx, DW&& dw, DB&& db = DB{})
232{
233 using T = row_value_t<X>;
234 const auto in = as_row(x);
235 const auto g = as_row(dy);
236 const auto gx = as_mutable_row(dx);
237 const auto gw = as_mutable_row(dw);
238 const auto gb = as_mutable_row(db);
239 assert(in.size() == w.cols && g.size() == w.rows && w.data.size() == w.rows * w.cols && gw.size() == w.rows * w.cols
240 && (gx.empty() || gx.size() == w.cols) && (gb.empty() || gb.size() == w.rows));
241 for (std::size_t i = 0; i < w.rows; ++i) {
242 const auto r = w.row(i);
243 const T d = g[i];
244 if (!gx.empty())
245 for (std::size_t j = 0; j < w.cols; ++j) {
246 const T p = detail::linear::widen<T>(r[j]) * d;
247 gx[j] += p;
248 }
249 for (std::size_t j = 0; j < w.cols; ++j) {
250 const T p = d * in[j];
251 gw[i * w.cols + j] += p;
252 }
253 if (!gb.empty())
254 gb[i] += d;
255 }
256}

263template <quant_block Block, std::size_t M, mutable_row Y, row B = std::span<const float>>
264 requires std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
265constexpr void linear(const quantized_weights<Block>& w, const std::array<block_q8_0, M>& x, Y&& y, const B& bias = B{})
266{
267 const auto out = as_mutable_row(y);
268 const auto b = as_row(bias);
269 assert(M * quant_block_size == w.cols && out.size() == w.rows && w.blocks.size() == w.rows * M && (b.empty() || b.size() == w.rows));
270 for (std::size_t i = 0; i < w.rows; ++i) {
271 const float d = dot(w.row(i), std::span<const block_q8_0>(x));
272 out[i] = b.empty() ? d : b[i] + d;
273 }
274}

290 // C = alpha op(A) op(B) + beta C. The shapes are right by construction and C never overlaps A or B, so the
291 // backend's product always succeeds
292 template <typename B, typename T, typename L, typename R, typename C>
293 constexpr void product(const B& backend, T alpha, const L& lhs, const R& rhs, T beta, const C& c)
294 {
295 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
296 assert(done.is_ok());
297 }

395template <math::math_number T, typename B>
396 requires linalg::blas_backend<B, T>
397constexpr void linear_sequence(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
398 std::type_identity_t<std::span<T>> ys, std::type_identity_t<std::span<const T>> bias = {})
399{
400 assert(w.cols > 0 && xs.size() % w.cols == 0);
401 const std::size_t n = xs.size() / w.cols;
402 assert(ys.size() == n * w.rows && w.data.size() == w.rows * w.cols && (bias.empty() || bias.size() == w.rows));
403 if (n == 0 || w.rows == 0)
404 return;
405 using detail::linear::view;
406 detail::linear::product(backend, T(1), view(xs, n, w.cols), linalg::transposed(view(w.data, w.rows, w.cols)), T(0),
407 view(ys, n, w.rows));
408 if (!bias.empty())
409 add_bias_rows(backend, ys, bias);
410}

421template <math::math_number T, typename B>
422 requires linalg::blas_backend<B, T>
423constexpr void linear_sequence_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
424 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
425 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db = {})
426{
427 assert(w.cols > 0 && xs.size() % w.cols == 0);
428 const std::size_t n = xs.size() / w.cols;
429 assert(dys.size() == n * w.rows && (dxs.empty() || dxs.size() == xs.size()) && dw.size() == w.rows * w.cols
430 && (db.empty() || db.size() == w.rows));
431 if (n == 0 || w.rows == 0)
432 return;
433 using detail::linear::view;
434 const auto dy = view(dys, n, w.rows);
435 if (!dxs.empty())
436 detail::linear::product(backend, T(1), dy, view(w.data, w.rows, w.cols), T(1), view(dxs, n, w.cols));
437 detail::linear::product(backend, T(1), linalg::transposed(dy), view(xs, n, w.cols), T(1), view(dw, w.rows, w.cols));
438 if (!db.empty())
439 sum_rows(backend, dys, db);
440}

452template <math::math_number T, typename B>
453 requires linalg::blas_backend<B, T>
454constexpr void linear_heads(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
455 std::type_identity_t<std::span<T>> out, std::type_identity_t<std::span<const T>> bias, std::size_t heads,
456 std::size_t sequences = 1)
457{
458 assert(heads > 0 && sequences > 0 && w.rows % heads == 0 && w.cols > 0 && xs.size() % (w.cols * sequences) == 0);
459 const std::size_t d = w.rows / heads, n = xs.size() / w.cols / sequences;
460 assert(out.size() == sequences * n * w.rows);
461 for (std::size_t s = 0; s < sequences; ++s)
462 for (std::size_t h = 0; h < heads; ++h)
463 linear_sequence(backend, dense_weights<T>{w.data.subspan(h * d * w.cols, d * w.cols), d, w.cols}, xs.subspan(s * n * w.cols, n * w.cols),
464 out.subspan((s * heads + h) * n * d, n * d), bias.empty() ? bias : bias.subspan(h * d, d));
465}

472template <math::math_number T, typename B>
473 requires linalg::blas_backend<B, T>
474constexpr void linear_heads_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
475 std::type_identity_t<std::span<const T>> douts, std::type_identity_t<std::span<T>> dxs,
476 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db, std::size_t heads,
477 std::size_t sequences = 1)
478{
479 assert(heads > 0 && sequences > 0 && w.rows % heads == 0 && w.cols > 0 && xs.size() % (w.cols * sequences) == 0);
480 const std::size_t d = w.rows / heads, n = xs.size() / w.cols / sequences, rows = d * w.cols;
481 assert(douts.size() == sequences * n * w.rows && dw.size() == w.rows * w.cols);

486 dw.subspan(h * rows, rows), db.empty() ? db : db.subspan(h * d, d));
487}
