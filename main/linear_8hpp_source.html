195template <typename S, row X, mutable_row Y, row B = std::span<const row_value_t<X>>>
196 requires std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<B>>
197constexpr void linear(const dense_weights<S>& w, const X& x, Y&& y, const B& bias = B{})
198{
199 using T = row_value_t<X>;
200 const auto in = as_row(x);
201 const auto out = as_mutable_row(y);
202 const auto b = as_row(bias);
203 assert(in.size() == w.cols && out.size() == w.rows && w.data.size() == w.rows * w.cols && (b.empty() || b.size() == w.rows));
204 for (std::size_t i = 0; i < w.rows; ++i) {
205 const auto r = w.row(i);
206 T acc = b.empty() ? T(0) : b[i];
207 for (std::size_t j = 0; j < w.cols; ++j) {
208 const T p = detail::linear::widen<T>(r[j]) * in[j];
209 acc += p;
210 }
211 out[i] = acc;
212 }
213}

227template <typename S, row X, row DY, mutable_row DX, mutable_row DW, mutable_row DB = std::span<row_value_t<X>>>
228 requires std::same_as<row_value_t<X>, row_value_t<DY>> && std::same_as<row_value_t<X>, row_value_t<DX>>
229 && std::same_as<row_value_t<X>, row_value_t<DW>> && std::same_as<row_value_t<X>, row_value_t<DB>>
230constexpr void linear_backward(const dense_weights<S>& w, const X& x, const DY& dy, DX&& dx, DW&& dw, DB&& db = DB{})
231{
232 using T = row_value_t<X>;
233 const auto in = as_row(x);
234 const auto g = as_row(dy);
235 const auto gx = as_mutable_row(dx);
236 const auto gw = as_mutable_row(dw);
237 const auto gb = as_mutable_row(db);
238 assert(in.size() == w.cols && g.size() == w.rows && w.data.size() == w.rows * w.cols && gw.size() == w.rows * w.cols
239 && (gx.empty() || gx.size() == w.cols) && (gb.empty() || gb.size() == w.rows));
240 for (std::size_t i = 0; i < w.rows; ++i) {
241 const auto r = w.row(i);
242 const T d = g[i];
243 if (!gx.empty())
244 for (std::size_t j = 0; j < w.cols; ++j) {
245 const T p = detail::linear::widen<T>(r[j]) * d;
246 gx[j] += p;
247 }
248 for (std::size_t j = 0; j < w.cols; ++j) {
249 const T p = d * in[j];
250 gw[i * w.cols + j] += p;
251 }
252 if (!gb.empty())
253 gb[i] += d;
254 }
255}

262template <quant_block Block, std::size_t M, mutable_row Y, row B = std::span<const float>>
263 requires std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
264constexpr void linear(const quantized_weights<Block>& w, const std::array<block_q8_0, M>& x, Y&& y, const B& bias = B{})
265{
266 const auto out = as_mutable_row(y);
267 const auto b = as_row(bias);
268 assert(M * quant_block_size == w.cols && out.size() == w.rows && w.blocks.size() == w.rows * M && (b.empty() || b.size() == w.rows));
269 for (std::size_t i = 0; i < w.rows; ++i) {
270 const float d = dot(w.row(i), std::span<const block_q8_0>(x));
271 out[i] = b.empty() ? d : b[i] + d;
272 }
273}

289 // C = alpha op(A) op(B) + beta C. The shapes are right by construction and C never overlaps A or B, so the
290 // backend's product always succeeds
291 template <typename B, typename T, typename L, typename R, typename C>
292 constexpr void product(const B& backend, T alpha, const L& lhs, const R& rhs, T beta, const C& c)
293 {
294 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
295 assert(done.is_ok());
296 }

394template <math::math_number T, typename B>
395 requires linalg::blas_backend<B, T>
396constexpr void linear_sequence(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
397 std::type_identity_t<std::span<T>> ys, std::type_identity_t<std::span<const T>> bias = {})
398{
399 assert(w.cols > 0 && xs.size() % w.cols == 0);
400 const std::size_t n = xs.size() / w.cols;
401 assert(ys.size() == n * w.rows && w.data.size() == w.rows * w.cols && (bias.empty() || bias.size() == w.rows));
402 if (n == 0 || w.rows == 0)
403 return;
404 using detail::linear::view;
405 detail::linear::product(backend, T(1), view(xs, n, w.cols), linalg::transposed(view(w.data, w.rows, w.cols)), T(0),
406 view(ys, n, w.rows));
407 if (!bias.empty())
408 add_bias_rows(backend, ys, bias);
409}

420template <math::math_number T, typename B>
421 requires linalg::blas_backend<B, T>
422constexpr void linear_sequence_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
423 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
424 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db = {})
425{
426 assert(w.cols > 0 && xs.size() % w.cols == 0);
427 const std::size_t n = xs.size() / w.cols;
428 assert(dys.size() == n * w.rows && (dxs.empty() || dxs.size() == xs.size()) && dw.size() == w.rows * w.cols
429 && (db.empty() || db.size() == w.rows));
430 if (n == 0 || w.rows == 0)
431 return;
432 using detail::linear::view;
433 const auto dy = view(dys, n, w.rows);
434 if (!dxs.empty())
435 detail::linear::product(backend, T(1), dy, view(w.data, w.rows, w.cols), T(1), view(dxs, n, w.cols));
436 detail::linear::product(backend, T(1), linalg::transposed(dy), view(xs, n, w.cols), T(1), view(dw, w.rows, w.cols));
437 if (!db.empty())
438 sum_rows(backend, dys, db);
439}

448template <math::math_number T, typename B>
449 requires linalg::blas_backend<B, T>
450constexpr void linear_heads(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
451 std::type_identity_t<std::span<T>> out, std::type_identity_t<std::span<const T>> bias, std::size_t heads)
452{
453 assert(heads > 0 && w.rows % heads == 0 && w.cols > 0 && xs.size() % w.cols == 0);
454 const std::size_t d = w.rows / heads, n = xs.size() / w.cols;
455 assert(out.size() == n * w.rows);
456 for (std::size_t h = 0; h < heads; ++h)
457 linear_sequence(backend, dense_weights<T>{w.data.subspan(h * d * w.cols, d * w.cols), d, w.cols}, xs, out.subspan(h * n * d, n * d),
458 bias.empty() ? bias : bias.subspan(h * d, d));
459}

465template <math::math_number T, typename B>
466 requires linalg::blas_backend<B, T>
467constexpr void linear_heads_backward(const B& backend, const dense_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
468 std::type_identity_t<std::span<const T>> douts, std::type_identity_t<std::span<T>> dxs,
469 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db, std::size_t heads)
470{
471 assert(heads > 0 && w.rows % heads == 0 && w.cols > 0 && xs.size() % w.cols == 0);
472 const std::size_t d = w.rows / heads, n = xs.size() / w.cols, rows = d * w.cols;
473 assert(douts.size() == n * w.rows && dw.size() == w.rows * w.cols);
474 for (std::size_t h = 0; h < heads; ++h)
475 linear_sequence_backward(backend, dense_weights<T>{w.data.subspan(h * rows, rows), d, w.cols}, xs, douts.subspan(h * n * d, n * d), dxs,
476 dw.subspan(h * rows, rows), db.empty() ? db : db.subspan(h * d, d));
477}
