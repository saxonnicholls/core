156template <typename S, row X, mutable_row Y, row B = std::span<const row_value_t<X>>>
157 requires std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<B>>
158constexpr void linear(const dense_weights<S>& w, const X& x, Y&& y, const B& bias = B{})
159{
160 using T = row_value_t<X>;
161 const auto in = as_row(x);
162 const auto out = as_mutable_row(y);
163 const auto b = as_row(bias);
164 assert(in.size() == w.cols && out.size() == w.rows && w.data.size() == w.rows * w.cols && (b.empty() || b.size() == w.rows));
165 for (std::size_t i = 0; i < w.rows; ++i) {
166 const auto r = w.row(i);
167 T acc = b.empty() ? T(0) : b[i];
168 for (std::size_t j = 0; j < w.cols; ++j) {
169 const T p = detail::linear::widen<T>(r[j]) * in[j];
170 acc += p;
171 }
172 out[i] = acc;
173 }
174}

181template <quant_block Block, std::size_t M, mutable_row Y, row B = std::span<const float>>
182 requires std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
183constexpr void linear(const quantized_weights<Block>& w, const std::array<block_q8_0, M>& x, Y&& y, const B& bias = B{})
184{
185 const auto out = as_mutable_row(y);
186 const auto b = as_row(bias);
187 assert(M * quant_block_size == w.cols && out.size() == w.rows && w.blocks.size() == w.rows * M && (b.empty() || b.size() == w.rows));
188 for (std::size_t i = 0; i < w.rows; ++i) {
189 const float d = dot(w.row(i), std::span<const block_q8_0>(x));
190 out[i] = b.empty() ? d : b[i] + d;
191 }
192}
