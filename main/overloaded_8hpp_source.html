174 // What an element passed to overloaded{...} / first_match{...} becomes: a function pointer is

179 // The index of the first true flag, or sizeof...(Bs) when none is. A loop, not a chain.
180 template <bool... Bs>
181 constexpr std::size_t first_true() noexcept
182 {
183 constexpr bool flags[]{Bs..., false}; // the trailing false keeps the array non-empty
184 std::size_t i = 0;
185 while (i < sizeof...(Bs) && !flags[i])
186 ++i;
187 return i;
188 }
