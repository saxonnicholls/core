150[[nodiscard]] constexpr std::size_t rope_section_axis(rope_mode m, const rope_sections& s, std::size_t i) noexcept
151{
152 if (m == rope_mode::normal || m == rope_mode::neox || s.total() == 0)
153 return 0;
154 const std::size_t sector = i % s.total();
155 if (m == rope_mode::imrope) {
156 if (sector % 3 == 1 && sector < 3 * s.h)
157 return 1;
158 if (sector % 3 == 2 && sector < 3 * s.w)
159 return 2;
160 if (sector % 3 == 0 && sector < 3 * s.t)
161 return 0;
162 return 3;
163 }
164 return sector < s.t ? 0 : sector < s.t + s.h ? 1 : sector < s.t + s.h + s.w ? 2 : 3;
165}

169 // The first pair of axis a's section in one period (vision mode restarts each axis's frequencies there)
170 constexpr std::size_t section_start(const rope_sections& s, std::size_t a) noexcept
171 {
172 return a == 0 ? 0 : a == 1 ? s.t : a == 2 ? s.t + s.h : s.t + s.h + s.w;
173 }

175 template <typename T>
176 constexpr T frequency(rope_mode m, const rope_sections& s, std::size_t i, std::size_t pairs, T base)
177 {
178 if (m != rope_mode::vision)
179 return rope_frequency<T>(i, 2 * pairs, base);
180 const std::size_t sector = i % s.total(), j = sector - section_start(s, rope_section_axis(m, s, i));
181 return rope_frequency<T>(j, pairs, base); // ggml: n_dims = d/2, theta_scale = base^(-2 / n_dims)
182 }

194template <mutable_row C, mutable_row S>
195 requires std::same_as<row_value_t<C>, row_value_t<S>>
196constexpr void rope_mode_angles(const rope_position& position, C&& cos, S&& sin, std::type_identity_t<row_value_t<C>> base, rope_mode mode,
197 const rope_sections& sections = {}, std::span<const row_value_t<C>> factors = {},
198 std::type_identity_t<row_value_t<C>> amplitude = 1)
199{
200 using T = row_value_t<C>;
201 const auto c = as_mutable_row(cos);
202 const auto s = as_mutable_row(sin);
203 const bool sectioned = mode != rope_mode::normal && mode != rope_mode::neox;
204 assert(c.size() == s.size() && (factors.empty() || factors.size() == c.size()));
205 assert(!sectioned || sections.t > 0 || sections.h > 0 || sections.w > 0);
206 (void)sectioned;
207 for (std::size_t i = 0; i < c.size(); ++i) {
208 const std::size_t axis = rope_section_axis(mode, sections, i);
209 T f = detail::rope_modes::frequency<T>(mode, sections, i, c.size(), base);
210 if (!factors.empty()) {
211 assert(factors[i] > T(0));
212 f = f / factors[i];
213 }
214 const T angle = T(position.at(axis)) * f;
215 c[i] = amplitude == T(1) ? math::cos(angle) : amplitude * math::cos(angle);
216 s[i] = amplitude == T(1) ? math::sin(angle) : amplitude * math::sin(angle);
217 }
218}

303template <mutable_row X>
304constexpr void rope_shift(X&& x, std::int64_t delta, std::type_identity_t<row_value_t<X>> base, rope_layout layout = rope_layout::half,
305 std::span<const row_value_t<X>> factors = {})
306{
307 using T = row_value_t<X>;
308 const auto v = as_mutable_row(x);
309 const std::size_t h = v.size() / 2;
310 assert(v.size() % 2 == 0 && (factors.empty() || factors.size() == h));
311 if (delta == 0)
312 return;
313 // rope by rope_angles(|delta|) (rope_inverse for a negative shift), one pair at a time so no table is needed
314 const T distance = T(delta < 0 ? -delta : delta), sign = delta < 0 ? T(-1) : T(1);
315 for (std::size_t i = 0; i < h; ++i) {
316 T f = rope_frequency<T>(i, 2 * h, base);
317 if (!factors.empty())
318 f = f / factors[i];
319 const T angle = distance * f, c = math::cos(angle), sn = sign * math::sin(angle);
320 T& a = layout == rope_layout::half ? v[i] : v[2 * i];
321 T& b = layout == rope_layout::half ? v[i + h] : v[2 * i + 1];
322 const T x1 = a, x2 = b;
323 a = x1 * c - x2 * sn;
324 b = x1 * sn + x2 * c;
325 }
326}
