96 // DCT-II of in(j), j < n, handing y_k for k < count to out(k, y_k).
97 template <typename T, typename In, typename Out>
98 constexpr void dct2(const fft_plan<T>& plan, In&& in, std::size_t count, Out&& out, std::span<complex<T>> work, dct_norm norm) noexcept
99 {
100 const std::size_t n = plan.size();
101 assert(work.size() >= dct_work_size(plan) && count <= n);
102 const auto v = work.first(n);
103 for (std::size_t j = 0; 2 * j < n; ++j)
104 v[j] = {in(2 * j), T(0)};
105 for (std::size_t j = 0; 2 * j + 1 < n; ++j)
106 v[n - 1 - j] = {in(2 * j + 1), T(0)};
107 plan.template execute<false>(v, work.subspan(n));
108 const T s0 = norm == dct_norm::ortho ? inv_sqrt<T>(4 * n) : T(1), s = norm == dct_norm::ortho ? inv_sqrt<T>(2 * n) : T(1);
109 for (std::size_t k = 0; k < count; ++k)
110 out(k, T(2) * (twiddle<T>(k, 4 * n) * v[k]).re * (k == 0 ? s0 : s));
111 }

113 // DCT-III of in(j), j < n, handing y_k for every k to out(k, y_k).
114 template <typename T, typename In, typename Out>
115 constexpr void dct3(const fft_plan<T>& plan, In&& in, Out&& out, std::span<complex<T>> work, dct_norm norm) noexcept
116 {
117 const std::size_t n = plan.size();
118 assert(work.size() >= dct_work_size(plan));
119 const auto v = work.first(n);
120 const T s0 = norm == dct_norm::ortho ? inv_sqrt<T>(n) : T(1), s = norm == dct_norm::ortho ? inv_sqrt<T>(2 * n) : T(1);
121 for (std::size_t k = 0; k < n; ++k) {
122 const T a = in(k) * (k == 0 ? s0 : s);
123 const T b = k == 0 ? T(0) : in(n - k) * s;
124 v[k] = conj(twiddle<T>(k, 4 * n)) * complex<T>{a, -b};
125 }
126 plan.template execute<true>(v, work.subspan(n));
127 for (std::size_t j = 0; 2 * j < n; ++j)
128 out(2 * j, v[j].re);
129 for (std::size_t j = 0; 2 * j + 1 < n; ++j)
130 out(2 * j + 1, v[n - 1 - j].re);
131 }

133 // DCT-IV of in(j), j < n, through a transform of length 2n.
134 template <typename T, typename In, typename Out>
135 constexpr void dct4(const fft_plan<T>& plan, In&& in, Out&& out, std::span<complex<T>> work, dct_norm norm) noexcept
136 {
137 const std::size_t m = plan.size(), n = m / 2;
138 assert(m % 2 == 0 && work.size() >= dct_work_size(plan));
139 const auto v = work.first(m);
140 for (std::size_t j = 0; j < n; ++j)
141 v[j] = twiddle<T>(j, 4 * n) * in(j);
142 for (std::size_t j = n; j < m; ++j)
143 v[j] = {};
144 plan.template execute<false>(v, work.subspan(m));
145 const T s = norm == dct_norm::ortho ? inv_sqrt<T>(2 * n) : T(1);
146 for (std::size_t k = 0; k < n; ++k)
147 out(k, T(2) * (twiddle<T>(2 * k + 1, 8 * n) * v[k]).re * s);
148 }

205template <math::math_number T>
206constexpr void dst2(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> x, std::span<std::type_identity_t<T>> out,

208{

256template <math::math_number T>
257constexpr void dct2_backward(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> grad_y, std::span<std::type_identity_t<T>> grad_x,
258 std::span<complex<std::type_identity_t<T>>> work, dct_norm norm = dct_norm::none) noexcept
259{
260 assert(grad_y.size() <= plan.size() && grad_x.size() == plan.size());
261 // Unnormalised, C2^T = C3 diag(2, 1, ..., 1); orthonormal, the transpose is the inverse, the orthonormal DCT-III.
262 const T first = norm == dct_norm::ortho ? T(1) : T(2);
263 detail::dct::dct3(plan, [grad_y, first](std::size_t k) { return k < grad_y.size() ? grad_y[k] * (k == 0 ? first : T(1)) : T(0); },
264 [grad_x](std::size_t j, T v) { grad_x[j] = v; }, work, norm);
265}
