110struct uuid
111{
113 std::array<std::uint8_t, 16> bytes{};
114
119 [[nodiscard]] static constexpr uuid nil() noexcept { return {}; }
124 [[nodiscard]] static constexpr uuid max() noexcept
125 {
126 uuid u;
127 u.bytes.fill(0xFF);
128 return u;
129 }
130
135 [[nodiscard]] static constexpr uuid from_u64(std::uint64_t high, std::uint64_t low) noexcept
136 {
137 uuid u;
138 for (std::size_t i = 0; i < 8; ++i) {
139 u.bytes[i] = static_cast<std::uint8_t>(high >> (56 - 8 * i));
140 u.bytes[8 + i] = static_cast<std::uint8_t>(low >> (56 - 8 * i));
141 }
142 return u;
143 }
144
147 [[nodiscard]] constexpr std::uint64_t high() const noexcept { return load(0); }
150 [[nodiscard]] constexpr std::uint64_t low() const noexcept { return load(8); }
151
154 [[nodiscard]] constexpr bool is_nil() const noexcept { return (high() | low()) == 0; }
157 [[nodiscard]] constexpr bool is_max() const noexcept { return (high() & low()) == ~std::uint64_t{0}; }
158
163 [[nodiscard]] constexpr unsigned version() const noexcept { return bytes[6] >> 4U; }
164
167 [[nodiscard]] constexpr uuid_variant variant() const noexcept
168 {
169 const unsigned b = bytes[8];
170 if ((b & 0x80U) == 0) return uuid_variant::ncs;
171 if ((b & 0x40U) == 0) return uuid_variant::rfc9562;
172 if ((b & 0x20U) == 0) return uuid_variant::microsoft;
173 return uuid_variant::future;
174 }
175
182 [[nodiscard]] constexpr std::optional<std::chrono::sys_time<std::chrono::milliseconds>> timestamp() const noexcept
183 {
184 if (version() != 7 || variant() != uuid_variant::rfc9562) return std::nullopt;
185 const auto ms = static_cast<std::chrono::milliseconds::rep>(high() >> 16);
186 return std::chrono::sys_time<std::chrono::milliseconds>(std::chrono::milliseconds(ms));
187 }
188
191 [[nodiscard]] std::span<const std::byte, 16> as_bytes() const noexcept
192 {
193 return std::span<const std::byte, 16>(reinterpret_cast<const std::byte*>(bytes.data()), 16);
194 }
195
200 friend constexpr bool operator==(const uuid& a, const uuid& b) noexcept
201 {
202 return a.high() == b.high() && a.low() == b.low();
203 }
204
209 friend constexpr std::strong_ordering operator<=>(const uuid& a, const uuid& b) noexcept
210 {
211 if (const auto c = a.high() <=> b.high(); c != 0) return c;
212 return a.low() <=> b.low();
213 }
214
215private:
216 // Big-endian load; compilers turn this into one load + byte swap.
217 constexpr std::uint64_t load(std::size_t at) const noexcept
218 {
219 std::uint64_t v = 0;
220 for (std::size_t i = 0; i < 8; ++i)
221 v = (v << 8) | bytes[at + i];
222 return v;
223 }

228// The 64-bit xmx finaliser snicholls::hash_combine uses (a bijection: full avalanche, no collisions of its own).
229constexpr std::uint64_t uuid_mix(std::uint64_t x) noexcept
230{
231 constexpr std::uint64_t m = 0xe9846af9b1a615dULL;
232 x ^= x >> 32;
233 x *= m;
234 x ^= x >> 32;
235 x *= m;
236 x ^= x >> 28;
237 return x;
238}

262{
263 uuid u{random};
264 u.bytes[6] = static_cast<std::uint8_t>((u.bytes[6] & 0x0FU) | 0x40U);

279[[nodiscard]] constexpr uuid make_uuid_v7(std::uint64_t unix_ts_ms, std::uint16_t rand_a, std::uint64_t rand_b) noexcept
280{
281 const std::uint64_t high = ((unix_ts_ms & 0xFFFF'FFFF'FFFFULL) << 16) | 0x7000U | (rand_a & 0x0FFFU);
282 const std::uint64_t low = 0x8000'0000'0000'0000ULL | (rand_b & 0x3FFF'FFFF'FFFF'FFFFULL);
283 return uuid::from_u64(high, low);

288template <typename Name>
289constexpr uuid make_uuid_v5_from(const uuid& name_space, Name name) noexcept
290{
291 sha1 h;
292 h.update(std::span<const std::uint8_t>(name_space.bytes));
293 h.update(name);
294 const sha1::digest_type digest = h.finish();
295 uuid u;
296 for (std::size_t i = 0; i < 16; ++i)
297 u.bytes[i] = digest[i]; // the first 128 of 160 bits
298 u.bytes[6] = static_cast<std::uint8_t>((u.bytes[6] & 0x0FU) | 0x50U);
299 u.bytes[8] = static_cast<std::uint8_t>((u.bytes[8] & 0x3FU) | 0x80U);
300 return u;
301}

342constexpr std::to_chars_result to_chars(char* first, char* last, const uuid& u) noexcept
343{
344 if (last - first < static_cast<std::ptrdiff_t>(uuid_text_size)) return {last, std::errc::value_too_large};
345 constexpr const char* digits = "0123456789abcdef";
346 char* p = first;
347 for (std::size_t i = 0; i < 16; ++i) {
348 if (i == 4 || i == 6 || i == 8 || i == 10) *p++ = '-';
349 *p++ = digits[u.bytes[i] >> 4U];
350 *p++ = digits[u.bytes[i] & 0x0FU];
351 }
352 return {p, std::errc{}};
353}

418// -1 for a non-hex character.
419inline constexpr std::array<std::int8_t, 256> uuid_hex_values = [] {
420 std::array<std::int8_t, 256> t{};
421 for (auto& v : t)
422 v = -1;
423 for (int c = 0; c < 10; ++c)
424 t[static_cast<std::size_t>('0' + c)] = static_cast<std::int8_t>(c);
425 for (int c = 0; c < 6; ++c) {
426 t[static_cast<std::size_t>('a' + c)] = static_cast<std::int8_t>(10 + c);
427 t[static_cast<std::size_t>('A' + c)] = static_cast<std::int8_t>(10 + c);
428 }
429 return t;
430}();

432// Offsets of the 16 byte pairs in the canonical and compact forms.
433inline constexpr std::array<std::uint8_t, 16> uuid_canonical_offsets{0, 2, 4, 6, 9, 11, 14, 16,
434 19, 21, 24, 26, 28, 30, 32, 34};
435
436// Decodes 16 hex pairs at p + offsets (canonical) or p + 2i (compact). false on any non-hex character.
437template <bool Canonical>
438constexpr bool decode_uuid_hex(const char* p, uuid& out) noexcept
439{
440 int bad = 0;
441 uuid u;
442 for (std::size_t i = 0; i < 16; ++i) {
443 const std::size_t at = Canonical ? uuid_canonical_offsets[i] : 2 * i;
444 const int hi = uuid_hex_values[static_cast<unsigned char>(p[at])];
445 const int lo = uuid_hex_values[static_cast<unsigned char>(p[at + 1])];
446 bad |= hi | lo; // negative (sign bit set) if either is not hex
447 u.bytes[i] = static_cast<std::uint8_t>(((hi & 0x0F) << 4) | (lo & 0x0F));
448 }
449 if (bad < 0) return false;
450 out = u;
451 return true;
452}

454constexpr const char* parse_canonical_uuid(const char* first, const char* last, uuid& out) noexcept
455{
456 if (last - first < static_cast<std::ptrdiff_t>(uuid_text_size)) return nullptr;
457 if (first[8] != '-' || first[13] != '-' || first[18] != '-' || first[23] != '-') return nullptr;
458 return decode_uuid_hex<true>(first, out) ? first + uuid_text_size : nullptr;
459}

461constexpr bool starts_with_urn_uuid(const char* first, const char* last) noexcept
462{
463 constexpr std::string_view prefix = "urn:uuid:";
464 if (last - first < static_cast<std::ptrdiff_t>(prefix.size())) return false;
465 for (std::size_t i = 0; i < prefix.size(); ++i) {
466 char c = first[i];
467 if (c >= 'A' && c <= 'Z') c = static_cast<char>(c - 'A' + 'a');
468 if (c != prefix[i]) return false;
469 }
470 return true;
471}

489constexpr std::from_chars_result from_chars(const char* first, const char* last, uuid& out,
490 uuid_parse_options options = {}) noexcept
491{
492 const std::from_chars_result fail{first, std::errc::invalid_argument};
493 if (first == last) return fail;
494 uuid u;
495 if (*first == '{') {
496 if (!options.braces) return fail;
497 const char* end = detail::parse_canonical_uuid(first + 1, last, u);
498 if (end == nullptr || end == last || *end != '}') return fail;
499 out = u;
500 return {end + 1, std::errc{}};
501 }
502 if (*first == 'u' || *first == 'U') {
503 if (!options.urn || !detail::starts_with_urn_uuid(first, last)) return fail;
504 const char* end = detail::parse_canonical_uuid(first + 9, last, u);
505 if (end == nullptr) return fail;
506 out = u;
507 return {end, std::errc{}};
508 }
509 if (const char* end = detail::parse_canonical_uuid(first, last, u); end != nullptr) {
510 out = u;
511 return {end, std::errc{}};
512 }
513 if (options.compact && last - first >= 32 && detail::decode_uuid_hex<false>(first, u)) {
514 out = u;
515 return {first + 32, std::errc{}};
516 }
517 return fail;
518}

527template <detail::uuid_char_string S>
528[[nodiscard]] constexpr std::optional<uuid> parse_uuid(const S& text, uuid_parse_options options = {}) noexcept
529{
530 const auto sv = to_string_view(text);
531 const char* const last = sv.data() + sv.size();
532 uuid u;
533 const auto [end, ec] = from_chars(sv.data(), last, u, options);
534 if (ec != std::errc{} || end != last) return std::nullopt;
535 return u;
536}
