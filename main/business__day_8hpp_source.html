119 // The first business day at or after `day` stepping by `step` (+1 or -1) days, within the search limit.
120 template <typename C>
121 constexpr result<std::chrono::sys_days, business_day_errc> roll(std::chrono::sys_days day, int step, const C& calendar)
122 {
123 for (int i = 0; i <= max_business_day_search; ++i, day += std::chrono::days(step))
124 if (static_cast<bool>(calendar.is_business_day(day)))
125 return ok(day);
126 return err(business_day_errc::no_business_day);
127 }

129 constexpr std::chrono::month month_of(std::chrono::sys_days day) noexcept
130 {
131 return std::chrono::year_month_day(day).month();
132 }
133
134 // Roll by `step`; if that leaves the month, roll the other way.
135 template <typename C>
136 constexpr result<std::chrono::sys_days, business_day_errc> roll_modified(std::chrono::sys_days day, int step, const C& calendar)
137 {
138 const auto r = roll(day, step, calendar);
139 if (r && month_of(*r) != month_of(day))
140 return roll(day, -step, calendar);
141 return r;
142 }
143 } // namespace detail

153 template <holiday_calendar C>
154 [[nodiscard]] constexpr result<std::chrono::sys_days, business_day_errc>
155 adjust(std::chrono::sys_days day, business_day_convention convention, const C& calendar) noexcept(noexcept(calendar.is_business_day(day)))
156 {
157 switch (convention) {
158 case business_day_convention::following: return detail::roll(day, 1, calendar);
159 case business_day_convention::modified_following: return detail::roll_modified(day, 1, calendar);
160 case business_day_convention::preceding: return detail::roll(day, -1, calendar);
161 case business_day_convention::modified_preceding: return detail::roll_modified(day, -1, calendar);
162 case business_day_convention::unadjusted: break;
163 }
164 return ok(day);
165 }
