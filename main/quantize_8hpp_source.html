446 // C's conversion of a float to an integer as x86 performs it (cvttss2si): toward zero, and INT_MIN for NaN or
447 // a value out of range. ggml's code meets those on degenerate blocks (a subnormal amax makes its inverse
448 // infinite), where C leaves the conversion undefined; this is ggml's result on x86 (ARM saturates instead)
449 constexpr int to_int(float x)
450 {
451 return x >= -2147483648.0f && x < 2147483648.0f ? static_cast<int>(x) : std::numeric_limits<int>::min();
452 }

454 // roundf, then the conversion to an integer: halves away from zero, the fraction x - trunc(x) is exact
455 constexpr int round_to_int(float x)
456 {
457 if (!(x >= -2147483648.0f && x < 2147483648.0f))
458 return std::numeric_limits<int>::min();
459 const int t = static_cast<int>(x);
460 const float frac = x - static_cast<float>(t);
461 return t + (frac >= 0.5f ? 1 : 0) - (frac <= -0.5f ? 1 : 0);
462 }

494 // ggml's make_qx_quants: a symmetric scale for n numbers, codes L in [0, 2 nmax), searched over 19 scales when
495 // rmse_type is not 0, weighted by qw (or by x^2, 1, |x|, sqrt|x| for rmse_type 1 to 4 when qw is null)
496 constexpr float make_qx_quants(int n, int nmax, const float* x, std::int8_t* L, int rmse_type, const float* qw)
497 {
498 float max = 0, amax = 0;
499 for (int i = 0; i < n; ++i) {
500 const float ax = abs(x[i]);
501 if (ax > amax) {
502 amax = ax;
503 max = x[i];
504 }
505 }
506 if (amax < group_max_eps) {
507 for (int i = 0; i < n; ++i)
508 L[i] = 0;
509 return 0.0f;
510 }
511 float iscale = float(-nmax) / max;
512 if (rmse_type == 0) {
513 for (int i = 0; i < n; ++i) {
514 const float v = iscale * x[i];
515 L[i] = static_cast<std::int8_t>(nmax + imax(-nmax, imin(nmax - 1, nearest_int(v))));
516 }
517 return 1 / iscale;
518 }
519 bool return_early = false;
520 if (rmse_type < 0) {
521 rmse_type = -rmse_type;
522 return_early = true;
523 }
524 const auto weight = [&](int i) {
525 if (qw)
526 return qw[i];
527 if (rmse_type == 1) {
528 const float w = x[i] * x[i];
529 return w;
530 }
531 return rmse_type == 2 ? 1.0f : rmse_type == 3 ? abs(x[i]) : sqrt(abs(x[i]));
532 };
533 float sumlx = 0, suml2 = 0;
534 for (int i = 0; i < n; ++i) {
535 const float v = iscale * x[i];
536 const int l = imax(-nmax, imin(nmax - 1, nearest_int(v)));
537 L[i] = static_cast<std::int8_t>(l + nmax);
538 const float w = weight(i);
539 const float wx = w * x[i];
540 const float wxl = wx * float(l);
541 sumlx += wxl;
542 const float wl = w * float(l);
543 const float wll = wl * float(l);
544 suml2 += wll;
545 }
546 float scale = suml2 != 0.0f ? sumlx / suml2 : 0.0f;
547 if (return_early) {
548 const float inv = 1 / iscale;
549 const float sum = scale + inv;
550 return suml2 > 0 ? 0.5f * sum : inv;
551 }
552 float best = scale * sumlx;
553 for (int is = -9; is <= 9; ++is) {
554 if (is == 0)
555 continue;
556 const float step = 0.1f * float(is);
557 const float top = float(nmax) + step;
558 iscale = -top / max;
559 sumlx = suml2 = 0;
560 for (int i = 0; i < n; ++i) {
561 const float v = iscale * x[i];
562 const int l = imax(-nmax, imin(nmax - 1, nearest_int(v)));
563 const float w = weight(i);
564 const float wx = w * x[i];
565 const float wxl = wx * float(l);
566 sumlx += wxl;
567 const float wl = w * float(l);
568 const float wll = wl * float(l);
569 suml2 += wll;
570 }
571 const float lhs = sumlx * sumlx;
572 const float rhs = best * suml2;
573 if (suml2 > 0 && lhs > rhs) {
574 for (int i = 0; i < n; ++i) {
575 const float v = iscale * x[i];
576 L[i] = static_cast<std::int8_t>(nmax + imax(-nmax, imin(nmax - 1, nearest_int(v))));
577 }
578 scale = sumlx / suml2;
579 best = scale * sumlx;
580 }
581 }
582 return scale;
583 }

585 // ggml's make_qkx2_quants (weights given) and make_qkx3_quants (weights or x^2): an affine scale and minimum for n
586 // numbers, codes L in [0, nmax], searched over nstep + 1 scales; the minimum, negated, goes to the_min
587 template <bool Qkx3>
588 constexpr float make_qkx_quants(int n, int nmax, const float* x, const float* weights, std::uint8_t* L, float* the_min,
589 std::uint8_t* Laux, float rmin, float rdelta, int nstep, bool use_mad)
590 {
591 const auto weight = [&](int i) {
592 if (weights)
593 return weights[i];
594 const float w = x[i] * x[i];
595 return w;
596 };
597 float min = x[0], max = x[0];
598 float sum_w = weight(0);
599 float sum_x = sum_w * x[0];
600 for (int i = 1; i < n; ++i) {
601 if (x[i] < min)
602 min = x[i];
603 if (x[i] > max)
604 max = x[i];
605 const float w = weight(i);
606 sum_w += w;
607 const float wx = w * x[i];
608 sum_x += wx;
609 }
610 if (min > 0)
611 min = 0;
612 if (Qkx3 ? max <= min : max == min) {
613 for (int i = 0; i < n; ++i)
614 L[i] = 0;
615 *the_min = -min;
616 return 0.0f;
617 }
618 float iscale = float(nmax) / (max - min);
619 float scale = 1 / iscale;
620 float best_error = 0;
621 const auto error = [&](float s, float m, const std::uint8_t* codes) {
622 float e = 0;
623 for (int i = 0; i < n; ++i) {
624 const float sl = s * float(codes[i]);
625 const float slm = sl + m;
626 float diff = slm - x[i];
627 if (use_mad) {
628 diff = abs(diff);
629 } else {
630 const float d2 = diff * diff;
631 diff = d2;
632 }
633 const float wd = weight(i) * diff;
634 e += wd;
635 }
636 return e;
637 };
638 for (int i = 0; i < n; ++i) {
639 const float v = iscale * (x[i] - min);
640 L[i] = static_cast<std::uint8_t>(imax(0, imin(nmax, nearest_int(v))));
641 }
642 best_error = error(scale, min, L);
643 if (nstep < 1) {
644 *the_min = -min;
645 return scale;
646 }
647 for (int is = 0; is <= nstep; ++is) {
648 const float step = rdelta * float(is);
649 const float top = rmin + step;
650 const float top2 = top + float(nmax);
651 iscale = top2 / (max - min); // min is the best fit so far, which moves
652 float sum_l = 0, sum_l2 = 0, sum_xl = 0;
653 for (int i = 0; i < n; ++i) {
654 const float v = iscale * (x[i] - min);
655 int l = 0;
656 if constexpr (Qkx3)
657 l = v > 0 ? nearest_int(v < float(nmax) ? v : float(nmax)) : 0;
658 else
659 l = imax(0, imin(nmax, nearest_int(v)));
660 Laux[i] = static_cast<std::uint8_t>(l);
661 const float w = weight(i);
662 const float wl = w * float(l);
663 sum_l += wl;
664 const float wll = wl * float(l);
665 sum_l2 += wll;
666 const float wlx = wl * x[i];
667 sum_xl += wlx;
668 }
669 const float a = sum_w * sum_l2;
670 const float b = sum_l * sum_l;
671 const float D = a - b;
672 if (D > 0) {
673 const float s1 = sum_w * sum_xl;
674 const float s2 = sum_x * sum_l;
675 float this_scale = (s1 - s2) / D;
676 const float m1 = sum_l2 * sum_x;
677 const float m2 = sum_l * sum_xl;
678 float this_min = (m1 - m2) / D;
679 if (this_min > 0) {
680 this_min = 0;
681 this_scale = sum_xl / sum_l2;
682 }
683 const float cur_error = error(this_scale, this_min, Laux);
684 if (cur_error < best_error) {
685 for (int i = 0; i < n; ++i)
686 L[i] = Laux[i];
687 best_error = cur_error;
688 scale = this_scale;
689 min = this_min;
690 }
691 }
692 }
693 *the_min = -min;
694 return scale;
695 }

697 // ggml's make_qp_quants: a positive scale for n non-negative numbers (scales of sub-blocks), codes L in [0, nmax],
698 // weighted by quant_weights
699 constexpr float make_qp_quants(int n, int nmax, const float* x, std::uint8_t* L, const float* quant_weights)
700 {
701 float max = 0;
702 for (int i = 0; i < n; ++i)
703 max = fmax(max, x[i]);
704 if (max < group_max_eps) {
705 for (int i = 0; i < n; ++i)
706 L[i] = 0;
707 return 0.0f;
708 }
709 float iscale = float(nmax) / max;
710 for (int i = 0; i < n; ++i) {
711 const float v = iscale * x[i];
712 L[i] = static_cast<std::uint8_t>(nearest_int(v));
713 }
714 const float scale = 1 / iscale;
715 const auto mse_of = [&](float s, auto code) {
716 float mse = 0;
717 for (int i = 0; i < n; ++i) {
718 const float sl = s * float(code(i));
719 const float diff = x[i] - sl;
720 const float wd = quant_weights[i] * diff;
721 const float wdd = wd * diff;
722 mse += wdd;
723 }
724 return mse;
725 };
726 float best_mse = mse_of(scale, [&](int i) { return int(L[i]); });
727 for (int is = -4; is <= 4; ++is) {
728 if (is == 0)
729 continue;
730 const float step = 0.1f * float(is);
731 const float top = step + float(nmax);
732 const float iscale_is = top / max;
733 const float scale_is = 1 / iscale_is;
734 const float mse = mse_of(scale_is, [&](int i) {
735 const float v = iscale_is * x[i];
736 return imin(nmax, nearest_int(v));
737 });
738 if (mse < best_mse) {
739 best_mse = mse;
740 iscale = iscale_is;
741 }
742 }
743 float sumlx = 0, suml2 = 0;
744 for (int i = 0; i < n; ++i) {
745 const float v = iscale * x[i];
746 const int l = imin(nmax, nearest_int(v));
747 L[i] = static_cast<std::uint8_t>(l);
748 const float w = quant_weights[i];
749 const float wx = w * x[i];
750 const float wxl = wx * float(l);
751 sumlx += wxl;
752 const float wl = w * float(l);
753 const float wll = wl * float(l);
754 suml2 += wll;
755 }
756 for (int itry = 0; itry < 5; ++itry) {
757 int n_changed = 0;
758 for (int i = 0; i < n; ++i) {
759 const float w = quant_weights[i];
760 const float wx = w * x[i];
761 const float wxl = wx * float(L[i]);
762 float slx = sumlx - wxl;
763 const float wl = w * float(L[i]);
764 const float wll = wl * float(L[i]);
765 float sl2 = suml2 - wll;
766 if (slx > 0 && sl2 > 0) {
767 const float xs = x[i] * sl2;
768 const float v = xs / slx;
769 const int new_l = imin(nmax, nearest_int(v));
770 if (new_l != int(L[i])) {
771 const float wxn = wx * float(new_l);
772 slx += wxn;
773 const float wn = w * float(new_l);
774 const float wnn = wn * float(new_l);
775 sl2 += wnn;
776 const float s1 = slx * slx;
777 const float lhs = s1 * suml2;
778 const float s2 = sumlx * sumlx;
779 const float rhs = s2 * sl2;
780 if (lhs > rhs) {
781 L[i] = static_cast<std::uint8_t>(new_l);
782 sumlx = slx;
783 suml2 = sl2;
784 ++n_changed;
785 }
786 }
787 }
788 }
789 if (n_changed == 0)
790 break;
791 }
792 return suml2 > 0.0f ? sumlx / suml2 : 0.0f;
793 }

795 // ggml's sigma2 of a row for the importance weights: the mean square (sum x^2 / n_per_row)
796 constexpr float mean_square(std::span<const float> x)
797 {
798 float sum_x2 = 0;
799 for (const float v : x) {
800 const float v2 = v * v;
801 sum_x2 += v2;
802 }
803 return sum_x2 / float(x.size());
804 }

814 // ggml's ggml_fp32_to_ue4m3 and ggml_ue4m3_to_fp32 (which returns half the value, to pair with doubled E2M1)
815 constexpr std::uint8_t fp32_to_ue4m3(float x)
816 {
817 if (!(x > 0.0f))
818 return 0;
819 if (x > 448.0f)
820 x = 448.0f;
821 const std::uint32_t bits = std::bit_cast<std::uint32_t>(x);
822 const int fp32_exp = int((bits >> 23) & 0xff) - 127;
823 const int fp32_man = int((bits >> 20) & 0x7);
824 int exp = fp32_exp + 7;
825 if (exp <= 0) {
826 const float scaled = x * 512.0f;
827 const float shifted = scaled + 0.5f;
828 int man = static_cast<int>(shifted);
829 man = man > 7 ? 7 : man;
830 return man < 1 ? std::uint8_t{0} : static_cast<std::uint8_t>(man);
831 }
832 if (exp >= 15)
833 return 0x7e;
834 int man = fp32_man + int((bits >> 19) & 1);
835 if (man > 7) {
836 man = 0;
837 if (++exp >= 15)
838 return 0x7e;
839 }
840 return static_cast<std::uint8_t>((exp << 3) | man);
841 }
842
843 constexpr float ue4m3_half(std::uint8_t x)
844 {
845 if (x == 0 || x == 0x7f)
846 return 0.0f;
847 const int exp = (x >> 3) & 0xf;
848 const int man = x & 0x7;
849 // ldexpf((float) man, -9) or ldexpf(1 + man / 8, exp - 7), then halved: every step exact
850 const float raw = exp == 0 ? float(man) * 0x1p-9f
851 : std::bit_cast<float>(static_cast<std::uint32_t>(exp - 7 + 127) << 23) * (1.0f + float(man) / 8.0f);
852 return raw * 0.5f;
853 }

856 {
857 std::uint8_t best = 0;

869 // Q4_0's code for v at inverse scale id, offset 8.5 (or 16.5 for Q5_0): MIN(top, (int8_t)(v · id + offset))
870 constexpr std::uint8_t offset_code(float v, float id, float offset, int top)
871 {
872 const float scaled = v * id;
873 const float shifted = scaled + offset;
874 const int q = static_cast<std::int8_t>(to_int(shifted));
875 return static_cast<std::uint8_t>(q < top ? q : top);

878 // Q4_1's code for v: MIN(15, (int8_t)((v - min) · id + 0.5)); Q5_1's has no MIN: (uint8_t)(...)

885 const int q = static_cast<std::int8_t>(to_int(shifted));
886 return static_cast<std::uint8_t>(q < top ? q : top);
