108constexpr std::int64_t days_from_civil(std::int64_t y, unsigned m, unsigned d) noexcept
109{
110 y -= m <= 2;
111 const std::int64_t era = (y >= 0 ? y : y - 399) / 400;
112 const auto yoe = static_cast<unsigned>(y - era * 400); // [0, 399]
113 const unsigned doy = (153 * (m > 2 ? m - 3 : m + 9) + 2) / 5 + d - 1; // [0, 365]
114 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy; // [0, 146096]
115 return era * 146097 + static_cast<std::int64_t>(doe) - 719468;
116}

122constexpr civil_date civil_from_days(std::int64_t z) noexcept
123{
124 z += 719468;
125 const std::int64_t era = (z >= 0 ? z : z - 146096) / 146097;
126 const auto doe = static_cast<unsigned>(z - era * 146097); // [0, 146096]
127 const unsigned yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365; // [0, 399]
128 const unsigned doy = doe - (365 * yoe + yoe / 4 - yoe / 100); // [0, 365]
129 const unsigned mp = (5 * doy + 2) / 153; // [0, 11]
130 const unsigned d = doy - (153 * mp + 2) / 5 + 1; // [1, 31]
131 const unsigned m = mp < 10 ? mp + 3 : mp - 9; // [1, 12]
132 return {static_cast<std::int64_t>(yoe) + era * 400 + (m <= 2), m, d};
133}

143constexpr civil_date civil_from_days_fast(std::int64_t z) noexcept
144{
