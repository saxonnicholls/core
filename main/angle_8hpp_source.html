117 template <linalg::scalar T>
118 [[nodiscard]] constexpr T wrap_pi(T x) noexcept(linalg::detail::nothrow_scalar<T>)
119 {
120 const T two_pi = T(2) * pi<T>;
121 T y = x + pi<T>;
122 y -= two_pi * static_cast<T>(math::floor(y / two_pi));
123 if (y >= two_pi)
124 y -= two_pi;
125 if (y < T(0))
126 y = T(0);
127 return y - pi<T>;
128 }

147 template <linalg::scalar T>
148 [[nodiscard]] constexpr T wrap_elevation(T x) noexcept(linalg::detail::nothrow_scalar<T>)
149 {
150 if (!(x == x) || x == std::numeric_limits<T>::infinity() || x == -std::numeric_limits<T>::infinity())
151 return std::numeric_limits<T>::quiet_NaN(); // no direction; and the fold below would cast NaN to int
152 const T two_pi = T(2) * pi<T>;
153 T y = x - two_pi * static_cast<T>(math::floor(x / two_pi));
154 if (y >= two_pi)
155 y -= two_pi;
156 const T quarter = pi<T> / T(2);
157 const auto n = static_cast<int>(math::floor(y / quarter));
158 if (n == 1 || n == 2)
159 return pi<T> - y;
160 if (n == 3)
161 return y - two_pi;
162 if (n >= 4)
163 return T(0);
164 return y;
165 }

226 template <linalg::scalar T>
227 [[nodiscard]] constexpr T circular_mean(std::span<const T> angles, std::span<const T> weights = {}) noexcept(linalg::detail::nothrow_scalar<T>)
228 {
229 T s(0), c(0);
230 for (std::size_t i = 0; i < angles.size(); ++i) {
231 const T w = weights.empty() ? T(1) : weights[i];
232 s += w * static_cast<T>(math::sin(angles[i]));
233 c += w * static_cast<T>(math::cos(angles[i]));
234 }
235 if (s == T(0) && c == T(0))
236 return T(0);
237 return static_cast<T>(math::atan2(s, c));
238 }

249 template <auto Kinds, linalg::scalar T, std::size_t N, std::size_t P>
250 [[nodiscard]] constexpr linalg::fixed_vector<T, N> weighted_mean(const std::array<linalg::fixed_vector<T, N>, P>& points, const std::array<T, P>& weights) noexcept(linalg::detail::nothrow_scalar<T>)
251 {
252 static_assert(Kinds.size() == N, "snicholls::geometry::weighted_mean: one angle_kind per component");
253 linalg::fixed_vector<T, N> m{};
254 for (std::size_t i = 0; i < N; ++i) {
255 if (Kinds[i] == angle_kind::linear) {
256 T s(0);
257 for (std::size_t p = 0; p < P; ++p)
258 s += weights[p] * points[p][i];
259 m[i] = s;
260 } else {
261 std::array<T, P> a{};
262 for (std::size_t p = 0; p < P; ++p)
263 a[p] = points[p][i];
264 m[i] = wrap(Kinds[i], circular_mean<T>(std::span<const T>(a), std::span<const T>(weights)));
265 }
266 }
267 return m;
268 }
