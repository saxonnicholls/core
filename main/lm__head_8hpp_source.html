81template <typename S, row H, mutable_row L>
82 requires std::same_as<row_value_t<H>, row_value_t<L>>
83constexpr row_value_t<H> lm_head_loss(const dense_weights<S>& head, const H& hs, std::span<const std::size_t> targets,
84 L&& logits)
85{
86 using T = row_value_t<H>;
87 const auto h = as_row(hs);
88 const auto z = as_mutable_row(logits);
89 const std::size_t n = targets.size(), d = head.cols;
90 assert(h.size() == n * d && z.size() == head.rows);
91 T sum = T(0);
92 for (std::size_t t = 0; t < n; ++t) {
93 assert(targets[t] < head.rows);
94 linear(head, h.subspan(t * d, d), z);
95 sum += cross_entropy(z, targets[t]);
96 }
97 return n == 0 ? T(0) : sum / T(n);
98}

108template <typename S, row H, mutable_row L, mutable_row DH, mutable_row DW>
109 requires std::same_as<row_value_t<H>, row_value_t<L>> && std::same_as<row_value_t<H>, row_value_t<DH>>
110 && std::same_as<row_value_t<H>, row_value_t<DW>>
111constexpr row_value_t<H> lm_head_loss(const dense_weights<S>& head, const H& hs, std::span<const std::size_t> targets,
112 L&& logits, DH&& dhs, DW&& dhead)
113{
114 using T = row_value_t<H>;
115 const auto h = as_row(hs);
116 const auto z = as_mutable_row(logits);
117 const auto gh = as_mutable_row(dhs);
118 const std::size_t n = targets.size(), d = head.cols;
119 assert(h.size() == n * d && z.size() == head.rows && gh.size() == h.size());
120 T sum = T(0);
121 for (std::size_t t = 0; t < n; ++t) {
122 assert(targets[t] < head.rows);
123 const auto ht = h.subspan(t * d, d);
124 linear(head, ht, z);
125 sum += cross_entropy_with_gradient(z, targets[t], z, T(1) / T(n));
126 linear_backward(head, ht, std::span<const T>(z), gh.subspan(t * d, d), dhead);
127 }
128 return n == 0 ? T(0) : sum / T(n);
129}

151template <math::math_number T, typename B>
152 requires linalg::blas_backend<B, T>
153constexpr T lm_head_loss_sequence(const B& backend, const dense_weights<T>& head, std::type_identity_t<std::span<const T>> hs,
154 std::span<const std::uint32_t> targets, std::type_identity_t<std::span<T>> logits,
155 std::type_identity_t<std::span<T>> losses, std::type_identity_t<std::span<T>> dhs,
156 std::type_identity_t<std::span<T>> dhead, std::size_t classes = 0, std::type_identity_t<T> weight = T(1),
157 std::size_t denominator = 0)
158{
159 const std::size_t n = targets.size(), m = denominator == 0 ? n : denominator;
160 assert(hs.size() == n * head.cols && logits.size() == n * head.rows && losses.size() == n && dhs.size() == hs.size()
161 && classes <= head.rows);
162 if (n == 0)
163 return T(0);
164 linear_sequence(backend, head, hs, logits);
165 const T scale = weight == T(1) ? T(1) / T(m) : weight / T(m);
166 if (classes == 0 || classes == head.rows)
167 cross_entropy_rows(backend, logits, targets, losses, scale);
168 else
169 cross_entropy_rows(backend, logits, targets, losses, scale, classes);
170 linear_sequence_backward(backend, head, hs, std::span<const T>(logits), dhs, dhead);
171 const T mean = sum_values(backend, std::span<const T>(losses)) / T(m);
172 return weight == T(1) ? mean : weight * mean;
173}
