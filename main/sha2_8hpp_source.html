103 // One compression for both word sizes: the round constants, the rotation amounts and the round count are the
104 // only differences (FIPS 180-4 6.2.2 and 6.4.2). Eight rounds at a time with the working variables' roles
105 // rotated through the calls instead of moved, and the schedule kept in a ring of 16 words.
106 template <typename W, std::size_t Rounds, const W* K, unsigned S0a, unsigned S0b, unsigned S0c, unsigned S1a,
107 unsigned S1b, unsigned S1c, unsigned s0a, unsigned s0b, unsigned s0c, unsigned s1a, unsigned s1b,
108 unsigned s1c>
109 constexpr void sha2_compress(std::array<W, 8>& h, const W (&block)[16]) noexcept
110 {
111 W w[16]{};
112 for (std::size_t i = 0; i < 16; ++i)
113 w[i] = block[i];
114 W a = h[0], b = h[1], c = h[2], d = h[3], e = h[4], f = h[5], g = h[6], hh = h[7];
115 const auto round = [](W x0, W x1, W x2, W& x3, W x4, W x5, W x6, W& x7, W k, W wt) {
116 const W t1 = static_cast<W>(x7 + (rotr(x4, S1a) ^ rotr(x4, S1b) ^ rotr(x4, S1c)) + ((x4 & x5) ^ (~x4 & x6)) + k + wt);
117 const W t2 = static_cast<W>((rotr(x0, S0a) ^ rotr(x0, S0b) ^ rotr(x0, S0c)) + ((x0 & x1) ^ (x0 & x2) ^ (x1 & x2)));
118 x3 = static_cast<W>(x3 + t1);
119 x7 = static_cast<W>(t1 + t2);
120 };
121 const auto schedule = [&w](std::size_t t) {
122 const W x = w[(t + 1) & 15], y = w[(t + 14) & 15];
123 const W sig0 = rotr(x, s0a) ^ rotr(x, s0b) ^ static_cast<W>(x >> s0c);
124 const W sig1 = rotr(y, s1a) ^ rotr(y, s1b) ^ static_cast<W>(y >> s1c);
125 w[t & 15] = static_cast<W>(w[t & 15] + sig0 + w[(t + 9) & 15] + sig1);
126 return w[t & 15];
127 };
128 for (std::size_t t = 0; t < Rounds; t += 8) {
129 if (t >= 16)
130 for (std::size_t i = 0; i < 8; ++i)
131 schedule(t + i);
132 round(a, b, c, d, e, f, g, hh, K[t], w[t & 15]);
133 round(hh, a, b, c, d, e, f, g, K[t + 1], w[(t + 1) & 15]);
134 round(g, hh, a, b, c, d, e, f, K[t + 2], w[(t + 2) & 15]);
135 round(f, g, hh, a, b, c, d, e, K[t + 3], w[(t + 3) & 15]);
136 round(e, f, g, hh, a, b, c, d, K[t + 4], w[(t + 4) & 15]);
137 round(d, e, f, g, hh, a, b, c, K[t + 5], w[(t + 5) & 15]);
138 round(c, d, e, f, g, hh, a, b, K[t + 6], w[(t + 6) & 15]);
139 round(b, c, d, e, f, g, hh, a, K[t + 7], w[(t + 7) & 15]);
140 }
141 h[0] = static_cast<W>(h[0] + a);
142 h[1] = static_cast<W>(h[1] + b);
143 h[2] = static_cast<W>(h[2] + c);
144 h[3] = static_cast<W>(h[3] + d);
145 h[4] = static_cast<W>(h[4] + e);
146 h[5] = static_cast<W>(h[5] + f);
147 h[6] = static_cast<W>(h[6] + g);
148 h[7] = static_cast<W>(h[7] + hh);
149 }

152 struct sha256_policy
153 {
154 using word = std::uint32_t;
155 using state = std::array<word, 8>;
156 static constexpr std::size_t block_size = 64;
157 static constexpr std::size_t length_size = 8;
158 static constexpr std::size_t digest_size = DigestSize;
159 static constexpr bool big_endian = true;
160 static constexpr state initial = Initial;
161 static constexpr void compress(state& h, const word (&w)[16]) noexcept
162 {
