79 // The row of an alphabetic code, or iso4217_rows.size(). An index, not a pointer: GCC cannot compare a
80 // pointer into the table with nullptr in a constant expression.
81 constexpr std::size_t iso4217_index(std::string_view code) noexcept
82 {
83 std::size_t lo = 0, hi = iso4217_rows.size();
84 while (lo < hi) {
85 const std::size_t mid = lo + (hi - lo) / 2;
86 const std::string_view c = iso4217_rows[mid].code;
87 if (c == code)
88 return mid;
89 if (c < code)
90 lo = mid + 1;
91 else
92 hi = mid;
93 }
94 return iso4217_rows.size();
95 }
96 } // namespace detail

121 [[nodiscard]] constexpr const iso4217_entry* find_iso4217(std::uint16_t numeric) noexcept
122 {
123 const iso4217_entry* historic = nullptr;
124 for (const auto& e : detail::iso4217_rows) {
125 if (e.numeric != numeric)
126 continue;
127 if (e.withdrawn.empty())
128 return &e;
129 historic = historic ? historic : &e;
130 }
131 return historic;
132 }

137 [[nodiscard]] constexpr std::optional<asset> iso4217_asset(std::string_view code) noexcept
138 {
139 const std::size_t i = detail::iso4217_index(code);
140 if (i >= detail::iso4217_rows.size() || detail::iso4217_rows[i].scale < 0)
141 return std::nullopt;
142 return make_asset(detail::iso4217_rows[i].code, static_cast<unsigned>(detail::iso4217_rows[i].scale));
