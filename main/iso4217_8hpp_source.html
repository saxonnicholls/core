100 // The row of an alphabetic code, or iso4217_rows.size(). An index, not a pointer: GCC cannot compare a
101 // pointer into the table with nullptr in a constant expression.
102 constexpr std::size_t iso4217_index(std::string_view code) noexcept
103 {
104 std::size_t lo = 0, hi = iso4217_rows.size();
105 while (lo < hi) {
106 const std::size_t mid = lo + (hi - lo) / 2;
107 const std::string_view c = iso4217_rows[mid].code;
108 if (c == code)
109 return mid;
110 if (c < code)
111 lo = mid + 1;
112 else
113 hi = mid;
114 }
115 return iso4217_rows.size();
116 }
117 } // namespace detail

147 [[nodiscard]] constexpr const iso4217_entry* find_iso4217(std::uint16_t numeric) noexcept
148 {
149 const iso4217_entry* historic = nullptr;
150 for (const auto& e : detail::iso4217_rows) {
151 if (e.numeric != numeric)
152 continue;
153 if (e.withdrawn.empty())
154 return &e;
155 historic = historic ? historic : &e;
156 }
157 return historic;
158 }

163 [[nodiscard]] constexpr std::optional<asset> iso4217_asset(std::string_view code) noexcept
164 {
165 const std::size_t i = detail::iso4217_index(code);
166 if (i >= detail::iso4217_rows.size() || detail::iso4217_rows[i].scale < 0)
167 return std::nullopt;
168 return make_asset(detail::iso4217_rows[i].code, static_cast<unsigned>(detail::iso4217_rows[i].scale));
169 }
