66 "example.test wants you to sign in with your Ethereum account:\n"

69 namespace detail {
70 template <typename Probe>
71 concept wallet_verifier_probe = std::is_default_constructible_v<Probe> &&
72 wallet_verifier<std::remove_cvref_t<decltype(std::declval<Probe&>().verifier())>> &&
73 requires(Probe& p, std::string_view m, capability_errc c, std::string_view r) {
74 { p.signer_address() } -> std::convertible_to<std::string>;
75 { p.other_address() } -> std::convertible_to<std::string>;
76 { p.sign(m) } -> std::convertible_to<std::vector<std::byte>>;
77 p.provider_fails(c, r);
78 };

80 template <typename Probe>
81 struct wallet_kit_case {
82 std::string address;
83 std::vector<std::byte> signature;
84 explicit wallet_kit_case(Probe& p) : address(p.signer_address()), signature(p.sign(wallet_conformance_message)) {}
85 };

94 [[nodiscard]] capability_conformance_report run_wallet_verifier_conformance()
95 {
96 capability_conformance_report report;
97 using detail::wallet_kit_case;
98 constexpr auto message = wallet_conformance_message;
99 const auto is = [](const capability_result<void>& r, capability_errc code) { return r.is_err() && r.unwrap_err().code == code; };
100
101 detail::capability_check<Probe>(report, "the signer verifies", [&](Probe& p, const auto& expect) {
102 const wallet_kit_case<Probe> c{p};
103 expect(p.verifier().verify({1, c.address, message, c.signature}).is_ok(), "the signer's signature verifies");
104 });
105
106 detail::capability_check<Probe>(report, "another address is signature_invalid", [&](Probe& p, const auto& expect) {
107 const wallet_kit_case<Probe> c{p};
108 const std::string other = p.other_address();
109 expect(is(p.verifier().verify({1, other, message, c.signature}), capability_errc::signature_invalid),
110 "the signature does not verify for another wallet");
111 });
112
113 detail::capability_check<Probe>(report, "a tampered message is signature_invalid", [&](Probe& p, const auto& expect) {
114 const wallet_kit_case<Probe> c{p};
115 std::string tampered(message);
116 tampered[0] = 'E';
117 expect(is(p.verifier().verify({1, c.address, tampered, c.signature}), capability_errc::signature_invalid),
118 "one changed character is signature_invalid");
119 });
120
121 detail::capability_check<Probe>(report, "a malformed signature is signature_invalid", [&](Probe& p, const auto& expect) {
122 wallet_kit_case<Probe> c{p};
123 if (!c.signature.empty())
124 c.signature.pop_back();
125 expect(is(p.verifier().verify({1, c.address, message, c.signature}), capability_errc::signature_invalid),
126 "a signature one byte short is signature_invalid");
127 expect(p.verifier().verify({1, c.address, message, {}}).is_err(), "an empty signature does not verify");
128 });
129
130 detail::capability_check<Probe>(report, "an address not in the contract's form is invalid_argument", [&](Probe& p, const auto& expect) {
131 const wallet_kit_case<Probe> c{p};
132 std::string upper = c.address;
133 for (auto& ch : upper)
134 if (ch >= 'a' && ch <= 'f')
135 ch = static_cast<char>(ch - 'a' + 'A');
136 if (upper == c.address)
137 upper[2] = 'g';
138 for (const std::string_view a : {std::string_view{upper}, std::string_view{"0x1234"}, std::string_view{}})
139 expect(is(p.verifier().verify({1, a, message, c.signature}), capability_errc::invalid_argument),
140 "an address with capitals, too short or empty is invalid_argument");
141 });
142
143 detail::capability_check<Probe>(report, "a provider failure never leaks its response", [&](Probe& p, const auto& expect) {
144 const wallet_kit_case<Probe> c{p};
145 p.provider_fails(capability_errc::unavailable, capability_provider_response);
146 const auto r = p.verifier().verify({1, c.address, message, c.signature});
147 expect(is(r, capability_errc::unavailable), "the provider's failure is returned");
148 if (r.is_err())
149 expect(!capability_error_leaks(r.unwrap_err(), capability_redaction_marker),
150 "neither provider_code nor message carries the provider's response");
151 });
152
153 return report;
154 }
