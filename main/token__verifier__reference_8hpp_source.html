116 [[nodiscard]] std::string issue(token_claims claims)
117 {
118 const std::string body = std::string(prefix) + std::to_string(issued_.size());
119 issued_.push_back({std::move(claims), false});
120 return body + "." + std::string(tag(body).view());
121 }

125 void expire(std::string_view token)

134 [[nodiscard]] capability_result<token_claims> verify(std::string_view token) const
135 {
136 if (injected_) {
137 const capability_error e = *injected_;
138 injected_.reset();
139 return err(e);
140 }
141 const auto i = index_of(token);
142 if (!i)
143 return fail(capability_errc::unauthorized, {}, "the token does not verify");
144 const auto& entry = issued_[*i];
145 if (entry.expired || (entry.claims.expires_at && *entry.claims.expires_at <= stamp()))
146 return fail(capability_errc::stale, {}, "the token has expired");
147 return ok(entry.claims);
148 }
149
153 void fail_next(capability_errc code, std::string_view response = {}) noexcept
154 {
155 injected_ = provider_error(code, response.substr(0, response.find(' ')));
156 }
157
158 private:
159 // The clock's time, in the seconds every record of this reference carries.
160 [[nodiscard]] utc_s stamp() const noexcept(noexcept(std::declval<const Clock&>().now()))
161 {
162 return detail::clock_now<std::chrono::seconds>(clock_);
163 }
164 struct entry {
165 token_claims claims;
166 bool expired;
167 };
168
169 [[nodiscard]] digest_hex<32> tag(std::string_view body) const noexcept
170 {
171 return hmac<sha256>::mac(key_.reveal(), body).hex();
172 }
173
174 // The serial of a token this verifier issued, after a constant-time check of its tag.
175 [[nodiscard]] std::optional<std::size_t> index_of(std::string_view token) const noexcept
176 {
177 if (!token.starts_with(prefix))
178 return std::nullopt;
179 const auto dot = token.find('.', prefix.size());
180 if (dot == std::string_view::npos)
181 return std::nullopt;
182 const auto body = token.substr(0, dot);
183 const auto presented = token.substr(dot + 1);
184 if (!constant_time_equal(tag(body).view(), presented))
185 return std::nullopt;
186 std::size_t serial = 0;
187 const auto digits = body.substr(prefix.size());
188 const auto [end, ec] = std::from_chars(digits.data(), digits.data() + digits.size(), serial);
189 if (ec != std::errc{} || end != digits.data() + digits.size() || serial >= issued_.size())
190 return std::nullopt;
191 return serial;
192 }

194 secret<std::string> key_;

196 std::vector<entry> issued_;
197 mutable std::optional<capability_error> injected_;
198 };
