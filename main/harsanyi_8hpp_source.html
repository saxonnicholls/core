61 template <game_value T>
62 constexpr void to_dividends(std::span<T> table)
63 {
64 assert(std::has_single_bit(table.size()));
65 for (std::size_t bit = 1; bit < table.size(); bit <<= 1)
66 for (std::size_t s = 0; s < table.size(); ++s)
67 if ((s & bit) != 0)
68 table[s] = table[s] - table[s ^ bit];
69 }

120 template <game_value T>
121 [[nodiscard]] constexpr dynamic_attribution<T> shapley_from_dividends(std::span<const T> dividends)
122 {
123 assert(std::has_single_bit(dividends.size()));
124 const auto n = static_cast<std::size_t>(std::countr_zero(dividends.size()));
125 dynamic_attribution<T> out{};
126 out.values.resize(n);
127 out.baseline = dividends[0];
128 out.total = dividends[0];
129 for (coalition s = 1; s < dividends.size(); ++s) {
130 out.total = out.total + dividends[s];
131 const T share = dividends[s] * (game_scalar_t<T>(1) / static_cast<game_scalar_t<T>>(coalition_size(s)));
132 for (coalition members = s; members != 0; members &= members - 1) {
133 const auto i = static_cast<std::size_t>(std::countr_zero(members));
134 out.values[i] = out.values[i] + share;
135 }
136 }
137 return out;
138 }
