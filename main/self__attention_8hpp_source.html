129template <std::size_t DModel, std::size_t DHead, weight_matrix W, math::math_number T, row X, mutable_row Out,
130 typename R = no_rope_t>
131 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
132 && (std::same_as<R, no_rope_t> || (rope_source<R> && DHead % 2 == 0))
133constexpr void self_attention_head(const self_attention_weights<W, T>& w, const X& xs,
134 const self_attention_activations<T>& act, Out&& out, std::size_t window = 0,
135 const R& rope_src = no_rope, rope_layout layout = rope_layout::half,
136 attention_mask mask = attention_mask::causal, std::size_t block = 1)
137{
138 static_assert(DModel > 0 && DHead > 0, "snicholls::ml::self_attention_head: a token and a head have channels");
139 const auto x = as_row(xs);
140 const auto y = as_mutable_row(out);
141 const std::size_t n = x.size() / DModel;
142 assert(x.size() == n * DModel && y.size() == n * DHead);
143 assert(act.q.size() == n * DHead && act.k.size() == n * DHead && act.v.size() == n * DHead
144 && (act.lse.empty() || act.lse.size() == n));
145 assert(w.q.rows == DHead && w.k.rows == DHead && w.v.rows == DHead && w.q.cols == DModel && w.k.cols == DModel
146 && w.v.cols == DModel);
147 for (std::size_t t = 0; t < n; ++t) {
148 const auto in = prepare_input<DModel>(w.q, x.subspan(t * DModel, DModel));
149 const auto q = act.q.subspan(t * DHead, DHead), k = act.k.subspan(t * DHead, DHead);
150 linear(w.q, in, q, w.q_bias);
151 linear(w.k, in, k, w.k_bias);
152 linear(w.v, in, act.v.subspan(t * DHead, DHead), w.v_bias);
153 if constexpr (!std::same_as<R, no_rope_t>) {
154 rope(q, rope_src.cos_at(t), rope_src.sin_at(t), layout);
155 rope(k, rope_src.cos_at(t), rope_src.sin_at(t), layout);
156 }
157 }
158 const T scale = T(1) / math::sqrt(T(DHead));
159 for (std::size_t t = 0; t < n; ++t) {
160 const key_range r = attention_keys(mask, t, n, window, block);
161 const T l = attend(std::span<const T>(act.q.subspan(t * DHead, DHead)),
162 std::span<const T>(act.k.subspan(r.first * DHead, r.size() * DHead)),
163 std::span<const T>(act.v.subspan(r.first * DHead, r.size() * DHead)),
164 y.subspan(t * DHead, DHead), scale);
165 if (!act.lse.empty())
166 act.lse[t] = l;
167 }
168}

206template <std::size_t DModel, std::size_t DHead, typename S, math::math_number T, row X, row O, row DO, mutable_row DX,
207 typename R = no_rope_t>
208 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<O>, T> && std::same_as<row_value_t<DO>, T>
209 && std::same_as<row_value_t<DX>, T> && (std::same_as<R, no_rope_t> || (rope_source<R> && DHead % 2 == 0))
210constexpr void self_attention_head_backward(const self_attention_weights<dense_weights<S>, T>& w, const X& xs,
211 const self_attention_activations<T>& act, const O& out, const DO& dout,
212 DX&& dxs, const self_attention_gradients<T>& grads, std::size_t window = 0,
213 const R& rope_src = no_rope, rope_layout layout = rope_layout::half,
214 attention_mask mask = attention_mask::causal, std::size_t block = 1)
215{
216 const auto x = as_row(xs);
217 const auto y = as_row(out);
218 const auto g = as_row(dout);
219 const auto gx = as_mutable_row(dxs);
220 const std::size_t n = x.size() / DModel;
221 assert(x.size() == n * DModel && y.size() == n * DHead && g.size() == n * DHead && (gx.empty() || gx.size() == x.size()));
222 assert(act.q.size() == n * DHead && act.k.size() == n * DHead && act.v.size() == n * DHead && act.lse.size() == n);
223 assert(grads.q.size() == n * DHead && grads.k.size() == n * DHead && grads.v.size() == n * DHead);
224 std::fill(grads.k.begin(), grads.k.end(), T(0));
225 std::fill(grads.v.begin(), grads.v.end(), T(0));
226 const T scale = T(1) / math::sqrt(T(DHead));
227 for (std::size_t t = 0; t < n; ++t) {
228 const key_range r = attention_keys(mask, t, n, window, block);
229 const auto keys = std::span<const T>(act.k.subspan(r.first * DHead, r.size() * DHead));
230 const auto values = std::span<const T>(act.v.subspan(r.first * DHead, r.size() * DHead));
231 attend_backward(std::span<const T>(act.q.subspan(t * DHead, DHead)), keys, values, y.subspan(t * DHead, DHead),
232 g.subspan(t * DHead, DHead), act.lse[t], scale, grads.q.subspan(t * DHead, DHead),
233 grads.k.subspan(r.first * DHead, r.size() * DHead),
234 grads.v.subspan(r.first * DHead, r.size() * DHead));
235 }
236 const std::span<T> no_dx{};
237 for (std::size_t t = 0; t < n; ++t) {
238 const auto dq = grads.q.subspan(t * DHead, DHead), dk = grads.k.subspan(t * DHead, DHead);
239 if constexpr (!std::same_as<R, no_rope_t>) {
240 rope_inverse(dq, rope_src.cos_at(t), rope_src.sin_at(t), layout);
241 rope_inverse(dk, rope_src.cos_at(t), rope_src.sin_at(t), layout);
242 }
243 const auto xt = x.subspan(t * DModel, DModel);
244 const auto dxt = gx.empty() ? no_dx : gx.subspan(t * DModel, DModel);
245 linear_backward(w.q, xt, dq, dxt, grads.wq, grads.q_bias);
246 linear_backward(w.k, xt, dk, dxt, grads.wk, grads.k_bias);
247 linear_backward(w.v, xt, grads.v.subspan(t * DHead, DHead), dxt, grads.wv, grads.v_bias);
248 }
249}
