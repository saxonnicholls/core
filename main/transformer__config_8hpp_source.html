110 [[nodiscard]] constexpr std::size_t q_size() const noexcept { return n_heads * head_dim; }
113 [[nodiscard]] constexpr std::size_t kv_size() const noexcept { return n_kv_heads * head_dim; }
116 [[nodiscard]] constexpr std::size_t group() const noexcept { return n_heads / n_kv_heads; }
119 [[nodiscard]] constexpr rope_layout layout() const noexcept { return rope == rope_kind::interleaved ? rope_layout::interleaved : rope_layout::half; }
120};

126[[nodiscard]] constexpr bool valid(const transformer_config& c) noexcept
127{
128 return c.d_model > 0 && c.n_heads > 0 && c.n_kv_heads > 0 && c.head_dim > 0 && c.d_ff > 0 && c.n_heads % c.n_kv_heads == 0
129 && (c.rope == rope_kind::none || c.head_dim % 2 == 0) && c.norm_eps > 0 && c.attention_softcap >= 0
130 && (c.rope == rope_kind::none || c.rope_base > 1);
131}
