103template <math::math_number T, typename B>
104 requires linalg::blas_backend<B, T>
105constexpr void muon_orthogonalise(const B& backend, std::span<const T> g, std::size_t rows, std::size_t cols, std::span<T> out,
106 std::span<T> work)
107{
108 using right = linalg::layout_right;
109 assert(g.size() == rows * cols && out.size() == g.size() && work.size() >= 2 * rows * cols + 2 * std::min(rows, cols) * std::min(rows, cols));
110 const bool tall = rows > cols;
111 const std::size_t r = tall ? cols : rows, s = tall ? rows : cols; // X is r x s, r <= s
112 const auto x = work.first(r * s), x2 = work.subspan(r * s, r * s), a = work.subspan(2 * r * s, r * r),
113 bm = work.subspan(2 * r * s + r * r, r * r);
114 T norm2 = T(0);
115 for (const T v : g)
116 norm2 += v * v;
117 const T scale = T(1) / (math::sqrt(norm2) + T(1e-7));
118 for (std::size_t i = 0; i < rows; ++i)
119 for (std::size_t j = 0; j < cols; ++j)
120 (tall ? x[j * s + i] : x[i * s + j]) = g[i * cols + j] * scale;
121 using ns = muon_newton_schulz<T>;
122 const linalg::matrix_view<T, right> x2v(x2.data(), {r, s}), av(a.data(), {r, r}), bv(bm.data(), {r, r});
123 const linalg::matrix_view<const T, right> xc(x.data(), {r, s}), ac(a.data(), {r, r}), bc(bm.data(), {r, r});
124 // The shapes are right by construction and C never overlaps A or B, so every product succeeds
125 const auto product = [&](T alpha, const auto& lhs, const auto& rhs, T beta, const auto& c) {
126 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
127 assert(done.is_ok());
128 };
129 for (int step = 0; step < ns::steps; ++step) {
130 product(T(1), xc, linalg::transposed(xc), T(0), av); // A = X Xᵀ
131 for (std::size_t i = 0; i < r * r; ++i)
132 bm[i] = ns::b * a[i];
133 product(ns::c, ac, ac, T(1), bv); // B = b A + c A²
134 for (std::size_t i = 0; i < r * s; ++i)
135 x2[i] = ns::a * x[i];
136 product(T(1), bc, xc, T(1), x2v); // X ← a X + B X
137 std::copy(x2.begin(), x2.end(), x.begin());
138 }
139 for (std::size_t i = 0; i < rows; ++i)
140 for (std::size_t j = 0; j < cols; ++j)
141 out[i * cols + j] = tall ? x[j * s + i] : x[i * s + j];
142}

153template <math::math_number T, typename B>
154 requires linalg::blas_backend<B, T>
155constexpr void muon_step(const B& backend, std::span<T> w, std::span<const T> g, std::span<T> momentum, std::size_t rows,
156 std::size_t cols, const muon_config<T>& config, std::span<T> work)
157{
158 const std::size_t n = rows * cols;
159 assert(w.size() == n && g.size() == n && momentum.size() == n && work.size() >= muon_work_size(rows, cols));
160 const auto update = work.first(n), rest = work.subspan(n);
161 for (std::size_t i = 0; i < n; ++i) {
162 momentum[i] = config.momentum * momentum[i] + g[i];
163 update[i] = config.nesterov ? g[i] + config.momentum * momentum[i] : momentum[i];
164 }
165 const auto x = rest.first(n);
166 muon_orthogonalise(backend, std::span<const T>(update), rows, cols, x, rest.subspan(n));
167 const T step = config.lr * math::sqrt(std::max(T(1), T(rows) / T(cols)));
168 for (std::size_t i = 0; i < n; ++i)
169 w[i] -= step * x[i];
170}
