124template <math::math_number T, typename B>
125 requires linalg::blas_backend<B, T>
126constexpr void muon_orthogonalise(const B& backend, std::span<const T> g, std::size_t rows, std::size_t cols, std::span<T> out,
127 std::span<T> work)
128{
129 using right = linalg::layout_right;
130 const std::size_t n = rows * cols, r = std::min(rows, cols);
131 assert(g.size() == n && out.size() == n && work.size() >= 2 * n + 2 * r * r);
132 const bool tall = rows > cols;
133 // X ping-pongs between out and x2; A and B are r x r
134 const auto x2 = work.first(n), a = work.subspan(n, r * r), bm = work.subspan(n + r * r, r * r);
135 const T scale = T(1) / (math::sqrt(sum_squares(backend, g)) + T(1e-7));
136 copy_rows(backend, g, out);
137 scale_rows(backend, out, scale);
138 using ns = muon_newton_schulz<T>;
139 const linalg::matrix_view<T, right> av(a.data(), {r, r}), bv(bm.data(), {r, r});
140 const linalg::matrix_view<const T, right> ac(a.data(), {r, r}), bc(bm.data(), {r, r});
141 // The shapes are right by construction and C never overlaps A or B, so every product succeeds
142 const auto product = [&](T alpha, const auto& lhs, const auto& rhs, T beta, const auto& c) {
143 [[maybe_unused]] const auto done = linalg::gemm(backend, alpha, lhs, rhs, beta, c);
144 assert(done.is_ok());
145 };
146 std::span<T> x = out, next = x2;
147 for (int step = 0; step < ns::steps; ++step) {
148 const linalg::matrix_view<const T, right> xc(x.data(), {rows, cols});
149 const linalg::matrix_view<T, right> nv(next.data(), {rows, cols});
150 if (tall)
151 product(T(1), linalg::transposed(xc), xc, T(0), av); // A = Xᵀ X
152 else
153 product(T(1), xc, linalg::transposed(xc), T(0), av); // A = X Xᵀ
154 copy_rows(backend, std::span<const T>(a), bm);
155 product(ns::c, ac, ac, ns::b, bv); // B = b A + c A²
156 copy_rows(backend, std::span<const T>(x), next);
157 if (tall)
158 product(T(1), xc, bc, ns::a, nv); // X ← a X + X B
159 else
160 product(T(1), bc, xc, ns::a, nv); // X ← a X + B X
161 std::swap(x, next);
162 }
163 if (x.data() != out.data())
164 copy_rows(backend, std::span<const T>(x), out);
165}

181template <math::math_number T, typename B>
182 requires linalg::blas_backend<B, T>
183constexpr void muon_step(const B& backend, std::span<T> w, std::span<const T> g, std::span<T> momentum, std::size_t rows,
184 std::size_t cols, const muon_config<T>& config, std::span<T> work)
185{
186 const std::size_t n = rows * cols;
187 assert(w.size() == n && g.size() == n && momentum.size() == n && work.size() >= muon_work_size(rows, cols));
188 const auto update = work.first(n), x = work.subspan(n, n), rest = work.subspan(2 * n);
189 scale_rows(backend, momentum, config.momentum); // momentum = g + μ momentum
190 add_rows(backend, g, momentum);
191 copy_rows(backend, std::span<const T>(momentum), update);
192 if (config.nesterov) { // update = g + μ momentum
193 scale_rows(backend, update, config.momentum);
194 add_rows(backend, g, update);
195 }
196 muon_orthogonalise(backend, std::span<const T>(update), rows, cols, x, rest);
197 scale_rows(backend, x, -(config.lr * math::sqrt(std::max(T(1), T(rows) / T(cols))))); // w = -step X + w
198 add_rows(backend, std::span<const T>(x), w);
199}
