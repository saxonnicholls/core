256 // ggml's neighbour list of the off-grid point u, visited in its order: every grid point whose squared distance
257 // to u is one of the nwant smallest, by distance then index. score(g) is a candidate's error; the first with
258 // the least error wins (ggml's strict <). Returns -1 when no score is below FLT_MAX.
259 template <std::size_t N, typename Score>
260 constexpr int best_neighbour(const lattice<N>& grid, std::uint16_t u, Score score)
261 {
262 std::array<int, N> d2{};
263 int level[3] = {std::numeric_limits<int>::max(), std::numeric_limits<int>::max(), std::numeric_limits<int>::max()};
264 for (std::size_t g = 0; g < N; ++g) {
265 int s = 0;
266 for (int k = 0; k < grid.dims; ++k) {
267 const int pos = 2 * int((u >> (grid.bits * k)) & ((1u << grid.bits) - 1)) + 1;
268 const int diff = grid.coordinate(g, k) - pos;
269 s += diff * diff;
270 }
271 d2[g] = s;
272 // keep the nwant smallest distinct distances, ascending
273 for (int l = 0; l < grid.nwant; ++l) {
274 if (s == level[l])
275 break;
276 if (s < level[l]) {
277 for (int m = grid.nwant - 1; m > l; --m)
278 level[m] = level[m - 1];
279 level[l] = s;
280 break;
281 }
282 }
283 }
284 float best = flt_max;
285 int best_g = -1;
286 for (int l = 0; l < grid.nwant && level[l] != std::numeric_limits<int>::max(); ++l)
287 for (std::size_t g = 0; g < N; ++g)
288 if (d2[g] == level[l]) {
289 const float e = score(g);
290 if (e < best) {
291 best = e;
292 best_g = int(g);
293 }
294 }
295 return best_g;
296 }

298 // ggml's iq2_find_best_neighbour / iq3_find_best_neighbour: the neighbour nearest scale · grid in the weighted
299 // norm; writes its levels to L
300 template <std::size_t N>
301 constexpr int find_best_neighbour(const lattice<N>& grid, std::uint16_t u, const float* xval, const float* weight, float scale, std::int8_t* L)
302 {
303 const int g = best_neighbour(grid, u, [&](std::size_t j) {
304 float d2 = 0;
305 for (int i = 0; i < grid.dims; ++i) {
306 const float q = float(grid.coordinate(j, i));
307 const float sq = scale * q;
308 const float diff = sq - xval[i];
309 const float wd = weight[i] * diff;
310 const float wdd = wd * diff;
311 d2 += wdd;
312 }
313 return d2;
314 });
315 assert(g >= 0 && "snicholls::ml: no grid point near a group (a NaN in the row?)");
316 for (int i = 0; i < grid.dims; ++i)
317 L[i] = static_cast<std::int8_t>(grid.level(std::size_t(g), i));
318 return g;
319 }

350 // sumqx and sumq2 of levels L (q = 2l + 1) against xval with weights
351 constexpr void sums(int n, const float* weight, const float* xval, const std::int8_t* L, float& sumqx, float& sumq2)
352 {
353 sumqx = 0;
354 sumq2 = 0;
355 for (int i = 0; i < n; ++i) {
356 const float w = weight[i];
357 const float q = float(2 * L[i] + 1);
358 const float wx = w * xval[i];
359 const float wxq = wx * q;
360 sumqx += wxq;
361 const float wq = w * q;
362 const float wqq = wq * q;
363 sumq2 += wqq;
364 }
365 }

367 // |x| into xval with the signs in s, and with even parity: when an odd number are negative, the one of least
368 // weight · x^2 is flipped (IQ2_XXS, IQ2_XS, IQ3_XXS store 7 sign bits)
369 constexpr std::uint8_t even_signs(const float* xb, const float* weight, float* xval)
370 {
371 int nflip = 0;
372 std::uint8_t s = 0;
373 for (int i = 0; i < 8; ++i) {
374 if (xb[i] >= 0) {
375 xval[i] = xb[i];
376 } else {
377 xval[i] = -xb[i];
378 ++nflip;
379 s = static_cast<std::uint8_t>(s | (1 << i));
380 }
381 }
382 if (nflip % 2) {
383 int imin_ = 0;
384 const float w0 = weight[0] * xb[0];
385 float min = w0 * xb[0];
386 for (int i = 1; i < 8; ++i) {
387 const float wi = weight[i] * xb[i];
388 const float ax = wi * xb[i];
389 if (ax < min) {
390 min = ax;
391 imin_ = i;
392 }
393 }
394 xval[imin_] = -xval[imin_];
395 s = static_cast<std::uint8_t>(s ^ (1 << imin_));
396 }
397 return static_cast<std::uint8_t>(s & 127);
398 }

432 // IQ2_XXS (sub-blocks of 32, kMaxQ 3): ggml's quantize_row_iq2_xxs_impl
433 constexpr void quantize(std::span<const float> x, std::span<const float> qw_row, std::span<block_iq2_xxs> y)
434 {
435 constexpr int kMaxQ = 3;
436 std::array<float, 8> scales{};
437 std::array<float, 32> weight{}, xval{}, waux{};
438 std::array<std::int8_t, 32> L{}, Laux{};
439 std::array<std::uint8_t, 4> block_signs{};
440 for (std::size_t ibl = 0; ibl < y.size(); ++ibl) {
441 std::array<std::uint32_t, 16> q2{};
442 y[ibl].d = detail::quantize::fp16(0.0f);
443 float largest_scale = 0;
444 const float* xbl = x.data() + 256 * ibl;
445 const float sigma2 = sum_squares(xbl, 256) / 256.0f;
446 for (int ib = 0; ib < 8; ++ib) {
447 const float* xb = xbl + 32 * ib;
448 const float* qw = qw_row.data() + 256 * ibl + 32 * ib;
449 for (int i = 0; i < 32; ++i)
450 weight[i] = detail::quantize::importance_weight(qw[i], sigma2, xb[i]);
451 for (int i = 0; i < 32; ++i)
452 waux[i] = detail::quantize::sqrt(weight[i]);
453 for (int k = 0; k < 4; ++k)
454 block_signs[k] = even_signs(xb + 8 * k, weight.data() + 8 * k, xval.data() + 8 * k);
455 float max = xval[0];
456 for (int i = 1; i < 32; ++i)
457 max = fmax(max, xval[i]);
458 if (max < detail::quantize::group_max_eps) {
459 scales[ib] = 0;
460 L = {};
461 continue;
462 }
463 std::array<std::uint8_t, 32> Lq{};
464 float scale = detail::quantize::make_qp_quants(32, kMaxQ + 1, xval.data(), Lq.data(), weight.data());
465 for (int i = 0; i < 32; ++i)
466 L[i] = static_cast<std::int8_t>(Lq[i]);
467 const float eff_max = scale * float(kMaxQ);
468 if (eff_max <= 0) {
469 scales[ib] = 0;
470 L = {};
471 continue;
472 }
473 float best = 0;
474 for (int is = -6; is <= 6; ++is) {
475 const float step = float(is) * 0.1f;
476 const float top = float(2 * kMaxQ - 1) + step;
477 const float id = top / eff_max;
478 const float this_scale = 1 / id;
479 for (int k = 0; k < 4; ++k) {
480 for (int i = 0; i < 8; ++i)
481 Laux[8 * k + i] = static_cast<std::int8_t>(imax(0, imin(kMaxQ - 1, level_of(id, xval[8 * k + i]))));
482 const std::uint16_t u = pack(iq2xxs, Laux.data() + 8 * k);
483 if (iq_grid_find(iq2xxs.index, u) < 0)
484 find_best_neighbour(iq2xxs, u, xval.data() + 8 * k, waux.data() + 8 * k, this_scale, Laux.data() + 8 * k);
485 }
486 float sumqx = 0, sumq2 = 0;
487 sums(32, weight.data(), xval.data(), Laux.data(), sumqx, sumq2);
488 if (sumq2 > 0 && sumqx * sumqx > best * sumq2) {
489 scale = sumqx / sumq2;
490 best = scale * sumqx;
491 L = Laux;
492 }
493 }
494 if (scale > 0) {
495 const float id = 1 / scale;
496 for (int k = 0; k < 4; ++k) {
497 std::array<std::int8_t, 8> l8{};
498 for (int i = 0; i < 8; ++i)
499 l8[i] = static_cast<std::int8_t>(imax(0, imin(kMaxQ - 1, level_of(id, xval[8 * k + i]))));
500 const std::uint16_t u = pack(iq2xxs, l8.data());
501 int g = iq_grid_find(iq2xxs.index, u);
502 if (g < 0)
503 g = find_best_neighbour(iq2xxs, u, xval.data() + 8 * k, waux.data() + 8 * k, scale, L.data() + 8 * k);
504 for (int i = 0; i < 8; ++i)
505 L[8 * k + i] = static_cast<std::int8_t>(iq2xxs.level(std::size_t(g), i));
506 }
507 float sumqx = 0, sumq2 = 0;
508 sums(32, weight.data(), xval.data(), L.data(), sumqx, sumq2);
509 if (sumq2 > 0)
510 scale = sumqx / sumq2;
511 }
512 if (scale < 0) {
513 scale = -scale;
514 for (int k = 0; k < 4; ++k)
515 block_signs[k] = static_cast<std::uint8_t>((~block_signs[k]) & 127);
516 }
517 for (int k = 0; k < 4; ++k) {
518 const int g = iq_grid_find(iq2xxs.index, pack(iq2xxs, L.data() + 8 * k));
519 assert(g >= 0 && "snicholls::ml: IQ2_XXS found a point not on the grid");
520 q2[2 * ib] |= std::uint32_t(g) << (8 * k);
521 q2[2 * ib + 1] |= std::uint32_t(block_signs[k]) << (7 * k);
522 }
523 scales[ib] = scale;
524 largest_scale = fmax(largest_scale, scale);
525 }
526 if (largest_scale == 0.0f) {
527 y[ibl].qs = {};
528 continue;
529 }
530 const float d = largest_scale / 31;
531 y[ibl].d = detail::quantize::fp16(d);
532 const float id = 1 / d;
533 for (int ib = 0; ib < 8; ++ib)
534 q2[2 * ib + 1] |= std::uint32_t(scale_code(id, scales[ib])) << 28;
535 for (int i = 0; i < 16; ++i) {
536 y[ibl].qs[2 * i] = static_cast<std::uint16_t>(q2[i]);
537 y[ibl].qs[2 * i + 1] = static_cast<std::uint16_t>(q2[i] >> 16);
538 }
539 }
540 }

542 // IQ2_XS and IQ2_S (sub-blocks of 16, kMaxQ 3): ggml's quantize_row_iq2_xs_impl and quantize_row_iq2_s_impl,
543 // which differ in their grid, signs (7 with parity, or 8), the weights without an importance row, the group
544 // threshold, the scale's fudge factor and the layout
545 template <bool S, typename Block>
546 constexpr void quantize_iq2_16(std::span<const float> x, std::span<const float> qw_row, std::span<Block> y)
547 {
548 constexpr int kMaxQ = 3;
549 constexpr auto& grid = []() -> const auto& {
550 if constexpr (S)
551 return iq2s;
552 else
553 return iq2xs;
554 }();
555 std::array<float, 16> scales{};
556 std::array<float, 16> weight{}, xval{}, waux{};
557 std::array<std::int8_t, 16> L{}, Laux{};
558 std::array<bool, 2> on_grid{}, on_grid_aux{};
559 std::array<std::uint8_t, 2> block_signs{};
560 for (std::size_t ibl = 0; ibl < y.size(); ++ibl) {
561 std::array<std::uint16_t, 32> q2{};
562 y[ibl] = Block{};
563 y[ibl].d = detail::quantize::fp16(0.0f);
564 float largest_scale = 0;
565 const float* xbl = x.data() + 256 * ibl;
566 const float sumx2 = sum_squares(xbl, 256);
567 float sigma2 = 0;
568 if constexpr (S) {
569 const float two = 2 * sumx2;
570 sigma2 = two / 256.0f;
571 } else {
572 sigma2 = sumx2 / 256.0f;
573 }
574 for (int ib = 0; ib < 16; ++ib) {
575 const float* xb = xbl + 16 * ib;
576 if (S && qw_row.empty()) {
577 const float quarter = 0.25f * sigma2;
578 for (int i = 0; i < 16; ++i) {
579 const float x2 = xb[i] * xb[i];
580 weight[i] = quarter + x2;
581 }
582 } else {
583 const float* qw = qw_row.data() + 256 * ibl + 16 * ib;
584 for (int i = 0; i < 16; ++i)
585 weight[i] = detail::quantize::importance_weight(qw[i], sigma2, xb[i]);
586 }
587 for (int i = 0; i < 16; ++i)
588 waux[i] = detail::quantize::sqrt(weight[i]);
589 for (int k = 0; k < 2; ++k)
590 block_signs[k] = S ? plain_signs(xb + 8 * k, xval.data() + 8 * k) : even_signs(xb + 8 * k, weight.data() + 8 * k, xval.data() + 8 * k);
591 float max = xval[0];
592 for (int i = 1; i < 16; ++i)
593 max = fmax(max, xval[i]);
594 L = {};
595 if (max < (S ? 1e-8f : detail::quantize::group_max_eps)) {
596 scales[ib] = 0;
597 continue;
598 }
599 float best = 0;
600 float scale = max / float(2 * kMaxQ - 1);
601 on_grid = {true, true};
602 for (int is = -9; is <= 9; ++is) {
603 const float step = float(is) * 0.1f;
604 const float top = float(2 * kMaxQ - 1) + step;
605 const float id = top / max;
606 const float this_scale = 1 / id;
607 for (int k = 0; k < 2; ++k) {
608 for (int i = 0; i < 8; ++i)
609 Laux[8 * k + i] = static_cast<std::int8_t>(imax(0, imin(kMaxQ - 1, level_of(id, xval[8 * k + i]))));
610 const std::uint16_t u = pack(grid, Laux.data() + 8 * k);
611 on_grid_aux[k] = true;
612 if (iq_grid_find(grid.index, u) < 0) {
613 on_grid_aux[k] = false;
614 find_best_neighbour(grid, u, xval.data() + 8 * k, waux.data() + 8 * k, this_scale, Laux.data() + 8 * k);
615 }
616 }
617 float sumqx = 0, sumq2 = 0;
618 sums(16, weight.data(), xval.data(), Laux.data(), sumqx, sumq2);
619 if (sumq2 > 0 && sumqx * sumqx > best * sumq2) {
620 scale = sumqx / sumq2;
621 best = scale * sumqx;
622 L = Laux;
623 on_grid = on_grid_aux;
624 }
625 }
626 const int n_not_ongrid = int(!on_grid[0]) + int(!on_grid[1]);
627 if (n_not_ongrid > 0 && scale > 0) {
628 const float id = 1 / scale;
629 for (int k = 0; k < 2; ++k) {
630 if (on_grid[k])
631 continue;
632 for (int i = 0; i < 8; ++i)
633 L[8 * k + i] = static_cast<std::int8_t>(imax(0, imin(kMaxQ - 1, level_of(id, xval[8 * k + i]))));
634 const std::uint16_t u = pack(grid, L.data() + 8 * k);
635 if (iq_grid_find(grid.index, u) < 0)
636 find_best_neighbour(grid, u, xval.data() + 8 * k, waux.data() + 8 * k, scale, L.data() + 8 * k);
637 }
638 float sumqx = 0, sumq2 = 0;
639 sums(16, weight.data(), xval.data(), L.data(), sumqx, sumq2);
640 if (sumq2 > 0)
641 scale = sumqx / sumq2;
642 }
643 if (scale < 0) {
644 scale = -scale;
645 for (int k = 0; k < 2; ++k)
646 block_signs[k] = static_cast<std::uint8_t>(S ? ~block_signs[k] : (~block_signs[k]) & 127);
647 }
648 for (int k = 0; k < 2; ++k) {
649 const int g = iq_grid_find(grid.index, pack(grid, L.data() + 8 * k));
650 assert(g >= 0 && "snicholls::ml: IQ2 found a point not on the grid");
651 if constexpr (S) {
652 const int i8 = 2 * ib + k;
653 y[ibl].qs[i8] = static_cast<std::uint8_t>(g & 255);
654 y[ibl].qh[i8 / 4] = static_cast<std::uint8_t>(y[ibl].qh[i8 / 4] | ((g >> 8) << (2 * (i8 % 4))));
655 y[ibl].qs[32 + i8] = block_signs[k];
656 } else {
657 q2[2 * ib + k] = static_cast<std::uint16_t>(g | (block_signs[k] << 9));
658 }
659 }
660 scales[ib] = scale;
661 largest_scale = fmax(largest_scale, scale);
662 }
663 if (largest_scale == 0.0f) {
664 if constexpr (!S)
665 y[ibl].qs = {};
666 continue;
667 }
668 const float d = largest_scale / 31;
669 if constexpr (S) {
670 const float fudged = d * 0.9875f;
671 y[ibl].d = detail::quantize::fp16(fudged);
672 } else {
673 y[ibl].d = detail::quantize::fp16(d);
674 }
675 const float id = 1 / d;
676 for (int ib = 0; ib < 16; ++ib) {
677 const int l = scale_code(id, scales[ib]);
678 if (ib % 2 == 0)
679 y[ibl].scales[ib / 2] = static_cast<std::uint8_t>(l);
680 else
681 y[ibl].scales[ib / 2] = static_cast<std::uint8_t>(y[ibl].scales[ib / 2] | (l << 4));
682 }
683 if constexpr (!S)
684 y[ibl].qs = q2;
685 }
686 }

688 // IQ3_XXS and IQ3_S (groups of 4, kMaxQ 8): ggml's quantize_row_iq3_xxs_impl(256, ...) and
689 // quantize_row_iq3_s_impl(32, ...)
690 template <bool S, typename Block>
691 constexpr void quantize_iq3(std::span<const float> x, std::span<const float> qw_row, std::span<Block> y)
692 {
693 constexpr int kMaxQ = 8;
694 constexpr auto& grid = []() -> const auto& {
695 if constexpr (S)
696 return iq3s;
697 else
698 return iq3xxs;
699 }();
700 std::array<float, 8> scales{};
701 std::array<float, 32> weight{}, xval{}, waux{};
702 std::array<std::int8_t, 32> L{}, Laux{};
703 std::array<bool, 8> on_grid{}, on_grid_aux{};
704 std::array<std::uint8_t, 4> block_signs{};
705 for (std::size_t ibl = 0; ibl < y.size(); ++ibl) {
706 std::array<std::uint8_t, 104> q3{}; // IQ3_XXS: 64 grid indices, 8 words of scales and signs
707 y[ibl] = Block{};
708 y[ibl].d = detail::quantize::fp16(0.0f);
709 float largest_scale = 0;
710 std::size_t qs_off = 0, signs_off = 0;
711 (void)qs_off, (void)signs_off;
712 const float* xbl = x.data() + 256 * ibl;
713 const float two = 2 * sum_squares(xbl, 256);
714 const float sigma2 = two / 256.0f;
715 for (int ib = 0; ib < 8; ++ib) {
716 const float* xb = xbl + 32 * ib;
717 if (!qw_row.empty()) {
718 const float* qw = qw_row.data() + 256 * ibl + 32 * ib;
719 for (int i = 0; i < 32; ++i)
720 weight[i] = detail::quantize::importance_weight(qw[i], sigma2, xb[i]);
721 } else {
722 for (int i = 0; i < 32; ++i)
723 weight[i] = xb[i] * xb[i];
724 }
725 for (int i = 0; i < 32; ++i)
726 waux[i] = detail::quantize::sqrt(weight[i]);
727 for (int k = 0; k < 4; ++k)
728 block_signs[k] = S ? plain_signs(xb + 8 * k, xval.data() + 8 * k) : even_signs(xb + 8 * k, weight.data() + 8 * k, xval.data() + 8 * k);
729 float max = xval[0];
730 for (int i = 1; i < 32; ++i)
731 max = fmax(max, xval[i]);
732 L = {};
733 if (S ? max == 0.0f : max < 1e-8f) {
734 scales[ib] = 0;
735 continue; // IQ3_S: ggml's qs and signs pointers do not advance, so the next sub-block takes these bytes
736 }
737 float best = 0;
738 float scale = max / float(2 * kMaxQ - 1);
739 on_grid.fill(!S);
740 for (int is = S ? -9 : -15; is <= (S ? 9 : 15); ++is) {
741 const float step = float(is) * 0.2f;
742 const float top = float(2 * kMaxQ - 1) + step;
743 const float id = top / max;
744 const float this_scale = 1 / id;
745 for (int k = 0; k < 8; ++k) {
746 for (int i = 0; i < 4; ++i)
747 Laux[4 * k + i] = static_cast<std::int8_t>(imax(0, imin(kMaxQ - 1, level_of(id, xval[4 * k + i]))));
748 const std::uint16_t u = pack(grid, Laux.data() + 4 * k);
749 on_grid_aux[k] = true;
750 if (iq_grid_find(grid.index, u) < 0) {
751 on_grid_aux[k] = false;
752 find_best_neighbour(grid, u, xval.data() + 4 * k, waux.data() + 4 * k, this_scale, Laux.data() + 4 * k);
753 }
754 }
755 float sumqx = 0, sumq2 = 0;
756 sums(32, weight.data(), xval.data(), Laux.data(), sumqx, sumq2);
757 if (sumq2 > 0 && sumqx * sumqx > best * sumq2) {
758 scale = sumqx / sumq2;
759 best = scale * sumqx;
760 L = Laux;
761 on_grid = on_grid_aux;
762 }
763 }
764 int n_not_ongrid = 0;
765 for (int k = 0; k < 8; ++k)
766 n_not_ongrid += on_grid[k] ? 0 : 1;
767 if (n_not_ongrid > 0 && scale > 0) {
768 const float id = 1 / scale;
769 for (int k = 0; k < 8; ++k) {
770 if (!S && on_grid[k]) // IQ3_S requantises every group
771 continue;
772 std::array<std::int8_t, 4> l4{};
773 for (int i = 0; i < 4; ++i)
774 l4[i] = static_cast<std::int8_t>(imax(0, imin(kMaxQ - 1, level_of(id, xval[4 * k + i]))));
775 const std::uint16_t u = pack(grid, l4.data());
776 int g = iq_grid_find(grid.index, u);
777 if (g < 0)
778 g = find_best_neighbour(grid, u, xval.data() + 4 * k, waux.data() + 4 * k, scale, L.data() + 4 * k);
779 for (int i = 0; i < 4; ++i)
780 L[4 * k + i] = static_cast<std::int8_t>(grid.level(std::size_t(g), i));
781 }
782 float sumqx = 0, sumq2 = 0;
783 sums(32, weight.data(), xval.data(), L.data(), sumqx, sumq2);
784 if (sumq2 > 0)
785 scale = sumqx / sumq2;
786 }
787 if (scale < 0) {
788 scale = -scale;
789 for (int k = 0; k < 4; ++k)
790 block_signs[k] = static_cast<std::uint8_t>(S ? ~block_signs[k] : (~block_signs[k]) & 127);
791 }
792 for (int k = 0; k < 8; ++k) {
793 const int g = iq_grid_find(grid.index, pack(grid, L.data() + 4 * k));
794 assert(g >= 0 && "snicholls::ml: IQ3 found a point not on the grid");
795 if constexpr (S) {
796 y[ibl].qs[qs_off + std::size_t(k)] = static_cast<std::uint8_t>(g & 255);
797 y[ibl].qh[ib] = static_cast<std::uint8_t>(y[ibl].qh[ib] | ((g >> 8) << k));
798 } else {
799 q3[8 * ib + k] = static_cast<std::uint8_t>(g);
800 }
801 }
802 if constexpr (S) {
803 qs_off += 8;
804 for (int k = 0; k < 4; ++k)
805 y[ibl].signs[signs_off + std::size_t(k)] = block_signs[k];
806 signs_off += 4;
807 } else {
808 store32(q3.data() + 64 + 4 * ib, block_signs[0] | (std::uint32_t(block_signs[1]) << 7) | (std::uint32_t(block_signs[2]) << 14)
809 | (std::uint32_t(block_signs[3]) << 21));
810 }
811 scales[ib] = scale;
812 largest_scale = fmax(largest_scale, scale);
813 }
814 if (largest_scale == 0.0f)
815 continue; // the block stays zero
816 const float d = largest_scale / 31;
817 const float fudged = d * (S ? 1.033f : 1.0125f);
818 y[ibl].d = detail::quantize::fp16(fudged);
819 const float id = 1 / d;
820 if constexpr (S) {
821 for (int ib = 0; ib < 8; ib += 2)
822 y[ibl].scales[ib / 2] = static_cast<std::uint8_t>(scale_code(id, scales[ib]) | (scale_code(id, scales[ib + 1]) << 4));
823 } else {
824 for (int ib = 0; ib < 8; ++ib)
825 store32(q3.data() + 64 + 4 * ib, load32(q3.data() + 64 + 4 * ib) | (std::uint32_t(scale_code(id, scales[ib])) << 28));
826 for (int i = 0; i < 96; ++i)
827 y[ibl].qs[i] = q3[i];
828 }
829 }
830 }

832 // IQ1's three values with the shift: -1 + δ, δ, 1 + δ and -1 - δ, -δ, 1 - δ, δ = 1/8
833 inline constexpr float iq1_delta = 0.125f;
834 inline constexpr std::array<float, 3> x_p{-1 + iq1_delta, iq1_delta, 1 + iq1_delta};
835 inline constexpr std::array<float, 3> x_m{-1 - iq1_delta, -iq1_delta, 1 - iq1_delta};

837 // The order qsort with ggml's iq1_sort_helper gives: ascending by value; equal values stay in their order
838 template <std::size_t N>
839 constexpr std::array<int, N> sorted_order(const float* xb)
840 {
841 std::array<int, N> idx{};
842 for (std::size_t j = 0; j < N; ++j)
843 idx[j] = int(j);
844 for (std::size_t j = 1; j < N; ++j) {
845 const int v = idx[j];
846 std::size_t k = j;
847 while (k > 0 && xb[v] < xb[idx[k - 1]]) {
848 idx[k] = idx[k - 1];
849 --k;
850 }
851 idx[k] = v;
852 }
853 return idx;
854 }

856 // ggml's iq1_find_best_neighbour2: the neighbour nearest scale · xx[level] in the weighted norm
857 constexpr int iq1_neighbour(std::uint16_t u, const float* xb, const float* weight, float scale, const std::array<float, 3>& xx, std::int8_t* L)
858 {
859 const auto score = [&](std::size_t j) {
860 float d2 = 0;
861 for (int i = 0; i < 8; ++i) {
862 const float q = xx[std::size_t(iq1s.level(j, i))];
863 const float sq = scale * q;
864 const float diff = sq - xb[i];
865 const float wd = weight[i] * diff;
866 const float wdd = wd * diff;
867 d2 += wdd;
868 }
869 return d2;
870 };
871 int g = best_neighbour(iq1s, u, score);
872 if (g < 0) { // ggml's fallback reads past the group here; core searches the whole grid
873 float best = flt_max;
874 for (std::size_t j = 0; j < 2048; ++j) {
875 const float e = score(j);
876 if (e < best) {
877 best = e;
878 g = int(j);
879 }
880 }
881 }
882 assert(g >= 0 && "snicholls::ml: IQ1 found no grid point");
883 for (int i = 0; i < 8; ++i)
884 L[i] = static_cast<std::int8_t>(iq1s.level(std::size_t(g), i));
885 return g;
886 }

888 // IQ1_S: ggml's quantize_row_iq1_s_impl
889 constexpr void quantize(std::span<const float> x, std::span<const float> qw_row, std::span<block_iq1_s> y)
890 {
891 constexpr int bs = 32;
892 std::array<float, 8> scales{};
893 std::array<float, bs> weight{};
894 std::array<std::int8_t, bs> L{};
895 std::array<std::int8_t, 8> shifts{};
896 for (std::size_t ibl = 0; ibl < y.size(); ++ibl) {
897 y[ibl] = block_iq1_s{};
898 y[ibl].d = detail::quantize::fp16(0.0f);
899 float largest_scale = 0;
900 const float* xbl = x.data() + 256 * ibl;
901 const float two = 2 * sum_squares(xbl, 256);
902 const float sigma2 = two / 256.0f;
903 for (int ib = 0; ib < 8; ++ib) {
904 const float* xb = xbl + bs * ib;
905 const float* qw = qw_row.data() + 256 * ibl + bs * ib;
906 for (int i = 0; i < bs; ++i)
907 weight[i] = detail::quantize::importance_weight(qw[i], sigma2, xb[i]);
908 float max = detail::quantize::abs(xb[0]);
909 for (int i = 1; i < bs; ++i)
910 max = fmax(max, detail::quantize::abs(xb[i]));
911 if (max < 1e-12f) {
912 scales[ib] = 0;
913 shifts[ib] = 1;
914 L.fill(1);
915 continue;
916 }
917 const auto idx = sorted_order<bs>(xb);
918 std::array<float, bs + 1> sumx{}, sumw{};
919 for (int j = 0; j < bs; ++j) {
920 const int i = idx[std::size_t(j)];
921 const float wx = weight[i] * xb[i];
922 sumx[j + 1] = sumx[j] + wx;
923 sumw[j + 1] = sumw[j] + weight[i];
924 }
925 float best_score = -flt_max, scale = max;
926 int besti1 = -1, besti2 = -1, best_shift = 0;
927 const auto try_split = [&](int i1, int i2, const std::array<float, 3>& v, int shift) {
928 const float a = sumx[i1] - sumx[0], b = sumx[i2] - sumx[i1], c = sumx[bs] - sumx[i2];
929 const float av = a * v[0], bv = b * v[1], cv = c * v[2];
930 const float ab = av + bv;
931 const float sumqx = ab + cv;
932 const float A = sumw[i1] - sumw[0], B = sumw[i2] - sumw[i1], C = sumw[bs] - sumw[i2];
933 const float Av = A * v[0], Bv = B * v[1], Cv = C * v[2];
934 const float Avv = Av * v[0], Bvv = Bv * v[1], Cvv = Cv * v[2];
935 const float AB = Avv + Bvv;
936 const float sumq2 = AB + Cvv;
937 if (sumq2 > 0 && sumqx * sumqx > best_score * sumq2) {
938 scale = sumqx / sumq2;
939 best_score = scale * sumqx;
940 besti1 = i1;
941 besti2 = i2;
942 best_shift = shift;
943 }
944 };
945 for (int i1 = 0; i1 <= bs; ++i1)
946 for (int i2 = i1; i2 <= bs; ++i2) {
947 try_split(i1, i2, x_p, 1);
948 try_split(i1, i2, x_m, -1);
949 }
950 if (besti1 < 0 || besti2 < 0 || best_shift == 0) {
951 scales[ib] = 0;
952 shifts[ib] = 1;
953 L.fill(1);
954 continue;
955 }
956 for (int j = 0; j < bs; ++j)
957 L[std::size_t(idx[std::size_t(j)])] = static_cast<std::int8_t>(j < besti1 ? 0 : j < besti2 ? 1 : 2);
958 if (scale < 0) {
959 for (int j = 0; j < bs; ++j)
960 L[j] = static_cast<std::int8_t>(2 - L[j]);
961 scale = -scale;
962 best_shift = -best_shift;
963 }
964 bool all_on_grid = true;
965 const auto& xx = best_shift == 1 ? x_p : x_m;
966 std::array<int, 4> index{};
967 for (int k = 0; k < 4; ++k) {
968 const std::uint16_t u = pack(iq1s, L.data() + 8 * k);
969 int g = iq_grid_find(iq1s.index, u);
970 if (g < 0) {
971 all_on_grid = false;
972 g = iq1_neighbour(u, xb + 8 * k, weight.data() + 8 * k, scale, xx, L.data() + 8 * k);
973 }
974 index[k] = g;
975 }
976 if (!all_on_grid) {
977 float sumqx = 0, sumq2 = 0;
978 for (int k = 0; k < 4; ++k)
979 for (int j = 0; j < 8; ++j) {
980 const float w = weight[8 * k + j];
981 const float q = xx[std::size_t(iq1s.level(std::size_t(index[k]), j))];
982 const float wq = w * q;
983 const float wqx = wq * xb[8 * k + j];
984 sumqx += wqx;
985 const float wqq = wq * q;
986 sumq2 += wqq;
987 }
988 if (sumqx > 0 && sumq2 > 0)
989 scale = sumqx / sumq2;
990 }
991 std::uint16_t h = 0;
992 for (int k = 0; k < 4; ++k) {
993 y[ibl].qs[4 * ib + k] = static_cast<std::uint8_t>(index[k] & 255);
994 h = static_cast<std::uint16_t>(h | ((index[k] >> 8) << (3 * k)));
995 }
996 y[ibl].qh[ib] = h;
997 scales[ib] = scale;
998 shifts[ib] = static_cast<std::int8_t>(best_shift);
999 largest_scale = fmax(largest_scale, scale);
1000 }
1001 if (largest_scale == 0.0f)
1002 continue;
1003 const float d = largest_scale / 15;
1004 const float fudged = d * 1.125f;
1005 y[ibl].d = detail::quantize::fp16(fudged);
1006 const float id = 1 / d;
1007 for (int ib = 0; ib < 8; ++ib) {
1008 int l = imax(0, imin(7, level_of(id, scales[ib])));
1009 if (shifts[ib] == -1)
1010 l |= 8;
1011 y[ibl].qh[ib] = static_cast<std::uint16_t>(y[ibl].qh[ib] | (l << 12));
1012 }
1013 }
1014 }

1016 // IQ1_M: ggml's quantize_row_iq1_m_impl
1017 constexpr void quantize(std::span<const float> x, std::span<const float> qw_row, std::span<block_iq1_m> y)
1018 {
1019 constexpr int bs = 16;
1020 constexpr std::array<std::uint8_t, 4> masks{0x00, 0x80, 0x08, 0x88};
1021 std::array<float, 16> scales{};
1022 std::array<float, bs> weight{};
1023 std::array<std::int8_t, bs> L{};
1024 std::array<std::int8_t, 16> shifts{};
1025 const auto weights_of = [&](const float* xb, std::size_t offset, float sigma2) {
1026 if (!qw_row.empty()) {
1027 for (int i = 0; i < bs; ++i)
1028 weight[i] = detail::quantize::importance_weight(qw_row[offset + std::size_t(i)], sigma2, xb[i]);
1029 } else {
1030 for (int i = 0; i < bs; ++i)
1031 weight[i] = xb[i] * xb[i];
1032 }
1033 };
1034 const auto shift_values = [](int best_k, int k) -> const std::array<float, 3>& {
1035 if (k == 0)
1036 return best_k < 2 ? x_p : x_m;
1037 return best_k % 2 == 0 ? x_p : x_m;
1038 };
1039 for (std::size_t ibl = 0; ibl < y.size(); ++ibl) {
1040 y[ibl] = block_iq1_m{};
1041 float largest_scale = 0;
1042 const float* xbl = x.data() + 256 * ibl;
1043 const float two = 2 * sum_squares(xbl, 256);
1044 const float sigma2 = two / 256.0f;
1045 for (int ib = 0; ib < 16; ++ib) {
1046 const float* xb = xbl + bs * ib;
1047 weights_of(xb, 256 * ibl + std::size_t(bs * ib), sigma2);
1048 float max = detail::quantize::abs(xb[0]);
1049 for (int i = 1; i < bs; ++i)
1050 max = fmax(max, detail::quantize::abs(xb[i]));
1051 if (max < 1e-7f) {
1052 scales[ib] = 0;
1053 shifts[ib] = 0;
1054 L.fill(1);
1055 continue;
1056 }
1057 const auto idx = sorted_order<bs>(xb);
1058 float best_score = -flt_max, scale = max;
1059 int besti1 = -1, besti2 = -1, best_k = -1;
1060 std::array<std::array<float, bs + 1>, 2> px{}, pw{};
1061 for (int j = 0; j < bs; ++j) {
1062 const int i = idx[std::size_t(j)];
1063 const int h = i < bs / 2 ? 0 : 1;
1064 const float wx = weight[i] * xb[i];
1065 px[h][j + 1] = px[h][j] + wx;
1066 px[1 - h][j + 1] = px[1 - h][j];
1067 pw[h][j + 1] = pw[h][j] + weight[i];
1068 pw[1 - h][j + 1] = pw[1 - h][j];
1069 }
1070 const std::array<float, 2> txs{px[0][bs], px[1][bs]};
1071 const std::array<float, 2> tws{pw[0][bs], pw[1][bs]};
1072 const float p2 = x_p[2], m2 = x_m[2];
1073 const auto diff_sq = [](float a, float b) {
1074 const float a2 = a * a;
1075 const float b2 = b * b;
1076 return a2 - b2;
1077 };
1078 const float cp1 = diff_sq(x_p[0], x_p[1]), cp2 = diff_sq(x_p[1], x_p[2]);
1079 const float cm1 = diff_sq(x_m[0], x_m[1]), cm2 = diff_sq(x_m[1], x_m[2]);
1080 for (int i1 = 0; i1 <= bs; ++i1)
1081 for (int i2 = i1; i2 <= bs; ++i2) {
1082 std::array<float, 2> qx_p{}, qx_m{}, q2_p{}, q2_m{};
1083 for (int h = 0; h < 2; ++h) {
1084 const float sx = px[h][i1] + px[h][i2];
1085 const float tp = txs[h] * p2;
1086 qx_p[h] = tp - sx;
1087 const float tm = txs[h] * m2;
1088 qx_m[h] = tm - sx;
1089 const auto q2 = [&](float v2, float c1, float c2) {
1090 const float t = tws[h] * v2;
1091 const float tt = t * v2;
1092 const float a = pw[h][i1] * c1;
1093 const float s1 = tt + a;
1094 const float b = pw[h][i2] * c2;
1095 return s1 + b;
1096 };
1097 q2_p[h] = q2(p2, cp1, cp2);
1098 q2_m[h] = q2(m2, cm1, cm2);
1099 }
1100 const std::array<float, 4> sumqx{qx_p[0] + qx_p[1], qx_p[0] + qx_m[1], qx_m[0] + qx_p[1], qx_m[0] + qx_m[1]};
1101 const std::array<float, 4> sumq2{q2_p[0] + q2_p[1], q2_p[0] + q2_m[1], q2_m[0] + q2_p[1], q2_m[0] + q2_m[1]};
1102 for (int k = 0; k < 4; ++k)
1103 if (sumq2[k] > 0 && sumqx[k] * sumqx[k] > best_score * sumq2[k]) {
1104 scale = sumqx[k] / sumq2[k];
1105 best_score = scale * sumqx[k];
1106 besti1 = i1;
1107 besti2 = i2;
1108 best_k = k;
1109 }
1110 }
1111 if (besti1 < 0 || besti2 < 0 || best_k < 0) {
1112 scales[ib] = 0;
1113 shifts[ib] = 0;
1114 L.fill(1);
1115 continue;
1116 }
1117 for (int j = 0; j < bs; ++j)
1118 L[std::size_t(idx[std::size_t(j)])] = static_cast<std::int8_t>(j < besti1 ? 0 : j < besti2 ? 1 : 2);
1119 if (scale < 0) {
1120 for (int j = 0; j < bs; ++j)
1121 L[j] = static_cast<std::int8_t>(2 - L[j]);
1122 scale = -scale;
1123 best_k = 3 - best_k;
1124 }
1125 bool all_on_grid = true;
1126 std::array<int, 2> index{};
1127 for (int k = 0; k < 2; ++k) {
1128 const std::uint16_t u = pack(iq1s, L.data() + 8 * k);
1129 int g = iq_grid_find(iq1s.index, u);
1130 if (g < 0) {
1131 all_on_grid = false;
1132 g = iq1_neighbour(u, xb + 8 * k, weight.data() + 8 * k, scale, shift_values(best_k, k), L.data() + 8 * k);
1133 }
1134 index[k] = g;
1135 }
1136 if (!all_on_grid) {
1137 float sumqx = 0, sumq2 = 0;
1138 for (int k = 0; k < 2; ++k) {
1139 const auto& xx = shift_values(best_k, k);
1140 for (int j = 0; j < 8; ++j) {
1141 const float w = weight[8 * k + j];
1142 const float q = xx[std::size_t(iq1s.level(std::size_t(index[k]), j))];
1143 const float wq = w * q;
1144 const float wqx = wq * xb[8 * k + j];
1145 sumqx += wqx;
1146 const float wqq = wq * q;
1147 sumq2 += wqq;
1148 }
1149 }
1150 if (sumqx > 0 && sumq2 > 0)
1151 scale = sumqx / sumq2;
1152 }
1153 y[ibl].qs[2 * ib] = static_cast<std::uint8_t>(index[0] & 255);
1154 y[ibl].qs[2 * ib + 1] = static_cast<std::uint8_t>(index[1] & 255);
1155 y[ibl].qh[ib] = static_cast<std::uint8_t>((index[0] >> 8) | ((index[1] >> 8) << 4));
1156 scales[ib] = scale;
1157 shifts[ib] = static_cast<std::int8_t>(best_k);
1158 largest_scale = fmax(largest_scale, scale);
1159 }
1160 if (largest_scale == 0.0f)
1161 continue;
1162 std::array<std::uint16_t, 4> sc{};
1163 float d = largest_scale / 15;
1164 const float id = 1 / d;
1165 float sumqx = 0, sumq2 = 0;
1166 for (int ib = 0; ib < 16; ++ib) {
1167 const int l = imax(0, imin(7, level_of(id, scales[ib])));
1168 sc[ib / 4] = static_cast<std::uint16_t>(sc[ib / 4] | (l << (3 * (ib % 4))));
1169 y[ibl].qh[ib] = static_cast<std::uint8_t>(y[ibl].qh[ib] | masks[std::size_t(shifts[ib])]);
1170 const float* xb = xbl + bs * ib;
1171 weights_of(xb, 256 * ibl + std::size_t(bs * ib), sigma2);
1172 for (int k = 0; k < 2; ++k) {
1173 const auto& xx = shift_values(shifts[ib], k);
1174 const int g = y[ibl].qs[2 * ib + k] + ((y[ibl].qh[ib] << (8 - 4 * k)) & 0x700);
1175 for (int j = 0; j < 8; ++j) {
1176 const float w = weight[8 * k + j];
1177 const float v = xx[std::size_t(iq1s.level(std::size_t(g), j))];
1178 const float q = v * float(2 * l + 1);
1179 const float wq = w * q;
1180 const float wqx = wq * xb[8 * k + j];
1181 sumqx += wqx;
1182 const float wqq = wq * q;
1183 sumq2 += wqq;
1184 }
1185 }
1186 }
1187 if (sumq2 > 0)
1188 d = sumqx / sumq2;
1189 const float fudged = d * 1.1125f;
1190 const std::uint16_t s = detail::quantize::fp16(fudged).bits;
1191 sc[0] = static_cast<std::uint16_t>(sc[0] | ((s & 0x000f) << 12));
1192 sc[1] = static_cast<std::uint16_t>(sc[1] | ((s & 0x00f0) << 8));
1193 sc[2] = static_cast<std::uint16_t>(sc[2] | ((s & 0x0f00) << 4));
1194 sc[3] = static_cast<std::uint16_t>(sc[3] | (s & 0xf000));
1195 for (int k = 0; k < 4; ++k) {
1196 y[ibl].scales[2 * k] = static_cast<std::uint8_t>(sc[k]);
1197 y[ibl].scales[2 * k + 1] = static_cast<std::uint8_t>(sc[k] >> 8);
1198 }
1199 }
1200 }

1218 // ggml's quantize_row_iq4_nl_impl for one super-block (32 for IQ4_NL, 256 for IQ4_XS); L persists between
1219 // calls, as ggml's buffer does
1220 constexpr void iq4(int super_block_size, const float* x, float16& dh, std::uint8_t* q4, std::uint16_t* scales_h, std::uint8_t* scales_l,
1221 std::uint8_t* L, const float* quant_weights, int ntry)
1222 {
1223 constexpr int block_size = 32;
1224 float sigma2 = sum_squares(x, super_block_size);
1225 sigma2 *= 2.0f / float(super_block_size);
1226 for (int i = 0; i < super_block_size / 2; ++i)
1227 q4[i] = 0;
1228 dh = detail::quantize::fp16(0.0f);
1229 const auto& mid = iq4nl_mid;
1230 std::array<float, 8> scales{};
1231 std::array<float, block_size> weight{};
1232 float largest_scale = 0, amax_scale = 0;
1233 const int nb = super_block_size / block_size;
1234 for (int ib = 0; ib < nb; ++ib) {
1235 const float* xb = x + ib * block_size;
1236 std::uint8_t* Lb = L + ib * block_size;
1237 if (quant_weights) {
1238 const float* qw = quant_weights + ib * block_size;
1239 for (int j = 0; j < block_size; ++j)
1240 weight[j] = detail::quantize::importance_weight(qw[j], sigma2, xb[j]);
1241 } else {
1242 for (int j = 0; j < block_size; ++j)
1243 weight[j] = xb[j] * xb[j];
1244 }
1245 float amax = 0, max = 0;
1246 for (int j = 0; j < block_size; ++j) {
1247 const float ax = detail::quantize::abs(xb[j]);
1248 if (ax > amax) {
1249 amax = ax;
1250 max = xb[j];
1251 }
1252 }
1253 if (amax < detail::quantize::group_max_eps) {
1254 scales[ib] = 0;
1255 continue;
1256 }
1257 const float v0 = float(iq4nl_values[0]);
1258 float d = ntry > 0 ? -max / v0 : max / v0;
1259 const int nt = ntry >= 0 ? 2 * ntry + 2 : 1;
1260 std::array<float, 16> idt{}, sqx{}, sq2{};
1261 idt[0] = 1 / d;
1262 for (int t = 1; t < nt; ++t)
1263 idt[t] = float(t - 1 - ntry + iq4nl_values[0]) / max;
1264 for (int j = 0; j < block_size; ++j) {
1265 const float v = idt[0] * xb[j];
1266 Lb[j] = static_cast<std::uint8_t>(best_index_iq4nl(mid, v));
1267 }
1268 for (int j = 0; j < block_size; ++j) {
1269 const float w = weight[j];
1270 for (int t = 0; t < nt; ++t) {
1271 const float v = idt[t] * xb[j];
1272 const float q = float(iq4nl_values[std::size_t(best_index_iq4nl(mid, v))]);
1273 const float wq = w * q;
1274 const float wqx = wq * xb[j];
1275 sqx[t] += wqx;
1276 const float wqq = wq * q;
1277 sq2[t] += wqq;
1278 }
1279 }
1280 d = sq2[0] > 0 ? sqx[0] / sq2[0] : 0.0f;
1281 float best = d * sqx[0];
1282 for (int t = 1; t < nt; ++t)
1283 if (sq2[t] > 0 && sqx[t] * sqx[t] > best * sq2[t]) {
1284 d = sqx[t] / sq2[t];
1285 best = d * sqx[t];
1286 }
1287 scales[ib] = d;
1288 const float abs_d = detail::quantize::abs(d);
1289 if (abs_d > amax_scale) {
1290 amax_scale = abs_d;
1291 largest_scale = d;
1292 }
1293 }
1294 if (nb > 1) {
1295 *scales_h = 0;
1296 const float d = -largest_scale / 32;
1297 dh = detail::quantize::fp16(d);
1298 const float id = d != 0.0f ? 1 / d : 0.0f;
1299 for (int ib = 0; ib < nb; ++ib) {
1300 const float v = id * scales[ib];
1301 int l = imax(-32, imin(31, nearest_int(v)));
1302 const float dl = d * float(l);
1303 const float idl = dl != 0.0f ? 1 / dl : 0.0f;
1304 std::uint8_t* Lb = L + ib * block_size;
1305 const float* xb = x + ib * block_size;
1306 for (int j = 0; j < block_size; ++j) {
1307 const float s = idl * xb[j];
1308 Lb[j] = static_cast<std::uint8_t>(best_index_iq4nl(mid, s));
1309 }
1310 l += 32;
1311 const auto l_l = static_cast<std::uint8_t>(l & 0xf);
1312 const auto l_h = static_cast<std::uint8_t>(l >> 4);
1313 if (ib % 2 == 0)
1314 scales_l[ib / 2] = l_l;
1315 else
1316 scales_l[ib / 2] = static_cast<std::uint8_t>(scales_l[ib / 2] | (l_l << 4));
1317 *scales_h = static_cast<std::uint16_t>(*scales_h | (l_h << (2 * (ib % 8))));
1318 }
1319 } else {
1320 dh = detail::quantize::fp16(scales[0]);
1321 if (ntry > 0) {
1322 const float id = scales[0] != 0.0f ? 1 / scales[0] : 0.0f;
1323 for (int j = 0; j < super_block_size; ++j) {
1324 const float s = id * x[j];
1325 L[j] = static_cast<std::uint8_t>(best_index_iq4nl(mid, s));
1326 }
1327 }
1328 }
1329 for (int i = 0; i < super_block_size / 32; ++i)
1330 for (int j = 0; j < 16; ++j)
1331 q4[16 * i + j] = static_cast<std::uint8_t>(L[32 * i + j] | (L[32 * i + 16 + j] << 4));
1332 }
