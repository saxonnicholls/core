385 // The first comparator's ordering if it is not equal, else the next one's.
386 struct lexicographic_op {
387 template <class A, class B, class C>
388 static constexpr auto step(const A& a, const B& b, const C& c)
389 {
390 return std::invoke(c, a, b);
391 }
392
393 template <class A, class B, class C, class C2, class... Cs>
394 static constexpr auto step(const A& a, const B& b, const C& c, const C2& c2, const Cs&... cs)
395 {
396 return then_with_fn{}(std::invoke(c, a, b), [&] { return step(a, b, c2, cs...); });
397 }
