142template <math::math_number T>
143constexpr pair_loss<T> preference_loss(const dpo& c, T chosen, std::type_identity_t<T> rejected, std::type_identity_t<T> ref_chosen,
144 std::type_identity_t<T> ref_rejected) noexcept(std::is_floating_point_v<T>)
145{
146 const T beta = T(c.beta);
147 const T rw = beta * (chosen - ref_chosen), rl = beta * (rejected - ref_rejected);
148 const T z = rw - rl;
149 const T s = sigmoid(-z);
150 return {softplus(-z), -beta * s, beta * s, rw, rl};
151}

160template <math::math_number T>
161constexpr pair_loss<T> preference_loss(const ipo& c, T chosen, std::type_identity_t<T> rejected, std::type_identity_t<T> ref_chosen,
162 std::type_identity_t<T> ref_rejected) noexcept(std::is_floating_point_v<T>)
163{
164 const T tau = T(c.tau);
165 const T lw = chosen - ref_chosen, ll = rejected - ref_rejected;
166 const T e = (lw - ll) - T(1) / (T(2) * tau);
167 return {e * e, T(2) * e, -T(2) * e, tau * lw, tau * ll};
168}

175template <math::math_number T>
176constexpr pair_loss<T> preference_loss(const simpo& c, T chosen, std::type_identity_t<T> rejected, std::type_identity_t<T> = T(0),
177 std::type_identity_t<T> = T(0)) noexcept(std::is_floating_point_v<T>)
178{
179 const T beta = T(c.beta);
180 const T rw = beta * chosen, rl = beta * rejected;
181 const T z = (rw - rl) - T(c.gamma);
182 const T s = sigmoid(-z);
183 return {softplus(-z), -beta * s, beta * s, rw, rl};
184}

189template <math::math_number T>
190constexpr T log_odds_from_logprob(T a) noexcept(std::is_floating_point_v<T>)
191{
192 assert(a < T(0));
193 return a - math::log(-math::expm1(a));
194}

202template <math::math_number T>
203constexpr pair_loss<T> preference_loss(const orpo& c, T chosen, std::type_identity_t<T> rejected, std::type_identity_t<T> = T(0),
204 std::type_identity_t<T> = T(0)) noexcept(std::is_floating_point_v<T>)
205{
206 const T lambda = T(c.lambda);
207 const T z = log_odds_from_logprob(chosen) - log_odds_from_logprob(rejected);
208 const T s = sigmoid(-z);
209 // d log odds(a) / da = 1 / (1 - e^a) = -1 / expm1(a)
210 const T dw = T(-1) / math::expm1(chosen), dl = T(-1) / math::expm1(rejected);
211 return {-chosen + lambda * softplus(-z), T(-1) - lambda * s * dw, lambda * s * dl, lambda * chosen, lambda * rejected};
212}

217template <math::math_number T>
218struct preference_metrics
219{
220 std::uint64_t pairs = 0;
221 std::uint64_t correct = 0;
222 T loss = T(0);
223 T chosen = T(0);
224 T rejected = T(0);
225
227 constexpr void add(const pair_loss<T>& p) noexcept(std::is_floating_point_v<T>)
228 {
229 ++pairs;
230 correct += p.chosen_reward > p.rejected_reward ? 1 : 0;
231 loss += p.loss;
232 chosen += p.chosen_reward;
233 rejected += p.rejected_reward;
234 }
235
237 constexpr void merge(const preference_metrics& o) noexcept(std::is_floating_point_v<T>)
238 {
239 pairs += o.pairs;
240 correct += o.correct;
241 loss += o.loss;
242 chosen += o.chosen;
243 rejected += o.rejected;
244 }
245
247 [[nodiscard]] constexpr T mean_loss() const noexcept(std::is_floating_point_v<T>) { return pairs == 0 ? T(0) : loss / T(pairs); }
249 [[nodiscard]] constexpr T mean_chosen_reward() const noexcept(std::is_floating_point_v<T>) { return pairs == 0 ? T(0) : chosen / T(pairs); }
251 [[nodiscard]] constexpr T mean_rejected_reward() const noexcept(std::is_floating_point_v<T>) { return pairs == 0 ? T(0) : rejected / T(pairs); }
253 [[nodiscard]] constexpr T mean_margin() const noexcept(std::is_floating_point_v<T>) { return mean_chosen_reward() - mean_rejected_reward(); }
255 [[nodiscard]] constexpr double accuracy() const noexcept { return pairs == 0 ? 0.0 : static_cast<double>(correct) / static_cast<double>(pairs); }
256};

269template <math::math_number T, typename B, typename P>
270constexpr T preference_loss_rows(const B&, const P& params, std::span<const T> chosen, std::type_identity_t<std::span<const T>> rejected,
271 std::type_identity_t<std::span<const T>> ref_chosen, std::type_identity_t<std::span<const T>> ref_rejected,
272 std::type_identity_t<std::span<T>> d_chosen, std::type_identity_t<std::span<T>> d_rejected,
273 preference_metrics<T>& metrics) noexcept(std::is_floating_point_v<T>)
274{
275 const std::size_t n = chosen.size();
276 assert(rejected.size() == n && d_chosen.size() == n && d_rejected.size() == n && (ref_chosen.empty() || ref_chosen.size() == n)
277 && (ref_rejected.empty() || ref_rejected.size() == n));
278 if (n == 0)
279 return T(0);
280 const T inv = T(1) / T(n);
281 T sum = T(0);
282 for (std::size_t i = 0; i < n; ++i) {
283 const pair_loss<T> p = preference_loss(params, chosen[i], rejected[i], ref_chosen.empty() ? T(0) : ref_chosen[i],
284 ref_rejected.empty() ? T(0) : ref_rejected[i]);
285 d_chosen[i] = p.d_chosen * inv;
286 d_rejected[i] = p.d_rejected * inv;
287 sum += p.loss;
288 metrics.add(p);
289 }
290 return sum * inv;
291}
