92 constexpr bool packed(std::span<const std::size_t> starts, std::size_t rows) noexcept
93 {
94 if (starts.size() < 2 || starts.front() != 0 || starts.back() != rows)
95 return false;
96 for (std::size_t s = 1; s < starts.size(); ++s)
97 if (starts[s] < starts[s - 1])
98 return false;
99 return true;
100 }

110template <math::math_number T>
111constexpr void pool_rows(pooling_kind kind, std::type_identity_t<std::span<const T>> rows, std::size_t width,
112 std::span<const std::size_t> starts, std::span<T> out)
113{
114 assert(width > 0 && rows.size() % width == 0 && detail::pooling::packed(starts, rows.size() / width));
115 const std::size_t sequences = starts.size() - 1;
116 if (kind == pooling_kind::none) {
117 assert(out.size() == rows.size());
118 // bounded by both sizes, which the assert makes equal: GCC's -Warray-bounds cannot see that under NDEBUG
119 const std::size_t n = rows.size() < out.size() ? rows.size() : out.size();
120 for (std::size_t i = 0; i < n; ++i)
121 out[i] = rows[i];
122 return;
123 }
124 assert(out.size() == sequences * width);
125 for (std::size_t s = 0; s < sequences; ++s) {
126 const std::size_t first = starts[s], n = starts[s + 1] - first;
127 const auto dst = out.subspan(s * width, width);
128 if (kind == pooling_kind::mean) {
129 mean_pool_rows<detail::pooling::host, T>(detail::pooling::host{}, rows.subspan(first * width, n * width), width, dst);
130 continue;
131 }
132 assert(n > 0);
133 const std::size_t r = kind == pooling_kind::last ? first + n - 1 : first;
134 for (std::size_t c = 0; c < width; ++c)
135 dst[c] = rows[r * width + c];
136 }
137}

146template <math::math_number T>
147constexpr void pool_rows_backward(pooling_kind kind, std::type_identity_t<std::span<const T>> dout, std::size_t width,
148 std::span<const std::size_t> starts, std::span<T> drows)
149{
150 assert(width > 0 && drows.size() % width == 0 && detail::pooling::packed(starts, drows.size() / width));
151 const std::size_t sequences = starts.size() - 1;
152 if (kind == pooling_kind::none) {
153 assert(dout.size() == drows.size());
154 for (std::size_t i = 0; i < drows.size(); ++i)
155 drows[i] += dout[i];
156 return;
157 }
158 assert(dout.size() == sequences * width);
159 for (std::size_t s = 0; s < sequences; ++s) {
160 const std::size_t first = starts[s], n = starts[s + 1] - first;
161 const auto g = dout.subspan(s * width, width);
162 if (kind == pooling_kind::mean) {
163 mean_pool_rows_backward<detail::pooling::host, T>(detail::pooling::host{}, g, width, drows.subspan(first * width, n * width));
164 continue;
165 }
166 const std::size_t r = kind == pooling_kind::last ? first + n - 1 : first;
167 for (std::size_t c = 0; c < width; ++c)
168 drows[r * width + c] += g[c];
169 }

208template <typename S, math::math_number T>
209constexpr void rank_head(const rank_head_weights<S, T>& w, std::type_identity_t<std::span<const T>> pooled,
210 std::type_identity_t<std::span<T>> scratch, std::type_identity_t<std::span<T>> scores)
211{
212 std::span<const T> x = pooled;
213 if (w.cls.rows > 0) {
214 assert(w.cls.cols == pooled.size() && scratch.size() == w.cls.rows);
215 linear(w.cls, pooled, scratch, w.cls_bias);
216 for (T& v : scratch)
217 switch (w.activation) {
218 case rank_activation::tanh: v = math::tanh(v); break;
219 case rank_activation::gelu: v = gelu(v); break;
220 case rank_activation::relu: v = relu(v); break;
221 case rank_activation::identity: break;
222 }
223 if (!w.norm_weight.empty())
224 layer_norm(std::span<const T>(scratch), w.norm_weight, w.norm_bias, scratch, w.norm_eps);
225 x = scratch;
226 }
227 if (w.out.rows > 0) {
228 assert(w.out.cols == x.size() && scores.size() == w.out.rows);
229 linear(w.out, x, scores, w.out_bias);
230 } else {
231 assert(scores.size() == x.size());
232 std::copy(x.begin(), x.end(), scores.begin());
233 }
234 if (w.softmax)
235 softmax(std::span<const T>(scores), scores);
236}
