69 // The tokens past the first K - 1 read only the inputs: the same sums as the general loop, without its branch, with
70 // the common widths' tap counts known to the compiler (fully unrolled)
71 template <std::size_t Taps, bool Bias, typename T>
72 constexpr void body(const T* taps, const T* bias, std::size_t ch, std::size_t width, const T* x, T* y, std::size_t first, std::size_t n)
73 {
74 const std::size_t K = Taps == 0 ? width : Taps;
75 for (std::size_t i = first; i < n; ++i) {
76 const T* window = x + (i + 1 - K) * ch;
77 T* out = y + i * ch;
78 for (std::size_t c = 0; c < ch; ++c) {
79 T sum = T(0);
80 for (std::size_t j = 0; j < K; ++j)
81 sum += taps[c * K + j] * window[j * ch + c];
82 if constexpr (Bias)
83 out[c] = sum + bias[c];
84 else
85 out[c] = sum;
86 }
87 }
88 }

90 template <typename W, typename T>
91 constexpr void body_for_width(const W& w, const T* x, T* y, std::size_t first, std::size_t n)
92 {
93 const T* bias = w.bias.empty() ? nullptr : w.bias.data();
94 const auto run = [&](auto taps, auto with_bias) {
95 body<decltype(taps)::value, decltype(with_bias)::value>(w.data.data(), bias, w.channels, w.width, x, y, first, n);
96 };
97 const auto by_width = [&](auto with_bias) {
98 switch (w.width) {
99 case 2: run(std::integral_constant<std::size_t, 2>{}, with_bias); break;
100 case 3: run(std::integral_constant<std::size_t, 3>{}, with_bias); break;
101 case 4: run(std::integral_constant<std::size_t, 4>{}, with_bias); break;
102 default: run(std::integral_constant<std::size_t, 0>{}, with_bias); break;
103 }
104 };
105 if (bias == nullptr)
106 by_width(std::false_type{});
107 else
108 by_width(std::true_type{});
109 }

140template <typename B, math::math_number T>
141constexpr void short_conv_forward(const B&, const short_conv_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
142 std::type_identity_t<std::span<T>> ys, std::size_t sequences = 1,
143 std::type_identity_t<std::span<T>> states = {})
144{
145 const std::size_t ch = w.channels, K = w.width, sk = short_conv_state_size(ch, K);
146 assert(K > 0 && w.data.size() == ch * K && (w.bias.empty() || w.bias.size() == ch) && sequences > 0);
147 assert(xs.size() % (ch * sequences) == 0 && ys.size() == xs.size() && (states.empty() || states.size() == sequences * sk));
148 const std::size_t n = xs.size() / (ch * sequences);
149 for (std::size_t s = 0; s < sequences; ++s) {
150 const auto x = xs.subspan(s * n * ch, n * ch);
151 const auto y = ys.subspan(s * n * ch, n * ch);
152 const auto st = states.empty() ? std::span<T>{} : states.subspan(s * sk, sk);
153 const std::size_t head = n < K - 1 ? n : K - 1; // the tokens that read the state
154 for (std::size_t i = 0; i < head; ++i)
155 for (std::size_t c = 0; c < ch; ++c) {
156 T sum = T(0);
157 for (std::size_t j = 0; j < K; ++j) {
158 const std::size_t back = K - 1 - j; // inputs before token i
159 const T in = back <= i ? x[(i - back) * ch + c] : st.empty() ? T(0) : st[(K - 1 + i - back) * ch + c];
160 sum += w.data[c * K + j] * in;
161 }
162 y[i * ch + c] = w.bias.empty() ? sum : sum + w.bias[c];
163 }
164 detail::short_conv::body_for_width(w, x.data(), y.data(), head, n);
165 for (std::size_t r = 0; r + 1 < K && !st.empty(); ++r) { // the last K - 1 inputs, oldest first
166 const std::size_t from = r + n; // in the old state's rows, then x's
167 for (std::size_t c = 0; c < ch; ++c)
168 st[r * ch + c] = from < K - 1 ? st[from * ch + c] : x[(from - (K - 1)) * ch + c];
169 }
170 }
171}

183template <typename B, math::math_number T>
184constexpr void short_conv_backward(const B&, const short_conv_weights<T>& w, std::type_identity_t<std::span<const T>> xs,
185 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
186 std::type_identity_t<std::span<T>> dw, std::type_identity_t<std::span<T>> db,
187 std::size_t sequences = 1, std::type_identity_t<std::span<const T>> initial = {},
188 std::type_identity_t<std::span<T>> dinitial = {})
189{
190 const std::size_t ch = w.channels, K = w.width, sk = short_conv_state_size(ch, K);
191 assert(K > 0 && w.data.size() == ch * K && sequences > 0 && xs.size() % (ch * sequences) == 0);
192 assert(dys.size() == xs.size() && dxs.size() == xs.size() && dw.size() == ch * K && (db.empty() || db.size() == ch));
193 assert((initial.empty() || initial.size() == sequences * sk) && (dinitial.empty() || dinitial.size() == sequences * sk));
194 const std::size_t n = xs.size() / (ch * sequences);
195 for (std::size_t s = 0; s < sequences; ++s) {
196 const auto x = xs.subspan(s * n * ch, n * ch);
197 const auto dy = dys.subspan(s * n * ch, n * ch);
198 const auto dx = dxs.subspan(s * n * ch, n * ch);
199 for (std::size_t i = 0; i < n; ++i)
200 for (std::size_t c = 0; c < ch; ++c) {
201 T sum = T(0); // the outputs token i feeds: i + K - 1 - j
202 for (std::size_t j = 0; j < K; ++j)
203 if (i + K - 1 - j < n)
204 sum += w.data[c * K + j] * dy[(i + K - 1 - j) * ch + c];
205 dx[i * ch + c] = sum;
206 }
207 for (std::size_t c = 0; c < ch; ++c)
208 for (std::size_t j = 0; j < K; ++j) {
209 T sum = T(0);
210 for (std::size_t i = 0; i < n; ++i) {
211 const std::size_t back = K - 1 - j;
212 const T in = back <= i ? x[(i - back) * ch + c] : initial.empty() ? T(0) : initial[s * sk + (K - 1 + i - back) * ch + c];
213 sum += dy[i * ch + c] * in;
214 }
215 dw[c * K + j] += sum;
216 }
217 if (!db.empty())
218 for (std::size_t i = 0; i < n; ++i)
219 for (std::size_t c = 0; c < ch; ++c)
220 db[c] += dy[i * ch + c];
221 if (!dinitial.empty())
222 for (std::size_t r = 0; r + 1 < K; ++r) // state row r feeds output i through tap r - i
223 for (std::size_t c = 0; c < ch; ++c) {
224 T sum = T(0);
225 for (std::size_t i = 0; i <= r && i < n; ++i)
226 sum += w.data[c * K + (r - i)] * dy[i * ch + c];
227 dinitial[s * sk + r * ch + c] = sum;
228 }
229 }
230}

237template <math::math_number T>
238constexpr void short_conv_step(const short_conv_weights<T>& w, std::type_identity_t<std::span<const T>> x,
239 std::type_identity_t<std::span<T>> y, std::type_identity_t<std::span<T>> state)
240{
241 const std::size_t ch = w.channels, K = w.width;
242 assert(K > 0 && x.size() == ch && y.size() == ch && state.size() == short_conv_state_size(ch, K));
243 for (std::size_t c = 0; c < ch; ++c) {
244 T sum = T(0);
245 for (std::size_t j = 0; j < K; ++j)
246 sum += w.data[c * K + j] * (j + 1 < K ? state[j * ch + c] : x[c]);
247 y[c] = w.bias.empty() ? sum : sum + w.bias[c];
248 }
249 for (std::size_t r = 0; r + 1 < K; ++r)
250 for (std::size_t c = 0; c < ch; ++c)
251 state[r * ch + c] = r + 2 < K ? state[(r + 1) * ch + c] : x[c];
252}
