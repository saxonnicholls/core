116 [[nodiscard]] constexpr capability_result<https_url> parse_https_url(std::string_view url) noexcept
117 {
118 constexpr std::string_view scheme = "https://";
119 if (!url.starts_with(scheme) || url.find('#') != std::string_view::npos)
120 return fail(capability_errc::invalid_argument, {}, "not an https URL");
121 std::string_view rest = url.substr(scheme.size());
122 const auto end = rest.find_first_of("/?");
123 std::string_view authority = rest.substr(0, end);
124 https_url out;
125 out.target = end == std::string_view::npos ? std::string_view{} : rest.substr(end);
126 if (const auto colon = authority.find(':'); colon != std::string_view::npos) {
127 const auto digits = authority.substr(colon + 1);
128 if (digits.empty() || digits.size() > 5)
129 return fail(capability_errc::invalid_argument, {}, "not a port");
130 std::uint32_t port = 0;
131 for (const char c : digits) {
132 if (c < '0' || c > '9')
133 return fail(capability_errc::invalid_argument, {}, "not a port");
134 port = port * 10 + static_cast<std::uint32_t>(c - '0');
135 }
136 if (port == 0 || port > 65535)
137 return fail(capability_errc::invalid_argument, {}, "not a port");
138 out.port = static_cast<std::uint16_t>(port);
139 authority = authority.substr(0, colon);
140 }
141 if (authority.empty())
142 return fail(capability_errc::invalid_argument, {}, "no host");
143 for (const char c : authority)
144 if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '-' || c == '.'))
145 return fail(capability_errc::invalid_argument, {}, "not a host name");
146 out.host = authority;
147 return ok(out);
148 }
