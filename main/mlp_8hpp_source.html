82template <mlp_kind K, math::math_number T>
83constexpr T mlp_activation(T x)
84{
85 switch (K) {
86 case mlp_kind::gelu: return gelu(x);
87 case mlp_kind::gelu_tanh: return gelu_tanh(x);
88 case mlp_kind::relu_squared: return relu_squared(x);
89 case mlp_kind::swiglu: return silu(x);
90 }
91 return x;
92}

99template <transformer_config C, weight_matrix W, math::math_number T, row X, mutable_row Y>
100 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Y>, T>
101constexpr void mlp_forward(const mlp_weights<W, T>& w, const X& x, Y&& y)
102{
103 static_assert(valid(C), "snicholls::ml::mlp_forward: the transformer_config is not valid");
104 assert(w.up.rows == C.d_ff && w.up.cols == C.d_model && w.down.rows == C.d_model && w.down.cols == C.d_ff);
105 std::array<T, C.d_ff> h{};
106 const auto in = prepare_input<C.d_model>(w.up, x);
107 linear(w.up, in, h, w.up_bias);
108 if constexpr (C.mlp == mlp_kind::swiglu) {
109 assert(w.gate.rows == C.d_ff && w.gate.cols == C.d_model);
110 std::array<T, C.d_ff> g{};
111 linear(w.gate, in, g, w.gate_bias);
112 for (std::size_t i = 0; i < C.d_ff; ++i)
113 h[i] = silu(g[i]) * h[i];
114 } else {
115 for (T& v : h)
116 v = mlp_activation<C.mlp>(v);
117 }
118 linear(w.down, prepare_input<C.d_ff>(w.down, h), y, w.down_bias);
119}

125template <math::math_number T, std::size_t K>
126struct moe_route
127{
128 std::array<std::size_t, K> expert{};
129 std::array<T, K> weight{};
130 std::size_t count = 0;
131};
132
137template <std::size_t K, row R>
138 requires(K > 0)
139constexpr moe_route<row_value_t<R>, K> moe_gate(const R& router_logits)
140{
141 using T = row_value_t<R>;
142 const auto best = top_k<K>(router_logits);
143 assert(best.count > 0);
144 moe_route<T, K> out{};
145 out.count = best.count;
146 T sum = T(0);
147 for (std::size_t i = 0; i < best.count; ++i) {
148 out.expert[i] = best.index[i];
149 out.weight[i] = math::exp(best.value[i] - best.value[0]);
150 sum += out.weight[i];
151 }
152 for (std::size_t i = 0; i < best.count; ++i)
153 out.weight[i] /= sum;
154 return out;
155}
