210 // The affinities of one row: z itself, softmax(z) or sigmoid(z), by the configuration's scoring
211 template <transformer_config C, typename T>
212 constexpr void affinities(std::span<const T> z, std::array<T, C.experts>& s)
213 {
214 switch (C.scoring) {
215 case moe_scoring::softmax_top_k:
216 for (std::size_t e = 0; e < C.experts; ++e)
217 s[e] = z[e];
218 return;
219 case moe_scoring::softmax: softmax(z, s); return;
220 case moe_scoring::sigmoid:
221 for (std::size_t e = 0; e < C.experts; ++e)
222 s[e] = sigmoid(z[e]);
223 return;
224 }
225 }

227 // The router probabilities of one row the balance loss reads: softmax(z), or sigmoid(z) normalised over every expert
228 template <transformer_config C, typename T>
229 constexpr void probabilities(std::span<const T> z, std::array<T, C.experts>& p)
230 {
231 if (C.scoring != moe_scoring::sigmoid) {
232 softmax(z, p);
233 return;
234 }
235 T sum = T(0);
236 for (std::size_t e = 0; e < C.experts; ++e) {
237 p[e] = sigmoid(z[e]);
238 sum += p[e];
239 }
240 for (std::size_t e = 0; e < C.experts; ++e)
241 p[e] = p[e] / sum;
242 }

244 // The gradient in the logits from the gradient g in probabilities(z), added into dz
245 template <transformer_config C, typename T>
246 constexpr void probabilities_backward(std::span<const T> z, const std::array<T, C.experts>& p, const std::array<T, C.experts>& g,
247 std::span<T> dz)
248 {
249 T dot = T(0);
250 for (std::size_t e = 0; e < C.experts; ++e)
251 dot += p[e] * g[e];
252 if (C.scoring != moe_scoring::sigmoid) {
253 for (std::size_t e = 0; e < C.experts; ++e)
254 dz[e] += p[e] * (g[e] - dot);
255 return;
256 }
257 T sum = T(0);
258 for (std::size_t e = 0; e < C.experts; ++e)
259 sum += sigmoid(z[e]);
260 for (std::size_t e = 0; e < C.experts; ++e)
261 dz[e] += (g[e] - dot) / sum * sigmoid_derivative(z[e]);
262 }

264 // The chosen experts' softmax_top_k weights before the scale: u_i / S with u_i = exp(z_i - m)
265 template <transformer_config C, typename T>
266 constexpr std::array<T, C.top_k> top_softmax(std::span<const T> z, std::span<const std::uint32_t> ids)
267 {
268 std::array<T, C.top_k> q{};
269 T m = z[ids[0]];
270 for (std::size_t i = 1; i < C.top_k; ++i)
271 m = z[ids[i]] > m ? z[ids[i]] : m;
272 T sum = T(0);
273 for (std::size_t i = 0; i < C.top_k; ++i) {
274 q[i] = math::exp(z[ids[i]] - m);
275 sum += q[i];
276 }
277 for (std::size_t i = 0; i < C.top_k; ++i)
278 q[i] /= sum;
279 return q;
280 }

282 // The K experts of the largest keys, best first, ties to the lower expert: over every expert, or, with
283 // node-limited routing, over the experts of the group_top_k groups whose two largest keys sum the highest
284 template <transformer_config C, typename T>
285 constexpr void choose(const std::array<T, C.experts>& key, std::span<std::uint32_t> ids)
286 {
287 constexpr std::size_t E = C.experts, K = C.top_k, G = C.expert_groups;
288 if constexpr (G == 0) {
289 const auto best = top_k<K>(key);
290 for (std::size_t i = 0; i < K; ++i)
291 ids[i] = static_cast<std::uint32_t>(best.index[i]);
292 } else {
293 constexpr std::size_t n = E / G, J = C.group_top_k, two = n < 2 ? n : 2;
294 std::array<T, G> score{};
295 for (std::size_t g = 0; g < G; ++g) {
296 const auto top = top_k<two>(std::span<const T>(key).subspan(g * n, n));
297 score[g] = top.value[0];
298 for (std::size_t i = 1; i < two; ++i)
299 score[g] = score[g] + top.value[i];
300 }
301 const auto groups = top_k<J>(score);
302 std::array<std::size_t, J> chosen = groups.index;
303 for (std::size_t i = 1; i < J; ++i) // ascending, so the candidates stay in expert order
304 for (std::size_t j = i; j > 0 && chosen[j - 1] > chosen[j]; --j) {
305 const std::size_t t = chosen[j];
306 chosen[j] = chosen[j - 1];
307 chosen[j - 1] = t;
308 }
309 std::array<T, J * n> candidate{};
310 for (std::size_t j = 0; j < J; ++j)
311 for (std::size_t i = 0; i < n; ++i)
312 candidate[j * n + i] = key[chosen[j] * n + i];
313 const auto best = top_k<K>(candidate);
314 for (std::size_t i = 0; i < K; ++i)
315 ids[i] = static_cast<std::uint32_t>(chosen[best.index[i] / n] * n + best.index[i] % n);
316 }
317 }

319 // One row of moe_route_rows: the affinities, the keys, the choice, the weights
320 template <transformer_config C, typename T>
321 constexpr void route_row(std::span<const T> z, std::span<const T> bias, std::span<std::uint32_t> ids, std::span<T> w)
322 {
323 constexpr std::size_t E = C.experts, K = C.top_k;
324 constexpr T scale = route_scale_v<C, T>;
325 std::array<T, E> s{}, key{};
326 affinities<C, T>(z, s);
327 for (std::size_t e = 0; e < E; ++e)
328 key[e] = bias.empty() ? s[e] : s[e] + bias[e];
329 choose<C, T>(key, ids);
330 switch (C.scoring) {
331 case moe_scoring::softmax_top_k: {
332 const auto q = top_softmax<C, T>(z, ids);
333 for (std::size_t i = 0; i < K; ++i)
334 w[i] = q[i] * scale;
335 return;
336 }
337 case moe_scoring::softmax:
338 for (std::size_t i = 0; i < K; ++i)
339 w[i] = s[ids[i]] * scale;
340 return;
341 case moe_scoring::sigmoid: {
342 T sum = T(0);
343 for (std::size_t i = 0; i < K; ++i)
344 sum += s[ids[i]];
345 for (std::size_t i = 0; i < K; ++i)
346 w[i] = s[ids[i]] / sum * scale;
347 return;
348 }
349 }
350 }

394template <transformer_config C, typename B, math::math_number T>
395 requires(C.experts > 0)
396constexpr void moe_route_rows_backward(const B&, std::type_identity_t<std::span<const T>> logits, std::span<const std::uint32_t> experts,
397 std::type_identity_t<std::span<const T>> dweights, std::span<T> dlogits)
398{
399 static_assert(valid(C), "snicholls::ml::moe_route_rows_backward: the transformer_config is not valid");
400 constexpr std::size_t E = C.experts, K = C.top_k;
401 const std::size_t rows = logits.size() / E;
402 assert(logits.size() == rows * E && experts.size() == rows * K && dweights.size() == rows * K && dlogits.size() == rows * E);
403 constexpr T scale = route_scale_v<C, T>;
404 for (std::size_t t = 0; t < rows; ++t) {
405 const auto z = logits.subspan(t * E, E);
406 const auto ids = experts.subspan(t * K, K);
407 const auto dw = dweights.subspan(t * K, K);
408 const auto g = dlogits.subspan(t * E, E);
409 switch (C.scoring) {
410 case moe_scoring::softmax_top_k: {
411 const auto q = detail::moe::top_softmax<C, T>(z, ids);
412 T dot = T(0);
413 for (std::size_t i = 0; i < K; ++i)
414 dot += q[i] * (scale * dw[i]);
415 for (T& v : g)
416 v = T(0);
417 for (std::size_t i = 0; i < K; ++i)
418 g[ids[i]] = q[i] * (scale * dw[i] - dot);
419 break;
420 }
421 case moe_scoring::softmax: {
422 std::array<T, E> p{}, dp{};
423 softmax(z, p);
424 for (std::size_t i = 0; i < K; ++i)
425 dp[ids[i]] = scale * dw[i];
426 softmax_backward(p, dp, g);
427 break;
428 }
429 case moe_scoring::sigmoid: {
430 std::array<T, K> s{};
431 T sum = T(0);
432 for (std::size_t i = 0; i < K; ++i) {
433 s[i] = sigmoid(z[ids[i]]);
434 sum += s[i];
435 }
436 T dot = T(0);
437 for (std::size_t i = 0; i < K; ++i)
438 dot += s[i] / sum * (scale * dw[i]);
439 for (T& v : g)
440 v = T(0);
441 for (std::size_t i = 0; i < K; ++i)
442 g[ids[i]] = (scale * dw[i] - dot) / sum * sigmoid_derivative(z[ids[i]]);
443 break;
444 }
445 }
446 }

827template <transformer_config C, typename B, math::math_number T>
828 requires(C.experts > 0)
829constexpr T moe_balance_loss(const B&, std::type_identity_t<std::span<const T>> logits, std::span<const std::uint32_t> experts,
830 std::span<T> dlogits, std::size_t groups = 1, std::size_t denominator = 0)
831{
832 constexpr std::size_t E = C.experts, K = C.top_k;
833 const std::size_t rows = logits.size() / E;
834 assert(groups > 0 && rows % groups == 0 && logits.size() == rows * E && experts.size() == rows * K
835 && (dlogits.empty() || dlogits.size() == rows * E));
836 const std::size_t n = rows / groups, den = denominator == 0 ? groups : denominator;
837 if (n == 0)
838 return T(0);
839 const T ae = balance_loss_weight_v<C, T> * T(E);
840 T loss = T(0);
841 std::array<T, E> p{}, sum_p{}, f{}, g{};
842 for (std::size_t s = 0; s < groups; ++s) {
843 std::array<std::uint32_t, E> counts{};
844 for (std::size_t j = s * n * K; j < (s + 1) * n * K; ++j)
845 ++counts[experts[j]];
846 sum_p.fill(T(0));
847 for (std::size_t t = s * n; t < (s + 1) * n; ++t) {
848 detail::moe::probabilities<C, T>(logits.subspan(t * E, E), p);
849 for (std::size_t e = 0; e < E; ++e)
850 sum_p[e] += p[e];
851 }
852 T dot = T(0);
853 for (std::size_t e = 0; e < E; ++e) {
854 f[e] = T(counts[e]) / T(n * K);
855 dot += f[e] * (sum_p[e] / T(n));
856 }
857 loss += ae * dot;
858 if (dlogits.empty())
859 continue;
860 for (std::size_t e = 0; e < E; ++e)
861 g[e] = ae * f[e] / T(n * den);
862 for (std::size_t t = s * n; t < (s + 1) * n; ++t) {
863 const auto z = logits.subspan(t * E, E);
864 detail::moe::probabilities<C, T>(z, p);
865 detail::moe::probabilities_backward<C, T>(z, p, g, dlogits.subspan(t * E, E));
866 }
867 }
868 return loss / T(den);
869}

878template <transformer_config C, typename B, math::math_number T>
879 requires(C.experts > 0)
880constexpr T moe_z_loss(const B&, std::type_identity_t<std::span<const T>> logits, std::span<T> dlogits, std::size_t denominator = 0)
881{
882 constexpr std::size_t E = C.experts;
883 const std::size_t rows = logits.size() / E;
884 assert(logits.size() == rows * E && (dlogits.empty() || dlogits.size() == rows * E));
885 const std::size_t den = denominator == 0 ? rows : denominator;
886 if (rows == 0)
887 return T(0);
888 constexpr T beta = z_loss_weight_v<C, T>;
889 T sum = T(0);
890 std::array<T, E> p{};
891 for (std::size_t t = 0; t < rows; ++t) {
892 const auto z = logits.subspan(t * E, E);
893 const T lse = logsumexp(z);
894 sum += lse * lse;
895 if (dlogits.empty())
896 continue;
897 softmax(z, p);
898 const T c = T(2) * beta * lse / T(den);
899 for (std::size_t e = 0; e < E; ++e)
900 dlogits[t * E + e] += c * p[e];
901 }
902 return beta * sum / T(den);
903}

910 requires(E > 0)
911struct moe_load
912{

917 constexpr void add(std::span<const std::uint32_t> step) noexcept
918 {
919 for (std::size_t e = 0; e < E && e < step.size(); ++e)
920 counts[e] += step[e];
921 }
922
925 constexpr void merge(const moe_load& other) noexcept
926 {
927 for (std::size_t e = 0; e < E; ++e)
928 counts[e] += other.counts[e];
929 }

933 {
934 std::uint64_t t = 0;
935 for (const auto c : counts)
936 t += c;
937 return t;
938 }
939
941 [[nodiscard]] constexpr std::uint64_t max() const noexcept
942 {
943 std::uint64_t m = 0;
944 for (const auto c : counts)
945 m = c > m ? c : m;
946 return m;
947 }
948
950 [[nodiscard]] constexpr std::uint64_t min() const noexcept
951 {
952 std::uint64_t m = counts[0];
953 for (const auto c : counts)
954 m = c < m ? c : m;
955 return m;
956 }
957
959 [[nodiscard]] constexpr double mean() const noexcept { return double(total()) / double(E); }
960

968 {
969 if (total() == 0)
970 return 0.0;
971 const double m = mean();
972 double s = 0;
973 for (const auto c : counts) {
974 const double d = double(c) - m;
975 s += d * d;
976 }
977 return math::sqrt(s / double(E)) / m;
978 }

1298template <transformer_config C, math::math_number T, row X, mutable_row Y>
1299 requires(C.experts > 0) && std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Y>, T>
1300constexpr void mlp_forward(const moe_weights<T>& w, const X& x, Y&& y)
1301{
1302 static_assert(valid(C), "snicholls::ml::mlp_forward: the transformer_config is not valid");
1303 constexpr std::size_t d = C.d_model, E = C.experts, K = C.top_k;
1304 const auto xs = as_row(x);
1305 const auto out = as_mutable_row(y);
1306 assert(xs.size() == d && out.size() == d && w.router.rows == E && w.router.cols == d);
1307 std::array<T, E> z{};
1308 linear(w.router, xs, z);
1309 std::array<std::uint32_t, K> ids{};
1310 std::array<T, K> weights{};
1311 detail::moe::route_row<C, T>(z, w.selection_bias, ids, weights);
1312 std::array<T, d> yp{};
1313 for (std::size_t s = 0; s < K; ++s) {
1314 mlp_forward<moe_expert_config(C)>(detail::moe::expert(w.experts, ids[s]), xs, yp);
1315 for (std::size_t i = 0; i < d; ++i) {
1316 const T v = weights[s] * yp[i];
1317 out[i] = s == 0 ? v : out[i] + v;
1318 }
