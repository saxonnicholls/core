126template <transformer_config C, math::math_number T, std::size_t N, row Q, mutable_row Out>
127 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<Out>, T>
128constexpr T attention_head(const Q& q, const kv_runs<T, N>& runs, Out&& out)
129{
130 const auto qs = as_row(q);
131 assert(qs.size() == C.head_dim);
132 const T scale = T(1) / math::sqrt(T(C.head_dim));
133 attention_accumulator<T, C.head_dim> acc{};
134 for (std::size_t r = 0; r < N; ++r)
135 for (std::size_t j = 0; j < runs.keys[r].size() / C.head_dim; ++j) {
136 T s = scale * detail::attention::dot(qs, runs.keys[r].subspan(j * C.head_dim, C.head_dim));
137 if constexpr (C.attention_softcap > 0)
138 s = softcap(s, attention_softcap_v<C, T>);
139 acc.add(s, runs.values[r].subspan(j * C.head_dim, C.head_dim));
140 }
141 acc.result(out);
142 return acc.log_sum_exp();
143}

147 // RoPE then, or before, the QK norm of one head, as the configuration says
148 template <transformer_config C, typename T, typename R>
149 constexpr void position(std::span<T> head, std::span<const T> norm, std::size_t pos, const R& rope_src)
150 {
151 constexpr T eps = norm_eps_v<C, T>;
152 if constexpr (C.qk_norm == qk_norm_kind::before_rope)
153 rms_norm(head, norm, head, eps);
154 if constexpr (C.rope != rope_kind::none)
155 ml::rope(head, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
156 if constexpr (C.qk_norm == qk_norm_kind::after_rope)
157 rms_norm(head, norm, head, eps);
158 }

171template <transformer_config C, weight_matrix W, math::math_number T, row X, kv_store<T> S, typename R, mutable_row Out>
172 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
173 && (C.rope == rope_kind::none || rope_source<R>)
174constexpr void attention_forward(const attention_weights<W, T>& w, const X& x, std::size_t pos, const S& cache,
175 std::size_t layer, const R& rope_src, Out&& out)
176{
177 static_assert(valid(C), "snicholls::ml::attention_forward: the transformer_config is not valid");
178 assert(w.q.rows == C.q_size() && w.k.rows == C.kv_size() && w.v.rows == C.kv_size() && w.o.rows == C.d_model
179 && w.q.cols == C.d_model && w.k.cols == C.d_model && w.v.cols == C.d_model && w.o.cols == C.q_size());
180 constexpr std::size_t d = C.head_dim;
181 std::array<T, C.q_size()> q{};
182 std::array<T, C.kv_size()> k{}, v{};
183 const auto in = prepare_input<C.d_model>(w.q, x);
184 linear(w.q, in, q, w.q_bias);
185 linear(w.k, in, k, w.k_bias);
186 linear(w.v, in, v, w.v_bias);
187 for (std::size_t h = 0; h < C.n_heads; ++h)
188 detail::attention_layer::position<C>(std::span<T>(q).subspan(h * d, d), w.q_norm, pos, rope_src);
189 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
190 detail::attention_layer::position<C>(std::span<T>(k).subspan(h * d, d), w.k_norm, pos, rope_src);
191 cache.append(layer, pos, k, v);
192 std::array<T, C.q_size()> y{};
193 const key_range visible = attention_window(pos, C.window);
194 for (std::size_t h = 0; h < C.n_heads; ++h)
195 attention_head<C>(std::span<const T>(q).subspan(h * d, d), cache.visible(layer, h / C.group(), visible),
196 std::span<T>(y).subspan(h * d, d));
197 linear(w.o, prepare_input<C.q_size()>(w.o, y), out, w.o_bias);
198}

215template <transformer_config C, math::math_number T, row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK,
216 mutable_row DV>
217 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<K>, T> && std::same_as<row_value_t<V>, T>
218 && std::same_as<row_value_t<O>, T> && std::same_as<row_value_t<DO>, T> && std::same_as<row_value_t<DQ>, T>
219 && std::same_as<row_value_t<DK>, T> && std::same_as<row_value_t<DV>, T>
220constexpr void attention_head_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
221 std::type_identity_t<T> lse, DQ&& dq, DK&& dkeys, DV&& dvalues)
222{
223 const T scale = T(1) / math::sqrt(T(C.head_dim));
224 if constexpr (C.attention_softcap == 0) {
225 attend_backward(q, keys, values, out, dout, lse, scale, dq, dkeys, dvalues);
226 } else {
227 constexpr std::size_t d = C.head_dim;
228 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values), o = as_row(out), g = as_row(dout);
229 const auto gq = as_mutable_row(dq), gk = as_mutable_row(dkeys), gv = as_mutable_row(dvalues);
230 assert(qs.size() == d && o.size() == d && g.size() == d && gq.size() == d && ks.size() == vs.size()
231 && ks.size() % d == 0 && gk.size() == ks.size() && gv.size() == vs.size());
232 std::fill(gq.begin(), gq.end(), T(0));
233 if (lse == -std::numeric_limits<T>::infinity())
234 return;
235 constexpr T cap = attention_softcap_v<C, T>;
236 const T big_d = detail::attention::dot(g, o);
237 for (std::size_t j = 0; j < ks.size() / d; ++j) {
238 const auto k = ks.subspan(j * d, d);
239 const auto v = vs.subspan(j * d, d);
240 const T r = scale * detail::attention::dot(qs, k);
241 const T p = math::exp(softcap(r, cap) - lse);
242 const T ds = p * (detail::attention::dot(g, v) - big_d) * softcap_derivative(r, cap);
243 for (std::size_t i = 0; i < d; ++i) {
244 gv[j * d + i] += p * g[i];
245 gq[i] += scale * ds * k[i];
246 gk[j * d + i] += scale * ds * qs[i];
247 }
248 }
249 }

297 // The backward of position(): the gradient in a head after it, to the gradient in its raw projection, in place
298 template <transformer_config C, typename T, typename R>
299 constexpr void position_backward(std::span<T> d, std::span<const T> raw, std::span<const T> norm,
300 std::span<T> dnorm, std::size_t pos, const R& rope_src)
301 {
302 if constexpr (C.qk_norm == qk_norm_kind::after_rope) {
303 constexpr T eps = norm_eps_v<C, T>;
304 std::array<T, C.head_dim> x{}, scratch{};
305 std::copy(raw.begin(), raw.end(), x.begin()); // the norm's input: the rotated projection
306 if constexpr (C.rope != rope_kind::none)
307 ml::rope(x, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
308 const T rstd = rms_norm(x, norm, scratch, eps);
309 rms_norm_backward(d, x, norm, rstd, d, dnorm);
310 }
311 if constexpr (C.rope != rope_kind::none)
312 rope_inverse(d, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
313 if constexpr (C.qk_norm == qk_norm_kind::before_rope) {
314 constexpr T eps = norm_eps_v<C, T>;
315 std::array<T, C.head_dim> x{}, scratch{};
316 std::copy(raw.begin(), raw.end(), x.begin());
317 const T rstd = rms_norm(x, norm, scratch, eps);
318 rms_norm_backward(d, x, norm, rstd, d, dnorm);
319 }
320 }

322 // The forward after the projections, for token t of n, shared by both paths: the raw copies, then QK norm and
323 // RoPE of its heads in place (the query in act.q, the keys in act.k's [kv_head][t] rows)
324 template <transformer_config C, typename W, typename T, typename R>
325 constexpr void position_token(const attention_weights<W, T>& w, const attention_activations<T>& act, std::size_t n,
326 std::size_t t, const R& rope_src)
327 {
328 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
329 const auto q = act.q.subspan(t * qn, qn);
330 if constexpr (C.qk_norm != qk_norm_kind::none) {
331 std::copy(q.begin(), q.end(), act.q_raw.begin() + std::ptrdiff_t(t * qn));
332 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
333 std::copy_n(act.k.begin() + std::ptrdiff_t((h * n + t) * d), d, act.k_raw.begin() + std::ptrdiff_t(t * kn + h * d));
334 }
335 for (std::size_t h = 0; h < C.n_heads; ++h)
336 position<C>(q.subspan(h * d, d), w.q_norm, t, rope_src);
337 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
338 position<C>(act.k.subspan((h * n + t) * d, d), w.k_norm, t, rope_src);
339 }

341 // Token t's heads over its visible keys: its rows of act.y and act.lse
342 template <transformer_config C, typename T>
343 constexpr void attend_token(const attention_activations<T>& act, std::size_t n, std::size_t t)
344 {
345 constexpr std::size_t d = C.head_dim, qn = C.q_size();
346 const key_range r = attention_window(t, C.window);
347 const auto yt = act.y.subspan(t * qn, qn);
348 for (std::size_t h = 0; h < C.n_heads; ++h) {
349 const std::size_t kvh = h / C.group();
350 kv_runs<T, 1> run{};
351 run.keys[0] = std::span<const T>(act.k).subspan((kvh * n + r.first) * d, r.size() * d);
352 run.values[0] = std::span<const T>(act.v).subspan((kvh * n + r.first) * d, r.size() * d);
353 act.lse[t * C.n_heads + h] = attention_head<C>(std::span<const T>(act.q).subspan(t * qn + h * d, d), run, yt.subspan(h * d, d));
354 }
355 }

357 // The backward of token t's heads from the gradient in their outputs: its query gradients written, its keys'
358 // and values' added into
359 template <transformer_config C, typename T>
360 constexpr void heads_backward_token(const attention_activations<T>& act, std::span<const T> dy,
361 const attention_gradients<T>& grads, std::size_t n, std::size_t t)
362 {
363 constexpr std::size_t d = C.head_dim, qn = C.q_size();
364 const auto yt = std::span<const T>(act.y).subspan(t * qn, qn);
365 const key_range r = attention_window(t, C.window);
366 for (std::size_t h = 0; h < C.n_heads; ++h) {
367 const std::size_t kvh = h / C.group(), at = (kvh * n + r.first) * d;
368 attention_head_backward<C, T>(std::span<const T>(act.q).subspan(t * qn + h * d, d),
369 std::span<const T>(act.k).subspan(at, r.size() * d),
370 std::span<const T>(act.v).subspan(at, r.size() * d), yt.subspan(h * d, d),
371 dy.subspan(h * d, d), act.lse[t * C.n_heads + h], grads.q.subspan(t * qn + h * d, d),
372 grads.k.subspan(at, r.size() * d), grads.v.subspan(at, r.size() * d));
373 }
374 }

376 // QK norm and RoPE undone for token t, in place: its row of grads.q and its [kv_head][t] rows of grads.k
377 template <transformer_config C, typename W, typename T, typename R>
378 constexpr void position_backward_token(const attention_weights<W, T>& w, const attention_activations<T>& act,
379 const attention_gradients<T>& grads, std::size_t n, std::size_t t, const R& rope_src)
380 {
381 if constexpr (C.qk_norm != qk_norm_kind::none || C.rope != rope_kind::none) {
382 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
383 constexpr bool raw = C.qk_norm != qk_norm_kind::none;
384 for (std::size_t h = 0; h < C.n_heads; ++h)
385 position_backward<C>(grads.q.subspan(t * qn + h * d, d),
386 raw ? std::span<const T>(act.q_raw).subspan(t * qn + h * d, d) : std::span<const T>{},
387 w.q_norm, grads.q_norm, t, rope_src);
388 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
389 position_backward<C>(grads.k.subspan((h * n + t) * d, d),
390 raw ? std::span<const T>(act.k_raw).subspan(t * kn + h * d, d) : std::span<const T>{},
391 w.k_norm, grads.k_norm, t, rope_src);
392 }
393 }

395 // Sequence s of a batch of sequences of n tokens: every per-sequence view is a slice, the batch's being the
396 // sequences' own laid end to end
397 template <typename T>
398 constexpr std::span<T> part(std::span<T> x, std::size_t s, std::size_t size)
399 {
400 return x.empty() ? x : x.subspan(s * size, size);
401 }
402 template <transformer_config C, typename T>
403 constexpr attention_activations<T> sequence(const attention_activations<T>& a, std::size_t n, std::size_t s)
404 {
405 constexpr std::size_t qn = C.q_size(), kn = C.kv_size();
406 return {part(a.q, s, n * qn), part(a.k, s, n * kn), part(a.v, s, n * kn), part(a.q_raw, s, n * qn),
407 part(a.k_raw, s, n * kn), part(a.y, s, n * qn), part(a.lse, s, n * C.n_heads)};
408 }
409 template <transformer_config C, typename T>
410 constexpr attention_gradients<T> sequence(const attention_gradients<T>& g, std::size_t n, std::size_t s)
411 {
412 constexpr std::size_t qn = C.q_size(), kn = C.kv_size();
413 attention_gradients<T> r = g;
414 r.q = part(g.q, s, n * qn);
415 r.k = part(g.k, s, n * kn);
416 r.v = part(g.v, s, n * kn);
417 r.y = part(g.y, s, n * qn);
418 return r;
419 }

508template <transformer_config C, weight_matrix W, math::math_number T, row X, typename R, mutable_row Out>
509 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
510 && (C.rope == rope_kind::none || rope_source<R>)
511constexpr void attention_forward_sequence(const attention_weights<W, T>& w, const X& xs, const R& rope_src,
512 const attention_activations<T>& act, Out&& out)
513{
514 static_assert(valid(C), "snicholls::ml::attention_forward_sequence: the transformer_config is not valid");
515 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
516 const auto x = as_row(xs);
517 const auto y = as_mutable_row(out);
518 const std::size_t n = x.size() / C.d_model;
519 assert(x.size() == n * C.d_model && y.size() == n * C.d_model);
520 detail::attention_layer::check_activations(act, n, qn, kn, C.n_heads, C.qk_norm != qk_norm_kind::none);
521 std::array<T, kn> k{}, v{};
522 for (std::size_t t = 0; t < n; ++t) {
523 const auto in = prepare_input<C.d_model>(w.q, x.subspan(t * C.d_model, C.d_model));
524 linear(w.q, in, act.q.subspan(t * qn, qn), w.q_bias);
525 linear(w.k, in, k, w.k_bias);
526 linear(w.v, in, v, w.v_bias);
527 for (std::size_t h = 0; h < C.n_kv_heads; ++h) {
528 std::copy_n(k.begin() + std::ptrdiff_t(h * d), d, act.k.begin() + std::ptrdiff_t((h * n + t) * d));
529 std::copy_n(v.begin() + std::ptrdiff_t(h * d), d, act.v.begin() + std::ptrdiff_t((h * n + t) * d));
530 }
531 detail::attention_layer::position_token<C>(w, act, n, t, rope_src);
532 detail::attention_layer::attend_token<C>(act, n, t);
533 linear(w.o, prepare_input<qn>(w.o, std::span<const T>(act.y).subspan(t * qn, qn)), y.subspan(t * C.d_model, C.d_model), w.o_bias);
534 }
535}

557template <transformer_config C, typename B, math::math_number T, typename R>
558 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
559constexpr void attention_forward_sequence(const B& backend, const attention_weights<dense_weights<T>, T>& w,
560 std::type_identity_t<std::span<const T>> xs, const R& rope_src,
561 const attention_activations<T>& act, std::type_identity_t<std::span<T>> out,
562 std::size_t sequences = 1)
563{
564 static_assert(valid(C), "snicholls::ml::attention_forward_sequence: the transformer_config is not valid");
565 constexpr std::size_t qn = C.q_size(), kn = C.kv_size();
566 const std::size_t rows = xs.size() / C.d_model, n = rows / sequences;
567 assert(sequences > 0 && xs.size() == n * sequences * C.d_model && out.size() == xs.size());
568 detail::attention_layer::check_activations(act, rows, qn, kn, C.n_heads, C.qk_norm != qk_norm_kind::none);
569 linear_sequence(backend, w.q, xs, act.q, w.q_bias);
570 linear_heads(backend, w.k, xs, act.k, w.k_bias, C.n_kv_heads, sequences); // the keys and values head by head
571 linear_heads(backend, w.v, xs, act.v, w.v_bias, C.n_kv_heads, sequences);

592template <transformer_config C, typename S, math::math_number T, row X, typename R, row DO, mutable_row DX>
593 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<DO>, T> && std::same_as<row_value_t<DX>, T>
594 && (C.rope == rope_kind::none || rope_source<R>)
595constexpr void attention_backward(const attention_weights<dense_weights<S>, T>& w, const X& xs, const R& rope_src,
596 const attention_activations<T>& act, const DO& dout, DX&& dxs,
597 const attention_gradients<T>& grads)
598{
599 static_assert(valid(C), "snicholls::ml::attention_backward: the transformer_config is not valid");
600 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
601 const auto x = as_row(xs);
602 const auto g = as_row(dout);
603 const auto gx = as_mutable_row(dxs);
604 const std::size_t n = x.size() / C.d_model;
605 assert(g.size() == x.size() && (gx.empty() || gx.size() == x.size()));
606 assert(grads.q.size() == n * qn && grads.k.size() == n * kn && grads.v.size() == n * kn);
607 std::fill(grads.k.begin(), grads.k.end(), T(0));
608 std::fill(grads.v.begin(), grads.v.end(), T(0));
609 for (std::size_t t = 0; t < n; ++t) {
610 std::array<T, qn> dy{};
611 linear_backward(w.o, std::span<const T>(act.y).subspan(t * qn, qn), g.subspan(t * C.d_model, C.d_model), dy, grads.wo, grads.o_bias);
612 detail::attention_layer::heads_backward_token<C>(act, std::span<const T>(dy), grads, n, t);
613 }
614 const std::span<T> no_dx{};
615 for (std::size_t t = 0; t < n; ++t) {
616 detail::attention_layer::position_backward_token<C>(w, act, grads, n, t, rope_src);
617 std::array<T, kn> dk{}, dv{};
618 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
619 for (std::size_t i = 0; i < d; ++i) {
620 dk[h * d + i] = grads.k[(h * n + t) * d + i];
621 dv[h * d + i] = grads.v[(h * n + t) * d + i];
622 }
623 const auto xt = x.subspan(t * C.d_model, C.d_model);
624 const auto dxt = gx.empty() ? no_dx : gx.subspan(t * C.d_model, C.d_model);
625 linear_backward(w.q, xt, grads.q.subspan(t * qn, qn), dxt, grads.wq, grads.q_bias);
626 linear_backward(w.k, xt, dk, dxt, grads.wk, grads.k_bias);
627 linear_backward(w.v, xt, dv, dxt, grads.wv, grads.v_bias);
628 }
629}

652template <transformer_config C, typename B, math::math_number T, typename R>
653 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
654constexpr void attention_backward(const B& backend, const attention_weights<dense_weights<T>, T>& w,
655 std::type_identity_t<std::span<const T>> xs, const R& rope_src, const attention_activations<T>& act,
656 std::type_identity_t<std::span<const T>> dout, std::type_identity_t<std::span<T>> dxs,
657 const attention_gradients<T>& grads, std::size_t sequences = 1)
658{
659 static_assert(valid(C), "snicholls::ml::attention_backward: the transformer_config is not valid");
660 [[maybe_unused]] constexpr std::size_t qn = C.q_size(); // each its own declaration: MSVC applies the attribute
661 [[maybe_unused]] constexpr std::size_t kn = C.kv_size(); // to the first declarator only (C4189)
662 const std::size_t rows = xs.size() / C.d_model, n = rows / sequences;
663 assert(sequences > 0 && rows == n * sequences && dout.size() == xs.size() && (dxs.empty() || dxs.size() == xs.size()));
664 assert(grads.q.size() == rows * qn && grads.k.size() == rows * kn && grads.v.size() == rows * kn && grads.y.size() == rows * qn);
665 fill_rows(backend, grads.k, T(0));
666 fill_rows(backend, grads.v, T(0));
667 fill_rows(backend, grads.y, T(0));
668 linear_sequence_backward(backend, w.o, std::span<const T>(act.y), dout, grads.y, grads.wo, grads.o_bias);
669 attention_attend_rows_backward<C>(backend, act, std::span<const T>(grads.y), grads, n, sequences);
670 attention_position_rows_backward<C>(backend, w, act, grads, n, rope_src, sequences);
671 linear_sequence_backward(backend, w.q, xs, std::span<const T>(grads.q), dxs, grads.wq, grads.q_bias);
672 linear_heads_backward(backend, w.k, xs, std::span<const T>(grads.k), dxs, grads.wk, grads.k_bias, C.n_kv_heads, sequences);
673 linear_heads_backward(backend, w.v, xs, std::span<const T>(grads.v), dxs, grads.wv, grads.v_bias, C.n_kv_heads, sequences);
674}
