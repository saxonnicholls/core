114template <transformer_config C, math::math_number T, std::size_t N, row Q, mutable_row Out>
115 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<Out>, T>
116constexpr T attention_head(const Q& q, const kv_runs<T, N>& runs, Out&& out)
117{
118 const auto qs = as_row(q);
119 assert(qs.size() == C.head_dim);
120 const T scale = T(1) / math::sqrt(T(C.head_dim));
121 attention_accumulator<T, C.head_dim> acc{};
122 for (std::size_t r = 0; r < N; ++r)
123 for (std::size_t j = 0; j < runs.keys[r].size() / C.head_dim; ++j) {
124 T s = scale * detail::attention::dot(qs, runs.keys[r].subspan(j * C.head_dim, C.head_dim));
125 if constexpr (C.attention_softcap > 0)
126 s = softcap(s, attention_softcap_v<C, T>);
127 acc.add(s, runs.values[r].subspan(j * C.head_dim, C.head_dim));
128 }
129 acc.result(out);
130 return acc.log_sum_exp();
131}

135 // RoPE then, or before, the QK norm of one head, as the configuration says
136 template <transformer_config C, typename T, typename R>
137 constexpr void position(std::span<T> head, std::span<const T> norm, std::size_t pos, const R& rope_src)
138 {
139 constexpr T eps = norm_eps_v<C, T>;
140 if constexpr (C.qk_norm == qk_norm_kind::before_rope)
141 rms_norm(head, norm, head, eps);
142 if constexpr (C.rope != rope_kind::none)
143 ml::rope(head, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
144 if constexpr (C.qk_norm == qk_norm_kind::after_rope)
145 rms_norm(head, norm, head, eps);
146 }

159template <transformer_config C, weight_matrix W, math::math_number T, row X, kv_store<T> S, typename R, mutable_row Out>
160 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
161 && (C.rope == rope_kind::none || rope_source<R>)
162constexpr void attention_forward(const attention_weights<W, T>& w, const X& x, std::size_t pos, const S& cache,
163 std::size_t layer, const R& rope_src, Out&& out)
164{
165 static_assert(valid(C), "snicholls::ml::attention_forward: the transformer_config is not valid");
166 assert(w.q.rows == C.q_size() && w.k.rows == C.kv_size() && w.v.rows == C.kv_size() && w.o.rows == C.d_model
167 && w.q.cols == C.d_model && w.k.cols == C.d_model && w.v.cols == C.d_model && w.o.cols == C.q_size());
168 constexpr std::size_t d = C.head_dim;
169 std::array<T, C.q_size()> q{};
170 std::array<T, C.kv_size()> k{}, v{};
171 const auto in = prepare_input<C.d_model>(w.q, x);
172 linear(w.q, in, q, w.q_bias);
173 linear(w.k, in, k, w.k_bias);
174 linear(w.v, in, v, w.v_bias);
175 for (std::size_t h = 0; h < C.n_heads; ++h)
176 detail::attention_layer::position<C>(std::span<T>(q).subspan(h * d, d), w.q_norm, pos, rope_src);
177 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
178 detail::attention_layer::position<C>(std::span<T>(k).subspan(h * d, d), w.k_norm, pos, rope_src);
179 cache.append(layer, pos, k, v);
180 std::array<T, C.q_size()> y{};
181 const key_range visible = attention_window(pos, C.window);
182 for (std::size_t h = 0; h < C.n_heads; ++h)
183 attention_head<C>(std::span<const T>(q).subspan(h * d, d), cache.visible(layer, h / C.group(), visible),
184 std::span<T>(y).subspan(h * d, d));
185 linear(w.o, prepare_input<C.q_size()>(w.o, y), out, w.o_bias);
186}
