122template <transformer_config C, math::math_number T, std::size_t N, row Q, mutable_row Out>
123 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<Out>, T>
124constexpr T attention_head(const Q& q, const kv_runs<T, N>& runs, Out&& out)
125{
126 const auto qs = as_row(q);
127 assert(qs.size() == C.head_dim);
128 const T scale = T(1) / math::sqrt(T(C.head_dim));
129 attention_accumulator<T, C.head_dim> acc{};
130 for (std::size_t r = 0; r < N; ++r)
131 for (std::size_t j = 0; j < runs.keys[r].size() / C.head_dim; ++j) {
132 T s = scale * detail::attention::dot(qs, runs.keys[r].subspan(j * C.head_dim, C.head_dim));
133 if constexpr (C.attention_softcap > 0)
134 s = softcap(s, attention_softcap_v<C, T>);
135 acc.add(s, runs.values[r].subspan(j * C.head_dim, C.head_dim));
136 }
137 acc.result(out);
138 return acc.log_sum_exp();
139}

143 // RoPE then, or before, the QK norm of one head, as the configuration says
144 template <transformer_config C, typename T, typename R>
145 constexpr void position(std::span<T> head, std::span<const T> norm, std::size_t pos, const R& rope_src)
146 {
147 constexpr T eps = norm_eps_v<C, T>;
148 if constexpr (C.qk_norm == qk_norm_kind::before_rope)
149 rms_norm(head, norm, head, eps);
150 if constexpr (C.rope != rope_kind::none)
151 ml::rope(head, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
152 if constexpr (C.qk_norm == qk_norm_kind::after_rope)
153 rms_norm(head, norm, head, eps);
154 }

167template <transformer_config C, weight_matrix W, math::math_number T, row X, kv_store<T> S, typename R, mutable_row Out>
168 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
169 && (C.rope == rope_kind::none || rope_source<R>)
170constexpr void attention_forward(const attention_weights<W, T>& w, const X& x, std::size_t pos, const S& cache,
171 std::size_t layer, const R& rope_src, Out&& out)
172{
173 static_assert(valid(C), "snicholls::ml::attention_forward: the transformer_config is not valid");
174 assert(w.q.rows == C.q_size() && w.k.rows == C.kv_size() && w.v.rows == C.kv_size() && w.o.rows == C.d_model
175 && w.q.cols == C.d_model && w.k.cols == C.d_model && w.v.cols == C.d_model && w.o.cols == C.q_size());
176 constexpr std::size_t d = C.head_dim;
177 std::array<T, C.q_size()> q{};
178 std::array<T, C.kv_size()> k{}, v{};
179 const auto in = prepare_input<C.d_model>(w.q, x);
180 linear(w.q, in, q, w.q_bias);
181 linear(w.k, in, k, w.k_bias);
182 linear(w.v, in, v, w.v_bias);
183 for (std::size_t h = 0; h < C.n_heads; ++h)
184 detail::attention_layer::position<C>(std::span<T>(q).subspan(h * d, d), w.q_norm, pos, rope_src);
185 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
186 detail::attention_layer::position<C>(std::span<T>(k).subspan(h * d, d), w.k_norm, pos, rope_src);
187 cache.append(layer, pos, k, v);
188 std::array<T, C.q_size()> y{};
189 const key_range visible = attention_window(pos, C.window);
190 for (std::size_t h = 0; h < C.n_heads; ++h)
191 attention_head<C>(std::span<const T>(q).subspan(h * d, d), cache.visible(layer, h / C.group(), visible),
192 std::span<T>(y).subspan(h * d, d));
193 linear(w.o, prepare_input<C.q_size()>(w.o, y), out, w.o_bias);
194}

211template <transformer_config C, math::math_number T, row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK,
212 mutable_row DV>
213 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<K>, T> && std::same_as<row_value_t<V>, T>
214 && std::same_as<row_value_t<O>, T> && std::same_as<row_value_t<DO>, T> && std::same_as<row_value_t<DQ>, T>
215 && std::same_as<row_value_t<DK>, T> && std::same_as<row_value_t<DV>, T>
216constexpr void attention_head_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
217 std::type_identity_t<T> lse, DQ&& dq, DK&& dkeys, DV&& dvalues)
218{
219 const T scale = T(1) / math::sqrt(T(C.head_dim));
220 if constexpr (C.attention_softcap == 0) {
221 attend_backward(q, keys, values, out, dout, lse, scale, dq, dkeys, dvalues);
222 } else {
223 constexpr std::size_t d = C.head_dim;
224 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values), o = as_row(out), g = as_row(dout);
225 const auto gq = as_mutable_row(dq), gk = as_mutable_row(dkeys), gv = as_mutable_row(dvalues);
226 assert(qs.size() == d && o.size() == d && g.size() == d && gq.size() == d && ks.size() == vs.size()
227 && ks.size() % d == 0 && gk.size() == ks.size() && gv.size() == vs.size());
228 std::fill(gq.begin(), gq.end(), T(0));
229 if (lse == -std::numeric_limits<T>::infinity())
230 return;
231 constexpr T cap = attention_softcap_v<C, T>;
232 const T big_d = detail::attention::dot(g, o);
233 for (std::size_t j = 0; j < ks.size() / d; ++j) {
234 const auto k = ks.subspan(j * d, d);
235 const auto v = vs.subspan(j * d, d);
236 const T r = scale * detail::attention::dot(qs, k);
237 const T p = math::exp(softcap(r, cap) - lse);
238 const T ds = p * (detail::attention::dot(g, v) - big_d) * softcap_derivative(r, cap);
239 for (std::size_t i = 0; i < d; ++i) {
240 gv[j * d + i] += p * g[i];
241 gq[i] += scale * ds * k[i];
242 gk[j * d + i] += scale * ds * qs[i];
243 }
244 }
245 }

287 // The backward of position(): the gradient in a head after it, to the gradient in its raw projection, in place
288 template <transformer_config C, typename T, typename R>
289 constexpr void position_backward(std::span<T> d, std::span<const T> raw, std::span<const T> norm,
290 std::span<T> dnorm, std::size_t pos, const R& rope_src)
291 {
292 if constexpr (C.qk_norm == qk_norm_kind::after_rope) {
293 constexpr T eps = norm_eps_v<C, T>;
294 std::array<T, C.head_dim> x{}, scratch{};
295 std::copy(raw.begin(), raw.end(), x.begin()); // the norm's input: the rotated projection
296 if constexpr (C.rope != rope_kind::none)
297 ml::rope(x, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
298 const T rstd = rms_norm(x, norm, scratch, eps);
299 rms_norm_backward(d, x, norm, rstd, d, dnorm);
300 }
301 if constexpr (C.rope != rope_kind::none)
302 rope_inverse(d, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
303 if constexpr (C.qk_norm == qk_norm_kind::before_rope) {
304 constexpr T eps = norm_eps_v<C, T>;
305 std::array<T, C.head_dim> x{}, scratch{};
306 std::copy(raw.begin(), raw.end(), x.begin());
307 const T rstd = rms_norm(x, norm, scratch, eps);
308 rms_norm_backward(d, x, norm, rstd, d, dnorm);
309 }
310 }

312 // The forward after the projections, for token t of n, shared by both paths: the raw copies, then QK norm and
313 // RoPE of its heads in place (the query in act.q, the keys in act.k's [kv_head][t] rows)
314 template <transformer_config C, typename W, typename T, typename R>
315 constexpr void position_token(const attention_weights<W, T>& w, const attention_activations<T>& act, std::size_t n,
316 std::size_t t, const R& rope_src)
317 {
318 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
319 const auto q = act.q.subspan(t * qn, qn);
320 if constexpr (C.qk_norm != qk_norm_kind::none) {
321 std::copy(q.begin(), q.end(), act.q_raw.begin() + std::ptrdiff_t(t * qn));
322 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
323 std::copy_n(act.k.begin() + std::ptrdiff_t((h * n + t) * d), d, act.k_raw.begin() + std::ptrdiff_t(t * kn + h * d));
324 }
325 for (std::size_t h = 0; h < C.n_heads; ++h)
326 position<C>(q.subspan(h * d, d), w.q_norm, t, rope_src);
327 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
328 position<C>(act.k.subspan((h * n + t) * d, d), w.k_norm, t, rope_src);
329 }

331 // Token t's heads over its visible keys: its rows of act.y and act.lse
332 template <transformer_config C, typename T>
333 constexpr void attend_token(const attention_activations<T>& act, std::size_t n, std::size_t t)
334 {
335 constexpr std::size_t d = C.head_dim, qn = C.q_size();
336 const key_range r = attention_window(t, C.window);
337 const auto yt = act.y.subspan(t * qn, qn);
338 for (std::size_t h = 0; h < C.n_heads; ++h) {
339 const std::size_t kvh = h / C.group();
340 kv_runs<T, 1> run{};
341 run.keys[0] = std::span<const T>(act.k).subspan((kvh * n + r.first) * d, r.size() * d);
342 run.values[0] = std::span<const T>(act.v).subspan((kvh * n + r.first) * d, r.size() * d);
343 act.lse[t * C.n_heads + h] = attention_head<C>(std::span<const T>(act.q).subspan(t * qn + h * d, d), run, yt.subspan(h * d, d));
344 }
345 }

347 // The backward of token t's heads from the gradient in their outputs: its query gradients written, its keys'
348 // and values' added into
349 template <transformer_config C, typename T>
350 constexpr void heads_backward_token(const attention_activations<T>& act, std::span<const T> dy,
351 const attention_gradients<T>& grads, std::size_t n, std::size_t t)
352 {
353 constexpr std::size_t d = C.head_dim, qn = C.q_size();
354 const auto yt = std::span<const T>(act.y).subspan(t * qn, qn);
355 const key_range r = attention_window(t, C.window);
356 for (std::size_t h = 0; h < C.n_heads; ++h) {
357 const std::size_t kvh = h / C.group(), at = (kvh * n + r.first) * d;
358 attention_head_backward<C, T>(std::span<const T>(act.q).subspan(t * qn + h * d, d),
359 std::span<const T>(act.k).subspan(at, r.size() * d),
360 std::span<const T>(act.v).subspan(at, r.size() * d), yt.subspan(h * d, d),
361 dy.subspan(h * d, d), act.lse[t * C.n_heads + h], grads.q.subspan(t * qn + h * d, d),
362 grads.k.subspan(at, r.size() * d), grads.v.subspan(at, r.size() * d));
363 }
364 }

366 // QK norm and RoPE undone for token t, in place: its row of grads.q and its [kv_head][t] rows of grads.k
367 template <transformer_config C, typename W, typename T, typename R>
368 constexpr void position_backward_token(const attention_weights<W, T>& w, const attention_activations<T>& act,
369 const attention_gradients<T>& grads, std::size_t n, std::size_t t, const R& rope_src)
370 {
371 if constexpr (C.qk_norm != qk_norm_kind::none || C.rope != rope_kind::none) {
372 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
373 constexpr bool raw = C.qk_norm != qk_norm_kind::none;
374 for (std::size_t h = 0; h < C.n_heads; ++h)
375 position_backward<C>(grads.q.subspan(t * qn + h * d, d),
376 raw ? std::span<const T>(act.q_raw).subspan(t * qn + h * d, d) : std::span<const T>{},
377 w.q_norm, grads.q_norm, t, rope_src);
378 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
379 position_backward<C>(grads.k.subspan((h * n + t) * d, d),
380 raw ? std::span<const T>(act.k_raw).subspan(t * kn + h * d, d) : std::span<const T>{},
381 w.k_norm, grads.k_norm, t, rope_src);
382 }
383 }

393 }

455template <transformer_config C, weight_matrix W, math::math_number T, row X, typename R, mutable_row Out>
456 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
457 && (C.rope == rope_kind::none || rope_source<R>)
458constexpr void attention_forward_sequence(const attention_weights<W, T>& w, const X& xs, const R& rope_src,
459 const attention_activations<T>& act, Out&& out)
460{
461 static_assert(valid(C), "snicholls::ml::attention_forward_sequence: the transformer_config is not valid");
462 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
463 const auto x = as_row(xs);
464 const auto y = as_mutable_row(out);
465 const std::size_t n = x.size() / C.d_model;
466 assert(x.size() == n * C.d_model && y.size() == n * C.d_model);
467 detail::attention_layer::check_activations(act, n, qn, kn, C.n_heads, C.qk_norm != qk_norm_kind::none);
468 std::array<T, kn> k{}, v{};
469 for (std::size_t t = 0; t < n; ++t) {
470 const auto in = prepare_input<C.d_model>(w.q, x.subspan(t * C.d_model, C.d_model));
471 linear(w.q, in, act.q.subspan(t * qn, qn), w.q_bias);
472 linear(w.k, in, k, w.k_bias);
473 linear(w.v, in, v, w.v_bias);
474 for (std::size_t h = 0; h < C.n_kv_heads; ++h) {
475 std::copy_n(k.begin() + std::ptrdiff_t(h * d), d, act.k.begin() + std::ptrdiff_t((h * n + t) * d));
476 std::copy_n(v.begin() + std::ptrdiff_t(h * d), d, act.v.begin() + std::ptrdiff_t((h * n + t) * d));
477 }
478 detail::attention_layer::position_token<C>(w, act, n, t, rope_src);
479 detail::attention_layer::attend_token<C>(act, n, t);
480 linear(w.o, prepare_input<qn>(w.o, std::span<const T>(act.y).subspan(t * qn, qn)), y.subspan(t * C.d_model, C.d_model), w.o_bias);
481 }

498template <transformer_config C, typename B, math::math_number T, typename R>
499 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
500constexpr void attention_forward_sequence(const B& backend, const attention_weights<dense_weights<T>, T>& w,
501 std::type_identity_t<std::span<const T>> xs, const R& rope_src,
502 const attention_activations<T>& act, std::type_identity_t<std::span<T>> out)
503{

505 constexpr std::size_t qn = C.q_size(), kn = C.kv_size();

513 attention_attend_rows<C>(backend, act, n);
514 linear_sequence(backend, w.o, std::span<const T>(act.y), out, w.o_bias);
515}

532template <transformer_config C, typename S, math::math_number T, row X, typename R, row DO, mutable_row DX>
533 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<DO>, T> && std::same_as<row_value_t<DX>, T>
534 && (C.rope == rope_kind::none || rope_source<R>)
535constexpr void attention_backward(const attention_weights<dense_weights<S>, T>& w, const X& xs, const R& rope_src,
536 const attention_activations<T>& act, const DO& dout, DX&& dxs,
537 const attention_gradients<T>& grads)
538{
539 static_assert(valid(C), "snicholls::ml::attention_backward: the transformer_config is not valid");
540 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
541 const auto x = as_row(xs);
542 const auto g = as_row(dout);
543 const auto gx = as_mutable_row(dxs);
544 const std::size_t n = x.size() / C.d_model;
545 assert(g.size() == x.size() && (gx.empty() || gx.size() == x.size()));
546 assert(grads.q.size() == n * qn && grads.k.size() == n * kn && grads.v.size() == n * kn);
547 std::fill(grads.k.begin(), grads.k.end(), T(0));
548 std::fill(grads.v.begin(), grads.v.end(), T(0));
549 for (std::size_t t = 0; t < n; ++t) {
550 std::array<T, qn> dy{};
551 linear_backward(w.o, std::span<const T>(act.y).subspan(t * qn, qn), g.subspan(t * C.d_model, C.d_model), dy, grads.wo, grads.o_bias);
552 detail::attention_layer::heads_backward_token<C>(act, std::span<const T>(dy), grads, n, t);
553 }
554 const std::span<T> no_dx{};
555 for (std::size_t t = 0; t < n; ++t) {
556 detail::attention_layer::position_backward_token<C>(w, act, grads, n, t, rope_src);
557 std::array<T, kn> dk{}, dv{};
558 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
559 for (std::size_t i = 0; i < d; ++i) {
560 dk[h * d + i] = grads.k[(h * n + t) * d + i];
561 dv[h * d + i] = grads.v[(h * n + t) * d + i];
562 }
563 const auto xt = x.subspan(t * C.d_model, C.d_model);
564 const auto dxt = gx.empty() ? no_dx : gx.subspan(t * C.d_model, C.d_model);
565 linear_backward(w.q, xt, grads.q.subspan(t * qn, qn), dxt, grads.wq, grads.q_bias);
566 linear_backward(w.k, xt, dk, dxt, grads.wk, grads.k_bias);
567 linear_backward(w.v, xt, dv, dxt, grads.wv, grads.v_bias);
568 }
569}

586template <transformer_config C, typename B, math::math_number T, typename R>
587 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
588constexpr void attention_backward(const B& backend, const attention_weights<dense_weights<T>, T>& w,
589 std::type_identity_t<std::span<const T>> xs, const R& rope_src, const attention_activations<T>& act,
590 std::type_identity_t<std::span<const T>> dout, std::type_identity_t<std::span<T>> dxs,
591 const attention_gradients<T>& grads)
592{
593 static_assert(valid(C), "snicholls::ml::attention_backward: the transformer_config is not valid");
594 [[maybe_unused]] constexpr std::size_t qn = C.q_size(); // each its own declaration: MSVC applies the attribute
595 [[maybe_unused]] constexpr std::size_t kn = C.kv_size(); // to the first declarator only (C4189)
596 const std::size_t n = xs.size() / C.d_model;
597 assert(dout.size() == xs.size() && (dxs.empty() || dxs.size() == xs.size()));
598 assert(grads.q.size() == n * qn && grads.k.size() == n * kn && grads.v.size() == n * kn && grads.y.size() == n * qn);
599 fill_rows(backend, grads.k, T(0));
600 fill_rows(backend, grads.v, T(0));
601 fill_rows(backend, grads.y, T(0));
602 linear_sequence_backward(backend, w.o, std::span<const T>(act.y), dout, grads.y, grads.wo, grads.o_bias);
603 attention_attend_rows_backward<C>(backend, act, std::span<const T>(grads.y), grads, n);
604 attention_position_rows_backward<C>(backend, w, act, grads, n, rope_src);
605 linear_sequence_backward(backend, w.q, xs, std::span<const T>(grads.q), dxs, grads.wq, grads.q_bias);
606 linear_heads_backward(backend, w.k, xs, std::span<const T>(grads.k), dxs, grads.wk, grads.k_bias, C.n_kv_heads);
607 linear_heads_backward(backend, w.v, xs, std::span<const T>(grads.v), dxs, grads.wv, grads.v_bias, C.n_kv_heads);
608}
