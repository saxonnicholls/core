125 capability_result<key_info> create(const key_spec& spec)
126 {
127 if (auto e = take_injected())
128 return err(*e);
129 if (spec.algorithm != signature_algorithm::hs256)
130 return fail(capability_errc::unsupported, {}, "the reference holds HS256 keys only");
131 auto label = make_id<short_text>(spec.label);
132 if (!label)
133 return err(label.unwrap_err());
134 const auto id = *provider_id::make("key-" + std::to_string(++next_id_));
135 entry e{spec.algorithm, *label, {}};
136 e.versions.push_back(make_version(1));
137 const auto [it, inserted] = keys_.emplace(std::string(id.view()), std::move(e));
138 static_cast<void>(inserted);
139 return ok(describe(it->first, it->second, it->second.versions.back()));
140 }
141
145 capability_result<key_info> rotate(std::string_view id)
146 {
147 if (auto e = take_injected())
148 return err(*e);
149 const auto it = keys_.find(id);
150 if (it == keys_.end())
151 return fail(capability_errc::not_found);
152 auto& versions = it->second.versions;
153 if (versions.back().state != key_state::active)
154 return fail(capability_errc::conflict, {}, "a retired key does not rotate");
155 versions.push_back(make_version(static_cast<std::uint32_t>(versions.size() + 1)));
156 versions[versions.size() - 2].state = key_state::retired;
157 return ok(describe(it->first, it->second, versions.back()));
158 }
159
163 capability_result<void> retire(std::string_view id)
164 {
165 if (auto e = take_injected())
166 return err(*e);
167 const auto it = keys_.find(id);
168 if (it == keys_.end())
169 return fail(capability_errc::not_found);
170 for (auto& v : it->second.versions)
171 v.state = key_state::retired;
172 return ok();
173 }
174
178 capability_result<void> destroy(std::string_view id)
179 {
180 if (auto e = take_injected())
181 return err(*e);
182 const auto it = keys_.find(id);
183 if (it == keys_.end())
184 return fail(capability_errc::not_found);
185 keys_.erase(it);
186 return ok();
187 }

202 [[nodiscard]] capability_result<std::vector<key_info>> list() const
203 {
204 std::vector<key_info> out;
205 for (const auto& [id, e] : keys_)
206 for (const auto& v : e.versions)
207 out.push_back(describe(id, e, v));
208 return ok(std::move(out));
209 }

214 capability_result<signature> sign(const sign_request& r)
215 {
216 if (auto e = take_injected())
217 return err(*e);
218 if (r.kind == payload_kind::digest)
219 return fail(capability_errc::unsupported, {}, "HS256 hashes for itself: no digests");
220 const auto it = keys_.find(r.key_id);
221 if (it == keys_.end())
222 return fail(capability_errc::not_found);
223 const version& v = it->second.versions.back();
224 if (v.state != key_state::active)
225 return fail(capability_errc::conflict, {}, "the key is retired");
226 return ok(signature{signature_algorithm::hs256, *provider_id::make(it->first), v.number, mac(v, r.payload)});
227 }

232 [[nodiscard]] capability_result<public_key> public_key_of(const key_ref& k) const
233 {
234 const auto it = keys_.find(k.key_id);
235 if (it == keys_.end() || k.version > it->second.versions.size())
236 return fail(capability_errc::not_found);
237 return fail(capability_errc::unsupported, {}, "HS256 has no public key");
238 }
239
244 [[nodiscard]] bool verify(const signature& s, std::span<const std::byte> payload) const
245 {
246 const auto it = keys_.find(s.key_id.view());
247 if (it == keys_.end() || s.algorithm != signature_algorithm::hs256 || s.key_version == 0 ||
248 s.key_version > it->second.versions.size())
249 return false;
250 const auto expected = mac(it->second.versions[s.key_version - 1], payload);
251 return constant_time_equal(std::span<const std::byte>(expected), std::span<const std::byte>(s.bytes));
252 }

289 version make_version(std::uint32_t number)
290 {
291 std::string bytes(key_bytes, '\0');
292 random_(std::as_writable_bytes(std::span<char>(bytes.data(), bytes.size())));
293 return version{number, key_state::active, stamp(), secret<std::string>{std::move(bytes)}};
294 }

296 static std::vector<std::byte> mac(const version& v, std::span<const std::byte> payload)
297 {
298 const auto tag = hmac<sha256>::mac(v.material.reveal(), payload);
299 return {tag.bytes.begin(), tag.bytes.end()};
300 }

302 static key_info describe(const std::string& id, const entry& e, const version& v)
303 {
304 return key_info{*provider_id::make(id), v.number, e.algorithm, v.state, e.label, v.created_at};
305 }
