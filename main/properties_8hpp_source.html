108 template <typename T, typename Keep>
109 constexpr std::optional<interaction_violation<T>> find_second_difference(std::size_t n, std::span<const T> table, Keep keep)
110 {
111 const coalition all = grand_coalition(n);
112 for (coalition s = 0; s <= all; ++s) {
113 const coalition missing = all & ~s;
114 for (coalition a = missing; a != 0; a &= a - 1) {
115 const auto i = static_cast<std::size_t>(std::countr_zero(a));
116 for (coalition b = a & (a - 1); b != 0; b &= b - 1) {
117 const auto j = static_cast<std::size_t>(std::countr_zero(b));
118 const T d = table[s | singleton(i) | singleton(j)] - table[s | singleton(i)] - table[s | singleton(j)] + table[s];
119 if (!keep(d))
120 return interaction_violation<T>{s, i, j, d};
121 }
122 }
123 }
124 return std::nullopt;
125 }

195 template <std::size_t N, game V>
196 requires ordered_game_value<game_value_t<V>>
197 [[nodiscard]] constexpr std::optional<monotonicity_violation<game_value_t<V>>> find_monotonicity_violation(const V& v,
198 game_value_t<V> tolerance = {})
199 {
200 using T = game_value_t<V>;
201 std::optional<monotonicity_violation<T>> found;
202 detail::with_table(N, v, [&](std::span<const T> table) {
203 const coalition all = grand_coalition(N);
204 for (coalition s = 0; s < all && !found; ++s)
205 for (coalition missing = all & ~s; missing != 0 && !found; missing &= missing - 1) {
206 const auto i = static_cast<std::size_t>(std::countr_zero(missing));
207 const T lift = table[with(s, i)] - table[s];
208 if (lift < T{} - tolerance)
209 found = monotonicity_violation<T>{s, i, lift};
210 }
211 });
212 return found;

222 template <std::size_t N, game V>
223 requires ordered_game_value<game_value_t<V>>
224 [[nodiscard]] constexpr std::optional<superadditivity_violation<game_value_t<V>>> find_superadditivity_violation(const V& v,
225 game_value_t<V> tolerance = {})
226 {
227 static_assert(N <= 20, "snicholls::game::find_superadditivity_violation: 3^N pairs; at most 20 players");
228 using T = game_value_t<V>;
229 std::optional<superadditivity_violation<T>> found;
230 detail::with_table(N, v, [&](std::span<const T> table) {
231 const coalition all = grand_coalition(N);
232 for (coalition s = 1; s <= all && !found; ++s)
233 for_each_subset_of(all & ~s, [&](coalition t) {
234 if (found || t == 0 || t < s)
235 return;
236 const T shortfall = table[s | t] - table[s] - table[t];
237 if (shortfall < T{} - tolerance)
238 found = superadditivity_violation<T>{s, t, shortfall};
239 });
240 });
241 return found;
242 }

323 template <std::size_t N, game V>
324 requires ordered_game_value<game_value_t<V>>
325 [[nodiscard]] constexpr std::optional<core_violation<game_value_t<V>>> find_core_violation(const V& v,
326 const std::array<game_value_t<V>, N>& allocation,
327 game_value_t<V> tolerance = {})
328 {
329 using T = game_value_t<V>;
330 std::optional<core_violation<T>> found;
331 detail::with_table(N, v, [&](std::span<const T> table) {
332 const coalition all = grand_coalition(N);
333 const T base = table[0];
334 for (coalition s = 1; s <= all && !found; ++s) {
335 T given{};
336 for (coalition m = s; m != 0; m &= m - 1)
337 given = given + allocation[static_cast<std::size_t>(std::countr_zero(m))];
338 const T shortfall = given - (table[s] - base);
339 const bool wrong_total = s == all && (shortfall < T{} - tolerance || T{} + tolerance < shortfall);
340 if (wrong_total || shortfall < T{} - tolerance)
341 found = core_violation<T>{s, shortfall};
342 }
343 });
344 return found;
345 }
