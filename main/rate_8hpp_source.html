109 [[nodiscard]] static constexpr result<rate, money_errc> parse(const asset& from, const asset& to,
110 std::string_view text) noexcept
111 {
112 const std::size_t point = text.find('.');
113 if (text.empty() || text == "." ||
114 (point != std::string_view::npos && text.find('.', point + 1) != std::string_view::npos))
115 return err(money_errc::malformed);
116 const std::size_t places = point == std::string_view::npos ? 0 : text.size() - point - 1;
117 if (places > 77)
118 return err(money_errc::overflow);
119 uint256 n;
120 for (const char c : text) {
121 if (c == '.')
122 continue;
123 if (c < '0' || c > '9')
124 return err(money_errc::malformed);
125 const auto t = uint256::checked_mul(n, 10);
126 const auto s = t ? uint256::checked_add(*t, static_cast<std::uint64_t>(c - '0')) : std::nullopt;
127 if (!s)
128 return err(money_errc::overflow);
129 n = *s;
130 }
131 return make(from, to, n, uint256::pow10(static_cast<unsigned>(places)));
132 }

139 [[nodiscard]] static constexpr result<rate, money_errc> from_decimal(const asset& from, const asset& to,
140 const decimal& d) noexcept
141 {
142 if (d.is_negative())
143 return err(money_errc::malformed);
144 const uint256 m = static_cast<std::uint64_t>(d.mantissa());
145 const int e = d.exponent();
146 if (e >= 0)
147 return make(from, to, m * uint256::pow10(static_cast<unsigned>(e)), 1);
148 return make(from, to, m, uint256::pow10(static_cast<unsigned>(-e)));
149 }

238 [[nodiscard]] constexpr result<residual, money_errc> sum_residuals(std::span<const residual> parts,
239 const uint256& denominator) noexcept
240 {
241 if (denominator.is_zero())
242 return err(money_errc::malformed);
243 uint256 up{}, down{}; // the positive and the negative parts, summed apart: one carry chain per part
244 for (const residual& part : parts) {
245 if (part.denominator != denominator)
246 return err(money_errc::malformed);
247 uint256& side = part.negative ? down : up;
248 const auto sum = uint256::checked_add(side, part.numerator);
249 if (!sum)
250 return err(money_errc::overflow);
251 side = *sum;
252 }
253 return ok(up >= down ? residual{false, up - down, denominator} : residual{true, down - up, denominator});
254 }

283 [[nodiscard]] constexpr result<priced_rate, money_errc> convert_rate(const rate& r, unsigned places, rounding mode) noexcept
284 {
285 if (places > 77)
286 return err(money_errc::overflow);
287 const uint256 scale = uint256::pow10(places);
288 const auto qr = detail::mul_div(r.numerator(), scale, r.denominator());
289 if (!qr)
290 return err(money_errc::overflow);
291 auto [q, rem] = *qr;
292 const uint256& d = r.denominator();
293 residual left{false, rem, d};
294 if (rounds_away(q, left, mode)) {
295 const auto up = uint256::checked_add(q, 1);
296 if (!up)
297 return err(money_errc::overflow);
298 q = *up;
299 left = residual{true, d - rem, d}; // the rounded rate overshoots by (d - rem) / d
300 }
301 return ok(priced_rate{*rate::make(r.from(), r.to(), q, scale), left});
302 }

309 [[nodiscard]] constexpr result<conversion, money_errc> convert(const money& amount, const rate& r, rounding mode) noexcept
310 {
311 if (amount.asset_of() != r.from())
312 return err(money_errc::asset_mismatch);
313 const auto n = uint256::checked_mul(r.numerator(), uint256::pow10(r.to().scale()));
314 const auto d = uint256::checked_mul(r.denominator(), uint256::pow10(r.from().scale()));
315 if (!n || !d)
316 return err(money_errc::overflow);
317 const auto qr = detail::mul_div(amount.magnitude(), *n, *d);
318 if (!qr)
319 return err(money_errc::overflow);
320 auto [q, rem] = *qr;
321 residual left{amount.is_negative(), rem, *d};
322 if (detail::rounds_away(q, rem, *d, amount.is_negative(), mode)) {
323 const auto up = uint256::checked_add(q, 1);
324 if (!up)
325 return err(money_errc::overflow);
326 q = *up;
327 left = residual{!amount.is_negative(), *d - rem, *d}; // the rounded value overshoots by (d - r) / d
328 }
329 if (left.numerator.is_zero())
330 left.negative = false;
331 return ok(conversion{money::from_units(r.to(), q, amount.is_negative()), left});
332 }
