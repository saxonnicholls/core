382 consteval std::array<Word, N> expand(const std::array<std::uint16_t, N>& index, int bits, int dims,
383 const std::array<std::uint8_t, Levels>& levels)
384 {
385 std::array<Word, N> grid{};
386 for (std::size_t g = 0; g < N; ++g) {
387 Word w = 0;
388 for (int k = 0; k < dims; ++k)
389 w |= Word(levels[(index[g] >> (bits * k)) & ((1u << bits) - 1)]) << (8 * k);
390 grid[g] = w;
391 }
392 return grid;
393 }

462{
