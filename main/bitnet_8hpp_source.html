90 // clamp to [lo, hi], then round half to even (PyTorch's round, which commutes with an integer clamp); a NaN
91 // passes the clamp and nearest_int's bit trick makes it 0
92 // |x| by clearing the sign bit: no branch on the sign of random data
93 constexpr float fabs(float x)
94 {
95 return std::bit_cast<float>(std::bit_cast<std::uint32_t>(x) & 0x7fffffffu);
96 }

98 constexpr int round_clamp(float v, int lo, int hi)
99 {
100 const float c = v < float(lo) ? float(lo) : v > float(hi) ? float(hi) : v;
101 return quantize::nearest_int(c);
102 }

110template <row W>
111 requires std::same_as<row_value_t<W>, float>
112constexpr float absmean_quantize(const W& w, std::span<std::int8_t> q, float eps = bitnet_eps) noexcept
113{
114 const auto in = as_row(w);
115 assert(q.size() == in.size());
116 double sum = 0;
117 for (const float v : in)
118 sum += double(detail::bitnet::fabs(v));
119 const float mean = in.empty() ? 0.0f : float(sum / double(in.size()));
120 const float gamma = mean > eps ? mean : eps;
121 const float s = 1.0f / gamma;
122 for (std::size_t i = 0; i < in.size(); ++i) {
123 const float v = in[i] * s;
124 q[i] = static_cast<std::int8_t>(detail::bitnet::round_clamp(v, -1, 1));
125 }
126 return gamma;
127}

160template <int Bits = 8, row X>
161 requires(Bits >= 2 && Bits <= 8 && std::same_as<row_value_t<X>, float>)
162constexpr float absmax_quantize(const X& x, std::span<std::int8_t> q, float eps = bitnet_eps) noexcept
163{
164 const auto in = as_row(x);
165 assert(q.size() == in.size());
166 constexpr int top = (1 << (Bits - 1)) - 1;
167 float amax = 0.0f;
168 for (const float v : in)
169 amax = detail::quantize::fmax(amax, detail::quantize::abs(v));
170 const float s = float(top) / (amax > eps ? amax : eps);
171 for (std::size_t i = 0; i < in.size(); ++i) {
172 const float v = in[i] * s;
173 q[i] = static_cast<std::int8_t>(detail::bitnet::round_clamp(v, -top - 1, top));
174 }
175 return s;
176}

219template <row D, row X, mutable_row DX>
220 requires std::same_as<row_value_t<D>, float> && std::same_as<row_value_t<X>, float> && std::same_as<row_value_t<DX>, float>
221constexpr void ste_backward(const D& dout, const X& x, float s, int qmin, int qmax, DX&& dx, ste mode = ste::identity) noexcept
222{
223 const auto g = as_row(dout);
224 const auto in = as_row(x);
225 const auto out = as_mutable_row(dx);
226 assert(g.size() == in.size() && out.size() == in.size());
227 const float lo = float(qmin) - 0.5f, hi = float(qmax) + 0.5f;
228 for (std::size_t i = 0; i < in.size(); ++i) {
229 const float v = in[i] * s;
230 out[i] = mode == ste::identity || (v >= lo && v <= hi) ? g[i] : 0.0f;
231 }
232}

238constexpr std::int32_t ternary_dot(std::span<const std::int8_t> t, std::span<const std::int8_t> q) noexcept
239{
240 assert(t.size() == q.size());
241 std::int32_t s = 0;
242 for (std::size_t j = 0; j < t.size(); ++j)
243 s += std::int32_t(t[j]) * std::int32_t(q[j]);
244 return s;
245}

262template <row X, mutable_row Y>
263 requires std::same_as<row_value_t<X>, float> && std::same_as<row_value_t<Y>, float>
264constexpr void bitlinear(const bitlinear_weights& w, const X& x, Y&& y, std::span<std::int8_t> codes) noexcept
265{
266 const auto in = as_row(x);
267 const auto out = as_mutable_row(y);
268 assert(in.size() == w.cols && out.size() == w.rows && codes.size() == w.cols && w.ternary.size() == w.rows * w.cols);
269 const float s = absmax_quantize<8>(in, codes);
270 for (std::size_t i = 0; i < w.rows; ++i) {
271 const std::int32_t acc = ternary_dot(w.ternary.subspan(i * w.cols, w.cols), codes);
272 const float a = float(acc) / s;
273 out[i] = a * w.scale;
274 }
275}
