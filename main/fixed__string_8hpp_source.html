148 // All comparisons go through string_view, so a fixed string compares
149 // naturally against another fixed string of ANY length, a string_view,
150 // a std::string, or a literal - with no ambiguity from the implicit
151 // literal constructor. (No defaulted operator== is needed: a class with
152 // a public character array is already a structural type, so it works as
153 // a non-type template parameter regardless.)
157 constexpr bool operator==(view_type other) const noexcept { return view() == other; }
161 constexpr auto operator<=>(view_type other) const noexcept { return view() <=> other; }
162
163 // Fixed string against fixed string, any two lengths. Without these, a == b
164 // converts each side to the other's view_type, so the normal and the
165 // reversed (rewritten) member candidates tie: ambiguous on Clang for
166 // different lengths, a warning on Clang and GCC, an error on MSVC even for
167 // equal lengths (and so for std::equal_to). An exact match on both
168 // arguments beats both, and the non-reversed form beats the reversed one.
174 template<size_type M>
175 friend constexpr bool operator==(const basic_fixed_string& a, const basic_fixed_string<CharT, M>& b) noexcept {
176 return a.view() == b.view();
177 }

183 template<size_type M>
184 friend constexpr std::strong_ordering operator<=>(const basic_fixed_string& a,
185 const basic_fixed_string<CharT, M>& b) noexcept {
186 // compare(), not view() <=> view(): libc++ before LLVM 16 (Xcode 14) has no
187 // operator<=> for basic_string_view, and compare() is everywhere
188 const int c = a.view().compare(b.view());
189 return c < 0 ? std::strong_ordering::less
190 : c > 0 ? std::strong_ordering::greater
191 : std::strong_ordering::equal;
192 }

194 // Concatenation: fixed_string<N> + fixed_string<M> -> fixed_string<N+M>
199 template<size_type M>
200 constexpr basic_fixed_string<CharT, N + M>
201 operator+(const basic_fixed_string<CharT, M>& rhs) const noexcept {
202 basic_fixed_string<CharT, N + M> out;
203 for (size_type i = 0; i < N; ++i) out.elems[i] = elems[i];
204 for (size_type i = 0; i < M; ++i) out.elems[N + i] = rhs.elems[i];
205 out.elems[N + M] = CharT{};
206 return out;
207 }

209 // Compile-time substring
214 template<size_type Pos, size_type Count>
215 constexpr basic_fixed_string<CharT, Count> substr() const noexcept {
216 static_assert(Pos + Count <= N, "basic_fixed_string::substr: range out of bounds");
217 basic_fixed_string<CharT, Count> out;
218 for (size_type i = 0; i < Count; ++i) out.elems[i] = elems[Pos + i];
219 out.elems[Count] = CharT{};
220 return out;
221 }

230 // Convenience wrappers mirroring the standard character-type family. These
231 // are class templates (each deriving from basic_fixed_string and carrying its
232 // own deduction guide) rather than alias templates, because alias-template
233 // CTAD (P1814) is not supported by every clang release - so `fixed_string("abc")`
234 // must deduce through a real class template. basic_fixed_string above remains
235 // the type to name for non-type template parameters (Tag<"object1">).
239#define SNICHOLLS_FIXED_STRING_WRAPPER(Name, CharT) \
240 template<std::size_t N> \
241 struct Name : basic_fixed_string<CharT, N> { \
242 using basic_fixed_string<CharT, N>::basic_fixed_string; \
243 constexpr Name(const basic_fixed_string<CharT, N>& base) noexcept \
244 : basic_fixed_string<CharT, N>(base) {} \
245 }; \
246 template<std::size_t N> Name(const CharT (&)[N]) -> Name<N - 1>;
247
248 SNICHOLLS_FIXED_STRING_WRAPPER(fixed_string, char)
249 SNICHOLLS_FIXED_STRING_WRAPPER(wfixed_string, wchar_t)
250 SNICHOLLS_FIXED_STRING_WRAPPER(u8fixed_string, char8_t)
251 SNICHOLLS_FIXED_STRING_WRAPPER(u16fixed_string, char16_t)
252 SNICHOLLS_FIXED_STRING_WRAPPER(u32fixed_string, char32_t)
253#undef SNICHOLLS_FIXED_STRING_WRAPPER
254}
