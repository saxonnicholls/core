88template <typename B, math::math_number T>
89constexpr void token_shift(const B&, std::type_identity_t<std::span<const T>> xs, std::type_identity_t<std::span<const T>> mu,
90 std::span<T> out, std::size_t sequences = 1, std::type_identity_t<std::span<T>> states = {})
91{
92 const std::size_t d = mu.size();
93 assert(d > 0 && sequences > 0 && xs.size() % (d * sequences) == 0 && out.size() == xs.size()
94 && (states.empty() || states.size() == sequences * d));
95 const std::size_t n = xs.size() / (d * sequences);
96 for (std::size_t s = 0; s < sequences; ++s) {
97 for (std::size_t i = 0; i < n; ++i)
98 for (std::size_t c = 0; c < d; ++c) {
99 const std::size_t t = s * n + i;
100 const T prev = i > 0 ? xs[(t - 1) * d + c] : states.empty() ? T(0) : states[s * d + c];
101 out[t * d + c] = (prev - xs[t * d + c]) * mu[c] + xs[t * d + c];
102 }
103 if (!states.empty() && n > 0)
104 for (std::size_t c = 0; c < d; ++c)
105 states[s * d + c] = xs[(s * n + n - 1) * d + c];
106 }
107}

114template <typename B, math::math_number T>
115constexpr void token_shift_backward(const B&, std::type_identity_t<std::span<const T>> xs, std::type_identity_t<std::span<const T>> mu,
116 std::type_identity_t<std::span<const T>> dout, std::span<T> dxs,
117 std::type_identity_t<std::span<T>> dmu, std::size_t sequences = 1,
118 std::type_identity_t<std::span<const T>> initial = {}, std::type_identity_t<std::span<T>> dinitial = {})
119{
120 const std::size_t d = mu.size();
121 assert(d > 0 && sequences > 0 && xs.size() % (d * sequences) == 0 && dout.size() == xs.size() && dxs.size() == xs.size()
122 && dmu.size() == d && (initial.empty() || initial.size() == sequences * d) && (dinitial.empty() || dinitial.size() == sequences * d));
123 const std::size_t n = xs.size() / (d * sequences);
124 for (std::size_t s = 0; s < sequences; ++s)
125 for (std::size_t i = 0; i < n; ++i)
126 for (std::size_t c = 0; c < d; ++c) {
127 const std::size_t t = s * n + i;
128 const T prev = i > 0 ? xs[(t - 1) * d + c] : initial.empty() ? T(0) : initial[s * d + c];
129 const T g = dout[t * d + c];
130 dmu[c] += (prev - xs[t * d + c]) * g;
131 dxs[t * d + c] = g - g * mu[c] + (i + 1 < n ? dout[(t + 1) * d + c] * mu[c] : T(0));
132 if (i == 0 && !dinitial.empty())
133 dinitial[s * d + c] = g * mu[c];
134 }
135}

139template <math::math_number T>
140constexpr void token_shift_step(std::type_identity_t<std::span<const T>> x, std::type_identity_t<std::span<const T>> mu,
141 std::span<T> out, std::type_identity_t<std::span<T>> state)
142{
143 assert(x.size() == mu.size() && out.size() == x.size() && state.size() == x.size());
144 for (std::size_t c = 0; c < x.size(); ++c) {
145 out[c] = (state[c] - x[c]) * mu[c] + x[c];
146 state[c] = x[c];
147 }

219 template <transformer_config C, typename T>
220 struct wkv6
221 {
222 using value_type = T;
223 static constexpr std::size_t H = C.mixer_heads, N = C.mixer_state;
224 static constexpr std::size_t state_size = H * N * N, work_size = 0;
225
226 wkv6_inputs<T> in{};
227 std::span<T> o{};
228 std::span<const T> dout{};
229 wkv6_gradients<T> g{};
230
231 constexpr void update(std::size_t row, std::span<T> state, bool output) const
232 {
233 for (std::size_t h = 0; h < H; ++h) {
234 const std::size_t base = (row * H + h) * N;
235 const auto s = state.subspan(h * N * N, N * N);
236 std::array<T, N> out{};
237 for (std::size_t n = 0; n < N; ++n) {
238 const T k = in.k[base + n], r = in.r[base + n], u = in.u[h * N + n], w = math::exp(in.lw[base + n]);
239 for (std::size_t p = 0; p < N; ++p) {
240 const T kv = in.v[base + p] * k, prev = s[n * N + p];
241 out[p] += (kv * u + prev) * r;
242 s[n * N + p] = prev * w + kv;
243 }
244 }
245 if (output)
246 for (std::size_t p = 0; p < N; ++p)
247 o[base + p] = out[p];
248 }
249 }
250 constexpr void step(std::size_t row, std::span<T> state) const { update(row, state, true); }
251 constexpr void advance(std::size_t row, std::span<T> state) const { update(row, state, false); }
252
253 constexpr void step_backward(std::size_t row, std::span<const T> prev, std::span<T> dstate, std::span<T>) const
254 {
255 for (std::size_t h = 0; h < H; ++h) {
256 const std::size_t base = (row * H + h) * N;
257 const auto s0 = prev.subspan(h * N * N, N * N);
258 const auto ds = dstate.subspan(h * N * N, N * N);
259 std::array<T, N> dv{};
260 for (std::size_t n = 0; n < N; ++n) {
261 const T k = in.k[base + n], r = in.r[base + n], u = in.u[h * N + n], w = math::exp(in.lw[base + n]);
262 T dr = T(0), dk = T(0), du = T(0), dw = T(0);
263 for (std::size_t p = 0; p < N; ++p) {
264 const T v = in.v[base + p], kv = v * k, d = dout[base + p];
265 dr += d * (kv * u + s0[n * N + p]);
266 const T dt = d * r; // the gradient in kv · u + prev
267 const T dkv = dt * u + ds[n * N + p]; // kv feeds the output and the state
268 du += dt * kv;
269 dw += ds[n * N + p] * s0[n * N + p];
270 dk += dkv * v;
271 dv[p] += dkv * k;
272 ds[n * N + p] = ds[n * N + p] * w + dt;
273 }
274 g.r[base + n] = dr;
275 g.k[base + n] = dk;
276 g.lw[base + n] = dw * w;
277 g.u[h * N + n] += du;
278 }
279 for (std::size_t p = 0; p < N; ++p)
280 g.v[base + p] = dv[p];
281 }
282 }
283 };

285 template <transformer_config C, typename T>
286 struct wkv7
287 {
288 using value_type = T;
289 static constexpr std::size_t H = C.mixer_heads, N = C.mixer_state;
290 static constexpr std::size_t state_size = H * N * N, work_size = 0;
291
292 wkv7_inputs<T> in{};
293 std::span<T> o{};
294 std::span<const T> dout{};
295 wkv7_gradients<T> g{};
296
297 constexpr void update(std::size_t row, std::span<T> state, bool output) const
298 {
299 for (std::size_t h = 0; h < H; ++h) {
300 const std::size_t base = (row * H + h) * N;
301 const auto s = state.subspan(h * N * N, N * N);
302 std::array<T, N> sa{}, out{};
303 for (std::size_t n = 0; n < N; ++n)
304 for (std::size_t p = 0; p < N; ++p)
305 sa[p] += in.a[base + n] * s[n * N + p];
306 for (std::size_t n = 0; n < N; ++n) {
307 const T k = in.k[base + n], w = math::exp(in.lw[base + n]), b = in.b[base + n], r = in.r[base + n];
308 for (std::size_t p = 0; p < N; ++p) {
309 s[n * N + p] = s[n * N + p] * w + in.v[base + p] * k + sa[p] * b;
310 out[p] += s[n * N + p] * r;
311 }
312 }
313 if (output)
314 for (std::size_t p = 0; p < N; ++p)
315 o[base + p] = out[p];
316 }
317 }
318 constexpr void step(std::size_t row, std::span<T> state) const { update(row, state, true); }
319 constexpr void advance(std::size_t row, std::span<T> state) const { update(row, state, false); }
320
321 constexpr void step_backward(std::size_t row, std::span<const T> prev, std::span<T> dstate, std::span<T>) const
322 {
323 for (std::size_t h = 0; h < H; ++h) {
324 const std::size_t base = (row * H + h) * N;
325 const auto s0 = prev.subspan(h * N * N, N * N);
326 const auto ds = dstate.subspan(h * N * N, N * N);
327 std::array<T, N> sa{}, dsa{}, dv{};
328 for (std::size_t n = 0; n < N; ++n)
329 for (std::size_t p = 0; p < N; ++p)
330 sa[p] += in.a[base + n] * s0[n * N + p];
331 for (std::size_t n = 0; n < N; ++n) {
332 const T k = in.k[base + n], w = math::exp(in.lw[base + n]), b = in.b[base + n], r = in.r[base + n];
333 T dr = T(0), dw = T(0), dk = T(0), db = T(0);
334 for (std::size_t p = 0; p < N; ++p) {
335 const T v = in.v[base + p], d = dout[base + p];
336 dr += d * (s0[n * N + p] * w + v * k + sa[p] * b);
337 const T dsp = ds[n * N + p] + r * d;
338 dw += dsp * s0[n * N + p];
339 dk += dsp * v;
340 dv[p] += dsp * k;
341 db += dsp * sa[p];
342 dsa[p] += dsp * b;
343 ds[n * N + p] = dsp * w;
344 }
345 g.r[base + n] = dr;
346 g.lw[base + n] = dw * w;
347 g.k[base + n] = dk;
348 g.b[base + n] = db;
349 }
350 for (std::size_t n = 0; n < N; ++n) { // sa = aᵀ S_{t-1}: into a and the previous state
351 T da = T(0);
352 for (std::size_t p = 0; p < N; ++p) {
353 da += dsa[p] * s0[n * N + p];
354 ds[n * N + p] += in.a[base + n] * dsa[p];
355 }
356 g.a[base + n] = da;
357 }
358 for (std::size_t p = 0; p < N; ++p)
359 g.v[base + p] = dv[p];
360 }
361 }
362 };
