60 {
61 const double p1 = static_cast<double>(coordinate(a, 0)), p2 = static_cast<double>(coordinate(b, 0));
62 const double dl = static_cast<double>(coordinate(b, 1)) - static_cast<double>(coordinate(a, 1));
63 const double sp = math::sin((p2 - p1) / 2), sl = math::sin(dl / 2);
64 double h = sp * sp + math::cos(p1) * math::cos(p2) * sl * sl;
65 h = h < 0 ? 0 : h > 1 ? 1 : h;
66 return 2 * radius * math::asin(math::sqrt(h));
67 }
