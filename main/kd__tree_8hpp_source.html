92 template <typename Point, std::size_t K, typename Traits = kd_point_traits>
93 class kd_tree {
94 static_assert(K > 0, "snicholls::containers::kd_tree: the dimension is at least 1");
95 static constexpr std::size_t max_depth = 64;
96 template <typename Q>
97 static constexpr bool nothrow_read = noexcept(Traits::coordinate(std::declval<const Point&>(), std::size_t{}))
98 && noexcept(Traits::coordinate(std::declval<const Q&>(), std::size_t{}));

100 struct frame {
101 std::size_t lo, hi, depth;
102 double plane2; // the squared distance to the plane that bounds this range, a lower bound on its points
103 };
104
105 public:
109 constexpr kd_tree(std::span<const Point> points, std::span<std::uint32_t> indices) noexcept(nothrow_read<Point>)
110 : points_(points), index_(indices.first(indices.size() < points.size() ? indices.size() : points.size()))
111 {
112 for (std::size_t i = 0; i < index_.size(); ++i)
113 index_[i] = static_cast<std::uint32_t>(i);
114 std::array<frame, max_depth> stack{};
115 std::size_t top = 0;
116 if (index_.size() > 1)
117 stack[top++] = {0, index_.size(), 0, 0};
118 while (top > 0) {
119 const frame f = stack[--top];
120 const std::size_t mid = f.lo + (f.hi - f.lo) / 2;
121 const std::size_t axis = f.depth % K;
122 std::nth_element(index_.begin() + static_cast<std::ptrdiff_t>(f.lo), index_.begin() + static_cast<std::ptrdiff_t>(mid),
123 index_.begin() + static_cast<std::ptrdiff_t>(f.hi),
124 [&](std::uint32_t a, std::uint32_t b) { return Traits::coordinate(points_[a], axis) < Traits::coordinate(points_[b], axis); });
125 if (mid - f.lo > 1)
126 stack[top++] = {f.lo, mid, f.depth + 1, 0};
127 if (f.hi - (mid + 1) > 1)
128 stack[top++] = {mid + 1, f.hi, f.depth + 1, 0};
129 }
130 }

133 [[nodiscard]] constexpr std::size_t size() const noexcept { return index_.size(); }
135 [[nodiscard]] constexpr bool empty() const noexcept { return index_.empty(); }

141 template <typename Q>
142 [[nodiscard]] constexpr std::optional<kd_neighbour> nearest(const Q& q) const noexcept(nothrow_read<Q>)
143 {
144 if (empty())
145 return std::nullopt;
146 kd_neighbour best{0, std::numeric_limits<double>::infinity()};
147 search(q, [&](std::uint32_t i, double d2) {
148 if (d2 < best.distance2)
149 best = {i, d2};
150 }, [&] { return best.distance2; });
151 return best;
152 }

159 template <typename Q>
160 constexpr std::size_t k_nearest(const Q& q, std::span<kd_neighbour> out) const noexcept(nothrow_read<Q>)
161 {
162 const std::size_t k = out.size() < size() ? out.size() : size();
163 if (k == 0)
164 return 0;
165 std::size_t have = 0;
166 search(q, [&](std::uint32_t i, double d2) {
167 if (have == k && !(d2 < out[k - 1].distance2))
168 return;
169 std::size_t j = have < k ? have++ : k - 1;
170 for (; j > 0 && out[j - 1].distance2 > d2; --j)
171 out[j] = out[j - 1];
172 out[j] = {i, d2};
173 }, [&] { return have < k ? std::numeric_limits<double>::infinity() : out[k - 1].distance2; });
174 return k;
175 }

198 template <typename Q, typename F>
199 constexpr void for_each_in_box(const Q& lo, const Q& hi, F&& f) const noexcept(nothrow_read<Q> && noexcept(f(std::uint32_t{})))
200 {
201 std::array<frame, max_depth> stack{};
202 std::size_t top = 0;
203 if (!empty())
204 stack[top++] = {0, index_.size(), 0, 0};
205 while (top > 0) {
206 const frame fr = stack[--top];
207 const std::size_t mid = fr.lo + (fr.hi - fr.lo) / 2;
208 const std::size_t axis = fr.depth % K;
209 const auto& p = points_[index_[mid]];
210 bool inside = true;
211 for (std::size_t a = 0; a < K && inside; ++a) {
212 const double c = Traits::coordinate(p, a);
213 inside = c >= Traits::coordinate(lo, a) && c <= Traits::coordinate(hi, a);
214 }
215 if (inside)
216 f(index_[mid]);
217 const double c = Traits::coordinate(p, axis);
218 if (mid > fr.lo && Traits::coordinate(lo, axis) <= c)
219 stack[top++] = {fr.lo, mid, fr.depth + 1, 0};
220 if (mid + 1 < fr.hi && Traits::coordinate(hi, axis) >= c)
221 stack[top++] = {mid + 1, fr.hi, fr.depth + 1, 0};
222 }
223 }

226 template <typename Q>
227 constexpr double distance2(const Q& q, std::uint32_t i) const
228 {
229 double s = 0;
230 for (std::size_t a = 0; a < K; ++a) {
231 const double d = Traits::coordinate(points_[i], a) - Traits::coordinate(q, a);
232 s += d * d;
233 }
234 return s;
235 }

237 // Visits candidates nearest-first along the tree, pruning ranges whose plane is beyond bound()
238 template <typename Q, typename Visit, typename Bound>
239 constexpr void search(const Q& q, Visit&& visit, Bound&& bound) const
240 {
241 std::array<frame, max_depth> stack{};
242 std::size_t top = 0;
243 if (!empty())
244 stack[top++] = {0, index_.size(), 0, 0};
245 while (top > 0) {
246 const frame fr = stack[--top];
247 if (fr.plane2 > bound())
248 continue;
249 const std::size_t mid = fr.lo + (fr.hi - fr.lo) / 2;
250 const std::size_t axis = fr.depth % K;
251 const std::uint32_t node = index_[mid];
252 visit(node, distance2(q, node));
253 const double diff = Traits::coordinate(q, axis) - Traits::coordinate(points_[node], axis);
254 const bool left_near = diff < 0;
255 const frame lo_child{fr.lo, mid, fr.depth + 1, left_near ? fr.plane2 : (fr.plane2 > diff * diff ? fr.plane2 : diff * diff)};
256 const frame hi_child{mid + 1, fr.hi, fr.depth + 1, left_near ? (fr.plane2 > diff * diff ? fr.plane2 : diff * diff) : fr.plane2};
257 // Push the far child first so the near one is searched first
258 const frame& near = left_near ? lo_child : hi_child;
259 const frame& far = left_near ? hi_child : lo_child;
260 if (far.hi > far.lo)
261 stack[top++] = far;
262 if (near.hi > near.lo)
263 stack[top++] = near;
264 }
265 }

269 };
