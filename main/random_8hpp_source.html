144 template <full_range_generator G>
145 [[nodiscard]] constexpr std::uint64_t uniform_index(G& g, std::uint64_t n) noexcept(noexcept(g()))
146 {
147 assert(n > 0);
148 snicholls::detail::wide_product m = snicholls::detail::umul128(random_u64(g), n);
149 if (m.low < n) {
150 const std::uint64_t threshold = (std::uint64_t{0} - n) % n;
151 while (m.low < threshold)
152 m = snicholls::detail::umul128(random_u64(g), n);
153 }
154 return m.high;
155 }

167 template <typename Refill>
168 [[nodiscard]] constexpr std::uint32_t uniform_index32(std::uint32_t bits, std::uint32_t n, Refill&& refill) noexcept(noexcept(refill()))
169 {
170 assert(n > 0);
171 std::uint64_t m = static_cast<std::uint64_t>(bits) * n;
172 auto low = static_cast<std::uint32_t>(m);
173 if (low < n) {
174 const std::uint32_t threshold = (std::uint32_t{0} - n) % n;
175 while (low < threshold) {
176 m = static_cast<std::uint64_t>(static_cast<std::uint32_t>(refill())) * n;
177 low = static_cast<std::uint32_t>(m);
178 }
179 }
180 return static_cast<std::uint32_t>(m >> 32);
181 }
