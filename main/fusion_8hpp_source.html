74 template <linalg::scalar T, std::size_t N>
75 [[nodiscard]] constexpr result<gaussian<T, N>, linalg::error> covariance_intersection(std::span<const gaussian<T, N>> estimates, std::span<const T> weights) noexcept(linalg::detail::nothrow_scalar<T>)
76 {
77 if (estimates.empty() || estimates.size() != weights.size())
78 return err(linalg::error::dimension_mismatch);
79 linalg::fixed_matrix<T, N, N> info{};
80 linalg::fixed_vector<T, N> vec{};
81 for (std::size_t i = 0; i < estimates.size(); ++i) {
82 if (weights[i] == T(0))
83 continue;
84 const auto y = linalg::spd_inverse(estimates[i].covariance);
85 if (!y)
86 return err(y.unwrap_err_unchecked());
87 info += *y * weights[i];
88 vec += *y * estimates[i].mean * weights[i];
89 }
90 const auto l = linalg::cholesky(info);
91 if (!l)
92 return err(l.unwrap_err_unchecked());
93 return ok(gaussian<T, N>{linalg::cholesky_solve(*l, vec), linalg::symmetrise(linalg::cholesky_solve(*l, linalg::fixed_matrix<T, N, N>::identity()))});
94 }

118 template <linalg::scalar T, std::size_t N>
119 [[nodiscard]] constexpr result<T, linalg::error> optimal_omega(const gaussian<T, N>& a, const gaussian<T, N>& b, T lo = T(0), T hi = T(1),
120 ci_criterion criterion = ci_criterion::trace) noexcept(linalg::detail::nothrow_scalar<T>)
121 {
122 const auto ya = linalg::spd_inverse(a.covariance);
123 const auto yb = linalg::spd_inverse(b.covariance);
124 if (!ya || !yb)
125 return err(linalg::error::not_positive_definite);
126 // Cost of ω from the information matrices alone; a singular combination costs +∞
127 auto cost = [&](T w) {
128 const auto info = *ya * w + *yb * (T(1) - w);
129 const auto l = linalg::cholesky(info);
130 if (!l)
131 return std::numeric_limits<T>::infinity();
132 if (criterion == ci_criterion::determinant)
133 return -linalg::cholesky_log_det(*l);
134 return linalg::trace(linalg::cholesky_solve(*l, linalg::fixed_matrix<T, N, N>::identity()));
135 };
136 const T g = T(0.6180339887498948482);
137 T x0 = lo, x3 = hi;
138 T x1 = x3 - g * (x3 - x0), x2 = x0 + g * (x3 - x0);
139 T f1 = cost(x1), f2 = cost(x2);
140 for (int i = 0; i < 80; ++i) {
141 if (f1 <= f2) {
142 x3 = x2;
143 x2 = x1;
144 f2 = f1;
145 x1 = x3 - g * (x3 - x0);
146 f1 = cost(x1);
147 } else {
148 x0 = x1;
149 x1 = x2;
150 f1 = f2;
151 x2 = x0 + g * (x3 - x0);
152 f2 = cost(x2);
153 }
154 }
155 T best = (x0 + x3) / T(2);
156 T fb = cost(best);
157 if (cost(lo) < fb) {
158 best = lo;
159 fb = cost(lo);
160 }
161 if (cost(hi) < fb)
162 best = hi;
163 return ok(best);
164 }

176 template <linalg::scalar T, std::size_t N>
177 constexpr bool fast_ci_weights(std::span<const gaussian<T, N>> estimates, std::span<T> weights, std::span<const T> lo = {}, std::span<const T> hi = {}) noexcept(linalg::detail::nothrow_scalar<T>)
178 {
179 const std::size_t n = estimates.size();
180 if (n == 0 || weights.size() != n || (!lo.empty() && lo.size() != n) || (!hi.empty() && hi.size() != n))
181 return false;
182 T lo_sum(0), hi_sum(0);
183 for (std::size_t i = 0; i < n; ++i) {
184 lo_sum += lo.empty() ? T(0) : lo[i];
185 hi_sum += hi.empty() ? T(1) : hi[i];
186 }
187 if (lo_sum > T(1) || hi_sum < T(1))
188 return false;
189 T s(0);
190 for (std::size_t i = 0; i < n; ++i) {
191 const T t = linalg::trace(estimates[i].covariance);
192 weights[i] = t > T(0) ? T(1) / t : T(0);
193 s += weights[i];
194 }
195 for (std::size_t i = 0; i < n; ++i)
196 weights[i] = s > T(0) ? weights[i] / s : T(1) / static_cast<T>(n);
197 // Clamp: ωᵢ = clamp(λ rᵢ, loᵢ, hiᵢ), rᵢ the proportional weights, with λ such that Σ ωᵢ = 1. The sum is
198 // non-decreasing and piecewise linear in λ, with breakpoints at loᵢ / rᵢ and hiᵢ / rᵢ: take the largest
199 // breakpoint at which it is at most 1 and the smallest at which it is at least 1, and interpolate (no
200 // breakpoint lies between them). O(n²), exact up to rounding.
201 auto bound = [&](std::size_t i, bool upper) { return upper ? (hi.empty() ? T(1) : hi[i]) : (lo.empty() ? T(0) : lo[i]); };
202 auto clamped = [&](std::size_t i, T lambda) {
203 const T w = lambda * weights[i], l = bound(i, false), h = bound(i, true);
204 return w < l ? l : w > h ? h : w;
205 };
206 auto total = [&](T lambda) {
207 T t(0);
208 for (std::size_t i = 0; i < n; ++i)
209 t += clamped(i, lambda);
210 return t;
211 };
212 T below(0), f_below = total(T(0)), above(-1), f_above(0);
213 for (std::size_t k = 0; k < 2 * n; ++k) {
214 const std::size_t i = k / 2;
215 if (!(weights[i] > T(0)))
216 continue;
217 const T lambda = bound(i, k % 2 == 1) / weights[i];
218 const T f = total(lambda);
219 if (f <= T(1) && lambda > below) {
220 below = lambda;
221 f_below = f;
222 }
223 if (f >= T(1) && (above < T(0) || lambda < above)) {
224 above = lambda;
225 f_above = f;
226 }
227 }
228 const T lambda = above < T(0) || !(f_above > f_below) ? below : below + (above - below) * (T(1) - f_below) / (f_above - f_below);
229 for (std::size_t i = 0; i < n; ++i)
230 weights[i] = clamped(i, lambda);
231 return true;
232 }
