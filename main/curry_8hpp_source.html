94 // Part 0 is f, parts 1... the arguments so far.
95 template <std::size_t N>
96 struct curry_op {
97 // The last arguments: call f.
98 template <std::size_t... I, class S, class... A>
99 requires(sizeof...(I) + sizeof...(A) == N)
100 static constexpr auto call(std::index_sequence<0, I...>, S&& s, A&&... a)
101 SNICHOLLS_FN_RETURNS(std::invoke(detail::part<0>(std::forward<S>(s)),
102 detail::part<I>(std::forward<S>(s))..., std::forward<A>(a)...))
103
104 // Not yet: a new closure with these arguments added.
105 template <std::size_t... I, class S, class... A>
106 requires(sizeof...(A) > 0 && sizeof...(I) + sizeof...(A) < N)
107 static constexpr auto call(std::index_sequence<0, I...>, S&& s, A&&... a) noexcept(
108 nothrow_store_v<part_t<0, S>, part_t<I, S>..., A...>)
109 {
110 return detail::make<curry_op>(detail::part<0>(std::forward<S>(s)),
111 detail::part<I>(std::forward<S>(s))..., std::forward<A>(a)...);
112 }
113 };

115 // g(a0)(a1)...(ak): one call per argument.
116 template <std::size_t K>
117 struct apply_each {
118 template <class G, class A0, class... A>
119 static constexpr auto call(G&& g, A0&& a0, A&&... a)
120 SNICHOLLS_FN_RETURNS(apply_each<K - 1>::call(std::invoke(std::forward<G>(g), std::forward<A0>(a0)),
121 std::forward<A>(a)...))
122 };

197 template <class F>
198 [[nodiscard]] constexpr auto curry(F&& f) noexcept(detail::nothrow_store_v<F>)
199 {
200 static_assert(single_signature_callable<F>,
201 "snicholls::fn::curry: a generic or overloaded callable has no arity of its own; give it: "
202 "curry<N>(f)");
203 return fn::curry<detail::arity_of<F>()>(std::forward<F>(f));
204 }
