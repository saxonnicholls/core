88 // The butterflies, ggml's order: passes of length 1, 2, 4, ...; within a pass, pairs (i + j, i + len + j)
89 template <typename T>
90 constexpr void butterflies(T* x, std::size_t n) noexcept
91 {
92 for (std::size_t len = 1; len < n; len <<= 1)
93 for (std::size_t i = 0; i < n; i += 2 * len)
94 for (std::size_t j = 0; j < len; ++j) {
95 const T u = x[i + j];
96 const T v = x[i + len + j];
97 x[i + j] = u + v;
98 x[i + len + j] = u - v;
99 }
100 }

102 // 1 / sqrt(n) in T, as ggml's 1.0f / sqrtf((float) n)
103 template <typename T>
104 constexpr T inverse_sqrt(std::size_t n) noexcept
105 {
106 const T root = math::sqrt(static_cast<T>(n));
107 return T(1) / root;
108 }

112 {

130[[nodiscard]] constexpr int hadamard_entry(std::size_t i, std::size_t j) noexcept
131{
132 return std::popcount(i & j) % 2 == 0 ? 1 : -1;

150constexpr void hadamard(X&& x) noexcept
151{
152 using T = row_value_t<X>;
153 const auto r = as_mutable_row(x);
154 assert(is_hadamard_order(r.size()));
155 detail::hadamard::scale(r.data(), r.size(), detail::hadamard::inverse_sqrt<T>(r.size()));

162template <mutable_row X>
163 requires std::floating_point<row_value_t<X>>
164constexpr void block_hadamard(X&& x, std::size_t block) noexcept
165{
166 using T = row_value_t<X>;
167 const auto r = as_mutable_row(x);
168 assert(is_hadamard_order(block) && r.size() % block == 0);
169 const T s = detail::hadamard::inverse_sqrt<T>(block);
170 for (std::size_t b = 0; b < r.size(); b += block) {
171 detail::hadamard::scale(r.data() + b, block, s);
172 detail::hadamard::butterflies(r.data() + b, block);
173 }
174}

179constexpr void hadamard_signs(std::span<std::int8_t> signs, std::uint64_t seed) noexcept
180{
181 stats::wyrand g(seed);
182 std::uint64_t bits = 0;
183 for (std::size_t k = 0; k < signs.size(); ++k) {
184 if (k % 64 == 0)
185 bits = g();
186 signs[k] = (bits >> (k % 64)) & 1 ? std::int8_t{-1} : std::int8_t{1};
187 }
188}

194template <mutable_row X>
195 requires std::floating_point<row_value_t<X>>
196constexpr void random_hadamard(X&& x, std::span<const std::int8_t> signs, std::size_t block) noexcept
197{
198 const auto r = as_mutable_row(x);
199 assert(signs.size() == r.size());
200 for (std::size_t i = 0; i < r.size(); ++i) {
201 assert(signs[i] == 1 || signs[i] == -1);
202 r[i] = signs[i] < 0 ? -r[i] : r[i];
203 }
204 block_hadamard(r, block);
205}

232constexpr void rotate_rows(W&& w, std::size_t cols, std::span<const std::int8_t> signs, std::size_t block) noexcept
233{
234 const auto m = as_mutable_row(w);
235 assert(cols != 0 && m.size() % cols == 0);
236 for (std::size_t i = 0; i < m.size(); i += cols)
237 random_hadamard(m.subspan(i, cols), signs, block);

244template <std::floating_point T>
245constexpr void hadamard_matrix(std::size_t n, std::span<T> m) noexcept
246{
247 assert(is_hadamard_order(n) && m.size() == n * n);
248 const float root = math::sqrt(static_cast<float>(n));
249 const T v = static_cast<T>(1.0 / static_cast<double>(root));
250 for (std::size_t i = 0; i < n; ++i)
251 for (std::size_t j = 0; j < n; ++j)
252 m[i * n + j] = hadamard_entry(i, j) < 0 ? -v : v;
253}
