74 // sigmoid(v) and sigmoid(-v) = 1 - sigmoid(v), both from one e^-|v|, so neither cancels.
75 template <typename T>
76 struct sigmoid_pair
77 {
78 T s;
79 T one_minus_s;
80 };
81
82 template <typename T>
83 constexpr sigmoid_pair<T> sigmoid_both(T v)
84 {
85 const T e = math::exp(-math::abs(v));
86 const T r = T(1) / (T(1) + e);
87 if (v >= T(0))
88 return {r, e * r};
89 return {e * r, r};
90 }
91} // namespace detail::activations

97template <math::math_argument X>
98constexpr math::real_t<X> sigmoid(X x)
99{
100 using T = math::real_t<X>;
101 const T v = static_cast<T>(x);
102 if (v >= T(0))
103 return T(1) / (T(1) + math::exp(-v));
104 const T e = math::exp(v);
105 return e / (T(1) + e);
106}

135constexpr math::real_t<X> softplus(X x)
136{
137 using T = math::real_t<X>;
138 const T v = static_cast<T>(x);
139 if (math::is_nan(v))
140 return v;

160template <math::math_argument X>
161constexpr math::real_t<X> logit(X p)
162{
163 using T = math::real_t<X>;
164 const T v = static_cast<T>(p);
165 if (!(v >= T(0) && v <= T(1))) // NaN, or outside [0, 1]: no arithmetic on a NaN, which a constant expression refuses
166 return std::numeric_limits<T>::quiet_NaN();
167 return math::log(v) - math::log1p(-v);

266template <math::math_argument X>
267constexpr math::real_t<X> gelu_derivative(X x)
268{
269 using T = math::real_t<X>;
270 using L = math::detail::limits<T>;
271 const T v = static_cast<T>(x);
272 const T cdf = T(0.5) * math::erfc(-v * T(L::inv_sqrt2));
273 const T pdf = math::exp(T(-0.5) * v * v) / T(L::sqrt_2pi);
274 return cdf + v * pdf;

279 // u = sqrt(2/pi) (x + 0.044715 x^3), GPT-2's approximation of GELU; returns 2u.
280 template <typename T>
281 constexpr T gelu_tanh_2u(T v)
282 {
283 using L = math::detail::limits<T>;
284 const T sqrt_2_over_pi = T(2) / T(L::sqrt_2pi);
285 return T(2) * sqrt_2_over_pi * (v + T(0.044715) * v * v * v);
286 }

305template <math::math_argument X>
306constexpr math::real_t<X> gelu_tanh_derivative(X x)
307{
308 using T = math::real_t<X>;
309 using L = math::detail::limits<T>;
310 const T v = static_cast<T>(x);
311 const auto [s, one_minus_s] = detail::activations::sigmoid_both(detail::activations::gelu_tanh_2u(v));
312 const T du = T(2) / T(L::sqrt_2pi) * (T(1) + T(3) * T(0.044715) * v * v);
313 return s + T(2) * v * s * one_minus_s * du;
314}

472template <math::math_number T>
473struct xielu_parameters
474{
475 T alpha_p = T(0.8);
476 T alpha_n = T(0.8);
477 T beta = T(0.5);
478 T eps = T(-1e-6);
479};
480
486template <math::math_argument X>
487constexpr math::real_t<X> xielu(X x, const xielu_parameters<math::real_t<X>>& p)
488{
489 using T = math::real_t<X>;
490 const T v = static_cast<T>(x);
491 if (v > T(0))
492 return p.alpha_p * v * v + p.beta * v;
493 const T m = v < p.eps ? v : p.eps;
494 return (math::expm1(m) - v) * p.alpha_n + p.beta * v;
495}

502template <math::math_argument X>
503constexpr math::real_t<X> xielu_derivative(X x, const xielu_parameters<math::real_t<X>>& p)
504{
505 using T = math::real_t<X>;
506 const T v = static_cast<T>(x);
507 if (v > T(0))
508 return T(2) * p.alpha_p * v + p.beta;
509 const T de = v < p.eps ? math::exp(v) : T(0);
510 return (de - T(1)) * p.alpha_n + p.beta;
511}
