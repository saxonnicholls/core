109 [[nodiscard]] constexpr digest<32> personal_sign_hash(std::string_view message) noexcept
110 {
111 std::array<char, 20> digits{};
112 std::size_t n = digits.size();
113 std::size_t length = message.size();
114 do {
115 digits[--n] = static_cast<char>('0' + length % 10);
116 length /= 10;
117 } while (length != 0);
118 keccak256 h;
119 h.update(std::string_view{"\x19" "Ethereum Signed Message:\n"});
120 h.update(std::string_view{digits.data() + n, digits.size() - n});
121 h.update(message);
122 return h.finish();
123 }
