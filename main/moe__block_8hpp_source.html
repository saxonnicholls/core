117template <transformer_config C, typename B, math::math_number T, typename R>
118 requires linalg::blas_backend<B, T> && (C.experts > 0) && (C.rope == rope_kind::none || rope_source<R>)
119constexpr moe_losses<T> block_forward_sequence(const B& backend, const moe_block_weights<T>& w, std::type_identity_t<std::span<T>> xs,
120 const R& rope_src, const moe_block_activations<T>& act, std::size_t sequences = 1)
121{
122 constexpr std::size_t d = C.d_model;
123 [[maybe_unused]] const std::size_t n = xs.size() / d;
124 assert(xs.size() == n * d && act.x0.size() == n * d && act.h1.size() == n * d && act.x1.size() == n * d && act.h2.size() == n * d);
125 copy_rows(backend, std::span<const T>(xs), act.x0);
126 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm1_weight, w.norm1_bias, act.h1);
127 attention_forward_sequence<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, xs, sequences);
128 add_rows(backend, std::span<const T>(act.x0), xs);
129 copy_rows(backend, std::span<const T>(xs), act.x1);
130 block_norm_rows<C>(backend, std::span<const T>(xs), w.norm2_weight, w.norm2_bias, act.h2);
131 const auto losses = moe_forward<C>(backend, w.moe, std::span<const T>(act.h2), xs, act.moe, sequences);
132 add_rows(backend, std::span<const T>(act.x1), xs);
133 return losses;
134}

147template <transformer_config C, typename B, math::math_number T, typename R>
148 requires linalg::blas_backend<B, T> && (C.experts > 0) && (C.rope == rope_kind::none || rope_source<R>)
149constexpr void block_backward(const B& backend, const moe_block_weights<T>& w, const R& rope_src, const moe_block_activations<T>& act,
150 std::type_identity_t<std::span<T>> dxs, const moe_block_gradients<T>& grads, std::size_t sequences = 1)
151{
152 constexpr std::size_t d = C.d_model;
153 [[maybe_unused]] const std::size_t n = dxs.size() / d;
154 assert(dxs.size() == n * d && act.x0.size() == n * d && grads.scratch.size() == n * d);
155 fill_rows(backend, grads.scratch, T(0));
156 moe_backward<C>(backend, w.moe, std::span<const T>(act.h2), act.moe, std::span<const T>(dxs), grads.scratch, grads.moe, sequences);
157 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x1), w.norm2_weight, w.norm2_bias,
158 dxs, grads.norm2_weight, grads.norm2_bias);
159 fill_rows(backend, grads.scratch, T(0));
160 attention_backward<C>(backend, w.attention, std::span<const T>(act.h1), rope_src, act.attention, std::span<const T>(dxs), grads.scratch,
161 grads.attention, sequences);
162 block_norm_rows_backward_add<C>(backend, std::span<const T>(grads.scratch), std::span<const T>(act.x0), w.norm1_weight, w.norm1_bias,
163 dxs, grads.norm1_weight, grads.norm1_bias);
164}
