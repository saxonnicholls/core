101inline constexpr std::size_t nf4_block_size = 64;
103inline constexpr std::size_t nf4_group_blocks = 256;
104
106constexpr std::size_t nf4_code_bytes(std::size_t n) noexcept { return n / 2; }
108constexpr std::size_t nf4_block_count(std::size_t n) noexcept { return n / nf4_block_size; }
110constexpr std::size_t nf4_group_count(std::size_t n) noexcept { return (nf4_block_count(n) + nf4_group_blocks - 1) / nf4_group_blocks; }
111
114constexpr std::uint8_t nf4_code(float x) noexcept
115{
116 std::uint8_t best = 0;
117 float best_d = x - nf4_values[0] < 0 ? nf4_values[0] - x : x - nf4_values[0];
118 for (std::uint8_t c = 1; c < 16; ++c) {
119 const float d = x - nf4_values[c] < 0 ? nf4_values[c] - x : x - nf4_values[c];
120 if (d < best_d) {
121 best_d = d;
122 best = c;
123 }
124 }
125 return best;
126}

130struct nf4_weights
131{
132 std::span<const std::uint8_t> codes{};
133 std::span<const float8_e4m3> block_scales{};
134 std::span<const float> group_scales{};
135 float scale_mean = 0.0f;
136 std::size_t rows = 0;
137 std::size_t cols = 0;
138
140 [[nodiscard]] constexpr float block_scale(std::size_t block) const noexcept
141 {
142 return decode_table<float8_e4m3>[block_scales[block].bits] * group_scales[block / nf4_group_blocks] + scale_mean;
143 }
144
146 [[nodiscard]] constexpr std::uint8_t code(std::size_t index) const noexcept
147 {
148 const std::uint8_t byte = codes[index / 2];
149 return static_cast<std::uint8_t>(index % 2 == 0 ? byte >> 4 : byte & 0x0f);
150 }
151
153 [[nodiscard]] constexpr float at(std::size_t i, std::size_t j) const noexcept
154 {
155 const std::size_t index = i * cols + j;
156 return nf4_values[code(index)] * block_scale(index / nf4_block_size);
157 }
158};

164template <row W>
165 requires std::same_as<row_value_t<W>, float>
166constexpr float nf4_quantize(const W& w, std::span<std::uint8_t> codes, std::span<float8_e4m3> block_scales,
167 std::span<float> group_scales) noexcept
168{
169 const auto x = as_row(w);
170 const std::size_t n = x.size(), blocks = nf4_block_count(n);
171 assert(n % nf4_block_size == 0 && codes.size() == nf4_code_bytes(n) && block_scales.size() == blocks
172 && group_scales.size() == nf4_group_count(n));
173 const auto absmax = [&](std::size_t b) {
174 float m = 0.0f;
175 for (std::size_t j = b * nf4_block_size; j < (b + 1) * nf4_block_size; ++j)
176 m = (x[j] < 0 ? -x[j] : x[j]) > m ? (x[j] < 0 ? -x[j] : x[j]) : m;
177 return m;
178 };
179 double sum = 0.0;
180 for (std::size_t b = 0; b < blocks; ++b) {
181 const float c2 = absmax(b);
182 sum += static_cast<double>(c2);
183 for (std::size_t j = b * nf4_block_size; j < (b + 1) * nf4_block_size; j += 2) {
184 const auto hi = c2 == 0.0f ? std::uint8_t{7} : nf4_code(x[j] / c2);
185 const auto lo = c2 == 0.0f ? std::uint8_t{7} : nf4_code(x[j + 1] / c2);
186 codes[j / 2] = static_cast<std::uint8_t>((hi << 4) | lo);
187 }
188 }
189 const float mean = blocks == 0 ? 0.0f : static_cast<float>(sum / static_cast<double>(blocks));
190 for (std::size_t g = 0; g < group_scales.size(); ++g) {
191 const std::size_t end = (g + 1) * nf4_group_blocks < blocks ? (g + 1) * nf4_group_blocks : blocks;
192 float m = 0.0f;
193 for (std::size_t b = g * nf4_group_blocks; b < end; ++b) {
194 const float d = absmax(b) - mean;
195 m = (d < 0 ? -d : d) > m ? (d < 0 ? -d : d) : m;
196 }
197 const float s1 = m / 448.0f;
198 group_scales[g] = s1;
199 for (std::size_t b = g * nf4_group_blocks; b < end; ++b)
200 block_scales[b] = s1 == 0.0f ? float8_e4m3::from_bits(0) : to_minifloat<float8_e4m3>(static_cast<double>((absmax(b) - mean) / s1));
201 }
202 return mean;
203}

206constexpr void nf4_dequantize(const nf4_weights& w, std::span<float> out) noexcept
207{
208 assert(out.size() == w.rows * w.cols);
209 for (std::size_t i = 0; i < w.rows; ++i)
210 for (std::size_t j = 0; j < w.cols; ++j)
211 out[i * w.cols + j] = w.at(i, j);
212}

227template <row X, mutable_row Y, row B = std::span<const float>>
228 requires std::same_as<row_value_t<X>, float> && std::same_as<row_value_t<Y>, float> && std::same_as<row_value_t<B>, float>
229constexpr void linear(const nf4_weights& w, const X& x, Y&& y, const B& bias = B{}) noexcept
230{
231 const auto in = as_row(x);
232 const auto out = as_mutable_row(y);
233 const auto b = as_row(bias);
234 assert(in.size() == w.cols && out.size() == w.rows && w.cols % nf4_block_size == 0 && (b.empty() || b.size() == w.rows));
235 for (std::size_t i = 0; i < w.rows; ++i) {
236 float acc = b.empty() ? 0.0f : b[i];
237 for (std::size_t j0 = 0; j0 < w.cols; j0 += nf4_block_size) {
238 const std::size_t base = i * w.cols + j0;
239 const float scale = w.block_scale(base / nf4_block_size);
240 for (std::size_t j = 0; j < nf4_block_size; ++j) {
241 const float wij = nf4_values[w.code(base + j)] * scale;
242 const float p = wij * in[j0 + j];
243 acc += p;
244 }
245 }
246 out[i] = acc;
247 }
248}

252constexpr void linear_backward_input(const nf4_weights& w, std::span<const float> dy, std::span<float> dx) noexcept
253{
254 assert(dy.size() == w.rows && dx.size() == w.cols);
255 for (std::size_t i = 0; i < w.rows; ++i)
256 for (std::size_t j = 0; j < w.cols; ++j) {
257 const float p = w.at(i, j) * dy[i];
258 dx[j] += p;
259 }
260}

264template <typename B>
265constexpr void linear_sequence(const B&, const nf4_weights& w, std::span<const float> xs, std::span<float> ys, std::span<const float> bias = {}) noexcept
266{
267 assert(w.cols > 0 && xs.size() % w.cols == 0 && ys.size() == xs.size() / w.cols * w.rows);
268 for (std::size_t t = 0; t < xs.size() / w.cols; ++t)
269 linear(w, xs.subspan(t * w.cols, w.cols), ys.subspan(t * w.rows, w.rows), bias);
270}
