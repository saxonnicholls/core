121 template <transformer_config C, typename T>
122 struct scan
123 {
124 using value_type = T;
125 static constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups;
126 static constexpr std::size_t state_size = H * N * P, work_size = N * P;
127
128 ssd_inputs<T> in{};
129 std::span<T> y{};
130 std::span<const T> dy{};
131 ssd_gradients<T> g{};
132
133 constexpr void update(std::size_t row, std::span<T> state, bool output) const
134 {
135 for (std::size_t h = 0; h < H; ++h) {
136 const T dt = in.dt[row * H + h], alpha = math::exp(dt * in.a[h]);
137 const std::size_t grp = h / (H / G);
138 const auto x = in.x.subspan((row * H + h) * P, P);
139 const auto b = in.b.subspan((row * G + grp) * N, N), c = in.c.subspan((row * G + grp) * N, N);
140 const auto s = state.subspan(h * N * P, N * P);
141 std::array<T, P> xdt{}, out{};
142 for (std::size_t p = 0; p < P; ++p)
143 xdt[p] = x[p] * dt;
144 for (std::size_t n = 0; n < N; ++n)
145 for (std::size_t p = 0; p < P; ++p) {
146 s[n * P + p] = s[n * P + p] * alpha + b[n] * xdt[p];
147 out[p] += s[n * P + p] * c[n];
148 }
149 if (output)
150 for (std::size_t p = 0; p < P; ++p)
151 y[(row * H + h) * P + p] = out[p];
152 }
153 }
154 constexpr void step(std::size_t row, std::span<T> state) const { update(row, state, true); }
155 constexpr void advance(std::size_t row, std::span<T> state) const { update(row, state, false); }

157 constexpr void step_backward(std::size_t row, std::span<const T> prev, std::span<T> dstate, std::span<T> work) const
158 {
159 for (std::size_t i = 0; i < G * N; ++i)
160 g.b[row * G * N + i] = g.c[row * G * N + i] = T(0);
161 for (std::size_t h = 0; h < H; ++h) {
162 const T dt = in.dt[row * H + h], A = in.a[h], alpha = math::exp(dt * A);
163 const std::size_t grp = h / (H / G);
164 const auto x = in.x.subspan((row * H + h) * P, P);
165 const auto b = in.b.subspan((row * G + grp) * N, N), c = in.c.subspan((row * G + grp) * N, N);
166 const auto dyh = dy.subspan((row * H + h) * P, P);
167 const auto s0 = prev.subspan(h * N * P, N * P);
168 const auto ds = dstate.subspan(h * N * P, N * P);
169 const auto db = g.b.subspan((row * G + grp) * N, N), dc = g.c.subspan((row * G + grp) * N, N);
170 T dalpha = T(0);
171 std::array<T, P> dxdt{};
172 for (std::size_t n = 0; n < N; ++n) {
173 T dcn = T(0), dbn = T(0);
174 for (std::size_t p = 0; p < P; ++p) {
175 const T st = s0[n * P + p] * alpha + b[n] * (x[p] * dt);
176 dcn += dyh[p] * st;
177 const T d = ds[n * P + p] + c[n] * dyh[p];
178 dbn += d * (x[p] * dt);
179 dxdt[p] += b[n] * d;
180 dalpha += d * s0[n * P + p];
181 ds[n * P + p] = d * alpha;
182 }
183 dc[n] += dcn;
184 db[n] += dbn;
185 }
186 T ddt = T(0);
187 for (std::size_t p = 0; p < P; ++p) {
188 g.x[(row * H + h) * P + p] = dxdt[p] * dt;
189 ddt += dxdt[p] * x[p];
190 }
191 g.dt[row * H + h] = ddt + dalpha * alpha * A;
192 g.a[h] += dalpha * alpha * dt;
193 }
194 static_cast<void>(work);
195 }
196 };

198 template <transformer_config C, typename T>
199 constexpr void check(const ssd_inputs<T>& in, std::size_t rows)
200 {
201 [[maybe_unused]] constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups;
202 static_assert(valid(C) && C.mixer_heads > 0, "snicholls::ml::ssd: the configuration has no mixer shape");
203 static_cast<void>(H), static_cast<void>(P), static_cast<void>(N), static_cast<void>(G); // read only by the assert (MSVC, C4189)
204 assert(in.x.size() == rows * H * P && in.dt.size() == rows * H && in.a.size() == H && in.b.size() == rows * G * N
205 && in.c.size() == rows * G * N);
206 static_cast<void>(rows);
207 static_cast<void>(in);
208 }

219template <transformer_config C, typename B, math::math_number T>
220constexpr void ssd_forward(const B&, const ssd_inputs<T>& in, std::type_identity_t<std::span<T>> y, std::size_t sequences,
221 std::type_identity_t<std::span<T>> states, std::type_identity_t<std::span<T>> checkpoints = {})
222{
223 const std::size_t rows = in.dt.size() / C.mixer_heads;
224 detail::ssd::check<C>(in, rows);
225 assert(sequences > 0 && rows % sequences == 0 && y.size() == in.x.size());
226 recurrence_forward(detail::ssd::scan<C, T>{in, y, {}, {}}, rows / sequences, sequences, C.mixer_chunk, states, checkpoints);
227}

234template <transformer_config C, typename B, math::math_number T>
235constexpr void ssd_forward_chunked(const B&, const ssd_inputs<T>& in, std::type_identity_t<std::span<T>> y, std::size_t sequences,
236 std::type_identity_t<std::span<T>> states, std::type_identity_t<std::span<T>> checkpoints = {})
237{
238 constexpr std::size_t H = C.mixer_heads, P = C.mixer_head_dim, N = C.mixer_state, G = C.mixer_groups, L = C.mixer_chunk;
239 constexpr std::size_t S = H * N * P;
240 const std::size_t rows = in.dt.size() / H;
241 detail::ssd::check<C>(in, rows);
242 assert(sequences > 0 && rows % sequences == 0 && y.size() == in.x.size() && states.size() == sequences * S);
243 const std::size_t n = rows / sequences, m = recurrence_checkpoints(n, L);
244 assert(checkpoints.empty() || checkpoints.size() == sequences * m * S);
245 for (std::size_t sq = 0; sq < sequences; ++sq)
246 for (std::size_t ck = 0; ck < m; ++ck) {
247 const std::size_t first = sq * n + ck * L, len = n - ck * L < L ? n - ck * L : L;
248 const auto state = states.subspan(sq * S, S);
249 if (!checkpoints.empty())
250 for (std::size_t i = 0; i < S; ++i)
251 checkpoints[(sq * m + ck) * S + i] = state[i];
252 for (std::size_t h = 0; h < H; ++h) {
253 const std::size_t grp = h / (H / G);
254 const auto s = state.subspan(h * N * P, N * P);
255 std::array<T, L> la{}; // the cumulative log-decay
256 for (std::size_t j = 0; j < len; ++j)
257 la[j] = (j == 0 ? T(0) : la[j - 1]) + in.dt[(first + j) * H + h] * in.a[h];
258 for (std::size_t j = 0; j < len; ++j) {
259 const auto cj = in.c.subspan(((first + j) * G + grp) * N, N);
260 const auto yj = y.subspan(((first + j) * H + h) * P, P);
261 for (std::size_t p = 0; p < P; ++p)
262 yj[p] = T(0);
263 for (std::size_t i = 0; i <= j; ++i) { // the diagonal block
264 const auto bi = in.b.subspan(((first + i) * G + grp) * N, N);
265 T cb = T(0);
266 for (std::size_t q = 0; q < N; ++q)
267 cb += cj[q] * bi[q];
268 const T w = cb * math::exp(la[j] - la[i]) * in.dt[(first + i) * H + h];
269 const auto xi = in.x.subspan(((first + i) * H + h) * P, P);
270 for (std::size_t p = 0; p < P; ++p)
271 yj[p] += w * xi[p];
272 }
273 const T decay = math::exp(la[j]); // the state carried in
274 for (std::size_t p = 0; p < P; ++p) {
275 T cs = T(0);
276 for (std::size_t q = 0; q < N; ++q)
277 cs += cj[q] * s[q * P + p];
278 yj[p] += decay * cs;
279 }
280 }
281 const T total = math::exp(la[len - 1]); // the state passed on
282 for (std::size_t q = 0; q < N; ++q)
283 for (std::size_t p = 0; p < P; ++p) {
284 T add = T(0);
285 for (std::size_t i = 0; i < len; ++i)
286 add += math::exp(la[len - 1] - la[i]) * in.dt[(first + i) * H + h] * in.b[((first + i) * G + grp) * N + q]
287 * in.x[((first + i) * H + h) * P + p];
288 s[q * P + p] = total * s[q * P + p] + add;
289 }
290 }
291 }
