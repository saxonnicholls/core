62 // The raw signature of type_name<T>(), with T spelled out by the compiler.
63 template <typename T>
64 constexpr std::string_view raw_type_signature() noexcept
65 {
66#if defined(__clang__) || defined(__GNUC__)
67 return __PRETTY_FUNCTION__;
68#elif defined(_MSC_VER)
69 return __FUNCSIG__;
70#else
71 return "?";
72#endif
73 }

75 constexpr bool is_identifier_character(char c) noexcept
76 {
77 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
78 }

80 // How much to drop at position i: a class-key MSVC writes or a library's inline namespace, as a
81 // whole word only, so that foo::my_struct keeps its name and std:: stays on std::__cxx11::string.
82 constexpr std::size_t type_name_noise_at(std::string_view raw, std::size_t i) noexcept
83 {
84 if (i > 0 && is_identifier_character(raw[i - 1]))
85 return 0;
86 constexpr std::string_view noise[] = {"struct ", "class ", "enum ", "union ", "__cxx11::", "__1::"};
87 for (const std::string_view token : noise)
88 if (raw.substr(i, token.size()) == token)
89 return token.size();
90 return 0;
91 }

93 // The part of the signature that is T.
94 template <typename T>
95 constexpr std::string_view cut_type_name() noexcept
96 {
97 constexpr std::string_view sig = raw_type_signature<T>();
98#if defined(__clang__)
99 constexpr std::string_view head = "[T = ";
100 constexpr std::string_view tail = "]";
101#elif defined(__GNUC__)
102 constexpr std::string_view head = "[with T = ";
103 constexpr std::string_view tail = ";";
104#elif defined(_MSC_VER)
105 constexpr std::string_view head = "raw_type_signature<";
106 constexpr std::string_view tail = ">(void)";
107#else
108 return sig;
109#endif
110 const std::size_t start = sig.find(head);
111 if (start == std::string_view::npos)
112 return sig;
113 const std::size_t from = start + head.size();
114 std::size_t end = sig.find(tail, from);
115 if (end == std::string_view::npos)
116 end = sig.size();
117 while (end > from && sig[end - 1] == ' ') // MSVC writes "> >" and leaves one before the tail
118 --end;
119 return sig.substr(from, end - from);
120 }

122 constexpr std::size_t normalised_size(std::string_view raw) noexcept
123 {
124 std::size_t size = 0;
125 for (std::size_t i = 0; i < raw.size();) {
126 const std::size_t noise = type_name_noise_at(raw, i);
127 size += noise == 0 ? 1 : 0;
128 i += noise == 0 ? 1 : noise;
129 }
130 return size;
131 }

133 // The normalised name, exactly as long as it needs to be: no limit, and no bytes wasted per type.
134 template <typename T>
135 constexpr auto make_type_name_storage() noexcept
136 {
137 constexpr std::string_view raw = cut_type_name<T>();
138 std::array<char, normalised_size(raw) + 1> out{};
139 std::size_t size = 0;
140 for (std::size_t i = 0; i < raw.size();) {
141 const std::size_t noise = type_name_noise_at(raw, i);
142 if (noise == 0)
143 out[size++] = raw[i];
144 i += noise == 0 ? 1 : noise;
145 }
146 return out;
147 }
