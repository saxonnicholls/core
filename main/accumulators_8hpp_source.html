135 template <std::floating_point T>
136 struct compensated_sum {
137 T total{};
138 T compensation{};
139
142 constexpr void add(T x) noexcept
143 {
144 // Neumaier: the low part lost when the smaller of the two is added to the larger
145 const T t = total + x;
146 const T abs_total = total < 0 ? -total : total;
147 const T abs_x = x < 0 ? -x : x;
148 compensation += abs_total >= abs_x ? (total - t) + x : (x - t) + total;
149 total = t;
150 }
151
153 constexpr void merge(const compensated_sum& other) noexcept
154 {
155 add(other.total);
156 compensation += other.compensation;
157 }
158
160 [[nodiscard]] constexpr T value() const noexcept { return total + compensation; }
161 };

219 struct mean {
220 std::uint64_t n = 0;
221 T m{};
222
225 constexpr void add(T x) noexcept
226 {
227 ++n;
228 m += (x - m) / static_cast<T>(n);

230
232 constexpr void merge(const mean& other) noexcept
233 {
234 if (other.n == 0)

248 template <std::floating_point T>
249 struct variance {
250 std::uint64_t n = 0;
251 T m{};
252 T m2{};
253
256 constexpr void add(T x) noexcept
257 {
258 ++n;
259 const T delta = x - m;
260 m += delta / static_cast<T>(n);
261 m2 += delta * (x - m);
262 }
263
265 constexpr void merge(const variance& other) noexcept
266 {
267 if (other.n == 0)
268 return;
269 const std::uint64_t total = n + other.n;
270 const T delta = other.m - m;
271 m += delta * static_cast<T>(other.n) / static_cast<T>(total);
272 m2 += other.m2 + delta * delta * static_cast<T>(n) * static_cast<T>(other.n) / static_cast<T>(total);
273 n = total;
274 }
275
277 [[nodiscard]] constexpr T value() const noexcept { return n < 2 ? std::numeric_limits<T>::quiet_NaN() : m2 / static_cast<T>(n - 1); }
280 [[nodiscard]] constexpr T mean() const noexcept { return n == 0 ? std::numeric_limits<T>::quiet_NaN() : m; }
283 [[nodiscard]] T standard_deviation() const noexcept { return std::sqrt(value()); }
286 [[nodiscard]] constexpr T population_variance() const noexcept { return n == 0 ? std::numeric_limits<T>::quiet_NaN() : m2 / static_cast<T>(n); }
287 };

292 template <std::floating_point T>
293 struct covariance {
294 std::uint64_t n = 0;
295 T mx{};
296 T my{};
297 T cxy{};
298 T m2x{};
299 T m2y{};
300
304 constexpr void add(T x, T y) noexcept
305 {
306 ++n;
307 const T dx = x - mx;
308 mx += dx / static_cast<T>(n);
309 const T dy = y - my;
310 my += dy / static_cast<T>(n);
311 cxy += dx * (y - my);
312 m2x += dx * (x - mx);
313 m2y += dy * (y - my);
314 }
315
318 template <typename P>
319 requires requires(const P& q) {
320 T(q.first);
321 T(q.second);
322 }
323 constexpr void add(const P& p) noexcept
324 {
325 add(static_cast<T>(p.first), static_cast<T>(p.second));
326 }
327
329 constexpr void merge(const covariance& other) noexcept
330 {
331 if (other.n == 0)
332 return;
333 const std::uint64_t total = n + other.n;
334 const T dx = other.mx - mx, dy = other.my - my;
335 const T w = static_cast<T>(n) * static_cast<T>(other.n) / static_cast<T>(total);
336 cxy += other.cxy + dx * dy * w;
337 m2x += other.m2x + dx * dx * w;
338 m2y += other.m2y + dy * dy * w;
339 mx += dx * static_cast<T>(other.n) / static_cast<T>(total);
340 my += dy * static_cast<T>(other.n) / static_cast<T>(total);
341 n = total;
342 }
343
345 [[nodiscard]] constexpr T value() const noexcept { return n < 2 ? std::numeric_limits<T>::quiet_NaN() : cxy / static_cast<T>(n - 1); }
348 [[nodiscard]] T correlation() const noexcept
349 {
350 return (n < 2 || m2x <= 0 || m2y <= 0) ? std::numeric_limits<T>::quiet_NaN() : cxy / std::sqrt(m2x * m2y);
351 }
352
354 [[nodiscard]] constexpr T beta() const noexcept { return (n < 2 || m2x <= 0) ? std::numeric_limits<T>::quiet_NaN() : cxy / m2x; }
355 };

360 template <std::floating_point T>
361 struct ewma {
362 T alpha = T(0.06);
363 T m{};
364 T var{};
365 bool started = false;
366
369 constexpr void add(T x) noexcept
370 {
371 if (!started) {
372 m = x;
373 started = true;
374 return;
375 }
376 const T delta = x - m;
377 m += alpha * delta;
378 var = (1 - alpha) * (var + alpha * delta * delta);
379 }
380
382 [[nodiscard]] constexpr T value() const noexcept { return started ? m : std::numeric_limits<T>::quiet_NaN(); }
385 [[nodiscard]] constexpr T variance() const noexcept { return started ? var : std::numeric_limits<T>::quiet_NaN(); }
388 [[nodiscard]] T volatility() const noexcept { return std::sqrt(variance()); }
389 };

395 template <std::floating_point T, std::size_t Bins>
396 struct histogram {
397 static_assert(Bins >= 1, "snicholls::stats::histogram: at least one bin");
398 T lo = 0;
399 T hi = 1;
400 std::array<std::uint64_t, Bins> bins{};
401 std::uint64_t below = 0;
402 std::uint64_t above = 0;
403
406 constexpr void add(T x) noexcept
407 {
408 if (!(x >= lo)) {
409 ++below;
410 return;
411 }
412 if (x >= hi) {
413 ++above;
414 return;
415 }
416 auto i = static_cast<std::size_t>((x - lo) / (hi - lo) * static_cast<T>(Bins));
417 ++bins[i < Bins ? i : Bins - 1];
418 }
419
421 constexpr void merge(const histogram& other) noexcept
422 {
423 for (std::size_t i = 0; i < Bins; ++i)
424 bins[i] += other.bins[i];
425 below += other.below;
426 above += other.above;
427 }
428
430 [[nodiscard]] constexpr const std::array<std::uint64_t, Bins>& value() const noexcept { return bins; }
434 [[nodiscard]] constexpr T edge(std::size_t i) const noexcept { return lo + (hi - lo) * static_cast<T>(i) / static_cast<T>(Bins); }
435 };

466 template <typename Acc, std::size_t B = 200>
467 struct bootstrapped {
468 static_assert(B >= 2, "snicholls::stats::bootstrapped: at least two replicates");
469 Acc estimate{};
470 std::array<Acc, B> replicates{};
471 wyrand rng{0x5eed};
472
475 constexpr void reseed(std::uint64_t seed) noexcept { rng = wyrand(seed); }
476
480 template <typename T>
481 constexpr void add(const T& x)
482 {
483 estimate.add(x);
484 for (std::size_t b = 0; b < B; b += 2) {
485 const std::uint64_t bits = rng();
486 for (std::uint32_t k = detail::poisson1_from(static_cast<std::uint32_t>(bits)); k > 0; --k)
487 replicates[b].add(x);
488 if (b + 1 < B)
489 for (std::uint32_t k = detail::poisson1_from(static_cast<std::uint32_t>(bits >> 32)); k > 0; --k)
490 replicates[b + 1].add(x);
491 }
492 }
493
496 constexpr void merge(const bootstrapped& other)
497 requires mergeable<Acc>
498 {
499 estimate.merge(other.estimate);
500 for (std::size_t b = 0; b < B; ++b)
501 replicates[b].merge(other.replicates[b]);
502 }
503
507 [[nodiscard]] bootstrap_distribution value() const
508 {
509 bootstrap_distribution out;
510 out.estimate = static_cast<double>(estimate.value());
511 out.replicates.reserve(B);
512 for (const Acc& r : replicates)
513 out.replicates.push_back(static_cast<double>(r.value()));
514 return out;
515 }
516 };
