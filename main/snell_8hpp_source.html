122 template <std::size_t States, std::size_t Horizon, typename Payoff>
123 [[nodiscard]] constexpr fixed_snell<States, Horizon> snell_envelope(const std::array<std::array<double, States>, States>& p, Payoff&& g,
124 double discount = 1.0)
125 {
126 fixed_snell<States, Horizon> out{};
127 for (std::size_t s = 0; s < States; ++s) {
128 out.payoff[Horizon][s] = std::invoke(g, Horizon, s);
129 out.value[Horizon][s] = out.payoff[Horizon][s];
130 out.continuation[Horizon][s] = -std::numeric_limits<double>::infinity();
131 }
132 for (std::size_t t = Horizon; t-- > 0;)
133 for (std::size_t s = 0; s < States; ++s) {
134 double expected = 0;
135 for (std::size_t j = 0; j < States; ++j)
136 expected += p[s][j] * out.value[t + 1][j];
137 out.continuation[t][s] = discount * expected;
138 out.payoff[t][s] = std::invoke(g, t, s);
139 out.value[t][s] = out.payoff[t][s] > out.continuation[t][s] ? out.payoff[t][s] : out.continuation[t][s];
140 }
141 return out;
142 }

189 template <typename Payoff>
190 [[nodiscard]] snell snell_envelope(const markov_chain& chain, std::size_t horizon, Payoff&& g, double discount = 1.0)
191 {
192 const std::size_t n = chain.size();
193 snell out;
194 out.states = n;
195 out.horizon = horizon;
196 out.value.assign((horizon + 1) * n, 0.0);
197 out.continuation.assign((horizon + 1) * n, -std::numeric_limits<double>::infinity());
198 out.payoff.assign((horizon + 1) * n, 0.0);
199 for (std::size_t s = 0; s < n; ++s) {
200 out.payoff[horizon * n + s] = std::invoke(g, horizon, s);
201 out.value[horizon * n + s] = out.payoff[horizon * n + s];
202 }
203 for (std::size_t t = horizon; t-- > 0;) {
204 const std::span<const double> next(out.value.data() + (t + 1) * n, n);
205 for (std::size_t s = 0; s < n; ++s) {
206 const double c = discount * chain.expect(s, next);
207 const double stop = std::invoke(g, t, s);
208 out.continuation[t * n + s] = c;
209 out.payoff[t * n + s] = stop;
210 out.value[t * n + s] = stop > c ? stop : c;
211 }
212 }
213 return out;
214 }

250 template <std::size_t Steps, typename T, std::size_t Branches, typename Payoff>
251 [[nodiscard]] constexpr lattice_snell<Steps, Branches, T> lattice_envelope(const std::array<T, Branches>& stencil, Payoff&& g,
252 std::type_identity_t<T> discount = T(1))
253 {
254 constexpr std::size_t width = Branches - 1;
255 lattice_snell<Steps, Branches, T> out{};
256 std::array<T, width * Steps + 1> row{};
257 for (std::size_t j = 0; j <= width * Steps; ++j)
258 row[j] = static_cast<T>(std::invoke(g, Steps, j));
259 for (std::size_t t = Steps + 1; t-- > 0;) {
260 if (t < Steps) {
261 const auto node = [&](std::size_t j) { // in place: node j reads j .. j + width, none yet overwritten but j
262 T expected = 0;
263 for (std::size_t k = 0; k < Branches; ++k)
264 expected += stencil[k] * row[j + k];
265 const T continuation = discount * expected, stop = static_cast<T>(std::invoke(g, t, j));
266 row[j] = stop > continuation ? stop : continuation;
267 return stop > continuation; // stopping strictly beats continuing
268 };
269 std::size_t j = 0, first = out.no_node, last = out.no_node;
270 for (; j <= width * t && first == out.no_node; ++j) // up to the first exercise, then only the last is tracked
271 first = last = node(j) ? j : out.no_node;
272 for (; j <= width * t; ++j)
273 if (node(j))
274 last = j;
275 out.first_exercise[t] = first;
276 out.last_exercise[t] = last;
277 }
278 if (t < out.head_steps)
279 for (std::size_t j = 0; j <= width * t; ++j)
280 out.head[t][j] = row[j];
281 }
282 out.value = row[0];
283 return out;
284 }
