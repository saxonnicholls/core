204template <math::math_number T>
205constexpr complex<T> sqrt(const complex<T>& z) noexcept
206{
207 const T r = abs(z);
208 if (r == T(0))
209 return {T(0), T(0)};
210 const T a = math::sqrt((r + math::abs(z.re)) / T(2));
211 if (z.re >= T(0))
212 return {a, z.im / (T(2) * a)};
213 return {math::abs(z.im) / (T(2) * a), z.im < T(0) ? -a : a};
214}

224 // (cos, sin) of (pi / 2) r / n for 0 <= r <= n / 2: an angle of at most pi / 4.
225 template <typename C>
226 constexpr complex<C> first_octant(std::size_t r, std::size_t n) noexcept
227 {
228 const C phi = math::math_limits<C>::pi / C(2) * (C(static_cast<long long>(r)) / C(static_cast<long long>(n)));
229 if (2 * r == n) { // pi / 4: one value for both, so twiddle(n - k) is conj(twiddle(k)) exactly
230 const C c = math::cos(phi);
231 return {c, c};
232 }
233 return {math::cos(phi), math::sin(phi)};
234 }
235
236 // (cos, sin) of 2 pi k / n with k < n, the octant reduced in integers so the quarter turns are exact.
237 template <typename C>
238 constexpr complex<C> unit(std::size_t k, std::size_t n) noexcept
239 {
240 const std::size_t q = (4 * k) / n, r = 4 * k - q * n; // 2 pi k / n = (pi / 2) (q + r / n)
241 complex<C> c = 2 * r <= n ? first_octant<C>(r, n) : [&] {
242 const complex<C> s = first_octant<C>(n - r, n);
243 return complex<C>{s.im, s.re};
244 }();
245 for (std::size_t i = 0; i < q; ++i)
246 c = {-c.im, c.re};
247 return c;
248 }
249} // namespace detail::complex_

259template <math::math_number T>
260constexpr complex<T> twiddle(std::size_t k, std::size_t n) noexcept
261{
262 const complex<compute_t<T>> c = detail::complex_::unit<compute_t<T>>(k % n, n);
