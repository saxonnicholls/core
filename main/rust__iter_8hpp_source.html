99 // A forward iterator over a view with operator[] and size(), holding a copy of the (small) view.
100 template <class V>
101 class index_iterator {
102 public:
103 using value_type = std::remove_cvref_t<decltype(std::declval<const V&>()[0])>;
104 using difference_type = std::ptrdiff_t;
105 using iterator_concept = std::forward_iterator_tag;
106
107 constexpr index_iterator() noexcept = default;
108 constexpr index_iterator(const V& v, std::size_t i) noexcept : v_(v), i_(i) {}
109
110 constexpr decltype(auto) operator*() const noexcept { return v_[i_]; }
111 constexpr index_iterator& operator++() noexcept
112 {
113 ++i_;
114 return *this;
115 }
116 constexpr index_iterator operator++(int) noexcept
117 {
118 index_iterator t = *this;
119 ++i_;
120 return t;
121 }
122 friend constexpr bool operator==(const index_iterator& a, const index_iterator& b) noexcept { return a.i_ == b.i_; }
123
124 private:
125 V v_{};
126 std::size_t i_ = 0;
127 };

142 template <class T, std::size_t W = std::dynamic_extent>
143 class slice_view : public std::ranges::view_interface<slice_view<T, W>> {
144 public:
146 constexpr slice_view() noexcept = default;
153 constexpr slice_view(T* data, std::size_t size, std::size_t width, std::size_t step, std::size_t count) noexcept
154 : data_(data), size_(size), width_(width), step_(step), count_(count)
155 {
156 }
157
161 [[nodiscard]] constexpr std::span<T, W> operator[](std::size_t i) const noexcept
162 {
163 assert(i < count_);
164 const std::size_t at = i * step_;
165 return std::span<T, W>(data_ + at, std::min(width_, size_ - at));
166 }
167
169 [[nodiscard]] constexpr std::size_t size() const noexcept { return count_; }
172 [[nodiscard]] constexpr detail::index_iterator<slice_view> begin() const noexcept { return {*this, 0}; }
175 [[nodiscard]] constexpr detail::index_iterator<slice_view> end() const noexcept { return {*this, count_}; }
178 [[nodiscard]] constexpr std::span<T> remainder() const noexcept
179 {
180 const std::size_t used = count_ == 0 ? 0 : std::min(size_, (count_ - 1) * step_ + width_);
181 return step_ == width_ ? std::span<T>(data_ + used, size_ - used) : std::span<T>();
182 }
183
184 private:
185 T* data_ = nullptr;
186 std::size_t size_ = 0;
187 std::size_t width_ = 0;
188 std::size_t step_ = 1;
189 std::size_t count_ = 0;
190 };

205
209 [[nodiscard]] constexpr T& operator[](std::size_t i) const noexcept
210 {
211 assert(i < size());
212 return data_[i * step_];

214
216 [[nodiscard]] constexpr std::size_t size() const noexcept { return size_ == 0 ? 0 : (size_ - 1) / step_ + 1; }
219 [[nodiscard]] constexpr detail::index_iterator<step_view> begin() const noexcept { return {*this, 0}; }
222 [[nodiscard]] constexpr detail::index_iterator<step_view> end() const noexcept { return {*this, size()}; }
223
224 private:
225 T* data_ = nullptr;
226 std::size_t size_ = 0;
227 std::size_t step_ = 1;
228 };

263 struct dedup_by_key_fn {
264 template <fn::detail::std_array A, class K, class T = typename std::remove_cvref_t<A>::value_type,
265 std::size_t N = std::tuple_size_v<std::remove_cvref_t<A>>>
266 constexpr containers::inplace_vector<T, N> operator()(const A& xs, K&& key) const
267 {
268 containers::inplace_vector<T, N> out;
269 for (std::size_t i = 0; i < N; ++i)
270 if (i == 0 || !(std::invoke(key, xs[i]) == std::invoke(key, xs[i - 1]))) out.push_back(xs[i]);
271 return out;
272 }

321 // The best element by a three-way comparison: Last ties go to the later element (Rust's max), else the earlier.
322 template <bool Last>
323 struct best_by_fn {
324 template <std::ranges::input_range R, class C, class V = range_value_t<R>>
325 constexpr std::optional<V> operator()(R&& r, C&& cmp) const
326 {
327 std::optional<V> best;
328 for (auto&& x : r) {
329 if (!best.has_value()) {
330 best.emplace(x);
331 continue;
332 }
333 const auto c = std::invoke(cmp, std::as_const(x), std::as_const(*best));
334 if (Last ? c >= 0 : c < 0) *best = x;
335 }
336 return best;
337 }
338 };

340 template <bool Last>
341 struct best_by_key_fn {
342 template <std::ranges::input_range R, class K, class V = range_value_t<R>,
343 class Key = std::remove_cvref_t<std::invoke_result_t<K&, const V&>>>
344 constexpr std::optional<V> operator()(R&& r, K&& key) const
345 {
346 std::optional<V> best;
347 std::optional<Key> best_key;
348 for (auto&& x : r) {
349 Key k = std::invoke(key, std::as_const(x));
350 if (best.has_value()) {
351 const auto c = snicholls::synth_three_way(k, *best_key);
352 if (!(Last ? c >= 0 : c < 0)) continue;
353 }
354 best.emplace(x);
355 best_key.emplace(std::move(k));
356 }
357 return best;
358 }

396 struct try_for_each_fn {
397 template <std::ranges::input_range R, class F,
398 class M = std::remove_cvref_t<std::invoke_result_t<F&, std::ranges::range_reference_t<R>>>>
399 constexpr auto operator()(R&& r, F&& f) const -> decltype(detail::success(std::type_identity<M>{}))
400 {
401 for (auto&& x : r)
402 if (M m = std::invoke(f, std::forward<decltype(x)>(x)); !detail::succeeded(m)) return m;
403 return detail::success(std::type_identity<M>{});
404 }
