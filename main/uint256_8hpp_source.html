309 constexpr uint256& operator<<=(unsigned n) noexcept
310 {
311 if (n >= 256) {
312 limbs_ = {};
313 return *this;
314 }
315 const std::size_t limb_shift = n / 64;
316 const unsigned bit_shift = n % 64;
317 std::array<std::uint64_t, 4> out{};
318 for (std::size_t i = 0; i < 4; ++i) {
319 if (i < limb_shift)
320 continue;
321 const std::size_t from = i - limb_shift;
322 out[i] = limbs_[from] << bit_shift;
323 if (bit_shift != 0 && from > 0)
324 out[i] |= limbs_[from - 1] >> (64 - bit_shift);
325 }
326 limbs_ = out;
327 return *this;
328 }

333 constexpr uint256& operator>>=(unsigned n) noexcept
334 {
335 if (n >= 256) {
336 limbs_ = {};
337 return *this;
338 }
339 const std::size_t limb_shift = n / 64;

493 {

511 [[nodiscard]] static constexpr std::optional<uint256> from_hex(std::string_view text) noexcept
512 {
513 if (text.size() >= 2 && text[0] == '0' && (text[1] == 'x' || text[1] == 'X'))
514 text.remove_prefix(2);
515 if (text.empty())
516 return std::nullopt;
517 while (text.size() > 1 && text[0] == '0')
518 text.remove_prefix(1);
519 if (text.size() > 64)
520 return std::nullopt;
521 uint256 acc;
522 for (const char c : text) {
523 std::uint64_t d = 0;
524 if (c >= '0' && c <= '9')
525 d = static_cast<std::uint64_t>(c - '0');
526 else if (c >= 'a' && c <= 'f')
527 d = static_cast<std::uint64_t>(c - 'a' + 10);
528 else if (c >= 'A' && c <= 'F')
529 d = static_cast<std::uint64_t>(c - 'A' + 10);
530 else
531 return std::nullopt;
532 acc <<= 4;
533 acc.limbs_[0] |= d;
534 }
535 return acc;
536 }

565 [[nodiscard]] static constexpr std::optional<uint256> parse_units(std::string_view text, unsigned decimals) noexcept
566 {
567 if (decimals > 77)
568 return std::nullopt;
569 const std::size_t point = text.find('.');
570 const std::string_view whole = text.substr(0, point);
571 const std::string_view fraction = point == std::string_view::npos ? std::string_view{} : text.substr(point + 1);
572 if (whole.empty() && fraction.empty())
573 return std::nullopt;
574 uint256 acc;
575 const auto push = [&acc](char c) {
576 return c >= '0' && c <= '9' && acc.mul_small(10) == 0 && acc.add_small(static_cast<std::uint64_t>(c - '0')) == 0;
577 };
578 for (const char c : whole)
579 if (!push(c))
580 return std::nullopt;
581 for (std::size_t i = 0; i < fraction.size(); ++i) {
582 const char c = fraction[i];
583 if (i < decimals) {
584 if (!push(c))
585 return std::nullopt;
586 } else if (c != '0') {
587 return std::nullopt; // a digit the token cannot hold; never rounded
588 }
589 }
590 for (std::size_t i = fraction.size(); i < decimals; ++i)
591 if (acc.mul_small(10) != 0)
592 return std::nullopt;
593 return acc;
594 }

652 // Add with carry out: returns 1 on overflow.
653 constexpr std::uint64_t add_with_carry(const uint256& other) noexcept
654 {
655 std::uint64_t carry = 0;
656 for (std::size_t i = 0; i < 4; ++i) {
657 const std::uint64_t sum = limbs_[i] + other.limbs_[i];
658 std::uint64_t c = sum < limbs_[i] ? 1u : 0u;
659 const std::uint64_t with_carry = sum + carry;
660 c += with_carry < sum ? 1u : 0u;
661 limbs_[i] = with_carry;
662 carry = c;
663 }
664 return carry;
665 }

667 // Subtract with borrow out: returns 1 when other > *this.
668 constexpr std::uint64_t sub_with_borrow(const uint256& other) noexcept
669 {
670 std::uint64_t borrow = 0;
671 for (std::size_t i = 0; i < 4; ++i) {
672 const std::uint64_t diff = limbs_[i] - other.limbs_[i];
673 std::uint64_t b = limbs_[i] < other.limbs_[i] ? 1u : 0u;
674 const std::uint64_t with_borrow = diff - borrow;
675 b += diff < borrow ? 1u : 0u;
676 limbs_[i] = with_borrow;
677 borrow = b;
678 }
679 return borrow;
680 }

682 // Multiply by a 64-bit value in place; returns the limb that fell off the top (0 when it fits).
683 constexpr std::uint64_t mul_small(std::uint64_t m) noexcept
684 {
685 std::uint64_t carry = 0;
686 for (std::size_t i = 0; i < 4; ++i) {
687 const detail::wide_product p = detail::umul128(limbs_[i], m);
688 const std::uint64_t low = p.low + carry;
689 limbs_[i] = low;
690 carry = p.high + (low < carry ? 1u : 0u);
691 }
692 return carry;
693 }

698 // Divide in place by a 32-bit value; returns the remainder.
699 constexpr std::uint32_t divmod_small(std::uint32_t d) noexcept
700 {
701 std::uint64_t rem = 0;
702 for (int i = 3; i >= 0; --i) {
703 const auto idx = static_cast<std::size_t>(i);
704 const std::uint64_t high = (rem << 32) | (limbs_[idx] >> 32);
705 const std::uint64_t qh = high / d;
706 rem = high % d;
707 const std::uint64_t low = (rem << 32) | (limbs_[idx] & 0xffffffffu);
708 const std::uint64_t ql = low / d;
709 rem = low % d;
710 limbs_[idx] = (qh << 32) | ql;
711 }
712 return static_cast<std::uint32_t>(rem);
713 }

715 // Whether both values fit in 64 bits: every high limb is zero.
716 static constexpr bool both_fit_64(const uint256& a, const uint256& b) noexcept
717 {
718 return (a.limbs_[1] | a.limbs_[2] | a.limbs_[3] | b.limbs_[1] | b.limbs_[2] | b.limbs_[3]) == 0;
719 }

721 // The full 512-bit product, as eight limbs.
722 static constexpr std::array<std::uint64_t, 8> mul_full(const uint256& a, const uint256& b) noexcept
723 {
724 std::array<std::uint64_t, 8> r{};
725 for (std::size_t i = 0; i < 4; ++i) {
726 std::uint64_t carry = 0;
727 for (std::size_t j = 0; j < 4; ++j) {
728 const detail::wide_product p = detail::umul128(a.limbs_[i], b.limbs_[j]);
729 std::uint64_t low = r[i + j] + p.low;
730 std::uint64_t c = low < p.low ? 1u : 0u;
731 const std::uint64_t with_carry = low + carry;
732 c += with_carry < low ? 1u : 0u;
733 r[i + j] = with_carry;
734 carry = p.high + c; // p.high <= 2^64 - 2, so no overflow
735 }
736 r[i + 4] = carry;
737 }
738 return r;
739 }

741 // Knuth's algorithm D (TAOCP 4.3.1), base 2^32, in the form of Hacker's Delight's divmnu.
742 // The 64-bit fast path, small enough to inline at every call: one hardware division, the same bits.
743 static constexpr std::pair<uint256, uint256> divmod_any(const uint256& dividend, const uint256& divisor) noexcept
744 {
745 if (both_fit_64(dividend, divisor))
746 return {uint256{dividend.limbs_[0] / divisor.limbs_[0]}, uint256{dividend.limbs_[0] % divisor.limbs_[0]}};
747 return divmod_knuth(dividend, divisor);
748 }

750 static constexpr std::pair<uint256, uint256> divmod_knuth(const uint256& dividend, const uint256& divisor) noexcept
751 {
752 std::array<std::uint32_t, 8> u{}, v{};
753 for (std::size_t i = 0; i < 4; ++i) {
754 u[2 * i] = static_cast<std::uint32_t>(dividend.limbs_[i]);
755 u[2 * i + 1] = static_cast<std::uint32_t>(dividend.limbs_[i] >> 32);
756 v[2 * i] = static_cast<std::uint32_t>(divisor.limbs_[i]);
757 v[2 * i + 1] = static_cast<std::uint32_t>(divisor.limbs_[i] >> 32);
758 }
759 int n = 8;
760 while (n > 1 && v[static_cast<std::size_t>(n - 1)] == 0)
761 --n;
762 int u_digits = 8;
763 while (u_digits > 0 && u[static_cast<std::size_t>(u_digits - 1)] == 0)
764 --u_digits;
765 if (u_digits < n)
766 return {uint256{}, dividend};
767
768 std::array<std::uint32_t, 8> q{}, r{};
769 if (n == 1) { // short division
770 std::uint64_t k = 0;
771 for (int j = u_digits - 1; j >= 0; --j) {
772 const auto idx = static_cast<std::size_t>(j);
773 const std::uint64_t cur = (k << 32) | u[idx];
774 q[idx] = static_cast<std::uint32_t>(cur / v[0]);
775 k = cur % v[0];
776 }
777 r[0] = static_cast<std::uint32_t>(k);
778 return {from_digits(q), from_digits(r)};
779 }
780
781 const int m = u_digits - n;
782 const int s = std::countl_zero(v[static_cast<std::size_t>(n - 1)]); // normalise so vn[n-1] has its top bit set
783 std::array<std::uint32_t, 8> vn{};
784 std::array<std::uint32_t, 9> un{};
785 for (int i = n - 1; i > 0; --i)
786 vn[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(
787 (static_cast<std::uint64_t>(v[static_cast<std::size_t>(i)]) << s) |
788 (static_cast<std::uint64_t>(v[static_cast<std::size_t>(i - 1)]) >> (32 - s)));
789 vn[0] = v[0] << s;
790 un[static_cast<std::size_t>(u_digits)] = static_cast<std::uint32_t>(
791 static_cast<std::uint64_t>(u[static_cast<std::size_t>(u_digits - 1)]) >> (32 - s));
792 for (int i = u_digits - 1; i > 0; --i)
793 un[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(
794 (static_cast<std::uint64_t>(u[static_cast<std::size_t>(i)]) << s) |
795 (static_cast<std::uint64_t>(u[static_cast<std::size_t>(i - 1)]) >> (32 - s)));
796 un[0] = u[0] << s;
797
798 constexpr std::uint64_t base = std::uint64_t{1} << 32;
799 for (int j = m; j >= 0; --j) {
800 const auto jn = static_cast<std::size_t>(j + n);
801 const std::uint64_t numerator = (static_cast<std::uint64_t>(un[jn]) << 32) | un[jn - 1];
802 std::uint64_t qhat = numerator / vn[static_cast<std::size_t>(n - 1)];
803 std::uint64_t rhat = numerator % vn[static_cast<std::size_t>(n - 1)];
804 while (qhat >= base ||
805 qhat * vn[static_cast<std::size_t>(n - 2)] > ((rhat << 32) | un[jn - 2])) {
806 --qhat;
807 rhat += vn[static_cast<std::size_t>(n - 1)];
808 if (rhat >= base)
809 break;
810 }
811 // multiply and subtract
812 std::int64_t k = 0;
813 std::int64_t t = 0;
814 for (int i = 0; i < n; ++i) {
815 const auto ij = static_cast<std::size_t>(i + j);
816 const std::uint64_t p = qhat * vn[static_cast<std::size_t>(i)];
817 t = static_cast<std::int64_t>(un[ij]) - k - static_cast<std::int64_t>(p & 0xffffffffu);
818 un[ij] = static_cast<std::uint32_t>(t);
819 k = static_cast<std::int64_t>(p >> 32) - (t >> 32);
820 }
821 t = static_cast<std::int64_t>(un[jn]) - k;
822 un[jn] = static_cast<std::uint32_t>(t);
823 q[static_cast<std::size_t>(j)] = static_cast<std::uint32_t>(qhat);
824 if (t < 0) { // qhat was one too large: add back
825 --q[static_cast<std::size_t>(j)];
826 std::uint64_t carry = 0;
827 for (int i = 0; i < n; ++i) {
828 const auto ij = static_cast<std::size_t>(i + j);
829 const std::uint64_t sum = static_cast<std::uint64_t>(un[ij]) + vn[static_cast<std::size_t>(i)] + carry;
830 un[ij] = static_cast<std::uint32_t>(sum);
831 carry = sum >> 32;
832 }
833 un[jn] = static_cast<std::uint32_t>(un[jn] + carry);
834 }
835 }
836 // the remainder is un, un-normalised
837 for (int i = 0; i < n; ++i) {
838 const auto idx = static_cast<std::size_t>(i);
839 r[idx] = static_cast<std::uint32_t>((static_cast<std::uint64_t>(un[idx]) >> s) |
840 (static_cast<std::uint64_t>(un[idx + 1]) << (32 - s)));
841 }
842 return {from_digits(q), from_digits(r)};
843 }
844
845 static constexpr uint256 from_digits(const std::array<std::uint32_t, 8>& d) noexcept
846 {
847 return from_limbs(d[0] | (static_cast<std::uint64_t>(d[1]) << 32), d[2] | (static_cast<std::uint64_t>(d[3]) << 32),
848 d[4] | (static_cast<std::uint64_t>(d[5]) << 32), d[6] | (static_cast<std::uint64_t>(d[7]) << 32));
849 }
850
851 std::array<std::uint64_t, 4> limbs_{};
852 };
