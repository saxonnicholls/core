92 template <std::size_t N>
93 struct inplace_string {
94 static_assert(N >= 1 && N <= 0xFFFFFFFFu, "snicholls::inplace_string: the capacity must be 1 to 2^32 - 1");
95
97 using size_type = detail::inplace_string_size_t<N>;
98
100 char chars_[N]{};
102 size_type size_{};
103
105 constexpr inplace_string() noexcept = default;
106
110 template <std::size_t M>
111 requires(M <= N + 1)
112 consteval inplace_string(const char (&literal)[M]) noexcept // NOLINT(google-explicit-constructor)
113 {
114 for (std::size_t i = 0; i + 1 < M; ++i)
115 chars_[i] = literal[i];
116 size_ = static_cast<size_type>(M - 1);
117 }
118
120 // make's constructor: the length is already checked
121 constexpr inplace_string(std::string_view text, detail::inplace_string_fits) noexcept
122 : size_(static_cast<size_type>(text.size()))
123 {
124 std::copy_n(text.data(), text.size(), chars_); // memmove at run time
125 }
127
131 [[nodiscard]] static constexpr result<inplace_string, inplace_string_errc> make(std::string_view text) noexcept
132 {
133 if (text.size() > N)
134 return err(inplace_string_errc::too_long);
135 // constructed in place in the result, so the characters are written once
136 return result<inplace_string, inplace_string_errc>(std::in_place, text, detail::inplace_string_fits{});
137 }
138
141 [[nodiscard]] constexpr std::string_view view() const noexcept { return {chars_, size_}; }
144 [[nodiscard]] constexpr const char* data() const noexcept { return chars_; }
147 [[nodiscard]] constexpr std::size_t size() const noexcept { return size_; }
150 [[nodiscard]] constexpr bool empty() const noexcept { return size_ == 0; }
153 [[nodiscard]] static constexpr std::size_t capacity() noexcept { return N; }
156 [[nodiscard]] constexpr std::uint64_t hash() const noexcept { return fnv1a_64(view()); }
