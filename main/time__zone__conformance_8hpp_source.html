61 concept time_zone_probe = std::is_default_constructible_v<Probe> &&

67 // Calls f(before, after) for every change of rules found by hourly sampling of [from, to).
68 template <typename Z, typename F>
69 void for_each_transition(const Z& z, std::chrono::sys_seconds from, std::chrono::sys_seconds to, F&& f)
70 {
71 for (auto t = from; t < to; t += std::chrono::hours{1}) {
72 const zone_info here = z.info(t);
73 if (here.end < to && here.end > t) {
74 const zone_info next = z.info(here.end);
75 if (next.offset != here.offset)
76 f(here, next);
77 t = here.end - std::chrono::hours{1};
78 }
79 }
80 }

82 constexpr std::chrono::local_seconds local_of(std::chrono::sys_seconds t, std::chrono::seconds offset) noexcept
83 {
84 return std::chrono::local_seconds(t.time_since_epoch() + offset);

92 template <typename Probe>

94 [[nodiscard]] capability_conformance_report run_time_zone_conformance()

96 using namespace std::chrono;
97 capability_conformance_report report;

99 detail::capability_check<Probe>(report, "the rules hold at the instant", [](Probe& p, const auto& expect) {
100 const auto& z = p.zone();
101 bool holds = true, bounded = true;
102 for (auto t = p.from(); t < p.to(); t += hours{1}) {
103 const zone_info i = z.info(t);
104 holds = holds && i.begin <= t && t < i.end;
105 bounded = bounded && i.offset > -hours{24} && i.offset < hours{24} && !i.abbrev.empty();
106 }
107 expect(holds, "begin <= t < end at every hour");
108 expect(bounded, "offsets are under a day and every interval has an abbreviation");
109 });
110
111 detail::capability_check<Probe>(report, "intervals follow one another", [](Probe& p, const auto& expect) {
112 const auto& z = p.zone();
113 bool follows = true;
114 for (auto t = p.from(); t < p.to(); t += hours{1}) {
115 const zone_info i = z.info(t);
116 if (i.end < p.to())
117 follows = follows && z.info(i.end).begin == i.end && z.info(i.end - seconds{1}) == i;
118 }
119 expect(follows, "the next interval begins where one ends");
120 });
121
122 detail::capability_check<Probe>(report, "a unique wall-clock time round-trips", [](Probe& p, const auto& expect) {
123 const auto& z = p.zone();
124 bool round_trips = true, classified = true;
125 for (auto t = p.from(); t < p.to(); t += hours{1}) {
126 const utc_s u{t};
127 const auto l = to_local(z, u);
128 const local_info li = z.info(floor<seconds>(l));
129 if (li.kind == local_time_kind::unique) {
130 const auto back = from_local(z, l);
131 round_trips = round_trips && back && *back == u;
132 } else {
133 // a time that is not unique must still be one of its candidates
134 classified = classified && li.kind == local_time_kind::ambiguous &&
135 (li.first.offset == offset_at(z, u) || li.second.offset == offset_at(z, u));
136 }
137 }
138 expect(round_trips, "from_local(to_local(t)) == t whenever the local time is unique");
139 expect(classified, "a local time reached from an instant is never nonexistent");
140 });
141
142 detail::capability_check<Probe>(report, "a gap is nonexistent and resolves by policy", [](Probe& p, const auto& expect) {
143 const auto& z = p.zone();
144 detail::for_each_transition(z, p.from(), p.to(), [&](const zone_info& before, const zone_info& after) {
145 if (after.offset <= before.offset)
146 return;
147 const auto width = after.offset - before.offset;
148 const local_seconds in_gap = detail::local_of(after.begin, before.offset) + width / 2;
149 const local_info li = z.info(in_gap);
150 expect(li.kind == local_time_kind::nonexistent, "a time in the gap is nonexistent");
151 expect(li.first.offset == before.offset && li.second.offset == after.offset, "first before, second after");
152 const auto rejected = from_local(z, in_gap);
153 expect(rejected.is_err() && rejected.unwrap_err() == time_zone_errc::nonexistent_local_time, "rejected by default");
154 const auto next = from_local(z, in_gap, {fold_policy::reject, gap_policy::next_valid});
155 expect(next && next->sys_time() == after.begin, "next_valid is the transition");
156 const auto shifted = from_local(z, in_gap, {fold_policy::reject, gap_policy::shift_forward});
157 expect(shifted && shifted->sys_time() == sys_seconds(in_gap.time_since_epoch() - before.offset),
158 "shift_forward keeps the earlier offset");
159 });
160 });
161
162 detail::capability_check<Probe>(report, "a fold is ambiguous and resolves by policy", [](Probe& p, const auto& expect) {
163 const auto& z = p.zone();
164 detail::for_each_transition(z, p.from(), p.to(), [&](const zone_info& before, const zone_info& after) {
165 if (after.offset >= before.offset)
166 return;
167 const auto width = before.offset - after.offset;
168 const local_seconds in_fold = detail::local_of(after.begin, after.offset) + width / 2;
169 const local_info li = z.info(in_fold);
170 expect(li.kind == local_time_kind::ambiguous, "a time in the fold is ambiguous");
171 expect(li.first.offset == before.offset && li.second.offset == after.offset, "first is the earlier offset's");
172 const auto rejected = from_local(z, in_fold);
173 expect(rejected.is_err() && rejected.unwrap_err() == time_zone_errc::ambiguous_local_time, "rejected by default");
174 const auto early = from_local(z, in_fold, {fold_policy::earliest, gap_policy::reject});
175 const auto late = from_local(z, in_fold, {fold_policy::latest, gap_policy::reject});
176 expect(early && late && *early < *late && *late - *early == width, "earliest and latest are the width apart");
177 });
178 });
179
180 detail::capability_check<Probe>(report, "the name is stable", [](Probe& p, const auto& expect) {
181 const auto& z = p.zone();
182 const std::string first{z.name()};
183 expect(!first.empty(), "the zone has a name");
184 expect(z.name() == first, "the name does not change");
185 });

187 return report;
188 }
