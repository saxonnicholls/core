78 // The sum of the weights; NaN if any weight is negative or NaN, so that usable() refuses it
79 template <typename T>
80 constexpr T total(std::span<const T> w) noexcept
81 {
82 T s(0);
83 for (const T x : w) {
84 if (!(x >= T(0)))
85 return std::numeric_limits<T>::quiet_NaN();
86 s += x;
87 }
88 return s;
89 }

91 template <typename T>
92 constexpr bool usable(T s) noexcept
93 {
94 return s > T(0) && s < std::numeric_limits<T>::infinity();
95 }

97 // Walks the cumulative weights against the increasing u(i) in [0, 1), writing M indices
98 template <typename T, typename U>
99 constexpr void walk(std::span<const T> w, T sum, std::span<std::size_t> out, U&& u) noexcept(noexcept(u(std::size_t{})))
100 {
101 std::size_t j = 0;
102 T cum = w.empty() ? T(0) : w[0] / sum;
103 const std::size_t last = w.size() - 1;
104 for (std::size_t i = 0; i < out.size(); ++i) {
105 const T ui = u(i);
106 while (ui >= cum && j < last)
107 cum += w[++j] / sum;
108 // Never land on a particle of zero weight (rounding at the end of the cumulative sum)
109 while (w[j] == T(0) && j > 0)
110 --j;
111 out[i] = j;
112 }
113 }

121 template <typename T>
122 [[nodiscard]] constexpr T effective_sample_size(std::span<const T> weights) noexcept
123 {
124 T s(0), s2(0);
125 for (const T x : weights) {
126 s += x;
127 s2 += x * x;
128 }
129 return s2 > T(0) ? s * s / s2 : T(0);
130 }

146 template <typename T>
147 [[nodiscard]] constexpr T log_sum_exp(std::span<const T> x) noexcept
148 {
149 T m = -std::numeric_limits<T>::infinity();
150 for (const T v : x)
151 if (v > m)
152 m = v;
153 if (m == -std::numeric_limits<T>::infinity() || m == std::numeric_limits<T>::infinity())
154 return m;
155 T s(0);
156 for (const T v : x)
157 s += static_cast<T>(math::exp(v - m));
158 return m + static_cast<T>(math::log(s));
159 }

165 template <typename T>
166 constexpr T normalise_log_weights(std::span<T> log_weights) noexcept
167 {
168 const T z = log_sum_exp(std::span<const T>(log_weights));
169 if (z == -std::numeric_limits<T>::infinity() || z == std::numeric_limits<T>::infinity())
170 return z;
171 for (T& v : log_weights)
172 v -= z;
173 return z;
174 }

184 template <typename T, full_range_generator G>
185 constexpr bool systematic_resample(std::span<const T> weights, std::span<std::size_t> indices, G& g) noexcept(noexcept(g()))
186 {
187 const T sum = detail::resample::total(weights);
188 if (!detail::resample::usable(sum) || indices.empty())
189 return detail::resample::usable(sum);
190 const T m = static_cast<T>(indices.size());
191 const T u0 = static_cast<T>(uniform_real(g));
192 detail::resample::walk(weights, sum, indices, [&](std::size_t i) noexcept { return (static_cast<T>(i) + u0) / m; });
193 return true;
194 }

224 template <typename T, full_range_generator G>
225 constexpr bool multinomial_resample(std::span<const T> weights, std::span<std::size_t> indices, G& g) noexcept(noexcept(g()))
226 {
227 const T sum = detail::resample::total(weights);
228 if (!detail::resample::usable(sum) || indices.empty())
229 return detail::resample::usable(sum);
230 // The order statistics of M uniforms are S_k / S_{M+1} for S_k the partial sums of M + 1 exponentials. Draw
231 // S_{M+1} first as a whole (a sum of M + 1 exponentials is not cheaper to draw directly), so draw the M + 1
232 // exponentials twice from a copy of the generator: once to total them, once to walk.
233 auto exp1 = [](G& gg) { return -static_cast<T>(math::log(T(1) - static_cast<T>(uniform_real(gg)))); };
234 G copy = g;
235 T total(0);
236 for (std::size_t i = 0; i <= indices.size(); ++i)
237 total += exp1(copy);
238 T partial(0);
239 detail::resample::walk(weights, sum, indices, [&](std::size_t) {
240 partial += exp1(g);
241 return partial / total;
242 });
243 static_cast<void>(exp1(g)); // the (M + 1)-th, so g ends where the copy did
244 return true;
245 }

258 template <typename T, full_range_generator G>
259 constexpr bool residual_resample(std::span<const T> weights, std::span<T> residual, std::span<std::size_t> indices, G& g,
260 residual_method method = residual_method::systematic) noexcept(noexcept(g()))
261 {
262 const T sum = detail::resample::total(weights);
263 if (!detail::resample::usable(sum) || residual.size() < weights.size())
264 return false;
265 const T m = static_cast<T>(indices.size());
266 std::size_t k = 0;
267 for (std::size_t j = 0; j < weights.size(); ++j) {
268 const T expected = m * (weights[j] / sum); // the ratio first: no overflow for weights near the largest double
269 const auto copies = static_cast<std::size_t>(math::floor(expected));
270 for (std::size_t c = 0; c < copies && k < indices.size(); ++c)
271 indices[k++] = j;
272 residual[j] = expected - static_cast<T>(copies);
273 }
274 const auto rest = indices.subspan(k);
275 if (rest.empty())
276 return true;
277 const std::span<const T> r(residual.data(), weights.size());
278 if (!detail::resample::usable(detail::resample::total(r))) {
279 // Rounding left draws but no residual weight: give them to the heaviest particle
280 std::size_t best = 0;
281 for (std::size_t j = 1; j < weights.size(); ++j)
282 if (weights[j] > weights[best])
283 best = j;
284 for (auto& x : rest)
285 x = best;
286 } else if (method == residual_method::systematic)
287 systematic_resample(r, rest, g);
288 else if (method == residual_method::stratified)
289 stratified_resample(r, rest, g);
290 else
291 multinomial_resample(r, rest, g);
292 return true;
293 }
