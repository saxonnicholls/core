63 [[nodiscard]] constexpr std::optional<std::uint64_t> checked_binomial(std::uint64_t n, std::uint64_t k) noexcept
64 {
65 if (k > n)
66 return 0;
67 k = k < n - k ? k : n - k;
68 std::uint64_t result = 1;
69 for (std::uint64_t i = 1; i <= k; ++i) {
70 // result * (n - k + i) / i, exactly: result is C(n - k + i - 1, i - 1), divisible after the multiply
71 // result * (n - k + i) is divisible by i; with g = gcd(result, i), i / g divides the factor,
72 // so (result / g) * (factor / (i / g)) is exact and overflows only if C(n, k) itself does
73 const std::uint64_t factor = n - k + i;
74 std::uint64_t g = result, h = i;
75 while (h != 0) {
76 const std::uint64_t r = g % h;
77 g = h;
78 h = r;
79 }
80 const std::uint64_t a = result / g, b = factor / (i / g);
81 if (b != 0 && a > std::numeric_limits<std::uint64_t>::max() / b)
82 return std::nullopt;
83 result = a * b;
84 }
85 return result;
86 }

103 template <std::size_t N, std::size_t K, typename F>
104 constexpr void for_each_combination(F&& f) noexcept(noexcept(f(std::declval<const std::array<std::size_t, K>&>())))
105 {
106 if constexpr (K > N) {
107 return;
108 } else {
109 std::array<std::size_t, K> row{};
110 for (std::size_t i = 0; i < K; ++i)
111 row[i] = i;
112 while (true) {
113 f(std::as_const(row));
114 std::size_t i = K;
115 while (i > 0 && row[i - 1] == N - K + i - 1)
116 --i;
117 if (i == 0)
118 return;
119 ++row[i - 1];
120 for (std::size_t j = i; j < K; ++j)
121 row[j] = row[j - 1] + 1;
122 }
123 }
124 }

188 consteval std::optional<std::size_t> checked_power(std::size_t n, std::size_t k) noexcept
189 {
190 std::size_t c = 1;
191 for (std::size_t i = 0; i < k; ++i) {
192 if (n != 0 && c > std::numeric_limits<std::size_t>::max() / n)
193 return std::nullopt;
194 c *= n;
195 }
196 return c;

215 template <std::size_t N, std::size_t K, typename F>
216 constexpr void for_each_tuple(F&& f) noexcept(noexcept(f(std::declval<const std::array<std::size_t, K>&>())))
217 {
218 if constexpr (N == 0 && K > 0) {
219 return;
220 } else {
221 std::array<std::size_t, K> row{};
222 while (true) {
223 f(std::as_const(row));
224 std::size_t i = K;
225 while (i > 0 && row[i - 1] == N - 1) {
226 row[i - 1] = 0;
227 --i;
228 }
229 if (i == 0)
230 return;
231 ++row[i - 1];
232 }
233 }

244 [[nodiscard]] constexpr std::array<std::array<T, K>, tuple_count_v<N, K>> tuples(const std::array<T, N>& values)
245 {

247 std::size_t next = 0;
248 for_each_tuple<N, K>([&](const std::array<std::size_t, K>& row) {
249 for (std::size_t i = 0; i < K; ++i)
250 rows[next][i] = values[row[i]];
251 ++next;
