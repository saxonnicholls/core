303 {

380 template <linear_operator Op, typename M = identity_preconditioner>
381 requires preconditioner<M, typename Op::value_type>
382 [[nodiscard]] constexpr krylov_result<typename Op::value_type> conjugate_gradient(const Op& op, std::span<const typename Op::value_type> b, std::span<typename Op::value_type> x,
383 std::span<typename Op::value_type> work, const krylov_options<typename Op::value_type>& options = {},
384 const M& m = {}) noexcept(detail::nothrow_krylov<Op, M>)
385 {
386 using T = typename Op::value_type;
387 using result_t = krylov_result<T>;
388 const std::size_t n = b.size();
389 if (op.rows() != n || op.cols() != n || x.size() != n || work.size() < conjugate_gradient_workspace(n))
390 return result_t{krylov_status::invalid_input, 0, T(0)};
391 // Without a preconditioner z is r itself: no copy, the same numbers
392 constexpr bool plain = std::is_same_v<M, identity_preconditioner>;
393 const auto r = work.subspan(0, n), z = plain ? r : work.subspan(n, n), p = work.subspan(2 * n, n), q = work.subspan(3 * n, n);
394 const T bnorm = detail::krylov_norm(b);
395 if (bnorm == T(0)) {
396 for (std::size_t i = 0; i < n; ++i)
397 x[i] = T(0);
398 return result_t{krylov_status::converged, 0, T(0)};
399 }
400 const T threshold = detail::krylov_threshold(options, bnorm);
401 const std::size_t limit = options.max_iterations != 0 ? options.max_iterations : 2 * n;
402 op.apply(std::span<const T>(x), q);
403 for (std::size_t i = 0; i < n; ++i)
404 r[i] = b[i] - q[i];
405 T rnorm = detail::krylov_norm(std::span<const T>(r));
406 if (rnorm <= threshold)
407 return result_t{krylov_status::converged, 0, rnorm};
408 if (!plain)
409 m.apply(std::span<const T>(r), z);
410 T rz = detail::krylov_dot(std::span<const T>(r), std::span<const T>(z));
411 if (!(rz > T(0)))
412 return result_t{krylov_status::breakdown, 0, rnorm};
413 for (std::size_t i = 0; i < n; ++i)
414 p[i] = z[i];
415 for (std::size_t k = 1; k <= limit; ++k) {
416 op.apply(std::span<const T>(p), q);
417 const T pq = detail::krylov_dot(std::span<const T>(p), std::span<const T>(q));
418 if (!(pq > T(0)))
419 return result_t{krylov_status::breakdown, k - 1, rnorm};
420 const T alpha = rz / pq;
421 for (std::size_t i = 0; i < n; ++i) {
422 x[i] += alpha * p[i];
423 r[i] -= alpha * q[i];
424 }
425 rnorm = detail::krylov_norm(std::span<const T>(r));
426 if (rnorm <= threshold)
427 return result_t{krylov_status::converged, k, rnorm};
428 if (!plain)
429 m.apply(std::span<const T>(r), z);
430 const T rz_new = detail::krylov_dot(std::span<const T>(r), std::span<const T>(z));
431 if (!(rz_new > T(0)))
432 return result_t{krylov_status::breakdown, k, rnorm};
433 const T beta = rz_new / rz;
434 rz = rz_new;
435 for (std::size_t i = 0; i < n; ++i)
436 p[i] = z[i] + beta * p[i];
437 }
438 return result_t{krylov_status::iteration_limit, limit, rnorm};
439 }

454 template <linear_operator Op, typename M = identity_preconditioner>
455 requires preconditioner<M, typename Op::value_type>
456 [[nodiscard]] constexpr krylov_result<typename Op::value_type> minres(const Op& op, std::span<const typename Op::value_type> b, std::span<typename Op::value_type> x,
457 std::span<typename Op::value_type> work, const krylov_options<typename Op::value_type>& options = {},
458 const M& m = {}) noexcept(detail::nothrow_krylov<Op, M>)
459 {
460 using T = typename Op::value_type;
461 using result_t = krylov_result<T>;
462 using cspan = std::span<const T>;
463 const std::size_t n = b.size();
464 if (op.rows() != n || op.cols() != n || x.size() != n || work.size() < minres_workspace(n))
465 return result_t{krylov_status::invalid_input, 0, T(0)};
466 auto v = work.subspan(0, n), y = work.subspan(n, n), r1 = work.subspan(2 * n, n), r2 = work.subspan(3 * n, n);
467 auto w = work.subspan(4 * n, n), w1 = work.subspan(5 * n, n), w2 = work.subspan(6 * n, n);
468 m.apply(b, y);
469 const T bb = detail::krylov_dot(b, cspan(y));
470 if (bb < T(0))
471 return result_t{krylov_status::breakdown, 0, T(0)};
472 if (bb == T(0)) {
473 for (std::size_t i = 0; i < n; ++i)
474 x[i] = T(0);
475 return result_t{krylov_status::converged, 0, T(0)};
476 }
477 const T threshold = detail::krylov_threshold(options, static_cast<T>(math::sqrt(bb)));
478 const std::size_t limit = options.max_iterations != 0 ? options.max_iterations : 2 * n;
479 op.apply(cspan(x), y);
480 for (std::size_t i = 0; i < n; ++i)
481 r1[i] = r2[i] = b[i] - y[i];
482 m.apply(cspan(r1), y);
483 const T b1 = detail::krylov_dot(cspan(r1), cspan(y));
484 if (b1 < T(0))
485 return result_t{krylov_status::breakdown, 0, T(0)};
486 const T beta1 = static_cast<T>(math::sqrt(b1));
487 if (beta1 <= threshold)
488 return result_t{krylov_status::converged, 0, beta1};
489 for (std::size_t i = 0; i < n; ++i)
490 w[i] = w2[i] = T(0);
491 T oldb(0), beta = beta1, dbar(0), epsln(0), phibar = beta1, cs(-1), sn(0);
492 for (std::size_t k = 1; k <= limit; ++k) {
493 // Lanczos: v = y / β, y = A v - (β / β_old) r1 - (α / β) r2, then r1 ← r2, r2 ← y, y ← M⁻¹ r2
494 const T s = T(1) / beta;
495 for (std::size_t i = 0; i < n; ++i)
496 v[i] = s * y[i];
497 op.apply(cspan(v), y);
498 if (k >= 2) {
499 const T f = beta / oldb;
500 for (std::size_t i = 0; i < n; ++i)
501 y[i] -= f * r1[i];
502 }
503 const T alfa = detail::krylov_dot(cspan(v), cspan(y));
504 const T g = alfa / beta;
505 for (std::size_t i = 0; i < n; ++i)
506 y[i] -= g * r2[i];
507 std::swap(r1, r2); // r1 ← r2
508 std::swap(r2, y); // r2 ← the new vector; y takes the old r1's storage
509 m.apply(cspan(r2), y);
510 oldb = beta;
511 const T bsq = detail::krylov_dot(cspan(r2), cspan(y));
512 if (bsq < T(0))
513 return result_t{krylov_status::breakdown, k - 1, phibar};
514 beta = static_cast<T>(math::sqrt(bsq));
515 // The tridiagonal's next column through the previous rotation, then a new rotation
516 const T oldeps = epsln;
517 const T delta = cs * dbar + sn * alfa;
518 const T gbar = sn * dbar - cs * alfa;
519 epsln = sn * beta;
520 dbar = -cs * beta;
521 T gamma = static_cast<T>(math::sqrt(gbar * gbar + beta * beta));
522 gamma = gamma > std::numeric_limits<T>::epsilon() ? gamma : std::numeric_limits<T>::epsilon();
523 cs = gbar / gamma;
524 sn = beta / gamma;
525 const T phi = cs * phibar;
526 phibar = sn * phibar;
527 // Directions: w1 ← w2, w2 ← w, w ← (v - ε w1 - δ w2) / γ
528 const T denom = T(1) / gamma;
529 std::swap(w1, w2); // w1 ← w2
530 std::swap(w2, w); // w2 ← w; w takes the old w1's storage, overwritten next
531 for (std::size_t i = 0; i < n; ++i) {
532 w[i] = (v[i] - oldeps * w1[i] - delta * w2[i]) * denom;
533 x[i] += phi * w[i];
534 }
535 if (phibar <= threshold)
536 return result_t{krylov_status::converged, k, phibar};
537 }
538 return result_t{krylov_status::iteration_limit, limit, phibar};
539 }
