124constexpr key_range attention_keys(attention_mask mask, std::size_t pos, std::size_t length, std::size_t window = 0,
125 std::size_t block = 1) noexcept
126{
127 if (pos >= length)
128 return {0, 0};
129 switch (mask) {
130 case attention_mask::bidirectional:
131 return {0, length};
132 case attention_mask::block_causal: {
133 const std::size_t b = block == 0 ? 1 : block, end = (pos / b + 1) * b;
134 return {0, end < length ? end : length};
135 }
136 case attention_mask::causal:
137 break;
138 }
139 return attention_window(pos, window);
140}

145 constexpr T dot(std::span<const T> a, std::span<const T> b)
146 {
147 T s = T(0);
148 for (std::size_t i = 0; i < a.size(); ++i)
149 s += a[i] * b[i];
150 return s;
151 }

153 // One step of the recurrence: fold value v with score s into (max, sum, acc)
154 template <typename T>
155 constexpr void fold(T& max, T& sum, std::span<T> acc, T s, std::span<const T> v)
156 {
157 if (s == -std::numeric_limits<T>::infinity())
158 return;
159 if (s > max) {
160 const T scale = math::exp(max - s);
161 for (std::size_t i = 0; i < acc.size(); ++i)
162 acc[i] = acc[i] * scale + v[i];
163 sum = sum * scale + T(1);
164 max = s;
165 } else {
166 const T w = math::exp(s - max);
167 for (std::size_t i = 0; i < acc.size(); ++i)
168 acc[i] += w * v[i];
169 sum += w;
170 }
171 }

178template <math::math_number T, std::size_t D>
179struct attention_accumulator
180{
181 online_softmax<T> weights{};
182 std::array<T, D> acc{};
183
187 template <row V>
188 requires std::same_as<row_value_t<V>, T>
189 constexpr void add(T score, const V& value)
190 {
191 const auto v = as_row(value);
192 assert(v.size() == D);
193 detail::attention::fold(weights.max, weights.sum, std::span<T>(acc), score, v);
194 }
195
198 constexpr void merge(const attention_accumulator& other)
199 {
200 if (other.weights.sum == T(0))
201 return;
202 if (weights.sum == T(0)) {
203 *this = other;
204 return;
205 }
206 const T m = std::max(weights.max, other.weights.max);
207 const T a = math::exp(weights.max - m), b = math::exp(other.weights.max - m);
208 for (std::size_t i = 0; i < D; ++i)
209 acc[i] = acc[i] * a + other.acc[i] * b;
210 weights.sum = weights.sum * a + other.weights.sum * b;
211 weights.max = m;
212 }
213
216 template <mutable_row Out>
217 requires std::same_as<row_value_t<Out>, T>
218 constexpr void result(Out&& out) const
219 {
220 const auto y = as_mutable_row(out);
221 assert(y.size() == D);
222 for (std::size_t i = 0; i < D; ++i)
223 y[i] = weights.sum == T(0) ? T(0) : acc[i] / weights.sum;
224 }
225
228 [[nodiscard]] constexpr T log_sum_exp() const { return weights.log_sum_exp(); }
229};

238template <row Q, row K, row V, mutable_row Out>
239 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
240 && std::same_as<row_value_t<Q>, row_value_t<Out>>
241constexpr row_value_t<Q> attend(const Q& q, const K& keys, const V& values, Out&& out, std::type_identity_t<row_value_t<Q>> scale)
242{
243 using T = row_value_t<Q>;
244 const auto qs = as_row(q);
245 const auto ks = as_row(keys);
246 const auto vs = as_row(values);
247 const auto y = as_mutable_row(out);
248 const std::size_t d = qs.size();
249 assert(y.size() == d && ks.size() == vs.size() && (d == 0 || ks.size() % d == 0));
250 std::fill(y.begin(), y.end(), T(0));
251 T max = -std::numeric_limits<T>::infinity(), sum = T(0);
252 for (std::size_t j = 0; d != 0 && j < ks.size() / d; ++j)
253 detail::attention::fold(max, sum, y, scale * detail::attention::dot(qs, ks.subspan(j * d, d)), vs.subspan(j * d, d));
254 for (T& v : y)
255 v = sum == T(0) ? T(0) : v / sum;
256 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
257}

273template <row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK, mutable_row DV>
274 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
275 && std::same_as<row_value_t<Q>, row_value_t<O>> && std::same_as<row_value_t<Q>, row_value_t<DO>>
276 && std::same_as<row_value_t<Q>, row_value_t<DQ>> && std::same_as<row_value_t<Q>, row_value_t<DK>>
277 && std::same_as<row_value_t<Q>, row_value_t<DV>>
278constexpr void attend_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
279 std::type_identity_t<row_value_t<Q>> lse, std::type_identity_t<row_value_t<Q>> scale, DQ&& dq,
280 DK&& dkeys, DV&& dvalues)
281{
282 using T = row_value_t<Q>;
283 const auto qs = as_row(q);
284 const auto ks = as_row(keys);
285 const auto vs = as_row(values);
286 const auto o = as_row(out);
287 const auto g = as_row(dout);
288 const auto gq = as_mutable_row(dq);
289 const auto gk = as_mutable_row(dkeys);
290 const auto gv = as_mutable_row(dvalues);
291 const std::size_t d = qs.size();
292 assert(o.size() == d && g.size() == d && gq.size() == d && ks.size() == vs.size() && gk.size() == ks.size()
293 && gv.size() == vs.size() && (d == 0 || ks.size() % d == 0));
294 std::fill(gq.begin(), gq.end(), T(0));
295 if (d == 0 || lse == -std::numeric_limits<T>::infinity())
296 return;
297 const T big_d = detail::attention::dot(g, o);
298 for (std::size_t j = 0; j < ks.size() / d; ++j) {
299 const auto k = ks.subspan(j * d, d);
300 const auto v = vs.subspan(j * d, d);
301 const T p = math::exp(scale * detail::attention::dot(qs, k) - lse);
302 const T ds = p * (detail::attention::dot(g, v) - big_d);
303 for (std::size_t i = 0; i < d; ++i) {
304 gv[j * d + i] += p * g[i];
305 gq[i] += scale * ds * k[i];
306 gk[j * d + i] += scale * ds * qs[i];
307 }
308 }
