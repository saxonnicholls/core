88// k if den == 10^k (k = 0..9), otherwise -1.
89constexpr int decimal_exponent(std::intmax_t den) noexcept
90{
91 int k = 0;
92 while (den > 1 && den % 10 == 0 && k < 10) {
93 den /= 10;
94 ++k;
95 }
96 return den == 1 && k <= 9 ? k : -1;
97}

99// Decimal digits of a second a Period resolves: s = 0, ms = 3, us = 6, ns = 9; whole-second multiples
100// (minutes, hours, days) = 0; anything else (1/3 s, 7/10 s) = -1.
101template <typename Period>
102inline constexpr int utc_fraction_digits =
103 Period::den == 1 ? 0 : (Period::num == 1 ? decimal_exponent(Period::den) : -1);

107// A Duration snicholls::utc can hold: a signed integral std::chrono::duration whose period is 1/10^k s (k = 0..9) or
108// a whole number of seconds (minutes, hours, days).
109template <typename D>
110concept utc_duration = detail::is_chrono_duration<D>::value && std::is_integral_v<typename D::rep> &&
111 std::is_signed_v<typename D::rep> && detail::utc_fraction_digits<typename D::period> >= 0;

216 template <typename Rep2, typename Period2>
217 requires std::is_integral_v<Rep2> && std::is_signed_v<Rep2>
218 [[nodiscard]] static constexpr bool is_representable(const std::chrono::duration<Rep2, Period2>& d) noexcept
219 {
220 using D2 = std::chrono::duration<Rep2, Period2>;
221 if constexpr (std::ratio_less_equal_v<Period2, period> && sizeof(Rep2) <= sizeof(rep)) {
222 return true; // finer (or equal) unit, no wider rep: flooring only shrinks the magnitude
223 } else if constexpr (std::ratio_less_equal_v<Period2, period>) {
224 // Floor in the wider Rep2 (flooring straight into Duration would narrow before we could check).
225 const auto f = std::chrono::floor<std::chrono::duration<Rep2, period>>(d);
226 return f.count() >= std::numeric_limits<rep>::min() && f.count() <= std::numeric_limits<rep>::max();
227 } else {
228 constexpr D2 hi = std::chrono::floor<D2>(Duration::max());
229 constexpr D2 lo = std::chrono::ceil<D2>(Duration::min());
230 return d >= lo && d <= hi;
231 }
232 }

269 [[nodiscard]] static constexpr utc from_timespec(const std::timespec& ts) noexcept
270 {
271 const auto s = static_cast<std::int64_t>(ts.tv_sec);
272 const auto ns = static_cast<std::int64_t>(ts.tv_nsec);
273 if constexpr (std::ratio_less_equal_v<period, std::ratio<1>>) {
274 // A negative second with a fraction is (s + 1) s minus the rest, so an instant in the last second
275 // above min() never passes through the whole second s, which Duration cannot hold (found by the
276 // portability tests: utc<>::min().to_timespec() did not round-trip).
277 if (s < 0 && ns > 0)
278 return from_unix(std::chrono::duration<std::int64_t>(s + 1)) -
279 std::chrono::ceil<Duration>(std::chrono::nanoseconds(1'000'000'000 - ns));
280 }
281 return from_unix(std::chrono::duration<std::int64_t>(s)) + std::chrono::floor<Duration>(std::chrono::nanoseconds(ns));
282 }
