89// k if den == 10^k (k = 0..9), otherwise -1.
90constexpr int decimal_exponent(std::intmax_t den) noexcept
91{
92 int k = 0;
93 while (den > 1 && den % 10 == 0 && k < 10) {
94 den /= 10;
95 ++k;
96 }
97 return den == 1 && k <= 9 ? k : -1;
98}

100// Decimal digits of a second a Period resolves: s = 0, ms = 3, us = 6, ns = 9; whole-second multiples
101// (minutes, hours, days) = 0; anything else (1/3 s, 7/10 s) = -1.
102template <typename Period>
103inline constexpr int utc_fraction_digits =
104 Period::den == 1 ? 0 : (Period::num == 1 ? decimal_exponent(Period::den) : -1);

108// A Duration snicholls::utc can hold: a signed integral std::chrono::duration whose period is 1/10^k s (k = 0..9) or
109// a whole number of seconds (minutes, hours, days).
110template <typename D>
111concept utc_duration = detail::is_chrono_duration<D>::value && std::is_integral_v<typename D::rep> &&
112 std::is_signed_v<typename D::rep> && detail::utc_fraction_digits<typename D::period> >= 0;

217 template <typename Rep2, typename Period2>
218 requires std::is_integral_v<Rep2> && std::is_signed_v<Rep2>
219 [[nodiscard]] static constexpr bool is_representable(const std::chrono::duration<Rep2, Period2>& d) noexcept
220 {
221 using D2 = std::chrono::duration<Rep2, Period2>;
222 if constexpr (std::ratio_less_equal_v<Period2, period> && sizeof(Rep2) <= sizeof(rep)) {
223 return true; // finer (or equal) unit, no wider rep: flooring only shrinks the magnitude
224 } else if constexpr (std::ratio_less_equal_v<Period2, period>) {
225 // Floor in the wider Rep2 (flooring straight into Duration would narrow before we could check).
226 const auto f = std::chrono::floor<std::chrono::duration<Rep2, period>>(d);
227 return f.count() >= std::numeric_limits<rep>::min() && f.count() <= std::numeric_limits<rep>::max();
228 } else {
229 constexpr D2 hi = std::chrono::floor<D2>(Duration::max());
230 constexpr D2 lo = std::chrono::ceil<D2>(Duration::min());
231 return d >= lo && d <= hi;
232 }
233 }

270 [[nodiscard]] static constexpr utc from_timespec(const std::timespec& ts) noexcept
271 {
272 const auto s = static_cast<std::int64_t>(ts.tv_sec);
273 const auto ns = static_cast<std::int64_t>(ts.tv_nsec);
274 if constexpr (std::ratio_less_equal_v<period, std::ratio<1>>) {
275 // A negative second with a fraction is (s + 1) s minus the rest, so an instant in the last second
276 // above min() never passes through the whole second s, which Duration cannot hold (found by the
277 // portability tests: utc<>::min().to_timespec() did not round-trip).
278 if (s < 0 && ns > 0)
279 return from_unix(std::chrono::duration<std::int64_t>(s + 1)) -
280 std::chrono::ceil<Duration>(std::chrono::nanoseconds(1'000'000'000 - ns));
281 }
282 return from_unix(std::chrono::duration<std::int64_t>(s)) + std::chrono::floor<Duration>(std::chrono::nanoseconds(ns));
283 }
