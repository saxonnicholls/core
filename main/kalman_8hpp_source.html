220 template <auto Kinds, typename T, std::size_t N, std::size_t M>
221 constexpr result<kalman_update<T, N, M>, linalg::error> core(const gaussian<T, N>& prediction, const linalg::fixed_vector<T, M>& z,
222 const measurement_prediction<T, N, M>& mp) noexcept(linalg::detail::nothrow_scalar<T>)
223 {
224 const auto l = linalg::cholesky(mp.covariance);
225 // The update is built inside the result, the only object returned, so it is never copied
226 result<kalman_update<T, N, M>, linalg::error> out{std::in_place};
227 if (!l) {
228 out = err(l.unwrap_err_unchecked());
229 return out;
230 }
231 kalman_update<T, N, M>& u = *out;
232 u.innovation = geometry::residual<Kinds>(z, mp.mean);
233 u.innovation_covariance = mp.covariance;
234 u.gain = linalg::transpose(linalg::cholesky_solve(*l, linalg::transpose(mp.cross_covariance)));
235 u.distance2 = whitened_squared_norm(u.innovation, *l);
236 const T log_2pi = static_cast<T>(math::log(T(2) * geometry::pi<T>));
237 u.log_likelihood = -(u.distance2 + linalg::cholesky_log_det(*l) + static_cast<T>(M) * log_2pi) / T(2);
238 u.posterior.mean = prediction.mean + u.gain * u.innovation;
239 u.posterior.covariance = linalg::symmetrise(prediction.covariance - linalg::sandwich(u.gain, mp.covariance));
240 return out;
241 }

269 template <measurement_model S>
270 [[nodiscard]] constexpr result<kalman_update<typename S::value_type, S::state_dim, S::measurement_dim>, linalg::error> update(const gaussian<typename S::value_type, S::state_dim>& prediction,
271 const linalg::fixed_vector<typename S::value_type, S::measurement_dim>& z, const S& sensor,
272 covariance_form form = covariance_form::standard) noexcept(linalg::detail::nothrow_scalar<typename S::value_type>)
273 {
274 using T = typename S::value_type;
275 constexpr std::size_t N = S::state_dim;
276 auto u = detail::kalman::core<S::kinds>(prediction, z, predict_measurement(prediction, sensor));
277 if (u && form == covariance_form::joseph) {
278 const auto h = sensor.jacobian(prediction.mean);
279 const auto ikh = linalg::fixed_matrix<T, N, N>::identity() - u->gain * h;
280 u->posterior.covariance = linalg::symmetrise(linalg::sandwich(ikh, prediction.covariance) + linalg::sandwich(u->gain, sensor.covariance()));
281 }
282 return u;
283 }

374 template <linalg::scalar T, std::size_t N>
375 [[nodiscard]] constexpr result<gaussian<T, N>, linalg::error> rts_smooth(const gaussian<T, N>& filtered, const gaussian<T, N>& predicted, const gaussian<T, N>& smoothed_next,
376 const linalg::fixed_matrix<T, N, N>& f) noexcept(linalg::detail::nothrow_scalar<T>)
377 {
378 const auto l = linalg::cholesky(predicted.covariance);
379 if (!l)
380 return err(l.unwrap_err_unchecked());
381 const auto c = linalg::transpose(linalg::cholesky_solve(*l, f * filtered.covariance)); // P_f Fᵀ P_p⁻¹
382 return ok(gaussian<T, N>{filtered.mean + c * (smoothed_next.mean - predicted.mean),
383 linalg::symmetrise(filtered.covariance + c * (smoothed_next.covariance - predicted.covariance) * linalg::transpose(c))});
384 }

395 template <std::size_t M, typename F, linalg::scalar T, std::size_t N>
396 [[nodiscard]] constexpr linalg::fixed_matrix<T, M, N> numerical_jacobian(F&& f, const linalg::fixed_vector<T, N>& x)
397 {
398 const T step = static_cast<T>(math::pow(std::numeric_limits<T>::epsilon(), T(1) / T(3)));
399 linalg::fixed_matrix<T, M, N> j{};
400 for (std::size_t k = 0; k < N; ++k) {
401 const T ax = x[k] < T(0) ? -x[k] : x[k];
402 const T h = step * (ax > T(1) ? ax : T(1));
403 auto xp = x, xm = x;
404 xp[k] += h;
405 xm[k] -= h;
406 const linalg::fixed_vector<T, M> d = (f(xp) - f(xm)) / ((xp[k] - xm[k]));
407 for (std::size_t i = 0; i < M; ++i)
408 j(i, k) = d[i];
409 }
410 return j;
411 }

427 template <std::array Pos, std::array Vel, linalg::scalar T, std::size_t N, std::size_t D>
428 [[nodiscard]] constexpr linalg::fixed_vector<T, N> alpha_beta_update(const linalg::fixed_vector<T, N>& x, const linalg::fixed_vector<T, D>& z, T alpha, T beta, T dt) noexcept(linalg::detail::nothrow_scalar<T>)
429 {
430 static_assert(Pos.size() == D && Vel.size() == D, "snicholls::tracking::alpha_beta_update: one position and one velocity index per measured component");
431 auto y = x;
432 for (std::size_t i = 0; i < D; ++i) {
433 const auto p = static_cast<std::size_t>(Pos[i]), v = static_cast<std::size_t>(Vel[i]);
434 const T r = z[i] - x[p];
435 y[p] = x[p] + alpha * r;
436 y[v] = x[v] + beta / dt * r;
437 }
438 return y;
439 }
