91 // Indices in [0, n) from 32-bit halves of 64-bit draws: two per call of the generator.
92 template <full_range_generator G>
93 class index_draws {
94 public:
95 constexpr index_draws(G& g, std::uint32_t n) noexcept : g_(&g), n_(n) {}
96
97 constexpr std::uint32_t next() noexcept(noexcept(std::declval<G&>()()))
98 {
99 return uniform_index32(bits32(), n_, [this] { return bits32(); });
100 }
101
102 // A whole resample with the state in locals: stores to `out` cannot then alias it.
103 constexpr void fill(std::span<std::uint32_t> out) noexcept(noexcept(std::declval<G&>()()))
104 {
105 G& g = *g_;
106 const std::uint32_t n = n_;
107 std::uint64_t bits = bits_;
108 bool have = have_;
109 const auto next32 = [&]() -> std::uint32_t {
110 if (have) {
111 have = false;
112 return static_cast<std::uint32_t>(bits >> 32);
113 }
114 bits = random_u64(g);
115 have = true;
116 return static_cast<std::uint32_t>(bits);
117 };
118 for (std::uint32_t& i : out)
119 i = uniform_index32(next32(), n, next32);
120 bits_ = bits;
121 have_ = have;
122 }
123
124 constexpr std::uint32_t bits32() noexcept(noexcept(std::declval<G&>()()))
125 {
126 if (have_) {
127 have_ = false;
128 return static_cast<std::uint32_t>(bits_ >> 32);
129 }
130 bits_ = random_u64(*g_);
131 have_ = true;
132 return static_cast<std::uint32_t>(bits_);
133 }
134
135 private:
136 G* g_;
137 std::uint32_t n_;
138 std::uint64_t bits_ = 0;
139 bool have_ = false;
140 };

142 // Poisson(1) by inversion on 32 bits: thresholds floor(F(k) 2^32), the last a sentinel.
143 consteval std::array<std::uint64_t, 13> poisson1_thresholds() noexcept
144 {
145 double e_inverse = 0, term = 1; // e^-1 = sum (-1)^j / j!
146 for (int j = 0; j < 25; ++j) {
147 e_inverse += (j % 2 == 0 ? term : -term);
148 term /= j + 1;
149 }
150 std::array<std::uint64_t, 13> t{};
151 double p = e_inverse, cdf = 0;
152 for (std::size_t k = 0; k < 12; ++k) {
153 cdf += p;
154 p /= static_cast<double>(k + 1);
155 t[k] = static_cast<std::uint64_t>(cdf * 4294967296.0);
156 }
157 t[12] = std::uint64_t{1} << 32;
158 return t;
159 }
160 inline constexpr std::array<std::uint64_t, 13> poisson1 = poisson1_thresholds();
161
162 // Branch-free to a count of 5, which is 99.94% of draws; above that a loop, so rare that it is
163 // always predicted not taken.
164 constexpr std::uint32_t poisson1_from(std::uint32_t bits) noexcept
165 {
166 const std::uint64_t b = bits;
167 std::uint32_t k = 0;
168 for (std::size_t j = 0; j < 5; ++j)
169 k += static_cast<std::uint32_t>(b >= poisson1[j]);
170 if (b >= poisson1[4]) [[unlikely]]
171 while (b >= poisson1[k])
172 ++k;
173 return k;
174 }

290 template <full_range_generator G, typename F>
291 void for_each_poisson_resample(std::size_t n, std::size_t resamples, G& g, F&& f)
292 {
293 assert(n > 0);
294 std::vector<std::uint32_t> counts(n);
295 for (std::size_t b = 0; b < resamples; ++b) {
296 std::size_t k = 0;
297 for (; k + 1 < n; k += 2) {
298 const std::uint64_t bits = random_u64(g);
299 counts[k] = detail::poisson1_from(static_cast<std::uint32_t>(bits));
300 counts[k + 1] = detail::poisson1_from(static_cast<std::uint32_t>(bits >> 32));
301 }
302 if (k < n)
303 counts[k] = detail::poisson1_from(static_cast<std::uint32_t>(random_u64(g)));
304 f(std::span<const std::uint32_t>(counts));
305 }
306 }

343 template <full_range_generator G, typename F>
344 void for_each_stationary_resample(std::size_t n, double mean_block_length, std::size_t resamples, G& g, F&& f)
345 {
346 assert(n > 0 && n <= 0xFFFFFFFFu && mean_block_length >= 1.0);
347 std::vector<std::uint32_t> indices(n);
348 detail::index_draws<G> draws(g, static_cast<std::uint32_t>(n));
349 const auto restart = static_cast<std::uint64_t>((1.0 / mean_block_length) * 4294967296.0);
350 for (std::size_t b = 0; b < resamples; ++b) {
351 std::uint32_t current = draws.next();
352 indices[0] = current;
353 for (std::size_t k = 1; k < n; ++k) {
354 if (draws.bits32() < restart)
355 current = draws.next();
356 else
357 current = current + 1 == n ? 0 : current + 1;
358 indices[k] = current;
359 }
360 f(std::span<const std::uint32_t>(indices));
361 }

374 template <full_range_generator G, typename F>
375 void for_each_block_resample(std::size_t n, std::size_t block_length, std::size_t resamples, G& g, F&& f)
376 {
377 assert(n > 0 && n <= 0xFFFFFFFFu && block_length >= 1 && block_length <= n);
378 std::vector<std::uint32_t> indices(n);
379 detail::index_draws<G> draws(g, static_cast<std::uint32_t>(n - block_length + 1));
380 for (std::size_t b = 0; b < resamples; ++b) {
381 for (std::size_t k = 0; k < n;) {
382 const std::uint32_t start = draws.next();
383 for (std::size_t j = 0; j < block_length && k < n; ++j, ++k)
384 indices[k] = static_cast<std::uint32_t>(start + j);
385 }
386 f(std::span<const std::uint32_t>(indices));
387 }

406 [[nodiscard]] inline block_lengths optimal_block_length(std::span<const double> x)
407 {
408 const std::size_t n = x.size();
409 assert(n >= 8);
410 double mean = 0;
411 for (const double v : x)
412 mean += v;
413 mean /= static_cast<double>(n);
414 std::vector<double> eps(n);
415 for (std::size_t i = 0; i < n; ++i)
416 eps[i] = x[i] - mean;
417 const double nd = static_cast<double>(n);
418 const double b_max = std::ceil(std::min(3 * std::sqrt(nd), nd / 3));
419 const std::size_t kn = std::max<std::size_t>(5, static_cast<std::size_t>(std::log10(nd)));
420 const std::size_t m_max = static_cast<std::size_t>(std::ceil(std::sqrt(nd))) + kn;
421 const double cv = 2 * std::sqrt(std::log10(nd) / nd);
422 std::vector<double> acv(m_max + 1), abs_acorr(m_max + 1);
423 std::optional<std::size_t> opt_m;
424 const auto dot = [&](std::size_t a0, std::size_t b0, std::size_t len) {
425 double s = 0;
426 for (std::size_t k = 0; k < len; ++k)
427 s += eps[a0 + k] * eps[b0 + k];
428 return s;
429 };
430 for (std::size_t i = 0; i <= m_max && i + 1 < n; ++i) {
431 const double v1 = dot(i + 1, i + 1, n - i - 1);
432 const double v2 = dot(0, 0, n - i - 1);
433 const double cross_prod = dot(i, 0, n - i);
434 acv[i] = cross_prod / nd;
435 abs_acorr[i] = std::abs(cross_prod) / std::sqrt(v1 * v2);
436 if (i >= kn && !opt_m) {
437 bool insignificant = true;
438 for (std::size_t k = i - kn; k < i; ++k)
439 insignificant = insignificant && abs_acorr[k] < cv;
440 if (insignificant)
441 opt_m = i - kn;
442 }
443 }
444 std::size_t m = opt_m ? 2 * std::max<std::size_t>(*opt_m, 1) : m_max;
445 m = std::min(m, m_max);
446 double g = 0, lr_acv = acv[0];
447 for (std::size_t k = 1; k <= m; ++k) {
448 const double ratio = static_cast<double>(k) / static_cast<double>(m);
449 const double lambda = ratio <= 0.5 ? 1.0 : 2 * (1 - ratio);
450 g += 2 * lambda * static_cast<double>(k) * acv[k];
451 lr_acv += 2 * lambda * acv[k];
452 }
453 const double d_sb = 2 * lr_acv * lr_acv;
454 const double d_cb = 4.0 / 3.0 * lr_acv * lr_acv;
455 block_lengths out;
456 out.stationary = std::min(std::cbrt(2 * g * g / d_sb) * std::cbrt(nd), b_max);
457 out.circular = std::min(std::cbrt(2 * g * g / d_cb) * std::cbrt(nd), b_max);
458 return out;
459 }

465 struct bootstrap_distribution {
466 double estimate = 0;
467 std::vector<double> replicates;
468
471 [[nodiscard]] double mean() const noexcept
472 {
473 if (replicates.empty())
474 return std::nan("");
475 return std::accumulate(replicates.begin(), replicates.end(), 0.0) / static_cast<double>(replicates.size());
476 }
477
480 [[nodiscard]] double standard_error() const noexcept
481 {
482 if (replicates.size() < 2)
483 return std::nan("");
484 const double m = mean();
485 double sum = 0;
486 for (const double r : replicates)
487 sum += (r - m) * (r - m);
488 return std::sqrt(sum / static_cast<double>(replicates.size() - 1));
489 }
490
493 [[nodiscard]] double bias() const noexcept { return mean() - estimate; }
494
499 [[nodiscard]] double quantile(double q) const
500 {
501 if (replicates.empty())
502 return std::nan("");
503 std::vector<double> sorted = replicates;
504 std::sort(sorted.begin(), sorted.end());
505 const double h = q * static_cast<double>(sorted.size() - 1);
506 const auto lo = static_cast<std::size_t>(std::floor(h));
507 const std::size_t hi = std::min(lo + 1, sorted.size() - 1);
508 return sorted[lo] + (h - static_cast<double>(lo)) * (sorted[hi] - sorted[lo]);
509 }
510
515 [[nodiscard]] std::pair<double, double> percentile_interval(double confidence) const
516 {
517 return {quantile((1 - confidence) / 2), quantile((1 + confidence) / 2)};
518 }
519
529 [[nodiscard]] std::pair<double, double> bca_interval(double confidence, std::span<const double> jackknife) const
530 {
531 double below = 0; // a replicate equal to the estimate counts half, as in SciPy
532 for (const double r : replicates)
533 below += r < estimate ? 1.0 : (r == estimate ? 0.5 : 0.0);
534 const double z0 = math::normal_quantile(below / static_cast<double>(replicates.size()));
535 double jmean = 0;
536 for (const double j : jackknife)
537 jmean += j;
538 jmean /= static_cast<double>(jackknife.size());
539 double num = 0, den = 0;
540 for (const double j : jackknife) {
541 const double d = jmean - j;
542 num += d * d * d;
543 den += d * d;
544 }
545 const double a = den > 0 ? num / (6 * std::pow(den, 1.5)) : 0.0;
546 const auto adjusted = [&](double alpha) {
547 const double z = math::normal_quantile(alpha);
548 return math::normal_cdf(z0 + (z0 + z) / (1 - a * (z0 + z)));
549 };
550 const double lo = adjusted((1 - confidence) / 2), hi = adjusted((1 + confidence) / 2);
551 if (std::isnan(lo) || std::isnan(hi))
552 return {std::nan(""), std::nan("")};
553 return {quantile(lo), quantile(hi)};
554 }
555
560 [[nodiscard]] std::pair<double, double> basic_interval(double confidence) const
561 {
562 const auto [lo, hi] = percentile_interval(confidence);
563 return {2 * estimate - hi, 2 * estimate - lo};
564 }
565 };

569 struct weighted_moments {
570 double weight = 0;
571 double mean = 0;
572 double variance = 0;
573
576 [[nodiscard]] double standard_deviation() const noexcept { return std::sqrt(variance); }
577 };
578
585 [[nodiscard]] inline weighted_moments moments_of(std::span<const double> data, std::span<const std::uint32_t> counts, double centre = 0) noexcept
586 {
587 assert(data.size() == counts.size());
588 double w = 0, s1 = 0, s2 = 0;
589 for (std::size_t i = 0; i < data.size(); ++i) {
590 const double c = static_cast<double>(counts[i]);
591 const double x = data[i] - centre;
592 w += c;
593 s1 += c * x;
594 s2 += c * x * x;
595 }
596 weighted_moments m;
597 m.weight = w;
598 m.mean = w > 0 ? centre + s1 / w : std::nan("");
599 m.variance = w > 1 ? (s2 - s1 * s1 / w) / (w - 1) : 0.0;
600 return m;
601 }

621 template <full_range_generator G, typename F>
622 [[nodiscard]] bootstrap_distribution bootstrap_moments(std::span<const double> data, std::size_t resamples, G& g, F&& statistic,
623 resampling how = resampling::multinomial)
624 {
625 bootstrap_distribution out;
626 const std::vector<std::uint32_t> ones(data.size(), 1u);
627 const double centre = data.empty() ? 0.0 : std::accumulate(data.begin(), data.end(), 0.0) / static_cast<double>(data.size());
628 out.estimate = statistic(moments_of(data, ones, centre));
629 out.replicates.reserve(resamples);
630 const auto collect = [&](std::span<const std::uint32_t> counts) { out.replicates.push_back(statistic(moments_of(data, counts, centre))); };
631 if (how == resampling::poisson)
632 for_each_poisson_resample(data.size(), resamples, g, collect);
633 else
634 for_each_resample_counts(data.size(), resamples, g, collect);
635 return out;
636 }

646 template <std::ranges::random_access_range R, typename Statistic>
647 [[nodiscard]] std::vector<double> jackknife(const R& data, Statistic&& statistic)
648 {
649 using T = std::ranges::range_value_t<R>;
650 const std::vector<T> all(std::ranges::begin(data), std::ranges::end(data));
651 std::vector<T> without(all.size() - 1);
652 std::vector<double> out;
653 out.reserve(all.size());
654 for (std::size_t i = 0; i < all.size(); ++i) {
655 std::copy(all.begin(), all.begin() + static_cast<std::ptrdiff_t>(i), without.begin());
656 std::copy(all.begin() + static_cast<std::ptrdiff_t>(i) + 1, all.end(), without.begin() + static_cast<std::ptrdiff_t>(i));
657 out.push_back(statistic(std::span<const T>(without)));
658 }
659 return out;
660 }

673 template <std::ranges::random_access_range R, full_range_generator G, typename Statistic>
674 [[nodiscard]] bootstrap_distribution bootstrap(const R& data, std::size_t resamples, G& g, Statistic&& statistic)
675 {
676 using T = std::ranges::range_value_t<R>;
677 const std::vector<T> original(std::ranges::begin(data), std::ranges::end(data));
678 bootstrap_distribution out;
679 out.estimate = statistic(std::span<const T>(original));
680 out.replicates.reserve(resamples);
681 std::vector<T> resample(original.size());
682 for_each_resample(original.size(), resamples, g, [&](std::span<const std::uint32_t> indices) {
683 for (std::size_t k = 0; k < indices.size(); ++k)
684 resample[k] = original[indices[k]];
685 out.replicates.push_back(statistic(std::span<const T>(resample)));
686 });
687 return out;
688 }
