103 [[nodiscard]] capability_result<void> verify(const wallet_signature_request& r) const
104 {
105 if (injected_) {
106 const capability_error e = *injected_;
107 injected_.reset();
108 return err(e);
109 }
110 if (!is_lowercase_evm_address(r.address))
111 return fail(capability_errc::invalid_argument, {}, "not a lowercase EVM address");
112 const auto it = wallets_.find(r.address);
113 if (it == wallets_.end() || r.signature.size() != signature_bytes)
114 return fail(capability_errc::signature_invalid);
115 const auto expected = test_signature(it->second.reveal(), r.message);
116 if (!constant_time_equal(std::span<const std::byte>(expected), r.signature))
117 return fail(capability_errc::signature_invalid);
118 return ok();
119 }

124 void fail_next(capability_errc code, std::string_view response = {}) noexcept

129 private:
130 static std::array<std::byte, signature_bytes> test_signature(std::string_view key, std::string_view message) noexcept
131 {
132 const auto h = personal_sign_hash(message);
133 const auto r = hmac<sha256>::mac(key, h);
134 const auto s = hmac<sha256>::mac(key, r);
135 std::array<std::byte, signature_bytes> out{};
136 for (std::size_t i = 0; i < 32; ++i) {
137 out[i] = r.bytes[i];
138 out[32 + i] = s.bytes[i];
139 }
140 out[64] = std::byte{0x1b};
141 return out;
142 }
