86 // Constrained with requires rather than by SFINAE in the return type: fsm/state_machine.hpp reads a generic
87 // guard's signature as `&F::template operator()<Event>`, and Apple Clang 17 crashes on that address when the

114 template <class L, class R>
115 concept pointwise_operands = pointwise_<L> || pointwise_<R>;

117 // An operand as a function: a pointwise's function, or a constant.
118 template <pointwise_ P>
119 constexpr auto as_function(P&& p) noexcept(nothrow_store_v<decltype((std::forward<P>(p).f))>)

121 return std::forward<P>(p).f;

144 template <bool And>
145 struct pointwise_logical_op {
146 template <std::size_t... I, class S, class... A>
147 static constexpr auto call(std::index_sequence<I...>, S&& s, A&&... a)
148 noexcept(noexcept(static_cast<bool>(std::invoke(detail::part<0>(s), a...))) &&
149 noexcept(static_cast<bool>(std::invoke(detail::part<1>(s), a...))))
150 -> decltype(static_cast<bool>(std::invoke(detail::part<0>(s), a...)) &&
151 static_cast<bool>(std::invoke(detail::part<1>(s), a...)))
152 {
153 if (static_cast<bool>(std::invoke(detail::part<0>(s), a...)) != And) return !And;
154 return static_cast<bool>(std::invoke(detail::part<1>(s), a...));
155 }
156 };
