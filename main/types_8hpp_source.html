116 // Whether T's arithmetic, comparisons and copies cannot throw: true for float and double, false for an
117 // allocating multiprecision type, whose bad_alloc must propagate rather than terminate.
118 template <typename T>
119 inline constexpr bool nothrow_scalar = std::is_nothrow_copy_constructible_v<T> && std::is_nothrow_copy_assignable_v<T> &&
120 noexcept(std::declval<T>() + std::declval<T>()) && noexcept(std::declval<T>() - std::declval<T>()) &&
121 noexcept(std::declval<T>() * std::declval<T>()) && noexcept(std::declval<T>() / std::declval<T>()) &&
122 noexcept(std::declval<T>() < std::declval<T>()) && noexcept(T(0));

124 // Where a view or a span starts, for the aliasing test below.
125 template <typename T, std::size_t R, typename L>
126 constexpr const void* start_of(const md_span<T, R, L>& x) noexcept
127 {
128 return x.data_handle();
129 }
130 template <typename T, std::size_t E>
131 constexpr const void* start_of(const std::span<T, E>& x) noexcept
132 {
133 return x.data();
134 }

136 // Whether two non-empty views or spans start at the same element: the cheap aliasing test of the free functions.
137 // Views that overlap some other way (a sub-view, interleaved strides) are a precondition violation, not detected.
138 template <typename X, typename Y>
139 constexpr bool same_storage(const X& x, const Y& y) noexcept
140 {
141 return !x.empty() && !y.empty() && start_of(x) == start_of(y);
142 }
