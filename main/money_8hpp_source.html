116 // Whether q + r/d, of the given sign, rounds up in magnitude (to q + 1) under `mode`. 0 <= r < d.
117 [[nodiscard]] constexpr bool rounds_away(const uint256& q, const uint256& r, const uint256& d, bool negative,
118 rounding mode) noexcept
119 {
120 if (r.is_zero())
121 return false;
122 const uint256 rest = d - r; // r > rest: above half; r == rest: a tie
123 switch (mode) {
124 case rounding::toward_zero: return false;
125 case rounding::away_from_zero: return true;
126 case rounding::floor: return negative;
127 case rounding::ceiling: return !negative;
128 case rounding::half_up: return r >= rest;
129 case rounding::half_down: return r > rest;
130 case rounding::half_even: return r > rest || (r == rest && q.bit(0));
131 }
132 return false;
133 }

279 [[nodiscard]] constexpr std::optional<signed_magnitude> signed_add(bool an, const uint256& a, bool bn,
280 const uint256& b) noexcept
281 {

380 [[nodiscard]] constexpr result<money, money_errc> parse_money(const asset& a, std::string_view text,
381 std::optional<rounding> mode = std::nullopt) noexcept
382 {
383 bool negative = false;
384 if (!text.empty() && (text.front() == '-' || text.front() == '+')) {
385 negative = text.front() == '-';
386 text.remove_prefix(1);
387 }
388 const std::size_t point = text.find('.');
389 const std::string_view whole = text.substr(0, point);
390 const std::string_view fraction = point == std::string_view::npos ? std::string_view{} : text.substr(point + 1);
391 if (whole.empty() && fraction.empty())
392 return err(money_errc::malformed);
393 const auto is_digits = [](std::string_view s) {
394 for (const char c : s)
395 if (c < '0' || c > '9')
396 return false;
397 return true;
398 };
399 if (!is_digits(whole) || !is_digits(fraction))
400 return err(money_errc::malformed);
401 const unsigned scale = a.scale();
402 uint256 units;
403 const auto push = [&units](char c) {
404 const auto t = uint256::checked_mul(units, 10);
405 if (!t)
406 return false;
407 const auto s = uint256::checked_add(*t, static_cast<std::uint64_t>(c - '0'));
408 if (!s)
409 return false;
410 units = *s;
411 return true;
412 };
413 for (const char c : whole)
414 if (!push(c))
415 return err(money_errc::overflow);
416 for (std::size_t i = 0; i < fraction.size() && i < scale; ++i)
417 if (!push(fraction[i]))
418 return err(money_errc::overflow);
419 for (std::size_t i = fraction.size(); i < scale; ++i)
420 if (!push('0'))
421 return err(money_errc::overflow);
422 if (fraction.size() > scale) {
423 const std::string_view rest = fraction.substr(scale);
424 bool any = false;
425 for (const char c : rest)
426 any = any || c != '0';
427 if (any) {
428 if (!mode)
429 return err(money_errc::too_many_digits);
430 // the rest as a fraction r / d of one minor unit, compared without forming 10^len when it is long
431 const std::size_t len = rest.size() > 77 ? 77 : rest.size();
432 uint256 r;
433 for (std::size_t i = 0; i < len; ++i)
434 r = r * 10 + static_cast<std::uint64_t>(rest[i] - '0');
435 const uint256 d = uint256::pow10(static_cast<unsigned>(len));
436 bool beyond = false; // nonzero digits past the 77th make a tie impossible and r slightly larger
437 for (std::size_t i = len; i < rest.size(); ++i)
438 beyond = beyond || rest[i] != '0';
439 bool away = detail::rounds_away(units, r, d, negative, *mode);
440 if (beyond && r == d - r && (*mode == rounding::half_down || *mode == rounding::half_even))
441 away = true; // above the tie, not on it
442 if (away) {
443 const auto s = uint256::checked_add(units, 1);
444 if (!s)
445 return err(money_errc::overflow);
446 units = *s;
447 }
448 }
449 }
450 return ok(money::from_units(a, units, negative));
451 }

459 [[nodiscard]] constexpr result<money, money_errc> from_decimal(const asset& a, const decimal& d,
460 std::optional<rounding> mode = std::nullopt) noexcept
461 {
462 const bool negative = d.is_negative();
463 const std::int64_t m = d.mantissa();
464 const uint256 magnitude = m < 0 ? uint256(0ull - static_cast<std::uint64_t>(m)) : uint256(static_cast<std::uint64_t>(m));
465 const int shift = d.exponent() + static_cast<int>(a.scale()); // units = magnitude * 10^shift
466 if (shift >= 0) {
467 const auto units = uint256::checked_mul(magnitude, uint256::pow10(static_cast<unsigned>(shift)));
468 if (!units)
469 return err(money_errc::overflow);
470 return ok(money::from_units(a, *units, negative));
471 }
472 const uint256 divisor = uint256::pow10(static_cast<unsigned>(-shift));
473 auto [q, r] = magnitude.divmod(divisor);
474 if (!r.is_zero()) {
475 if (!mode)
476 return err(money_errc::too_many_digits);
477 if (detail::rounds_away(q, r, divisor, negative, *mode))
478 q = q + 1;
479 }
480 return ok(money::from_units(a, q, negative));
481 }

486 [[nodiscard]] constexpr std::optional<decimal> to_decimal(const money& m) noexcept
487 {
488 uint256 units = m.magnitude();
489 int exponent = -static_cast<int>(m.asset_of().scale());
490 while (!units.is_zero()) { // trailing zeros belong in the exponent
491 const auto [q, r] = units.divmod(10);
492 if (!r.is_zero())
493 break;
494 units = q;
495 ++exponent;
496 }
497 const auto small = units.to_uint64();
498 if (!small || *small > 999'999'999'999'999'999ull)
499 return std::nullopt;
500 const auto mantissa = static_cast<std::int64_t>(*small);
501 return decimal::from_parts(m.is_negative() ? -mantissa : mantissa, units.is_zero() ? 0 : exponent);
502 }

516 constexpr std::to_chars_result to_chars(char* first, char* last, const money& m,
517 money_digits digits = money_digits::minimal) noexcept
518 {
519 // the magnitude's decimal digits, least significant first
520 std::array<char, 80> rev{};
521 std::size_t n = 0;
522 uint256 v = m.magnitude();
523 const uint256 chunk = uint256::pow10(19);
524 do {
525 const auto [q, r] = v.divmod(chunk);
526 std::uint64_t part = r.low64();
527 const bool last_chunk = q.is_zero();
528 for (int i = 0; i < 19 && (part != 0 || !last_chunk); ++i) {
529 rev[n++] = static_cast<char>('0' + part % 10);
530 part /= 10;
531 }
532 v = q;
533 } while (!v.is_zero());
534 if (n == 0)
535 rev[n++] = '0';
536 const std::size_t scale = m.asset_of().scale();
537 while (n <= scale) // at least one digit before the point
538 rev[n++] = '0';
539 std::size_t frac = scale;
540 if (digits == money_digits::minimal)
541 while (frac > 0 && rev[scale - frac] == '0')
542 --frac;
543 const std::size_t whole = n - scale;
544 // the text in a local buffer first, then copied with a loop bounded by both ends: GCC's -Wstringop-overflow
545 // cannot see that a size check alone keeps the writes inside a small destination
546 std::array<char, 82> text{};
547 std::size_t size = 0;
548 if (m.is_negative())
549 text[size++] = '-';
550 for (std::size_t i = 0; i < whole; ++i)
551 text[size++] = rev[n - 1 - i];
552 if (frac > 0) {
553 text[size++] = '.';
554 for (std::size_t i = 0; i < frac; ++i)
555 text[size++] = rev[scale - 1 - i];
556 }
557 if (static_cast<std::size_t>(last - first) < size)
558 return {last, std::errc::value_too_large};
559 char* out = first;
560 for (std::size_t i = 0; i < size && out != last; ++i)
561 *out++ = text[i];
562 return {out, std::errc{}};
563 }
