174 constexpr bool ladder_add(money& a, const money& b, money_errc& e) noexcept
175 {
176 auto s = add(a, b);
177 if (!s) {
178 e = s.unwrap_err();
179 return false;
180 }
181 a = *s;
182 return true;
183 }

194 [[nodiscard]] constexpr result<cash_ladder, money_errc> make_cash_ladder(std::span<const cash_flow> flows,
195 const cash_ladder_terms& terms,
196 std::span<cash_rung> rungs) noexcept
197 {
198 const asset& a = terms.opening.asset_of();
199 if (terms.threshold.asset_of() != a)
200 return err(money_errc::asset_mismatch);
201 const money zero = money::zero(a);
202 cash_ladder out;
203 out.start = terms.opening;
204 out.overdue = {zero, zero, 0, 0};
205 money_errc e{};
206 std::size_t used = 0;
207
208 // the rung for date d, inserted in order when it is new; nullptr when the span is full
209 const auto rung_for = [&](std::chrono::sys_days d) -> cash_rung* {
210 std::size_t lo = 0, hi = used;
211 while (lo < hi) {
212 const std::size_t mid = lo + (hi - lo) / 2;
213 if (rungs[mid].on < d)
214 lo = mid + 1;
215 else
216 hi = mid;
217 }
218 if (lo < used && rungs[lo].on == d)
219 return &rungs[lo];
220 if (used == rungs.size())
221 return nullptr;
222 for (std::size_t i = used; i > lo; --i)
223 rungs[i] = rungs[i - 1];
224 rungs[lo] = cash_rung{d, zero, zero, zero, zero};
225 ++used;
226 return &rungs[lo];
227 };
228
229 for (const cash_flow& f : flows) {
230 if (f.amount.asset_of() != a)
231 return err(money_errc::asset_mismatch);
232 const bool in = !f.amount.is_negative();
233 std::chrono::sys_days on = f.on;
234 if (on < terms.as_of) {
235 if (!detail::ladder_add(in ? out.overdue.inflow : out.overdue.outflow, in ? f.amount : -f.amount, e))
236 return err(e);
237 ++(in ? out.overdue.inflows : out.overdue.outflows);
238 const overdue_treatment t = in ? terms.overdue.inflows : terms.overdue.outflows;
239 if (t == overdue_treatment::report_only)
240 continue;
241 if (t == overdue_treatment::count) {
242 if (!detail::ladder_add(out.start, f.amount, e))
243 return err(e);
244 continue;
245 }
246 on = terms.as_of; // due today
247 }
248 cash_rung* r = rung_for(on);
249 if (r == nullptr)
250 return err(money_errc::buffer_too_small);
251 if (!detail::ladder_add(in ? r->inflow : r->outflow, in ? f.amount : -f.amount, e))
252 return err(e);
253 }
254
255 money position = out.start;
256 out.lowest = position;
257 if (detail::ladder_below(position, terms.threshold))
258 out.first_below = terms.as_of;
259 for (std::size_t i = 0; i < used; ++i) {
260 cash_rung& r = rungs[i];
261 const auto net = subtract(r.inflow, r.outflow);
262 if (!net)
263 return err(net.unwrap_err());
264 r.net = *net;
265 if (!detail::ladder_add(position, r.net, e))
266 return err(e);
267 r.cumulative = position;
268 if (detail::ladder_below(position, out.lowest))
269 out.lowest = position;
270 if (!out.first_below && detail::ladder_below(position, terms.threshold))
271 out.first_below = r.on;
272 }
273 out.closing = position;
274 out.rungs = rungs.first(used);
275 return ok(out);
276 }
