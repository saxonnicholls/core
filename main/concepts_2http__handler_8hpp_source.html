99 [[nodiscard]] constexpr std::optional<http_method> parse_http_method(std::string_view s) noexcept
100 {
101 for (const auto m : {http_method::get, http_method::head, http_method::post, http_method::put, http_method::patch,

140 [[nodiscard]] inline http_response respond(int status, std::string body = {},
141 std::string_view content_type = "text/plain; charset=utf-8")
142 {
143 http_response r;
144 r.status = status;
145 if (!body.empty())
146 r.headers.push_back({"Content-Type", std::string{content_type}});
147 r.body = std::move(body);
148 return r;
149 }
