76 // NaN by its bits: the exponent all ones and a non-zero fraction. Not x != x, which MSVC's constant evaluation
77 // gets wrong.
78 constexpr bool is_nan(double x) noexcept
79 {
80 const auto bits = std::bit_cast<std::uint64_t>(x);
81 return (bits & 0x7ff0000000000000ull) == 0x7ff0000000000000ull && (bits & 0x000fffffffffffffull) != 0;
82 }

113 struct threshold_colours {
115 struct step {
116 double from;
117 colour tint;
118 };
119 colour below;
120 std::array<step, 8> steps{};
121 std::size_t count = 0;
122
126 consteval threshold_colours(colour below_first, std::initializer_list<step> from) : below(below_first)
127 {
128 if (from.size() > steps.size())
129 throw "snicholls::threshold_colours: at most eight cut points";
130 for (const step& st : from) {
131 if (count > 0 && !(st.from > steps[count - 1].from))
132 throw "snicholls::threshold_colours: cut points must ascend";
133 steps[count++] = st;
134 }
135 }
136
140 [[nodiscard]] constexpr std::optional<colour> of(double x) const noexcept
141 {
142 if (detail::display::is_nan(x))
143 return std::nullopt;
144 colour c = below;
145 for (std::size_t i = 0; i < count; ++i)
146 if (x >= steps[i].from)
147 c = steps[i].tint;
148 return c;
149 }
150 };

186 struct number_format {
187 std::string_view spec;
188
191 consteval number_format(std::string_view s) : spec(s) // NOLINT(google-explicit-constructor): number_format{"{:+.2f}"}
192 {
193 const auto digit = [](char c) { return c >= '0' && c <= '9'; };
194 const auto align = [](char c) { return c == '<' || c == '>' || c == '^'; };
195 if (s.size() < 2 || s.front() != '{' || s.back() != '}')
196 throw "snicholls::number_format: not one {...} replacement field";
197 std::string_view f = s.substr(1, s.size() - 2);
198 if (f.empty())
199 return;
200 if (f.front() != ':')
201 throw "snicholls::number_format: the field must be {} or start {:";
202 f.remove_prefix(1);
203 const auto take = [&f](auto pred) {
204 const bool took = !f.empty() && pred(f.front());
205 if (took)
206 f.remove_prefix(1);
207 return took;
208 };
209 if (f.size() >= 2 && align(f[1]) && f[0] != '{' && f[0] != '}')
210 f.remove_prefix(2);
211 else
212 take(align);
213 take([](char c) { return c == '+' || c == '-' || c == ' '; });
214 take([](char c) { return c == '#'; });
215 take([](char c) { return c == '0'; });
216 while (take(digit)) {
217 }
218 if (take([](char c) { return c == '.'; }) && !take(digit))
219 throw "snicholls::number_format: a precision needs digits";
220 while (take(digit)) {
221 }
222 take([](char c) { return c == 'L'; });
223 take([](char c) { return std::string_view{"aAeEfFgG"}.find(c) != std::string_view::npos; });
224 if (!f.empty())
225 throw "snicholls::number_format: not a std::format specification for a double";
226 }
227 };

369 inline void write_json_string(std::ostream& os, std::string_view s)
370 {
371 os << '"';
372 for (const char c : s) {
373 if (c == '"' || c == '\\')
374 os << '\\' << c;
375 else if (c == '\n')
376 os << "\\n";
377 else
378 os << c;
379 }
380 os << '"';
