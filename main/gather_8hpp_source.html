86template <typename B, math::math_number T>
87constexpr void gather_rows(const B&, std::type_identity_t<std::span<const T>> src, std::span<const std::uint32_t> indices,
88 std::span<T> out, std::size_t width, std::size_t sequences = 1) noexcept(std::is_floating_point_v<T>)
89{
90 assert(width > 0 && sequences > 0 && src.size() % (width * sequences) == 0 && indices.size() % sequences == 0
91 && out.size() == indices.size() * width);
92 const std::size_t n = src.size() / width / sequences, k = indices.size() / sequences;
93 for (std::size_t s = 0; s < sequences; ++s)
94 for (std::size_t j = 0; j < k; ++j) {
95 const std::size_t i = indices[s * k + j];
96 assert(i < n);
97 std::copy_n(src.begin() + std::ptrdiff_t((s * n + i) * width), width, out.begin() + std::ptrdiff_t((s * k + j) * width));
98 }
99}

150template <typename B, math::math_number T>
151constexpr void concat_rows(const B&, std::type_identity_t<std::span<const T>> first, std::type_identity_t<std::span<const T>> second,
152 std::span<T> out, [[maybe_unused]] std::size_t width, std::size_t sequences = 1) noexcept(std::is_floating_point_v<T>)
153{
154 assert(width > 0 && sequences > 0 && first.size() % (width * sequences) == 0 && second.size() % (width * sequences) == 0
155 && out.size() == first.size() + second.size());
156 const std::size_t a = first.size() / sequences, b = second.size() / sequences;
157 for (std::size_t s = 0; s < sequences; ++s) {
158 std::copy_n(first.begin() + std::ptrdiff_t(s * a), a, out.begin() + std::ptrdiff_t(s * (a + b)));
159 std::copy_n(second.begin() + std::ptrdiff_t(s * b), b, out.begin() + std::ptrdiff_t(s * (a + b) + a));
160 }
