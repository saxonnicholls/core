126template <transformer_config C, typename B>
127 requires(C.experts > 0)
128constexpr void moe_receive_offsets(const B&, std::span<const std::span<const std::uint32_t>> counts, std::size_t rank,
129 std::span<std::uint32_t> receive_offsets)
130{
131 const std::size_t ranks = counts.size(), first = first_expert(rank, C.experts, ranks), local = experts_per_rank(C.experts, ranks);
132 assert(ranks > 0 && C.experts % ranks == 0 && rank < ranks && receive_offsets.size() == ranks + 1);
133 receive_offsets[0] = 0;
134 for (std::size_t s = 0; s < ranks; ++s) {
135 assert(counts[s].size() == C.experts);
136 std::uint32_t n = 0;
137 for (std::size_t j = 0; j < local; ++j)
138 n += counts[s][first + j];
139 receive_offsets[s + 1] = receive_offsets[s] + n;
140 }
141}

150template <transformer_config C, typename B>
151 requires(C.experts > 0)
152constexpr void moe_receive_layout(const B&, std::span<const std::span<const std::uint32_t>> counts, std::size_t rank,
153 std::span<std::uint32_t> expert_offsets, std::span<std::uint32_t> order)
154{
155 const std::size_t ranks = counts.size(), first = first_expert(rank, C.experts, ranks), local = experts_per_rank(C.experts, ranks);
156 assert(ranks > 0 && C.experts % ranks == 0 && rank < ranks && expert_offsets.size() == local + 1);
157 std::uint32_t p = 0;
158 for (std::size_t j = 0; j < local; ++j) {
159 expert_offsets[j] = p;
160 std::uint32_t source_start = 0; // where source s's run starts in the received rows
161 for (std::size_t s = 0; s < ranks; ++s) {
162 std::uint32_t within = 0; // where expert first + j starts within source s's run
163 for (std::size_t i = 0; i < j; ++i)
164 within += counts[s][first + i];
165 for (std::uint32_t i = 0; i < counts[s][first + j]; ++i)
166 order[p++] = source_start + within + i;
167 for (std::size_t i = 0; i < local; ++i)
168 source_start += counts[s][first + i];
169 }
170 }
171 expert_offsets[local] = p;
172 assert(order.size() == p);
173}

185template <typename B, typename T>
186 requires std::is_trivially_copyable_v<T>
187constexpr void moe_all_to_all(const B&, std::span<const std::span<const T>> send, std::span<const std::span<const std::uint32_t>> send_offsets,
188 std::size_t width, std::span<const std::span<T>> receive)
189{
190 const std::size_t ranks = send.size();
191 assert(send_offsets.size() == ranks && receive.size() == ranks);
192 for (std::size_t r = 0; r < ranks; ++r) {
193 std::size_t at = 0;
194 for (std::size_t s = 0; s < ranks; ++s) {
195 assert(send_offsets[s].size() == ranks + 1);
196 for (std::size_t i = std::size_t(send_offsets[s][r]) * width; i < std::size_t(send_offsets[s][r + 1]) * width; ++i)
197 receive[r][at++] = send[s][i];
198 }
199 assert(at == receive[r].size());
200 }
201}
