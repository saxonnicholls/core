144 // A standard floating-point type's limits, from std::numeric_limits: what long double uses, whatever it is.
145 // The splits of ln 2 and pi/2 are double's, whose parts together carry about 85 and 119 bits.
146 template <typename T>
147 struct ieee_limits
148 {
149 using compute_type = T;
150 using L = std::numeric_limits<T>;
151 // Long double literals, not std::numbers: libc++ gives those only double's precision.
152 static constexpr T ln2_full = T(0.693147180559945309417232121458176568L);
153 static constexpr T pi = T(3.14159265358979323846264338327950288L);
154 static constexpr T inv_sqrt_pi = T(0.564189583547756286948079451560772586L);
155 static constexpr T two_over_sqrt_pi = 2 * inv_sqrt_pi;
156 static constexpr T sqrt2 = T(1.41421356237309504880168872420969808L);
157 static constexpr T inv_sqrt2 = T(0.707106781186547524400844362104849039L);
158 static constexpr T sqrt_2pi = T(2.50662827463100050241576528481104525L);
159 static constexpr T half_log_2pi = T(0.918938533204672741780329736405617640L);
160 static constexpr T ln2_hi = math_limits<double>::ln2_hi;
161 static constexpr T ln2_lo = math_limits<double>::ln2_lo;
162 static constexpr T ln2_lo2 = T(1.16122272293625324218337828526629425e-26L); // ln 2 - ln2_hi - ln2_lo
163 static constexpr T inv_ln2 = T(1.44269504088896340735992468100189214L);
164 // Up to 64 bits, double's three Cody-Waite parts (about 100 bits of pi/2) suffice. Wider (binary128) takes pi/2
165 // as four non-overlapping doubles, about 212 bits: each k * part is exact in 113 bits.
166 static constexpr bool wide = L::digits > 64;
167 static constexpr T pio2_1 = wide ? T(0x1.921fb54442d18p+0) : T(math_limits<double>::pio2_1);
168 static constexpr T pio2_2 = wide ? T(0x1.1a62633145c07p-54) : T(math_limits<double>::pio2_2);
169 static constexpr T pio2_3 = wide ? T(-0x1.f1976b7ed8fbcp-110) : T(math_limits<double>::pio2_3);
170 static constexpr T pio2_4 = wide ? T(0x1.4cf98e804177dp-164) : T(0);
171 static constexpr T inv_pio2 = T(0.636619772367581343075535053490057448L);
172 static constexpr T exp_overflow = T(L::max_exponent) * ln2_full;
173 static constexpr T exp_underflow = T(L::min_exponent - L::digits - 1) * ln2_full;
174 static constexpr T whole_from = [] {
175 T p = 1;
176 for (int i = 0; i < L::digits; ++i)
177 p *= 2;
178 return p;
179 }();
180 static constexpr T integral_below = T(4.0e18);
181 static constexpr T series_tolerance = L::epsilon() * T(1e-4);
182 static constexpr T trig_tolerance = L::epsilon() * T(1e-6);
183 static constexpr T gamma_tolerance = L::epsilon() / 20;
184 static constexpr T tiny = L::min() / L::epsilon();
185 static constexpr T tanh_one = T(L::digits) * ln2_full / 2 + 1;
186 static constexpr T erf_one = L::digits <= 53 ? T(6.5) : (L::digits <= 64 ? T(6.7) : T(9.0));
187 static constexpr T erfc_zero = L::digits <= 53 ? T(27) : T(107);
188 static constexpr int erfc_terms = L::digits <= 53 ? 60 : (L::digits <= 64 ? 100 : 400);

190 // Halve the exponent by powers of four, then (1 + m) / 2 is within 25% of sqrt(m) for m in [1, 4).
191 static constexpr T sqrt_seed(T x) noexcept
192 {
193 T y = 1;
194 constexpr T big = T(18446744073709551616.0L), half_big = T(4294967296.0L); // 2^64, 2^32
195 while (x >= big) { x /= big; y *= half_big; }
196 while (x < 1 / big) { x *= big; y /= half_big; }
197 while (x >= 4) { x /= 4; y *= 2; }
198 while (x < 1) { x *= 4; y /= 2; }
199 return y * (1 + x) / 2;
200 }
201 static constexpr int sqrt_steps = 8;
202 };

233 template <typename T>
234 constexpr bool nan_(T x) { return x != x; }
235 template <typename T>
236 constexpr bool inf_(T x) { return x == std::numeric_limits<T>::infinity() || x == -std::numeric_limits<T>::infinity(); }
237 template <typename T>
238 constexpr T abs_(T x) { return x < 0 ? -x : x; }

240 template <typename T>
241 constexpr T trunc_(T x)
242 {
243 if (nan_(x) || inf_(x))
244 return x;
245 T a = abs_(x);
246 if (a >= limits<T>::whole_from) // already integral
247 return x;
248 // Wider than long long: take off the whole multiple of 2^32 first, then the rest fits
249 const T high = a >= T(4611686018427387904.0L) ? T(static_cast<long long>(a / T(4294967296.0L))) * T(4294967296.0L) : T(0);
250 T r = high + static_cast<T>(static_cast<long long>(a - high));
251 return x < 0 ? -r : r;
252 }

254 template <typename T>
255 constexpr T round_(T x)
256 {
257 // x - trunc(x) is exact, so the half is judged on x itself: trunc(x + 0.5) would round 0.49999997f up
258 const T t = trunc_(x);
259 const T frac = x - t;
260 return frac >= T(0.5) ? t + T(1) : frac <= T(-0.5) ? t - T(1) : t;
261 }

263 template <typename T>
264 constexpr bool integral_(T x)
265 {
266 return !nan_(x) && !inf_(x) && x == trunc_(x) && abs_(x) < limits<T>::integral_below;

269// x * 2^k without <cmath>. k is at most ~2100 in magnitude for finite results.
270template <typename T>
271constexpr T ldexp2(T x, long long k)
272{
273 while (k > 0)
274 {
275 x *= T(2.0);
276 --k;
277 if (inf_(x))
278 return x;
279 }
280 while (k < 0)
281 {
282 x *= T(0.5);
283 ++k;
284 if (x == T(0.0))
285 return x;
286 }
287 return x;
288}

290template <typename T>
291constexpr T exp_ct(T x)
292{
293 using L = limits<T>;
294 if (nan_(x)) return x;
295 if (x > L::exp_overflow) return std::numeric_limits<T>::infinity();
296 if (x < L::exp_underflow) return T(0.0);
297
298 const long long k = static_cast<long long>(round_(x * L::inv_ln2));
299 const T kd = static_cast<T>(k);
300 const T r = ((x - kd * L::ln2_hi) - kd * L::ln2_lo) - kd * L::ln2_lo2; // |r| <= ~0.347
301
302 T term = T(1.0);
303 T sum = T(1.0);
304 for (int n = 1; n < 40; ++n)
305 {
306 term *= r / T(n);
307 sum += term;
308 if (abs_(term) < L::series_tolerance * abs_(sum))
309 break;
310 }
311 return ldexp2(sum, k);
312}

314template <typename T>
315constexpr T log_ct(T x)
316{
317 using L = limits<T>;
318 if (nan_(x)) return x;
319 if (x < T(0.0)) return std::numeric_limits<T>::quiet_NaN();
320 if (x == T(0.0)) return -std::numeric_limits<T>::infinity();
321 if (inf_(x)) return x;
322
323 // Normalize x = m * 2^e with m in [1, 2), then center on 1.
324 T m = x;
325 long long e = 0;
326 while (m >= T(2.0)) { m *= T(0.5); ++e; }
327 while (m < T(1.0)) { m *= T(2.0); --e; }
328 if (m > L::sqrt2)
329 {
330 m *= T(0.5);
331 ++e;
332 }
333
334 // ln m = 2 atanh((m-1)/(m+1))
335 const T t = (m - T(1.0)) / (m + T(1.0));
336 const T t2 = t * t;
337 T term = t;
338 T sum = t;
339 for (int n = 3; n < 60; n += 2)
340 {
341 term *= t2;
342 const T contrib = term / T(n);
343 sum += contrib;
344 if (abs_(contrib) < L::series_tolerance * abs_(sum))
345 break;
346 }
347 const T lnm = T(2.0) * sum;
348 const T ed = static_cast<T>(e);
349 return ed * L::ln2_hi + ((lnm + ed * L::ln2_lo) + ed * L::ln2_lo2);
350}

352template <typename T>
353constexpr T sin_series(T r)
354{
355 const T r2 = r * r;
356 T term = r;
357 T sum = r;
358 for (int n = 1; n < 20; ++n)
359 {
360 term *= -r2 / ((T(2.0) * T(n)) * (T(2.0) * T(n) + T(1.0)));
361 sum += term;
362 if (abs_(term) < limits<T>::trig_tolerance)
363 break;
364 }
365 return sum;
366}

368template <typename T>
369constexpr T cos_series(T r)
370{
371 const T r2 = r * r;
372 T term = T(1.0);
373 T sum = T(1.0);
374 for (int n = 1; n < 20; ++n)
375 {
376 term *= -r2 / ((T(2.0) * T(n) - T(1.0)) * (T(2.0) * T(n)));

378 if (abs_(term) < limits<T>::trig_tolerance)

384// Reduce x to r with x = k*(pi/2) + r, |r| <= pi/4; writes quadrant k mod 4.
385template <typename T>
386struct reduced { T r; int quadrant; };
387
388template <typename T>
389constexpr reduced<T> reduce_pio2(T x)
390{
391 using L = limits<T>;
392 const T kd = round_(x * L::inv_pio2);
393 const long long k = static_cast<long long>(kd);
394 const T r = (((x - kd * L::pio2_1) - kd * L::pio2_2) - kd * L::pio2_3) - kd * L::pio2_4;
395 int q = static_cast<int>(k % 4);
396 if (q < 0)
397 q += 4;
398 return {r, q};
399}

403{

429template <typename T>
430constexpr T sqrt_ct(T x)
431{
432 if (nan_(x)) return x;
433 if (x < T(0.0)) return std::numeric_limits<T>::quiet_NaN();
434 if (x == T(0.0) || inf_(x)) return x;
435
436 // An estimate from the type's limits, then Newton to full precision.
437 T y = limits<T>::sqrt_seed(x);
438 for (int i = 0; i < limits<T>::sqrt_steps; ++i)
439 y = T(0.5) * (y + x / y);
440 return y;
441}

443// pow_int is TOTAL: it never divides by zero. `1.0 / pow_int(0.0, n)` is a
444// division by zero, which is UB and therefore a hard error in a constant
445// expression. The
446// result of 0^(-n) is a pole, so the honest kernel answer is +inf, which is
447// also exactly what the run-time std::pow returns. Keeping the compile-time
448// and run-time paths in agreement is this header's contract.
449template <typename T>
450constexpr T pow_int(T a, long long n)
451{
452 if (n < 0)
453 {
454 const T d = pow_int(a, -n);
455 return d == T(0.0) ? std::numeric_limits<T>::infinity() : T(1.0) / d;
456 }
457 T result = T(1.0);
458 T base = a;
459 while (n > 0)
460 {
461 if (n & 1)
462 result *= base;
463 base *= base;
464 n >>= 1;
465 }
466 return result;
467}

469template <typename T>
470constexpr T pow_ct(T a, T b)
471{
472 if (b == T(0.0)) return T(1.0);
473 if (a == T(1.0)) return T(1.0);
474 if (nan_(a) || nan_(b)) return std::numeric_limits<T>::quiet_NaN();
475 if (integral_(b))
476 return pow_int(a, static_cast<long long>(b));
477 if (a < T(0.0)) return std::numeric_limits<T>::quiet_NaN(); // non-integral power of a negative base
478 if (a == T(0.0)) return b > T(0.0) ? T(0.0) : std::numeric_limits<T>::infinity();
479 return exp_ct(b * log_ct(a));
480}

506// e^x - 1 without the cancellation near 0: the series below |x| = 1, exp - 1 above. A negative x goes
507// through m = e^-x - 1, whose series has no alternating terms: e^x - 1 = -m / (1 + m).
508template <typename T>
509constexpr T expm1_ct(T x)
510{
511 if (nan_(x)) return x;
512 if (abs_(x) >= T(1.0)) return exp_ct(x) - T(1.0);
513 if (x < T(0.0))
514 {
515 const T m = expm1_ct(-x);
516 return -m / (T(1.0) + m);
517 }
518 T term = x;
519 T sum = x;
520 for (int n = 2; n < 60; ++n)
521 {
522 term *= x / T(n);
523 sum += term;
524 if (abs_(term) < limits<T>::series_tolerance * abs_(sum))
525 break;
526 }
527 return sum;
528}

530// log(1 + x) without the cancellation near 0: log(u) * x / (u - 1), with u = 1 + x rounded (Goldberg's
531// correction), which cancels the rounding error of u.
532template <typename T>
533constexpr T log1p_ct(T x)
534{
535 if (nan_(x)) return x;
536 if (x < T(-1.0)) return std::numeric_limits<T>::quiet_NaN();
537 if (x == T(-1.0)) return -std::numeric_limits<T>::infinity();
538 if (inf_(x)) return x;
539 const T u = T(1.0) + x;
540 if (u == T(1.0)) return x;
541 return log_ct(u) * (x / (u - T(1.0)));
542}

544// atan via two argument-halvings atan(y) = 2 atan(y/(1+sqrt(1+y^2))) followed
545// by the Maclaurin series, with the range reduction atan(x)=pi/2-atan(1/x).
546template <typename T>
547constexpr T atan_ct(T x)
548{
549 using L = limits<T>;
550 if (nan_(x)) return x;
551 if (x < T(0.0)) return -atan_ct(-x);
552 if (inf_(x)) return L::pi / T(2.0);
553 if (x > T(1.0)) return L::pi / T(2.0) - atan_ct(T(1.0) / x);
554
555 T y = x;
556 for (int i = 0; i < 2; ++i)
557 y = y / (T(1.0) + sqrt_ct(T(1.0) + y * y));
558 const T y2 = y * y;
559 T term = y, sum = y;
560 for (int n = 1; n < 60; ++n)
561 {
562 term *= -y2;
563 const T contrib = term / (T(2.0) * T(n) + T(1.0));
564 sum += contrib;
565 if (abs_(contrib) < L::series_tolerance * abs_(sum))
566 break;
567 }
568 return T(4.0) * sum; // 2^2 from the two halvings
569}

588// erf: Maclaurin series on |x| <= 2 (alternating, mild cancellation there),
589// Laplace continued fraction for erfc beyond, evaluated backward.
590template <typename T>
591constexpr T erf_ct(T x)
592{
593 using L = limits<T>;
594 if (nan_(x)) return x;
595 if (x < T(0.0)) return -erf_ct(-x);
596 if (x > L::erf_one) return T(1.0); // erfc below half an ulp of 1
597
598 if (x <= T(2.0))
599 {
600 // 2/sqrt(pi) * sum (-1)^n x^(2n+1) / (n! (2n+1))
601 const T x2 = x * x;
602 T term = x; // (-1)^n x^(2n+1) / n!
603 T sum = x;
604 for (int n = 1; n < 80; ++n)
605 {
606 term *= -x2 / T(n);
607 const T contrib = term / (T(2.0) * T(n) + T(1.0));
608 sum += contrib;
609 if (abs_(contrib) < L::series_tolerance * abs_(sum))
610 break;
611 }
612 return L::two_over_sqrt_pi * sum;
613 }
614
615 // erfc(x) = e^{-x^2}/sqrt(pi) * 1/(x + (1/2)/(x + 1/(x + (3/2)/(x + ...))))
616 T k = x;
617 for (int n = L::erfc_terms; n >= 1; --n)
618 k = x + (T(n) / T(2.0)) / k;
619 const T erfc = exp_ct(-x * x) * L::inv_sqrt_pi / k;
620 return T(1.0) - erfc;
621}

623// erfc without cancellation in the tail: the continued fraction beyond 2, 1 - erf below, 2 - erfc(-x) for x < 0.
624template <typename T>
625constexpr T erfc_ct(T x)
626{
627 using L = limits<T>;
628 if (nan_(x)) return x;
629 if (x < T(0.0)) return T(2.0) - erfc_ct(-x);
630 if (x <= T(2.0)) return T(1.0) - erf_ct(x);
631 if (x > L::erfc_zero) return T(0.0);
632 T k = x;
633 for (int n = L::erfc_terms; n >= 1; --n)
634 k = x + (T(n) / T(2.0)) / k;
635 return exp_ct(-x * x) * L::inv_sqrt_pi / k;
636}

858// The QUADRANT-AWARE arc tangent, in C's argument order: atan2(y, x) is the
859// angle of the point (x, y), in (-pi, pi].
860//
861// Note the order, because three conventions collide here and getting it
862// backwards is a silent wrong answer rather than an error:
863//
864// C / C++ / SymPy / numpy atan2(y, x) y FIRST
865// Mathematica ArcTan[x, y] x FIRST
866//
867// This kernel follows C, because it is a C++ function called atan2 and any
868// other choice would ambush every caller. The SYMBOLIC layer in functions.hpp
869// stores the node in Mathematica's order so that `wolfram()` prints correctly,
870// and provides both spellings with the relationship pinned by a test.
877template <math_argument Y, math_argument X>
878constexpr real_t<std::common_type_t<Y, X>> atan2(Y y, X x)
879{
880 using T = real_t<std::common_type_t<Y, X>>;
881 const T pi_t = static_cast<T>(detail::limits<T>::pi);
882 const T yt = static_cast<T>(y), xt = static_cast<T>(x);
883 if (xt > T(0.0))
884 return atan(yt / xt);
885 if (xt < T(0.0))
886 return yt >= T(0.0) ? atan(yt / xt) + pi_t : atan(yt / xt) - pi_t;
887 if (yt > T(0.0))
888 return pi_t / T(2.0);
889 if (yt < T(0.0))
890 return -pi_t / T(2.0);
891 return T(0.0); // atan2(0, 0): 0 by convention, and the one value with no angle
892}

982template <typename T>
983constexpr T normal_quantile_c(T p_in)
984{
985 const T p = static_cast<T>(p_in);
986 if (detail::nan_(p) || p < T(0.0) || p > T(1.0)) return std::numeric_limits<T>::quiet_NaN();
987 if (p == T(0.0)) return -std::numeric_limits<T>::infinity();
988 if (p == T(1.0)) return std::numeric_limits<T>::infinity();
989 const auto A = [](int i) { return T(detail::acklam_a[i]); };
990 const auto B = [](int i) { return T(detail::acklam_b[i]); };
991 const auto C = [](int i) { return T(detail::acklam_c[i]); };
992 const auto D = [](int i) { return T(detail::acklam_d[i]); };
993 const T low = T(0.02425);
994 T x = T(0);
995 if (p < low) {
996 const T q = sqrt(T(-2.0) * log(p));
997 x = (((((C(0) * q + C(1)) * q + C(2)) * q + C(3)) * q + C(4)) * q + C(5)) / ((((D(0) * q + D(1)) * q + D(2)) * q + D(3)) * q + T(1.0));
998 } else if (p <= T(1.0) - low) {
999 const T q = p - T(0.5), r = q * q;
1000 x = (((((A(0) * r + A(1)) * r + A(2)) * r + A(3)) * r + A(4)) * r + A(5)) * q /
1001 (((((B(0) * r + B(1)) * r + B(2)) * r + B(3)) * r + B(4)) * r + T(1.0));
1002 } else {
1003 const T q = sqrt(T(-2.0) * log(T(1.0) - p));
1004 x = -(((((C(0) * q + C(1)) * q + C(2)) * q + C(3)) * q + C(4)) * q + C(5)) / ((((D(0) * q + D(1)) * q + D(2)) * q + D(3)) * q + T(1.0));
1005 }
1006 // One Halley step on F(x) - p, using the accurate tail of erfc
1007 const T e = normal_cdf(x) - p;
1008 const T u = e * static_cast<T>(detail::limits<T>::sqrt_2pi) * exp(T(0.5) * x * x);
1009 return x - u / (T(1.0) + T(0.5) * x * u);
1010}

1028constexpr T lgamma_c(T x_in)
1029{
1030 const T pi_t = static_cast<T>(detail::limits<T>::pi);

1032 if (detail::nan_(x)) return x;
1033 if (x < T(0.5))

1035 constexpr double c[] = {0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313, -176.61502916214059,

1061template <typename T>
1062constexpr T gamma_p_c(T a_in, T x_in)
1063{
1064 using L = detail::limits<T>;
1065 const T a = static_cast<T>(a_in), x = static_cast<T>(x_in);
1066 if (detail::nan_(a) || detail::nan_(x) || a <= T(0.0) || x < T(0.0)) return std::numeric_limits<T>::quiet_NaN();
1067 if (x == T(0.0)) return T(0.0);
1068 const T front = exp(-x + a * log(x) - lgamma(a));
1069 if (x < a + T(1.0)) {
1070 T term = T(1.0) / a, sum = term;
1071 for (int n = 1; n < 1000; ++n) {
1072 term *= x / (a + T(n));
1073 sum += term;
1074 if (abs(term) < abs(sum) * static_cast<T>(L::gamma_tolerance))
1075 break;
1076 }
1077 return sum * front;
1078 }
1079 const T tiny = static_cast<T>(L::tiny);
1080 T b = x + T(1.0) - a, c = T(1.0) / tiny, d = T(1.0) / b, h = d;
1081 for (int i = 1; i < 1000; ++i) {
1082 const T an = -T(i) * (T(i) - a);
1083 b += T(2.0);
1084 d = an * d + b;
1085 d = abs(d) < tiny ? tiny : d;
1086 c = b + an / c;
1087 c = abs(c) < tiny ? tiny : c;
1088 d = T(1.0) / d;
1089 const T delta = d * c;
1090 h *= delta;
1091 if (abs(delta - T(1.0)) < static_cast<T>(L::gamma_tolerance))
1092 break;
1093 }
1094 return T(1.0) - front * h;
1095}
