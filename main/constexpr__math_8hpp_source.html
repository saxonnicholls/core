144 // A standard floating-point type's limits, from std::numeric_limits: what long double uses, whatever it is.
145 // The splits of ln 2 and pi/2 are double's, whose parts together carry about 85 and 119 bits.
146 template <typename T>
147 struct ieee_limits
148 {
149 using compute_type = T;
150 using L = std::numeric_limits<T>;
151 // Long double literals, not std::numbers: libc++ gives those only double's precision.
152 static constexpr T ln2_full = T(0.693147180559945309417232121458176568L);
153 static constexpr T pi = T(3.14159265358979323846264338327950288L);
154 static constexpr T inv_sqrt_pi = T(0.564189583547756286948079451560772586L);
155 static constexpr T two_over_sqrt_pi = 2 * inv_sqrt_pi;
156 static constexpr T sqrt2 = T(1.41421356237309504880168872420969808L);
157 static constexpr T inv_sqrt2 = T(0.707106781186547524400844362104849039L);
158 static constexpr T sqrt_2pi = T(2.50662827463100050241576528481104525L);
159 static constexpr T half_log_2pi = T(0.918938533204672741780329736405617640L);
160 static constexpr T ln2_hi = math_limits<double>::ln2_hi;
161 static constexpr T ln2_lo = math_limits<double>::ln2_lo;
162 static constexpr T ln2_lo2 = T(1.16122272293625324218337828526629425e-26L); // ln 2 - ln2_hi - ln2_lo
163 static constexpr T inv_ln2 = T(1.44269504088896340735992468100189214L);
164 // Up to 64 bits, double's three Cody-Waite parts (about 100 bits of pi/2) suffice. Wider (binary128) takes pi/2
165 // as four non-overlapping doubles, about 212 bits: each k * part is exact in 113 bits.
166 static constexpr bool wide = L::digits > 64;
167 static constexpr T pio2_1 = wide ? T(0x1.921fb54442d18p+0) : T(math_limits<double>::pio2_1);
168 static constexpr T pio2_2 = wide ? T(0x1.1a62633145c07p-54) : T(math_limits<double>::pio2_2);
169 static constexpr T pio2_3 = wide ? T(-0x1.f1976b7ed8fbcp-110) : T(math_limits<double>::pio2_3);
170 static constexpr T pio2_4 = wide ? T(0x1.4cf98e804177dp-164) : T(0);
171 static constexpr T inv_pio2 = T(0.636619772367581343075535053490057448L);
172 static constexpr T exp_overflow = T(L::max_exponent) * ln2_full;
173 static constexpr T exp_underflow = T(L::min_exponent - L::digits - 1) * ln2_full;
174 static constexpr T whole_from = [] {
175 T p = 1;
176 for (int i = 0; i < L::digits; ++i)
177 p *= 2;
178 return p;
179 }();
180 static constexpr T integral_below = T(4.0e18);
181 static constexpr T series_tolerance = L::epsilon() * T(1e-4);
182 static constexpr T trig_tolerance = L::epsilon() * T(1e-6);
183 static constexpr T gamma_tolerance = L::epsilon() / 20;
184 static constexpr T tiny = L::min() / L::epsilon();
185 static constexpr T tanh_one = T(L::digits) * ln2_full / 2 + 1;
186 static constexpr T erf_one = L::digits <= 53 ? T(6.5) : (L::digits <= 64 ? T(6.7) : T(9.0));
187 static constexpr T erfc_zero = L::digits <= 53 ? T(27) : T(107);
188 static constexpr int erfc_terms = L::digits <= 53 ? 60 : (L::digits <= 64 ? 100 : 400);
189
190 // Halve the exponent by powers of four, then (1 + m) / 2 is within 25% of sqrt(m) for m in [1, 4).
191 static constexpr T sqrt_seed(T x) noexcept
192 {
193 T y = 1;
194 constexpr T big = T(18446744073709551616.0L), half_big = T(4294967296.0L); // 2^64, 2^32
195 while (x >= big) { x /= big; y *= half_big; }
196 while (x < 1 / big) { x *= big; y /= half_big; }
197 while (x >= 4) { x /= 4; y *= 2; }
198 while (x < 1) { x *= 4; y /= 2; }
199 return y * (1 + x) / 2;
200 }
201 static constexpr int sqrt_steps = 8;
202 };

240 template <typename T>
241 constexpr T trunc_(T x)
242 {
243 if (nan_(x) || inf_(x))
244 return x;
245 T a = abs_(x);
246 if (a >= limits<T>::whole_from) // already integral
247 return x;
248 // Wider than long long: take off the whole multiple of 2^32 first, then the rest fits
249 const T high = a >= T(4611686018427387904.0L) ? T(static_cast<long long>(a / T(4294967296.0L))) * T(4294967296.0L) : T(0);
250 T r = high + static_cast<T>(static_cast<long long>(a - high));
251 return x < 0 ? -r : r;
252 }

266// x * 2^k without <cmath>. k is at most ~2100 in magnitude for finite results.
267template <typename T>
268constexpr T ldexp2(T x, long long k)
269{
270 while (k > 0)
271 {
272 x *= T(2.0);
273 --k;
274 if (inf_(x))
275 return x;
276 }
277 while (k < 0)
278 {
279 x *= T(0.5);
280 ++k;
281 if (x == T(0.0))
282 return x;
283 }
284 return x;
285}

287template <typename T>
288constexpr T exp_ct(T x)
289{
290 using L = limits<T>;
291 if (nan_(x)) return x;
292 if (x > L::exp_overflow) return std::numeric_limits<T>::infinity();
293 if (x < L::exp_underflow) return T(0.0);
294
295 const long long k = static_cast<long long>(round_(x * L::inv_ln2));
296 const T kd = static_cast<T>(k);
297 const T r = ((x - kd * L::ln2_hi) - kd * L::ln2_lo) - kd * L::ln2_lo2; // |r| <= ~0.347
298
299 T term = T(1.0);
300 T sum = T(1.0);
301 for (int n = 1; n < 40; ++n)
302 {
303 term *= r / T(n);
304 sum += term;
305 if (abs_(term) < L::series_tolerance * abs_(sum))
306 break;
307 }
308 return ldexp2(sum, k);
309}

311template <typename T>
312constexpr T log_ct(T x)
313{
314 using L = limits<T>;
315 if (nan_(x)) return x;
316 if (x < T(0.0)) return std::numeric_limits<T>::quiet_NaN();
317 if (x == T(0.0)) return -std::numeric_limits<T>::infinity();
318 if (inf_(x)) return x;
319
320 // Normalize x = m * 2^e with m in [1, 2), then center on 1.
321 T m = x;
322 long long e = 0;
323 while (m >= T(2.0)) { m *= T(0.5); ++e; }
324 while (m < T(1.0)) { m *= T(2.0); --e; }
325 if (m > L::sqrt2)
326 {
327 m *= T(0.5);
328 ++e;
329 }
330
331 // ln m = 2 atanh((m-1)/(m+1))
332 const T t = (m - T(1.0)) / (m + T(1.0));
333 const T t2 = t * t;
334 T term = t;
335 T sum = t;
336 for (int n = 3; n < 60; n += 2)
337 {
338 term *= t2;
339 const T contrib = term / T(n);
340 sum += contrib;
341 if (abs_(contrib) < L::series_tolerance * abs_(sum))
342 break;
343 }
344 const T lnm = T(2.0) * sum;
345 const T ed = static_cast<T>(e);
346 return ed * L::ln2_hi + ((lnm + ed * L::ln2_lo) + ed * L::ln2_lo2);
347}

349template <typename T>
350constexpr T sin_series(T r)
351{
352 const T r2 = r * r;
353 T term = r;
354 T sum = r;
355 for (int n = 1; n < 20; ++n)
356 {
357 term *= -r2 / ((T(2.0) * T(n)) * (T(2.0) * T(n) + T(1.0)));
358 sum += term;
359 if (abs_(term) < limits<T>::trig_tolerance)
360 break;
361 }
362 return sum;
363}

365template <typename T>
366constexpr T cos_series(T r)
367{
368 const T r2 = r * r;
369 T term = T(1.0);
370 T sum = T(1.0);
371 for (int n = 1; n < 20; ++n)
372 {
373 term *= -r2 / ((T(2.0) * T(n) - T(1.0)) * (T(2.0) * T(n)));
374 sum += term;
375 if (abs_(term) < limits<T>::trig_tolerance)
376 break;
377 }
378 return sum;

381// Reduce x to r with x = k*(pi/2) + r, |r| <= pi/4; writes quadrant k mod 4.
382template <typename T>
383struct reduced { T r; int quadrant; };
384
385template <typename T>
386constexpr reduced<T> reduce_pio2(T x)
387{
388 using L = limits<T>;
389 const T kd = round_(x * L::inv_pio2);
390 const long long k = static_cast<long long>(kd);
391 const T r = (((x - kd * L::pio2_1) - kd * L::pio2_2) - kd * L::pio2_3) - kd * L::pio2_4;
392 int q = static_cast<int>(k % 4);
393 if (q < 0)
394 q += 4;
395 return {r, q};
396}

398template <typename T>
399constexpr T sin_ct(T x)
400{
401 if (nan_(x) || inf_(x)) return std::numeric_limits<T>::quiet_NaN();
402 const auto [r, q] = reduce_pio2(x);
403 switch (q)
404 {
405 case 0: return sin_series(r);
406 case 1: return cos_series(r);
407 case 2: return -sin_series(r);
408 default: return -cos_series(r);
409 }
410}

426template <typename T>
427constexpr T sqrt_ct(T x)
428{
429 if (nan_(x)) return x;
430 if (x < T(0.0)) return std::numeric_limits<T>::quiet_NaN();
431 if (x == T(0.0) || inf_(x)) return x;
432
433 // An estimate from the type's limits, then Newton to full precision.
434 T y = limits<T>::sqrt_seed(x);
435 for (int i = 0; i < limits<T>::sqrt_steps; ++i)
436 y = T(0.5) * (y + x / y);
437 return y;
438}

440// pow_int is TOTAL: it never divides by zero. `1.0 / pow_int(0.0, n)` is a
441// division by zero, which is UB and therefore a hard error in a constant
442// expression. The
443// result of 0^(-n) is a pole, so the honest kernel answer is +inf, which is
444// also exactly what the run-time std::pow returns. Keeping the compile-time
445// and run-time paths in agreement is this header's contract.
446template <typename T>
447constexpr T pow_int(T a, long long n)
448{
449 if (n < 0)
450 {
451 const T d = pow_int(a, -n);
452 return d == T(0.0) ? std::numeric_limits<T>::infinity() : T(1.0) / d;
453 }
454 T result = T(1.0);
455 T base = a;
456 while (n > 0)
457 {
458 if (n & 1)
459 result *= base;
460 base *= base;
461 n >>= 1;
462 }
463 return result;
464}

466template <typename T>
467constexpr T pow_ct(T a, T b)
468{
469 if (b == T(0.0)) return T(1.0);
470 if (a == T(1.0)) return T(1.0);
471 if (nan_(a) || nan_(b)) return std::numeric_limits<T>::quiet_NaN();
472 if (integral_(b))
473 return pow_int(a, static_cast<long long>(b));
474 if (a < T(0.0)) return std::numeric_limits<T>::quiet_NaN(); // non-integral power of a negative base
475 if (a == T(0.0)) return b > T(0.0) ? T(0.0) : std::numeric_limits<T>::infinity();
476 return exp_ct(b * log_ct(a));
477}

503// atan via two argument-halvings atan(y) = 2 atan(y/(1+sqrt(1+y^2))) followed
504// by the Maclaurin series, with the range reduction atan(x)=pi/2-atan(1/x).
505template <typename T>
506constexpr T atan_ct(T x)
507{
508 using L = limits<T>;
509 if (nan_(x)) return x;
510 if (x < T(0.0)) return -atan_ct(-x);
511 if (inf_(x)) return L::pi / T(2.0);
512 if (x > T(1.0)) return L::pi / T(2.0) - atan_ct(T(1.0) / x);
513
514 T y = x;
515 for (int i = 0; i < 2; ++i)
516 y = y / (T(1.0) + sqrt_ct(T(1.0) + y * y));
517 const T y2 = y * y;
518 T term = y, sum = y;
519 for (int n = 1; n < 60; ++n)
520 {
521 term *= -y2;
522 const T contrib = term / (T(2.0) * T(n) + T(1.0));
523 sum += contrib;
524 if (abs_(contrib) < L::series_tolerance * abs_(sum))
525 break;
526 }
527 return T(4.0) * sum; // 2^2 from the two halvings
528}

530template <typename T>
531constexpr T asin_ct(T x)
532{
533 using L = limits<T>;
534 if (nan_(x) || x < T(-1.0) || x > T(1.0)) return std::numeric_limits<T>::quiet_NaN();
535 if (x == T(1.0)) return L::pi / T(2.0);

547// erf: Maclaurin series on |x| <= 2 (alternating, mild cancellation there),
548// Laplace continued fraction for erfc beyond, evaluated backward.
549template <typename T>
550constexpr T erf_ct(T x)
551{
552 using L = limits<T>;
553 if (nan_(x)) return x;
554 if (x < T(0.0)) return -erf_ct(-x);
555 if (x > L::erf_one) return T(1.0); // erfc below half an ulp of 1
556
557 if (x <= T(2.0))
558 {
559 // 2/sqrt(pi) * sum (-1)^n x^(2n+1) / (n! (2n+1))
560 const T x2 = x * x;
561 T term = x; // (-1)^n x^(2n+1) / n!
562 T sum = x;
563 for (int n = 1; n < 80; ++n)
564 {
565 term *= -x2 / T(n);
566 const T contrib = term / (T(2.0) * T(n) + T(1.0));
567 sum += contrib;
568 if (abs_(contrib) < L::series_tolerance * abs_(sum))
569 break;
570 }
571 return L::two_over_sqrt_pi * sum;
572 }
573
574 // erfc(x) = e^{-x^2}/sqrt(pi) * 1/(x + (1/2)/(x + 1/(x + (3/2)/(x + ...))))
575 T k = x;
576 for (int n = L::erfc_terms; n >= 1; --n)
577 k = x + (T(n) / T(2.0)) / k;
578 const T erfc = exp_ct(-x * x) * L::inv_sqrt_pi / k;
579 return T(1.0) - erfc;
580}

582// erfc without cancellation in the tail: the continued fraction beyond 2, 1 - erf below, 2 - erfc(-x) for x < 0.
583template <typename T>
584constexpr T erfc_ct(T x)
585{
586 using L = limits<T>;
587 if (nan_(x)) return x;
588 if (x < T(0.0)) return T(2.0) - erfc_ct(-x);
589 if (x <= T(2.0)) return T(1.0) - erf_ct(x);
590 if (x > L::erfc_zero) return T(0.0);
591 T k = x;
592 for (int n = L::erfc_terms; n >= 1; --n)
593 k = x + (T(n) / T(2.0)) / k;
594 return exp_ct(-x * x) * L::inv_sqrt_pi / k;
595}

797// The QUADRANT-AWARE arc tangent, in C's argument order: atan2(y, x) is the
798// angle of the point (x, y), in (-pi, pi].
799//
800// Note the order, because three conventions collide here and getting it
801// backwards is a silent wrong answer rather than an error:
802//
803// C / C++ / SymPy / numpy atan2(y, x) y FIRST
804// Mathematica ArcTan[x, y] x FIRST
805//
806// This kernel follows C, because it is a C++ function called atan2 and any
807// other choice would ambush every caller. The SYMBOLIC layer in functions.hpp
808// stores the node in Mathematica's order so that `wolfram()` prints correctly,
809// and provides both spellings with the relationship pinned by a test.
816template <math_argument Y, math_argument X>
817constexpr real_t<std::common_type_t<Y, X>> atan2(Y y, X x)
818{
819 using T = real_t<std::common_type_t<Y, X>>;
820 const T pi_t = static_cast<T>(detail::limits<T>::pi);
821 const T yt = static_cast<T>(y), xt = static_cast<T>(x);
822 if (xt > T(0.0))
823 return atan(yt / xt);
824 if (xt < T(0.0))
825 return yt >= T(0.0) ? atan(yt / xt) + pi_t : atan(yt / xt) - pi_t;
826 if (yt > T(0.0))
827 return pi_t / T(2.0);
828 if (yt < T(0.0))
829 return -pi_t / T(2.0);
830 return T(0.0); // atan2(0, 0): 0 by convention, and the one value with no angle
831}

921template <typename T>
922constexpr T normal_quantile_c(T p_in)
923{
924 const T p = static_cast<T>(p_in);
925 if (detail::nan_(p) || p < T(0.0) || p > T(1.0)) return std::numeric_limits<T>::quiet_NaN();
926 if (p == T(0.0)) return -std::numeric_limits<T>::infinity();
927 if (p == T(1.0)) return std::numeric_limits<T>::infinity();
928 const auto A = [](int i) { return T(detail::acklam_a[i]); };
929 const auto B = [](int i) { return T(detail::acklam_b[i]); };
930 const auto C = [](int i) { return T(detail::acklam_c[i]); };
931 const auto D = [](int i) { return T(detail::acklam_d[i]); };
932 const T low = T(0.02425);
933 T x = T(0);
934 if (p < low) {
935 const T q = sqrt(T(-2.0) * log(p));
936 x = (((((C(0) * q + C(1)) * q + C(2)) * q + C(3)) * q + C(4)) * q + C(5)) / ((((D(0) * q + D(1)) * q + D(2)) * q + D(3)) * q + T(1.0));
937 } else if (p <= T(1.0) - low) {
938 const T q = p - T(0.5), r = q * q;
939 x = (((((A(0) * r + A(1)) * r + A(2)) * r + A(3)) * r + A(4)) * r + A(5)) * q /
940 (((((B(0) * r + B(1)) * r + B(2)) * r + B(3)) * r + B(4)) * r + T(1.0));
941 } else {
942 const T q = sqrt(T(-2.0) * log(T(1.0) - p));
943 x = -(((((C(0) * q + C(1)) * q + C(2)) * q + C(3)) * q + C(4)) * q + C(5)) / ((((D(0) * q + D(1)) * q + D(2)) * q + D(3)) * q + T(1.0));
944 }
945 // One Halley step on F(x) - p, using the accurate tail of erfc
946 const T e = normal_cdf(x) - p;
947 const T u = e * static_cast<T>(detail::limits<T>::sqrt_2pi) * exp(T(0.5) * x * x);
948 return x - u / (T(1.0) + T(0.5) * x * u);
949}

967constexpr T lgamma_c(T x_in)
968{
969 const T pi_t = static_cast<T>(detail::limits<T>::pi);

971 if (detail::nan_(x)) return x;
972 if (x < T(0.5))

974 constexpr double c[] = {0.99999999999980993, 676.5203681218851, -1259.1392167224028, 771.32342877765313, -176.61502916214059,

1000template <typename T>
1001constexpr T gamma_p_c(T a_in, T x_in)
1002{
1003 using L = detail::limits<T>;
1004 const T a = static_cast<T>(a_in), x = static_cast<T>(x_in);
1005 if (detail::nan_(a) || detail::nan_(x) || a <= T(0.0) || x < T(0.0)) return std::numeric_limits<T>::quiet_NaN();
1006 if (x == T(0.0)) return T(0.0);
1007 const T front = exp(-x + a * log(x) - lgamma(a));
1008 if (x < a + T(1.0)) {
1009 T term = T(1.0) / a, sum = term;
1010 for (int n = 1; n < 1000; ++n) {
1011 term *= x / (a + T(n));
1012 sum += term;
1013 if (abs(term) < abs(sum) * static_cast<T>(L::gamma_tolerance))
1014 break;
1015 }
1016 return sum * front;
1017 }
1018 const T tiny = static_cast<T>(L::tiny);
1019 T b = x + T(1.0) - a, c = T(1.0) / tiny, d = T(1.0) / b, h = d;
1020 for (int i = 1; i < 1000; ++i) {
1021 const T an = -T(i) * (T(i) - a);
1022 b += T(2.0);
1023 d = an * d + b;
1024 d = abs(d) < tiny ? tiny : d;
1025 c = b + an / c;
1026 c = abs(c) < tiny ? tiny : c;
1027 d = T(1.0) / d;
1028 const T delta = d * c;
1029 h *= delta;
1030 if (abs(delta - T(1.0)) < static_cast<T>(L::gamma_tolerance))
1031 break;
1032 }
1033 return T(1.0) - front * h;
1034}
