31 [[nodiscard]] constexpr std::optional<std::pair<uint256, uint256>> mul_div(const uint256& a, const uint256& b,
32 const uint256& d) noexcept
33 {
34 if (d.is_zero())
35 return std::nullopt;
36 // the 512-bit product, as eight 64-bit limbs
37 std::array<std::uint64_t, 8> p{};
38 for (std::size_t i = 0; i < 4; ++i) {
39 std::uint64_t carry = 0;
40 for (std::size_t j = 0; j < 4; ++j) {
41 const auto w = umul128(a.limb(i), b.limb(j));
42 std::uint64_t lo = w.low + carry;
43 std::uint64_t hi = w.high + (lo < carry ? 1u : 0u);
44 const std::uint64_t sum = p[i + j] + lo;
45 hi += sum < lo ? 1u : 0u;
46 p[i + j] = sum;
47 carry = hi;
48 }
49 p[i + 4] = carry;
50 }
51 // to 32-bit digits: u (16 + 1 for normalisation) and v (8)
52 std::array<std::uint32_t, 17> u{};
53 for (std::size_t i = 0; i < 8; ++i) {
54 u[2 * i] = static_cast<std::uint32_t>(p[i]);
55 u[2 * i + 1] = static_cast<std::uint32_t>(p[i] >> 32);
56 }
57 std::array<std::uint32_t, 8> v{};
58 for (std::size_t i = 0; i < 4; ++i) {
59 v[2 * i] = static_cast<std::uint32_t>(d.limb(i));
60 v[2 * i + 1] = static_cast<std::uint32_t>(d.limb(i) >> 32);
61 }
62 int n = 8;
63 while (v[static_cast<std::size_t>(n - 1)] == 0)
64 --n;
65 int m = 16;
66 while (m > 0 && u[static_cast<std::size_t>(m - 1)] == 0)
67 --m;
68 std::array<std::uint32_t, 16> q{};
69 std::array<std::uint32_t, 8> r{};
70 if (m < n) { // the product is smaller than the divisor
71 for (std::size_t i = 0; i < 8; ++i)
72 r[i] = u[i];
73 } else if (n == 1) { // a one-digit divisor: short division
74 std::uint64_t rem = 0;
75 for (int i = m - 1; i >= 0; --i) {
76 const std::uint64_t cur = (rem << 32) | u[static_cast<std::size_t>(i)];
77 q[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(cur / v[0]);
78 rem = cur % v[0];
79 }
80 r[0] = static_cast<std::uint32_t>(rem);
81 } else { // Knuth, TAOCP vol. 2, 4.3.1, algorithm D
82 int s = 0;
83 for (std::uint32_t top = v[static_cast<std::size_t>(n - 1)]; (top & 0x80000000u) == 0; top <<= 1)
84 ++s;
85 std::array<std::uint32_t, 8> vn{};
86 for (int i = n - 1; i > 0; --i)
87 vn[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(
88 (static_cast<std::uint64_t>(v[static_cast<std::size_t>(i)]) << s) |
89 (s == 0 ? 0 : static_cast<std::uint64_t>(v[static_cast<std::size_t>(i - 1)]) >> (32 - s)));
90 vn[0] = static_cast<std::uint32_t>(static_cast<std::uint64_t>(v[0]) << s);
91 std::array<std::uint32_t, 17> un{};
92 un[static_cast<std::size_t>(m)] =
93 s == 0 ? 0u : static_cast<std::uint32_t>(static_cast<std::uint64_t>(u[static_cast<std::size_t>(m - 1)]) >> (32 - s));
94 for (int i = m - 1; i > 0; --i)
95 un[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(
96 (static_cast<std::uint64_t>(u[static_cast<std::size_t>(i)]) << s) |
97 (s == 0 ? 0 : static_cast<std::uint64_t>(u[static_cast<std::size_t>(i - 1)]) >> (32 - s)));
98 un[0] = static_cast<std::uint32_t>(static_cast<std::uint64_t>(u[0]) << s);
99 const std::uint64_t base = 1ull << 32;
100 for (int j = m - n; j >= 0; --j) {
101 const auto jn = static_cast<std::size_t>(j + n);
102 const std::uint64_t num = (static_cast<std::uint64_t>(un[jn]) << 32) | un[jn - 1];
103 std::uint64_t qhat = num / vn[static_cast<std::size_t>(n - 1)];
104 std::uint64_t rhat = num % vn[static_cast<std::size_t>(n - 1)];
105 while (qhat >= base || qhat * vn[static_cast<std::size_t>(n - 2)] > ((rhat << 32) | un[jn - 2])) {
106 --qhat;
107 rhat += vn[static_cast<std::size_t>(n - 1)];
108 if (rhat >= base)
109 break;
110 }
111 std::int64_t borrow = 0;
112 std::uint64_t carry = 0;
113 for (int i = 0; i < n; ++i) {
114 const auto idx = static_cast<std::size_t>(i);
115 const std::uint64_t prod = qhat * vn[idx] + carry;
116 carry = prod >> 32;
117 const std::int64_t t = static_cast<std::int64_t>(un[idx + static_cast<std::size_t>(j)]) - borrow -
118 static_cast<std::int64_t>(prod & 0xffffffffu);
119 un[idx + static_cast<std::size_t>(j)] = static_cast<std::uint32_t>(t);
120 borrow = t < 0 ? 1 : 0;
121 }
122 const std::int64_t t = static_cast<std::int64_t>(un[jn]) - borrow - static_cast<std::int64_t>(carry);
123 un[jn] = static_cast<std::uint32_t>(t);
124 q[static_cast<std::size_t>(j)] = static_cast<std::uint32_t>(qhat);
125 if (t < 0) { // added back: qhat was one too large
126 q[static_cast<std::size_t>(j)] -= 1;
127 std::uint64_t c = 0;
128 for (int i = 0; i < n; ++i) {
129 const auto idx = static_cast<std::size_t>(i) + static_cast<std::size_t>(j);
130 const std::uint64_t sum = static_cast<std::uint64_t>(un[idx]) + vn[static_cast<std::size_t>(i)] + c;
131 un[idx] = static_cast<std::uint32_t>(sum);
132 c = sum >> 32;
133 }
134 un[jn] = static_cast<std::uint32_t>(un[jn] + c);
135 }
136 }
137 for (int i = 0; i < n; ++i) {
138 const auto idx = static_cast<std::size_t>(i);
139 r[idx] = static_cast<std::uint32_t>(
140 (static_cast<std::uint64_t>(un[idx]) >> s) |
141 (s == 0 ? 0 : static_cast<std::uint64_t>(un[idx + 1]) << (32 - s)));
142 }
143 }
144 for (std::size_t i = 8; i < 16; ++i)
145 if (q[i] != 0)
146 return std::nullopt; // the quotient needs more than 256 bits
147 const auto limb = [](const auto& digits, std::size_t i) {
148 return digits[2 * i] | (static_cast<std::uint64_t>(digits[2 * i + 1]) << 32);
149 };
150 return std::pair{uint256::from_limbs(limb(q, 0), limb(q, 1), limb(q, 2), limb(q, 3)),
151 uint256::from_limbs(limb(r, 0), limb(r, 1), limb(r, 2), limb(r, 3))};
152 }
