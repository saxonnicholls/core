72template <row R>
73constexpr std::size_t argmax(const R& r)
74{
75 const auto x = as_row(r);
76 assert(!x.empty());
77 std::size_t best = 0;
78 for (std::size_t i = 1; i < x.size(); ++i)
79 if (x[i] > x[best])
80 best = i;
81 return best;
82}

88template <row R>
89constexpr std::size_t sample(const R& probabilities, std::type_identity_t<row_value_t<R>> u)
90{
91 using T = row_value_t<R>;
92 const auto p = as_row(probabilities);
93 assert(!p.empty() && u >= T(0) && u < T(1));
94 T cdf = T(0);
95 std::size_t last = 0;
96 for (std::size_t i = 0; i < p.size(); ++i) {
97 cdf += p[i];
98 if (p[i] > T(0))
99 last = i;
100 if (u < cdf)
101 return i;
102 }
103 return last; // rounding left the sum below u
104}

111template <row R>
112constexpr std::size_t sample_logits(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
113 std::type_identity_t<row_value_t<R>> u)
114{
115 using T = row_value_t<R>;
116 const auto x = as_row(logits);
117 assert(!x.empty() && temperature >= T(0) && u >= T(0) && u < T(1));
118 const std::size_t best = argmax(x);
119 if (temperature == T(0))
120 return best;
121 const T max = x[best];
122 T sum = T(0);
123 for (const T v : x)
124 sum += math::exp((v - max) / temperature);
125 const T target = u * sum;
126 T cdf = T(0);
127 for (std::size_t i = 0; i < x.size(); ++i) {
128 cdf += math::exp((x[i] - max) / temperature);
129 if (target < cdf)
130 return i;
131 }
132 return best;
133}

139template <typename T, std::size_t K>
140struct top_k_result
141{
142 std::array<std::size_t, K> index{};
143 std::array<T, K> value{};
144 std::size_t count = 0;
145};
146
151template <std::size_t K, row R>
152 requires(K > 0)
153constexpr top_k_result<row_value_t<R>, K> top_k(const R& r)
154{
155 const auto x = as_row(r);
156 top_k_result<row_value_t<R>, K> out;
157 for (std::size_t i = 0; i < x.size(); ++i) {
158 if (out.count == K && !(x[i] > out.value[K - 1]))
159 continue;
160 std::size_t at = out.count < K ? out.count++ : K - 1;
161 for (; at > 0 && x[i] > out.value[at - 1]; --at) { // strictly greater: an earlier equal stays first
162 out.value[at] = out.value[at - 1];
163 out.index[at] = out.index[at - 1];
164 }
165 out.value[at] = x[i];
166 out.index[at] = i;
167 }
168 return out;
169}

177template <std::size_t K, row R>
178 requires(K > 0)
179constexpr std::size_t sample_top_k(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
180 std::type_identity_t<row_value_t<R>> u)
181{
182 const auto best = top_k<K>(logits);
183 assert(best.count > 0);
184 return best.index[sample_logits(std::span(best.value).first(best.count), temperature, u)];
185}

194template <row R>
195constexpr std::size_t sample_top_p(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
196 std::type_identity_t<row_value_t<R>> p, std::type_identity_t<row_value_t<R>> u,
197 std::span<std::size_t> scratch)
198{
199 using T = row_value_t<R>;
200 const auto x = as_row(logits);
201 assert(!x.empty() && scratch.size() == x.size() && p > T(0) && p <= T(1) && u >= T(0) && u < T(1));
202 if (temperature == T(0))
203 return argmax(x);
204 for (std::size_t i = 0; i < scratch.size(); ++i)
205 scratch[i] = i;
206 std::sort(scratch.begin(), scratch.end(), [&](std::size_t a, std::size_t b) { return x[a] > x[b] || (x[a] == x[b] && a < b); });
207 const T max = x[scratch[0]];
208 T sum = T(0);
209 for (const T v : x)
210 sum += math::exp((v - max) / temperature);
211 // Keep the most probable until their mass reaches p, then sample among them
212 std::size_t kept = 0;
213 T mass = T(0);
214 while (kept < scratch.size() && (kept == 0 || mass < p * sum))
215 mass += math::exp((x[scratch[kept++]] - max) / temperature);
216 const T target = u * mass;
217 T cdf = T(0);
218 for (std::size_t i = 0; i < kept; ++i) {
219 cdf += math::exp((x[scratch[i]] - max) / temperature);
220 if (target < cdf)
221 return scratch[i];
222 }
223 return scratch[kept - 1];
224}
