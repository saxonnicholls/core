80template <row R>
81constexpr std::size_t argmax(const R& r)
82{
83 const auto x = as_row(r);
84 assert(!x.empty());
85 std::size_t best = 0;
86 for (std::size_t i = 1; i < x.size(); ++i)
87 if (x[i] > x[best])
88 best = i;
89 return best;
90}

96template <row R>
97constexpr std::size_t sample(const R& probabilities, std::type_identity_t<row_value_t<R>> u)
98{
99 using T = row_value_t<R>;
100 const auto p = as_row(probabilities);
101 assert(!p.empty() && u >= T(0) && u < T(1));
102 T cdf = T(0);
103 std::size_t last = 0;
104 for (std::size_t i = 0; i < p.size(); ++i) {
105 cdf += p[i];
106 if (p[i] > T(0))
107 last = i;
108 if (u < cdf)
109 return i;
110 }
111 return last; // rounding left the sum below u
112}

119template <row R>
120constexpr std::size_t sample_logits(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
121 std::type_identity_t<row_value_t<R>> u)
122{
123 using T = row_value_t<R>;
124 const auto x = as_row(logits);
125 assert(!x.empty() && temperature >= T(0) && u >= T(0) && u < T(1));
126 const std::size_t best = argmax(x);
127 if (temperature == T(0))
128 return best;
129 const T max = x[best];
130 T sum = T(0);
131 for (const T v : x)
132 sum += math::exp((v - max) / temperature);
133 const T target = u * sum;
134 T cdf = T(0);
135 for (std::size_t i = 0; i < x.size(); ++i) {
136 cdf += math::exp((x[i] - max) / temperature);
137 if (target < cdf)
138 return i;
139 }
140 return best;
141}

147template <typename T, std::size_t K>
148struct top_k_result
149{
150 std::array<std::size_t, K> index{};
151 std::array<T, K> value{};
152 std::size_t count = 0;
153};
154
159template <std::size_t K, row R>
160 requires(K > 0)
161constexpr top_k_result<row_value_t<R>, K> top_k(const R& r)
162{
163 const auto x = as_row(r);
164 top_k_result<row_value_t<R>, K> out;
165 for (std::size_t i = 0; i < x.size(); ++i) {
166 if (out.count == K && !(x[i] > out.value[K - 1]))
167 continue;
168 std::size_t at = out.count < K ? out.count++ : K - 1;
169 for (; at > 0 && x[i] > out.value[at - 1]; --at) { // strictly greater: an earlier equal stays first
170 out.value[at] = out.value[at - 1];
171 out.index[at] = out.index[at - 1];
172 }
173 out.value[at] = x[i];
174 out.index[at] = i;
175 }
176 return out;
177}

185template <std::size_t K, row R>
186 requires(K > 0)
187constexpr std::size_t sample_top_k(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
188 std::type_identity_t<row_value_t<R>> u)
189{
190 const auto best = top_k<K>(logits);
191 assert(best.count > 0);
192 return best.index[sample_logits(std::span(best.value).first(best.count), temperature, u)];
193}

202template <row R>
203constexpr std::size_t sample_top_p(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
204 std::type_identity_t<row_value_t<R>> p, std::type_identity_t<row_value_t<R>> u,
205 std::span<std::size_t> scratch)
206{
207 using T = row_value_t<R>;
208 const auto x = as_row(logits);
209 assert(!x.empty() && scratch.size() == x.size() && p > T(0) && p <= T(1) && u >= T(0) && u < T(1));
210 if (temperature == T(0))
211 return argmax(x);
212 for (std::size_t i = 0; i < scratch.size(); ++i)
213 scratch[i] = i;
214 std::sort(scratch.begin(), scratch.end(), [&](std::size_t a, std::size_t b) { return x[a] > x[b] || (x[a] == x[b] && a < b); });
215 const T max = x[scratch[0]];
216 T sum = T(0);
217 for (const T v : x)
218 sum += math::exp((v - max) / temperature);
219 // Keep the most probable until their mass reaches p, then sample among them
220 std::size_t kept = 0;
221 T mass = T(0);
222 while (kept < scratch.size() && (kept == 0 || mass < p * sum))
223 mass += math::exp((x[scratch[kept++]] - max) / temperature);
224 const T target = u * mass;
225 T cdf = T(0);
226 for (std::size_t i = 0; i < kept; ++i) {
227 cdf += math::exp((x[scratch[i]] - max) / temperature);
228 if (target < cdf)
229 return scratch[i];
230 }
231 return scratch[kept - 1];
232}

239template <row R>
240constexpr row_value_t<R> min_p_threshold(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
241 std::type_identity_t<row_value_t<R>> min_p)
242{
243 using T [[maybe_unused]] = row_value_t<R>; // read only by the assert
244 const auto x = as_row(logits);
245 assert(!x.empty() && temperature > T(0) && min_p > T(0) && min_p <= T(1));
246 return x[argmax(x)] + temperature * math::log(min_p);
247}

255template <row R>
256constexpr std::size_t sample_min_p(const R& logits, std::type_identity_t<row_value_t<R>> temperature,
257 std::type_identity_t<row_value_t<R>> min_p, std::type_identity_t<row_value_t<R>> u)
258{
259 using T = row_value_t<R>;
260 const auto x = as_row(logits);
261 assert(!x.empty() && u >= T(0) && u < T(1));
262 const std::size_t best = argmax(x);
263 if (temperature == T(0))
264 return best;
265 const T cut = min_p_threshold(x, temperature, min_p), max = x[best];
266 T mass = T(0);
267 for (const T v : x)
268 if (v >= cut)
269 mass += math::exp((v - max) / temperature);
270 const T target = u * mass;
271 T cdf = T(0);
272 std::size_t last = best;
273 for (std::size_t i = 0; i < x.size(); ++i)
274 if (x[i] >= cut) {
275 cdf += math::exp((x[i] - max) / temperature);
276 last = i;
277 if (target < cdf)
278 return i;
279 }
280 return last;
281}

286template <math::math_number T>
287struct logit_penalties
288{
289 T repetition = T(1);
290 T presence = T(0);
291 T frequency = T(0);
292};

298template <mutable_row R, typename Token>
299 requires std::integral<Token>
300constexpr void apply_penalties(R&& logits, std::span<const Token> history, const logit_penalties<row_value_t<R>>& penalties)
301{
302 using T = row_value_t<R>;
303 const auto x = as_mutable_row(logits);
304 assert(penalties.repetition > T(0));
305 for (std::size_t i = 0; i < history.size(); ++i) {
306 const auto t = static_cast<std::size_t>(history[i]);
307 assert(t < x.size());
308 bool first = true;
309 std::size_t count = 0;
310 for (std::size_t j = 0; j < history.size(); ++j)
311 if (history[j] == history[i]) {
312 first = first && j >= i;
313 ++count;
314 }
315 if (!first)
316 continue; // each distinct token once, at its first occurrence
317 T& v = x[t];
318 v = v > T(0) ? v / penalties.repetition : v * penalties.repetition;
319 v -= penalties.presence + penalties.frequency * T(count);
320 }
321}
