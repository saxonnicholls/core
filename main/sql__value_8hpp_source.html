121 [[nodiscard]] constexpr bool may_retry(sql_errc code, sql_retry policy) noexcept
122 {
123 if (policy == sql_retry::never)
124 return false;
125 if (code == sql_errc::connection_failed || code == sql_errc::busy || code == sql_errc::serialization_failure)
126 return true;
127 return policy == sql_retry::idempotent && (code == sql_errc::connection_lost || code == sql_errc::timeout);

220 [[nodiscard]] inline sql_param view_of(const sql_value& v) noexcept
221 {
222 switch (kind_of(v)) {
223 case sql_kind::boolean: return std::get<1>(v);

331 [[nodiscard]] constexpr sql_result<bool> get_bool(std::size_t i) const noexcept
332 {
333 const auto c = cell(i);
334 if (!c)
335 return err(c.unwrap_err());
336 if (kind_of(*c) == sql_kind::boolean)
337 return ok(std::get<1>(*c));
338 if (kind_of(*c) == sql_kind::int64 && (std::get<2>(*c) == 0 || std::get<2>(*c) == 1))
339 return ok(std::get<2>(*c) == 1);
340 return sql_fail(sql_errc::type_mismatch, "not a boolean");
341 }
342
345 [[nodiscard]] constexpr sql_result<std::int64_t> get_int64(std::size_t i) const noexcept
346 {
347 const auto c = cell(i);
348 if (!c)
349 return err(c.unwrap_err());
350 if (kind_of(*c) == sql_kind::int64)
351 return ok(std::get<2>(*c));
352 if (kind_of(*c) == sql_kind::boolean)
353 return ok(std::int64_t{std::get<1>(*c) ? 1 : 0});
354 return sql_fail(sql_errc::type_mismatch, "not an integer");
355 }
356
359 [[nodiscard]] constexpr sql_result<double> get_double(std::size_t i) const noexcept
360 {
361 const auto c = cell(i);
362 if (!c)
363 return err(c.unwrap_err());
364 if (kind_of(*c) == sql_kind::float64)
365 return ok(std::get<3>(*c));
366 if (kind_of(*c) == sql_kind::int64) {
367 const std::int64_t v = std::get<2>(*c);
368 if (v > (std::int64_t{1} << 53) || v < -(std::int64_t{1} << 53))
369 return sql_fail(sql_errc::out_of_range, "an integer a double cannot hold exactly");
370 return ok(static_cast<double>(v));
371 }
372 return sql_fail(sql_errc::type_mismatch, "not a number");
373 }
374
377 [[nodiscard]] constexpr sql_result<std::string_view> get_text(std::size_t i) const noexcept
378 {
379 const auto c = cell(i);
380 if (!c)
381 return err(c.unwrap_err());
382 if (kind_of(*c) == sql_kind::text)
383 return ok(std::get<4>(*c));

389 [[nodiscard]] constexpr sql_result<std::span<const std::byte>> get_blob(std::size_t i) const noexcept

440 [[nodiscard]] constexpr sql_result<uuid> uuid_from_sql(const sql_param& p) noexcept
441 {
442 if (kind_of(p) != sql_kind::blob)
443 return sql_fail(sql_errc::type_mismatch, "a uuid is a blob");
444 const auto b = std::get<5>(p);
445 if (b.size() != 16)
446 return sql_fail(sql_errc::out_of_range, "a uuid is 16 bytes");
447 uuid u;
448 for (std::size_t i = 0; i < 16; ++i)
449 u.bytes[i] = static_cast<std::uint8_t>(b[i]);
450 return ok(u);
451 }

508 [[nodiscard]] constexpr sql_result<money> money_from_sql(const asset& a, const sql_param& p) noexcept
509 {
510 if (kind_of(p) != sql_kind::text)
511 return sql_fail(sql_errc::type_mismatch, "an amount is text");
512 const auto m = parse_money(a, std::get<4>(p));
513 if (m)
514 return ok(*m);
