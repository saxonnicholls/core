202 // A dimension: the keys seen in one position of the pivot, and each one's index.
203 template <typename K>
204 class dimension {
205 static_assert(std_hashable<K> && std::equality_comparable<K>,
206 "snicholls::stats::pivot: a key must be an enumeration, a bool, have a pivot_domain, or be hashable");
207
208 public:
209 static constexpr bool fixed = false;
210 std::optional<std::size_t> index(const K& k) const
211 {
212 const auto found = index_.find(k);
213 return found == index_.end() ? std::nullopt : std::optional<std::size_t>(found->second);
214 }
215 std::optional<std::size_t> index_or_add(const K& k)
216 {
217 const auto [it, added] = index_.try_emplace(k, keys_.size());
218 if (added)
219 keys_.push_back(k);
220 return it->second;
221 }
222 std::size_t size() const noexcept { return keys_.size(); }
223 const K& key(std::size_t i) const noexcept { return keys_[i]; }
224
225 private:
226 std::unordered_map<K, std::size_t> index_;
227 std::vector<K> keys_;
228 };

341 void add(const Row& row)
342 {
343 ++rows_;
344 std::array<std::size_t, dims> at{};
345 if (!place(row, at, std::make_index_sequence<dims>{})) {
346 ++dropped_;
347 return;
348 }

350 if (touched_[i] == 0) { // a new cell: give each bootstrap its own stream
351 touched_[i] = 1;

353 }
354 add_measures(cells_[i], row, std::index_sequence_for<As...>{});

413 template <typename F>
414 void for_each(F&& f) const
415 {
416 const auto caps = capacity_;
417 for (std::size_t i = 0; i < cells_.size(); ++i) {
418 if (touched_[i] == 0)
419 continue;
420 std::array<std::size_t, dims> at{};
421 std::size_t rest = i;

423 at[d] = rest % caps[d];

497 void print(std::ostream& os) const
498 {
499 std::vector<std::tuple<detail::key_t<Rs, Row>...>> row_keys;
500 std::vector<std::tuple<detail::key_t<Cs, Row>...>> column_keys;
501 for_each([&](const keys_type& keys, const cell_type&) {
502 const auto r = row_part(keys, std::make_index_sequence<row_dims>{});
503 const auto c = column_part(keys, std::make_index_sequence<sizeof...(Cs)>{});
504 if (std::find(row_keys.begin(), row_keys.end(), r) == row_keys.end())
505 row_keys.push_back(r);
506 if (std::find(column_keys.begin(), column_keys.end(), c) == column_keys.end())
507 column_keys.push_back(c);
508 });
509 const auto print_keys = [&os](const auto& t) {
510 std::apply(
511 [&os](const auto&... k) {
512 const char* separator = "";
513 ((os << separator, print_to(os, k), separator = "/"), ...);
514 },
515 t);
516 };
517 for (const auto& c : column_keys) {
518 os << '\t';
519 print_keys(c);
520 }
521 os << '\n';
522 for (const auto& r : row_keys) {
523 print_keys(r);
524 for (const auto& c : column_keys) {
525 os << '\t';
526 if (const auto v = value<0>(std::tuple_cat(r, c)))
527 detail::print_cell<first_getter>(os, *v);

529 os << '\n';

542 template <std::size_t D, typename K>
543 bool place_one(const K& key, std::array<std::size_t, dims>& at)
544 {
545 const std::optional<std::size_t> i = std::get<D>(dims_).index_or_add(key);
546 if (!i)
547 return false;
548 if constexpr (!fixed_dimension<D>) // a fixed dimension never grows
549 if (*i >= capacity_[D])
550 grow(D, *i + 1);
551 at[D] = *i;
552 return true;

620 // Row-major position; with every dimension fixed the strides are constants.
621 std::size_t flat(const std::array<std::size_t, dims>& at) const noexcept
622 {
623 const std::array<std::size_t, dims>& caps = all_fixed ? fixed_capacity : capacity_;
624 std::size_t i = 0;
625 for (std::size_t d = 0; d < dims; ++d)
626 i = i * caps[d] + at[d];
627 return i;
628 }
629
630 // Widen dimension d to hold `needed` keys, doubling, and move every cell to its new place.
631 void grow(std::size_t d, std::size_t needed)
632 {
633 std::array<std::size_t, dims> caps = capacity_;
634 std::size_t wanted = caps[d] == 0 ? 4 : caps[d];
635 while (wanted < needed)
636 wanted *= 2;
637 caps[d] = wanted;
638 std::size_t total = 1;
639 for (const std::size_t c : caps)
640 total *= c;
641 std::vector<cell_type> cells(total, prototype_);
642 std::vector<std::uint8_t> touched(total, 0);
643 for (std::size_t i = 0; i < cells_.size(); ++i) {
644 if (touched_[i] == 0)
645 continue;
646 std::size_t rest = i, j = 0, scale = 1;
647 std::array<std::size_t, dims> at{};
648 for (std::size_t k = dims; k-- > 0;) {
649 at[k] = rest % capacity_[k];
650 rest /= capacity_[k];
651 }
652 for (std::size_t k = dims; k-- > 0;) {
653 j += at[k] * scale;
654 scale *= caps[k];
655 }
656 cells[j] = std::move(cells_[i]);
657 touched[j] = 1;
658 }
659 capacity_ = caps;
660 cells_ = std::move(cells);
661 touched_ = std::move(touched);
662 }
