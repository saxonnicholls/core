85 constexpr auto operator()(coalition s) const
86 {
87 Point x = *from_;
88 for (coalition m = s; m != 0; m &= m - 1) {
89 const auto i = static_cast<std::size_t>(std::countr_zero(m));
90 x[i] = (*to_)[i];
91 }
92 return std::invoke(*f_, std::as_const(x));
93 }
