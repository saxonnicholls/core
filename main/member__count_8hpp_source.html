40// Converts to any member type T (by T&&, so move-only members count correctly). Only ever used in
41// unevaluated requires-expressions, so it is declared but deliberately never defined (and not
42// constexpr/inline, which would demand a definition).
43struct any_member
44{
45 std::size_t index;
46
47 template <typename T>
48 operator T&&() const noexcept;
49};
50
51template <typename S, std::size_t... I>
52constexpr bool is_initialisable_with(std::index_sequence<I...>) noexcept
53{
54 return requires { S{any_member{I}...}; };
55}

57template <typename S, std::size_t Lo, std::size_t Hi>
58constexpr std::size_t search_member_count() noexcept
59{
60 if constexpr (Lo == Hi) {
61 return Lo;
62 } else {
63 constexpr std::size_t mid = (Lo + Hi + 1) / 2;
64 if constexpr (is_initialisable_with<S>(std::make_index_sequence<mid>{}))
65 return search_member_count<S, mid, Hi>();
66 else
67 return search_member_count<S, Lo, mid - 1>();
68 }
69}

76template <typename S>

83template <plain_aggregate S>
84struct member_count
85 : std::integral_constant<std::size_t,
86 detail::search_member_count<std::remove_cvref_t<S>, 0, SNICHOLLS_MAX_STRUCT_MEMBERS + 1>()>
87{
88 static_assert(member_count::value <= SNICHOLLS_MAX_STRUCT_MEMBERS,
89 "snicholls::member_count: struct has more than SNICHOLLS_MAX_STRUCT_MEMBERS members");
90 static_assert(member_count::value > 0 || std::is_empty_v<std::remove_cvref_t<S>>,
91 "snicholls::member_count: cannot count members - reference members are not supported");
92};
