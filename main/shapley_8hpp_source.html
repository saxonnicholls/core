93 template <typename T, typename Values>
94 struct attribution {
95 // Copy-initialised, not braced: GCC 13 crashes (verify_ctor_sanity) constant-evaluating a braced
96 // member of a user's value type with default member initialisers.
97 T baseline = T();
98 T total = T();
99 Values values = Values();
100
103 [[nodiscard]] constexpr T attributed() const
104 {
105 T sum = baseline;
106 for (const T& a : values)
107 sum = sum + a;
108 return sum;
109 }
110
113 [[nodiscard]] constexpr T unattributed() const { return total - attributed(); }
114
117 void print(std::ostream& os) const
118 requires streamable<T>
119 {
120 os << "baseline " << baseline << ", [";
121 const char* separator = "";
122 for (const T& a : values) {
123 os << separator << a;
124 separator = ", ";
125 }
126 os << "], total " << total;
127 }
128 };

167 template <typename T>
168 constexpr game_scalar_t<T> shapley_weight(std::size_t n, std::size_t size) noexcept
169 {
170 using scalar = game_scalar_t<T>;
171 return scalar(1) / (static_cast<scalar>(n) * static_cast<scalar>(binomial(n - 1, size)));

189 // Formula (3): each edge S -> S + i of the hypercube, weighted |S|! (n - |S| - 1)! / n!.
190 template <typename T>
191 constexpr void shapley_from_table(std::size_t n, std::span<const T> table, std::span<T> out)
192 {
193 std::array<game_scalar_t<T>, 64> weight{};
194 for (std::size_t k = 0; k < n; ++k)
195 weight[k] = shapley_weight<T>(n, k);
196 const coalition all = grand_coalition(n);
197 for (coalition s = 0; s < all; ++s) {
198 const auto w = weight[coalition_size(s)];
199 const T& here = table[s];
200 for (coalition missing = all & ~s; missing != 0; missing &= missing - 1) {
201 const auto i = static_cast<std::size_t>(std::countr_zero(missing));
202 out[i] = out[i] + (table[s | singleton(i)] - here) * w;
203 }
204 }
205 }

207 template <typename T>
208 constexpr void banzhaf_from_table(std::size_t n, std::span<const T> table, std::span<T> out)
209 {
210 const game_scalar_t<T> w = n == 0 ? game_scalar_t<T>(1) : game_scalar_t<T>(1) / static_cast<game_scalar_t<T>>(table_size(n - 1));
211 const coalition all = grand_coalition(n);
212 for (coalition s = 0; s < all; ++s)
213 for (coalition missing = all & ~s; missing != 0; missing &= missing - 1) {
214 const auto i = static_cast<std::size_t>(std::countr_zero(missing));
215 out[i] = out[i] + (table[s | singleton(i)] - table[s]) * w;
216 }
217 }

219 template <game V, typename Values>
220 constexpr void sequential_walk(const V& v, std::span<const std::size_t> order, attribution<game_value_t<V>, Values>& out)
221 {
222 coalition seen = 0;
223 for (const std::size_t i : order) {
224 assert(i < order.size() && !contains(seen, i) && "snicholls::game::permuted_sequential: the order is not a permutation");
225 seen = with(seen, i);
226 }
227 out.baseline = std::invoke(v, coalition{0});
228 game_value_t<V> previous = out.baseline;
229 coalition s = 0;
230 for (const std::size_t i : order) {
231 s = with(s, i);
232 game_value_t<V> current = std::invoke(v, s);
233 out.values[i] = current - previous;
234 previous = std::move(current);
235 }
236 out.total = previous;
237 }

257 // Welford's running mean and variance, one per player.
258 template <std::floating_point T>
259 struct moments {
260 std::vector<T> mean, m2;
261 std::size_t count = 0;
262
263 explicit moments(std::size_t n) : mean(n), m2(n) {}
264
265 void add(std::span<const T> sample)
266 {
267 ++count;
268 for (std::size_t i = 0; i < mean.size(); ++i) {
269 const T delta = sample[i] - mean[i];
270 mean[i] += delta / static_cast<T>(count);
271 m2[i] += delta * (sample[i] - mean[i]);
272 }
273 }
274
275 T standard_error(std::size_t i) const noexcept
276 {
277 if (count < 2)
278 return std::numeric_limits<T>::quiet_NaN();
279 return std::sqrt(m2[i] / static_cast<T>(count - 1) / static_cast<T>(count));
280 }
281 };

283 template <game V, stats::full_range_generator G, typename Values>
284 void sample_orders(std::size_t n, const V& v, G& g, permutation_sampling options, sampled_attribution<game_value_t<V>, Values>& out)
285 {
286 using T = game_value_t<V>;
287 out.baseline = std::invoke(v, coalition{0});
288 out.total = std::invoke(v, grand_coalition(n));
289 out.evaluations = 2;
290 moments<T> stats(n);
291 std::vector<std::size_t> order(n);
292 for (std::size_t i = 0; i < n; ++i)
293 order[i] = i;
294 std::vector<T> forward(n), sample(n);
295 const auto walk = [&](auto first, auto last, std::vector<T>& into) {
296 T previous = out.baseline;
297 coalition s = 0;
298 for (auto it = first; it != last; ++it) {
299 s = with(s, *it);
300 const bool complete = s == grand_coalition(n);
301 T current = complete ? out.total : std::invoke(v, s);
302 out.evaluations += complete ? 0 : 1;
303 into[*it] = current - previous;
304 previous = current;
305 }
306 };
307 for (std::size_t p = 0; p < options.permutations && n > 0; ++p) {
308 stats::shuffle(std::span<std::size_t>(order), g);
309 walk(order.begin(), order.end(), forward);
310 if (options.antithetic) {
311 walk(order.rbegin(), order.rend(), sample);
312 for (std::size_t i = 0; i < n; ++i)
313 sample[i] = (sample[i] + forward[i]) / T(2);
314 } else {
315 sample = forward;
316 }
317 stats.add(std::span<const T>(sample));
318 }
319 for (std::size_t i = 0; i < n; ++i) {
320 out.values[i] = stats.mean[i];
321 out.standard_errors[i] = stats.standard_error(i);
322 }
323 out.samples = stats.count;
324 }

326 template <game V, stats::full_range_generator G, typename Values>
327 void sample_lifts(std::size_t n, const V& v, G& g, lift_sampling options, sampled_attribution<game_value_t<V>, Values>& out)
328 {
329 using T = game_value_t<V>;
330 out.baseline = std::invoke(v, coalition{0});
331 out.total = std::invoke(v, grand_coalition(n));
332 out.evaluations = 2;
333 std::vector<std::size_t> others;
334 T sum = T(0);
335 for (std::size_t i = 0; i < n; ++i) {
336 others.clear();
337 for (std::size_t j = 0; j < n; ++j)
338 if (j != i)
339 others.push_back(j);
340 T mean = T(0), m2 = T(0);
341 std::size_t count = 0;
342 for (std::size_t k = 0; k < options.samples_per_player; ++k) {
343 // |S| is uniform on 0 .. n-1 under the Shapley weights, then S uniform of that size
344 const auto size = static_cast<std::size_t>(stats::uniform_index(g, n));
345 coalition s = 0;
346 for (std::size_t j = 0; j < size; ++j) {
347 const auto pick = j + static_cast<std::size_t>(stats::uniform_index(g, others.size() - j));
348 std::swap(others[j], others[pick]);
349 s = with(s, others[j]);
350 }
351 const T lift = std::invoke(v, with(s, i)) - std::invoke(v, s);
352 out.evaluations += 2;
353 ++count;
354 const T delta = lift - mean;
355 mean += delta / static_cast<T>(count);
356 m2 += delta * (lift - mean);
357 }
358 out.values[i] = mean;
359 out.standard_errors[i] = count < 2 ? std::numeric_limits<T>::quiet_NaN()
360 : std::sqrt(m2 / static_cast<T>(count - 1) / static_cast<T>(count));
361 sum += mean;
362 }
363 out.samples = options.samples_per_player;
364 const T target = out.total - out.baseline;
365 if (options.scale_to_total && sum != T(0)) {
366 const T factor = target / sum;
367 for (std::size_t i = 0; i < n; ++i) {
368 out.values[i] *= factor;
369 out.standard_errors[i] *= std::abs(factor);
370 }
371 }
372 }

428 template <std::size_t N, game V>
429 [[nodiscard]] constexpr fixed_attribution<game_value_t<V>, N> shapley_by_permutations(const V& v)
430 {
431 static_assert(N <= 10, "snicholls::game::shapley_by_permutations: more than 10! orders; use shapley");
432 using T = game_value_t<V>;
433 fixed_attribution<T, N> out{};
434 detail::with_table(N, v, [&out](std::span<const T> table) {
435 out.baseline = table.front();
436 out.total = table.back();
437 for_each_permutation<N>([&](const std::array<std::size_t, N>& order) {
438 coalition s = 0;
439 for (const std::size_t i : order) {
440 const coalition next = with(s, i);
441 out.values[i] = out.values[i] + (table[next] - table[s]);
442 s = next;
443 }
444 });
445 const auto scale = game_scalar_t<T>(1) / static_cast<game_scalar_t<T>>(permutation_count_v<N>);
446 for (T& a : out.values)
447 a = a * scale;
448 });
449 return out;
450 }

481 template <std::size_t N, game V>
482 [[nodiscard]] constexpr std::array<std::array<game_value_t<V>, N>, N> shapley_interactions(const V& v)
483 {
484 static_assert(N <= max_table_players, "snicholls::game::shapley_interactions: too many players");
485 using T = game_value_t<V>;
486 std::array<std::array<T, N>, N> out{};
487 // With fewer than two players no coalition misses two, so the loop body never runs and nothing is divided.
488 detail::with_table(N, v, [&out](std::span<const T> table) {
489 using scalar = game_scalar_t<T>;
490 const coalition all = grand_coalition(N);
491 for (coalition s = 0; s <= all; ++s) {
492 const coalition missing = all & ~s;
493 if (coalition_size(missing) < 2)
494 continue;
495 const scalar w = scalar(1) / (static_cast<scalar>(N - 1) * static_cast<scalar>(binomial(N - 2, coalition_size(s))));
496 for (coalition a = missing; a != 0; a &= a - 1) {
497 const auto i = static_cast<std::size_t>(std::countr_zero(a));
498 for (coalition b = a & (a - 1); b != 0; b &= b - 1) {
499 const auto j = static_cast<std::size_t>(std::countr_zero(b));
500 const T delta = table[s | singleton(i) | singleton(j)] - table[s | singleton(i)] - table[s | singleton(j)] + table[s];
501 out[i][j] = out[i][j] + delta * w;
502 out[j][i] = out[i][j];
503 }
504 }
505 }
506 });
507 return out;
508 }
