91 template <typename T, typename Values>
92 struct attribution {
93 // Copy-initialised, not braced: GCC 13 crashes (verify_ctor_sanity) constant-evaluating a braced
94 // member of a user's value type with default member initialisers.
95 T baseline = T();
96 T total = T();
97 Values values = Values();
98
101 [[nodiscard]] constexpr T attributed() const
102 {
103 T sum = baseline;
104 for (const T& a : values)
105 sum = sum + a;
106 return sum;
107 }
108
111 [[nodiscard]] constexpr T unattributed() const { return total - attributed(); }
112
115 void print(std::ostream& os) const
116 requires streamable<T>
117 {
118 os << "baseline " << baseline << ", [";
119 const char* separator = "";
120 for (const T& a : values) {
121 os << separator << a;
122 separator = ", ";
123 }
124 os << "], total " << total;
125 }
126 };

165 template <typename T>
166 constexpr game_scalar_t<T> shapley_weight(std::size_t n, std::size_t size) noexcept
167 {
168 using scalar = game_scalar_t<T>;
169 return scalar(1) / (static_cast<scalar>(n) * static_cast<scalar>(binomial(n - 1, size)));

187 // Formula (3): each edge S -> S + i of the hypercube, weighted |S|! (n - |S| - 1)! / n!.
188 template <typename T>
189 constexpr void shapley_from_table(std::size_t n, std::span<const T> table, std::span<T> out)
190 {
191 std::array<game_scalar_t<T>, 64> weight{};
192 for (std::size_t k = 0; k < n; ++k)
193 weight[k] = shapley_weight<T>(n, k);
194 const coalition all = grand_coalition(n);
195 for (coalition s = 0; s < all; ++s) {
196 const auto w = weight[coalition_size(s)];
197 const T& here = table[s];
198 for (coalition missing = all & ~s; missing != 0; missing &= missing - 1) {
199 const auto i = static_cast<std::size_t>(std::countr_zero(missing));
200 out[i] = out[i] + (table[s | singleton(i)] - here) * w;
201 }
202 }
203 }

205 template <typename T>
206 constexpr void banzhaf_from_table(std::size_t n, std::span<const T> table, std::span<T> out)
207 {
208 const game_scalar_t<T> w = n == 0 ? game_scalar_t<T>(1) : game_scalar_t<T>(1) / static_cast<game_scalar_t<T>>(table_size(n - 1));
209 const coalition all = grand_coalition(n);
210 for (coalition s = 0; s < all; ++s)
211 for (coalition missing = all & ~s; missing != 0; missing &= missing - 1) {
212 const auto i = static_cast<std::size_t>(std::countr_zero(missing));
213 out[i] = out[i] + (table[s | singleton(i)] - table[s]) * w;
214 }

217 template <game V, typename Values>
218 constexpr void sequential_walk(const V& v, std::span<const std::size_t> order, attribution<game_value_t<V>, Values>& out)
219 {
220 coalition seen = 0;
221 for (const std::size_t i : order) {
222 assert(i < order.size() && !contains(seen, i) && "snicholls::game::permuted_sequential: the order is not a permutation");
223 seen = with(seen, i);
224 }
225 out.baseline = std::invoke(v, coalition{0});
226 game_value_t<V> previous = out.baseline;
227 coalition s = 0;
228 for (const std::size_t i : order) {
229 s = with(s, i);
230 game_value_t<V> current = std::invoke(v, s);
231 out.values[i] = current - previous;
232 previous = std::move(current);
233 }
234 out.total = previous;
235 }

255 // Welford's running mean and variance, one per player.
256 template <std::floating_point T>
257 struct moments {
258 std::vector<T> mean, m2;
259 std::size_t count = 0;
260
261 explicit moments(std::size_t n) : mean(n), m2(n) {}
262
263 void add(std::span<const T> sample)
264 {
265 ++count;
266 for (std::size_t i = 0; i < mean.size(); ++i) {
267 const T delta = sample[i] - mean[i];
268 mean[i] += delta / static_cast<T>(count);
269 m2[i] += delta * (sample[i] - mean[i]);
270 }
271 }
272
273 T standard_error(std::size_t i) const noexcept
274 {
275 if (count < 2)
276 return std::numeric_limits<T>::quiet_NaN();
277 return std::sqrt(m2[i] / static_cast<T>(count - 1) / static_cast<T>(count));
278 }
279 };

281 template <game V, stats::full_range_generator G, typename Values>
282 void sample_orders(std::size_t n, const V& v, G& g, permutation_sampling options, sampled_attribution<game_value_t<V>, Values>& out)
283 {
284 using T = game_value_t<V>;
285 out.baseline = std::invoke(v, coalition{0});
286 out.total = std::invoke(v, grand_coalition(n));
287 out.evaluations = 2;
288 moments<T> stats(n);
289 std::vector<std::size_t> order(n);
290 for (std::size_t i = 0; i < n; ++i)
291 order[i] = i;
292 std::vector<T> forward(n), sample(n);
293 const auto walk = [&](auto first, auto last, std::vector<T>& into) {
294 T previous = out.baseline;
295 coalition s = 0;
296 for (auto it = first; it != last; ++it) {
297 s = with(s, *it);
298 const bool complete = s == grand_coalition(n);
299 T current = complete ? out.total : std::invoke(v, s);
300 out.evaluations += complete ? 0 : 1;
301 into[*it] = current - previous;
302 previous = current;
303 }
304 };
305 for (std::size_t p = 0; p < options.permutations && n > 0; ++p) {
306 stats::shuffle(std::span<std::size_t>(order), g);
307 walk(order.begin(), order.end(), forward);
308 if (options.antithetic) {
309 walk(order.rbegin(), order.rend(), sample);
310 for (std::size_t i = 0; i < n; ++i)
311 sample[i] = (sample[i] + forward[i]) / T(2);
312 } else {
313 sample = forward;
314 }
315 stats.add(std::span<const T>(sample));
316 }
317 for (std::size_t i = 0; i < n; ++i) {
318 out.values[i] = stats.mean[i];
319 out.standard_errors[i] = stats.standard_error(i);
320 }
321 out.samples = stats.count;
322 }

324 template <game V, stats::full_range_generator G, typename Values>
325 void sample_lifts(std::size_t n, const V& v, G& g, lift_sampling options, sampled_attribution<game_value_t<V>, Values>& out)
326 {
327 using T = game_value_t<V>;
328 out.baseline = std::invoke(v, coalition{0});
329 out.total = std::invoke(v, grand_coalition(n));
330 out.evaluations = 2;
331 std::vector<std::size_t> others;
332 T sum = T(0);
333 for (std::size_t i = 0; i < n; ++i) {
334 others.clear();
335 for (std::size_t j = 0; j < n; ++j)
336 if (j != i)
337 others.push_back(j);
338 T mean = T(0), m2 = T(0);
339 std::size_t count = 0;
340 for (std::size_t k = 0; k < options.samples_per_player; ++k) {
341 // |S| is uniform on 0 .. n-1 under the Shapley weights, then S uniform of that size
342 const auto size = static_cast<std::size_t>(stats::uniform_index(g, n));
343 coalition s = 0;
344 for (std::size_t j = 0; j < size; ++j) {
345 const auto pick = j + static_cast<std::size_t>(stats::uniform_index(g, others.size() - j));
346 std::swap(others[j], others[pick]);
347 s = with(s, others[j]);
348 }
349 const T lift = std::invoke(v, with(s, i)) - std::invoke(v, s);
350 out.evaluations += 2;
351 ++count;
352 const T delta = lift - mean;
353 mean += delta / static_cast<T>(count);
354 m2 += delta * (lift - mean);
355 }
356 out.values[i] = mean;
357 out.standard_errors[i] = count < 2 ? std::numeric_limits<T>::quiet_NaN()
358 : std::sqrt(m2 / static_cast<T>(count - 1) / static_cast<T>(count));
359 sum += mean;
360 }
361 out.samples = options.samples_per_player;
362 const T target = out.total - out.baseline;
363 if (options.scale_to_total && sum != T(0)) {
364 const T factor = target / sum;
365 for (std::size_t i = 0; i < n; ++i) {
366 out.values[i] *= factor;
367 out.standard_errors[i] *= std::abs(factor);
368 }
369 }
370 }

426 template <std::size_t N, game V>
427 [[nodiscard]] constexpr fixed_attribution<game_value_t<V>, N> shapley_by_permutations(const V& v)
428 {
429 static_assert(N <= 10, "snicholls::game::shapley_by_permutations: more than 10! orders; use shapley");
430 using T = game_value_t<V>;
431 fixed_attribution<T, N> out{};
432 detail::with_table(N, v, [&out](std::span<const T> table) {
433 out.baseline = table.front();
434 out.total = table.back();
435 for_each_permutation<N>([&](const std::array<std::size_t, N>& order) {
436 coalition s = 0;
437 for (const std::size_t i : order) {
438 const coalition next = with(s, i);
439 out.values[i] = out.values[i] + (table[next] - table[s]);
440 s = next;
441 }
442 });
443 const auto scale = game_scalar_t<T>(1) / static_cast<game_scalar_t<T>>(permutation_count_v<N>);
444 for (T& a : out.values)
445 a = a * scale;
446 });
447 return out;
448 }

479 template <std::size_t N, game V>
480 [[nodiscard]] constexpr std::array<std::array<game_value_t<V>, N>, N> shapley_interactions(const V& v)
481 {
482 static_assert(N <= max_table_players, "snicholls::game::shapley_interactions: too many players");
483 using T = game_value_t<V>;
484 std::array<std::array<T, N>, N> out{};
485 // With fewer than two players no coalition misses two, so the loop body never runs and nothing is divided.
486 detail::with_table(N, v, [&out](std::span<const T> table) {
487 using scalar = game_scalar_t<T>;
488 const coalition all = grand_coalition(N);
489 for (coalition s = 0; s <= all; ++s) {
490 const coalition missing = all & ~s;
491 if (coalition_size(missing) < 2)
492 continue;
493 const scalar w = scalar(1) / (static_cast<scalar>(N - 1) * static_cast<scalar>(binomial(N - 2, coalition_size(s))));
494 for (coalition a = missing; a != 0; a &= a - 1) {
495 const auto i = static_cast<std::size_t>(std::countr_zero(a));
496 for (coalition b = a & (a - 1); b != 0; b &= b - 1) {
497 const auto j = static_cast<std::size_t>(std::countr_zero(b));
498 const T delta = table[s | singleton(i) | singleton(j)] - table[s | singleton(i)] - table[s | singleton(j)] + table[s];
499 out[i][j] = out[i][j] + delta * w;
500 out[j][i] = out[i][j];
501 }
502 }
503 }
504 });
505 return out;
506 }
