90constexpr result<std::size_t, bpe_error> sentencepiece_escape(std::string_view text, std::span<char> out, bool add_dummy_prefix = true) noexcept
91{
92 std::size_t at = 0;
93 const auto put_space = [&]() {
94 for (const char c : escaped_space) { // checked byte by byte, which GCC's -Warray-bounds can follow
95 if (at == out.size())
96 return false;
97 out[at++] = c;
98 }
99 return true;
100 };
101 if (text.empty())
102 return ok(std::size_t{0});
103 if (add_dummy_prefix && !put_space())
104 return err(bpe_error::output_too_small);
105 for (const char c : text) {
106 if (c == ' ') {
107 if (!put_space())
108 return err(bpe_error::output_too_small);
109 } else {
110 if (at == out.size())
111 return err(bpe_error::output_too_small);
112 out[at++] = c;
113 }
114 }
115 return ok(at);
116}

135template <typename Token>
136constexpr byte_fallback make_byte_fallback(Token&& token)
137{
138 byte_fallback f;
139 constexpr char hex[] = "0123456789ABCDEF";
140 for (std::size_t b = 0; b < 256; ++b) {
141 const char name[6] = {'<', '0', 'x', hex[b >> 4], hex[b & 15], '>'};
142 std::uint32_t id = token(std::string_view(name, 6));
143 if (id == no_token) {
144 const char single = static_cast<char>(b);
145 id = token(std::string_view(&single, 1));
146 }
147 f.ids[b] = id;
148 }
149 return f;
150}

162template <typename Token, typename Score>
163constexpr result<std::size_t, bpe_error> sentencepiece_bpe_encode(std::string_view escaped, Token&& token, Score&& score, const byte_fallback& bytes,
164 std::span<std::uint32_t> out, std::span<merge_entry<float>> scratch)
165{
166 // Not lambdas capturing the lookups: MSVC cannot read a captured reference to a function during constant
167 // evaluation.
168 struct priority
169 {
170 Token& token;
171 Score& score;
172 constexpr float operator()(std::string_view left, std::string_view right) const
173 {
174 const std::uint32_t id = token(std::string_view(left.data(), left.size() + right.size()));
175 return id == no_token ? merge_never<float> : -static_cast<float>(score(id));
176 }
177 };
178 return merge_encode(escaped, priority{token, score}, token, bytes, out, scratch);
179}

198template <typename Piece, typename Kind>
199constexpr result<std::size_t, bpe_error> sentencepiece_decode(std::span<const std::uint32_t> tokens, Piece&& piece, Kind&& kind,
200 std::span<char> out, sentencepiece_decode_options options = {})
201{
202 std::size_t at = 0;
203 bool first = options.strip_leading_space;
204 const auto put = [&](char c) {
205 if (at == out.size())
206 return false;
207 out[at++] = c;
208 return true;
209 };
210 const auto hex = [](char c) -> int {
211 if (c >= '0' && c <= '9')
212 return c - '0';
213 if (c >= 'A' && c <= 'F')
214 return c - 'A' + 10;
215 if (c >= 'a' && c <= 'f')
216 return c - 'a' + 10;
217 return -1;
218 };
219 for (const std::uint32_t id : tokens) {
220 const std::string_view p = piece(id);
221 if (p.empty())
222 return err(bpe_error::unknown_bytes);
223 const token_kind k = kind(id);
224 if ((k == token_kind::control || k == token_kind::unknown) && !options.render_special)
225 continue;
226 if (k == token_kind::byte && p.size() == 6 && p.substr(0, 3) == "<0x" && hex(p[3]) >= 0 && hex(p[4]) >= 0) {
227 if (!put(static_cast<char>(hex(p[3]) * 16 + hex(p[4]))))
228 return err(bpe_error::output_too_small);
229 first = false;
230 continue;
231 }
232 const bool escape = k != token_kind::user_defined && k != token_kind::control && k != token_kind::unknown;
233 for (std::size_t i = 0; i < p.size();) {
234 if (escape && p.substr(i, escaped_space.size()) == escaped_space) {
235 i += escaped_space.size();
236 if (first && i == escaped_space.size())
237 continue;
238 if (!put(' '))
239 return err(bpe_error::output_too_small);
240 continue;
241 }
242 if (!put(p[i++]))
243 return err(bpe_error::output_too_small);
244 }
245 first = false;
246 }
247 return ok(at);
248}
