57 // scipy's h: 1 at bin 0 and (even n) n / 2, 2 at the positive frequencies, 0 at the negative.
58 template <typename T>
59 constexpr T weight(std::size_t k, std::size_t n) noexcept
60 {
61 if (k == 0 || 2 * k == n)
62 return T(1);
63 return 2 * k < n ? T(2) : T(0);
64 }

66 template <typename T>
67 constexpr void apply(const fft_plan<T>& plan, std::span<complex<T>> z, std::span<complex<T>> work) noexcept
68 {
69 const std::size_t n = plan.size();
70 plan.template execute<false>(z, work);
71 const T inv = T(1) / T(static_cast<long long>(n));
72 for (std::size_t k = 0; k < n; ++k)
73 z[k] *= weight<T>(k, n) * inv;
74 plan.template execute<true>(z, work);
75 }
76} // namespace detail::hilbert

84template <math::math_number T>
85constexpr void analytic_signal(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> x, std::span<complex<std::type_identity_t<T>>> out,
86 std::span<complex<std::type_identity_t<T>>> work) noexcept
87{
88 const std::size_t n = plan.size();
89 assert(x.size() == n && out.size() == n);
90 if (n == 0)
91 return;
92 for (std::size_t i = 0; i < n; ++i)
93 out[i] = {x[i], T(0)};
94 detail::hilbert::apply(plan, out, work);
95}

103template <math::math_number T>
104constexpr void hilbert_transform(const fft_plan<T>& plan, std::span<const std::type_identity_t<T>> x, std::span<std::type_identity_t<T>> out,
105 std::span<complex<std::type_identity_t<T>>> work) noexcept
106{
107 const std::size_t n = plan.size();
108 assert(out.size() == n && work.size() >= n + plan.work_size());
109 const auto z = work.first(n);
110 analytic_signal(plan, x, z, work.subspan(n));
111 for (std::size_t i = 0; i < n; ++i)
112 out[i] = z[i].im;
113}

139template <math::math_number T>
140constexpr void analytic_signal_backward(const fft_plan<T>& plan, std::span<complex<std::type_identity_t<T>>> grad_z,
141 std::span<std::type_identity_t<T>> grad_x, std::span<complex<std::type_identity_t<T>>> work) noexcept
142{
143 // z = A x with A = F^-1 D F (D real, diagonal); for real x, dL/dx = Re(A^H g) = Re(F^H D F^-H g) = Re(F^-1 D F g),
144 // since F^H = n F^-1 and F^-H = F / n. So it is the same map applied to g, real part taken.
145 const std::size_t n = plan.size();
146 assert(grad_z.size() == n && grad_x.size() == n);
147 if (n == 0)
148 return;
149 detail::hilbert::apply(plan, grad_z, work);
150 for (std::size_t i = 0; i < n; ++i)
151 grad_x[i] = grad_z[i].re;
152}
