76template <math::math_number T>
77struct online_softmax
78{
79 T max = -std::numeric_limits<T>::infinity();
80 T sum = T(0);
81
84 constexpr void add(T x)
85 {
86 if (x == -std::numeric_limits<T>::infinity())
87 return;
88 if (x > max) {
89 sum = sum * math::exp(max - x) + T(1);
90 max = x;
91 } else {
92 sum += math::exp(x - max);
93 }
94 }
95
98 constexpr void merge(const online_softmax& other)
99 {
100 if (other.sum == T(0))
101 return;
102 if (sum == T(0)) {
103 *this = other;
104 return;
105 }
106 if (other.max > max) {
107 sum = sum * math::exp(max - other.max) + other.sum;
108 max = other.max;
109 } else {
110 sum += other.sum * math::exp(other.max - max);
111 }
112 }
113
116 [[nodiscard]] constexpr T log_sum_exp() const
117 {
118 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
119 }
120
124 [[nodiscard]] constexpr T probability(T x) const { return math::exp(x - max) / sum; }
125};

129 template <typename T>
130 constexpr T max_of(std::span<const T> in)
131 {
132 T m = -std::numeric_limits<T>::infinity();
133 for (const T x : in) {
134 if (x != x)
135 return x; // NaN
136 if (x > m)
137 m = x;
138 }
139 return m;
140 }
141
142 template <typename T>
143 constexpr T logsumexp(std::span<const T> in)
144 {
145 const T m = max_of(in);
146 if (m == -std::numeric_limits<T>::infinity() || m != m)
147 return m;
148 T s = T(0);
149 for (const T x : in)
150 s += math::exp(x - m);
151 return m + math::log(s);
152 }
153} // namespace detail::softmax

170template <row In, mutable_row Out>
171 requires std::same_as<row_value_t<In>, row_value_t<Out>>
172constexpr void softmax(const In& in, Out&& out)
173{
174 using T = row_value_t<In>;
175 const auto x = as_row(in);
176 const auto y = as_mutable_row(out);
177 assert(x.size() == y.size());
178 const T m = detail::softmax::max_of(x);
179 T s = T(0);
180 for (std::size_t i = 0; i < x.size(); ++i) {
181 y[i] = math::exp(x[i] - m);
182 s += y[i];
183 }
184 const T inv = T(1) / s;
185 for (T& p : y)
186 p *= inv;
187}

213template <row L, row D, mutable_row Out>
214 requires std::same_as<row_value_t<L>, row_value_t<D>> && std::same_as<row_value_t<L>, row_value_t<Out>>
215constexpr void log_softmax_backward(const L& logprobs, const D& dlogprobs, Out&& dlogits)
216{
217 using T = row_value_t<L>;
218 const auto lp = as_row(logprobs);
219 const auto d = as_row(dlogprobs);
220 const auto g = as_mutable_row(dlogits);
221 assert(lp.size() == d.size() && lp.size() == g.size());
222 T sum = T(0);
223 for (std::size_t i = 0; i < d.size(); ++i)
224 sum += d[i];
225 for (std::size_t i = 0; i < lp.size(); ++i)
226 g[i] = d[i] - math::exp(lp[i]) * sum;
227}

236template <row P, row D, mutable_row Out>
237 requires std::same_as<row_value_t<P>, row_value_t<D>> && std::same_as<row_value_t<P>, row_value_t<Out>>
238constexpr void softmax_backward(const P& probs, const D& dprobs, Out&& dlogits)
239{
240 using T = row_value_t<P>;
241 const auto p = as_row(probs);
242 const auto d = as_row(dprobs);
243 const auto g = as_mutable_row(dlogits);
244 assert(p.size() == d.size() && p.size() == g.size());
245 T dot = T(0);
246 for (std::size_t i = 0; i < p.size(); ++i)
247 dot += p[i] * d[i];
248 for (std::size_t i = 0; i < p.size(); ++i)
249 g[i] = p[i] * (d[i] - dot);
250}

275template <row In, mutable_row Out>
276 requires std::same_as<row_value_t<In>, row_value_t<Out>>
277constexpr row_value_t<In> cross_entropy_with_gradient(const In& logits, std::size_t target, Out&& dlogits,
278 std::type_identity_t<row_value_t<In>> scale = 1)
279{
280 using T = row_value_t<In>;
281 const auto x = as_row(logits);
282 const auto g = as_mutable_row(dlogits);
283 assert(target < x.size() && x.size() == g.size());
284 const T lse = detail::softmax::logsumexp(x);
285 const T loss = lse - x[target];
286 for (std::size_t i = 0; i < x.size(); ++i)
287 g[i] = scale * (math::exp(x[i] - lse) - (i == target ? T(1) : T(0)));
288 return loss;
289}
