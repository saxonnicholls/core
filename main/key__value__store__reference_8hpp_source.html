98 template <typename F>
99 bool kv_reference_read(const kv_reference_map& m, kv_bytes key, F& fn)
100 {
101 const auto it = m.find(key);
102 if (it == m.end())
103 return false;
104 fn(kv_bytes(it->second));
105 return true;
106 }

108 inline kv_reference_map::const_iterator kv_reference_start(const kv_reference_map& m, const kv_range& r)
109 {
110 return r.lower_open() ? m.upper_bound(r.lower()) : m.lower_bound(r.lower());
111 }
112
113 template <typename F>
114 std::size_t kv_reference_scan(const kv_reference_map& m, const kv_range& r, F& fn)
115 {
116 if (r.empty())
117 return 0;
118 std::size_t n = 0;
119 for (auto it = kv_reference_start(m, r); it != m.end() && !r.beyond(it->first); ++it) {
120 ++n;
121 if (!fn(kv_bytes(it->first), kv_bytes(it->second)))
122 break;
123 }
124 return n;
125 }
126 }

248 result<void, kv_errc> put(kv_bytes key, kv_bytes value)
249 {
250 return write([&]() -> result<void, kv_errc> {
251 if (const auto e = tick())
252 return err(*e);
253 auto copy = detail::kv_copy(value);
254 auto& m = writable();
255 if (const auto it = m.find(key); it != m.end())
256 it->second.swap(copy);
257 else
258 m.emplace(detail::kv_copy(key), std::move(copy));
259 ++sequence_;
260 return ok();
261 });
262 }
263
268 result<bool, kv_errc> insert(kv_bytes key, kv_bytes value)
269 {
270 return write([&]() -> result<bool, kv_errc> {
271 if (const auto e = tick())
272 return err(*e);
273 if (data_->find(key) != data_->end())
274 return ok(false);
275 writable().emplace(detail::kv_copy(key), detail::kv_copy(value));
276 ++sequence_;
277 return ok(true);
278 });
279 }

284 result<bool, kv_errc> erase(kv_bytes key)
285 {
286 return write([&]() -> result<bool, kv_errc> {
287 if (const auto e = tick())
288 return err(*e);
289 if (data_->find(key) == data_->end())
290 return ok(false);
291 auto& m = writable();
292 m.erase(m.find(key));
293 ++sequence_;
294 return ok(true);
295 });
296 }

301 result<std::size_t, kv_errc> erase_range(const kv_range& range)
302 {
303 return write([&]() -> result<std::size_t, kv_errc> {
304 if (const auto e = tick())
305 return err(*e);
306 if (range.empty())
307 return ok(std::size_t{0});
308 const auto first = detail::kv_reference_start(*data_, range);
309 if (first == data_->end() || range.beyond(first->first))
310 return ok(std::size_t{0});
311 auto& m = writable();
312 std::size_t n = 0;
313 for (auto it = detail::kv_reference_start(m, range); it != m.end() && !range.beyond(it->first); ++n)
314 it = m.erase(it);
315 ++sequence_;
316 return ok(n);
317 });
318 }

323 result<void, kv_errc> apply(std::span<const kv_write> batch)
324 {
325 return write([&]() -> result<void, kv_errc> {
326 if (batch.empty())
327 return ok();
328 auto next = std::make_shared<detail::kv_reference_map>(*data_);
329 for (const kv_write& w : batch) {
330 if (const auto e = tick())
331 return err(*e);
332 switch (w.op) {
333 case kv_op::put:
334 next->insert_or_assign(detail::kv_copy(w.key), detail::kv_copy(w.value));
335 break;
336 case kv_op::insert:
337 if (next->find(w.key) != next->end())
338 return err(kv_errc::conflict);
339 next->emplace(detail::kv_copy(w.key), detail::kv_copy(w.value));
340 break;
341 case kv_op::erase:
342 if (const auto it = next->find(w.key); it != next->end())
343 next->erase(it);
344 break;
345 }
346 }
347 data_ = std::move(next);
348 ++sequence_;
349 return ok();
350 });
351 }

382 {
383 return write([&]() -> result<kv_checkpoint_info, kv_errc> {
384 if (const auto e = tick())

387 return err(kv_errc::invalid_argument);
388 std::uint64_t bytes = 0;

391 checkpoints_.emplace(std::string(destination), saved{data_, sequence_});
392 return ok(kv_checkpoint_info{sequence_, bytes});

398 result<kv_verify_report, kv_errc> verify() const
399 {
400 if (unavailable_)
401 return err(kv_errc::unavailable);
402 kv_verify_report report{data_->size(), 0};
403 const detail::kv_bytes_order less;
404 for (auto it = data_->begin(); it != data_->end() && std::next(it) != data_->end(); ++it)
405 report.problems += !less(it->first, std::next(it)->first); // strictly ascending, no duplicates
406 return ok(report);

457 // An injected failure due now, spending it; otherwise counts one write toward it.
458 std::optional<kv_errc> tick() noexcept
459 {
460 if (!fault_)
461 return std::nullopt;
462 if (fault_after_ > 0) {
463 --fault_after_;
464 return std::nullopt;
465 }
466 return std::exchange(fault_, std::nullopt);
467 }
468
469 // A write: refused while unavailable; std::bad_alloc becomes an error (nothing has changed when it is thrown).
470 template <typename F>
471 auto write(F&& f) -> decltype(f())
472 {
473 if (unavailable_)
474 return err(kv_errc::unavailable);
475#if SNICHOLLS_RESULT_HAS_EXCEPTIONS
476 try {
477 return f();
478 } catch (const std::bad_alloc&) {
479 return err(kv_errc::out_of_memory);
480 }
481#else
482 return f();
483#endif
484 }
