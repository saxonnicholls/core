64 template <typename Probe>
65 concept key_store_probe = std::is_default_constructible_v<Probe> &&
66 key_store<std::remove_reference_t<decltype(std::declval<Probe&>().store())>> &&
67 requires(Probe& p, capability_errc c, std::string_view r) {
68 { p.supported() } -> std::same_as<key_spec>;
69 { p.unsupported() } -> std::same_as<std::optional<key_spec>>;
70 p.provider_fails(c, r);
71 };
