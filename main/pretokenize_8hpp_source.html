87 // One code point of the text: its value, category and length. Past the end: length 0, category unassigned.
88 struct unit
89 {
90 char32_t cp;
91 unicode_category cat;
92 std::size_t len;
93 };
94
95 constexpr unit at(std::string_view text, std::size_t i) noexcept
96 {
97 if (i >= text.size())
98 return {0, unicode_category::unassigned, 0};
99 const utf8_decoded d = decode_utf8(text, i);
100 return {d.code_point, category(d.code_point), d.length};
101 }
102
103 constexpr bool letter(const unit& u) noexcept { return u.len && is_letter(u.cat); }
104 constexpr bool number(const unit& u) noexcept { return u.len && is_number(u.cat); }
105 constexpr bool space(const unit& u) noexcept { return u.len && is_whitespace(u.cp); }
106 constexpr bool crlf(const unit& u) noexcept { return u.len && (u.cp == '\r' || u.cp == '\n'); }
107 constexpr bool letter_or_mark(const unit& u) noexcept { return u.len && (is_letter(u.cat) || is_mark(u.cat)); }
108 // [^\s\p{L}\p{N}]
109 constexpr bool other(const unit& u) noexcept { return u.len && !space(u) && !letter(u) && !number(u); }
110 // [^\r\n\p{L}\p{N}]
111 constexpr bool lead(const unit& u) noexcept { return u.len && !crlf(u) && !letter(u) && !number(u); }

113 // The bytes of the longest run of units satisfying pred from i (at most max units)
114 template <typename Pred>
115 constexpr std::size_t run(std::string_view text, std::size_t i, Pred pred, std::size_t max = ~std::size_t{0}) noexcept
116 {
117 std::size_t n = 0;
118 for (std::size_t k = 0; k < max; ++k) {
119 const unit u = at(text, i + n);
120 if (!pred(u))
121 break;
122 n += u.len;
123 }
124 return n;
125 }
126
127 // [A]?[B]+ with a greedy optional: the bytes matched at i, or 0
128 template <typename A, typename B>
129 constexpr std::size_t optional_then_run(std::string_view text, std::size_t i, A a, B b) noexcept
130 {
131 const unit c = at(text, i);
132 if (a(c))
133 if (const std::size_t n = run(text, i + c.len, b); n > 0)
134 return c.len + n;
135 return run(text, i, b);
136 }

138 // 's|'t|'re|'ve|'m|'ll|'d, case-sensitive or (?i:...): the bytes matched at i, or 0
139 constexpr bool folds_to(char32_t cp, char lower, bool fold) noexcept
140 {
141 if (cp == static_cast<char32_t>(lower))
142 return true;
143 return fold && (cp == static_cast<char32_t>(lower - 'a' + 'A') || (lower == 's' && cp == U'ſ'));
144 }
145
146 constexpr std::size_t contraction(std::string_view text, std::size_t i, bool fold) noexcept
147 {
148 if (i >= text.size() || text[i] != '\'')
149 return 0;
150 const unit a = at(text, i + 1);
151 for (const char c : {'s', 't'})
152 if (folds_to(a.cp, c, fold) && a.len)
153 return 1 + a.len;
154 const unit b = at(text, i + 1 + a.len);
155 const auto pair = [&](char x, char y) { return a.len && b.len && folds_to(a.cp, x, fold) && folds_to(b.cp, y, fold); };
156 if (pair('r', 'e') || pair('v', 'e'))
157 return 1 + a.len + b.len;
158 if (folds_to(a.cp, 'm', fold) && a.len)
159 return 1 + a.len;
160 if (pair('l', 'l'))
161 return 1 + a.len + b.len;
162 if (folds_to(a.cp, 'd', fold) && a.len)
163 return 1 + a.len;
164 return 0;
165 }

167 // ' ?' then a run of pred, then a run of tail: the bytes matched at i, or 0
168 template <typename Pred, typename Tail>
169 constexpr std::size_t space_run_tail(std::string_view text, std::size_t i, Pred pred, Tail tail) noexcept
170 {
171 const std::size_t sp = (i < text.size() && text[i] == ' ' && pred(at(text, i + 1))) ? 1 : 0;
172 const std::size_t n = run(text, i + sp, pred);
173 if (n == 0)
174 return 0;
175 return sp + n + run(text, i + sp + n, tail);
176 }

178 // \s*[\r\n]+ : up to and including the last \r or \n of the whitespace run at i, or 0
179 constexpr std::size_t whitespace_to_newline(std::string_view text, std::size_t i) noexcept
180 {
181 std::size_t n = 0, end = 0;
182 for (unit u = at(text, i); space(u); u = at(text, i + n)) {
183 n += u.len;
184 if (crlf(u))
185 end = n;
186 }
187 return end;
188 }
189
190 // \s+(?!\S)|\s+ : the whitespace run at i, less its last unit when a non-space follows and the run is longer
191 // than one unit
192 constexpr std::size_t whitespace(std::string_view text, std::size_t i) noexcept
193 {
194 std::size_t n = 0, last = 0, units = 0;
195 for (unit u = at(text, i); space(u); u = at(text, i + n)) {
196 last = u.len;
197 n += u.len;
198 ++units;
199 }
200 if (units > 1 && i + n < text.size())
201 return n - last;
202 return n;
203 }
204
205 constexpr bool is_crlf_or_slash(const unit& u) noexcept { return u.len && (u.cp == '\r' || u.cp == '\n' || u.cp == '/'); }
206 constexpr bool none(const unit&) noexcept { return false; }
207} // namespace detail::pretokenize

299struct gpt4o_pattern
300{
305 constexpr std::size_t operator()(std::string_view text, std::size_t at) const noexcept
306 {
307 namespace d = detail::pretokenize;
308 // Each word alternative with its optional lead: with the lead if that matches, else without
309 for (const auto word : {&gpt4o_pattern::cased_lower, &gpt4o_pattern::cased_upper}) {
310 const d::unit c = d::at(text, at);
311 if (d::lead(c))
312 if (const std::size_t n = word(text, at + c.len))
313 return c.len + n;
314 if (const std::size_t n = word(text, at))
315 return n;
316 }
317 if (const std::size_t n = d::run(text, at, d::number, 3))
318 return n;
319 if (const std::size_t n = d::space_run_tail(text, at, d::other, d::is_crlf_or_slash))
320 return n;
321 if (const std::size_t n = d::whitespace_to_newline(text, at))
322 return n;
323 return d::whitespace(text, at);
324 }

326private:
327 static constexpr bool upper(const detail::pretokenize::unit& u) noexcept { return u.len && is_uppercase_class(u.cat); }
328 static constexpr bool lower(const detail::pretokenize::unit& u) noexcept { return u.len && is_lowercase_class(u.cat); }
329
330 // [Upper]*[Lower]+(contraction)? at i: [Upper]* backtracks to the last position from which [Lower]+ matches
331 static constexpr std::size_t cased_lower(std::string_view text, std::size_t i) noexcept
332 {
333 namespace d = detail::pretokenize;
334 // The ends of the units of the maximal upper run, to backtrack over
335 std::size_t candidate = d::run(text, i, lower) > 0 ? i : ~std::size_t{0}; // [Upper]* empty
336 for (std::size_t p = i;;) {
337 const d::unit u = d::at(text, p);
338 if (!upper(u))
339 break;
340 p += u.len;
341 if (lower(d::at(text, p)))
342 candidate = p; // the longest [Upper]* after which a lower unit follows
343 }
344 if (candidate == ~std::size_t{0})
345 return 0;
346 const std::size_t end = candidate + d::run(text, candidate, lower);
347 return end - i + d::contraction(text, end, true);
348 }
349
350 // [Upper]+[Lower]*(contraction)? at i
351 static constexpr std::size_t cased_upper(std::string_view text, std::size_t i) noexcept
352 {
353 namespace d = detail::pretokenize;
354 const std::size_t u = d::run(text, i, upper);
355 if (u == 0)
356 return 0;
357 const std::size_t end = i + u + d::run(text, i + u, lower);
358 return end - i + d::contraction(text, end, true);
359 }

483struct deepseek_v3_pattern
484{
489 constexpr std::size_t operator()(std::string_view text, std::size_t at) const noexcept
490 {
491 namespace d = detail::pretokenize;
492 constexpr auto ascii_letter = [](const d::unit& u) { return u.len && ((u.cp | 0x20) >= 'a' && (u.cp | 0x20) <= 'z'); };
493 constexpr auto ascii_punct = [](const d::unit& u) {
494 return u.len && u.cp < 0x7F && u.cp > 0x20 && !((u.cp | 0x20) >= 'a' && (u.cp | 0x20) <= 'z') && !(u.cp >= '0' && u.cp <= '9');
495 };
496 if (ascii_punct(d::at(text, at)))
497 if (const std::size_t n = d::run(text, at + 1, ascii_letter))
498 return 1 + n;
499 constexpr auto punct_symbol = [](const d::unit& u) { return u.len && (is_punctuation(u.cat) || is_symbol(u.cat)); };
500 constexpr auto lead = [](const d::unit& u) { return u.len && !d::crlf(u) && !is_letter(u.cat) && !is_punctuation(u.cat) && !is_symbol(u.cat); };
501 if (const std::size_t n = d::optional_then_run(text, at, lead, d::letter_or_mark))
502 return n;
503 if (const std::size_t n = d::space_run_tail(text, at, punct_symbol, d::crlf))
504 return n;
505 if (const std::size_t n = d::whitespace_to_newline(text, at))
506 return n;
507 return d::whitespace(text, at);
508 }

595template <typename Pattern, typename Emit>
596constexpr void split_isolated(std::string_view text, const Pattern& pattern, Emit&& emit)
597{
598 std::size_t gap = 0, at = 0;
599 while (at < text.size()) {
600 const std::size_t n = pattern(text, at);
601 if (n == 0) {
602 at += decode_utf8(text, at).length;
603 continue;
604 }
605 if (gap < at)
606 emit(text.substr(gap, at - gap));
607 emit(text.substr(at, n));
608 at += n;
609 gap = at;
610 }
611 if (gap < text.size())
612 emit(text.substr(gap));
613}

621template <typename Emit, typename First, typename... Rest>
622constexpr void split_sequence(std::string_view text, Emit&& emit, const First& first, const Rest&... rest)
623{
624 if constexpr (sizeof...(Rest) == 0)
625 split_isolated(text, first, emit);
626 else
627 split_isolated(text, first, [&](std::string_view piece) { split_sequence(piece, emit, rest...); });
628}

659template <typename Emit>
660constexpr void pretokenize(pretokenizer kind, std::string_view text, Emit&& emit)
661{
662 switch (kind) {
663 case pretokenizer::gpt2:
664 return split_isolated(text, gpt2_pattern{}, emit);
665 case pretokenizer::llama3:
666 return split_isolated(text, llama3_pattern{}, emit);
667 case pretokenizer::qwen2:
668 return split_isolated(text, qwen2_pattern{}, emit);
669 case pretokenizer::qwen35:
670 return split_isolated(text, qwen35_pattern{}, emit);
671 case pretokenizer::gpt4o:
672 return split_isolated(text, gpt4o_pattern{}, emit);
673 case pretokenizer::deepseek_v3:
674 return split_sequence(text, emit, digits_pattern<3>{}, cjk_kana_pattern, deepseek_v3_pattern{});
675 case pretokenizer::deepseek_llm:
676 return split_sequence(text, emit, crlf_pattern{}, deepseek_letters_pattern, deepseek_punctuation_pattern,
677 trailing_space_pattern{}, cjk_wide_pattern, digits_pattern<>{});
678 case pretokenizer::deepseek_coder:
679 return split_sequence(text, emit, crlf_pattern{}, space_category_pattern<false>{}, space_category_pattern<true>{},
680 cjk_wide_pattern, digits_pattern<1>{});
681 case pretokenizer::falcon:
682 return split_sequence(text, emit, falcon_punctuation_pattern{}, gpt2_pattern{}, three_ascii_digits_pattern{});
683 case pretokenizer::starcoder:
684 return split_sequence(text, emit, digits_pattern<1>{}, gpt2_pattern{});
685 case pretokenizer::newlines:
686 return split_isolated(text, newline_pattern{}, emit);
687 }

732template <typename OnText, typename OnToken>
733constexpr void split_added_tokens(std::string_view text, const token_trie& added, OnText&& on_text, OnToken&& on_token)
734{
735 std::size_t gap = 0, at = 0;
736 while (at < text.size()) {
737 std::size_t n = 0;
738 const std::uint32_t id = added.order.empty() ? no_token : added.longest_prefix(text.substr(at), n);
739 if (id == no_token || n == 0) {
740 ++at;
741 continue;
742 }
743 if (gap < at)
744 on_text(text.substr(gap, at - gap));
745 on_token(id);
746 at += n;
747 gap = at;
748 }
749 if (gap < text.size())
750 on_text(text.substr(gap));
751}
