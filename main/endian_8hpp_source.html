129template <typename T>
130 requires(std::is_integral_v<T> && !std::is_same_v<T, bool>)
131constexpr T byteswap(T v) noexcept
132{
133 // All arithmetic in 64 bits: every shift is narrower than its operand for
134 // every T, so no compiler can warn that one might not be (MSVC C4333 did)
135 std::uint64_t in = static_cast<std::make_unsigned_t<T>>(v), out = 0;
136 for (std::size_t i = 0; i < sizeof(T); ++i) {
137 out = (out << 8) | (in & 0xffu);
138 in >>= 8;
139 }
140 return static_cast<T>(static_cast<std::make_unsigned_t<T>>(out));
141}

148template <byte_order_value T, byte_type B>
149constexpr void store_le(B* p, T v) noexcept
150{
151 const std::uint64_t b = detail::to_bits(v);
152 for (std::size_t i = 0; i < sizeof(T); ++i)
153 p[i] = static_cast<B>(static_cast<unsigned char>((b >> (8 * i)) & 0xffu));
154}

161template <byte_order_value T, byte_type B>
162constexpr void store_be(B* p, T v) noexcept
163{
164 const std::uint64_t b = detail::to_bits(v);
165 for (std::size_t i = 0; i < sizeof(T); ++i)
166 p[sizeof(T) - 1 - i] = static_cast<B>(static_cast<unsigned char>((b >> (8 * i)) & 0xffu));
167}

174template <byte_order_value T, byte_type B>
175constexpr T load_le(const B* p) noexcept
176{
177 std::uint64_t b = 0;
178 for (std::size_t i = 0; i < sizeof(T); ++i)
179 b |= std::uint64_t{detail::octet(p[i])} << (8 * i);
180 return detail::from_bits<T>(static_cast<detail::bits_t<T>>(b));
181}

188template <byte_order_value T, byte_type B>
189constexpr T load_be(const B* p) noexcept
190{
191 std::uint64_t b = 0;
192 for (std::size_t i = 0; i < sizeof(T); ++i)
193 b = (b << 8) | detail::octet(p[i]);
194 return detail::from_bits<T>(static_cast<detail::bits_t<T>>(b));
195}
