200 [[nodiscard]] static constexpr ann_storage suggested_storage(std::size_t n, std::size_t trees, std::size_t leaf_size) noexcept
201 {
202 const std::size_t leaf = leaf_size == 0 ? 1 : leaf_size;
203 const std::size_t per_tree = 4 * ((n + leaf - 1) / leaf) + 3;
204 const std::size_t nodes = trees * per_tree;
205 return {nodes, n * trees, bits ? 0 : nodes * D};
206 }

219 template <stats::full_range_generator G>
220 [[nodiscard]] static constexpr result<ann_forest, ann_error> build(std::span<const Point> points, std::size_t trees, std::size_t leaf_size, std::span<ann_node> nodes,
221 std::span<std::uint32_t> items, std::span<Scalar> normals, G& g, const Metric& m = Metric{})
222 {
223 const std::size_t n = points.size();
224 if (n == 0 || trees == 0 || leaf_size == 0 || n > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max()) || items.size() < n * trees)
225 return err(ann_error::invalid_parameters);
226 for (const auto& p : points)
227 if (metric::dimension(p) != D)
228 return err(ann_error::invalid_parameters);
229 if (nodes.size() < trees)
230 return err(ann_error::out_of_storage);
231 // Breadth-first over the node array itself: a pending node holds its item range in `first` and `count`
232 // and is marked by `left == -2`; its children are allocated at the end, so the array is the queue. No
233 // stack, no depth limit.
234 std::size_t used = trees;
235 for (std::size_t t = 0; t < trees; ++t) {
236 const std::size_t base = t * n;
237 for (std::size_t i = 0; i < n; ++i)
238 items[base + i] = static_cast<std::uint32_t>(i);
239 nodes[t] = {-2, -1, static_cast<std::uint32_t>(base), static_cast<std::uint32_t>(n), 0, 0};
240 }
241 for (std::size_t k = 0; k < used; ++k) {
242 ann_node& node = nodes[k];
243 if (node.left != -2)
244 continue;
245 const std::size_t lo = node.first, hi = lo + node.count;
246 if (node.count <= leaf_size) {
247 node.left = -1;
248 continue;
249 }
250 if (used + 2 > nodes.size())
251 return err(ann_error::out_of_storage);
252 if constexpr (!bits) {
253 if ((k + 1) * D > normals.size())
254 return err(ann_error::out_of_storage);
255 }
256 const std::size_t mid = split_range(points, items, lo, hi, k, node, normals, g, m);
257 const std::size_t left = used, right = used + 1;
258 used += 2;
259 nodes[left] = {-2, -1, static_cast<std::uint32_t>(lo), static_cast<std::uint32_t>(mid - lo), 0, 0};
260 nodes[right] = {-2, -1, static_cast<std::uint32_t>(mid), static_cast<std::uint32_t>(hi - mid), 0, 0};
261 ann_node& split = nodes[k];
262 split.left = static_cast<std::int32_t>(left);
263 split.right = static_cast<std::int32_t>(right);
264 split.first = 0;
265 split.count = 0;
266 }
267 const ann_layout layout{static_cast<std::uint32_t>(n), static_cast<std::uint32_t>(trees), static_cast<std::uint32_t>(leaf_size), static_cast<std::uint32_t>(used)};
268 return ok(ann_forest(layout, points, nodes.first(used), items.first(n * trees), normals.first(bits ? 0 : used * D), m));
269 }

302 template <metric::point Q>
303 requires metric::distance_for<Metric, Point, Q>
304 constexpr std::size_t query(const Q& q, std::span<ann_neighbour> out, std::span<std::uint32_t> candidates, std::span<ann_frontier> frontier) const
305 {
306 if (out.empty() || candidates.empty() || frontier.empty() || layout_.points == 0)
307 return 0;
308 std::size_t heap = 0;
309 auto less = [](const ann_frontier& a, const ann_frontier& b) { return a.priority < b.priority; };
310 auto push = [&](float priority, std::int32_t node) {
311 if (heap == frontier.size())
312 return; // full: drop it (lower recall, never a wrong answer)
313 frontier[heap++] = {priority, node};
314 std::push_heap(frontier.begin(), frontier.begin() + static_cast<std::ptrdiff_t>(heap), less);
315 };
316 for (std::uint32_t t = 0; t < layout_.trees; ++t)
317 push(std::numeric_limits<float>::infinity(), static_cast<std::int32_t>(t));
318 std::size_t found = 0;
319 while (heap > 0 && found < candidates.size()) {
320 std::pop_heap(frontier.begin(), frontier.begin() + static_cast<std::ptrdiff_t>(heap), less);
321 const ann_frontier top = frontier[--heap];
322 const ann_node& node = nodes_[static_cast<std::size_t>(top.node)];
323 if (node.count > 0) {
324 for (std::uint32_t i = 0; i < node.count && found < candidates.size(); ++i)
325 candidates[found++] = items_[node.first + i];
326 continue;
327 }
328 const float mg = static_cast<float>(margin(q, node, static_cast<std::size_t>(top.node)));
329 push(top.priority < mg ? top.priority : mg, node.right);
330 push(top.priority < -mg ? top.priority : -mg, node.left);
331 }
332 const auto cand = candidates.first(found);
333 std::sort(cand.begin(), cand.end());
334 const auto last = std::unique(cand.begin(), cand.end());
335 const std::size_t unique = static_cast<std::size_t>(last - cand.begin());
336 std::size_t have = 0;
337 const std::size_t k = out.size();
338 for (std::size_t c = 0; c < unique; ++c) {
339 const double d = static_cast<double>(metric_(points_[cand[c]], q));
340 if (have == k && !(d < out[k - 1].distance))
341 continue;
342 std::size_t j = have < k ? have++ : k - 1;
343 for (; j > 0 && out[j - 1].distance > d; --j)
344 out[j] = out[j - 1];
345 out[j] = {cand[c], d};
346 }
347 return have;
348 }

354 {
355 using W = metric::value_t<Point>;

360 static constexpr void normalise(vec& v)
361 {
362 Scalar s(0);
363 for (const Scalar x : v)
364 s += x * x;
365 if (s > Scalar(0)) {
366 const Scalar r = static_cast<Scalar>(math::sqrt(s));
367 for (Scalar& x : v)
368 x /= r;
369 }
370 }

382 // The signed distance of q from a split: hyperplane margin, or ±1 by a bit
383 template <typename Q>
384 constexpr double margin(const Q& q, const ann_node& node, std::size_t index) const
385 {
386 if constexpr (bits) {
387 return bit_of(q, node.split) ? 1.0 : -1.0;
388 } else {
389 double s = node.offset;
390 for (std::size_t i = 0; i < D; ++i)
391 s += static_cast<double>(normals_[index * D + i]) * static_cast<double>(coord(q, i));
392 return s;
393 }
394 }

396 // Chooses the split of items[lo, hi) and partitions them: negative side first. Returns the boundary.
397 template <typename G>
398 static constexpr std::size_t split_range(std::span<const Point> points, std::span<std::uint32_t> items, std::size_t lo, std::size_t hi, std::size_t index, ann_node& node,
399 std::span<Scalar> normals, G& g, const Metric& m)
400 {
401 static_cast<void>(m);
402 const std::size_t size = hi - lo;
403 auto positive = [&](std::uint32_t item) -> int { // 1 positive, 0 negative, -1 on the plane
404 if constexpr (bits) {
405 return bit_of(points[item], node.split) ? 1 : 0;
406 } else {
407 Scalar s = node.offset;
408 for (std::size_t i = 0; i < D; ++i)
409 s += normals[index * D + i] * coord(points[item], i);
410 return s > Scalar(0) ? 1 : s < Scalar(0) ? 0 : -1;
411 }
412 };
413 if constexpr (bits) {
414 constexpr std::uint32_t total = static_cast<std::uint32_t>(D * sizeof(metric::value_t<Point>) * 8);
415 node.split = static_cast<std::uint32_t>(stats::uniform_index(g, total));
416 for (int attempt = 0; attempt < 32; ++attempt) {
417 std::size_t ones = 0;
418 for (std::size_t i = lo; i < hi; ++i)
419 ones += bit_of(points[items[i]], node.split) ? 1u : 0u;
420 if (ones > 0 && ones < size)
421 break;
422 node.split = static_cast<std::uint32_t>(stats::uniform_index(g, total));
423 }
424 } else {
425 // Two-means over random members: two centroids, each weighted by its count
426 const std::size_t i0 = lo + static_cast<std::size_t>(stats::uniform_index(g, size));
427 std::size_t j0 = lo + static_cast<std::size_t>(stats::uniform_index(g, size - 1));
428 if (j0 >= i0)
429 ++j0;
430 vec p = load(points[items[i0]]), q = load(points[items[j0]]);
431 Scalar ic(1), jc(1);
432 for (int step = 0; step < 200; ++step) {
433 const vec x = load(points[items[lo + static_cast<std::size_t>(stats::uniform_index(g, size))]]);
434 Scalar di(0), dj(0);
435 for (std::size_t c = 0; c < D; ++c) {
436 di += (p[c] - x[c]) * (p[c] - x[c]);
437 dj += (q[c] - x[c]) * (q[c] - x[c]);
438 }
439 di *= ic;
440 dj *= jc;
441 if (di < dj) {
442 for (std::size_t c = 0; c < D; ++c)
443 p[c] = (p[c] * ic + x[c]) / (ic + Scalar(1));
444 ic += Scalar(1);
445 } else if (dj < di) {
446 for (std::size_t c = 0; c < D; ++c)
447 q[c] = (q[c] * jc + x[c]) / (jc + Scalar(1));
448 jc += Scalar(1);
449 }
450 }
451 vec normal{};
452 for (std::size_t c = 0; c < D; ++c)
453 normal[c] = p[c] - q[c];
454 normalise(normal);
455 Scalar offset(0);
456 if constexpr (!through_origin)
457 for (std::size_t c = 0; c < D; ++c)
458 offset -= normal[c] * (p[c] + q[c]) / Scalar(2);
459 for (std::size_t c = 0; c < D; ++c)
460 normals[index * D + c] = normal[c];
461 node.offset = static_cast<float>(offset);
462 }
463 // Partition: negative side first; a point on the plane goes to a random side
464 std::size_t i = lo, j = hi;
465 while (i < j) {
466 int s = positive(items[i]);
467 if (s < 0)
468 s = static_cast<int>(stats::uniform_index(g, 2));
469 if (s == 0)
470 ++i;
471 else
472 std::swap(items[i], items[--j]);
473 }
474 if (i == lo || i == hi) {
475 // Degenerate: shuffle and halve, and make the split neutral so a query explores both sides
476 stats::shuffle(items.subspan(lo, size), g);
477 if constexpr (!bits) {
478 for (std::size_t c = 0; c < D; ++c)
479 normals[index * D + c] = Scalar(0);
480 node.offset = 0;
481 }
482 return lo + size / 2;
483 }
484 return i;
485 }
