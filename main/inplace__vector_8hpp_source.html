234 constexpr iterator insert(const_iterator pos, const T& v)
235 {
236 if (size_ == N)
237 throw std::length_error("snicholls::containers::inplace_vector: full");
238 const auto i = static_cast<size_type>(pos - begin());
239 std::move_backward(data_.data() + i, data_.data() + size_, data_.data() + size_ + 1);
240 data_[i] = v;
241 ++size_;
242 return data_.data() + i;
243 }

248 constexpr iterator erase(const_iterator pos)
249 {
250 const auto i = static_cast<size_type>(pos - begin());
251 std::move(data_.data() + i + 1, data_.data() + size_, data_.data() + i);
252 data_[--size_] = T{};
253 return data_.data() + i;
254 }

288 template <typename T, std::size_t N, typename Pred>
289 constexpr std::size_t erase_if(inplace_vector<T, N>& v, Pred pred)
290 {
291 const auto kept = std::remove_if(v.begin(), v.end(), pred);
292 const auto removed = static_cast<std::size_t>(v.end() - kept);
293 v.resize(v.size() - removed);
294 return removed;
295 }
