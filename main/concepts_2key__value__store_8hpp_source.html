342 class kv_write_batch {
343 public:
351 template <kv_byte_source K, kv_byte_source V>
352 kv_write_batch& put(const K& key, const V& value) { return add(kv_op::put, kv_bytes_of(key), kv_bytes_of(value)); }
353
361 template <kv_byte_source K, kv_byte_source V>
362 kv_write_batch& insert(const K& key, const V& value) { return add(kv_op::insert, kv_bytes_of(key), kv_bytes_of(value)); }
363
369 template <kv_byte_source K>
370 kv_write_batch& erase(const K& key) { return add(kv_op::erase, kv_bytes_of(key), {}); }
371
374 [[nodiscard]] std::span<const kv_write> writes() const noexcept { return writes_; }
377 operator std::span<const kv_write>() const noexcept { return writes_; } // NOLINT(google-explicit-constructor)
380 [[nodiscard]] std::size_t size() const noexcept { return writes_.size(); }
383 [[nodiscard]] bool empty() const noexcept { return writes_.empty(); }
385 void clear() noexcept
386 {
387 writes_.clear();
388 storage_.clear();
389 }
390
391 private:
392 kv_write_batch& add(kv_op op, kv_bytes key, kv_bytes value)
393 {
394 auto block = std::make_unique<std::byte[]>(key.size() + value.size() + 1); // + 1: never a zero-length array
395 std::copy(key.begin(), key.end(), block.get());
396 std::copy(value.begin(), value.end(), block.get() + key.size());
397 writes_.reserve(writes_.size() + 1);
398 storage_.reserve(storage_.size() + 1);
399 writes_.push_back({op, kv_bytes(block.get(), key.size()), kv_bytes(block.get() + key.size(), value.size())});
400 storage_.push_back(std::move(block)); // cannot throw: room reserved
401 return *this;
402 }
403
404 std::vector<kv_write> writes_;
405 std::vector<std::unique_ptr<std::byte[]>> storage_;
406 };

425 template <typename S>
426 concept kv_readable = requires(const S& s, kv_bytes key, const kv_range& range, kv_read_visitor rv, kv_scan_visitor sv) {
427 { s.read(key, rv) } -> std::same_as<result<bool, kv_errc>>;
428 { s.contains(key) } -> std::same_as<result<bool, kv_errc>>;
429 { s.scan(range, sv) } -> std::same_as<result<std::size_t, kv_errc>>;

612 template <kv_readable S, typename F>
613 requires std::is_invocable_v<F&, kv_bytes, kv_bytes>
614 result<std::size_t, kv_errc> kv_scan(const S& store, const kv_range& range, F&& fn)
615 {
616 using answer = std::invoke_result_t<F&, kv_bytes, kv_bytes>;
617 static_assert(std::is_void_v<answer> || std::is_convertible_v<answer, bool>,
618 "snicholls::kv_scan: the visitor must return void or bool");
619 if constexpr (std::is_void_v<answer>)
620 return store.scan(range, [&](kv_bytes k, kv_bytes v) {
621 fn(k, v);
622 return true;
623 });
624 else
625 return store.scan(range, kv_scan_visitor(fn));
626 }

659 inline constexpr std::string_view kv_digest_v1_tag{"crtp.kv.digest.v1\0", 18};
660
672 template <kv_readable S>
673 result<digest<32>, kv_errc> kv_digest_v1(const S& store, const kv_range& range = kv_range::all())
674 {
675 const auto u64_le = [](std::uint64_t n) {
676 std::array<std::byte, 8> out{};
677 for (std::size_t i = 0; i < 8; ++i)
678 out[i] = static_cast<std::byte>(n >> (8 * i));
679 return out;
680 };
681 sha256 h;
682 h.update(kv_digest_v1_tag);
683 auto pairs = store.scan(range, [&](kv_bytes k, kv_bytes v) {
684 h.update(u64_le(k.size())).update(k).update(u64_le(v.size())).update(v);
685 return true;
686 });
687 if (pairs.is_err())
688 return err(pairs.unwrap_err_unchecked());
689 h.update(u64_le(*pairs));
690 return ok(h.finish());
691 }
692}
