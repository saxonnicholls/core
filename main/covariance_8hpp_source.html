72 // The data about each column's mean, as one contiguous T x N block.
73 template <std::ranges::input_range Rows>
74 std::vector<double> centred(const Rows& rows, std::size_t& t, std::size_t& n)
75 {
76 std::vector<double> x;
77 t = 0;
78 n = 0;
79 for (const auto& row : rows) {
80 const auto width = static_cast<std::size_t>(std::ranges::size(row));
81 assert(t == 0 || width == n);
82 n = width;
83 for (const auto& v : row)
84 x.push_back(static_cast<double>(v));
85 ++t;
86 }
87 for (std::size_t j = 0; j < n; ++j) {
88 double mean = 0;
89 for (std::size_t r = 0; r < t; ++r)
90 mean += x[r * n + j];
91 mean /= static_cast<double>(t);
92 for (std::size_t r = 0; r < t; ++r)
93 x[r * n + j] -= mean;
94 }
95 return x;
96 }

98 inline std::vector<double> cross(const std::vector<double>& x, std::size_t t, std::size_t n)
99 {
100 std::vector<double> s(n * n);
101 for (std::size_t r = 0; r < t; ++r)
102 for (std::size_t i = 0; i < n; ++i) {
103 const double xi = x[r * n + i];
104 for (std::size_t j = i; j < n; ++j)
105 s[i * n + j] += xi * x[r * n + j];
106 }
107 for (std::size_t i = 0; i < n; ++i)
108 for (std::size_t j = 0; j < i; ++j)
109 s[i * n + j] = s[j * n + i];
110 return s;
111 }

140 template <std::ranges::input_range Rows>
141 [[nodiscard]] covariance_matrix ledoit_wolf(const Rows& rows)
142 {
143 std::size_t t = 0, n = 0;
144 const std::vector<double> x = detail::centred(rows, t, n);
145 covariance_matrix out;
146 out.n = n;
147 if (t == 0 || n == 0)
148 return out;
149 const std::vector<double> xtx = detail::cross(x, t, n);
150 std::vector<double> x2(x.size());
151 for (std::size_t k = 0; k < x.size(); ++k)
152 x2[k] = x[k] * x[k];
153 const auto td = static_cast<double>(t), nd = static_cast<double>(n);
154 double trace = 0;
155 for (std::size_t i = 0; i < n; ++i)
156 trace += xtx[i * n + i] / td;
157 const double mu = trace / nd;
158 // beta_: the sum of the entries of X2' X2; delta_: the sum of the squared entries of X' X, over T^2
159 double beta_ = 0;
160 for (std::size_t r = 0; r < t; ++r) {
161 double row_sum = 0;
162 for (std::size_t j = 0; j < n; ++j)
163 row_sum += x2[r * n + j];
164 beta_ += row_sum * row_sum;
165 }
166 double delta_ = 0;
167 for (const double v : xtx)
168 delta_ += v * v;
169 delta_ /= td * td;
170 double beta = (beta_ / td - delta_) / (nd * td);
171 const double delta = (delta_ - 2.0 * mu * trace + nd * mu * mu) / nd;
172 beta = beta < delta ? beta : delta;
173 out.shrinkage = beta == 0 ? 0.0 : beta / delta;
174 out.target = mu;
175 out.values.resize(n * n);
176 for (std::size_t i = 0; i < n; ++i)
177 for (std::size_t j = 0; j < n; ++j)
178 out.values[i * n + j] = (1 - out.shrinkage) * xtx[i * n + j] / td + (i == j ? out.shrinkage * mu : 0.0);
179 return out;
180 }
