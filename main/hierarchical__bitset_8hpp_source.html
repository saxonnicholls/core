88 void resize(std::size_t n)
89 {
90 std::vector<std::vector<std::uint64_t>> levels;
91 std::size_t bits = n;
92 do {
93 const std::size_t words = (bits + 63) / 64;
94 levels.emplace_back(words == 0 ? 1 : words, 0);
95 bits = words;
96 } while (bits > 1);
97 levels_ = std::move(levels);
98 size_ = n;
99 }

148 [[nodiscard]] std::size_t find_next(std::size_t i) const noexcept
149 {
150 if (i >= size_) return npos;
151 std::size_t level = 0;
152 // Climb: look in the current word at or after the position; if empty, move to the next word via the parent.
153 for (;;) {
154 const auto& words = levels_[level];
155 const std::size_t w = i >> 6;
156 if (w >= words.size()) return npos;
157 const std::uint64_t bits = words[w] & (~std::uint64_t{0} << (i & 63));
158 if (bits) {
159 i = (w << 6) | static_cast<std::size_t>(std::countr_zero(bits));
160 break;
161 }
162 if (level + 1 == levels_.size()) return npos;
163 i = w + 1; // next word at this level == next bit at the level above
164 ++level;
165 if ((i >> 6) >= levels_[level].size()) return npos;
166 }
167 // Descend: take the lowest set bit of each word below.
168 while (level > 0) {
169 --level;
170 i = (i << 6) | static_cast<std::size_t>(std::countr_zero(levels_[level][i]));
171 }
172 return i < size_ ? i : npos;
173 }
174
178 [[nodiscard]] std::size_t find_prev(std::size_t i) const noexcept
179 {
180 if (size_ == 0) return npos;
181 if (i >= size_) i = size_ - 1;
182 std::size_t level = 0;
183 for (;;) {
184 const auto& words = levels_[level];
185 const std::size_t w = i >> 6;
186 const unsigned shift = 63 - static_cast<unsigned>(i & 63);
187 const std::uint64_t bits = words[w] & (~std::uint64_t{0} >> shift);
188 if (bits) {
189 i = (w << 6) | (63 - static_cast<std::size_t>(std::countl_zero(bits)));
190 break;
191 }
192 if (level + 1 == levels_.size() || w == 0) return npos;
193 i = w - 1;
194 ++level;
195 }
196 while (level > 0) {
197 --level;
198 i = (i << 6) | (63 - static_cast<std::size_t>(std::countl_zero(levels_[level][i])));
199 }
200 return i;
201 }

203private:
