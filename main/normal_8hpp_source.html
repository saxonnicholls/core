67 // A uniform in (0, 1), (2i + 1) 2^-53 for the top 52 bits i: every value is a double, the largest 1 - 2^-53
68 // (with 53 bits, (i + 1/2) 2^-53 rounds to 1 at the top and its quantile is infinite)
69 template <typename G>
70 constexpr double open_uniform(G& g) noexcept(noexcept(g()))
71 {
72 return static_cast<double>((random_u64(g) >> 12) * 2 + 1) * 0x1.0p-53;
73 }
74 } // namespace detail::normal

98 struct normal_box_muller {
102 template <full_range_generator G>
103 [[nodiscard]] constexpr double operator()(G& g) noexcept(noexcept(g()))
104 {
105 if (has_spare_) {
106 has_spare_ = false;
107 return spare_;
108 }
109 const double u1 = detail::normal::open_uniform(g); // in (0, 1): the logarithm is finite
110 const double u2 = static_cast<double>(random_u64(g) >> 11) * 0x1.0p-53;
111 const double r = math::sqrt(-2.0 * math::log(u1)), t = 2.0 * math::pi * u2;
112 spare_ = r * math::sin(t);
113 has_spare_ = true;
114 return r * math::cos(t);
115 }
116
117 private:
118 double spare_ = 0.0;
119 bool has_spare_ = false;
120 };
