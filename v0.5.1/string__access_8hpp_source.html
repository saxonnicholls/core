48inline void snicholls_character_array_is_not_nul_terminated() noexcept {}

50template <typename C, std::size_t N>
51constexpr void check_string_literal(const C (&a)[N]) noexcept
52{
53 if (std::is_constant_evaluated()) {
54 if (a[N - 1] != C{}) snicholls_character_array_is_not_nul_terminated();
55 }
56 else {
57 assert(a[N - 1] == C{} && "snicholls: a character array is a string literal; its last element must be NUL");
58 }
59}

64// Pointer to the first code unit (null only for a null C*).
65template <code_unit_sequence S>
66constexpr const code_unit_t<S>* data_of(const S& s) noexcept
67{
68 using unit = code_unit_t<S>;
69 constexpr carrier kind = classify<S>::kind;
70 if constexpr (kind == carrier::pointer) {
71 return s;
72 }
73 else if constexpr (kind == carrier::array) {
74 if constexpr (is_text_code_unit_v<unit>) check_string_literal(s);
75 return s;
76 }
77 else {
78 return s.data();
79 }
80}

90template <code_unit_sequence S>
91constexpr std::size_t length_of(const S& s) noexcept
92{
93 constexpr detail::carrier kind = detail::classify<S>::kind;
94 if constexpr (kind == detail::carrier::pointer) {
95 return s == nullptr ? 0 : std::char_traits<code_unit_t<S>>::length(s);
96 }
97 else if constexpr (length_kind_v<S> == length_kind::compile_time) {
98 if constexpr (kind == detail::carrier::array && string_like<S>) detail::check_string_literal(s);
99 return static_length_v<S>;
100 }
101 else {
102 return s.size();
103 }
104}

176// Sequences whose bytes are well defined and portable: byte sequences, and text of one-byte code units
177// (char, char8_t), whose bytes are its encoding (UTF-8). Wider text (wchar_t, char16_t, char32_t) is
178// rejected: its bytes depend on platform endianness and wchar_t's width - use std::as_bytes explicitly.
179template <typename S>
180concept byte_viewable = byte_sequence<S> || (string_like<S> && sizeof(code_unit_t<S>) == 1);
181
186template <byte_viewable S>
187constexpr std::span<const std::byte> as_bytes_view(const S& s) noexcept
188{
189 const auto* data = detail::data_of(s);
190 if constexpr (detail::classify<S>::kind == detail::carrier::pointer) {
191 if (data == nullptr) return {};
192 }
193 const std::size_t n = length_of(s);
194 if constexpr (std::is_same_v<code_unit_t<S>, std::byte>)
195 return std::span<const std::byte>(data, n);
196 else
197 return std::span<const std::byte>(reinterpret_cast<const std::byte*>(data), n);
198}
