149 constexpr void append_decimal(unsigned value) noexcept
150 {
151 char digits[10]{};
152 std::size_t n = 0;
153 do {
154 digits[n++] = static_cast<char>('0' + value % 10);
155 value /= 10;
156 } while (value != 0);
157 while (n != 0) append(digits[--n]);
158 }
159 };

165 constexpr colour_text<7> hex(colour c) noexcept
166 {
167 constexpr std::string_view digits = "0123456789abcdef";
168 colour_text<7> out;
169 out.append('#');
170 for (int shift = 20; shift >= 0; shift -= 4) out.append(digits[(c.rgb >> shift) & 0xFu]);
171 return out;
172 }

178 constexpr colour_text<20> ansi_start(colour c) noexcept
179 {
180 colour_text<20> out;
181 out.append("\x1b[");
182 if (c.ansi != ansi_colour::none) {
183 out.append_decimal(static_cast<unsigned>(c.ansi));
184 } else {
185 out.append("38;2;");
186 out.append_decimal((c.rgb >> 16) & 0xFFu);
187 out.append(';');
188 out.append_decimal((c.rgb >> 8) & 0xFFu);
189 out.append(';');
190 out.append_decimal(c.rgb & 0xFFu);
191 }
192 out.append('m');
193 return out;
194 }
