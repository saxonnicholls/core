131 [[nodiscard]] constexpr std::uint64_t binomial(std::size_t n, std::size_t k) noexcept
132 {
133 if (k > n)
134 return 0;
135 std::array<std::uint64_t, 65> row{};
136 row[0] = 1;
137 for (std::size_t r = 1; r <= n; ++r)
138 for (std::size_t c = r; c > 0; --c)
139 row[c] += row[c - 1];
140 return row[k];
141 }

198 template <typename F>
199 constexpr void for_each_coalition_of_size(std::size_t n, std::size_t k, F&& f) noexcept(noexcept(f(coalition{})))
200 {
201 assert(n < 64);
202 if (k > n)
203 return;
204 if (k == 0) {
205 f(coalition{0});
206 return;
207 }
208 const coalition limit = coalition{1} << n;
209 for (coalition s = (coalition{1} << k) - 1; s < limit;) {
210 f(s);
211 const coalition lowest = s & (~s + 1);
212 const coalition ripple = s + lowest;
213 s = (((ripple ^ s) >> 2) / lowest) | ripple;
214 }
215 }

226 template <typename F>
227 constexpr void for_each_coalition_gray(std::size_t n, F&& f) noexcept(noexcept(f(coalition{}, std::size_t{})))
228 {
229 assert(n < 64);
230 coalition s = 0;
231 f(s, no_player);
232 const coalition count = grand_coalition(n);
233 for (coalition step = 1; step <= count && step != 0; ++step) {
234 const auto player = static_cast<std::size_t>(std::countr_zero(step));
235 s ^= singleton(player);
236 f(s, player);
237 }
