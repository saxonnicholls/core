105 template <std::size_t States, std::size_t Horizon, typename Payoff>
106 [[nodiscard]] constexpr fixed_snell<States, Horizon> snell_envelope(const std::array<std::array<double, States>, States>& p, Payoff&& g,
107 double discount = 1.0)
108 {
109 fixed_snell<States, Horizon> out{};
110 for (std::size_t s = 0; s < States; ++s) {
111 out.payoff[Horizon][s] = std::invoke(g, Horizon, s);
112 out.value[Horizon][s] = out.payoff[Horizon][s];
113 out.continuation[Horizon][s] = -std::numeric_limits<double>::infinity();
114 }
115 for (std::size_t t = Horizon; t-- > 0;)
116 for (std::size_t s = 0; s < States; ++s) {
117 double expected = 0;
118 for (std::size_t j = 0; j < States; ++j)
119 expected += p[s][j] * out.value[t + 1][j];
120 out.continuation[t][s] = discount * expected;
121 out.payoff[t][s] = std::invoke(g, t, s);
122 out.value[t][s] = out.payoff[t][s] > out.continuation[t][s] ? out.payoff[t][s] : out.continuation[t][s];
123 }
124 return out;
125 }

172 template <typename Payoff>
173 [[nodiscard]] snell snell_envelope(const markov_chain& chain, std::size_t horizon, Payoff&& g, double discount = 1.0)
174 {
175 const std::size_t n = chain.size();
176 snell out;
177 out.states = n;
178 out.horizon = horizon;
179 out.value.assign((horizon + 1) * n, 0.0);
180 out.continuation.assign((horizon + 1) * n, -std::numeric_limits<double>::infinity());
181 out.payoff.assign((horizon + 1) * n, 0.0);
182 for (std::size_t s = 0; s < n; ++s) {
183 out.payoff[horizon * n + s] = std::invoke(g, horizon, s);
184 out.value[horizon * n + s] = out.payoff[horizon * n + s];
185 }
186 for (std::size_t t = horizon; t-- > 0;) {
187 const std::span<const double> next(out.value.data() + (t + 1) * n, n);
188 for (std::size_t s = 0; s < n; ++s) {
189 const double c = discount * chain.expect(s, next);
190 const double stop = std::invoke(g, t, s);
191 out.continuation[t * n + s] = c;
192 out.payoff[t * n + s] = stop;
193 out.value[t * n + s] = stop > c ? stop : c;
194 }
195 }
196 return out;
197 }
