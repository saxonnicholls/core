161 template <typename T, std::size_t Rank>
162 class md_span {
163 static_assert(Rank >= 1, "snicholls::containers::md_span: at least one dimension");
164
165 public:
169 constexpr md_span(T* data, std::array<std::size_t, Rank> extents) noexcept : data_(data), extents_(extents)
170 {
171 std::size_t stride = 1;
172 for (std::size_t d = Rank; d-- > 0;) {
173 strides_[d] = stride;
174 stride *= extents_[d];
175 }
176 size_ = stride;
177 }

186 {
187 std::size_t offset = 0, d = 0;
188 ((offset += static_cast<std::size_t>(is) * strides_[d++]), ...);
189 return data_[offset];
190 }

198 [[nodiscard]] constexpr std::size_t size() const noexcept { return size_; }
