75template <math::math_number T>
76struct online_softmax
77{
78 T max = -std::numeric_limits<T>::infinity();
79 T sum = T(0);
80
83 constexpr void add(T x)
84 {
85 if (x == -std::numeric_limits<T>::infinity())
86 return;
87 if (x > max) {
88 sum = sum * math::exp(max - x) + T(1);
89 max = x;
90 } else {
91 sum += math::exp(x - max);
92 }
93 }
94
97 constexpr void merge(const online_softmax& other)
98 {
99 if (other.sum == T(0))
100 return;
101 if (sum == T(0)) {
102 *this = other;
103 return;
104 }
105 if (other.max > max) {
106 sum = sum * math::exp(max - other.max) + other.sum;
107 max = other.max;
108 } else {
109 sum += other.sum * math::exp(other.max - max);
110 }
111 }
112
115 [[nodiscard]] constexpr T log_sum_exp() const
116 {
117 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
118 }
119
123 [[nodiscard]] constexpr T probability(T x) const { return math::exp(x - max) / sum; }
124};

128 template <typename T>
129 constexpr T max_of(std::span<const T> in)
130 {
131 T m = -std::numeric_limits<T>::infinity();
132 for (const T x : in) {
133 if (x != x)
134 return x; // NaN
135 if (x > m)
136 m = x;
137 }
138 return m;
139 }
140
141 template <typename T>
142 constexpr T logsumexp(std::span<const T> in)
143 {
144 const T m = max_of(in);
145 if (m == -std::numeric_limits<T>::infinity() || m != m)
146 return m;
147 T s = T(0);
148 for (const T x : in)
149 s += math::exp(x - m);
150 return m + math::log(s);
151 }

169template <row In, mutable_row Out>
170 requires std::same_as<row_value_t<In>, row_value_t<Out>>
171constexpr void softmax(const In& in, Out&& out)
172{
173 using T = row_value_t<In>;
174 const auto x = as_row(in);
175 const auto y = as_mutable_row(out);
176 assert(x.size() == y.size());
177 const T m = detail::softmax::max_of(x);
178 T s = T(0);
179 for (std::size_t i = 0; i < x.size(); ++i) {
180 y[i] = math::exp(x[i] - m);
181 s += y[i];
182 }
183 const T inv = T(1) / s;
184 for (T& p : y)
185 p *= inv;
186}

212template <row P, row D, mutable_row Out>
213 requires std::same_as<row_value_t<P>, row_value_t<D>> && std::same_as<row_value_t<P>, row_value_t<Out>>
214constexpr void softmax_backward(const P& probs, const D& dprobs, Out&& dlogits)
215{
216 using T = row_value_t<P>;
217 const auto p = as_row(probs);
218 const auto d = as_row(dprobs);
219 const auto g = as_mutable_row(dlogits);
220 assert(p.size() == d.size() && p.size() == g.size());
221 T dot = T(0);
222 for (std::size_t i = 0; i < p.size(); ++i)
223 dot += p[i] * d[i];
224 for (std::size_t i = 0; i < p.size(); ++i)
225 g[i] = p[i] * (d[i] - dot);
226}

251template <row In, mutable_row Out>
252 requires std::same_as<row_value_t<In>, row_value_t<Out>>
253constexpr row_value_t<In> cross_entropy_with_gradient(const In& logits, std::size_t target, Out&& dlogits,
254 std::type_identity_t<row_value_t<In>> scale = 1)
255{
256 using T = row_value_t<In>;
257 const auto x = as_row(logits);
258 const auto g = as_mutable_row(dlogits);
259 assert(target < x.size() && x.size() == g.size());
260 const T lse = detail::softmax::logsumexp(x);
261 const T loss = lse - x[target];
262 for (std::size_t i = 0; i < x.size(); ++i)
263 g[i] = scale * (math::exp(x[i] - lse) - (i == target ? T(1) : T(0)));
264 return loss;
265}
