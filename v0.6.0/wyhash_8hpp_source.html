113 // The reads the hash is built from, over anything with b[i] convertible to a byte: little-endian,
114 // assembled byte by byte, so they are allowed in a constant expression (where reinterpret_cast is
115 // not) and give the same bits on every host.
116 template <typename B>
117 struct wy_indexed {
118 const B& b;
119
120 constexpr std::uint64_t byte(std::size_t i) const noexcept
121 {
122 return static_cast<std::uint64_t>(static_cast<unsigned char>(b[i]));
123 }
124 constexpr std::uint64_t read8(std::size_t o) const noexcept
125 {
126 return byte(o) | (byte(o + 1) << 8) | (byte(o + 2) << 16) | (byte(o + 3) << 24) | (byte(o + 4) << 32) |
127 (byte(o + 5) << 40) | (byte(o + 6) << 48) | (byte(o + 7) << 56);
128 }
129 constexpr std::uint64_t read4(std::size_t o) const noexcept
130 {
131 return byte(o) | (byte(o + 1) << 8) | (byte(o + 2) << 16) | (byte(o + 3) << 24);
132 }
133 constexpr std::uint64_t read3(std::size_t o, std::size_t k) const noexcept
134 {
135 return (byte(o) << 16) | (byte(o + (k >> 1)) << 8) | byte(o + k - 1);
136 }
137 };

139 // The same reads from a pointer at run time: one memcpy each, which is one load, swapped on a
140 // big-endian host so the value is the little-endian one the constant-expression path produces.
141 struct wy_pointer {
142 const unsigned char* p;
143
144 std::uint64_t read8(std::size_t o) const noexcept
145 {
146 std::uint64_t v = 0;
147 std::memcpy(&v, p + o, sizeof v);
148 return std::endian::native == std::endian::little ? v : byteswap(v);
149 }
150 std::uint64_t read4(std::size_t o) const noexcept
151 {
152 std::uint32_t v = 0;
153 std::memcpy(&v, p + o, sizeof v);
154 return std::endian::native == std::endian::little ? v : byteswap(v);
155 }
156 std::uint64_t read3(std::size_t o, std::size_t k) const noexcept
157 {
158 return (static_cast<std::uint64_t>(p[o]) << 16) | (static_cast<std::uint64_t>(p[o + (k >> 1)]) << 8) |
159 p[o + k - 1];
160 }
161 };

178 // wyhash final version 4.3 over a reader, taken by value: a pointer or a reference, in a register.
179 template <typename Reader>
180 constexpr std::uint64_t wy_hash(const Reader r, std::size_t len, std::uint64_t seed) noexcept
181 {
182 seed ^= wymix(seed ^ wyhash_secret[0], wyhash_secret[1]);
183 std::uint64_t a = 0, hi = 0;
184 std::size_t p = 0; // the read offset, in place of the reference's moving pointer
185
186 if (len <= 16) {
187 if (len >= 4) {
188 const std::size_t off = (len >> 3) << 2;
189 a = (r.read4(0) << 32) | r.read4(off);
190 hi = (r.read4(len - 4) << 32) | r.read4(len - 4 - off);
191 } else if (len > 0) {
192 a = r.read3(0, len);
193 }
194 } else {
195 std::size_t i = len;
196 if (i > 48) {
197 std::uint64_t see1 = seed, see2 = seed;
198 do {
199 seed = wymix(r.read8(p) ^ wyhash_secret[1], r.read8(p + 8) ^ seed);
200 see1 = wymix(r.read8(p + 16) ^ wyhash_secret[2], r.read8(p + 24) ^ see1);
201 see2 = wymix(r.read8(p + 32) ^ wyhash_secret[3], r.read8(p + 40) ^ see2);
202 p += 48;
203 i -= 48;
204 } while (i > 48);
205 seed ^= see1 ^ see2;
206 }
207 while (i > 16) {
208 seed = wymix(r.read8(p) ^ wyhash_secret[1], r.read8(p + 8) ^ seed);
209 i -= 16;
210 p += 16;
211 }
212 a = r.read8(p + i - 16);
213 hi = r.read8(p + i - 8);
214 }
215
216 a ^= wyhash_secret[1];
217 hi ^= seed;
218 const wide_product m = umul128(a, hi);
219 return wymix(m.low ^ wyhash_secret[0] ^ len, m.high ^ wyhash_secret[1]);
220 }
