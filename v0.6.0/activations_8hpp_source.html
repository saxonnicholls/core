63template <math::math_argument X>
64constexpr math::real_t<X> sigmoid(X x)
65{
66 using T = math::real_t<X>;
67 const T v = static_cast<T>(x);
68 if (v >= T(0))
69 return T(1) / (T(1) + math::exp(-v));
70 const T e = math::exp(v);
71 return e / (T(1) + e);
72}

98template <math::math_argument X>
99constexpr math::real_t<X> softplus(X x)
100{
101 using T = math::real_t<X>;
102 const T v = static_cast<T>(x);
103 if (math::is_nan(v))
104 return v;
105 const T positive = v > T(0) ? v : T(0);
106 return positive + math::log1p(math::exp(-math::abs(v)));
107}

124template <math::math_argument X>
125constexpr math::real_t<X> logit(X p)
126{
127 using T = math::real_t<X>;
128 const T v = static_cast<T>(p);
129 if (!(v >= T(0) && v <= T(1))) // NaN, or outside [0, 1]: no arithmetic on a NaN, which a constant expression refuses
130 return std::numeric_limits<T>::quiet_NaN();
131 return math::log(v) - math::log1p(-v);
132}

219{

230template <math::math_argument X>
231constexpr math::real_t<X> gelu_derivative(X x)
232{
233 using T = math::real_t<X>;
234 using L = math::detail::limits<T>;
235 const T v = static_cast<T>(x);
236 const T cdf = T(0.5) * math::erfc(-v * T(L::inv_sqrt2));
237 const T pdf = math::exp(T(-0.5) * v * v) / T(L::sqrt_2pi);
238 return cdf + v * pdf;
239}

243 // u = sqrt(2/pi) (x + 0.044715 x^3), GPT-2's approximation of GELU; returns 2u.
244 template <typename T>
245 constexpr T gelu_tanh_2u(T v)
246 {
247 using L = math::detail::limits<T>;
248 const T sqrt_2_over_pi = T(2) / T(L::sqrt_2pi);
249 return T(2) * sqrt_2_over_pi * (v + T(0.044715) * v * v * v);
250 }

269template <math::math_argument X>
270constexpr math::real_t<X> gelu_tanh_derivative(X x)
271{
272 using T = math::real_t<X>;
273 using L = math::detail::limits<T>;
274 const T v = static_cast<T>(x);
275 const T s = sigmoid(detail::activations::gelu_tanh_2u(v));
276 const T du = T(2) / T(L::sqrt_2pi) * (T(1) + T(3) * T(0.044715) * v * v);
277 return s + T(2) * v * s * (T(1) - s) * du;
278}
