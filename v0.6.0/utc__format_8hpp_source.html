73template <snicholls::utc_duration D>
74struct std::formatter<snicholls::utc<D>, char>
75{
79 constexpr auto parse(std::format_parse_context& ctx)
80 {
81 auto it = ctx.begin();
82 rfc3339_ = it == ctx.end() || *it == '}';
83 return rfc3339_ ? it : chrono_.parse(ctx);
84 }
85
90 template <typename FormatContext>
91 auto format(const snicholls::utc<D>& t, FormatContext& ctx) const
92 {
93 if (!rfc3339_) return chrono_.format(t.sys_time(), ctx);
94 const snicholls::rfc3339_text text = snicholls::to_text(t);
95 if (!text) throw std::format_error("snicholls::utc: year outside 0000..9999 has no RFC 3339 form");
96 return std::copy(text.buffer.data(), text.buffer.data() + text.size, ctx.out());
97 }
98
99private:
100 std::formatter<std::chrono::sys_time<D>, char> chrono_;
101 bool rfc3339_ = true;
102};
