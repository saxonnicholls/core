298 constexpr uint256& operator<<=(unsigned n) noexcept
299 {
300 if (n >= 256) {
301 limbs_ = {};
302 return *this;
303 }
304 const std::size_t limb_shift = n / 64;
305 const unsigned bit_shift = n % 64;
306 std::array<std::uint64_t, 4> out{};
307 for (std::size_t i = 0; i < 4; ++i) {
308 if (i < limb_shift)
309 continue;
310 const std::size_t from = i - limb_shift;
311 out[i] = limbs_[from] << bit_shift;
312 if (bit_shift != 0 && from > 0)
313 out[i] |= limbs_[from - 1] >> (64 - bit_shift);
314 }
315 limbs_ = out;
316 return *this;
317 }

322 constexpr uint256& operator>>=(unsigned n) noexcept
323 {
324 if (n >= 256) {
325 limbs_ = {};
326 return *this;
327 }
328 const std::size_t limb_shift = n / 64;
329 const unsigned bit_shift = n % 64;
330 std::array<std::uint64_t, 4> out{};
331 for (std::size_t i = 0; i < 4; ++i) {
332 const std::size_t from = i + limb_shift;
333 if (from >= 4)
334 continue;
335 out[i] = limbs_[from] >> bit_shift;
336 if (bit_shift != 0 && from + 1 < 4)
337 out[i] |= limbs_[from + 1] << (64 - bit_shift);
338 }
339 limbs_ = out;
340 return *this;
341 }

477 [[nodiscard]] static constexpr std::optional<uint256> from_decimal(std::string_view text) noexcept
478 {
479 if (text.empty())
480 return std::nullopt;
481 uint256 acc;
482 for (const char c : text) {
483 if (c < '0' || c > '9')
484 return std::nullopt;
485 if (acc.mul_small(10) != 0)
486 return std::nullopt;
487 if (acc.add_small(static_cast<std::uint64_t>(c - '0')) != 0)
488 return std::nullopt;
489 }
490 return acc;

496 [[nodiscard]] static constexpr std::optional<uint256> from_hex(std::string_view text) noexcept
497 {
498 if (text.size() >= 2 && text[0] == '0' && (text[1] == 'x' || text[1] == 'X'))
499 text.remove_prefix(2);
500 if (text.empty())
501 return std::nullopt;
502 while (text.size() > 1 && text[0] == '0')
503 text.remove_prefix(1);
504 if (text.size() > 64)
505 return std::nullopt;
506 uint256 acc;
507 for (const char c : text) {
508 std::uint64_t d = 0;
509 if (c >= '0' && c <= '9')
510 d = static_cast<std::uint64_t>(c - '0');
511 else if (c >= 'a' && c <= 'f')
512 d = static_cast<std::uint64_t>(c - 'a' + 10);
513 else if (c >= 'A' && c <= 'F')
514 d = static_cast<std::uint64_t>(c - 'A' + 10);
515 else
516 return std::nullopt;
517 acc <<= 4;
518 acc.limbs_[0] |= d;
519 }
520 return acc;
521 }

526 [[nodiscard]] std::string to_string() const
527 {
528 if (is_zero())
529 return "0";
530 std::string out;
531 uint256 v = *this;
532 while (!v.is_zero()) {
533 const std::uint32_t chunk = v.divmod_small(1'000'000'000u); // nine digits at a time
534 std::string digits = std::to_string(chunk);
535 if (!v.is_zero())
536 digits.insert(0, 9 - digits.size(), '0');
537 out.insert(0, digits);
538 }
539 return out;

550 [[nodiscard]] static constexpr std::optional<uint256> parse_units(std::string_view text, unsigned decimals) noexcept
551 {
552 if (decimals > 77)
553 return std::nullopt;
554 const std::size_t point = text.find('.');
555 const std::string_view whole = text.substr(0, point);
556 const std::string_view fraction = point == std::string_view::npos ? std::string_view{} : text.substr(point + 1);
557 if (whole.empty() && fraction.empty())
558 return std::nullopt;
559 uint256 acc;
560 const auto push = [&acc](char c) {
561 return c >= '0' && c <= '9' && acc.mul_small(10) == 0 && acc.add_small(static_cast<std::uint64_t>(c - '0')) == 0;
562 };
563 for (const char c : whole)
564 if (!push(c))
565 return std::nullopt;
566 for (std::size_t i = 0; i < fraction.size(); ++i) {
567 const char c = fraction[i];
568 if (i < decimals) {
569 if (!push(c))
570 return std::nullopt;
571 } else if (c != '0') {
572 return std::nullopt; // a digit the token cannot hold; never rounded
573 }
574 }
575 for (std::size_t i = fraction.size(); i < decimals; ++i)
576 if (acc.mul_small(10) != 0)
577 return std::nullopt;
578 return acc;
579 }

627 {
628 constexpr const char* hex = "0123456789abcdef";

630 for (std::size_t i = 0; i < 4; ++i)

637 // Add with carry out: returns 1 on overflow.
638 constexpr std::uint64_t add_with_carry(const uint256& other) noexcept
639 {
640 std::uint64_t carry = 0;
641 for (std::size_t i = 0; i < 4; ++i) {
642 const std::uint64_t sum = limbs_[i] + other.limbs_[i];
643 std::uint64_t c = sum < limbs_[i] ? 1u : 0u;
644 const std::uint64_t with_carry = sum + carry;
645 c += with_carry < sum ? 1u : 0u;
646 limbs_[i] = with_carry;
647 carry = c;
648 }
649 return carry;
650 }
651
652 // Subtract with borrow out: returns 1 when other > *this.
653 constexpr std::uint64_t sub_with_borrow(const uint256& other) noexcept
654 {
655 std::uint64_t borrow = 0;
656 for (std::size_t i = 0; i < 4; ++i) {
657 const std::uint64_t diff = limbs_[i] - other.limbs_[i];
658 std::uint64_t b = limbs_[i] < other.limbs_[i] ? 1u : 0u;
659 const std::uint64_t with_borrow = diff - borrow;
660 b += diff < borrow ? 1u : 0u;
661 limbs_[i] = with_borrow;
662 borrow = b;
663 }
664 return borrow;
665 }

667 // Multiply by a 64-bit value in place; returns the limb that fell off the top (0 when it fits).
668 constexpr std::uint64_t mul_small(std::uint64_t m) noexcept
669 {
670 std::uint64_t carry = 0;
671 for (std::size_t i = 0; i < 4; ++i) {
672 const detail::wide_product p = detail::umul128(limbs_[i], m);
673 const std::uint64_t low = p.low + carry;
674 limbs_[i] = low;
675 carry = p.high + (low < carry ? 1u : 0u);
676 }
677 return carry;
678 }

683 // Divide in place by a 32-bit value; returns the remainder.
684 constexpr std::uint32_t divmod_small(std::uint32_t d) noexcept
685 {
686 std::uint64_t rem = 0;
687 for (int i = 3; i >= 0; --i) {
688 const auto idx = static_cast<std::size_t>(i);
689 const std::uint64_t high = (rem << 32) | (limbs_[idx] >> 32);
690 const std::uint64_t qh = high / d;
691 rem = high % d;
692 const std::uint64_t low = (rem << 32) | (limbs_[idx] & 0xffffffffu);
693 const std::uint64_t ql = low / d;
694 rem = low % d;
695 limbs_[idx] = (qh << 32) | ql;
696 }
697 return static_cast<std::uint32_t>(rem);
698 }

700 // The full 512-bit product, as eight limbs.
701 static constexpr std::array<std::uint64_t, 8> mul_full(const uint256& a, const uint256& b) noexcept
702 {
703 std::array<std::uint64_t, 8> r{};
704 for (std::size_t i = 0; i < 4; ++i) {
705 std::uint64_t carry = 0;
706 for (std::size_t j = 0; j < 4; ++j) {
707 const detail::wide_product p = detail::umul128(a.limbs_[i], b.limbs_[j]);
708 std::uint64_t low = r[i + j] + p.low;
709 std::uint64_t c = low < p.low ? 1u : 0u;
710 const std::uint64_t with_carry = low + carry;
711 c += with_carry < low ? 1u : 0u;
712 r[i + j] = with_carry;
713 carry = p.high + c; // p.high <= 2^64 - 2, so no overflow
714 }
715 r[i + 4] = carry;
716 }
717 return r;
718 }

720 // Knuth's algorithm D (TAOCP 4.3.1), base 2^32, in the form of Hacker's Delight's divmnu.
721 static constexpr std::pair<uint256, uint256> divmod_knuth(const uint256& dividend, const uint256& divisor) noexcept
722 {
723 std::array<std::uint32_t, 8> u{}, v{};
724 for (std::size_t i = 0; i < 4; ++i) {
725 u[2 * i] = static_cast<std::uint32_t>(dividend.limbs_[i]);
726 u[2 * i + 1] = static_cast<std::uint32_t>(dividend.limbs_[i] >> 32);
727 v[2 * i] = static_cast<std::uint32_t>(divisor.limbs_[i]);
728 v[2 * i + 1] = static_cast<std::uint32_t>(divisor.limbs_[i] >> 32);
729 }
730 int n = 8;
731 while (n > 1 && v[static_cast<std::size_t>(n - 1)] == 0)
732 --n;
733 int u_digits = 8;
734 while (u_digits > 0 && u[static_cast<std::size_t>(u_digits - 1)] == 0)
735 --u_digits;
736 if (u_digits < n)
737 return {uint256{}, dividend};
738
739 std::array<std::uint32_t, 8> q{}, r{};
740 if (n == 1) { // short division
741 std::uint64_t k = 0;
742 for (int j = u_digits - 1; j >= 0; --j) {
743 const auto idx = static_cast<std::size_t>(j);
744 const std::uint64_t cur = (k << 32) | u[idx];
745 q[idx] = static_cast<std::uint32_t>(cur / v[0]);
746 k = cur % v[0];
747 }
748 r[0] = static_cast<std::uint32_t>(k);
749 return {from_digits(q), from_digits(r)};
750 }
751
752 const int m = u_digits - n;
753 const int s = std::countl_zero(v[static_cast<std::size_t>(n - 1)]); // normalise so vn[n-1] has its top bit set
754 std::array<std::uint32_t, 8> vn{};
755 std::array<std::uint32_t, 9> un{};
756 for (int i = n - 1; i > 0; --i)
757 vn[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(
758 (static_cast<std::uint64_t>(v[static_cast<std::size_t>(i)]) << s) |
759 (static_cast<std::uint64_t>(v[static_cast<std::size_t>(i - 1)]) >> (32 - s)));
760 vn[0] = v[0] << s;
761 un[static_cast<std::size_t>(u_digits)] = static_cast<std::uint32_t>(
762 static_cast<std::uint64_t>(u[static_cast<std::size_t>(u_digits - 1)]) >> (32 - s));
763 for (int i = u_digits - 1; i > 0; --i)
764 un[static_cast<std::size_t>(i)] = static_cast<std::uint32_t>(
765 (static_cast<std::uint64_t>(u[static_cast<std::size_t>(i)]) << s) |
766 (static_cast<std::uint64_t>(u[static_cast<std::size_t>(i - 1)]) >> (32 - s)));
767 un[0] = u[0] << s;
768
769 constexpr std::uint64_t base = std::uint64_t{1} << 32;
770 for (int j = m; j >= 0; --j) {
771 const auto jn = static_cast<std::size_t>(j + n);
772 const std::uint64_t numerator = (static_cast<std::uint64_t>(un[jn]) << 32) | un[jn - 1];
773 std::uint64_t qhat = numerator / vn[static_cast<std::size_t>(n - 1)];
774 std::uint64_t rhat = numerator % vn[static_cast<std::size_t>(n - 1)];
775 while (qhat >= base ||
776 qhat * vn[static_cast<std::size_t>(n - 2)] > ((rhat << 32) | un[jn - 2])) {
777 --qhat;
778 rhat += vn[static_cast<std::size_t>(n - 1)];
779 if (rhat >= base)
780 break;
781 }
782 // multiply and subtract
783 std::int64_t k = 0;
784 std::int64_t t = 0;
785 for (int i = 0; i < n; ++i) {
786 const auto ij = static_cast<std::size_t>(i + j);
787 const std::uint64_t p = qhat * vn[static_cast<std::size_t>(i)];
788 t = static_cast<std::int64_t>(un[ij]) - k - static_cast<std::int64_t>(p & 0xffffffffu);
789 un[ij] = static_cast<std::uint32_t>(t);
790 k = static_cast<std::int64_t>(p >> 32) - (t >> 32);
791 }
792 t = static_cast<std::int64_t>(un[jn]) - k;
793 un[jn] = static_cast<std::uint32_t>(t);
794 q[static_cast<std::size_t>(j)] = static_cast<std::uint32_t>(qhat);
795 if (t < 0) { // qhat was one too large: add back
796 --q[static_cast<std::size_t>(j)];
797 std::uint64_t carry = 0;
798 for (int i = 0; i < n; ++i) {
799 const auto ij = static_cast<std::size_t>(i + j);
800 const std::uint64_t sum = static_cast<std::uint64_t>(un[ij]) + vn[static_cast<std::size_t>(i)] + carry;
801 un[ij] = static_cast<std::uint32_t>(sum);
802 carry = sum >> 32;
803 }
804 un[jn] = static_cast<std::uint32_t>(un[jn] + carry);
805 }
806 }
807 // the remainder is un, un-normalised
808 for (int i = 0; i < n; ++i) {
809 const auto idx = static_cast<std::size_t>(i);
810 r[idx] = static_cast<std::uint32_t>((static_cast<std::uint64_t>(un[idx]) >> s) |
811 (static_cast<std::uint64_t>(un[idx + 1]) << (32 - s)));
812 }
813 return {from_digits(q), from_digits(r)};
814 }

817 {
