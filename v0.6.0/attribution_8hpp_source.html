97 template <std::size_t K, std::floating_point T>
98 [[nodiscard]] constexpr taylor_explain<K, T> explain(const taylor_model<K, T>& model, const std::array<T, K>& moves, T actual) noexcept
99 {
100 taylor_explain<K, T> out{};
101 out.carry = model.carry;
102 out.explained = model.carry;
103 out.actual = actual;
104 for (std::size_t i = 0; i < K; ++i) {
105 out.first_order[i] = model.gradient[i] * moves[i];
106 out.by_factor[i] = out.first_order[i];
107 out.explained += out.first_order[i];
108 for (std::size_t j = 0; j < K; ++j) {
109 out.second_order[i][j] = T(0.5) * model.hessian[i][j] * moves[i] * moves[j];
110 out.by_factor[i] += out.second_order[i][j];
111 out.explained += out.second_order[i][j];
112 }
113 }
114 out.unexplained = actual - out.explained;
115 return out;
116 }

192 template <std::floating_point T>
193 [[nodiscard]] constexpr taylor_model<4, T> as_taylor_model(const greeks<T>& g) noexcept
194 {
195 taylor_model<4, T> model{};
196 model.gradient = {g.delta, g.vega, g.theta, g.rho};
197 model.hessian[0][0] = g.gamma;
198 model.hessian[1][1] = g.volga;
199 model.hessian[0][1] = g.vanna;
200 model.hessian[1][0] = g.vanna;
201 return model;

211 template <std::floating_point T>
212 [[nodiscard]] constexpr greeks_explain<T> explain(const greeks<T>& g, const market_move<T>& move, T actual) noexcept
213 {
214 const taylor_explain<4, T> terms = explain(as_taylor_model(g), std::array<T, 4>{move.spot, move.vol, move.time, move.rate}, actual);
215 greeks_explain<T> out{};
216 out.delta = terms.first_order[0];
217 out.vega = terms.first_order[1];
218 out.theta = terms.first_order[2];
219 out.rho = terms.first_order[3];
220 out.gamma = terms.second_order[0][0];
221 out.volga = terms.second_order[1][1];
222 out.vanna = terms.second_order[0][1] + terms.second_order[1][0];
223 out.explained = terms.explained;
224 out.actual = actual;
225 out.unexplained = terms.unexplained;
226 out.spot = terms.by_factor[0];
227 out.volatility = terms.by_factor[1];
228 out.time = terms.by_factor[2];
229 out.rates = terms.by_factor[3];
230 return out;

252 template <std::size_t K, std::floating_point T>
253 [[nodiscard]] constexpr risk_contribution<K, T> risk_contributions(const std::array<std::array<T, K>, K>& covariance,
254 const std::array<T, K>& weights) noexcept
255 {
256 risk_contribution<K, T> out{};
257 for (std::size_t i = 0; i < K; ++i) {
258 T marginal{};
259 for (std::size_t j = 0; j < K; ++j)
260 marginal += covariance[i][j] * weights[j];
261 out.to_variance[i] = weights[i] * marginal;
262 out.variance += out.to_variance[i];
263 }
264 out.volatility = out.variance > T(0) ? std::sqrt(out.variance) : T(0);
265 for (std::size_t i = 0; i < K; ++i)
266 out.to_volatility[i] = out.volatility > T(0) ? out.to_variance[i] / out.volatility : T(0);
267 return out;
268 }
