81template <utc_duration D>
82[[nodiscard]] solar_position sun_position(const utc<D>& t) noexcept
83{
84 constexpr double rad = std::numbers::pi / 180.0, deg = 180.0 / std::numbers::pi;
85 const double unix_days = std::chrono::duration<double, std::ratio<86400>>(t.time_since_epoch()).count();
86 const double jd = unix_days + 2440587.5;
87 const double jc = (jd - 2451545.0) / 36525.0;
88 const double l0 = std::fmod(280.46646 + jc * (36000.76983 + jc * 0.0003032), 360.0);
89 const double m = 357.52911 + jc * (35999.05029 - 0.0001537 * jc);
90 const double e = 0.016708634 - jc * (0.000042037 + 0.0000001267 * jc);
91 const double c = std::sin(m * rad) * (1.914602 - jc * (0.004817 + 0.000014 * jc)) + std::sin(2 * m * rad) * (0.019993 - 0.000101 * jc) +
92 std::sin(3 * m * rad) * 0.000289;
93 const double omega = 125.04 - 1934.136 * jc;
94 const double app_long = l0 + c - 0.00569 - 0.00478 * std::sin(omega * rad);
95 const double mean_obliq = 23.0 + (26.0 + (21.448 - jc * (46.815 + jc * (0.00059 - jc * 0.001813))) / 60.0) / 60.0;
96 const double obliq = mean_obliq + 0.00256 * std::cos(omega * rad);
97 const double decl = std::asin(std::sin(obliq * rad) * std::sin(app_long * rad)) * deg;
98 const double y = std::tan(obliq * rad / 2) * std::tan(obliq * rad / 2);
99 const double eot = 4 * deg *
100 (y * std::sin(2 * l0 * rad) - 2 * e * std::sin(m * rad) + 4 * e * y * std::sin(m * rad) * std::cos(2 * l0 * rad) -
101 0.5 * y * y * std::sin(4 * l0 * rad) - 1.25 * e * e * std::sin(2 * m * rad));
102 const double minutes_utc = (unix_days - std::floor(unix_days)) * 1440.0;
103 double lon = -(minutes_utc + eot - 720.0) / 4.0; // where apparent solar time is 12:00
104 lon = std::fmod(lon + 540.0, 360.0) - 180.0;
105 return {decl, eot, decl, lon};
106}

112[[nodiscard]] inline double solar_elevation_deg(const solar_position& sun, double latitude_deg, double longitude_deg) noexcept
113{
114 constexpr double rad = std::numbers::pi / 180.0;
115 const double h = (longitude_deg - sun.subsolar_longitude_deg) * rad;
116 const double s = std::sin(latitude_deg * rad) * std::sin(sun.declination_deg * rad) +
117 std::cos(latitude_deg * rad) * std::cos(sun.declination_deg * rad) * std::cos(h);
118 return std::asin(std::clamp(s, -1.0, 1.0)) / rad;
119}

134[[nodiscard]] inline std::vector<std::pair<double, double>> terminator(const solar_position& sun, double step_deg = 1.0)
135{
136 constexpr double rad = std::numbers::pi / 180.0;
137 // At an equinox tan(decl) -> 0: nudge it so the terminator is the (correct) near-vertical line.
138 const double tan_decl = std::abs(sun.declination_deg) < 1e-6 ? 1e-6 : std::tan(sun.declination_deg * rad);
139 std::vector<std::pair<double, double>> points;
140 for (double lon = -180.0; lon <= 180.0 + 1e-9; lon += step_deg)
141 points.emplace_back(lon, std::atan(-std::cos((lon - sun.subsolar_longitude_deg) * rad) / tan_decl) / rad);
142 return points;
143}
