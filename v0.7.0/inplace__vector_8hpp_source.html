235 constexpr iterator insert(const_iterator pos, const T& v)
236 {
237 if (size_ == N)
238 throw std::length_error("snicholls::containers::inplace_vector: full");
239 const auto i = static_cast<size_type>(pos - begin());
240 std::move_backward(data_.data() + i, data_.data() + size_, data_.data() + size_ + 1);
241 data_[i] = v;
242 ++size_;
243 return data_.data() + i;
244 }

249 constexpr iterator erase(const_iterator pos)
250 {
251 const auto i = static_cast<size_type>(pos - begin());
252 std::move(data_.data() + i + 1, data_.data() + size_, data_.data() + i);
253 data_[--size_] = T{};
254 return data_.data() + i;
255 }

289 template <typename T, std::size_t N, typename Pred>
290 constexpr std::size_t erase_if(inplace_vector<T, N>& v, Pred pred)
291 {
292 const auto kept = std::remove_if(v.begin(), v.end(), pred);
293 const auto removed = static_cast<std::size_t>(v.end() - kept);
294 v.resize(v.size() - removed);
295 return removed;
