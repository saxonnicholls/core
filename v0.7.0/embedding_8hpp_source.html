64template <row Table>
65constexpr std::span<const row_value_t<Table>> embedding_row(const Table& table, std::size_t id, std::size_t channels)
66{
67 const auto t = as_row(table);
68 assert((id + 1) * channels <= t.size());
69 return t.subspan(id * channels, channels);
70}

76template <row Table, mutable_row Out>
77 requires std::same_as<row_value_t<Table>, row_value_t<Out>>
78constexpr void embed(const Table& table, std::size_t id, Out&& out)
79{
80 const auto y = as_mutable_row(out);
81 const auto r = embedding_row(table, id, y.size());
82 for (std::size_t c = 0; c < y.size(); ++c)

92template <row Tokens, row Positions, mutable_row Out>
93 requires std::same_as<row_value_t<Tokens>, row_value_t<Out>> && std::same_as<row_value_t<Positions>, row_value_t<Out>>
94constexpr void embed_with_position(const Tokens& tokens, std::size_t id, const Positions& positions, std::size_t pos, Out&& out)
95{
96 const auto y = as_mutable_row(out);
97 const auto t = embedding_row(tokens, id, y.size());
98 const auto p = embedding_row(positions, pos, y.size());
