165[[nodiscard]] inline bool is_secret_key(std::string_view key)
166{
167 std::string upper(key);
168 std::transform(upper.begin(), upper.end(), upper.begin(), detail::ascii_upper);
169 constexpr std::string_view markers[] = {"PASSWORD", "PASSWD", "PASSPHRASE", "SECRET", "TOKEN", "CREDENTIAL",
170 "PRIVATE", "AUTH", "API_KEY", "APIKEY", "ACCESS_KEY", "COOKIE",
171 "SESSION", "SIGNATURE"};
172 for (const std::string_view marker : markers)
173 if (upper.find(marker) != std::string::npos)
174 return true;
175 return upper == "KEY" || upper.ends_with("_KEY") || upper.ends_with("_PWD");
176}

184[[nodiscard]] inline std::string mask_url_credentials(std::string_view value)
185{
186 const std::size_t scheme = value.find("://");
187 if (scheme == std::string_view::npos)
188 return std::string(value);
189 const std::size_t authority = scheme + 3;
190 const std::size_t end = value.find_first_of("/?#", authority);
191 const std::string_view host_part = value.substr(authority, end == std::string_view::npos ? value.npos : end - authority);
192 const std::size_t at = host_part.rfind('@');
193 if (at == std::string_view::npos)
194 return std::string(value);
195 const std::size_t colon = host_part.find(':');
196 if (colon == std::string_view::npos || colon > at)
197 return std::string(value);
198 std::string out(value.substr(0, authority + colon + 1));
199 out += masked_text;
200 out += value.substr(authority + at);
201 return out;
202}
