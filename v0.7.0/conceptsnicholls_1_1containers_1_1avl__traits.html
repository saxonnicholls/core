typename Traits::key_type;
{ Traits::hook(n) } -> std::same_as<avl_hook<T>&>;
{ Traits::key(c) } -> std::convertible_to<typename Traits::key_type>;
{ Traits::less(k, k) } -> std::convertible_to<bool>;
}
