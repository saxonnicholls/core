100 template <typename T>
101 constexpr T dot(std::span<const T> a, std::span<const T> b)
102 {
103 T s = T(0);
104 for (std::size_t i = 0; i < a.size(); ++i)
105 s += a[i] * b[i];
106 return s;
107 }

109 // One step of the recurrence: fold value v with score s into (max, sum, acc)
110 template <typename T>
111 constexpr void fold(T& max, T& sum, std::span<T> acc, T s, std::span<const T> v)
112 {
113 if (s == -std::numeric_limits<T>::infinity())
114 return;
115 if (s > max) {
116 const T scale = math::exp(max - s);
117 for (std::size_t i = 0; i < acc.size(); ++i)
118 acc[i] = acc[i] * scale + v[i];
119 sum = sum * scale + T(1);
120 max = s;
121 } else {
122 const T w = math::exp(s - max);
123 for (std::size_t i = 0; i < acc.size(); ++i)
124 acc[i] += w * v[i];
125 sum += w;
126 }
127 }

134template <math::math_number T, std::size_t D>
135struct attention_accumulator
136{
137 online_softmax<T> weights{};
138 std::array<T, D> acc{};
139
143 template <row V>
144 requires std::same_as<row_value_t<V>, T>
145 constexpr void add(T score, const V& value)
146 {
147 const auto v = as_row(value);
148 assert(v.size() == D);
149 detail::attention::fold(weights.max, weights.sum, std::span<T>(acc), score, v);
150 }
151
154 constexpr void merge(const attention_accumulator& other)
155 {
156 if (other.weights.sum == T(0))
157 return;
158 if (weights.sum == T(0)) {
159 *this = other;
160 return;
161 }
162 const T m = std::max(weights.max, other.weights.max);
163 const T a = math::exp(weights.max - m), b = math::exp(other.weights.max - m);
164 for (std::size_t i = 0; i < D; ++i)
165 acc[i] = acc[i] * a + other.acc[i] * b;
166 weights.sum = weights.sum * a + other.weights.sum * b;
167 weights.max = m;
168 }
169
172 template <mutable_row Out>
173 requires std::same_as<row_value_t<Out>, T>
174 constexpr void result(Out&& out) const
175 {
176 const auto y = as_mutable_row(out);
177 assert(y.size() == D);
178 for (std::size_t i = 0; i < D; ++i)
179 y[i] = weights.sum == T(0) ? T(0) : acc[i] / weights.sum;
180 }
181
184 [[nodiscard]] constexpr T log_sum_exp() const { return weights.log_sum_exp(); }
185};

194template <row Q, row K, row V, mutable_row Out>
195 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
196 && std::same_as<row_value_t<Q>, row_value_t<Out>>
197constexpr row_value_t<Q> attend(const Q& q, const K& keys, const V& values, Out&& out, std::type_identity_t<row_value_t<Q>> scale)
198{
199 using T = row_value_t<Q>;
200 const auto qs = as_row(q);
201 const auto ks = as_row(keys);
202 const auto vs = as_row(values);
203 const auto y = as_mutable_row(out);
204 const std::size_t d = qs.size();
205 assert(y.size() == d && ks.size() == vs.size() && (d == 0 || ks.size() % d == 0));
206 std::fill(y.begin(), y.end(), T(0));
207 T max = -std::numeric_limits<T>::infinity(), sum = T(0);
208 for (std::size_t j = 0; d != 0 && j < ks.size() / d; ++j)
209 detail::attention::fold(max, sum, y, scale * detail::attention::dot(qs, ks.subspan(j * d, d)), vs.subspan(j * d, d));
210 for (T& v : y)
211 v = sum == T(0) ? T(0) : v / sum;
212 return sum == T(0) ? -std::numeric_limits<T>::infinity() : max + math::log(sum);
213}

229template <row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK, mutable_row DV>
230 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
231 && std::same_as<row_value_t<Q>, row_value_t<O>> && std::same_as<row_value_t<Q>, row_value_t<DO>>
232 && std::same_as<row_value_t<Q>, row_value_t<DQ>> && std::same_as<row_value_t<Q>, row_value_t<DK>>
233 && std::same_as<row_value_t<Q>, row_value_t<DV>>
234constexpr void attend_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
235 std::type_identity_t<row_value_t<Q>> lse, std::type_identity_t<row_value_t<Q>> scale, DQ&& dq,
236 DK&& dkeys, DV&& dvalues)
237{
238 using T = row_value_t<Q>;
239 const auto qs = as_row(q);
240 const auto ks = as_row(keys);
241 const auto vs = as_row(values);
242 const auto o = as_row(out);
243 const auto g = as_row(dout);
244 const auto gq = as_mutable_row(dq);
245 const auto gk = as_mutable_row(dkeys);
246 const auto gv = as_mutable_row(dvalues);
247 const std::size_t d = qs.size();
248 assert(o.size() == d && g.size() == d && gq.size() == d && ks.size() == vs.size() && gk.size() == ks.size()
249 && gv.size() == vs.size() && (d == 0 || ks.size() % d == 0));
250 std::fill(gq.begin(), gq.end(), T(0));
251 if (d == 0 || lse == -std::numeric_limits<T>::infinity())
252 return;
253 const T big_d = detail::attention::dot(g, o);
254 for (std::size_t j = 0; j < ks.size() / d; ++j) {
255 const auto k = ks.subspan(j * d, d);
256 const auto v = vs.subspan(j * d, d);
257 const T p = math::exp(scale * detail::attention::dot(qs, k) - lse);
258 const T ds = p * (detail::attention::dot(g, v) - big_d);
259 for (std::size_t i = 0; i < d; ++i) {
260 gv[j * d + i] += p * g[i];
261 gq[i] += scale * ds * k[i];
262 gk[j * d + i] += scale * ds * qs[i];
263 }
264 }
265}
