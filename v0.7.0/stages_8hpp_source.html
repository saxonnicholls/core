83template <typename Pred, typename Next>
84struct filter_stage

137template <typename T, typename Next>
138struct batch_stage
139{
141 std::size_t size;
143 Next next;
145 std::vector<T> items{};
146
149 void operator()(const T& v)
150 {
151 if (items.capacity() < size) items.reserve(size);
152 items.push_back(v);
153 if (items.size() >= size) flush();
154 }
155
156 void flush()
157 {
158 if (items.empty()) return;
159 std::invoke(next, std::as_const(items));
160 items.clear();
161 }
162};
