97template <mlp_kind K, math::math_number T>
98constexpr T mlp_activation(T x)
99{
100 switch (K) {
101 case mlp_kind::gelu: return gelu(x);
102 case mlp_kind::gelu_tanh: return gelu_tanh(x);
103 case mlp_kind::relu_squared: return relu_squared(x);
104 case mlp_kind::swiglu: return silu(x);
105 }
106 return x;
107}

139template <mlp_kind K, math::math_number T>
140constexpr T mlp_activation_derivative(T x)
141{
142 switch (K) {
143 case mlp_kind::gelu: return gelu_derivative(x);
144 case mlp_kind::gelu_tanh: return gelu_tanh_derivative(x);
145 case mlp_kind::relu_squared: return relu_squared_derivative(x);
146 case mlp_kind::swiglu: return silu_derivative(x);
147 }
148 return T(1);

155 // The forward pass, saving the pre-activations into non-empty spans
156 template <transformer_config C, typename W, typename T, typename X, typename Y>
157 constexpr void forward(const mlp_weights<W, T>& w, const X& x, Y&& y, const mlp_activations<T>& save)
158 {
159 static_assert(valid(C), "snicholls::ml::mlp_forward: the transformer_config is not valid");
160 assert(w.up.rows == C.d_ff && w.up.cols == C.d_model && w.down.rows == C.d_model && w.down.cols == C.d_ff);
161 assert((save.up.empty() || save.up.size() == C.d_ff) && (save.gate.empty() || save.gate.size() == C.d_ff));
162 std::array<T, C.d_ff> h{};
163 const auto in = prepare_input<C.d_model>(w.up, x);
164 linear(w.up, in, h, w.up_bias);
165 if (!save.up.empty())
166 std::copy(h.begin(), h.end(), save.up.begin());
167 if constexpr (C.mlp == mlp_kind::swiglu) {
168 assert(w.gate.rows == C.d_ff && w.gate.cols == C.d_model);
169 std::array<T, C.d_ff> g{};
170 linear(w.gate, in, g, w.gate_bias);
171 if (!save.gate.empty())
172 std::copy(g.begin(), g.end(), save.gate.begin());
173 for (std::size_t i = 0; i < C.d_ff; ++i)
174 h[i] = silu(g[i]) * h[i];
175 } else {
176 for (T& v : h)
177 v = mlp_activation<C.mlp>(v);
178 }
179 linear(w.down, prepare_input<C.d_ff>(w.down, h), y, w.down_bias);
180 }
181} // namespace detail::mlp

221template <transformer_config C, typename S, math::math_number T, row X, row DY, mutable_row DX>
222 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<DY>, T> && std::same_as<row_value_t<DX>, T>
223constexpr void mlp_backward(const mlp_weights<dense_weights<S>, T>& w, const X& x, const mlp_activations<T>& saved,
224 const DY& dy, DX&& dx, const mlp_gradients<T>& grads)
225{
226 static_assert(valid(C), "snicholls::ml::mlp_backward: the transformer_config is not valid");
227 constexpr bool swiglu = C.mlp == mlp_kind::swiglu;
228 assert(saved.up.size() == C.d_ff && (!swiglu || saved.gate.size() == C.d_ff));
229 std::array<T, C.d_ff> h{}, dh{};
230 for (std::size_t i = 0; i < C.d_ff; ++i)
231 if constexpr (swiglu)
232 h[i] = silu(saved.gate[i]) * saved.up[i];
233 else
234 h[i] = mlp_activation<C.mlp>(saved.up[i]);
235 linear_backward(w.down, h, dy, dh, grads.down, grads.down_bias);
236 if constexpr (swiglu) {
237 std::array<T, C.d_ff> dgate{};
238 for (std::size_t i = 0; i < C.d_ff; ++i) {
239 dgate[i] = dh[i] * saved.up[i] * silu_derivative(saved.gate[i]);
240 dh[i] = dh[i] * silu(saved.gate[i]);
241 }
242 linear_backward(w.gate, x, dgate, dx, grads.gate, grads.gate_bias);
243 } else {
244 for (std::size_t i = 0; i < C.d_ff; ++i)
245 dh[i] = dh[i] * mlp_activation_derivative<C.mlp>(saved.up[i]);
246 }
247 linear_backward(w.up, x, dh, dx, grads.up, grads.up_bias);
248}

254template <transformer_config C, typename B, math::math_number T>
255constexpr void mlp_activate_rows(const B&, const mlp_activations<T>& saved)
256{
257 const std::size_t m = saved.up.size();
258 if constexpr (C.mlp == mlp_kind::swiglu) {
259 for (std::size_t i = 0; i < m; ++i)
260 saved.hidden[i] = silu(saved.gate[i]) * saved.up[i];
261 } else {
262 for (std::size_t i = 0; i < m; ++i)
263 saved.hidden[i] = mlp_activation<C.mlp>(saved.up[i]);
264 }
265}

273template <transformer_config C, typename B, math::math_number T>
274constexpr void mlp_activate_rows_backward(const B&, const mlp_activations<T>& saved, std::span<T> dh, std::span<T> dgate)
275{
276 const std::size_t m = dh.size();
277 if constexpr (C.mlp == mlp_kind::swiglu) {
278 for (std::size_t i = 0; i < m; ++i) {
279 dgate[i] = dh[i] * saved.up[i] * silu_derivative(saved.gate[i]);
280 dh[i] = dh[i] * silu(saved.gate[i]);
281 }
282 } else {
283 for (std::size_t i = 0; i < m; ++i)
284 dh[i] = dh[i] * mlp_activation_derivative<C.mlp>(saved.up[i]);
285 (void)dgate;
286 }

301template <transformer_config C, typename B, math::math_number T>
302 requires linalg::blas_backend<B, T>
303constexpr void mlp_forward_sequence(const B& backend, const mlp_weights<dense_weights<T>, T>& w, std::type_identity_t<std::span<const T>> xs,
304 std::type_identity_t<std::span<T>> ys, const mlp_activations<T>& save)
305{
306 static_assert(valid(C), "snicholls::ml::mlp_forward_sequence: the transformer_config is not valid");
307 constexpr bool swiglu = C.mlp == mlp_kind::swiglu;
308 const std::size_t n = xs.size() / C.d_model;
309 [[maybe_unused]] const std::size_t m = n * C.d_ff;
310 assert(xs.size() == n * C.d_model && ys.size() == xs.size() && save.up.size() == m && save.hidden.size() == m
311 && (!swiglu || save.gate.size() == m));
312 linear_sequence(backend, w.up, xs, save.up, w.up_bias);
313 if constexpr (swiglu)
314 linear_sequence(backend, w.gate, xs, save.gate, w.gate_bias);
315 mlp_activate_rows<C>(backend, save);
316 linear_sequence(backend, w.down, std::span<const T>(save.hidden), ys, w.down_bias);
317}

328template <transformer_config C, typename B, math::math_number T>
329 requires linalg::blas_backend<B, T>
330constexpr void mlp_backward_sequence(const B& backend, const mlp_weights<dense_weights<T>, T>& w,
331 std::type_identity_t<std::span<const T>> xs, const mlp_activations<T>& saved,
332 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
333 const mlp_gradients<T>& grads)
334{
335 static_assert(valid(C), "snicholls::ml::mlp_backward_sequence: the transformer_config is not valid");
336 constexpr bool swiglu = C.mlp == mlp_kind::swiglu;
337 const std::size_t n = xs.size() / C.d_model, m = n * C.d_ff;
338 assert(dys.size() == xs.size() && saved.up.size() == m && saved.hidden.size() == m && (!swiglu || saved.gate.size() == m)
339 && grads.scratch.size() >= (swiglu ? 2 : 1) * m);
340 const auto dh = grads.scratch.first(m);
341 const auto dgate = swiglu ? grads.scratch.subspan(m, m) : std::span<T>{};
342 fill_rows(backend, dh, T(0));
343 linear_sequence_backward(backend, w.down, std::span<const T>(saved.hidden), dys, dh, grads.down, grads.down_bias);
344 mlp_activate_rows_backward<C>(backend, saved, dh, dgate);
345 if constexpr (swiglu)
346 linear_sequence_backward(backend, w.gate, xs, std::span<const T>(dgate), dxs, grads.gate, grads.gate_bias);
347 linear_sequence_backward(backend, w.up, xs, std::span<const T>(dh), dxs, grads.up, grads.up_bias);
348}

354template <math::math_number T, std::size_t K>
355struct moe_route
356{
357 std::array<std::size_t, K> expert{};
358 std::array<T, K> weight{};
359 std::size_t count = 0;
360};
361
366template <std::size_t K, row R>
367 requires(K > 0)
368constexpr moe_route<row_value_t<R>, K> moe_gate(const R& router_logits)
369{
370 using T = row_value_t<R>;
371 const auto best = top_k<K>(router_logits);
372 assert(best.count > 0);
373 moe_route<T, K> out{};
374 out.count = best.count;
375 T sum = T(0);
376 for (std::size_t i = 0; i < best.count; ++i) {
377 out.expert[i] = best.index[i];
378 out.weight[i] = math::exp(best.value[i] - best.value[0]);
379 sum += out.weight[i];
380 }
381 for (std::size_t i = 0; i < best.count; ++i)
382 out.weight[i] /= sum;
383 return out;
384}

393template <math::math_number T, std::size_t K, row DW, mutable_row Out>
394 requires std::same_as<row_value_t<DW>, T> && std::same_as<row_value_t<Out>, T>
395constexpr void moe_gate_backward(const moe_route<T, K>& route, const DW& dweight, Out&& dlogits)
396{
397 const auto dw = as_row(dweight);
398 const auto g = as_mutable_row(dlogits);
399 assert(dw.size() >= route.count && route.count <= g.size());
400 T dot = T(0);
401 for (std::size_t i = 0; i < route.count; ++i)
402 dot += route.weight[i] * dw[i];
403 std::fill(g.begin(), g.end(), T(0));
404 for (std::size_t i = 0; i < route.count; ++i)
405 g[route.expert[i]] = route.weight[i] * (dw[i] - dot);
406}
