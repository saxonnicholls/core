133template <transformer_config C, math::math_number T, std::size_t N, row Q, mutable_row Out>
134 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<Out>, T>
135constexpr T attention_head(const Q& q, const kv_runs<T, N>& runs, Out&& out)
136{
137 const auto qs = as_row(q);
138 assert(qs.size() == C.head_dim);
139 const T scale = T(1) / math::sqrt(T(C.head_dim));
140 attention_accumulator<T, C.head_dim> acc{};
141 for (std::size_t r = 0; r < N; ++r)
142 for (std::size_t j = 0; j < runs.keys[r].size() / C.head_dim; ++j) {
143 T s = scale * detail::attention::dot(qs, runs.keys[r].subspan(j * C.head_dim, C.head_dim));
144 if constexpr (C.attention_softcap > 0)
145 s = softcap(s, attention_softcap_v<C, T>);
146 acc.add(s, runs.values[r].subspan(j * C.head_dim, C.head_dim));
147 }
148 acc.result(out);
149 return acc.log_sum_exp();
150}

154 // RoPE then, or before, the QK norm of one head, as the configuration says
155 template <transformer_config C, typename T, typename R>
156 constexpr void position(std::span<T> head, std::span<const T> norm, std::size_t pos, const R& rope_src)
157 {
158 constexpr T eps = norm_eps_v<C, T>;
159 if constexpr (C.qk_norm == qk_norm_kind::before_rope)
160 rms_norm(head, norm, head, eps);
161 if constexpr (C.rope != rope_kind::none)
162 ml::rope(head, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
163 if constexpr (C.qk_norm == qk_norm_kind::after_rope)
164 rms_norm(head, norm, head, eps);
165 }

178template <transformer_config C, weight_matrix W, math::math_number T, row X, kv_store<T> S, typename R, mutable_row Out>
179 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
180 && (C.rope == rope_kind::none || rope_source<R>)
181constexpr void attention_forward(const attention_weights<W, T>& w, const X& x, std::size_t pos, const S& cache,
182 std::size_t layer, const R& rope_src, Out&& out)
183{
184 static_assert(valid(C), "snicholls::ml::attention_forward: the transformer_config is not valid");
185 static_assert(C.mask == attention_mask::causal,
186 "snicholls::ml::attention_forward: decoding one token at a time is causal; run a bidirectional or block-causal layer over the sequence");
187 assert(w.q.rows == C.q_size() && w.k.rows == C.kv_size() && w.v.rows == C.kv_size() && w.o.rows == C.d_model
188 && w.q.cols == C.d_model && w.k.cols == C.d_model && w.v.cols == C.d_model && w.o.cols == C.q_size());
189 constexpr std::size_t d = C.head_dim;
190 std::array<T, C.q_size()> q{};
191 std::array<T, C.kv_size()> k{}, v{};
192 const auto in = prepare_input<C.d_model>(w.q, x);
193 linear(w.q, in, q, w.q_bias);
194 linear(w.k, in, k, w.k_bias);
195 linear(w.v, in, v, w.v_bias);
196 for (std::size_t h = 0; h < C.n_heads; ++h)
197 detail::attention_layer::position<C>(std::span<T>(q).subspan(h * d, d), w.q_norm, pos, rope_src);
198 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
199 detail::attention_layer::position<C>(std::span<T>(k).subspan(h * d, d), w.k_norm, pos, rope_src);
200 cache.append(layer, pos, k, v);
201 std::array<T, C.q_size()> y{};
202 const key_range visible = attention_window(pos, C.window);
203 for (std::size_t h = 0; h < C.n_heads; ++h)
204 attention_head<C>(std::span<const T>(q).subspan(h * d, d), cache.visible(layer, h / C.group(), visible),
205 std::span<T>(y).subspan(h * d, d));
206 linear(w.o, prepare_input<C.q_size()>(w.o, y), out, w.o_bias);
207}

224template <transformer_config C, math::math_number T, row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK,
225 mutable_row DV>
226 requires std::same_as<row_value_t<Q>, T> && std::same_as<row_value_t<K>, T> && std::same_as<row_value_t<V>, T>
227 && std::same_as<row_value_t<O>, T> && std::same_as<row_value_t<DO>, T> && std::same_as<row_value_t<DQ>, T>
228 && std::same_as<row_value_t<DK>, T> && std::same_as<row_value_t<DV>, T>
229constexpr void attention_head_backward(const Q& q, const K& keys, const V& values, const O& out, const DO& dout,
230 std::type_identity_t<T> lse, DQ&& dq, DK&& dkeys, DV&& dvalues)
231{
232 const T scale = T(1) / math::sqrt(T(C.head_dim));
233 if constexpr (C.attention_softcap == 0) {
234 attend_backward(q, keys, values, out, dout, lse, scale, dq, dkeys, dvalues);
235 } else {
236 constexpr std::size_t d = C.head_dim;
237 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values), o = as_row(out), g = as_row(dout);
238 const auto gq = as_mutable_row(dq), gk = as_mutable_row(dkeys), gv = as_mutable_row(dvalues);
239 assert(qs.size() == d && o.size() == d && g.size() == d && gq.size() == d && ks.size() == vs.size()
240 && ks.size() % d == 0 && gk.size() == ks.size() && gv.size() == vs.size());
241 std::fill(gq.begin(), gq.end(), T(0));
242 if (lse == -std::numeric_limits<T>::infinity())
243 return;
244 constexpr T cap = attention_softcap_v<C, T>;
245 const T big_d = detail::attention::dot(g, o);
246 for (std::size_t j = 0; j < ks.size() / d; ++j) {
247 const auto k = ks.subspan(j * d, d);
248 const auto v = vs.subspan(j * d, d);
249 const T r = scale * detail::attention::dot(qs, k);
250 const T p = math::exp(softcap(r, cap) - lse);
251 const T ds = p * (detail::attention::dot(g, v) - big_d) * softcap_derivative(r, cap);
252 for (std::size_t i = 0; i < d; ++i) {
253 gv[j * d + i] += p * g[i];
254 gq[i] += scale * ds * k[i];
255 gk[j * d + i] += scale * ds * qs[i];
256 }
257 }
258 }

307 // The backward of position(): the gradient in a head after it, to the gradient in its raw projection, in place
308 template <transformer_config C, typename T, typename R>
309 constexpr void position_backward(std::span<T> d, std::span<const T> raw, std::span<const T> norm,
310 std::span<T> dnorm, std::size_t pos, const R& rope_src)
311 {
312 if constexpr (C.qk_norm == qk_norm_kind::after_rope) {
313 constexpr T eps = norm_eps_v<C, T>;
314 std::array<T, C.head_dim> x{}, scratch{};
315 std::copy(raw.begin(), raw.end(), x.begin()); // the norm's input: the rotated projection
316 if constexpr (C.rope != rope_kind::none)
317 ml::rope(x, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
318 const T rstd = rms_norm(x, norm, scratch, eps);
319 rms_norm_backward(d, x, norm, rstd, d, dnorm);
320 }
321 if constexpr (C.rope != rope_kind::none)
322 rope_inverse(d, rope_src.cos_at(pos), rope_src.sin_at(pos), C.layout());
323 if constexpr (C.qk_norm == qk_norm_kind::before_rope) {
324 constexpr T eps = norm_eps_v<C, T>;
325 std::array<T, C.head_dim> x{}, scratch{};
326 std::copy(raw.begin(), raw.end(), x.begin());
327 const T rstd = rms_norm(x, norm, scratch, eps);
328 rms_norm_backward(d, x, norm, rstd, d, dnorm);
329 }
330 }

332 // The forward after the projections, for token t of n, shared by both paths: the raw copies, then QK norm and
333 // RoPE of its heads in place (the query in act.q, the keys in act.k's [kv_head][t] rows)
334 template <transformer_config C, typename W, typename T, typename R>
335 constexpr void position_token(const attention_weights<W, T>& w, const attention_activations<T>& act, std::size_t n,
336 std::size_t t, const R& rope_src)
337 {
338 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
339 const auto q = act.q.subspan(t * qn, qn);
340 if constexpr (C.qk_norm != qk_norm_kind::none) {
341 std::copy(q.begin(), q.end(), act.q_raw.begin() + std::ptrdiff_t(t * qn));
342 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
343 std::copy_n(act.k.begin() + std::ptrdiff_t((h * n + t) * d), d, act.k_raw.begin() + std::ptrdiff_t(t * kn + h * d));
344 }
345 for (std::size_t h = 0; h < C.n_heads; ++h)
346 position<C>(q.subspan(h * d, d), w.q_norm, t, rope_src);
347 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
348 position<C>(act.k.subspan((h * n + t) * d, d), w.k_norm, t, rope_src);
349 }

372 // Token t's heads over its visible keys: its rows of act.y and act.lse
373 template <transformer_config C, typename T>
374 constexpr void attend_token(const attention_activations<T>& act, std::size_t n, std::size_t t)
375 {
376 constexpr std::size_t d = C.head_dim, qn = C.q_size();
377 const key_range r = keys<C>(act, n, t);
378 const auto yt = act.y.subspan(t * qn, qn);
379 for (std::size_t h = 0; h < C.n_heads; ++h) {
380 const std::size_t kvh = h / C.group();
381 kv_runs<T, 1> run{};
382 run.keys[0] = std::span<const T>(act.k).subspan((kvh * n + r.first) * d, r.size() * d);
383 run.values[0] = std::span<const T>(act.v).subspan((kvh * n + r.first) * d, r.size() * d);
384 act.lse[t * C.n_heads + h] = attention_head<C>(std::span<const T>(act.q).subspan(t * qn + h * d, d), run, yt.subspan(h * d, d));
385 }
386 }

388 // The backward of token t's heads from the gradient in their outputs: its query gradients written, its keys'
389 // and values' added into
390 template <transformer_config C, typename T>
391 constexpr void heads_backward_token(const attention_activations<T>& act, std::span<const T> dy,
392 const attention_gradients<T>& grads, std::size_t n, std::size_t t)
393 {
394 constexpr std::size_t d = C.head_dim, qn = C.q_size();
395 const auto yt = std::span<const T>(act.y).subspan(t * qn, qn);
396 const key_range r = keys<C>(act, n, t);
397 for (std::size_t h = 0; h < C.n_heads; ++h) {
398 const std::size_t kvh = h / C.group(), at = (kvh * n + r.first) * d;
399 attention_head_backward<C, T>(std::span<const T>(act.q).subspan(t * qn + h * d, d),
400 std::span<const T>(act.k).subspan(at, r.size() * d),
401 std::span<const T>(act.v).subspan(at, r.size() * d), yt.subspan(h * d, d),
402 dy.subspan(h * d, d), act.lse[t * C.n_heads + h], grads.q.subspan(t * qn + h * d, d),
403 grads.k.subspan(at, r.size() * d), grads.v.subspan(at, r.size() * d));
404 }
405 }

407 // QK norm and RoPE undone for token t, in place: its row of grads.q and its [kv_head][t] rows of grads.k
408 template <transformer_config C, typename W, typename T, typename R>
409 constexpr void position_backward_token(const attention_weights<W, T>& w, const attention_activations<T>& act,
410 const attention_gradients<T>& grads, std::size_t n, std::size_t t, const R& rope_src)
411 {
412 if constexpr (C.qk_norm != qk_norm_kind::none || C.rope != rope_kind::none) {
413 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
414 constexpr bool raw = C.qk_norm != qk_norm_kind::none;
415 for (std::size_t h = 0; h < C.n_heads; ++h)
416 position_backward<C>(grads.q.subspan(t * qn + h * d, d),
417 raw ? std::span<const T>(act.q_raw).subspan(t * qn + h * d, d) : std::span<const T>{},
418 w.q_norm, grads.q_norm, t, rope_src);
419 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
420 position_backward<C>(grads.k.subspan((h * n + t) * d, d),
421 raw ? std::span<const T>(act.k_raw).subspan(t * kn + h * d, d) : std::span<const T>{},
422 w.k_norm, grads.k_norm, t, rope_src);
423 }
424 }

426 // Sequence s of a batch of sequences of n tokens: every per-sequence view is a slice, the batch's being the
427 // sequences' own laid end to end
428 template <typename T>
429 constexpr std::span<T> part(std::span<T> x, std::size_t s, std::size_t size)
430 {
431 return x.empty() ? x : x.subspan(s * size, size);
432 }
433 template <transformer_config C, typename T>
434 constexpr attention_activations<T> sequence(const attention_activations<T>& a, std::size_t n, std::size_t s)
435 {
436 constexpr std::size_t qn = C.q_size(), kn = C.kv_size();
437 return {part(a.q, s, n * qn), part(a.k, s, n * kn), part(a.v, s, n * kn), part(a.q_raw, s, n * qn),
438 part(a.k_raw, s, n * kn), part(a.y, s, n * qn), part(a.lse, s, n * C.n_heads), part(a.lengths, s, 1)};
439 }
440 template <transformer_config C, typename T>
441 constexpr attention_gradients<T> sequence(const attention_gradients<T>& g, std::size_t n, std::size_t s)
442 {
443 constexpr std::size_t qn = C.q_size(), kn = C.kv_size();
444 attention_gradients<T> r = g;
445 r.q = part(g.q, s, n * qn);
446 r.k = part(g.k, s, n * kn);
447 r.v = part(g.v, s, n * kn);
448 r.y = part(g.y, s, n * qn);
449 return r;
450 }

551template <transformer_config C, weight_matrix W, math::math_number T, row X, typename R, mutable_row Out>
552 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<Out>, T>
553 && (C.rope == rope_kind::none || rope_source<R>)
554constexpr void attention_forward_sequence(const attention_weights<W, T>& w, const X& xs, const R& rope_src,
555 const attention_activations<T>& act, Out&& out)
556{
557 static_assert(valid(C), "snicholls::ml::attention_forward_sequence: the transformer_config is not valid");
558 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
559 const auto x = as_row(xs);
560 const auto y = as_mutable_row(out);
561 const std::size_t n = x.size() / C.d_model;
562 assert(x.size() == n * C.d_model && y.size() == n * C.d_model);
563 detail::attention_layer::check_activations(act, n, qn, kn, C.n_heads, C.qk_norm != qk_norm_kind::none);
564 detail::attention_layer::check_lengths(act, n, 1);
565 std::array<T, kn> k{}, v{};
566 for (std::size_t t = 0; t < n; ++t) {
567 const auto in = prepare_input<C.d_model>(w.q, x.subspan(t * C.d_model, C.d_model));
568 linear(w.q, in, act.q.subspan(t * qn, qn), w.q_bias);
569 linear(w.k, in, k, w.k_bias);
570 linear(w.v, in, v, w.v_bias);
571 for (std::size_t h = 0; h < C.n_kv_heads; ++h) {
572 std::copy_n(k.begin() + std::ptrdiff_t(h * d), d, act.k.begin() + std::ptrdiff_t((h * n + t) * d));
573 std::copy_n(v.begin() + std::ptrdiff_t(h * d), d, act.v.begin() + std::ptrdiff_t((h * n + t) * d));
574 }
575 detail::attention_layer::position_token<C>(w, act, n, t, rope_src);
576 }
577 // Every key is in place before any token attends: a bidirectional or block-causal token reads later keys
578 for (std::size_t t = 0; t < n; ++t) {
579 detail::attention_layer::attend_token<C>(act, n, t);
580 linear(w.o, prepare_input<qn>(w.o, std::span<const T>(act.y).subspan(t * qn, qn)), y.subspan(t * C.d_model, C.d_model), w.o_bias);
581 }
582}

605 requires linalg::blas_backend<B, T> && (C.rope == rope_kind::none || rope_source<R>)
606constexpr void attention_forward_sequence(const B& backend, const attention_weights<dense_weights<T>, T>& w,
607 std::type_identity_t<std::span<const T>> xs, const R& rope_src,
608 const attention_activations<T>& act, std::type_identity_t<std::span<T>> out,
609 std::size_t sequences = 1)
610{

614 assert(sequences > 0 && xs.size() == n * sequences * C.d_model && out.size() == xs.size());

616 linear_sequence(backend, w.q, xs, act.q, w.q_bias);
617 linear_heads(backend, w.k, xs, act.k, w.k_bias, C.n_kv_heads, sequences); // the keys and values head by head
618 linear_heads(backend, w.v, xs, act.v, w.v_bias, C.n_kv_heads, sequences);

622}

639template <transformer_config C, typename S, math::math_number T, row X, typename R, row DO, mutable_row DX>
640 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<DO>, T> && std::same_as<row_value_t<DX>, T>
641 && (C.rope == rope_kind::none || rope_source<R>)
642constexpr void attention_backward(const attention_weights<dense_weights<S>, T>& w, const X& xs, const R& rope_src,
643 const attention_activations<T>& act, const DO& dout, DX&& dxs,
644 const attention_gradients<T>& grads)
645{
646 static_assert(valid(C), "snicholls::ml::attention_backward: the transformer_config is not valid");
647 constexpr std::size_t d = C.head_dim, qn = C.q_size(), kn = C.kv_size();
648 const auto x = as_row(xs);
649 const auto g = as_row(dout);
650 const auto gx = as_mutable_row(dxs);
651 const std::size_t n = x.size() / C.d_model;
652 assert(g.size() == x.size() && (gx.empty() || gx.size() == x.size()));
653 assert(grads.q.size() == n * qn && grads.k.size() == n * kn && grads.v.size() == n * kn);
654 std::fill(grads.k.begin(), grads.k.end(), T(0));
655 std::fill(grads.v.begin(), grads.v.end(), T(0));
656 for (std::size_t t = 0; t < n; ++t) {
657 std::array<T, qn> dy{};
658 linear_backward(w.o, std::span<const T>(act.y).subspan(t * qn, qn), g.subspan(t * C.d_model, C.d_model), dy, grads.wo, grads.o_bias);
659 detail::attention_layer::heads_backward_token<C>(act, std::span<const T>(dy), grads, n, t);
660 }
661 const std::span<T> no_dx{};
662 for (std::size_t t = 0; t < n; ++t) {
663 detail::attention_layer::position_backward_token<C>(w, act, grads, n, t, rope_src);
664 std::array<T, kn> dk{}, dv{};
665 for (std::size_t h = 0; h < C.n_kv_heads; ++h)
666 for (std::size_t i = 0; i < d; ++i) {
667 dk[h * d + i] = grads.k[(h * n + t) * d + i];
668 dv[h * d + i] = grads.v[(h * n + t) * d + i];
669 }
670 const auto xt = x.subspan(t * C.d_model, C.d_model);
671 const auto dxt = gx.empty() ? no_dx : gx.subspan(t * C.d_model, C.d_model);
672 linear_backward(w.q, xt, grads.q.subspan(t * qn, qn), dxt, grads.wq, grads.q_bias);
673 linear_backward(w.k, xt, dk, dxt, grads.wk, grads.k_bias);
674 linear_backward(w.v, xt, dv, dxt, grads.wv, grads.v_bias);
675 }
676}
