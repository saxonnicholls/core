144constexpr void mtp_input_ids(std::span<const std::uint32_t> ids, std::size_t depth, std::size_t sequences, std::uint32_t pad,
145 std::span<std::uint32_t> out)
146{
147 assert(sequences > 0 && ids.size() % sequences == 0 && out.size() == ids.size());
148 const std::size_t n = ids.size() / sequences;
149 assert(depth < n || n == 0);
150 for (std::size_t s = 0; s < sequences; ++s)
151 for (std::size_t i = 0; i < n; ++i)
152 out[s * n + i] = i + depth < n ? ids[s * n + i + depth] : pad;
153}

162template <transformer_config C, typename B, math::math_number T>
163constexpr void mtp_combine_rows(const B&, std::type_identity_t<std::span<const T>> hidden, std::type_identity_t<std::span<const T>> embedded,
164 std::type_identity_t<std::span<const T>> hidden_norm,
165 std::type_identity_t<std::span<const T>> embedding_norm, std::span<T> combined)
166{
167 constexpr std::size_t d = C.d_model;
168 constexpr T eps = norm_eps_v<C, T>;
169 const std::size_t rows = hidden.size() / d;
170 assert(hidden.size() == rows * d && embedded.size() == hidden.size() && combined.size() == 2 * hidden.size());
171 for (std::size_t t = 0; t < rows; ++t) {
172 rms_norm(hidden.subspan(t * d, d), hidden_norm, combined.subspan(2 * t * d, d), eps);
173 rms_norm(embedded.subspan(t * d, d), embedding_norm, combined.subspan(2 * t * d + d, d), eps);
174 }
175}

182template <transformer_config C, typename B, math::math_number T>
183constexpr void mtp_combine_rows_backward(const B&, std::type_identity_t<std::span<const T>> dcombined,
184 std::type_identity_t<std::span<const T>> hidden, std::type_identity_t<std::span<const T>> embedded,
185 std::type_identity_t<std::span<const T>> hidden_norm,
186 std::type_identity_t<std::span<const T>> embedding_norm, std::span<T> dhidden,
187 std::type_identity_t<std::span<T>> dembedded, std::type_identity_t<std::span<T>> dhidden_norm,
188 std::type_identity_t<std::span<T>> dembedding_norm)
189{
190 constexpr std::size_t d = C.d_model;
191 constexpr T eps = norm_eps_v<C, T>;
192 const std::size_t rows = hidden.size() / d;
193 assert(dcombined.size() == 2 * hidden.size() && dhidden.size() == hidden.size() && dembedded.size() == embedded.size());
194 std::array<T, d> scratch{}, dx{};
195 for (std::size_t t = 0; t < rows; ++t) {
196 const auto h = hidden.subspan(t * d, d), e = embedded.subspan(t * d, d);
197 T rstd = rms_norm(h, hidden_norm, scratch, eps);
198 rms_norm_backward(dcombined.subspan(2 * t * d, d), h, hidden_norm, rstd, dx, dhidden_norm);
199 for (std::size_t i = 0; i < d; ++i)
200 dhidden[t * d + i] += dx[i];
201 rstd = rms_norm(e, embedding_norm, scratch, eps);
202 rms_norm_backward(dcombined.subspan(2 * t * d + d, d), e, embedding_norm, rstd, dx, dembedding_norm);
203 for (std::size_t i = 0; i < d; ++i)
204 dembedded[t * d + i] += dx[i];
205 }
206}

248template <transformer_config C, typename B, math::math_number T, typename Block, typename BlockActivations, typename R>
249 requires linalg::blas_backend<B, T>
250constexpr auto mtp_forward_sequence(const B& backend, const mtp_weights<Block, T>& w, std::type_identity_t<std::span<const T>> hidden_prev,
251 std::type_identity_t<std::span<const T>> embedded, const R& rope_src,
252 const mtp_activations<BlockActivations, T>& act, std::size_t sequences = 1)
253{
254 assert(w.projection.rows == C.d_model && w.projection.cols == 2 * C.d_model && act.output.size() == hidden_prev.size()
255 && act.head_input.size() == hidden_prev.size());
256 mtp_combine_rows<C>(backend, hidden_prev, embedded, w.hidden_norm, w.embedding_norm, act.combined);
257 linear_sequence(backend, w.projection, std::span<const T>(act.combined), act.output);
258 return detail::mtp::then([&] { return block_forward_sequence<C>(backend, w.block, act.output, rope_src, act.block, sequences); },
259 [&] { block_norm_rows<C>(backend, std::span<const T>(act.output), w.output_norm, std::span<const T>{}, act.head_input); });
260}

277template <transformer_config C, typename B, math::math_number T, typename Block, typename BlockActivations, typename BlockGradients,
278 typename R>
279 requires linalg::blas_backend<B, T>
280constexpr void mtp_backward(const B& backend, const mtp_weights<Block, T>& w, std::type_identity_t<std::span<const T>> hidden_prev,
281 std::type_identity_t<std::span<const T>> embedded, const R& rope_src,
282 const mtp_activations<BlockActivations, T>& act, std::type_identity_t<std::span<const T>> dhead_input,
283 std::type_identity_t<std::span<const T>> doutput, std::type_identity_t<std::span<T>> dhidden_prev,
284 std::type_identity_t<std::span<T>> dembedded, const mtp_gradients<BlockGradients, T>& grads,
285 std::size_t sequences = 1)
286{
287 assert(grads.output.size() == hidden_prev.size() && grads.combined.size() == 2 * hidden_prev.size()
288 && (doutput.empty() || doutput.size() == hidden_prev.size()));
289 if (doutput.empty())
290 fill_rows(backend, grads.output, T(0));
291 else
292 copy_rows(backend, doutput, grads.output);
293 block_norm_rows_backward_add<C>(backend, dhead_input, std::span<const T>(act.output), w.output_norm, std::span<const T>{},
294 grads.output, grads.output_norm, std::span<T>{});
295 block_backward<C>(backend, w.block, rope_src, act.block, grads.output, grads.block, sequences);
296 fill_rows(backend, grads.combined, T(0));
297 linear_sequence_backward(backend, w.projection, std::span<const T>(act.combined), std::span<const T>(grads.output), grads.combined,
298 grads.projection);
299 mtp_combine_rows_backward<C>(backend, std::span<const T>(grads.combined), hidden_prev, embedded, w.hidden_norm, w.embedding_norm,
300 dhidden_prev, dembedded, grads.hidden_norm, grads.embedding_norm);
301}

317template <math::math_number T, typename B>
318 requires linalg::blas_backend<B, T>
319constexpr T mtp_loss_sequence(const B& backend, const dense_weights<T>& head, std::type_identity_t<std::span<const T>> head_input,
320 std::span<const std::uint32_t> targets, std::size_t depth, std::type_identity_t<std::span<T>> logits,
321 std::type_identity_t<std::span<T>> losses, std::type_identity_t<std::span<T>> dhead_input,
322 std::type_identity_t<std::span<T>> dhead, std::type_identity_t<T> weight, std::size_t sequences = 1)
323{
324 const std::size_t d = head.cols, rows = targets.size(), n = rows / sequences;
325 assert(sequences > 0 && rows == n * sequences && depth > 0 && depth < n && head_input.size() == rows * d
326 && dhead_input.size() == head_input.size() && losses.size() == rows && logits.size() >= (n - depth) * head.rows);
327 fill_rows(backend, losses, T(0));
328 T total = T(0);
329 const std::size_t m = n - depth;
330 for (std::size_t s = 0; s < sequences; ++s)
331 total += lm_head_loss_sequence(backend, head, head_input.subspan(s * n * d, m * d), targets.subspan(s * n + depth, m),
332 logits.first(m * head.rows), losses.subspan(s * n, m), dhead_input.subspan(s * n * d, m * d),
333 dhead, 0, weight, rows);
334 return total;
335}
