121template <std::size_t Rank>
122 requires(Rank >= 1 && Rank <= 3)
123struct conv_shape
124{
125 std::size_t in_channels = 0;
126 std::size_t out_channels = 0;
127 std::array<std::size_t, Rank> input{};
128 std::array<std::size_t, Rank> kernel{};
129 std::array<std::size_t, Rank> stride = detail::conv::ones<Rank>();
130 std::array<std::size_t, Rank> padding{};
131 std::array<std::size_t, Rank> dilation = detail::conv::ones<Rank>();
132 std::size_t groups = 1;
133
137 [[nodiscard]] constexpr bool valid() const noexcept
138 {
139 if (in_channels == 0 || out_channels == 0 || groups == 0 || in_channels % groups != 0 || out_channels % groups != 0)
140 return false;
141 for (std::size_t d = 0; d < Rank; ++d)
142 if (kernel[d] == 0 || stride[d] == 0 || dilation[d] == 0 || input[d] + 2 * padding[d] < dilation[d] * (kernel[d] - 1) + 1)
143 return false;
144 return true;
145 }
146
148 [[nodiscard]] constexpr std::array<std::size_t, Rank> output() const noexcept
149 {
150 std::array<std::size_t, Rank> o{};
151 for (std::size_t d = 0; d < Rank; ++d)
152 o[d] = (input[d] + 2 * padding[d] - dilation[d] * (kernel[d] - 1) - 1) / stride[d] + 1;
153 return o;
154 }
155
157 [[nodiscard]] constexpr std::size_t input_size() const noexcept { return detail::conv::product(input); }
160 [[nodiscard]] constexpr std::size_t output_size() const noexcept { return detail::conv::product(output()); }
163 [[nodiscard]] constexpr std::size_t kernel_size() const noexcept { return detail::conv::product(kernel); }
166 [[nodiscard]] constexpr std::size_t group_in_channels() const noexcept { return in_channels / groups; }
169 [[nodiscard]] constexpr std::size_t group_out_channels() const noexcept { return out_channels / groups; }
172 [[nodiscard]] constexpr std::size_t x_size() const noexcept { return in_channels * input_size(); }
175 [[nodiscard]] constexpr std::size_t y_size() const noexcept { return out_channels * output_size(); }
178 [[nodiscard]] constexpr std::size_t weight_size() const noexcept { return out_channels * group_in_channels() * kernel_size(); }
181 [[nodiscard]] constexpr std::size_t column_width() const noexcept { return in_channels * kernel_size(); }
184 [[nodiscard]] constexpr std::size_t column_size() const noexcept { return output_size() * column_width(); }
185
188 friend constexpr bool operator==(const conv_shape&, const conv_shape&) = default;
189};

210 // Steps row-major coordinates c within extents e; false after the last
211 template <std::size_t Rank>
212 constexpr bool next(std::array<std::size_t, Rank>& c, const std::array<std::size_t, Rank>& e) noexcept
213 {
214 for (std::size_t d = Rank; d-- > 0;) {
215 if (++c[d] < e[d])
216 return true;
217 c[d] = 0;
218 }
219 return false;
220 }

222 // The taps of dimension D on, inside the input: the range of each dimension's taps found once, no division
223 template <std::size_t D, std::size_t Rank, typename F>
224 constexpr void taps(const conv_shape<Rank>& s, const std::array<std::ptrdiff_t, Rank>& base, std::size_t t, std::size_t at, F& f)
225 {
226 if constexpr (D == Rank) {
227 f(t, at);
228 } else {
229 const auto dil = static_cast<std::ptrdiff_t>(s.dilation[D]), n = static_cast<std::ptrdiff_t>(s.input[D]);
230 std::size_t k = 0;
231 std::ptrdiff_t pos = base[D];
232 for (; k < s.kernel[D] && pos < 0; ++k) // the taps in the leading padding
233 pos += dil;
234 for (; k < s.kernel[D] && pos < n; ++k, pos += dil)
235 taps<D + 1>(s, base, t * s.kernel[D] + k, at * s.input[D] + static_cast<std::size_t>(pos), f);
236 }
237 }

239 // Where output o's window starts in each dimension of the input (negative in the padding)
240 template <std::size_t Rank>
241 constexpr std::array<std::ptrdiff_t, Rank> window(const conv_shape<Rank>& s, const std::array<std::size_t, Rank>& o) noexcept
242 {
243 std::array<std::ptrdiff_t, Rank> base{};
244 for (std::size_t d = 0; d < Rank; ++d)
245 base[d] = static_cast<std::ptrdiff_t>(o[d] * s.stride[d]) - static_cast<std::ptrdiff_t>(s.padding[d]);
246 return base;
247 }

249 // Calls f(tap, input position) for each tap of output o inside the input, taps in row-major order
250 template <std::size_t Rank, typename F>
251 constexpr void for_each_tap(const conv_shape<Rank>& s, const std::array<std::size_t, Rank>& o, F&& f)
252 {
253 taps<0>(s, window(s, o), 0, 0, f);

258 // y[oc][p] = sum over (ic of the group, tap) of w * x, from 0, in that order; then + bias
259 template <std::size_t Rank, typename T>
260 constexpr void forward(const conv_shape<Rank>& s, std::span<const T> x, std::span<const T> w, std::span<const T> bias, std::span<T> y)
261 {
262 const std::size_t cg = s.group_in_channels(), og = s.group_out_channels(), ks = s.kernel_size(), os = s.output_size(),
263 is = s.input_size();
264 const auto out = s.output();
265 for (std::size_t oc = 0; oc < s.out_channels; ++oc) {
266 const std::size_t g = oc / og;
267 std::array<std::size_t, Rank> o{};
268 for (std::size_t p = 0; p < os; ++p, next(o, out)) {
269 T sum = T(0);
270 const T* wr = w.data() + oc * cg * ks;
271 const T* xr = x.data() + g * cg * is;
272 const auto base = window(s, o);
273 for (std::size_t c = 0; c < cg; ++c, wr += ks, xr += is) {
274 auto f = [&](std::size_t t, std::size_t at) {
275 const T term = wr[t] * xr[at];
276 sum += term;
277 };
278 taps<0>(s, base, 0, 0, f);
279 }
280 y[oc * os + p] = bias.empty() ? sum : sum + bias[oc];
281 }
282 }
283 }

285 // dx = Wᵀ dy scattered back (written), in the order (oc, p, c, tap)
286 template <std::size_t Rank, typename T>
287 constexpr void data_gradient(const conv_shape<Rank>& s, std::span<const T> w, std::span<const T> dy, std::span<T> dx)
288 {
289 const std::size_t cg = s.group_in_channels(), og = s.group_out_channels(), ks = s.kernel_size(), os = s.output_size(),
290 is = s.input_size();
291 const auto out = s.output();
292 for (T& v : dx)
293 v = T(0);
294 for (std::size_t oc = 0; oc < s.out_channels; ++oc) {
295 const std::size_t g = oc / og;
296 std::array<std::size_t, Rank> o{};
297 for (std::size_t p = 0; p < os; ++p, next(o, out)) {
298 const T gy = dy[oc * os + p];
299 for (std::size_t c = 0; c < cg; ++c) {
300 const T* wr = w.data() + (oc * cg + c) * ks;
301 T* xr = dx.data() + (g * cg + c) * is;
302 for_each_tap(s, o, [&](std::size_t t, std::size_t at) {
303 const T term = wr[t] * gy;
304 xr[at] += term;
305 });
306 }
307 }
308 }
309 }

311 // The input position tap k of output o reads, or input_size() in the padding; no division
312 template <std::size_t Rank>
313 constexpr std::size_t source(const conv_shape<Rank>& s, const std::array<std::size_t, Rank>& o, const std::array<std::size_t, Rank>& k) noexcept
314 {
315 std::size_t at = 0;
316 for (std::size_t d = 0; d < Rank; ++d) {
317 const std::size_t padded = o[d] * s.stride[d] + k[d] * s.dilation[d]; // in the padded input
318 if (padded < s.padding[d] || padded - s.padding[d] >= s.input[d])
319 return s.input_size();
320 at = at * s.input[d] + (padded - s.padding[d]);
321 }
322 return at;
323 }

325 // dw[oc][c][t] += Σ_p dy[oc][p] · x[ic][src(p, t)], the sum over p in order, then added once
326 template <std::size_t Rank, typename T>
327 constexpr void weight_gradient(const conv_shape<Rank>& s, std::span<const T> x, std::span<const T> dy, std::span<T> dw)
328 {
329 const std::size_t cg = s.group_in_channels(), og = s.group_out_channels(), ks = s.kernel_size(), os = s.output_size(),
330 is = s.input_size();
331 const auto out = s.output();
332 for (std::size_t oc = 0; oc < s.out_channels; ++oc) {
333 const std::size_t g = oc / og;
334 for (std::size_t c = 0; c < cg; ++c) {
335 const T* xr = x.data() + (g * cg + c) * is;
336 const T* gr = dy.data() + oc * os;
337 std::array<std::size_t, Rank> k{};
338 for (std::size_t t = 0; t < ks; ++t, next(k, s.kernel)) {
339 T sum = T(0);
340 std::array<std::size_t, Rank> o{};
341 for (std::size_t p = 0; p < os; ++p, next(o, out)) {
342 const std::size_t at = source(s, o, k);
343 if (at == is)
344 continue;
345 const T term = gr[p] * xr[at];
346 sum += term;
347 }
348 dw[(oc * cg + c) * ks + t] += sum;
349 }
350 }
351 }
352 }

354 template <typename T>
355 constexpr void bias_gradient(std::size_t channels, std::size_t positions, std::span<const T> dy, std::span<T> db)
356 {
357 for (std::size_t oc = 0; oc < channels; ++oc) {
358 T sum = T(0);
359 for (std::size_t p = 0; p < positions; ++p)
360 sum += dy[oc * positions + p];
361 db[oc] += sum;
362 }

372template <std::size_t Rank, row X, mutable_row Cols>
373 requires std::same_as<row_value_t<X>, row_value_t<Cols>>
374constexpr void im2col(const conv_shape<Rank>& s, const X& x, Cols&& cols)
375{
376 using T = row_value_t<X>;
377 const auto in = as_row(x);
378 const auto c = as_mutable_row(cols);
379 assert(s.valid() && in.size() == s.x_size() && c.size() == s.column_size());
380 const std::size_t ks = s.kernel_size(), is = s.input_size(), width = s.column_width();
381 const auto out = s.output();
382 for (T& v : c)
383 v = T(0);
384 std::array<std::size_t, Rank> o{};
385 for (std::size_t p = 0; p < s.output_size(); ++p, detail::conv::next(o, out))
386 for (std::size_t ic = 0; ic < s.in_channels; ++ic) {
387 T* dst = c.data() + p * width + ic * ks;
388 const T* src = in.data() + ic * is;
389 detail::conv::for_each_tap(s, o, [&](std::size_t t, std::size_t at) { dst[t] = src[at]; });
390 }
391}

450template <std::size_t Rank, row X, row W, row B, mutable_row Y, mutable_row Cols>
451 requires std::same_as<row_value_t<X>, row_value_t<W>> && std::same_as<row_value_t<X>, row_value_t<B>>
452 && std::same_as<row_value_t<X>, row_value_t<Y>> && std::same_as<row_value_t<X>, row_value_t<Cols>>
453constexpr void conv_forward_im2col(const conv_shape<Rank>& s, const X& x, const W& w, const B& bias, Y&& y, Cols&& cols)
454{
455 using T = row_value_t<X>;
456 const auto ws = as_row(w);
457 const auto b = as_row(bias);
458 const auto out = as_mutable_row(y);
459 const auto c = as_mutable_row(cols);
460 assert(ws.size() == s.weight_size() && detail::conv::fits(b.size(), s.out_channels) && out.size() == s.y_size());
461 im2col(s, x, c);
462 const std::size_t cg = s.group_in_channels(), og = s.group_out_channels(), ks = s.kernel_size(), os = s.output_size(),
463 width = s.column_width(), inner = cg * ks;
464 for (std::size_t oc = 0; oc < s.out_channels; ++oc) {
465 const std::size_t first = (oc / og) * inner;
466 const auto wrow = ws.subspan(oc * inner, inner);
467 for (std::size_t p = 0; p < os; ++p) {
468 const auto crow = std::span<const T>(c).subspan(p * width + first, inner);
469 T sum = T(0);
470 for (std::size_t j = 0; j < inner; ++j) {
471 const T term = wrow[j] * crow[j];
472 sum += term;
473 }
474 out[oc * os + p] = b.empty() ? sum : sum + b[oc];
475 }
476 }

488template <std::size_t Rank, row X, row W, row DY, mutable_row DX, mutable_row DW, mutable_row DB>

518template <std::size_t Rank>
519 requires(Rank >= 1 && Rank <= 3)
520struct conv_transpose_shape
521{
522 std::size_t in_channels = 0;
523 std::size_t out_channels = 0;
524 std::array<std::size_t, Rank> input{};
525 std::array<std::size_t, Rank> kernel{};
526 std::array<std::size_t, Rank> stride = detail::conv::ones<Rank>();
527 std::array<std::size_t, Rank> padding{};
528 std::array<std::size_t, Rank> dilation = detail::conv::ones<Rank>();
529 std::size_t groups = 1;
530
533 [[nodiscard]] constexpr std::array<std::size_t, Rank> output() const noexcept
534 {
535 std::array<std::size_t, Rank> o{};
536 for (std::size_t d = 0; d < Rank; ++d)
537 o[d] = (input[d] - 1) * stride[d] + dilation[d] * (kernel[d] - 1) + 1 - 2 * padding[d];
538 return o;
539 }
540
542 [[nodiscard]] constexpr conv_shape<Rank> as_conv() const noexcept
543 {
544 return {out_channels, in_channels, output(), kernel, stride, padding, dilation, groups};
545 }
546
548 [[nodiscard]] constexpr bool valid() const noexcept
549 {
550 for (std::size_t d = 0; d < Rank; ++d)
551 if (input[d] == 0 || kernel[d] == 0 || stride[d] == 0 || dilation[d] == 0
552 || (input[d] - 1) * stride[d] + dilation[d] * (kernel[d] - 1) + 1 <= 2 * padding[d])
553 return false;
554 return as_conv().valid() && as_conv().output() == input;
555 }
556
558 [[nodiscard]] constexpr std::size_t x_size() const noexcept { return in_channels * detail::conv::product(input); }
561 [[nodiscard]] constexpr std::size_t y_size() const noexcept { return out_channels * detail::conv::product(output()); }
564 [[nodiscard]] constexpr std::size_t weight_size() const noexcept { return as_conv().weight_size(); }
565
568 friend constexpr bool operator==(const conv_transpose_shape&, const conv_transpose_shape&) = default;
569};
