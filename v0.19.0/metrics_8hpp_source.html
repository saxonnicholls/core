139 [[nodiscard]] constexpr result<gospa_result<T>, metric_error> gospa(std::span<const linalg::fixed_vector<T, D>> truths, std::span<const linalg::fixed_vector<T, D>> estimates, T c,
140 T p = T(1), T alpha = T(2)) noexcept(linalg::detail::nothrow_scalar<T>)
141 {
142 if (!(c > T(0)) || !(p >= T(1)) || !(alpha > T(0)) || alpha > T(2))
143 return err(metric_error::invalid_parameters);
144 const std::size_t m = truths.size(), n = estimates.size();
145 gospa_result<T> g{};
146 const T cp = static_cast<T>(math::pow(c, p));
147 T assigned(0);
148 std::size_t paired = 0;
149 if (m > 0 && n > 0) {
150 const auto s = assign(truths, estimates, c, p);
151 if (!s)
152 return err(s.unwrap_err_unchecked());
153 assigned = *s;
154 for (std::size_t i = 0; i < m; ++i) {
155 const auto j = row_to_col_[i];
156 if (j < 0)
157 continue;
158 const T d = linalg::norm(truths[i] - estimates[static_cast<std::size_t>(j)]);
159 if (d < c) {
160 g.localisation += static_cast<T>(math::pow(d, p));
161 ++paired;
162 }
163 }
164 }
165 g.missed = m - paired;
166 g.false_tracks = n - paired;
167 const std::size_t mismatch = m > n ? m - n : n - m;
168 g.distance = static_cast<T>(math::pow(assigned + cp / alpha * static_cast<T>(mismatch), T(1) / p));
169 return ok(g);
170 }
