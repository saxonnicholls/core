80template <row X, mutable_row Out>
81 requires std::same_as<row_value_t<X>, row_value_t<Out>>
82constexpr void timestep_embedding(const X& timesteps, std::size_t dim, Out&& out, std::type_identity_t<row_value_t<X>> max_period = 10000)
83{
84 using T = row_value_t<X>;
85 const auto t = as_row(timesteps);
86 const auto o = as_mutable_row(out);
87 assert(o.size() == t.size() * dim);
88 const std::size_t half = dim / 2;
89 const T log_period = math::log(max_period);
90 for (std::size_t i = 0; i < t.size(); ++i) {
91 const auto row_out = o.subspan(i * dim, dim);
92 for (std::size_t j = 0; j < half; ++j) {
93 const T freq = math::exp(-log_period * T(j) / T(half));
94 const T arg = t[i] * freq;
95 row_out[j] = math::cos(arg);
96 row_out[j + half] = math::sin(arg);
97 }
98 if (dim % 2 != 0)
99 row_out[2 * half] = T(0);
100 }
101}

109template <math::math_number T>
110[[nodiscard]] constexpr std::size_t arange_count(T start, T stop, T step)
111{
112 assert(step != T(0));
113 const T n = -math::floor(-((stop - start) / step));
114 return n > T(0) ? static_cast<std::size_t>(n) : 0;
115}

122constexpr void arange(std::type_identity_t<row_value_t<Out>> start, std::type_identity_t<row_value_t<Out>> step, Out&& out)
123{
124 using T = row_value_t<Out>;
125 const auto o = as_mutable_row(out);
126 for (std::size_t i = 0; i < o.size(); ++i) {
127 const T term = step * T(i);
128 o[i] = start + term;

135template <row X, mutable_row Y>
136 requires std::same_as<row_value_t<X>, row_value_t<Y>>
137constexpr void cumsum(const X& x, Y&& y)
138{
139 const auto in = as_row(x);
140 const auto out = as_mutable_row(y);
141 assert(out.size() == in.size());
142 for (std::size_t i = 0; i < in.size(); ++i)
143 out[i] = i == 0 ? in[0] : out[i - 1] + in[i];
144}

149template <row DY, mutable_row DX>
150 requires std::same_as<row_value_t<DY>, row_value_t<DX>>
151constexpr void cumsum_backward(const DY& dy, DX&& dx)
152{
153 const auto g = as_row(dy);
154 const auto gx = as_mutable_row(dx);
155 assert(gx.size() == g.size());
156 for (std::size_t i = g.size(); i-- > 0;)
157 gx[i] = i + 1 == g.size() ? g[i] : gx[i + 1] + g[i];
158}

171template <row X>
172constexpr void argsort(const X& x, std::span<std::size_t> indices, sort_order order = sort_order::ascending)
173{
174 const auto v = as_row(x);
175 assert(indices.size() == v.size());
176 for (std::size_t i = 0; i < indices.size(); ++i)
177 indices[i] = i;
178 if (order == sort_order::ascending)
179 std::sort(indices.begin(), indices.end(), [&](std::size_t a, std::size_t b) { return v[a] < v[b] || (!(v[b] < v[a]) && a < b); });
180 else
181 std::sort(indices.begin(), indices.end(), [&](std::size_t a, std::size_t b) { return v[a] > v[b] || (!(v[b] > v[a]) && a < b); });
182}

189template <row X, mutable_row Y>
190 requires std::same_as<row_value_t<X>, row_value_t<Y>>
191constexpr void top_k_mask(const X& x, std::size_t k, Y&& y, std::span<std::size_t> scratch)
192{
193 using T = row_value_t<X>;
194 const auto in = as_row(x);
195 const auto out = as_mutable_row(y);
196 assert(out.size() == in.size() && k <= in.size());
197 argsort(in, scratch, sort_order::descending);
198 for (std::size_t r = k; r < scratch.size(); ++r)
199 out[scratch[r]] = -std::numeric_limits<T>::infinity();
200 for (std::size_t r = 0; r < k; ++r)
201 out[scratch[r]] = in[scratch[r]];
202}

236template <row X, mutable_row Y>
237 requires std::same_as<row_value_t<X>, row_value_t<Y>>
238constexpr void tri(tri_kind kind, std::size_t rows, std::size_t cols, const X& x, Y&& y, std::type_identity_t<row_value_t<X>> fill = 0)
239{
240 const auto in = as_row(x);
241 const auto out = as_mutable_row(y);
242 assert(out.size() == in.size() && rows * cols > 0 && in.size() % (rows * cols) == 0);
243 for (std::size_t e = 0; e < in.size(); ++e) {
244 const std::size_t i = (e / cols) % rows, j = e % cols;
245 out[e] = in_triangle(kind, i, j) ? in[e] : fill;
246 }
247}

256template <row A, row B, mutable_row X>
257 requires std::same_as<row_value_t<A>, row_value_t<B>> && std::same_as<row_value_t<A>, row_value_t<X>>
258constexpr void solve_tri(const A& a, const B& b, X&& x, std::size_t n, std::size_t k)
259{
260 using T = row_value_t<A>;
261 const auto as = as_row(a);
262 const auto bs = as_row(b);
263 const auto xs = as_mutable_row(x);
264 assert(as.size() == n * n && bs.size() == n * k && xs.size() == n * k);
265 for (std::size_t c = 0; c < k; ++c)
266 for (std::size_t i = 0; i < n; ++i) {
267 T sum = T(0);
268 for (std::size_t t = 0; t < i; ++t) {
269 const T term = as[i * n + t] * xs[t * k + c];
270 sum += term;
271 }
272 assert(as[i * n + i] != T(0));
273 xs[i * k + c] = (bs[i * k + c] - sum) / as[i * n + i];
274 }
275}

286template <row A, row X, row DX, mutable_row DA, mutable_row DB>
287 requires std::same_as<row_value_t<A>, row_value_t<X>> && std::same_as<row_value_t<A>, row_value_t<DX>>
288 && std::same_as<row_value_t<A>, row_value_t<DA>> && std::same_as<row_value_t<A>, row_value_t<DB>>
289constexpr void solve_tri_backward(const A& a, const X& x, const DX& dx, DA&& da, DB&& db, std::size_t n, std::size_t k)
290{
291 using T = row_value_t<A>;
292 const auto as = as_row(a);
293 const auto xs = as_row(x);
294 const auto g = as_row(dx);
295 const auto ga = as_mutable_row(da);
296 const auto gb = as_mutable_row(db);
297 assert(as.size() == n * n && xs.size() == n * k && g.size() == n * k && gb.size() == n * k && (ga.empty() || ga.size() == n * n));
298 for (std::size_t c = 0; c < k; ++c)
299 for (std::size_t i = n; i-- > 0;) {
300 T sum = T(0);
301 for (std::size_t t = i + 1; t < n; ++t) {
302 const T term = as[t * n + i] * gb[t * k + c];
303 sum += term;
304 }
305 gb[i * k + c] = (g[i * k + c] - sum) / as[i * n + i];
306 }
307 if (ga.empty())
308 return;
309 for (std::size_t i = 0; i < n; ++i)
310 for (std::size_t j = 0; j <= i; ++j) {
311 T sum = T(0);
312 for (std::size_t c = 0; c < k; ++c) {
313 const T term = gb[i * k + c] * xs[j * k + c];
314 sum += term;
315 }
316 ga[i * n + j] -= sum;
317 }
318}
