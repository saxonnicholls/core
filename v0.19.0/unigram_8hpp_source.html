121 constexpr match_result match(std::string_view text, std::size_t at) const noexcept
122 {
123 if (blob.size() < 4 || !valid())
124 return {{}, 0};
125 const std::size_t units = array_bytes() / 4;
126 std::uint32_t node = base(unit(0));
127 std::size_t best = 0;
128 std::uint32_t value = 0;
129 for (std::size_t i = at; i < text.size(); ++i) {
130 const auto c = static_cast<unsigned char>(text[i]);
131 if (c == 0)
132 break;
133 node ^= c;
134 if (node >= units || (unit(node) & 0x800000FFu) != c)
135 break;
136 const bool leaf = (unit(node) >> 8) & 1u;
137 node ^= base(unit(node));
138 if (leaf) {
139 if (node >= units)
140 break;
141 best = i - at + 1;
142 value = unit(node) & 0x7FFFFFFFu;
143 }
144 }
145 if (best == 0)
146 return {{}, 0};
147 const std::string_view strings = blob.substr(4 + array_bytes());
148 if (value >= strings.size())
149 return {{}, 0};
150 const std::size_t end = strings.find('\0', value);
151 if (end == std::string_view::npos)
152 return {{}, 0};
153 return {strings.substr(value, end - value), best};
154 }
155
158 constexpr std::size_t max_replacement() const noexcept
159 {
160 if (blob.size() < 4 || !valid())
161 return 0;
162 const std::string_view strings = blob.substr(4 + array_bytes());
163 std::size_t longest = 0, run = 0;
164 for (const char c : strings) {
165 run = c == '\0' ? 0 : run + 1;
166 longest = run > longest ? run : longest;
167 }
168 return longest;
169 }
170
171private:
172 constexpr std::uint32_t read32(std::size_t i) const noexcept
173 {
174 return static_cast<std::uint32_t>(static_cast<unsigned char>(blob[i])) |
175 static_cast<std::uint32_t>(static_cast<unsigned char>(blob[i + 1])) << 8 |
176 static_cast<std::uint32_t>(static_cast<unsigned char>(blob[i + 2])) << 16 |
177 static_cast<std::uint32_t>(static_cast<unsigned char>(blob[i + 3])) << 24;
178 }
179 constexpr std::size_t array_bytes() const noexcept { return read32(0); }
180 constexpr std::uint32_t unit(std::size_t i) const noexcept { return read32(4 + 4 * i); }
181 static constexpr std::uint32_t base(std::uint32_t u) noexcept { return (u >> 10) << ((u & (1u << 9)) >> 6); }

211constexpr result<std::size_t, unigram_error> unigram_normalize(std::string_view text, const precompiled_charsmap& charsmap, const token_trie& user_defined,
212 unigram_normalizer_options options, std::span<char> out) noexcept
213{
214 if (!charsmap.valid())
215 return err(unigram_error::bad_charsmap);
216 const std::string_view space = options.escape_whitespaces ? escaped_space : std::string_view(" ");
217 const bool prepend = !options.treat_whitespace_as_suffix && options.add_dummy_prefix;
218 const bool append = options.treat_whitespace_as_suffix && options.add_dummy_prefix;
219 const bool merge = options.remove_extra_whitespaces;
220 std::size_t at = 0;
221 bool fail = false;
222 const auto put = [&](std::string_view s) {
223 if (out.size() - at < s.size()) {
224 fail = true;
225 return;
226 }
227 for (const char c : s)
228 out[at++] = c;
229 };
230 bool prepended = false, in_word = false;
231 for (std::size_t i = 0; i < text.size() && !fail;) {
232 std::string_view r;
233 std::size_t consumed = 0;
234 if (!user_defined.order.empty())
235 if (user_defined.longest_prefix(text.substr(i), consumed) == no_token)
236 consumed = 0;
237 if (consumed > 0) {
238 r = text.substr(i, consumed);
239 } else if (const auto m = charsmap.match(text, i); m.consumed > 0) {
240 r = m.replacement;
241 consumed = m.consumed;
242 } else if (const utf8_decoded d = decode_utf8(text, i); d.valid) {
243 r = text.substr(i, d.length);
244 consumed = d.length;
245 } else {
246 r = "\xEF\xBF\xBD";
247 consumed = 1;
248 }
249 for (const char c : r) {
250 if (c != ' ') {
251 if (!in_word) {
252 in_word = true;
253 if ((prepend && !prepended) || merge) {
254 put(space);
255 prepended = true;
256 }
257 }
258 put(std::string_view(&c, 1));
259 } else {
260 in_word = false;
261 if (!merge)
262 put(space);
263 }
264 }
265 i += consumed;
266 }
267 if (append)
268 put(space);
269 if (fail)
270 return err(unigram_error::output_too_small);
271 return ok(at);
272}

307template <typename Score, typename Kind>
308constexpr result<std::size_t, unigram_error> unigram_encode(std::string_view normalized, const token_trie& vocab, Score&& score, Kind&& kind,
309 unigram_unknown unknown, std::span<std::uint32_t> out,
310 std::span<unigram_lattice_entry> lattice)
311{
312 const std::size_t n = normalized.size();
313 if (n == 0)
314 return ok(std::size_t{0});
315 if (lattice.size() < n + 1)
316 return err(unigram_error::lattice_too_small);
317 if (out.size() < n)
318 return err(unigram_error::output_too_small);
319 constexpr double lowest = -1.7976931348623157e308;
320 for (std::size_t i = 0; i <= n; ++i)
321 lattice[i] = {lowest, 0, unknown.id};
322 lattice[0].score = 0;
323 for (std::size_t i = 0; i < n;) {
324 const std::size_t unit = decode_utf8(normalized, i).length;
325 const double here = lattice[i].score;
326 bool covered = false;
327 auto c = vocab.root();
328 for (std::size_t j = i; j < n && vocab.advance(c, normalized[j]); ++j) {
329 const std::uint32_t id = vocab.at(c);
330 if (id == no_token)
331 continue;
332 covered = covered || j + 1 - i == unit;
333 const double s = here + (kind(id) == token_kind::user_defined ? 0.0 : static_cast<double>(score(id)));
334 if (s > lattice[j + 1].score)
335 lattice[j + 1] = {s, static_cast<std::uint32_t>(i), id};
336 }
337 if (!covered && here + unknown.score > lattice[i + unit].score)
338 lattice[i + unit] = {here + unknown.score, static_cast<std::uint32_t>(i), unknown.id};
339 i += unit;
340 }
341 // Back from the end, one unknown for a run of them, then reversed
342 std::size_t count = 0;
343 bool previous_unknown = false;
344 for (std::size_t at = n;;) {
345 const unigram_lattice_entry& e = lattice[at];
346 const bool is_unknown = e.token == unknown.id;
347 if (!(previous_unknown && is_unknown))
348 out[count++] = e.token;
349 previous_unknown = is_unknown;
350 if (e.from == 0)
351 break;
352 at = e.from;
353 }
354 for (std::size_t a = 0, b = count - 1; a < b; ++a, --b) {
355 const std::uint32_t t = out[a];
356 out[a] = out[b];
357 out[b] = t;
358 }
359 return ok(count);
360}
