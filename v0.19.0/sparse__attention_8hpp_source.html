99template <row Q, row K, row V, mutable_row Out>
100 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
101 && std::same_as<row_value_t<Q>, row_value_t<Out>>
102constexpr row_value_t<Q> attend_selected(const Q& q, const K& keys, const V& values, std::span<const std::uint32_t> indices,
103 Out&& out, std::type_identity_t<row_value_t<Q>> scale,
104 std::type_identity_t<row_value_t<Q>> sink = -std::numeric_limits<row_value_t<Q>>::infinity())
105{
106 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values);
107 const auto y = as_mutable_row(out);
108 const std::size_t dk = qs.size(), dv = y.size(), n = dk == 0 ? 0 : ks.size() / dk;
109 assert(dk > 0 && ks.size() == n * dk && vs.size() == n * dv);
110 assert(std::all_of(indices.begin(), indices.end(), [n](std::uint32_t i) { return i < n; }));
111 (void)n;
112 return detail::attention_sink::forward(
113 indices.size(), [&](std::size_t j) { return scale * detail::attention::dot(qs, ks.subspan(indices[j] * dk, dk)); },
114 [&](std::size_t j) { return vs.subspan(indices[j] * dv, dv); }, y, sink);
115}

123template <row Q, row K, row V, row O, row DO, mutable_row DQ, mutable_row DK, mutable_row DV>
124 requires std::same_as<row_value_t<Q>, row_value_t<K>> && std::same_as<row_value_t<Q>, row_value_t<V>>
125 && std::same_as<row_value_t<Q>, row_value_t<O>> && std::same_as<row_value_t<Q>, row_value_t<DO>>
126 && std::same_as<row_value_t<Q>, row_value_t<DQ>> && std::same_as<row_value_t<Q>, row_value_t<DK>>
127 && std::same_as<row_value_t<Q>, row_value_t<DV>>
128constexpr row_value_t<Q> attend_selected_backward(const Q& q, const K& keys, const V& values, std::span<const std::uint32_t> indices,
129 const O& out, const DO& dout, std::type_identity_t<row_value_t<Q>> lse,
130 std::type_identity_t<row_value_t<Q>> scale,
131 std::type_identity_t<row_value_t<Q>> sink, DQ&& dq, DK&& dkeys, DV&& dvalues)
132{
133 using T = row_value_t<Q>;
134 const auto qs = as_row(q), ks = as_row(keys), vs = as_row(values), o = as_row(out), g = as_row(dout);
135 const auto gq = as_mutable_row(dq), gk = as_mutable_row(dkeys), gv = as_mutable_row(dvalues);
136 const std::size_t dk = qs.size(), dv = o.size(), n = dk == 0 ? 0 : ks.size() / dk;
137 assert(dk > 0 && ks.size() == n * dk && vs.size() == n * dv && g.size() == dv && gq.size() == dk && gk.size() == ks.size()
138 && gv.size() == vs.size());
139 (void)n;
140 std::fill(gq.begin(), gq.end(), T(0));
141 return detail::attention_sink::backward(
142 indices.size(), [&](std::size_t j) { return scale * detail::attention::dot(qs, ks.subspan(indices[j] * dk, dk)); },
143 [&](std::size_t j) { return vs.subspan(indices[j] * dv, dv); }, o, g, lse, sink,
144 [&](std::size_t j, T p, T ds) {
145 const std::size_t r = indices[j];
146 for (std::size_t i = 0; i < dv; ++i)
147 gv[r * dv + i] += p * g[i];
148 for (std::size_t i = 0; i < dk; ++i) {
149 gq[i] += scale * ds * ks[r * dk + i];
150 gk[r * dk + i] += scale * ds * qs[i];
151 }
152 });
153}

160template <row Q, row W, row K, mutable_row S>
161 requires std::same_as<row_value_t<Q>, row_value_t<W>> && std::same_as<row_value_t<Q>, row_value_t<K>>
162 && std::same_as<row_value_t<Q>, row_value_t<S>>
163constexpr void indexer_scores(const Q& q, const W& weights, const K& keys, S&& scores)
164{
165 using T = row_value_t<Q>;
166 const auto qs = as_row(q), w = as_row(weights), ks = as_row(keys);
167 const auto out = as_mutable_row(scores);
168 const std::size_t h = w.size(), d = h == 0 ? 0 : qs.size() / h, n = out.size();
169 assert(h > 0 && qs.size() == h * d && ks.size() == n * d);
170 for (std::size_t s = 0; s < n; ++s) {
171 T acc = T(0);
172 for (std::size_t j = 0; j < h; ++j) {
173 const T a = detail::attention::dot(qs.subspan(j * d, d), ks.subspan(s * d, d));
174 acc += w[j] * (a > T(0) ? a : T(0));
175 }
176 out[s] = acc;
177 }
178}

185template <row Q, row W, row K, row DS, mutable_row DQ, mutable_row DW, mutable_row DK>
186 requires std::same_as<row_value_t<Q>, row_value_t<W>> && std::same_as<row_value_t<Q>, row_value_t<K>>
187 && std::same_as<row_value_t<Q>, row_value_t<DS>> && std::same_as<row_value_t<Q>, row_value_t<DQ>>
188 && std::same_as<row_value_t<Q>, row_value_t<DW>> && std::same_as<row_value_t<Q>, row_value_t<DK>>
189constexpr void indexer_scores_backward(const Q& q, const W& weights, const K& keys, const DS& dscores, DQ&& dq, DW&& dweights,
190 DK&& dkeys)
191{
192 using T = row_value_t<Q>;
193 const auto qs = as_row(q), w = as_row(weights), ks = as_row(keys), g = as_row(dscores);
194 const auto gq = as_mutable_row(dq), gw = as_mutable_row(dweights), gk = as_mutable_row(dkeys);
195 const std::size_t h = w.size(), d = h == 0 ? 0 : qs.size() / h, n = g.size();
196 assert(h > 0 && qs.size() == h * d && ks.size() == n * d && gq.size() == qs.size() && gw.size() == h && gk.size() == ks.size());
197 for (std::size_t s = 0; s < n; ++s)
198 for (std::size_t j = 0; j < h; ++j) {
199 const T a = detail::attention::dot(qs.subspan(j * d, d), ks.subspan(s * d, d));
200 if (!(a > T(0)))
201 continue;
202 gw[j] += g[s] * a;
203 const T c = g[s] * w[j];
204 for (std::size_t i = 0; i < d; ++i) {
205 gq[j * d + i] += c * ks[s * d + i];
206 gk[s * d + i] += c * qs[j * d + i];
207 }
208 }
209}

215template <row S>
216constexpr std::size_t select_top_k(const S& scores, std::span<std::uint32_t> indices)
217{
218 const auto s = as_row(scores);
219 const std::size_t n = s.size(), k = std::min(indices.size(), n);
220 assert(n <= std::numeric_limits<std::uint32_t>::max());
221 // A min-heap of the best k under "a beats b": a higher score, or the same score at an earlier position
222 const auto beats = [&](std::uint32_t a, std::uint32_t b) { return s[a] > s[b] || (s[a] == s[b] && a < b); };
223 const auto heap = indices.first(k);
224 for (std::size_t i = 0; i < n; ++i) {
225 const auto pos = static_cast<std::uint32_t>(i);
226 if (i < k) {
227 heap[i] = pos;
228 std::push_heap(heap.begin(), heap.begin() + std::ptrdiff_t(i + 1), beats);
229 } else if (beats(pos, heap.front())) {
230 std::pop_heap(heap.begin(), heap.end(), beats);
231 heap.back() = pos;
232 std::push_heap(heap.begin(), heap.end(), beats);
233 }
234 }
235 std::sort(heap.begin(), heap.end());
236 return k;
237}

245template <row P, row S, mutable_row DS>
246 requires std::same_as<row_value_t<P>, row_value_t<S>> && std::same_as<row_value_t<P>, row_value_t<DS>>
247constexpr row_value_t<P> indexer_kl_loss(const P& target, const S& scores, DS&& dscores)
248{
249 using T = row_value_t<P>;
250 const auto p = as_row(target), z = as_row(scores);
251 const auto g = as_mutable_row(dscores);
252 const std::size_t n = z.size();
253 assert(p.size() == n && g.size() == n);
254 if (n == 0)
255 return T(0);
256 T max = z[0];
257 for (const T v : z)
258 max = v > max ? v : max;
259 T sum = T(0);
260 for (const T v : z)
261 sum += math::exp(v - max);
262 const T lse = max + math::log(sum);
263 T loss = T(0), mass = T(0);
264 for (std::size_t s = 0; s < n; ++s) {
265 mass += p[s];
266 if (p[s] > T(0))
267 loss += p[s] * (math::log(p[s]) - (z[s] - lse));
268 }
269 for (std::size_t s = 0; s < n; ++s)
270 g[s] = mass * math::exp(z[s] - lse) - p[s];
271 return loss;
272}
