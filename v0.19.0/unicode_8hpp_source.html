131constexpr utf8_decoded decode_utf8(std::string_view text, std::size_t at) noexcept
132{
133 if (at >= text.size())
134 return {0, 0, false};
135 const auto byte = [&](std::size_t i) -> std::uint32_t { return static_cast<unsigned char>(text[at + i]); };
136 const std::size_t left = text.size() - at;
137 const std::uint32_t b0 = byte(0);
138 constexpr utf8_decoded bad{replacement_character, 1, false};
139 if (b0 < 0x80)
140 return {b0, 1, true};
141 const auto cont = [&](std::size_t i, std::uint32_t lo = 0x80, std::uint32_t hi = 0xBF) {
142 return i < left && byte(i) >= lo && byte(i) <= hi;
143 };
144 if (b0 >= 0xC2 && b0 <= 0xDF)
145 return cont(1) ? utf8_decoded{((b0 & 0x1F) << 6) | (byte(1) & 0x3F), 2, true} : bad;
146 if (b0 >= 0xE0 && b0 <= 0xEF) {
147 const std::uint32_t lo = b0 == 0xE0 ? 0xA0 : 0x80, hi = b0 == 0xED ? 0x9F : 0xBF;
148 if (!cont(1, lo, hi) || !cont(2))
149 return bad;
150 return {((b0 & 0x0F) << 12) | ((byte(1) & 0x3F) << 6) | (byte(2) & 0x3F), 3, true};
151 }
152 if (b0 >= 0xF0 && b0 <= 0xF4) {
153 const std::uint32_t lo = b0 == 0xF0 ? 0x90 : 0x80, hi = b0 == 0xF4 ? 0x8F : 0xBF;
154 if (!cont(1, lo, hi) || !cont(2) || !cont(3))
155 return bad;
156 return {((b0 & 0x07) << 18) | ((byte(1) & 0x3F) << 12) | ((byte(2) & 0x3F) << 6) | (byte(3) & 0x3F), 4, true};
157 }
158 return bad;
159}

179constexpr std::size_t encode_utf8(char32_t cp, char* out) noexcept
180{
181 if ((cp >= 0xD800 && cp <= 0xDFFF) || cp > 0x10FFFF)
182 cp = replacement_character;
183 if (cp < 0x80) {
184 out[0] = static_cast<char>(cp);
185 return 1;
186 }
187 if (cp < 0x800) {
188 out[0] = static_cast<char>(0xC0 | (cp >> 6));
189 out[1] = static_cast<char>(0x80 | (cp & 0x3F));
190 return 2;
191 }
192 if (cp < 0x10000) {
193 out[0] = static_cast<char>(0xE0 | (cp >> 12));
194 out[1] = static_cast<char>(0x80 | ((cp >> 6) & 0x3F));
195 out[2] = static_cast<char>(0x80 | (cp & 0x3F));
196 return 3;
197 }
198 out[0] = static_cast<char>(0xF0 | (cp >> 18));
199 out[1] = static_cast<char>(0x80 | ((cp >> 12) & 0x3F));
200 out[2] = static_cast<char>(0x80 | ((cp >> 6) & 0x3F));
201 out[3] = static_cast<char>(0x80 | (cp & 0x3F));
202 return 4;
203}

208constexpr bool is_valid_utf8(std::string_view text) noexcept
209{
210 for (std::size_t i = 0; i < text.size();) {
211 const utf8_decoded d = decode_utf8(text, i);
212 if (!d.valid)
213 return false;
214 i += d.length;
215 }
216 return true;
217}

224constexpr std::size_t replace_invalid_utf8(std::string_view text, std::span<char> out) noexcept
225{
226 std::size_t at = 0;
227 for (std::size_t i = 0; i < text.size();) {
228 const utf8_decoded d = decode_utf8(text, i);
229 if (d.valid)
230 for (std::size_t k = 0; k < d.length; ++k)
231 out[at++] = text[i + k];
232 else
233 at += encode_utf8(replacement_character, out.data() + at);
234 i += d.length;
235 }
236 return at;
237}

241 // The run of the generated category table that holds cp: the last run starting at or before it
242 constexpr unicode_category category_search(char32_t cp) noexcept
243 {
244 const auto& t = snicholls::detail::unicode_tables::categories;
245 std::size_t lo = 0, hi = std::size(t); // the answer is in [lo, hi)
246 while (hi - lo > 1) {
247 const std::size_t mid = lo + (hi - lo) / 2;
248 if ((t[mid] >> 5) <= cp)
249 lo = mid;
250 else
251 hi = mid;
252 }
253 return static_cast<unicode_category>(t[lo] & 31u);
254 }

355constexpr char32_t to_lower(char32_t cp) noexcept
356{
357 if (cp < 128)
358 return cp >= 'A' && cp <= 'Z' ? cp + 32 : cp;
359 const auto& r = snicholls::detail::unicode_tables::lowercase_runs;
360 std::size_t lo = 0, hi = std::size(r) / 4; // the last run starting at or before cp is in [lo, hi)
361 if (cp < static_cast<char32_t>(r[0]))
362 return cp;
363 while (hi - lo > 1) {
364 const std::size_t mid = lo + (hi - lo) / 2;
365 if (static_cast<char32_t>(r[4 * mid]) <= cp)
366 lo = mid;
367 else
368 hi = mid;
369 }
370 const auto first = static_cast<char32_t>(r[4 * lo]);
371 const auto count = static_cast<char32_t>(r[4 * lo + 1]), stride = static_cast<char32_t>(r[4 * lo + 2]);
372 const char32_t offset = cp - first;
373 if (offset % stride != 0 || offset / stride >= count)
374 return cp;
375 return static_cast<char32_t>(static_cast<std::int32_t>(cp) + r[4 * lo + 3]);
376}

381constexpr std::uint8_t combining_class(char32_t cp) noexcept
382{
383 const auto& t = snicholls::detail::unicode_tables::combining_classes;
384 std::size_t lo = 0, hi = std::size(t);
385 while (lo < hi) { // the first range whose last code point is at or after cp
386 const std::size_t mid = lo + (hi - lo) / 2;
387 if (((t[mid] >> 8) & 0x1FFFFF) < cp)
388 lo = mid + 1;
389 else
390 hi = mid;
391 }
392 if (lo < std::size(t) && (t[lo] >> 29) <= cp)
393 return static_cast<std::uint8_t>(t[lo] & 0xFF);
394 return 0;
395}

399 // The direct canonical decomposition of cp: 0, 1 or 2 code points (Hangul excluded)
400 constexpr std::size_t decompose_once(char32_t cp, char32_t& a, char32_t& b) noexcept
401 {
402 const auto& t = snicholls::detail::unicode_tables::decompositions;
403 std::size_t lo = 0, hi = std::size(t);
404 while (lo < hi) {
405 const std::size_t mid = lo + (hi - lo) / 2;
406 if ((t[mid] >> 42) < cp)
407 lo = mid + 1;
408 else
409 hi = mid;
410 }
411 if (lo == std::size(t) || (t[lo] >> 42) != cp)
412 return 0;
413 a = static_cast<char32_t>((t[lo] >> 21) & 0x1FFFFF);
414 b = static_cast<char32_t>(t[lo] & 0x1FFFFF);
415 return b == 0 ? 1 : 2;
416 }
417} // namespace detail::unicode

426constexpr std::size_t decompose(char32_t cp, std::span<char32_t, max_decomposition> out) noexcept
427{
428 if (cp >= 0xAC00 && cp < 0xAC00 + 11172) { // Hangul, UAX #15
429 const char32_t s = cp - 0xAC00;
430 out[0] = 0x1100 + s / 588;
431 out[1] = 0x1161 + (s % 588) / 28;
432 if (s % 28 == 0)
433 return 2;
434 out[2] = 0x11A7 + s % 28;
435 return 3;
436 }
437 // Expand the first code point until it has no mapping; every second code point of a canonical mapping is
438 // itself fully decomposed, so only the first needs expanding again.
439 std::size_t n = 1;
440 out[0] = cp;
441 for (;;) {
442 char32_t a = 0, b = 0;
443 const std::size_t k = detail::unicode::decompose_once(out[0], a, b);
444 if (k == 0 || (k == 2 && n == max_decomposition))
445 return n;
446 if (k == 2) {
447 for (std::size_t i = n; i > 1; --i)
448 out[i] = out[i - 1];
449 out[1] = b;
450 ++n;
451 }
452 out[0] = a;
453 }
454}
