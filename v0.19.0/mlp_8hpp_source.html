217 // The forward pass, saving the pre-activations into non-empty spans
218 template <transformer_config C, typename W, typename T, typename X, typename Y>
219 constexpr void forward(const mlp_weights<W, T>& w, const X& x, Y&& y, const mlp_activations<T>& save)
220 {
221 static_assert(valid(C), "snicholls::ml::mlp_forward: the transformer_config is not valid");
222 assert(w.up.rows == C.d_ff && w.up.cols == C.d_model && w.down.rows == C.d_model && w.down.cols == C.d_ff);
223 assert((save.up.empty() || save.up.size() == C.d_ff) && (save.gate.empty() || save.gate.size() == C.d_ff));
224 std::array<T, C.d_ff> h{};
225 const auto in = prepare_input<C.d_model>(w.up, x);
226 linear(w.up, in, h, w.up_bias);
227 if (!save.up.empty())
228 std::copy(h.begin(), h.end(), save.up.begin());
229 if constexpr (gated(C.mlp)) {
230 assert(w.gate.rows == C.d_ff && w.gate.cols == C.d_model);
231 std::array<T, C.d_ff> g{};
232 linear(w.gate, in, g, w.gate_bias);
233 if (!save.gate.empty())
234 std::copy(g.begin(), g.end(), save.gate.begin());
235 for (std::size_t i = 0; i < C.d_ff; ++i)
236 h[i] = mlp_glu<C>(g[i], h[i]);
237 } else {
238 for (T& v : h)
239 v = mlp_activation<C.mlp>(v);
240 }
241 linear(w.down, prepare_input<C.d_ff>(w.down, h), y, w.down_bias);
242 }
243} // namespace detail::mlp

283template <transformer_config C, typename S, math::math_number T, row X, row DY, mutable_row DX>
284 requires std::same_as<row_value_t<X>, T> && std::same_as<row_value_t<DY>, T> && std::same_as<row_value_t<DX>, T>
285constexpr void mlp_backward(const mlp_weights<dense_weights<S>, T>& w, const X& x, const mlp_activations<T>& saved,
286 const DY& dy, DX&& dx, const mlp_gradients<T>& grads)
287{
288 static_assert(valid(C), "snicholls::ml::mlp_backward: the transformer_config is not valid");
289 constexpr bool swiglu = gated(C.mlp);
290 assert(saved.up.size() == C.d_ff && (!swiglu || saved.gate.size() == C.d_ff));
291 std::array<T, C.d_ff> h{}, dh{};
292 for (std::size_t i = 0; i < C.d_ff; ++i)
293 if constexpr (swiglu)
294 h[i] = mlp_glu<C>(saved.gate[i], saved.up[i]);
295 else
296 h[i] = mlp_activation<C.mlp>(saved.up[i]);
297 linear_backward(w.down, h, dy, dh, grads.down, grads.down_bias);
298 if constexpr (swiglu) {
299 std::array<T, C.d_ff> dgate{};
300 for (std::size_t i = 0; i < C.d_ff; ++i) {
301 const auto d = mlp_glu_backward<C>(saved.gate[i], saved.up[i], dh[i]);
302 dgate[i] = d.gate;
303 dh[i] = d.up;
304 }
305 linear_backward(w.gate, x, dgate, dx, grads.gate, grads.gate_bias);
306 } else {
307 for (std::size_t i = 0; i < C.d_ff; ++i)
308 dh[i] = dh[i] * mlp_activation_derivative<C.mlp>(saved.up[i]);
309 }
310 linear_backward(w.up, x, dh, dx, grads.up, grads.up_bias);
311}

317template <transformer_config C, typename B, math::math_number T>
318constexpr void mlp_activate_rows(const B&, const mlp_activations<T>& saved)
319{
320 const std::size_t m = saved.up.size();
321 if constexpr (gated(C.mlp)) {
322 for (std::size_t i = 0; i < m; ++i)
323 saved.hidden[i] = mlp_glu<C>(saved.gate[i], saved.up[i]);
324 } else {
325 for (std::size_t i = 0; i < m; ++i)
326 saved.hidden[i] = mlp_activation<C.mlp>(saved.up[i]);
327 }

336template <transformer_config C, typename B, math::math_number T>
337constexpr void mlp_activate_rows_backward(const B&, const mlp_activations<T>& saved, std::span<T> dh, std::span<T> dgate)
338{
339 const std::size_t m = dh.size();
340 if constexpr (gated(C.mlp)) {
341 for (std::size_t i = 0; i < m; ++i) {
342 const auto d = mlp_glu_backward<C>(saved.gate[i], saved.up[i], dh[i]);
343 dgate[i] = d.gate;
344 dh[i] = d.up;
345 }
346 } else {
347 for (std::size_t i = 0; i < m; ++i)
348 dh[i] = dh[i] * mlp_activation_derivative<C.mlp>(saved.up[i]);
349 (void)dgate;
350 }
351}

365template <transformer_config C, typename B, math::math_number T>
366 requires linalg::blas_backend<B, T>
367constexpr void mlp_forward_sequence(const B& backend, const mlp_weights<dense_weights<T>, T>& w, std::type_identity_t<std::span<const T>> xs,
368 std::type_identity_t<std::span<T>> ys, const mlp_activations<T>& save)
369{
370 static_assert(valid(C), "snicholls::ml::mlp_forward_sequence: the transformer_config is not valid");
371 constexpr bool swiglu = gated(C.mlp);
372 const std::size_t n = xs.size() / C.d_model;
373 [[maybe_unused]] const std::size_t m = n * C.d_ff;
374 assert(xs.size() == n * C.d_model && ys.size() == xs.size() && save.up.size() == m && save.hidden.size() == m
375 && (!swiglu || save.gate.size() == m));
376 linear_sequence(backend, w.up, xs, save.up, w.up_bias);
377 if constexpr (swiglu)
378 linear_sequence(backend, w.gate, xs, save.gate, w.gate_bias);
379 mlp_activate_rows<C>(backend, save);
380 linear_sequence(backend, w.down, std::span<const T>(save.hidden), ys, w.down_bias);
381}

392template <transformer_config C, typename B, math::math_number T>
393 requires linalg::blas_backend<B, T>
394constexpr void mlp_backward_sequence(const B& backend, const mlp_weights<dense_weights<T>, T>& w,
395 std::type_identity_t<std::span<const T>> xs, const mlp_activations<T>& saved,
396 std::type_identity_t<std::span<const T>> dys, std::type_identity_t<std::span<T>> dxs,
397 const mlp_gradients<T>& grads)
398{
399 static_assert(valid(C), "snicholls::ml::mlp_backward_sequence: the transformer_config is not valid");
400 constexpr bool swiglu = gated(C.mlp);
401 const std::size_t n = xs.size() / C.d_model, m = n * C.d_ff;
402 assert(dys.size() == xs.size() && saved.up.size() == m && saved.hidden.size() == m && (!swiglu || saved.gate.size() == m)
403 && grads.scratch.size() >= (swiglu ? 2 : 1) * m);
404 const auto dh = grads.scratch.first(m);
405 const auto dgate = swiglu ? grads.scratch.subspan(m, m) : std::span<T>{};
406 fill_rows(backend, dh, T(0));
407 linear_sequence_backward(backend, w.down, std::span<const T>(saved.hidden), dys, dh, grads.down, grads.down_bias);
408 mlp_activate_rows_backward<C>(backend, saved, dh, dgate);
409 if constexpr (swiglu)
410 linear_sequence_backward(backend, w.gate, xs, std::span<const T>(dgate), dxs, grads.gate, grads.gate_bias);
411 linear_sequence_backward(backend, w.up, xs, std::span<const T>(dh), dxs, grads.up, grads.up_bias);
412}

418template <math::math_number T, std::size_t K>
419struct moe_route
420{
421 std::array<std::size_t, K> expert{};
422 std::array<T, K> weight{};
423 std::size_t count = 0;
424};
425
430template <std::size_t K, row R>
431 requires(K > 0)
432constexpr moe_route<row_value_t<R>, K> moe_gate(const R& router_logits)
433{
434 using T = row_value_t<R>;
435 const auto best = top_k<K>(router_logits);
436 assert(best.count > 0);
437 moe_route<T, K> out{};
438 out.count = best.count;
439 T sum = T(0);
440 for (std::size_t i = 0; i < best.count; ++i) {
441 out.expert[i] = best.index[i];
442 out.weight[i] = math::exp(best.value[i] - best.value[0]);
443 sum += out.weight[i];
444 }
445 for (std::size_t i = 0; i < best.count; ++i)
446 out.weight[i] /= sum;
447 return out;
448}

457template <math::math_number T, std::size_t K, row DW, mutable_row Out>
458 requires std::same_as<row_value_t<DW>, T> && std::same_as<row_value_t<Out>, T>
459constexpr void moe_gate_backward(const moe_route<T, K>& route, const DW& dweight, Out&& dlogits)
460{
461 const auto dw = as_row(dweight);
462 const auto g = as_mutable_row(dlogits);
463 assert(dw.size() >= route.count && route.count <= g.size());
464 T dot = T(0);
465 for (std::size_t i = 0; i < route.count; ++i)
466 dot += route.weight[i] * dw[i];
467 std::fill(g.begin(), g.end(), T(0));
468 for (std::size_t i = 0; i < route.count; ++i)
469 g[route.expert[i]] = route.weight[i] * (dw[i] - dot);
470}
