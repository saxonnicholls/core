154 template <linalg::scalar T, std::size_t N>
155 [[nodiscard]] constexpr result<T, linalg::error> squared_hellinger(const gaussian<T, N>& a, const gaussian<T, N>& b) noexcept(linalg::detail::nothrow_scalar<T>)
156 {
157 const auto la = linalg::cholesky(a.covariance);
158 const auto lb = linalg::cholesky(b.covariance);
159 const auto lm = linalg::cholesky((a.covariance + b.covariance) / T(2));
160 if (!la || !lb || !lm)
161 return err(linalg::error::not_positive_definite);
162 const T log_coeff = (linalg::cholesky_log_det(*la) + linalg::cholesky_log_det(*lb)) / T(4) - linalg::cholesky_log_det(*lm) / T(2);
163 const T e = whitened_squared_norm(a.mean - b.mean, *lm) / T(8);
164 const T h2 = T(1) - static_cast<T>(math::exp(log_coeff - e));
165 return ok(h2 < T(0) ? T(0) : h2);
166 }

185 template <linalg::scalar T, std::size_t N>
186 [[nodiscard]] constexpr result<T, linalg::error> kl_divergence(const gaussian<T, N>& p, const gaussian<T, N>& q) noexcept(linalg::detail::nothrow_scalar<T>)
187 {
188 const auto lp = linalg::cholesky(p.covariance);
189 const auto lq = linalg::cholesky(q.covariance);
190 if (!lp || !lq)
191 return err(linalg::error::not_positive_definite);
192 const T tr = linalg::trace(linalg::cholesky_solve(*lq, p.covariance));
193 const T md = whitened_squared_norm(q.mean - p.mean, *lq);
194 return ok((tr + md - static_cast<T>(N) + linalg::cholesky_log_det(*lq) - linalg::cholesky_log_det(*lp)) / T(2));
195 }

232 template <linalg::scalar T>
233 [[nodiscard]] constexpr ellipse<T> confidence_ellipse(const linalg::fixed_matrix<T, 2, 2>& p, T k = T(1)) noexcept(linalg::detail::nothrow_scalar<T>)
234 {
235 const T a = p(0, 0), b = p(0, 1), d = p(1, 1);
236 const T mean = (a + d) / T(2);
237 const T h = static_cast<T>(math::sqrt((a - d) * (a - d) / T(4) + b * b));
238 const T l1 = mean + h, l2 = mean - h;
239 // The major axis is at θ = atan2(2b, a - d) / 2, in (-π/2, π/2]
240 const T ang = (b == T(0) && a == d) ? T(0) : static_cast<T>(math::atan2(T(2) * b, a - d)) / T(2);
241 return {k * static_cast<T>(math::sqrt(l1 > T(0) ? l1 : T(0))), k * static_cast<T>(math::sqrt(l2 > T(0) ? l2 : T(0))), ang};
242 }
